def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_cross_matrix(a):
    """Build T such that (T @ b)_k = (conv(a,b))_k."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[np.arange(end - start)]
    return T


def lp_min_peak(a, S, solver):
    """min t s.t. (a*b)_k <= t for all k, b>=0, sum(b)=S. Returns (b, t)."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T = build_cross_matrix(a)
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == S]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    t_val = float(t_var.value)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val * S / s
    return b_val, t_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Accelerated LP best-response for min-peak cross-correlation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    last_best_time = time.monotonic()
    momentum = 0.0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b, t_val = lp_min_peak(y, norm, solver)
        if b is None:
            continue
        n_iters += 1

        # Nesterov step with line search
        base_alpha = 2.0 / (t + 2.0)
        best_alpha = base_alpha
        best_obj = fast_autocorr(np.maximum((1 - base_alpha) * y + base_alpha * b, 0.0)).max()

        for trial in [base_alpha * 0.5, base_alpha * 1.5, base_alpha * 2.0, 1.0]:
            trial = min(max(trial, 0.0), 1.0)
            cand = np.maximum((1 - trial) * y + trial * b, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand * norm / cs
            sc = fast_autocorr(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial

        alpha = best_alpha
        y_new = np.maximum((1.0 - alpha) * y + alpha * b, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = np.maximum(y_new + momentum * (y_new - y), 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        for cand in [y, a]:
            sc = fast_autocorr(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                last_best_time = time.monotonic()
                stall = 0

        n_avg_add = 1
        avg_a += a
        n_avg += n_avg_add
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_best_time = time.monotonic()

        if t % 200 == 0:
            if time.monotonic() - last_best_time > 120:
                stall += 1
                if stall > 10:
                    if verbose:
                        print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                    break
                elif stall > 3:
                    y = best_a.copy()
                    avg_a = np.zeros(n)
                    n_avg = 0
                    stall = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_objective(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    lse = max_b + np.log(np.exp(shifted).sum()) / T_temp
    obj = 2.0 * n * lse / (s * s)
    return obj


def lse_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp

    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 120.0)

        x0 = a.copy()

        def obj_func(x):
            return lse_objective(x, n, T)

        def grad_func(x):
            return lse_grad(x, n, T)

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """Aggressive coordinate descent targeting peak lags."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-15:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    iter_count = 0
    while time.monotonic() - t0 < time_limit - 2:
        iter_count += 1
        b = fast_autocorr(a)
        max_b = b.max()
        # Find lags close to the peak
        threshold = max_b * 0.98
        peak_lags = np.where(b >= threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.argmax(b)
            peak_lags = np.array([peak_lags])
        peak_lags = peak_lags[:20]

        improved = False
        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 2:
                break
            # Find contributing pairs
            # (a*a)_k = sum_i a[i]*a[k-i] where 0<=i<n and 0<=k-i<n
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            # Top contributors
            top_k = min(8, len(idx))
            top_local = np.argsort(contrib)[::-1][:top_k]
            top_idx = idx[top_local]

            # Build peak mask: positions that contribute to any peak lag
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in peak_lags[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 <= je2:
                    peak_mask[js2:je2 + 1] = True

            for src in top_idx:
                if a[src] < 1e-10:
                    continue
                # Try moving mass from src to non-peak positions
                candidates = np.where(~peak_mask)[0]
                if len(candidates) < 5:
                    candidates = np.where(np.arange(n) != src)[0]
                # Sort by current value (prefer moving to low-value positions)
                candidates_sorted = candidates[np.argsort(a[candidates])]
                n_try = min(8, len(candidates_sorted))

                for dst in candidates_sorted[:n_try]:
                    for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random-ish moves to escape
            if stall % 3 == 0:
                for _ in range(50):
                    if time.monotonic() - t0 > time_limit - 2:
                        break
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.005, 0.2)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break

            # Perturbation
            if stall % 20 == 0 and stall > 0:
                n_pert = max(5, n // 50)
                pidx = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for p in pidx:
                    a_trial[p] *= rng.uniform(0.3, 2.0)
                try_update(a_trial)

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng, variant=None):
    if variant is None:
        variant = rng.integers(0, 12)
    if variant == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif variant == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif variant == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.5) ** 2)
    elif variant == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif variant == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif variant == 7:
        # Low-ripple pattern
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.2 * np.cos(6.0 * np.pi * x) + 0.1 * np.cos(12.0 * np.pi * x)
    elif variant == 8:
        # Slightly asymmetric
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 + x) * (1.0 - x) * (1.0 + 0.2 * np.cos(4.0 * np.pi * x))
    elif variant == 9:
        # Periodic with rational frequency
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 3.0 * np.arange(n) / n)
    elif variant == 10:
        # Bump + tail
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.4) ** 2) + 0.3 * np.exp(-3.0 * (x - 0.8) ** 2)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Determine best length via quick scan
    print("\n=== Phase 0: Length scan ===")
    n_parent = len(best_seq) if best_seq is not None else 1600
    scan_lengths = sorted(set([
        n_parent,
        max(800, n_parent - 200),
        n_parent + 200,
        n_parent + 400,
    ]))
    scan_lengths = [n for n in scan_lengths if 400 <= n <= 4000]
    scan_results = {}

    for i_scan, n_scan in enumerate(scan_lengths):
        if time.monotonic() - t_global > time_budget * 0.10:
            break
        remaining_scan = time_budget * 0.10 - (time.monotonic() - t_global)
        if remaining_scan < 10:
            break
        scan_time = min(remaining_scan / max(1, len(scan_lengths) - i_scan), 60.0)
        if scan_time < 8:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        do_try_update(scan_results[best_scan_n][0], f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LP-Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Deep LP n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_deep_n{n_cur}")

    # Phase 2: LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=50.0, T_end=1.0, n_stages=8)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 3: LP at extended length
    n_long = n_base + 300
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP n={n_long} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # Phase 4: Noisy restart LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP noisy restart n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        noise_scale = 0.03 * best_seq.mean()
        a_noisy = best_seq + rng.normal(0, noise_scale, n_cur)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_noisy = a_noisy * np.sqrt(2.0 * n_cur) / a_noisy.sum()
        a_out, _ = nesterov_lp_optimize(a_noisy, t4, solver, verbose=False)
        do_try_update(a_out, f"lp_noisy_n{n_cur}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t5, seed=12345)
        do_try_update(a_out, "coord_1")

    # Phase 6: LSE with lower temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LSE low-T n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t6, T_start=15.0, T_end=0.5, n_stages=5)
        do_try_update(a_out, "lse_low_T")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = max(remaining * 0.5, 10.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=99999)
        do_try_update(a_out, "coord_final")

    # Phase 8: One more short LP round on the best
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
        t8 = min(remaining * 0.4, 120.0)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=False)
        do_try_update(a_out, "lp_final")

        # Quick coord after
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = coordinate_descent_polish(a_norm, min(remaining2 * 0.5, 60.0), seed=31337)
            do_try_update(a_out, "coord_very_final")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
