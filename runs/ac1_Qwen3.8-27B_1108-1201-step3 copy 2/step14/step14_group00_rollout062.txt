```python
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def smoothed_objective(a, beta):
    conv = fast_autocorr(a)
    # Shift to avoid overflow
    max_conv = conv.max()
    if max_conv < 1e-30:
        return 0.0, np.zeros_like(a)
    shifted = beta * (conv - max_conv)
    log_sum_exp = max_conv * beta + np.log(np.exp(shifted).sum()) / beta
    return log_sum_exp, None

    n = len(a)
    # Compute softmax of conv * beta
    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Gradient of (1/beta) * log(sum(exp(beta*conv_k))) wrt a_j

    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated L-BFGS-B on log-sum-exp smoothed surrogate with annealing
2. Exact pair coordinate descent for local refinement
3. Multi-length exploration via interpolation
4. Perturbation-based restarts to escape local optima
Key difference from LP approach: O(n log n) per iteration enables thousands of
optimization steps, allowing thorough exploration of the landscape.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_smoothed_obj_grad(a, beta):
    """
    Compute smoothed objective f_beta(a) = 2n * LSE_beta(conv(a)) / sum(a)^2
    and its gradient. Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)

    conv = fast_autocorr(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    # Numerical safety
    max_shifted = shifted.max()
    if max_shifted > 500:
        shifted = shifted - max_shifted
        log_sum_exp = (max_c + (max_shifted + np.log(np.exp(shifted).sum())) / beta)
    else:
        log_sum_exp = max_c + np.log(np.exp(shifted).sum()) / beta

    val = 2.0 * n * log_sum_exp / (s * s)

    # Softmax weights
    if max_shifted > 500:
        weights = np.exp(shifted)
    else:
        weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum

    # Gradient of LSE w.r.t. a: 2 * cross_correlation(weights, a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(weights, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    grad_lse = 2.0 * cross

    # Full gradient of f = 2n*LSE/s^2
    grad = (2.0 * n / (s * s)) * grad_lse - 2.0 * val * a / s
    return val, grad


def optimize_lbfgs(a0, beta_schedule, time_limit, verbose=False, maxiter_per_beta=300):
    """
    Run L-BFGS-B with annealed beta schedule.
    Returns best sequence found.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    t0 = time.monotonic()

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() - t0 > time_limit:
            break

        x0 = np.maximum(a, 1e-10)
        x0 = x0 / x0.sum()

        def fun(x):
            v, _ = compute_smoothed_obj_grad(x, beta)
            return v

        def gfun(x):
            _, g = compute_smoothed_obj_grad(x, beta)
            return g

        try:
            res = minimize(fun, x0, jac=gfun, method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': maxiter_per_beta,
                                    'ftol': 1e-15, 'gtol': 1e-10,
                                    'maxls': 50})
        except Exception:
            continue

        a_new = np.maximum(res.x, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        conv_new = fast_autocorr(a_new)
        score_new = 2.0 * n * conv_new.max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            if verbose:
                print(f"      beta={beta:.0f}: improved to {best_score:.10f}")

        a = a_new

        # Also try raw res.x normalized
        a_raw = np.maximum(res.x, 0.0)
        sr = a_raw.sum()
        if sr > 1e-15:
            a_raw = a_raw / sr
            sc_raw = 2.0 * n * fast_autocorr(a_raw).max()
            if sc_raw < best_score:
                best_score = sc_raw
                best_a = a_raw.copy()
                a = a_raw.copy()

    return best_a, best_score


def pair_coordinate_descent(a0, time_limit, rng, verbose=False):
    """
    Pair coordinate descent: move mass between two positions to reduce max autocorrelation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    no_improve = 0
    pass_num = 0

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        if no_improve >= 6:
            break
        no_improve = 0
        pass_improved = False

        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue

            # Try random partners
            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Compute gain vector for moving delta from i to j
                # delta_conv[k] = contribution change at lag k
                # Removing delta from i: -delta * (a shifted by i) - delta * (a shifted by n-1-i)
                # Adding delta to j: +delta * (a shifted by j) + delta * (a shifted by n-1-j)
                g = np.zeros(size)

                # Effect of reducing a[i] by delta (first order in delta):
                # conv changes by -delta * (a shifted right by i) - delta * (a shifted right by n-1-i)
                lo1 = i
                hi1 = i + n
                if hi1 > size:
                    hi1 = size
                if lo1 < size:
                    g[lo1:hi1] -= a[0:hi1 - lo1]

                lo2 = n - 1 - i
                hi2 = n - 1 - i + n
                if hi2 > size:
                    hi2 = size
                if lo2 < size:
                    g[lo2:hi2] -= a[0:hi2 - lo2]

                # Effect of increasing a[j] by delta:
                lo3 = j
                hi3 = j + n
                if hi3 > size:
                    hi3 = size
                if lo3 < size:
                    g[lo3:hi3] += a[0:hi3 - lo3]

                lo4 = n - 1 - j
                hi4 = n - 1 - j + n
                if hi4 > size:
                    hi4 = size
                if lo4 < size:
                    g[lo4:hi4] += a[0:hi4 - lo4]

                # Find best delta in [0, ai]
                # We want to minimize max(conv + delta * g + second_order)
                # Second order at diagonal: conv[2i] gets -delta^2 (from a[i]^2 -> (a[i]-d)^2)
                #                        conv[2j] gets +2*a[j]*delta + delta^2

                candidates = [ai * f for f in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]]
                best_d = 0.0
                best_val = conv.max()

                for d in candidates:
                    if d <= 0 or d > ai:
                        continue
                    new_max = (conv + d * g).max()
                    # Second order corrections
                    di = 2 * i
                    if di < size:
                        second_i = conv[di] + d * g[di] - 2 * a[i] * d + d * d
                        # Actually the first-order already includes the change from a[i] change
                        # Let's just use the exact: new_conv[2i] = (a[i]-d)^2 + sum of cross terms
                        # The gain g already accounts for first order. Second order is -d*d at position 2i
                        # (since a[i]^2 becomes (a[i]-d)^2 = a[i]^2 - 2*a[i]*d + d^2,
                        #  and 2*a[i]*d is already in g)
                        new_max = max(new_max, conv[di] + d * g[di] - d * d)
                    dj = 2 * j
                    if dj < size:
                        new_max = max(new_max, conv[dj] + d * g[dj] + d * d)

                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    # Renormalize (shouldn't change sum since we moved mass)
                    # Actually sum stays the same, so no renormalization needed
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1

        if verbose and pass_num % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD pass={pass_num} ({elapsed:.1f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_seq(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_seq(a, rng, scale=0.05):
    """Add multiplicative noise."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

def subgradient_descent(a0, time_limit, rng, verbose=False):
    """
    Subgradient/proximal method: at each step, identify the peak lag,
    compute subgradient, take a step, project onto simplex.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    beta = 50.0
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        k_star = int(np.argmax(conv))

        # Subgradient of max at peak: grad_j = 2 * a[k_star - j] (if valid)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        # Step size (Polyak-type or fixed decreasing)
        gamma = 0.1 / (1.0 + 0.01 * t)

        # Gradient step
        a_new = a - gamma * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Occasionally add noise
        if t % 500 == 0:
            noise = rng.standard_normal(n) * 0.01
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.1f}s): best={best_score:.10f}")

    return best_a, best_score

            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Generated init: n={n_init}, score={best_score:.8f}")

    # === Phase 1: L-BFGS annealing at current length ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: L-BFGS annealing n={n_cur} ===")
    t1 = min(remaining() * 0.20, 300.0)
    if t1 > 30:
        a_start = best_seq.copy()
        # Beta schedule: start smooth, get sharper
        beta_sch = [50.0, 100.0, 200.0, 400.0, 800.0, 1500.0, 3000.0, 5000.0]
        # Scale beta by n to make it adaptive
        beta_sch = [b * n_cur / 1600.0 for b in beta_sch]
        a_out, _ = optimize_lbfgs(a_start, beta_sch, t1, verbose=True, maxiter_per_beta=500)
        try_update(a_out, f"lbfgs1_n{n_cur}")

    # === Phase 2: CD polish ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: CD polish n={n_cur} ===")
    t2 = min(remaining() * 0.12, 150.0)
    if t2 > 15:
        a_out, _ = pair_coordinate_descent(best_seq, t2, rng, verbose=True)
        try_update(a_out, f"cd1_n{n_cur}")

    # === Phase 3: More L-BFGS with different init ===
    n_cur = len(best_seq)
    remaining_check = remaining()
    if remaining_check > 100:
        print(f"\n=== Phase 3: L-BFGS perturbation restarts n={n_cur} ===")
        t3 = min(remaining_check * 0.15, 200.0)
        n_restarts = 3
        t3_each = t3 / n_restarts
        for r in range(n_restarts):
            if remaining() < 50:
                break
            pert = perturb_seq(best_seq, rng, scale=0.02 + 0.01 * r)
            beta_sch = [200.0, 500.0, 1000.0, 2000.0, 4000.0]
            beta_sch = [b * len(pert) / 1600.0 for b in beta_sch]
            a_out, _ = optimize_lbfgs(pert, beta_sch, t3_each, verbose=False, maxiter_per_beta=400)
            try_update(a_out, f"lbfgs_pert{r}_n{len(pert)}")
            # Quick CD
            if remaining() > 30:
                a_out2, _ = pair_coordinate_descent(a_out, min(20.0, remaining() * 0.3), rng, verbose=False)
                try_update(a_out2, f"cd_pert{r}_n{len(pert)}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration ===")
    lengths_to_try = []
    base_n = len(best_seq)
    for delta in [200, 400, 600, -200, -400]:
        nl = base_n + delta
        if 400 <= nl <= 5000:
            lengths_to_try.append(nl)
    # Also try doubling
    if base_n * 2 <= 5000:
        lengths_to_try.append(base_n * 2)
    # Remove duplicates and sort
    lengths_to_try = sorted(set(lengths_to_try))

    for nl in lengths_to_try:
        if remaining() < 60:
            break
        if nl == len(best_seq):
            continue
        print(f"  Trying n={nl}...")
        t_nl = min(remaining() * 0.12, 120.0)
        if t_nl < 20:
            break
        init_nl = interpolate_seq(best_seq, nl)
        beta_sch = [100.0, 300.0, 800.0, 2000.0, 5000.0]
        beta_sch = [b * nl / 1600.0 for b in beta_sch]
        a_out, _ = optimize_lbfgs(init_nl, beta_sch, t_nl * 0.7, verbose=False, maxiter_per_beta=300)
        try_update(a_out, f"lbfgs_n{nl}")
        if remaining() > 20:
            a_out2, _ = pair_coordinate_descent(a_out, t_nl * 0.3, rng, verbose=False)
            try_update(a_out2, f"cd_n{nl}")

    # === Phase 5: Subgradient method at best length ===
    n_cur = len(best_seq)
    if remaining() > 80:
        print(f"\n=== Phase 5: Subgradient n={n_cur} ===")
        t5 = min(remaining() * 0.12, 120.0)
        a_out, _ = subgradient_descent(best_seq, t5, rng, verbose=False)
        try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 6: Aggressive final L-BFGS with longer schedule ===
    n_cur = len(best_seq)
    if remaining() > 100:
        print(f"\n=== Phase 6: Final L-BFGS n={n_cur} ===")
        t6 = min(remaining() * 0.20, 250.0)
        # Finer annealing schedule
        beta_sch_final = [50.0, 100.0, 150.0, 200.0, 300.0, 500.0, 800.0, 1200.0,
                          2000.0, 3000.0, 5000.0, 8000.0, 12000.0]
        beta_sch_final = [b * n_cur / 1600.0 for b in beta_sch_final]
        a_out, _ = optimize_lbfgs(best_seq, beta_sch_final, t6, verbose=True, maxiter_per_beta=400)
        try_update(a_out, f"lbfgs_final_n{n_cur}")

    # === Phase 7: Final CD polish ===
    n_cur = len(best_seq)
    if remaining() > 30:
        print(f"\n=== Phase 7: Final CD n={n_cur} ===")
        t7 = min(remaining() * 0.40, 200.0)
        a_out, _ = pair_coordinate_descent(best_seq, t7, rng, verbose=True)
        try_update(a_out, f"cd_final_n{n_cur}")

    # === Phase 8: One more L-BFGS + CD cycle if time ===
    if remaining() > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 8: Extra cycle n={n_cur} ===")
        pert = perturb_seq(best_seq, rng, scale=0.01)
        t8 = remaining() * 0.6
        beta_sch = [500.0, 1000.0, 2000.0, 5000.0, 10000.0]
        beta_sch = [b * n_cur / 1600.0 for b in beta_sch]
        a_out, _ = optimize_lbfgs(pert, beta_sch, t8 * 0.6, verbose=False, maxiter_per_beta=500)
        try_update(a_out, f"extra_lbfgs_n{n_cur}")
        if remaining() > 15:
            a_out2, _ = pair_coordinate_descent(a_out, min(30.0, remaining() * 0.5), rng, verbose=False)
            try_update(a_out2, f"extra_cd_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
