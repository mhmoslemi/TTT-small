        k_int = int(k)
        # C[k] = sum_i y[i] * y[k-i], valid i: max(0, k-(n-1)) <= i <= min(n-1, k)
        i_start = max(0, k_int - (n - 1))
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            T[idx, i_start:i_end + 1] = y[k_int - i_end:k_int - i_start + 1][::-1]
    return T

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, K=50, refresh_interval=50):
    """Nesterov-accelerated Frank-Wolfe with reduced-constraint LP."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iter_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()

        # Periodically refresh constraints
        if iter_count % refresh_interval == 0:
            T = build_constraint_matrix(y, ac, K=K)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            iter_count += 1
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            iter_count += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            iter_count += 1
            continue
        g = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        # Check x as well
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.8f}")

    return best_a, best_m

def nesterov_lp_optimize_v2(a0, time_limit, solver, verbose=True, K=50, aggressive=False):
    """Nesterov FW with adaptive constraint count and occasional full check."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    T = None
    K_current = K

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()

        # Refresh constraints every 20 iterations or if K changed
        if T is None or t % 20 == 1:
            if aggressive and t > 100:
                K_current = max(20, K - t // 100)
            T = build_constraint_matrix(y, ac, K=K_current)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum

        # Check gap to detect convergence
        gap = M_curr - (T @ g).max()
        if gap < 1e-10 * M_curr:
            # Very close to convergence, try smaller steps
            alpha = 1.0 / (t + 1.0)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPv2 t={t} ({elapsed:.1f}s): J={J:.8f} K={K_current}")

    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent: move mass to reduce max convolution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        C = fast_conv(a)
        M_curr = C.max()
        k_max = int(np.argmax(C))

        # Compute sensitivity: dC[k_max]/da[i] = a[k_max-i] + a[k_max+i] (valid terms)
        sens = np.zeros(n)
        for i in range(n):
            s = 0.0
            j1 = k_max - i
            if 0 <= j1 < n:
                s += a[j1]
            j2 = k_max + i
            if 0 <= j2 < 2 * n - 1 and i != j2:
                # This term only exists if k_max + i < 2n-1, but a only has n elements
                pass
            # Actually C[k] = sum_i a[i]*a[k-i], so dC[k_max]/da[i] = a[k_max-i] (if valid)
            # plus a[i'] where i' + i = k_max, i.e., i' = k_max - i (same thing)
            # Wait: C[k_max] = sum_j a[j]*a[k_max-j]
            # dC[k_max]/da[i] = a[k_max-i] (from j=i term) + a[i] (from j=k_max-i term, if k_max-i != i)
            # Actually: d/d(a[i]) [sum_j a[j]*a[k_max-j]] = a[k_max-i] + (if k_max-i != i) a[i] * 0... 
            # No: C[k_max] = sum_j a[j] * a[k_max - j]
            # dC/d(a[i]) appears when j=i: a[k_max-i] (if 0<=k_max-i<n)
            # and when k_max-j=i, i.e., j=k_max-i: a[j]=a[k_max-i], derivative w.r.t a[i] gives a[k_max-i]
            # So dC[k_max]/da[i] = a[k_max-i] + a[k_max-i] = 2*a[k_max-i] if k_max-i != i
            # or a[k_max-i] = a[i] if k_max-i == i (i.e., k_max = 2i)
            if 0 <= k_max - i < n:
                if k_max - i == i:
                    sens[i] = a[i]
                else:
                    sens[i] = 2.0 * a[k_max - i]

        # Find best source (high sensitivity) and destination (low sensitivity)
        src = int(np.argmax(sens * a))  # weighted by current value
        # For destination, prefer low sensitivity and also consider all peaks
        top_peaks = np.argsort(C)[::-1][:10]
        dst_sens = np.zeros(n)
        for kp in top_peaks:
            kp_int = int(kp)
            for i in range(n):
                if 0 <= kp_int - i < n:
                    if kp_int - i == i:
                        dst_sens[i] += a[i]
                    else:
                        dst_sens[i] += 2.0 * a[kp_int - i]

        dst = int(np.argmin(dst_sens))
        if dst == src:
            dst = int(np.argmin(dst_sens[np.delete(np.arange(n), src)])) + (1 if np.argmin(dst_sens) > src else 0)
            if dst >= n:
                dst = 0

        improved = False
        for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]:
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-15:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation
            candidates = np.where(a > 1e-8)[0]
            if len(candidates) > 1:
                src_r = candidates[rng.integers(len(candidates))]
                dst_r = rng.integers(n)
                if dst_r != src_r:
                    frac_r = rng.uniform(0.002, 0.02)
                    amt_r = a[src_r] * frac_r
                    a[src_r] -= amt_r
                    a[dst_r] += amt_r
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[src_r] += amt_r
                        a[dst_r] -= amt_r
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.8f} stall={stall}")

    return best_a, best_m

def mirror_descent_optimize(a0, time_limit, top_k=30, beta=4.0, verbose=True):
    """Mirror descent with soft subgradient on the convolution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        step = 0.5 / (np.sqrt(t) + 1.0)
        C = fast_conv(a)

        # Soft subgradient: weighted sum of active constraint gradients
        top_idx = np.argsort(C)[::-1][:top_k]
        top_vals = np.maximum(C[top_idx], 1e-30)
        w = top_vals ** beta
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    if j == i:
                        grad[i] += wi * a[i]
                    else:
                        grad[i] += 2.0 * wi * a[j]

        # Mirror descent step (exponential family)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M_new = fast_conv(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.8f}")

    return best_a, best_m

def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0, 1, n)

    # Sine window
    s = np.sin(np.pi * x) ** 2
    inits.append(s / s.sum())

    # Cosine window
    c = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(c / c.sum())

    # Uniform
    u = np.ones(n)
    inits.append(u / u.sum())

    # Triangular
    tri = np.sin(np.pi * x)
    inits.append(tri / tri.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Random
    for _ in range(3):
        r = rng.random(n)
        r = np.maximum(r, 0.01)
        inits.append(r / r.sum())

    # Geometric (decaying from center)
    center = n // 2
    dist = np.abs(np.arange(n) - center) / center
    geo = np.exp(-dist * 2)
    inits.append(geo / geo.sum())

    # Two-peaked
    two = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    inits.append(two / two.sum())

    return inits

def propose_candidate(budget_s=1800.0):
    """
    Main search function: multi-phase optimization with LP-FW, coordinate polish,
    and mirror descent. Uses reduced-constraint LP for speed.
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    n_parent = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            n_parent = len(p)
            p_score = evaluate_safe(p)
            print(f"Parent: n={n_parent}, score={p_score:.8f}")
            best_seq = p / p.sum()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_parent = 1600
        best_seq = np.sin(np.pi * np.linspace(0, 1, n_parent)) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: LP-FW on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW (K=50) n={n_parent} ===")
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_lp_optimize(a_in, t1, solver, verbose=True, K=50, refresh_interval=30)
        try_update(a_out, f"lp1_n{n_parent}")

    # === Phase 2: LP-FW v2 with more aggressive K reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW v2 (adaptive K) n={len(best_seq)} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_lp_optimize_v2(a_in, t2, solver, verbose=True, K=40, aggressive=True)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # === Phase 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coordinate polish n={len(best_seq)} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t3, verbose=True)
        try_update(a_out, f"coord1_n{len(best_seq)}")

    # === Phase 4: Try different initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_best = len(best_seq)
        print(f"\n=== Phase 4: Alternative init + LP n={n_best} ===")
        inits = generate_diverse_inits(n_best, rng)
        t4_per = min(remaining * 0.15 / 3, 60.0)
        for idx, init in enumerate(inits[:3]):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            a_in = 0.7 * norm_sum1(best_seq) + 0.3 * init
            a_in = norm_sum1(a_in)
            a_out, _ = nesterov_lp_optimize(a_in, min(t4_per, remaining * 0.4), solver, verbose=False, K=40)
            try_update(a_out, f"alt{idx}_n{n_best}")

    # === Phase 5: Mirror descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Mirror descent n={len(best_seq)} ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t5, top_k=30, beta=4.0, verbose=True)
        try_update(a_out, f"md_n{len(best_seq)}")

    # === Phase 6: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final coord polish n={len(best_seq)} ===")
    if remaining > 30:
        t6 = min(remaining * 0.4, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True)
        try_update(a_out, f"coord2_n{len(best_seq)}")

    # === Phase 7: One more LP round if time allows ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = min(remaining * 0.5, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_lp_optimize(a_in, t7, solver, verbose=True, K=30)
        try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
