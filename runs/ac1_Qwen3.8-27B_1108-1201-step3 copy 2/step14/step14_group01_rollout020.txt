```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP best-response with Nesterov acceleration (scipy HiGHS solver)
2. Adaptive constraint screening with cross-convolution violation checks
3. LSE smoothed gradient descent for fast initial progress
4. Multi-length exploration with interpolation
5. Gradient-guided pair coordinate descent for final polish
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def eval_fn(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    return float(2.0 * n * fast_conv_self(a).max() / (S * S))


def build_A_ub(a, active_idx):
    """Build constraint matrix for LP: row k has a[k-i] at column i."""
    n = len(a)
    pieces = len(active_idx)
    A_ub = np.zeros((pieces, n))
    for r in range(pieces):
        k = active_idx[r]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_hi >= i_lo:
            idx = np.arange(i_lo, i_hi + 1)
            A_ub[r, idx] = a[k - idx]
    return A_ub


def lp_best_response(a, M, active_idx):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for k in active_idx, b >= 0."""
    n = len(a)
    A_ub = build_A_ub(a, active_idx)
    c = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.maximum(res.x, 0.0)


def lse_gradient_descent(a0, time_budget):
    """LSE smoothed gradient descent for fast initial progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(40.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(10, min(100, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_budget:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_max:
                best_max = sc
                best_a = a.copy()

        if si < n_stages:
            print(f"    LSE stage {si} T={T_val:.1f}: max={best_max:.8f}")

    return best_a, best_max


def nesterov_lp_optimize(a0, time_budget):
    """LP best-response with Nesterov acceleration and adaptive constraint screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / a.sum()

    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Initialize active set
    conv_init = fast_conv_self(y)
    threshold_init = conv_init.max() * 0.90
    active = set(np.where(conv_init >= threshold_init)[0].tolist())
    if len(active) < 30:
        active = set(np.where(conv_init >= conv_init.max() * 0.7)[0].tolist())

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_budget:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Periodically refresh active set
        if t % 100 == 1:
            thr = M * 0.92
            new_active = set(np.where(conv_y >= thr)[0].tolist())
            if len(new_active) < 30:
                new_active = set(np.where(conv_y >= M * 0.7)[0].tolist())
            # Union with current to avoid losing constraints
            active = active | new_active

        sorted_active = np.array(sorted(active))
        b = lp_best_response(y, M, sorted_active)
        if b is None:
            active = set(range(size))
            b = lp_best_response(y, M, np.arange(size))
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Check dropped constraints
        if len(active) < size:
            cross = fast_conv_cross(y, g)
            violated = np.where(cross > M * 1.001)[0]
            if len(violated) > 0:
                active.update(violated.tolist())
                sorted_active2 = np.array(sorted(active))
                b2 = lp_best_response(y, M, sorted_active2)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        max_y = fast_conv_self(y).max()
        if max_y < best_max:
            best_max = max_y
            best_a = y.copy()

        max_a = fast_conv_self(a).max()
        if max_a < best_max:
            best_max = max_a
            best_a = a.copy()

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Ne t={t} ({elapsed:.0f}s): max={best_max:.8f} |A|={len(active)}")

    return best_a, best_max


def pair_cd_optimize(a0, time_budget, rng):

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / a.sum()
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0

    while time.monotonic() - t0 < time_budget:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Candidate j's
            j_cands = set()
            for _ in range(6):
                j_cands.add(int(rng.integers(n)))
            peak_lags = np.where(conv >= cur_max * 0.95)[0]
            for kl in peak_lags[:4]:
                j_cands.add((kl - i) % n)
                j_cands.add((kl + i) % n)
            j_cands.discard(i)

            conv = fast_conv_self(a)
            cur_max = conv.max()

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_cands:
                if j == i:
                    continue
                aj = a[j]

                # Gain vector: change in conv when moving delta from i to j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # 1D optimization over delta in [0, ai]
                n_samples = 25
                deltas = np.linspace(0.01, ai * 0.99, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] -= 2 * ai * d - 2 * d * d
                    if 2 * j < size:
                        new_conv[2 * j] += 2 * aj * d + 2 * d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_max - 1e-12:
                        best_max = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_imp += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 4:
                n_pert = max(5, n // 40)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if total_imp % 200 == 0 and total_imp > 0:
            print(f"    PCD imp={total_imp} ({time.monotonic()-t0:.0f}s): max={best_max:.8f}")

    return best_a, best_max


def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(10 * np.pi * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(8 * np.pi * x) * np.cos(4 * np.pi * x)
    elif choice == 8:
        a = np.abs(np.sin(np.pi * x * 3.0)) ** 0.8 + 0.2
    else:
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = min(remaining * 0.12, 120.0)
    print(f"\n=== Phase 1: LSE GD n={n1} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1)
        try_update(a_out, f"lse1_n{n1}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2)
        try_update(a_out, f"lp_n{n2}")

    # Phase 3: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Pair CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng)
        try_update(a_out, f"pcd_n{n3}")

    # Phase 4: Second LP Nesterov round
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4)
        try_update(a_out, f"lp2_n{n4}")

    # Phase 5: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t5 = min(remaining * 0.30, 200.0)
    print(f"\n=== Phase 5: Longer length n={n_long} (budget={t5:.0f}s) ===")
    if t5 > 40 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, min(t5, 120.0))
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            a_out2, _ = pair_cd_optimize(best_seq, min(remaining * 0.5, 60.0), rng)
            try_update(a_out2, f"pcd_long")

    # Phase 6: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t_half = remaining * 0.5
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 60.0))
        try_update(a_out, f"lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.7, 60.0), rng)
            try_update(a_out, f"pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
