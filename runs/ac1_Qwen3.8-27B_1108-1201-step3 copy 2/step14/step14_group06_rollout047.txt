        else:
            eps = 0.03
            M_used = M_curr * (1.0 - eps)
        
        T = build_constraint_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr * 0.999, solver, T)
            if b is None:
                b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1
        
        # Adaptive step size based on improvement ratio
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.05), 1.0)
        
        # Nesterov extrapolation
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to(y_new, norm_val)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, norm_val)
        
        y = y_new
        a = a_new
        
        # Check improvements
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
        
        # Stalling detection
        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 8:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def subgradient_optimize(a0, time_limit, verbose=True):
    """Subgradient method with adaptive step size for the non-smooth objective."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, 1.0)  # Normalize to sum=1 for simplicity
    
    best_a = a.copy()
    # Objective with sum=1: 2n * max(conv)
    best_obj = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    last_improve = time.monotonic()
    stall = 0
    
    # Polyak-like step size
    best_so_far = best_obj
    c_step = 1.0
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_autocorr(a)
        max_val = conv_a.max()
        # Active lags (within 5% of max)
        threshold = max_val * 0.95
        active = np.where(conv_a >= threshold)[0]
        
        # Subgradient of max_k conv_k(a) at point a:
        # It's the average of gradients of active constraints
        # grad of conv_k(a) = 2 * T_k @ a (by symmetry of convolution)
        # Actually conv_k(a) = sum_i a[i]*a[k-i], so d(conv_k)/da[j] = a[k-j] + a[j-k] (if valid)
        # For the full convolution: d(conv_k)/da[j] = sum over valid terms
        
        # Compute subgradient more efficiently using FFT
        # The gradient of conv[a,a] wrt a is 2*conv[a, ·] (circular)
        # For linear: gradient of conv_k wrt a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        
        # Build the subgradient
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                # d conv[k] / da[j]
                val = 0.0
                if 0 <= k - j < n:
                    val += a[k - j]
                if 0 <= j - k < n:
                    val += a[j - k]
                grad[j] += val / len(active)
        
        # Normalize: we want to minimize max(conv) subject to sum(a)=1, a>=0
        # Project gradient onto the constraint set
        grad_proj = grad - np.mean(grad)  # Project onto sum=0 hyperplane
        grad_norm = np.linalg.norm(grad_proj)
        
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size
        t_curr = it + 1
        step = c_step / np.sqrt(t_curr)
        # Make step small enough to stay in feasible region
        a_trial = a - step * grad_proj
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s
        
        # Check improvement
        obj_trial = 2.0 * n * fast_autocorr(a_trial).max()
        if obj_trial < best_obj - 1e-10:
            best_obj = obj_trial
            best_a = a_trial.copy()
            last_improve = time.monotonic()
            best_so_far = obj_trial
            a = a_trial.copy()
        else:
            # Try smaller step
            for factor in [0.5, 0.1, 0.01]:
                a_trial2 = a - step * factor * grad_proj
                a_trial2 = np.maximum(a_trial2, 0.0)
                s2 = a_trial2.sum()
                if s2 < 1e-10:
                    continue
                a_trial2 = a_trial2 / s2
                obj2 = 2.0 * n * fast_autocorr(a_trial2).max()
                if obj2 < best_obj - 1e-10:
                    best_obj = obj2
                    best_a = a_trial2.copy()
                    a = a_trial2.copy()
                    last_improve = time.monotonic()
                    break
        
        # Reduce step if no improvement
        if time.monotonic() - last_improve > 30.0:
            c_step *= 0.7
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it} ({elapsed:.0f}s): best={best_obj:.8f}")
    
    return normalize_to(best_a, norm_val), best_obj / (2.0 * n)

def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    
    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

def lse_optimize(a0, time_limit, T_start=60.0, T_end=1.0, n_stages=4):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm_val)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)
        
        def obj_func(x, T_val=T):
            val, _ = lse_objective_and_grad(x, n, T_val)
            return val
        
        def grad_func(x, T_val=T):
            _, g = lse_objective_and_grad(x, n, T_val)
            return g
        
        try:
            # Use multiple starting points for the stage
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 500, 'maxfun': 2000,
                                           'gtol': 1e-10, 'ftol': 1e-12,
                                           'maxls': 80})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                a_candidate = normalize_to(a_candidate, norm_val)
                score_c = fast_autocorr(a_candidate).max()
                if score_c < best_score:
                    best_score = score_c
                    best_a = a_candidate.copy()
                    a = a_candidate.copy()
                    print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")
    
    return best_a, best_score

def coord_descent_improved(a0, time_limit, seed=42):
    """Improved coordinate descent with line search and peak targeting."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm_val)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        
        # Find elements contributing to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]
        
        # Also consider other high peaks
        top_peaks = np.argsort(b)[::-1][:20]
        all_targets = set()
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            for j2 in range(js2, je2 + 1):
                all_targets.add(j2)
        
        # Low-value elements (good targets for receiving mass)
        low_val = np.argsort(a)[:20]
        
        improved = False
        
        # Try moving mass from high contributors to low-value elements
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val:
                if dst == src:
                    continue
                # Line search: find best fraction to move
                best_frac = 0
                best_trial_score = best_score
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2, 0.35]:
                    amt = a[src] * frac
                    if a[src] - amt < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = normalize_to(a_trial, norm_val)
                    score = fast_autocorr(a_trial).max()
                    if score < best_trial_score:
                        best_trial_score = score
                        best_frac = frac
                
                if best_frac > 0 and best_trial_score < best_score - 1e-12:
                    amt = a[src] * best_frac
                    a[src] -= amt
                    a[dst] += amt
                    a = normalize_to(a, norm_val)
                    best_score = best_trial_score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            # Random moves
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = normalize_to(a_trial, norm_val)
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(6 * np.pi * x) + 0.05 * np.cos(10 * np.pi * x)
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        a = np.sin(np.pi * x) ** 2 * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif choice == 7:
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))  # sigmoid
        a = a - a.min() + 0.1
    elif choice == 8:
        a = np.cos(np.pi * x) ** 2 * np.sin(np.pi * x) ** 0.5
    else:
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 1.5
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
