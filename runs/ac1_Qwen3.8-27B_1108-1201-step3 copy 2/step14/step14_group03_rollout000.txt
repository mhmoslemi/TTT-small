            best_score = score_y
            best_a = y.copy()
            last_best_iter = n_iters

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_best_iter = n_iters

        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_best_iter = n_iters

        if n_iters - last_best_iter > 80:
            a = y.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    if verbose:
        print(f"    NLP done: {n_iters} iters, best={best_score:.8f}")
    return best_a, best_score


def pair_cd_window(a, time_limit, rng, window=30, verbose=True):
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    delta_vec = np.zeros(size)
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        src = np.arange(lo1 - i, hi1 - i)
                        delta_vec[lo1:hi1] += a[src]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 + i + 1)
                    if hi2 > lo2:
                        src2 = n - 1 - np.arange(lo2, hi2)
                        valid = (src2 >= 0) & (src2 < n)
                        delta_vec[lo2:hi2][valid] += a[src2[valid]]

                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        src3 = np.arange(lo3 - j, hi3 - j)
                        delta_vec[lo3:hi3] -= a[src3]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 + j + 1)
                    if hi4 > lo4:
                        src4 = n - 1 - np.arange(lo4, hi4)
                        valid4 = (src4 >= 0) & (src4 < n)
                        delta_vec[lo4:hi4][valid4] -= a[src4[valid4]]

                    delta_vec *= 2.0

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 25
                    deltas = np.linspace(0.0, a[i] * 0.99, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-13:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    if verbose:
        print(f"    PCD done: {n_passes} passes, best={best_score:.8f}")
    return best_a, best_score


def peak_shave(a, time_limit, verbose=True):
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_iter = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        conv = fast_autocorr(a)
        curr_max = conv.max()
        peak_indices = np.where(conv > curr_max * 0.999)[0]

        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            # Find which a[i] contribute most to conv[k]
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            contributions = np.zeros(n)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    contributions[i] = a[i] * a[j]
                    contributions[j] += a[i] * a[j]

            top_idx = np.argsort(contributions)[::-1]
            for i in top_idx[:5]:
                if contributions[i] < norm * norm * 1e-6:
                    continue
                # Try reducing a[i] and distributing to random positions
                for trial in range(3):
                    if time.monotonic() - t0 > time_limit:
                        break
                    frac = 0.1 + 0.1 * trial
                    d = a[i] * frac
                    if d < 1e-12:
                        continue
                    j = int(np.argmax(np.random.random(n)) ) if False else int(np.argmin(conv[0:n]))
                    # Better: move to position with lowest local autocorrelation impact
                    # Compute which j would minimize the new max
                    best_j = -1
                    best_new_max = curr_max
                    for j_candidate in range(0, n, max(1, n // 50)):
                        if j_candidate == i:
                            continue
                        # Quick estimate: check conv at key positions
                        new_conv_approx = conv.copy()
                        new_conv_approx[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        new_conv_approx[2 * j_candidate] = conv[2 * j_candidate] + 2 * a[j_candidate] * d + d * d
                        # Approximate cross terms
                        for pk in [i + j_candidate, abs(i - j_candidate)]:
                            if pk < size:
                                new_conv_approx[pk] = conv[pk] - 2 * a[i] * a[j_candidate] + 2 * d * (a[j_candidate] + a[i] - d)
                        nm = new_conv_approx.max()
                        if nm < best_new_max:
                            best_new_max = nm
                            best_j = j_candidate

                    if best_j >= 0 and best_new_max < curr_max * (1 - 1e-8):
                        a[i] -= d
                        a[best_j] += d
                        s = a.sum()
                        if s > 1e-15:
                            a = a * norm / s
                            improved = True
                            break

        if improved:
            stagnant = 0
            conv = fast_autocorr(a)
            cur_max = conv.max()
            if cur_max < best_score - 1e-14:
                best_score = cur_max
                best_a = a.copy()
        else:
            stagnant += 1
            if stagnant > 50:
                break

        if verbose and n_iter % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave iter={n_iter} ({elapsed:.0f}s): best={best_score:.8f}")

    if verbose:
        print(f"    PeakShave done: {n_iter} iters, best={best_score:.8f}")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def multi_start_explore(a_best, time_limit, rng, solver, verbose=True):
    n = len(a_best)
    norm = np.sqrt(2.0 * n)
    best_a = a_best.copy()
    best_score = fast_autocorr(a_best).max()
    t0 = time.monotonic()
    n_attempts = 0

    while time.monotonic() - t0 < time_limit:
        # Perturb the best solution
        epsilon = 0.02 + 0.03 * rng.random()
        a_pert = a_best + epsilon * (rng.random(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert * norm / s
        a_pert = a_pert / a_pert.sum()

        # Quick LP Nesterov from perturbed point
        remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(remaining * 0.3, 30.0)
        if t_sub < 5:
            break

        a_out, _ = nesterov_lp_optimize(a_pert, t_sub, solver, verbose=False)
        n_attempts += 1

        sc = fast_autocorr(a_out).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    MultiStart attempt {n_attempts}: improved to {best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov - main workhorse (55% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={remaining * 0.55:.0f}s) ===")
    t1 = remaining * 0.55
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: Pair CD polish (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: PCD n={n_best} (budget={remaining * 0.15:.0f}s) ===")
    t2 = remaining * 0.15
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t2, rng, window=25, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 3: Second LP Nesterov round (20% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP round 2 n={n_best} (budget={remaining * 0.20:.0f}s) ===")
    t3 = remaining * 0.20
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 4: Pair CD with larger window (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: PCD large window n={n_best} (budget={remaining * 0.15:.0f}s) ===")
    t4 = remaining * 0.15
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t4, rng, window=50, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # Phase 5: Peak shaving (10% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak shave n={n_best} (budget={remaining * 0.10:.0f}s) ===")
    t5 = remaining * 0.10
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t5, verbose=True)
        do_try_update(a_out, f"peak_n{n_best}")

    # Phase 6: Multi-start exploration (10% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 6: Multi-start n={n_best} (budget={remaining * 0.10:.0f}s) ===")
        t6 = remaining * 0.10
        if t6 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = multi_start_explore(a_norm, t6, rng, solver, verbose=True)
            do_try_update(a_out, f"multi_n{n_best}")

    # Phase 7: Final LP Nesterov squeeze (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 7: Final LP squeeze n={n_best} (budget={remaining * 0.6:.0f}s) ===")
        t7 = remaining * 0.6
        if t7 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
            do_try_update(a_out, f"lp3_n{n_best}")

    # Phase 8: Final PCD squeeze
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        print(f"\n=== Phase 8: Final PCD n={n_best} (budget={remaining * 0.5:.0f}s) ===")
        t8 = remaining * 0.5
        if t8 > 5:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_cd_window(a_norm, t8, rng, window=40, verbose=True)
            do_try_update(a_out, f"pcd3_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
