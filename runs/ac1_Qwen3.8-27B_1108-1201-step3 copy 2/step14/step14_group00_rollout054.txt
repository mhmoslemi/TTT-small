"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration (brief global phase)
2. Focused coordinate descent on peak-contributing positions (main refinement)
3. Mirror descent / multiplicative weights for diverse global moves
4. Multi-length exploration and final polish
Key improvements: pressure-based coordinate selection, O(n) move evaluation,
adaptive scheduling, and aggressive local search.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_local_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))

def mirror_descent_optimize(a0, budget, seed=0):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    conv = fast_autocorr(a)
    stall = 0
    for t in range(1, 100000000):
        if time.monotonic() - t0 > budget:
            break
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            grad[lo:hi + 1] += a[k_star - np.arange(lo, hi + 1)]
        lo = max(0, k_star)
        hi = min(n - 1, k_star + n - 1)
        if hi >= lo:
            grad[lo:hi + 1] += a[np.arange(lo, hi + 1) - k_star]
        gmax = grad.max()
        if gmax < 1e-15:
            break
        eta = 0.8 / (1.0 + 0.5 * np.log1p(t))
        a = a * np.exp(-eta * grad / gmax)
        s = a.sum()
        if s < 1e-30:
            a = np.ones(n) / n
        else:
            a = a / s
        conv = fast_autocorr(a)
        cs = 2.0 * n * conv.max()
        if cs < best_score - 1e-12:
            best_score = cs
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                a = best_a + 0.02 * rng.standard_normal(n)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                conv = fast_autocorr(a)
                stall = 0
    return best_a, best_score


def coord_descent_optimize(a0, budget, seed=0, verbose=False):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    conv = fast_autocorr(a)
    size = 2 * n - 1
    no_improve = 0
    n_passes = 0
    delta_fracs = np.array([0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7])
    while time.monotonic() - t0 < budget and no_improve < 200:
        n_passes += 1
        k_star = int(np.argmax(conv))
        pressure = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            pressure[lo:hi + 1] += a[k_star - np.arange(lo, hi + 1)]
        lo = max(0, k_star)
        hi = min(n - 1, k_star + n - 1)
        if hi >= lo:
            pressure[lo:hi + 1] += a[np.arange(lo, hi + 1) - k_star]
        order = np.argsort(pressure)
        n_top = max(8, n // 15)
        high_idx = order[-n_top:]
        low_idx = order[:n_top]
        pass_improved = False
        n_trials = max(n_top * 3, 40)
        for trial in range(n_trials):
            if time.monotonic() - t0 > budget:
                break
            i = int(high_idx[rng.integers(n_top)])
            j = int(low_idx[rng.integers(n_top)])
            if i == j:
                continue
            ai = a[i]
            if ai < 1e-12:
                continue
            best_delta = 0.0
            best_new_max = conv[k_star]
            shift_i = np.zeros(size)
            shift_i[i:i + n] = a
            shift_j = np.zeros(size)
            shift_j[j:j + n] = a
            lin = shift_i - shift_j
            for frac in delta_fracs:
                d = ai * frac
                new_max = (conv - 2.0 * d * lin).max()
                q2j = 2 * j
                qij = i + j
                q2i = 2 * i
                if q2j < size:
                    new_max = max(new_max, conv[q2j] - 2.0 * d * lin[q2j] + d * d)
                if qij < size:
                    new_max = max(new_max, conv[qij] - 2.0 * d * lin[qij] - 2.0 * d * d)
                if q2i < size:
                    new_max = max(new_max, conv[q2i] - 2.0 * d * lin[q2i] + d * d)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = d
            if best_delta > 0 and best_new_max < conv[k_star] - 1e-13:
                a[i] -= best_delta
                a[j] += best_delta
                a = np.maximum(a, 0.0)
                cs = a.sum()
                if cs < 1e-10:
                    a[i] += best_delta
                    a[j] -= best_delta
                    continue
                a = a / cs
                conv = fast_autocorr(a)
                cs2 = 2.0 * n * conv.max()
                if cs2 < best_score:
                    best_score = cs2
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[j] -= best_delta
                    a[i] += best_delta
                    a = a / a.sum()
                    conv = fast_autocorr(a)
        if not pass_improved:
            no_improve += 1
            if no_improve % 10 == 0:
                n_pert = max(3, n // 50)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                a[pert_idx] += 0.05 * a.mean()
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = fast_autocorr(a)
        else:
            no_improve = 0
        if verbose and n_passes % 50 == 0:
            print(f"      CD pass={n_passes}: score={best_score:.8f}")
    return best_a, best_score


def lp_nesterov_optimize(a0, budget, solver, seed=0, verbose=False):
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s0
    norm = np.sqrt(2.0 * n)
    a_norm = a * norm
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a_norm.copy()
    a_mom = a_norm.copy()
    active_set = None
    stall = 0
    size = 2 * n - 1
    for t in range(1, 5000000):
        if time.monotonic() - t0 > budget:
            break
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        if active_set is None or t % 50 == 0:
            tol = 0.03
            active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
            if len(active_set) < 10:
                active_set = np.where(conv_y > M_curr * 0.85)[0]
        n = len(y)
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_set), n))
        for row_i, k in enumerate(active_set):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                idx = k - j
                if 0 <= idx < n:
                    T[row_i, j] = y[idx]
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            b = None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            b = None
        if b is None:
            if t % 30 == 0:
                active_set = None
                b_var = cp.Variable(n, nonneg=True)
                T = np.zeros((size, n))
                for j in range(n):
                    T[j:j + n, j] = y
                prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_curr])
                try:
                    prob.solve(solver=solver, verbose=False)
                except Exception:
                    b = None
            if b is None:
                continue
            active_set = np.arange(size)
        b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b.sum()
        if b_sum < 1e-10:
            continue
        g = b * norm / b_sum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy
        if t == 1:
            a_mom = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.92)
            a_mom = y_new + momentum * (y_new - y)
            a_mom = np.maximum(a_mom, 0.0)
            sa = a_mom.sum()
            if sa < 1e-10:
                a_mom = y_new.copy()
            else:
                a_mom = a_mom * norm / sa
        y = y_new
        prev_best = best_score
        sy2 = y / norm
        cs_y = 2.0 * n * fast_autocorr(sy2).max()
        if cs_y < best_score:
            best_score = cs_y
            best_a = sy2.copy()
            stall = 0
        else:
            stall += 1
        am = a_mom / norm
        cs_am = 2.0 * n * fast_autocorr(am).max()
        if cs_am < best_score:
            best_score = cs_am
            best_a = am.copy()
            stall = 0
        if stall > 80:
            y = best_a * norm
            a_mom = best_a * norm
            active_set = None
            stall = 0
            if verbose:
                print(f"      LP restart at t={t}")
        if verbose and t % 500 == 0:
            print(f"      LP t={t}: score={best_score:.8f}")
    return best_a, best_score


def interpolate_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-10:
        return np.ones(new_n) / new_n
    return fi / ss
