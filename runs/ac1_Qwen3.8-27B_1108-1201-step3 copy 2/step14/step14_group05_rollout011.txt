    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax(x, T=1.0):
    x = np.asarray(x, dtype=np.float64)
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted / T)
    return exp_x / exp_x.sum()

def lse_gradient(a, T):
    """Compute gradient of LSE-smoothed max autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    w = softmax(b, T)
    # grad_j = 2 * sum_m w_{j+m} * a_m
    # This is the cross-correlation of w and a
    w_rev = w[::-1]
    # convolve(w_rev, a) gives length (2n-1) + n - 1 = 3n-2
    # We need elements at positions n-1, n, ..., 2n-2 (length n)
    # Actually, let me use np.correlate
    h = np.correlate(w, a, mode='valid')  # length n
    return 2.0 * h

def lp_max_sum(a, M, solver):
    """Solve max sum(b) s.t. T@b <= M, b >= 0 using ALL constraints."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint T @ b <= M
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Instead of building the full matrix, use the structure
    # But for cvxpy, we need to express this explicitly
    
    # For efficiency, build the constraint using the Toeplitz structure
    # T @ b has element k: sum_{j: 0<=k-j<n} a[k-j] * b[j]
    # This is the convolution of a and b (reversed)
    
    # For cvxpy, let's build the matrix explicitly (sparse)
    # Actually for n=1600, the matrix is 3199 x 1600, which is ~5M entries
    # Most are zero. Let's use sparse.
    
    # Build T as a scipy.sparse matrix
    import scipy.sparse as sp
    
    # T is (2n-1) x n
    # T[k, j] = a[k-j] if 0 <= k-j < n else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j+n):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
