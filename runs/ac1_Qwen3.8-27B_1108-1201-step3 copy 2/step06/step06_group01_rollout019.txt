Let me now think about a more sophisticated approach: **mirror descent with adaptive step size.**

Mirror descent on the simplex uses the Bregman divergence with respect to the negative entropy:
D_KL(p || q) = sum_i p[i] * log(p[i]/q[i])

The update is:
p_new[i] = p[i] * exp(-eta * g[i]) / Z

where g is the (sub)gradient and Z is a normalizing constant.

This is equivalent to the "exponentiated gradient" or "multiplicative weights" update.

The advantage of mirror descent over projected gradient is that it naturally respects the simplex constraint (no projection needed) and can be much more effective in high dimensions.

The step size eta can be chosen adaptively using the "zoom" algorithm of Hazan (2016), which tries increasing step sizes and picks the largest one that doesn't increase the objective.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_subgradient(a, active_indices):
    """Compute subgradient of max_k (a*a)[k] at point a."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active_indices)
    return g


def soft_max_obj_and_grad_z(a, T, n):
    """Compute soft-max smoothed objective and gradient in log-space."""
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n) * (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    # Gradient w.r.t. a: 2*n * sum_k w[k] * a[k-i] for each i
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    # Convert to log-space gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_soft_max(a0, time_limit, T_start=100.0, T_end=5.0, maxiter=5000, T_target=100):
    """L-BFGS refinement with soft-max smoothing and adaptive temperature."""
    from scipy.optimize import minimize
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    # Geometric temperature schedule
    n_stages = 5
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': iters_per_stage * 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=5000):
    """
    Nesterov dual averaging style LP refinement.
    Maintains a "virtual" iterate y that leads the actual iterate x.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()  # virtual iterate
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP for best response to y
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov dual averaging step size
        alpha = 2.0 / (it + 2)

        # Update y (virtual iterate)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Update a (actual iterate) using weighted average
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        # Also evaluate y
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, lr_init=0.1, n_steps=50000):
    """
    Mirror descent (exponentiated gradient) on the simplex.
    Uses subgradient of the max-conv objective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        # Find active indices (near-max)
        threshold = max_b - 0.02 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Mirror descent update in log-space
        z = z - lr * g
        a_new = softmax_from_z(z)

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Backup: keep old a, don't update
            z = z + lr * g  # undo

        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
        lr = max(lr, 1e-6)

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_passes=100):
    """Cyclic coordinate descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Optimize coordinate i: a[i] = t, others scaled
            if a[i] < 1e-15:
                continue
            # Try a few values
            base = a.copy()
            base[i] = 0
            base_sum = base.sum()
            if base_sum < 1e-15:
                continue
            t_new = a[i]
            best_t = t_new
            best_local = best_score
            # Golden section or just try a few
            for factor in [0.7, 0.85, 1.0, 1.15, 1.3]:
                trial_t = t_new * factor
                if trial_t < 0:
                    continue
                a_trial = base * (1.0 - trial_t) + trial_t * np.eye(n)[i]
                if a_trial.sum() < 1e-15:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_local:
                    best_local = sc
                    best_t = trial_t
            if best_t != t_new:
                a = base * (1.0 - best_t) + best_t * np.eye(n)[i]
                a = a / a.sum()
                sc = 2.0 * n * fast_convolve(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if p % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc pass={p} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=5000):
    """Aggressive local peak shaving with multiple coordinate adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_mid = b.copy()
        b_mid[0] = -np.inf
        b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.93, 0.88, 0.8]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing both i and j
            if len(top_contrib) >= 2:
                for ci in top_contrib[:3]:
                    i = idx[ci]
                    j = peak_k - i
                    if i == j or a[j] < 1e-14:
                        continue
                    for f1 in [0.95, 0.9]:
                        for f2 in [0.95, 0.9]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
            if not improved:
                # Try reducing all top contributors slightly
                top3 = top_contrib[:3]
                a_trial = a.copy()
                for ci in top3:
                    i = idx[ci]
                    a_trial[i] *= 0.95
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                if not improved:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely, returning inf on failure."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing the auto-convolution peak.
    Combines L-BFGS soft-max, mirror descent, LP-based Nesterov,
    coordinate descent, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: L-BFGS soft-max with geometric temperature (700s) ===
    print(f"\n=== Phase 1: L-BFGS Soft-Max (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 200->5, n={n_best}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_soft_max(best_seq, t_alloc, T_start=200.0, T_end=5.0, maxiter=8000)
        try_update(a_lb, "lbgfs_main")

        # Second L-BFGS at lower temperature
        rem = deadline - time.monotonic()
        t_alloc2 = min(rem * 0.15, 200.0)
        if t_alloc2 > 20:
            print(f"  L-BFGS T: 5->1, n={len(best_seq)}, budget={t_alloc2:.0f}s")
            a_lb2 = lbgfs_soft_max(best_seq, t_alloc2, T_start=5.0, T_end=1.0, maxiter=5000)
            try_update(a_lb2, "lbgfs_lowT")

    # === Phase 2: Nesterov LP refinement (350s) ===
    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  NesterovLP n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_np, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver)
        try_update(a_np, "nesterov_lp")

    # === Phase 3: Mirror descent (200s) ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  MirrorDesc n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_md, _ = mirror_descent_optimize(best_seq, t_alloc, lr_init=0.05)
        try_update(a_md, "mirror_desc")

    # === Phase 4: Coordinate descent (200s) ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 15:
        print(f"  CoordDesc n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_cd, _ = coordinate_descent(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")

    # === Phase 5: Peak shaving (200s) ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 15:
        print(f"  PeakShave n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave_local(best_seq, t_alloc)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Second L-BFGS + LP cycle (150s) ===
    print(f"\n=== Phase 6: Refinement Cycle (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t1 = min(rem * 0.5, 80.0)
        if t1 > 15:
            a_r1 = lbgfs_soft_max(best_seq, t1, T_start=3.0, T_end=0.5, maxiter=4000)
            try_update(a_r1, "lbgfs_R2")

        rem = deadline - time.monotonic()
        t2 = min(rem * 0.5, 70.0)
        if t2 > 15:
            a_r2, _ = nesterov_lp_refine(best_seq, t2, solver=solver, n_max_iters=1000)
            try_update(a_r2, "lp_R2")

    # === Phase 7: Try alternative lengths (100s) ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = [int(n_best * 1.25), int(n_best * 0.8), int(n_best * 1.5)]
    alt_lengths = [x for x in alt_lengths if 256 <= x <= 4096 and x != n_best]
    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.4, 80.0)
        if t_alloc < 15:
            break
        print(f"  Alt length n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt = lbgfs_soft_max(init_alt, t_alloc, T_start=50.0, T_end=2.0, maxiter=3000)
        try_update(a_alt, f"alt_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_lp = min(rem * 0.4, 40.0)
            if t_lp > 10:
                a_al, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=500)
                try_update(a_al, f"lp_alt_n{alt_n}")

    # === Phase 8: Final micro-polish (50s) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_final = min(rem * 0.5, 30.0)
        a_final = lbgfs_soft_max(best_seq, t_final, T_start=2.0, T_end=0.3, maxiter=2000)
        try_update(a_final, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.5, 20.0)
        a_ps_final, _ = peak_shave_local(best_seq, t_ps, n_steps=2000)
        try_update(a_ps_final, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
