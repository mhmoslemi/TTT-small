def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def log_softmax(z):
    z = z - z.max()
    return z - np.log(np.exp(z).sum())


def softmax(z):
    return np.exp(log_softmax(z))

    Compute the log-sum-exp smooth objective and its gradient.

    n = len(a)
    c = fast_convolve(a)

    scaled = c / tau - c_max / tau
    w = np.exp(scaled)

    w = w / w_sum

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
