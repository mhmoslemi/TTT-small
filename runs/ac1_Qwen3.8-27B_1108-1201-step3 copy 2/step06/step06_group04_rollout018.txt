We want to minimize the peak of the autocorrelation, which is equivalent to making |A(ω)|^2 as flat as possible (equal power spectrum). This is the "minimum peak sidelobe" problem.

However, Barker sequences are typically ±1 valued, not non-negative. For non-negative sequences, the problem is different.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved sparse LP-Frank-Wolfe
with adaptive constraint selection, structured perturbation, multi-length
exploration, and spectrally-guided coordinate descent. Key innovations: sparse
LP with top-K active constraints for speed, Nesterov-accelerated mirror descent,
and convolution-structure-aware restarts to escape local minima.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum=1 normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def sparse_lp_direction(f, M, n, top_k=300):
    """LP: max sum(g) s.t. (f*g)[k] <= M for top-K active constraints, g >= 0."""
    conv_f = fast_convolve(f)
    top_k_actual = min(top_k, len(conv_f))
    top_indices = np.argpartition(conv_f, -top_k_actual)[-top_k_actual:]
    
    A_ub = np.zeros((top_k_actual, n))
    for i, k in enumerate(top_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        A_ub[i, idx] = f[k - idx]
    
    c = -np.ones(n)
    b_ub = np.full(top_k_actual, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_obj_bound': 1e9})
    if res.success:
        return np.maximum(res.x, 0)
    return None


def dense_lp_direction(f, M, n):
    """LP with all constraints."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0)
    return None


def line_search(f, g, M, n, max_evals=20):
    """Line search for alpha in [0,1] minimizing max conv after mixing."""
    best_alpha = 0.0
    best_val = M
    evals = 0
    
    # Coarse scan
    alphas_coarse = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    for a in alphas_coarse:
        ft = f + a * (g - f)
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        v = fast_convolve(ft).max()
        evals += 1
        if v < best_val:
            best_val = v
            best_alpha = a
    
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    span = 0.12
    lo = max(0.0, best_alpha - span)
    hi = min(1.0, best_alpha + span)
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    
    def eval_alpha(a):
        nonlocal evals
        ft = f + a * (g - f)
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return M
        ft = ft / s
        v = fast_convolve(ft).max()
        evals += 1
        return v
    
    c_val = eval_alpha(c_pt)
    d_val = eval_alpha(d_pt)
    
    for _ in range(8):
        if evals >= max_evals:
            break
        if c_val < d_val:
            hi = d_pt
            d_pt = c_pt
            d_val = c_val
            c_pt = hi - gr * (hi - lo)
            c_val = eval_alpha(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            c_val = d_val
            d_pt = lo + gr * (hi - lo)
            d_val = eval_alpha(d_pt)
        if abs(hi - lo) < 1e-7:
            break
        mid = (lo + hi) / 2.0
        mid_val = eval_alpha(mid)
        if mid_val < best_val:
            best_val = mid_val
            best_alpha = mid
    
    return best_alpha, best_val

def _perturb_restart(f, best_f, n, seed_offset):
    """Structured perturbation to escape local minima."""
    rng = np.random.default_rng(seed_offset * 17 + 3)
    
    # Strategy: reduce peak contributors and boost valleys
    conv = fast_convolve(best_f)
    peak_idx = int(np.argmax(conv))
    js = max(0, peak_idx - n + 1)
    je = min(n - 1, peak_idx)
    
    f_new = best_f.copy()
    f_new[js:je+1] *= rng.uniform(0.7, 0.9)
    
    # Boost non-peak areas
    non_peak = np.ones(n, dtype=bool)
    non_peak[js:je+1] = False
    if non_peak.sum() > 0:
        f_new[non_peak] *= rng.uniform(1.0, 1.15)
    
    f_new = np.maximum(f_new, 0)
    f_new = f_new / f_new.sum()
    return f_new, fast_convolve(f_new).max()


def nesterov_mirror_descent(f0, time_limit, verbose=False):
    """Nesterov-accelerated multiplicative weights (mirror descent)."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)
    
    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.003
    momentum = 0.7
    v = np.zeros(n)  # velocity for Nesterov
    f_mom = f.copy()  # momentum variable
    no_improve = 0
    temperature = 0.003
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        f_look = f_mom + momentum * (f_mom - f) if it > 1 else f_mom
        f_look = np.maximum(f_look, 1e-10)
        f_look = f_look / f_look.sum()
        
        conv = fast_convolve(f_look)
        M = conv.max()
        
        # Subgradient at the peak
        k_peak = int(np.argmax(conv))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * f_look[k_peak - idx]
        
        # Also add contributions from near-peaks for smoothness
        n_peaks = min(8, len(conv))
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        pv = conv[top_peaks]
        weights = pv / (pv.max() + 1e-15)
        
        grad_full = np.zeros(n)
        for i, kk in enumerate(top_peaks):
            if weights[i] < 0.1:
                continue
            kk = int(kk)
            js2 = max(0, kk - n + 1)
            je2 = min(n - 1, kk)
            idx2 = np.arange(js2, je2 + 1)
            grad_full[idx2] += weights[i] * f_look[kk - idx2]
        
        # Multiplicative update with momentum
        v = momentum * v + (1 - momentum) * grad_full
        f = f * np.exp(-lr * v)
        f = np.maximum(f, 1e-10)
        f = f / f.sum()
        f_mom = f
        
        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9998
        temperature *= 1.005
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 29 + 5)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-10)
            f = f / f.sum()
            f_mom = f.copy()
            v = np.zeros(n)
            lr = max(0.002, lr * 2)
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_f, best_obj


def coord_descent_peaks(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()

        M = conv.max()
        improved = False
        
        # Focus on top peaks
        n_peaks = min(10, len(conv))
        peak_indices = np.argsort(conv)[-n_peaks:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue
            
            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(5, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                
                for frac in [0.4, 0.55, 0.7, 0.82, 1.3, 1.6]:
                    new_val = old_val * frac
                    if new_val < 0 or new_val > 0.8:

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute to least-contributing positions
                        contrib = f[idx] * f[k - idx]
                        least = np.argsort(contrib)[:len(idx)//2]
                        target_idx = idx[least]
                        tw = f[target_idx]
                        tw_sum = tw.sum()
                        if tw_sum > 1e-15:
                            trial[target_idx] += d * tw / tw_sum
                    
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Aggressive peak reduction
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = f.copy()
                trial[js:je+1] *= 0.9
                d = f.sum() - trial.sum()
                non_c = np.ones(n, dtype=bool)
                non_c[js:je+1] = False
                if non_c.sum() > 0:
                    nc_w = f[non_c]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_c] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_exponential(n, rate=2.0):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * x / n)
    # Make it "double-sided" for better autocorrelation
    b = np.exp(-rate * (n - 1 - x) / n)
    a = 0.5 * (a + b)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bell_shape(n, width=0.6, asym=0.0):
    x = np.arange(n) - n / 2.0 + asym * n
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 4
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_arcosh(n, spread=1.5):
    """Arcoshine-based shape."""
    x = np.linspace(-np.pi/2, np.pi/2, n)
    a = np.cosh(spread * x) ** (-1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate and track best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: sparse LP-FW, Nesterov mirror descent,
    peak coordinate descent, multi-length exploration, and structured restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")
    
    n_parent = len(parent)
    
    # === Phase 1: Multi-init LP-FW (parallel exploration) ===
    print("\n=== Phase 1: Multi-Init Optimization ===")
    
    init_candidates = [parent.copy()]
    for n_try in [700, 768, 850, 900]:
        if n_try == n_parent:
            init_candidates.append(parent.copy())
        else:
            init_candidates.append(interpolate_sequence(parent, n_try))
        init_candidates.append(make_gaussian(n_try, 0.2))
        init_candidates.append(make_gaussian(n_try, 0.3))
        init_candidates.append(make_exponential(n_try, 2.0))
        init_candidates.append(make_bell_shape(n_try, 0.6))
        init_candidates.append(make_arcosh(n_try, 1.2))
    
    # Score all inits and pick top 3
    init_scored = []
    for ic in init_candidates:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = obj_normalized(a_norm)
        init_scored.append((obj, a_norm))
        best_seq, best_score = try_sequence(a_norm, best_seq, best_score,
                                            f"init_n{len(a_norm)}")
    
    init_scored.sort(key=lambda x: x[0])
    top_inits = [x[1] for x in init_scored[:4]]
    print(f"  Top init obj: {init_scored[0][0]:.6f}")
    
    # Run LP-FW on top inits with time splitting
    n_inits = len(top_inits)
    t_per_init = min(120, (deadline - time.monotonic() - 600) / max(n_inits, 1))
    t_per_init = max(30, t_per_init)
    
    for i, init in enumerate(top_inits):
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        t_alloc = min(t_per_init, rem)
        print(f"  LP-FW init {i} (n={len(init)}): t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(init, time_limit=t_alloc, verbose=(i == 0))
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"lpfw_init{i}")
    
    # === Phase 2: Deep LP-FW on best ===
    if best_seq is not None:
        print(f"\n=== Phase 2: Deep LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_alloc = min(rem, 200)
            print(f"  Deep LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "deep_lpfw")
    
    # === Phase 3: Nesterov Mirror Descent ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Nesterov MD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"  MD: t={t_alloc:.0f}s")
            a_opt, obj = nesterov_mirror_descent(best_seq, time_limit=t_alloc)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "md")
    
    # === Phase 4: Peak Coordinate Descent ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Peak CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 200
        if rem > 15:
            t_alloc = min(rem, 80)
            print(f"  Peak CD: t={t_alloc:.0f}s")
            a_opt, obj = coord_descent_peaks(best_seq, time_limit=t_alloc)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "peak_cd")
    
    # === Phase 5: Multi-length exploration ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Multi-length ===")
        n_best = len(best_seq)
        target_lengths = [n_best - 32, n_best + 32, n_best - 64, n_best + 64,
                         n_best - 96, n_best + 96]
        target_lengths = [t for t in target_lengths if 400 <= t <= 1200]
        
        for tn in target_lengths[:4]:
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            if tn == n_best:
                continue
            f_interp = interpolate_sequence(best_seq, tn)
            best_seq, best_score = try_sequence(f_interp, best_seq, best_score,
                                                 f"interp_n{tn}")
            t_alloc = min(rem * 0.35, 60)
            if t_alloc < 20:
                continue
            a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"lpfw_n{tn}")
    
    # === Phase 6: Structured restarts + re-optimize ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Structured Restarts ===")
        n_best = len(best_seq)
        for attempt in range(4):
            rem = deadline - time.monotonic() - 50
            if rem < 25:
                break
            t_alloc = min(rem / (5 - attempt), 70)
            
            if attempt == 0:
                # Small random perturbation
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
            elif attempt == 1:
                # Symmetrize
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif attempt == 2:
                # Blend with uniform
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                # Peak-targeted reduction
                conv = fast_convolve(best_seq)
                pk = int(np.argmax(conv))
                js = max(0, pk - n_best + 1)
                je = min(n_best - 1, pk)
                f_init = best_seq.copy()
                f_init[js:je+1] *= 0.95
                f_init = f_init / f_init.sum()
            
            print(f"  Restart {attempt}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"restart_{attempt}")
    
    # === Phase 7: Final polish ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Final Polish ===")
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t1 = min(rem * 0.4, 40)
            t2 = min(rem * 0.3, 30)
            t3 = min(rem * 0.3, 30)
            
            if t1 > 5:
                a_opt, obj = coord_descent_peaks(best_seq, time_limit=t1)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_cd")
            
            rem2 = deadline - time.monotonic() - 5
            if rem2 > 5 and best_seq is not None:
                t2a = min(t2, rem2 - 3)
                if t2a > 5:
                    a_opt, obj = nesterov_mirror_descent(best_seq, time_limit=t2a)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_md")
            
            rem3 = deadline - time.monotonic() - 2
            if rem3 > 3 and best_seq is not None:
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=max(rem3, 1))
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_lpfw")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
