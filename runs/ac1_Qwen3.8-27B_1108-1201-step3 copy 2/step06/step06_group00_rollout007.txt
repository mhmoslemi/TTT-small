```python
def lp_peak_reduction(f, M, solver='hihps'):
    """
    Solve LP: max sum(b) s.t. b >= 0, (f*b)(k) <= M for all k.
    Returns optimized b.
    """
    n = len(f)
    # Build constraint matrix A where A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Constraint: A @ b <= M (for all k = 0, ..., 2n-2)
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n - 1, k) + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: max c^T b s.t. A b <= M, b >= 0
    # In scipy form: min -c^T b s.t. A b <= M, -I b <= 0
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='hihps')
    if res.success:
        return res.x
    return f  # fallback
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Adaptive subgradient with backtracking line search and peak-weighted gradients
2. LP-based direction finding with concentration penalty for basin exploration
3. L-BFGS with soft-max smoothing for smooth convergence
4. Active-set peak equalization via projected gradient on active peaks
5. Coordinate descent micro-optimization
6. Multi-length exploration around the parent length

Key improvements:
- Backtracking line search for robust step size selection
- LP directions with anti-concentration constraints
- Active-set method that directly equalizes top peaks
- More aggressive multi-length search
- Nesterov momentum with adaptive restarts

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def subgradient_step(a, top_k=None, temp=0.001):
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    scaled = (b - b_max) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    if top_k is not None and top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        w_sel = w[top_idx].copy()
        w_sel /= w_sel.sum()
        w_full = np.zeros(len(b))
        w_full[top_idx] = w_sel
        w = w_full
    g = np.zeros(n)
    active_idx = np.where(w > 1e-12)[0]
    for k in active_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * a[k - idx]
    return g, b, b_max


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.0005, temp_growth=1.02,
                         n_top=None, seed=42, backtracking=True):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        g, b, b_max = subgradient_step(a, top_k=n_top, temp=cur_temp)

        if backtracking:
            alpha = cur_lr
            best_alpha = alpha
            best_trial_obj = best_obj
            for _ in range(10):
                v = momentum * v - alpha * g
                a_trial = project_simplex(a + v)
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_obj:
                    best_a = a_trial.copy()
                    best_obj = obj_trial
                    best_alpha = alpha
                    no_improve = 0
                elif obj_trial < best_trial_obj:
                    best_trial_obj = obj_trial
                    best_alpha = alpha
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            a = project_simplex(a + momentum * v - best_alpha * g)
            v = momentum * v - best_alpha * g
        else:
            v = momentum * v - cur_lr * g
            a_new = project_simplex(a + v)
            obj_new = fast_convolve(a_new).max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new

        it_count += 1
        if no_improve > 800:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.8, lr_init * 0.1)
            no_improve = 0
        elif no_improve > 400:
            cur_lr *= 0.95
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def lp_direction(f, M, max_elem_ratio=3.0):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    f_max = f.max()
    upper_bound = max_elem_ratio * f_max
    bounds = [(0.0, upper_bound)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None
