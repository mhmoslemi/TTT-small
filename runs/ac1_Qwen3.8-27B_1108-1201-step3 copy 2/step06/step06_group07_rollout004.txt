"""
LP-based optimization with cutting plane for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP direction finding (state-of-the-art approach) with mirror descent fine-tuning,
multi-start diversification, adaptive step sizes, and coordinate descent polish.
Key innovation: efficient cutting-plane LP that only uses active constraints, enabling
many more iterations than naive approaches.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def solve_lp_active(f, M, n, active_keys):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for k in active_keys."""
    keys = sorted(active_keys)
    m = len(keys)
    if m == 0:
        return np.ones(n) / n
    A = np.zeros((m, n))
    b = np.full(m, M)
    for i, k in enumerate(keys):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        i_start = k - je
        i_end = k - js
        A[i, i_start:i_end + 1] = f[js:je + 1]
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        return None
    return result.x


def solve_lp_cutting(f, M, n, max_active=120, max_cp=15):
    """LP with cutting plane: start with top peaks, add violated constraints."""
    ff = fast_convolve(f)
    K = min(30, len(ff))
    top_peaks = np.argpartition(ff, -K)[-K:]
    active_set = set(int(k) for k in top_peaks)

    g = None
    for _ in range(max_cp):
        g = solve_lp_active(f, M, n, active_set)
        if g is None:
            return None
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        fg = np.convolve(f, g, mode='full')
        tol = M * 1e-9 + 1e-12
        violated = np.where(fg > M + tol)[0]
        if len(violated) == 0:
            return g
        if len(active_set) < max_active:
            for k in violated[:max_active - len(active_set)]:
                active_set.add(int(k))
        else:
            worst = int(violated[np.argmax(fg[violated])])
            active_set.add(worst)
    return g


def lp_optimize(a0, time_limit, verbose=False, label=""):
    """LP-based optimization following the state-of-the-art approach."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_maxb = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    t = 0.1
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_lp_cutting(a, M, n, max_active=100, max_cp=12)
        if g is None:
            no_improve += 10
            if no_improve > 30:
                break
            continue
        g_sum = g.sum()
        if g_sum < 1e-10:
            no_improve += 10
            continue
        g = g / g_sum

        best_trial = None
        best_trial_obj = best_maxb
        best_t = t

        for trial_t in [t, t * 0.5, t * 0.25, t * 1.5, t * 2, t * 3]:
            if trial_t <= 0.001 or trial_t > 0.95:
                continue
            a_trial = (1.0 - trial_t) * a + trial_t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_maxb = fast_convolve(a_trial).max()
            if trial_maxb < best_trial_obj - 1e-14:
                best_trial_obj = trial_maxb
                best_trial = a_trial.copy()
                best_t = trial_t

        if best_trial is not None:
            a = best_trial
            t = best_t
            if best_trial_obj < best_maxb - 1e-14:
                best_maxb = best_trial_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    {label} LP iter {it}: max_b={best_maxb:.10f}, obj={2*n*best_maxb:.8f}")
        else:
            no_improve += 1

        if no_improve > 5:
            t = max(t * 0.7, 0.02)
        if no_improve > 25:
            rng = np.random.default_rng(it * 104729 + 7)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0.0)
            a = a / a.sum()
            t = 0.1
            no_improve = 0

        if it % 10 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 0.0)
            ss = a_sym.sum()
            if ss > 1e-10:
                a_sym = a_sym / ss
                sym_maxb = fast_convolve(a_sym).max()
                if sym_maxb < best_maxb:
                    best_maxb = sym_maxb
                    best_a = a_sym.copy()
                    a = a_sym.copy()

        if t < 0.015:
            break

    return best_a, best_maxb


def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01,
                   lr_decay=0.9999, verbose=False, label=""):
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))
        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        g = compute_subgradient(a, peak_indices, w)
        best_lr = lr * 0.1
        best_trial_obj = 2.0 * n * M
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-10:
                continue
            log_update = -trial_lr * g
            log_update = np.clip(log_update, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_conv = fast_convolve(a_trial)
            trial_obj = 2.0 * n * trial_conv.max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < 2.0 * n * M * 0.999:
                    break
        log_update = -best_lr * g
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        conv = fast_convolve(a)
        new_obj = 2.0 * n * conv.max()
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * lr_decay, 1e-8)
        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.001 + 0.002 * min(no_improve / 5000, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0
        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = 2.0 * n * fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
        if lr < 1e-9:
            break
    return best_a, best_obj / (2.0 * n)


def coordinate_descent(a0, time_limit, verbose=False, label=""):

    a = a / a.sum()
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

            if time.monotonic() - t0 > time_limit:

            old_val = a[j]

                if denom > 1e-10:
