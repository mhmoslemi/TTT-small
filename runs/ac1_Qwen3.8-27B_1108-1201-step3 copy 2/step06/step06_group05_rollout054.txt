Let me think about the away-step FW more carefully.

Standard FW for minimizing F(x) over a polytope P:
1. x_0 in P
2. At each step:
   a. Find the subgradient g of F at x_t
   b. Find s_t = argmin_{s in P} <g, s> (the "linear minimization oracle")
   c. Find gamma_t = argmin_{gamma in [0,1]} F((1-gamma)*x_t + gamma*s_t)
   d. x_{t+1} = (1-gamma_t)*x_t + gamma_t*s_t

Away-step FW (Fisher, 2013):
Maintain a working set Δ = {v_1, ..., v_m} of vertices.
At each step:
1. Find new vertex s (linear minimization oracle)
2. Check if s is already in the working set
3. If not, add s to the working set
4. Solve: min_{γ, β} F(β * x_t + Σ γ_i * v_i - Σ δ_i * v_i)
   subject to γ_i >= 0, δ_i >= 0, Σ γ_i + Σ δ_i <= 1 (or similar constraint)
   This is a small QP (or we can linearize it)
5. Remove vertices with small coefficients from the working set

The key benefit of the away-step is that it can "move away" from bad vertices that are slowing down convergence.

For our problem, the "vertices" are the LP solutions g (which are not basis vectors but vertices of a different polytope). So the away-step is a bit different from the standard case.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-Frank-Wolfe with away-step for high-precision optimization
2. Newton-based peak equalization (active-set method) for final convergence
3. Multi-length exploration and diverse restarts for robustness

Key innovation: Novel Newton step that solves the KKT system for active
peak equalization, exploiting the Toeplitz structure of the Gram matrix
A_S @ A_S^T for efficiency. Combined with LP-FW away-step for fast
convergence to the neighborhood of the optimum.

import numpy as np
from scipy.optimize import linprog
from scipy.linalg import solve as slsolve
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_rows(f, indices):
    """
    Build rows of the convolution matrix A at specified row indices.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Returns a (len(indices) x n) matrix.
    """
    n = len(f)
    s = len(indices)
    result = np.zeros((s, n))
    for i, k in enumerate(indices):
        j_max = min(n - 1, k)
        j_min = max(0, k - n + 1)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            result[i, js] = f[k - js]
    return result


def compute_GRAM(f, S):
    """
    Compute A_S @ A_S^T where A_S is the submatrix of the convolution matrix
    at rows indexed by S. Uses direct computation (fast for |S| < 300).

    n = len(f)

                val = np.dot(f[ki - idx], f[kl - idx])
            else:

            result[i, l] = val
            result[l, i] = val

        M = c.max()

        else:

def newton_peak_equalize(f0, time_limit, verbose=False):
    """
    Newton-based peak equalization using active-set KKT conditions.
    
    At the optimum (interior), for the active set S of peaks:

    - sum(f) = 1
    - The gradient of (f*f)[k] (weighted by KKT multipliers) is constant
    
    We linearize and solve the resulting (|S|+1) x (|S|+1) system.
    """

    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    # Start with a small active set, grow it
    k_active = 2  # Number of peaks to equalize

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem_time = time_limit - (time.monotonic() - t0)
        if rem_time < 1.0:
            break

        c = fast_convolve(f)
        M = c.max()

        # Identify active set: top k_active peaks
        top_k = min(k_active, len(c))
        peak_idx = np.argsort(c)[::-1][:top_k]
        S = sorted(peak_idx.tolist())
        s_count = len(S)

        # Check if all active peaks are at (nearly) the same level
        peak_vals = np.array([c[k] for k in S])
        peak_spread = peak_vals.max() - peak_vals.min()

        if peak_spread < 1e-14 and s_count >= 2:
            # Try growing the active set
            if s_count < 40:
                k_active = min(s_count + 2, 40)
                continue
            elif s_count >= 40:
                # Also check the 41st peak
                if len(c) > s_count and c[np.argsort(c)[::-1][s_count]] > M - 1e-10:
                    k_active = min(s_count + 4, 80)
                    continue
                else:
                    no_improve += 1
                    if no_improve > 20:
                        break
                    continue
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Build the KKT system
        # A_S: s_count x n
        A_S = build_conv_matrix_rows(f, S)
        # b_k = M - c[k] for k in S (we want to "raise" sub-max peaks to M)
        # Or equivalently, we want (f*d)[k] = (M - c[k])/2 for all k in S
        b_vec = (M - np.array([c[k] for k in S])) / 2.0

        if np.max(np.abs(b_vec)) < 1e-16:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Compute Gram matrix G = A_S @ A_S^T
        G = compute_GRAM(f, S)
        # Compute A_S @ 1
        A1 = compute_AS_ones(f, S)

        # Build (s_count+1) x (s_count+1) system:
        # [G   A1 ] [lam]   [-b_vec]
        # [A1^T n ] [mu ] = [  0   ]
        sys_mat = np.zeros((s_count + 1, s_count + 1))
        sys_mat[:s_count, :s_count] = G
        sys_mat[:s_count, s_count] = A1
        sys_mat[s_count, :s_count] = A1
        sys_mat[s_count, s_count] = float(n)

        sys_vec = np.zeros(s_count + 1)
        sys_vec[:s_count] = -b_vec

        # Add small regularization for numerical stability
        sys_mat += np.eye(s_count + 1) * 1e-12

        try:
            sol = slsolve(sys_mat, sys_vec)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(sys_mat, sys_vec, rcond=None)[0]

        lam = sol[:s_count]
        mu = sol[s_count]

        # Compute direction: d = -A_S^T @ lam - mu * 1
        d = -A_S.T @ lam - mu * np.ones(n)

        # Ensure sum(d) = 0 (should be approximately so)
        d -= np.mean(d)

        # Check that d doesn't make f negative
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0 / (n * d_norm + 1)  # Initial step size
        alpha = min(alpha, 0.1)

        for _ in range(25):
            f_trial = f + alpha * d
            if np.min(f_trial) < -1e-12:
                alpha *= 0.5
                continue
            f_trial = np.maximum(f_trial, 0.0)
            f_trial = f_trial / f_trial.sum()
            c_trial = fast_convolve(f_trial)
            val = c_trial.max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.7

        if best_alpha > 1e-12:
            f_new = f + best_alpha * d
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            c_new = fast_convolve(f_new)
            M_new = c_new.max()

            if M_new < best_obj - 1e-15:
                best_obj = M_new
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
                # Slightly grow active set on success
                if it % 10 == 0 and k_active < 60:
                    k_active = min(k_active + 1, 60)
                if verbose and it % 10 == 0:
                    print(f"    Newton iter {it}: obj={best_obj:.12f}, |S|={s_count}, alpha={best_alpha:.4e}")
            else:
                no_improve += 1
                # Shrink active set if Newton step fails
                if no_improve % 5 == 0 and k_active > 2:
                    k_active = max(2, k_active - 1)
        else:
            no_improve += 1
            # Try smaller step or different active set
            if no_improve % 10 == 0:
                k_active = max(2, k_active - 2)

        if no_improve > 100:
            # Perturb and restart
            rng = np.random.default_rng(it * 41 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0003)
            k_active = 2
            no_improve = 0

    return best_f, best_obj


def smoothed_gd(f0, time_limit, verbose=False):
    """
    Smoothed gradient descent with log-sum-exp for the initial phase.
    Quick convergence to the right neighborhood.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    momentum = np.zeros(n)
    mu = 0.85

    beta_schedule = [4, 8, 16, 32, 64, 128, 256, 512]

    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.9:
            break

        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.1, (time_limit - (time.monotonic() - t0)) * 0.2)
        no_improve = 0
        it = 0

        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.9:
            it += 1
            c = fast_convolve(f)
            cur_max = c.max()

            # LSE gradient
            scaled = beta * c
            scaled -= scaled.max()
            exp_s = np.exp(scaled)
            w = exp_s / exp_s.sum()

            # Gradient: g_j = 2 * sum_i f_i * w_{j+i}
            w_rev = w[::-1].copy()
            conv_len = n + len(w_rev) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            lr = 0.5 / (g_norm + 1e-10)
            lr = min(lr, 1.0 / np.sqrt(n))

            momentum = mu * momentum - lr * g
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)

            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()

            if obj_trial <= cur_max + 1e-10:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                momentum *= 0.5
                no_improve += 1

            if no_improve > 60:
                rng = np.random.default_rng(it * 13 + int(beta) * 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                momentum = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    """Raised cosine (Hann window)."""
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_refine(f0, time_limit, verbose=False):
    """
    Coordinate descent: try adjusting pairs of nearby coordinates.
    More efficient than single-coordinate for this problem.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)

        # Try pairwise adjustments
        for j in range(0, n, 2):
            if time.monotonic() - t0 > time_limit:
                break
            if j + 1 >= n:
                break

            f1, f2 = f[j], f[j + 1]
            if f1 < 1e-14 and f2 < 1e-14:
                continue

            # Try shifting mass between j and j+1
            for frac in [0.3, 0.4, 0.45, 0.55, 0.6, 0.7]:
                total = f1 + f2
                new_f1 = frac * total
                new_f2 = (1 - frac) * total
                if abs(new_f1 - f1) < 1e-15:
                    continue

                f_trial = f.copy()
                f_trial[j] = new_f1
                f_trial[j + 1] = new_f2
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    f = f_trial
                    best_f = f.copy()
                    improved = True
                    f1, f2 = new_f1, new_f2

        if not improved:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-FW, Newton peak equalization,
    and coordinate refinement to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_raised_cosine(768)
        try_seq(parent, "init_rc")

    n_parent = len(parent)
    print(f"Starting with n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Quick smoothed GD (10% time) ===
    print("\n=== Phase 1: Smoothed Gradient Descent ===")
    t1_start = time.monotonic()
    rem = deadline - time.monotonic()
    t_alloc1 = min(rem * 0.08, 120)
    print(f"  Allocation: {t_alloc1:.0f}s")

    a_opt, obj = smoothed_gd(parent, time_limit=t_alloc1, verbose=False)
    try_seq(a_opt, "phase1_gd")

    # Also try a different init
    gauss_init = make_gaussian(n_parent, 0.28)
    a_opt2, obj2 = smoothed_gd(gauss_init, time_limit=min(t_alloc1 * 0.5, 50))
    try_seq(a_opt2, "phase1_gd_gauss")

    print(f"  Phase 1 time: {time.monotonic()-t1_start:.1f}s, best={best_score:.8f}")

    # === Phase 2: LP-FW (45% time) ===
    print("\n=== Phase 2: LP-FW Optimization ===")
    t2_start = time.monotonic()
    base = best_seq if best_seq is not None else a_opt
    rem = deadline - time.monotonic()
    t_alloc2 = min(rem * 0.45, 800)
    print(f"  Allocation: {t_alloc2:.0f}s")

    # Run LP-FW from best
    a_opt, obj = lp_fw_optimize(base, time_limit=t_alloc2 * 0.7, verbose=True)
    try_seq(a_opt, "phase2_lpfw")

    # Run from a perturbed version
    if time.monotonic() < deadline - 60:
        rem2 = deadline - time.monotonic() - 60
        t_alloc2b = min(rem2 * 0.3, 200)
        if t_alloc2b > 30:
            pert = project_simplex(base + rng.standard_normal(n_parent) * 0.005)
            a_opt3, obj3 = lp_fw_optimize(pert, time_limit=t_alloc2b, verbose=False)
            try_seq(a_opt3, "phase2_lpfw_pert")

    print(f"  Phase 2 time: {time.monotonic()-t2_start:.1f}s, best={best_score:.8f}")

    # === Phase 3: Newton Peak Equalization (30% time) ===
    print("\n=== Phase 3: Newton Peak Equalization ===")
    t3_start = time.monotonic()
    base = best_seq if best_seq is not None else a_opt
    rem = deadline - time.monotonic()
    t_alloc3 = min(rem * 0.30, 400)
    print(f"  Allocation: {t_alloc3:.0f}s")

    # Run Newton from best
    a_opt, obj = newton_peak_equalize(base, time_limit=t_alloc3 * 0.75, verbose=True)
    try_seq(a_opt, "phase3_newton")

    # Another Newton run from slightly perturbed
    if time.monotonic() < deadline - 40:
        rem2 = deadline - time.monotonic() - 40
        t_alloc3b = min(rem2 * 0.25, 100)
        if t_alloc3b > 20:
            pert = project_simplex(base + rng.standard_normal(len(base)) * 0.001)
            a_opt4, obj4 = newton_peak_equalize(pert, time_limit=t_alloc3b, verbose=False)
            try_seq(a_opt4, "phase3_newton_pert")

    print(f"  Phase 3 time: {time.monotonic()-t3_start:.1f}s, best={best_score:.8f}")

    # === Phase 4: Multi-length exploration (10% time) ===
    print("\n=== Phase 4: Multi-Length Exploration ===")
    t4_start = time.monotonic()
    base = best_seq if best_seq is not None else a_opt
    rem = deadline - time.monotonic()
    t_alloc4 = min(rem * 0.10, 150)
    print(f"  Allocation: {t_alloc4:.0f}s")

    for target_n in [896, 1024]:
        if time.monotonic() > deadline - 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")

        # Quick Newton on interpolated
        rem_rem = deadline - time.monotonic() - 30
        t_sub = min(rem_rem * 0.4, 60)
        if t_sub > 15:
            a_i, obj_i = newton_peak_equalize(f_interp, time_limit=t_sub, verbose=False)
            try_seq(a_i, f"newton_n{target_n}")

    print(f"  Phase 4 time: {time.monotonic()-t4_start:.1f}s, best={best_score:.8f}")

    # === Phase 5: Final combined polish (remaining time) ===
    print("\n=== Phase 5: Final Polish ===")
    t5_start = time.monotonic()
    base = best_seq if best_seq is not None else a_opt
    n_base = len(base)
    rem = deadline - time.monotonic()
    print(f"  Remaining: {rem:.0f}s")

    # Alternating Newton + LP-FW
    while time.monotonic() < deadline - 10:
        rem_now = deadline - time.monotonic() - 10
        if rem_now < 15:
            break

        # Newton step
        t_newton = min(rem_now * 0.5, 40)
        a_new, _ = newton_peak_equalize(base, time_limit=t_newton, verbose=False)
        try_seq(a_new, "polish_newton")
        if best_seq is not None:
            base = best_seq.copy()
            n_base = len(base)

        # LP-FW step
        rem_now = deadline - time.monotonic() - 10
        if rem_now < 15:
            break
        t_lpfw = min(rem_now * 0.5, 40)
        a_lpfw, _ = lp_fw_optimize(base, time_limit=t_lpfw, verbose=False)
        try_seq(a_lpfw, "polish_lpfw")
        if best_seq is not None:
            base = best_seq.copy()
            n_base = len(base)

        # Coordinate refinement
        rem_now = deadline - time.monotonic() - 10
        if rem_now < 10:
            break
        t_coord = min(rem_now * 0.3, 20)
        if t_coord > 5:
            a_coord, _ = coordinate_refine(base, time_limit=t_coord, verbose=False)
            try_seq(a_coord, "polish_coord")
            if best_seq is not None:
                base = best_seq.copy()
                n_base = len(base)

    print(f"  Phase 5 time: {time.monotonic()-t5_start:.1f}s, best={best_score:.8f}")

    # === Final: one more Newton squeeze ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        rem_final = deadline - time.monotonic() - 3
        if rem_final > 10:
            a_final, _ = newton_peak_equalize(best_seq, time_limit=rem_final, verbose=True)
            try_seq(a_final, "final_newton")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
