```python
"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses a sequential LP approach: at each step, solve an LP to find the best
direction that reduces the peak of the auto-convolution, then take a step
in that direction. Combined with subgradient polishing and coordinate descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import coo_matrix
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A where A[k,i] = f[i] if i <= k <= i+n-1."""
    rows_parts = []
    cols_parts = []
    vals_parts = []
    for i in range(n):
        r = np.arange(i, i + n)
        c = np.full(n, i)
        v = np.full(n, f[i])
        rows_parts.append(r)
        cols_parts.append(c)
        vals_parts.append(v)
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    vals = np.concatenate(vals_parts)
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_direction(f, M, verbose=False):
    """Solve the LP to find the best direction g >= 0 with (f*g) <= M, maximizing sum(g)."""
    n = len(f)
    g = cp.Variable(n)
    
    # Build constraint matrix
    A = build_conv_matrix(f, n)
    M_vec = np.full(2*n-1, M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M_vec])
    prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-7)
    
    if g.value is not None:
        g_val = np.maximum(g.value, 0)
        if verbose:
            print(f"    LP status: {prob.status}, sum(g) = {g_val.sum():.6f}")
        return g_val
    if verbose:
        print(f"    LP status: {prob.status}")
    return None

def subgradient_step(f, lr, momentum_v, momentum, temp, n_top):
    """One subgradient step on the simplex."""
    n = len(f)
    b = fast_convolve(f)
    b_max = b.max()
    
    # Temperature-weighted peaks
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Focus on top peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        w_sel /= w_sel.sum()
        w_new = np.zeros(len(b))
        w_new[top_idx] = w_sel
        w = w_new
    
    # Compute subgradient
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * f[k - idx]
    
    # Momentum step
    momentum_v = momentum * momentum_v - lr * g
    f_new = project_simplex(f + momentum_v)
    
    return f_new, momentum_v, b_max

def coordinate_polish(f, time_limit, n_sweeps_max=100):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.75, 1.0, 1.33, 2.0])
    
    for sweep in range(n_sweeps_max):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                # Renormalize (divide others proportionally to keep sum=1)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j = new_val
            if best_j != f[j]:
                f[j] = best_j
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
        if not improved:
            break
    return f, best_obj
```

```python
"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key algorithm: At each iteration, solve a reduced linear program to find
the best non-negative direction that keeps the auto-convolution below the
current peak. Take a step in that direction with line search. This is
complemented by subgradient polish and coordinate descent.

The reduced LP targets only the top-K peaks, making it fast (K=50) while
still providing strong descent directions.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import coo_matrix
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_reduced(f, n, top_k=50, eps=1e-8):
    """
    Solve reduced LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns g or None on failure.
    """
    b = fast_convolve(f)
    M = b.max()
    
    if M < 1e-15:
        return None
    
    # Select top_k peaks
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build constraint matrix (dense, K x n)
    A = np.zeros((K, n))
    for row_i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi+1] = f[lo:hi+1]
    
    # Solve LP
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        return None
    
    if g.value is None:
        return None
    
    g_val = np.maximum(g.value, 0)
    # Verify solution quality
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def subgradient_step(f, lr, v, momentum, temp, n_top):
    """One projected subgradient step with momentum."""
    n = len(f)
    b = fast_convolve(f)
    
    # Temperature-weighted peaks
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Focus on top peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        w_sum_sel = w_sel.sum()
        if w_sum_sel > 1e-30:
            w_sel /= w_sum_sel
        else:
            w_sel = np.ones(len(top_idx)) / len(top_idx)
        w_new = np.zeros(len(b))
        w_new[top_idx] = w_sel
        w = w_new
    
    # Compute subgradient: g[i] = sum_k w[k] * f[k-i] * I[0<=k-i<n]
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * f[k - idx]
    
    # Momentum step
    v = momentum * v - lr * g
    f_new = project_simplex(f + v)
    
    return f_new, v


def coordinate_polish(f, time_limit):
    """Coordinate descent polish on individual elements."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.9, 1.1, 1.4, 2.0])
    
    improved_global = True
    sweep = 0
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        sweep += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.999:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                f /= s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved_global = True
    
    return f, best_obj
```

```python
"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key algorithm: At each iteration, solve a reduced linear program targeting
the top-K peaks of the auto-convolution to find the best non-negative
direction. Take a step with line search. Adaptive K starts small (focus
on worst peaks) and grows. Complemented by subgradient polish,
multi-length exploration, and coordinate descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve_reduced(f, n, top_k, eps=1e-8):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns (g, M) or (None, M) on failure.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build dense constraint matrix (K x n)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

    """

        it += 1
        # Adaptive top_k: grow every 50 iterations
        if it % 50 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)

        g, M = lp_solve_reduced(f, n, top_k)

        if g is None:
            # Try with more peaks
            g, M = lp_solve_reduced(f, n, min(2 * top_k, 2 * n - 1))

        if S_g <= 1.0 + 1e-5:

def subgradient_step(f, lr, v, momentum, temp, n_top):
    """One projected subgradient step with momentum."""
    n = len(f)
    b = fast_convolve(f)

    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum

    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:
            w_sel = np.ones(len(top_idx)) / len(top_idx)
        w_new = np.zeros(len(b))
        w_new[top_idx] = w_sel
        w = w_new

    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * f[k - idx]

    v = momentum * v - lr * g
    f_new = project_simplex(f + v)
    return f_new, v

        if rem < 40:
            break
        t3 = min(rem * 0.2, 150.0)
        if t3 < 20:
            break
        # Interpolate
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  LP n={target_n}, budget={t3:.0f}s")
        f3, obj3 = run_lp_phase(f_interp, time_limit=t3, top_k_init=40,
                                top_k_max=120, label=f"lp_n{target_n}")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [n{target_n}]: {score3:.6f}")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 150.0)
    if t4 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t4:.0f}s, n={n_b}")
        f4, obj4 = run_subgradient_phase(best_seq, time_limit=t4,
                                         lr_init=0.002,
                                         n_top=max(30, n_b // 10))
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [sg]: {score4:.6f}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 80.0)
    if t5 > 10:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = coordinate_polish(best_seq, time_limit=t5)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [cd]: {score5:.6f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.4, 100.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}, top_k=100")
        f6, obj6 = run_lp_phase(best_seq, time_limit=t6, top_k_init=100,
                                top_k_max=200, label="lp_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {score6:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
```
