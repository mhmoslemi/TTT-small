"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe with
subgradient polishing, adaptive M-relaxation, structured initialization,
and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_val(f):
    """Objective for sum-normalized sequence."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()

def subgradient_step(f, lr, n_top=5):
    """
    Take a projected subgradient step for min max_k (f*f)(k) s.t. f>=0, sum(f)=1.
    The subgradient of max_k (f*f)(k) at f is the average of gradients at active k's.
    Gradient of (f*f)(k) w.r.t. f[j] = f[k-j] + f[j-k] (for valid indices)
    """
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    
    # Find active k's (within tolerance of max)
    tol = 1e-6
    active_k = np.where(conv >= M - tol)[0]
    
    if len(active_k) == 0:
        active_k = np.array([np.argmax(conv)])
    
    # Compute subgradient (average over active k's)
    sg = np.zeros(n)
    for k in active_k:
        k = int(k)
        # (f*f)(k) = sum_j f[j]*f[k-j]
        # d/d f[j] of (f*f)(k) = f[k-j] + f[j-k] (when both are valid)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                sg[j] += f[kj]
            jk = j - k
            if 0 <= jk < n:
                sg[j] += f[jk]
    
    sg /= len(active_k)
    
    # Projected gradient step on simplex
    f_new = f - lr * sg
    f_new = project_simplex(f_new)
    return f_new

def lp_direction(f, M, n, A_ub):
    """Solve LP for Frank-Wolfe direction."""
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def build_A_ub(f, n):
    """Build constraint matrix: A_ub[k,j] = f contribution to conv at position k."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # conv[k] includes f[j] * f[k-j], so A_ub[k,j] = f[k-j] if 0<=k-j<n
        for k in range(2 * n - 1):
            kj = k - j
            if 0 <= kj < n:
                A_ub[k, j] = f[kj]
    return A_ub

def build_A_ub_fast(f, n):
    """Build constraint matrix more efficiently using the Toeplitz structure."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A_ub[k, j] = f[k-j] for max(0, j) <= k <= min(2n-2, j+n-1)
        k_start = max(0, j)
        k_end = min(2 * n - 2, j + n - 1)
        # A_ub[k_start:k_end+1, j] = f[0 : k_end - k_start + 1]
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub

def fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with adaptive M and subgradient fallback."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    A_ub = build_A_ub_fast(f, n)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factor = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP
        g = lp_direction(f, M * m_factor, n, A_ub)
        
        if g is None:
            m_factor *= 1.2
            if m_factor > 8.0:
                # Subgradient fallback
                lr = 0.001
                f = subgradient_step(f, lr)
                conv = fast_convolve(f)
                M = conv.max()
                A_ub = build_A_ub_fast(f, n)
                m_factor = 1.0
                co = 2.0 * n * M
                if co < best_obj:
                    best_obj = co
                    best_f = f.copy()
                    no_improve = 0
                continue
            continue
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            m_factor *= 1.1

        g = g / gs
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            # Stuck, try subgradient

                A_ub = build_A_ub_fast(f, n)

                no_improve = 0
            if no_improve > 30:
                # Random perturbation

                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)

                A_ub = build_A_ub_fast(f, n)
                m_factor = 1.0
                no_improve = 0
            continue
        
        if diff > 1e-5:
            no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_a_val = M
        for a in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + a * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            cv = fast_convolve(ft).max()
            if cv < best_a_val:
                best_a_val = cv
                best_alpha = a
        
        if best_alpha < 1e-8:
            no_improve += 1
            m_factor = min(m_factor * 1.05, 5.0)
            if no_improve > 50:
                break
            continue
        
        f = np.maximum(f + best_alpha * (g - f), 0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()
        m_factor = 1.0
        
        co = 2.0 * n * M
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Rebuild A_ub periodically

        if verbose and it % 50 == 0:
            print(f"    FW iter {it}: obj={best_obj:.10f}, M={M:.8f}")

def refine_smooth(f0, time_limit, verbose=False):
    """
    Smooth refinement using projected subgradient with decreasing step size.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.005
    decay = 0.9995
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f = subgradient_step(f, lr, n_top=3)
        
        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= decay
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            lr = max(0.002, lr)
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_f, best_obj

def make_interpolated(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, np.maximum(f, 0))
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

def try_candidate(a, best_seq, best_score, label=""):
    """Evaluate and track best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe with
subgradient polishing, spectral initialization, adaptive trimming, and
multi-scale structured exploration. Key innovations: efficient constraint
matrix handling, subgradient interspersing, end-trimming for n reduction,

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def build_A_ub(f, n):
    """Build the constraint matrix for the LP: A_ub[k,j] = f[k-j] if valid."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = max(0, j)
        k_end = min(2 * n - 2, j + n - 1)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def subgradient_direction(f, n, n_active=8):
    """
    Compute subgradient of max_k (f*f)(k) at f.
    Returns (subgradient_vector, active_peak_indices).
    """
    conv = fast_convolve(f)
    M = conv.max()
    # Find peaks within tolerance of max
    tol = max(1e-8, M * 1e-6)
    active_k = np.where(conv >= M - tol)[0]
    # Limit to top n_active
    if len(active_k) > n_active:
        order = np.argsort(conv[active_k])[::-1][:n_active]
        active_k = active_k[order]

    sg = np.zeros(n)
    for k in active_k:
        k = int(k)
        # d/d f[j] of sum_{j'} f[j']*f[k-j'] = f[k-j] + f[j-k] (when valid)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                sg[j] += f[kj]
            jk = j - k
            if 0 <= jk < n:
                sg[j] += f[jk]

    sg /= max(1, len(active_k))
    return sg, active_k

def _solve_lp(f, M, n, A_ub):
    """Solve the LP for FW direction."""
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def subgradient_polish(f0, time_limit, verbose=False):

    restarts for escaping local minima.

    f = f / f.sum()
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.004
    decay = 0.9996
    no_improve = 0
    it = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg, active_k = subgradient_direction(f, n, n_active=6)
        f = project_simplex(f - lr * sg)

        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if co < best_obj - 1e-10:
                best_lr = lr
        else:
            no_improve += 1

        lr *= decay

        if no_improve > 50:
            # Adaptable restart
            rng = np.random.default_rng(it * 23 + 5)
            scale = 0.002 * (1.0 + 0.1 * (no_improve // 50))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            lr = max(best_lr * 0.7, 0.001)
            no_improve = 0

        if lr < 5e-7:
            break

        if verbose and it % 100 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def make_interpolated(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, np.maximum(f, 0))
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def trim_sequence(f, threshold=1e-6):
    """Trim near-zero elements from the ends of a sequence."""
    f = np.array(f, dtype=np.float64)
    n = len(f)
    start = 0
    end = n
    for i in range(n):
        if f[i] > threshold:
            start = i
            break
    for i in range(n - 1, -1, -1):
        if f[i] > threshold:
            end = i + 1
            break
    if end - start < 4:
        return f
    trimmed = f[start:end]
    s = trimmed.sum()
    if s > 1e-15:
        trimmed = trimmed / s
    return trimmed


def spectral_init(n, seed=0):
    """
    Initialize using flat-spectrum idea: start with random phase,
    then iterate to flatten the power spectrum while maintaining non-negativity.
    """
    rng = np.random.default_rng(seed)
    # Start with random real sequence
    a = rng.standard_normal(n)
    a = np.abs(a)
    a = a / a.sum()

    # Iterate: flatten spectrum
    for _ in range(20):
        F = np.fft.rfft(a, 2 * n - 1)
        P = np.abs(F) ** 2
        # Target: make P more uniform
        P_target = np.ones_like(P)
        P_target[0] = P[0]  # Keep DC
        # Blend
        P_new = 0.7 * P + 0.3 * P_target * (P.sum() / max(P_target.sum(), 1e-15))
        # Convert back to F (preserve phase)
        phase = F / (np.abs(F) + 1e-15)
        F_new = np.sqrt(P_new) * phase
        a_new = np.fft.irfft(F_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s

    return a


def make_sparse_structured(n, seed=42):
    """Create a structured sparse initialization inspired by Sidon-like patterns."""
    rng = np.random.default_rng(seed)
    # Choose a subset of positions (roughly n/3 positions)
    n_active = max(4, n // 3)
    positions = np.sort(rng.choice(n, size=n_active, replace=False))
    a = np.zeros(n)
    # Assign weights with some structure
    weights = rng.dirichlet(np.ones(n_active) * 2.0)
    a[positions] = weights
    return a


def make_bell(n, width=0.6):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.22):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def try_candidate(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence and track best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: LP-Frank-Wolfe with subgradient interspersing,
    spectral initialization, adaptive trimming, and multi-scale exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_score = p_score
            best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.22)
        best_seq, best_score = try_candidate(parent, best_seq, best_score, "gauss_init")

    n_parent = len(parent)

    # === Phase 0: Quick initialization survey ===
    print("=== Phase 0: Initialization Survey ===")
    init_pool = [parent.copy()]
    for n_try in [n_parent, n_parent - 64, n_parent + 64]:
        if n_try < 64:
            continue
        init_pool.append(make_interpolated(parent, n_try))
        init_pool.append(make_gaussian(n_try, 0.18))
        init_pool.append(make_gaussian(n_try, 0.28))
        init_pool.append(make_bell(n_try, 0.5))
        init_pool.append(make_bell(n_try, 0.7))
        init_pool.append(spectral_init(n_try, seed=n_try % 100))
        init_pool.append(make_sparse_structured(n_try, seed=n_try % 200))

    best_init = None
    best_init_obj = float('inf')
    for ic in init_pool:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = obj_normalized(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_candidate(a_norm, best_seq, best_score,
                                             f"init_n{len(a_norm)}")
    print(f"  Best init: obj_norm={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 1: LP-FW on best init ===
    print(f"\n=== Phase 1: LP-FW (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 1100
    t1 = max(40, min(rem, 300))
    print(f"  Allocated: {t1:.0f}s")
    a_opt, obj = fw_optimize(best_init, time_limit=t1, verbose=True)
    best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "fw1")

    # === Phase 2: Subgradient polish ===
    if best_seq is not None:
        print(f"\n=== Phase 2: Subgradient Polish ===")
        rem = deadline - time.monotonic() - 800
        t2 = max(20, min(rem, 120))
        print(f"  Allocated: {t2:.0f}s")
        a_opt, obj = subgradient_polish(best_seq, time_limit=t2, verbose=True)
        best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "sg1")

    # === Phase 3: Try trimming ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Trimming ===")
        for thresh in [1e-5, 1e-4, 5e-4, 1e-3]:
            trimmed = trim_sequence(best_seq, threshold=thresh)
            if len(trimmed) < len(best_seq):
                best_seq, best_score = try_candidate(trimmed, best_seq, best_score,
                                                     f"trim_{thresh}")

    # === Phase 4: Multi-length LP-FW ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    lengths_to_try = []
    n_base = len(base)
    for delta in [-96, -48, 48, 96, 128]:
        ln = n_base + delta
        if 64 <= ln <= 2000:
            lengths_to_try.append(ln)

    for target_n in lengths_to_try:
        rem = deadline - time.monotonic() - 400
        if rem < 80:
            break
        f_interp = make_interpolated(base, target_n)
        best_seq, best_score = try_candidate(f_interp, best_seq, best_score,
                                             f"interp_{target_n}")
        t_alloc = max(30, min(rem * 0.3, 120))
        if t_alloc < 20:
            continue
        print(f"  FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_candidate(a_opt, best_seq, best_score, f"fw_n{target_n}")

        # Quick SG polish
        rem2 = deadline - time.monotonic() - 200
        if rem2 > 15:
            a_opt2, _ = subgradient_polish(a_opt, time_limit=min(rem2 * 0.2, 25))
            best_seq, best_score = try_candidate(a_opt2, best_seq, best_score,
                                                 f"sg_n{target_n}")

    # === Phase 5: Deep FW on best ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Deep FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 300
        t5 = max(50, min(rem, 250))
        print(f"  Allocated: {t5:.0f}s")
        a_opt, obj = fw_optimize(best_seq, time_limit=t5, verbose=True)
        best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "fw_deep")

    # === Phase 6: Perturbation + re-optimize ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Perturbation Variants ===")
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 120
            if rem < 40:
                break
            t_alloc = max(25, min(rem / (6 - p), 100))

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif p == 2:
                f_init = 0.5 * (best_seq + best_seq[::-1])
                f_init = f_init / f_init.sum()
            elif p == 3:
                # Sparse perturbation: zero out random 20%
                mask = rng.random(n_best) < 0.8
                f_init = best_seq.copy()
                f_init[~mask] = 0
                s = f_init.sum()
                if s > 1e-15:
                    f_init /= s
                else:
                    f_init = best_seq.copy()
            else:
                # Cross with spectral init
                spec = spectral_init(n_best, seed=p * 7 + 100)
                f_init = 0.6 * best_seq + 0.4 * spec
                f_init = f_init / f_init.sum()

            print(f"  Variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = fw_optimize(f_init, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, f"pert_{p}")

            # Quick SG
            rem2 = deadline - time.monotonic() - 60
            if rem2 > 10:
                a_opt2, _ = subgradient_polish(a_opt, time_limit=min(rem2 * 0.3, 20))
                best_seq, best_score = try_candidate(a_opt2, best_seq, best_score,
                                                     f"pert_{p}_sg")

    # === Phase 7: Final polish ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Final Polish ===")
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            # Try trimming again
            trimmed = trim_sequence(best_seq, threshold=1e-5)
            if len(trimmed) < len(best_seq):
                best_seq, best_score = try_candidate(trimmed, best_seq, best_score, "fin_trim")

            t7 = min(rem, 50)
            t7a = t7 * 0.5
            print(f"  Final SG: {t7a:.0f}s")
            a_opt, _ = subgradient_polish(best_seq, time_limit=t7a)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "fin_sg")

            rem2 = deadline - time.monotonic() - 5
            if rem2 > 8 and best_seq is not None:
                t7b = min(rem2, 25)
                print(f"  Final FW: {t7b:.0f}s")
                a_opt, _ = fw_optimize(best_seq, time_limit=t7b)
                best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "fin_fw")

    # === Phase 8: One more trim + verify ===
    if best_seq is not None:
        trimmed = trim_sequence(best_seq, threshold=1e-6)
        if len(trimmed) < len(best_seq):
            best_seq, best_score = try_candidate(trimmed, best_seq, best_score, "last_trim")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(768, 0.22)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
