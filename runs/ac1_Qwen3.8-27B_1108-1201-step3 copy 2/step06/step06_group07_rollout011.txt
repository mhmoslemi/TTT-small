import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(a, g):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(A * G, fft_size)[:size]

def lp_direction(a, M, tol=1e-8):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)_k <= M + tol for all k."""
    n = len(a)
    
    # Build constraint matrix: each row k corresponds to (a*g)_k <= M
    # (a*g)_k = sum_j a_j * g_{k-j} where 0 <= j < n and 0 <= k-j < n
    # So for each k, the valid j range is max(0, k-n+1) <= j <= min(n-1, k)
    
    # The constraint matrix C has shape (2n-1, n)
    # C[k, j] = a[j] if 0 <= k-j < n, else 0
    
    # This is a banded matrix (bandwidth n in each direction)
    # Let's build it using sparse matrix
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        # g_j contributes to (a*g)_k for k = j, j+1, ..., j+n-1
        # (a*g)_k gets contribution a_{k-j} * g_j, so for fixed j,
        # the contribution to row k is a[k-j] (when 0 <= k-j < n)
        # So for column j, non-zero entries are at rows j to j+n-1, with values a[0] to a[n-1]
        for i in range(n):
            k = j + i  # row index
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    
    C = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: minimize -sum(g) => c = -1 for all variables
    c = -np.ones(n)
    
    # Constraints: C @ g <= M (for each k), g >= 0
    b = np.full(2*n-1, M + tol)
    
    res = linprog(c, A_ub=C, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return None
    
    g = res.x
    g = np.maximum(g, 0)
    ss = g.sum()
    if ss < 1e-10:
        return None
    return g / ss  # normalize to sum=1

LP-based greedy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses the state-of-the-art LP direction-finding approach: at each step, solve
an LP to find the most "spread out" direction compatible with the current
peak constraint, then blend. Combined with mirror descent polishing,
symmetrization, and diverse multi-start.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def build_lp_structure(n):
    """Precompute row/col structure for the LP constraint matrix."""
    rows = []
    cols = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_direction(a, M, rows, cols):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)_k <= M for all k."""
    n = len(a)
    num_constraints = 2 * n - 1
    vals = np.tile(a, n)
    C = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    c_obj = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if res.status != 0:
        return None
    return res.x


def lp_optimization(a0, time_limit=300, verbose=False, label=""):
    """LP-based greedy optimization loop."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a0, float('inf')
    a = a / s
    n = len(a)

    rows, cols = build_lp_structure(n)

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g_raw = solve_lp_direction(a, M, rows, cols)
        if g_raw is None:
            break
        g_raw = np.maximum(g_raw, 0)
        ss = g_raw.sum()
        if ss < 1e-10:
            break
        g = g_raw / ss

        best_t = None
        best_new_M = M
        for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            new_M = float(fast_convolve(a_new).max())
            if new_M < best_new_M - 1e-15:
                best_new_M = new_M
                best_t = t
                if new_M < M * 0.995:
                    break

        if best_t is None:
            no_improve += 1
            if no_improve > 50:
                # Try relaxed constraint
                g_raw = solve_lp_direction(a, M * 1.005, rows, cols)
                if g_raw is not None:
                    g_raw = np.maximum(g_raw, 0)
                    ss2 = g_raw.sum()
                    if ss2 > 1e-10:
                        g2 = g_raw / ss2
                        for t in [0.3, 0.2, 0.1, 0.05]:
                            a_new = (1.0 - t) * a + t * g2
                            a_new = np.maximum(a_new, 0)
                            a_new = a_new / a_new.sum()
                            new_M = float(fast_convolve(a_new).max())
                            if new_M < M - 1e-15:
                                a = a_new
                                M = new_M
                                no_improve = 0
                                break
                if no_improve > 100:
                    break
            continue

        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0)
        a = a / a.sum()
        M = float(fast_convolve(a).max())
        no_improve = 0

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print(f"    {label} LP iter {it}: obj={2*n*M:.8f}")

    return best_a, 2 * n * best_M


def mirror_descent_polish(a0, time_limit=60, top_k=20, temp_init=0.001,
                          lr_init=0.005, verbose=False, label=""):
    """Mirror descent for final polishing."""
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = float(conv.max())
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        g = compute_subgradient(a, peak_indices, w)

        # Backtracking
        best_lr = lr * 0.1
        best_trial = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-12:
                continue
            log_update = np.clip(-trial_lr * g, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            trial_obj = float(fast_convolve(a_trial).max())
            if trial_obj < best_trial:
                best_lr = trial_lr
                best_trial = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = np.clip(-best_lr * g, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        new_obj = float(fast_convolve(a).max())

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp = min(temp * 1.003, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * 0.9999, 1e-10)

        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.5
            no_improve = 0

        if it % 200 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = float(fast_convolve(a_sym).max())
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if lr < 1e-11:
            break

    return best_a, 2 * n * best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 6
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tukey(n, r=0.3):
    """Tukey (tapered cosine) window."""
    n = int(n)
    w = np.ones(n)
    m = int(n * r / 2)
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / r - 1) / (n * r / (2 * m))))
            w[n - 1 - i] = w[i]
    # Simpler: use the standard formula
    w = np.ones(n)
    m = int(n * r / 2)
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i - (r * n - 1)) / (r * n)))
            w[n - 1 - i] = w[i]
    a = np.maximum(w, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_uniform(n):
    return np.ones(n) / n


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym
