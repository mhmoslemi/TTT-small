    count = 0
    
    for idx, k in enumerate(use_k):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[count] = idx
            cols[count] = j
            vals[count] = f[k - j]
            count += 1
    
    rows = rows[:count]
    cols = cols[:count]
    vals = vals[:count]
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(n_used, n))
    b_vec = np.full(n_used, M)
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            return None
    except Exception:
        return None


def lp_optimize(a0, time_limit, n_constraints=None, verbose=True):
    """LP-based optimization on the simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        return a, float('inf')
    a = a / S
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        
        g = lp_solve_direction(a, M, n_constraints)
        
        if g is None or g.sum() < 1e-10:
            # Fallback: use subgradient direction
            n_peaks = max(5, n // 50)
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
            w = np.zeros(2*n-1)
            w[top_idx] = 1.0 / n_peaks
            fft_size = 1 << (2*n-1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g = irfft(np.conj(W) * A, fft_size)[:n]
            g = np.maximum(g, 0)
            if g.sum() < 1e-15:
                break
            g = g / g.sum()
        
        g = g / g.sum()
        
        # Line search
        best_t = 0
        best_obj_trial = M
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1 - t) * a + t * g
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_obj_trial:
                best_obj_trial = obj_trial
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            new_obj = fast_convolve(a).max()
            
            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={elapsed:.1f}s")
        
        if no_improve > 30:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
        
        if no_improve > 100:
            break
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit, lr_init=0.002, momentum=0.9, n_top=50):
    """Subgradient method with Nesterov momentum for polishing."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        b_max = b.max()
        
        # Subgradient using top peaks
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w = np.zeros(2*n-1)
        w[top_idx] = 1.0 / n_top
        
        fft_size = 1 << (2*n-1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g = 2 * irfft(np.conj(W) * A, fft_size)[:n]
        
        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 500:
            cur_lr *= 0.8
            no_improve = 0
        
        if cur_lr < 1e-8:
            break
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try several multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_dual_bump_init(n):
    """Two separated bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.3) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.15) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
