"""

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g_star, sum_g_star, dual_y) or None if infeasible."""
    # Build sparse constraint matrix
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # For each j, nonzeros at k = j, j+1, ..., j+n-1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if f[j] < 1e-15:
            continue
        k_start = j
        k_end = j + n  # exclusive
        rows.extend(range(k_start, min(k_end, 2*n-1)))
        cols.extend([j] * (min(k_end, 2*n-1) - k_start))
        vals.extend([f[j]] * (min(k_end, 2*n-1) - k_start))
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res.x.sum(), res.unequality_slack  # slack gives dual info
    return None

def solve_lp_dual(f, M, n):
    """Solve dual: min M*sum(y) s.t. (y * f_rev)[j] >= 1 for all j, y >= 0.
    Returns (y_star, obj_value) or None."""
    # (y * f_rev)[j] = sum_k y[k] * f[k-j] where 0 <= k-j < n
    # For each j, this is sum over k from j to j+n-1 of y[k] * f[k-j]
    # = sum over i from 0 to n-1 of y[j+i] * f[i]
    # So the constraint matrix B[j, k] = f[k-j] if 0 <= k-j < n, else 0
    # B[j, k] = f[k-j] if j <= k <= j+n-1
    # This is the same as A transposed!
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = j + i
            if k < 2*n-1 and f[i] >= 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    
    B = sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    # Dual: min M * sum(y) s.t. B^T @ y >= 1, y >= 0
    # In scipy form: min c^T y s.t. A_ub @ y <= b_ub, A_eq @ y = b_eq, bounds
    # min M*sum(y) s.t. -B^T @ y <= -1, y >= 0
    c_dual = np.full(2*n-1, M)
    A_ub_dual = -B.T.tocsr()
    b_ub_dual = -np.ones(n)
    bounds_dual = [(0, None)] * (2*n-1)
    
    res = linprog(c_dual, A_ub=A_ub_dual, b_ub=b_ub_dual, bounds=bounds_dual, method='highs')
    if res.success:
        return res.x, res.fun
    return None

def make_hann(n):
    """Hann window normalized to sum 1."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length with smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Gaussian smoothing
    sigma = max(1.0, new_n * 0.01)
    kernel_size = int(6 * sigma) | 1
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) / sigma) ** 2
    kernel /= kernel.sum()
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with dual
certificates, multi-length exploration, and smoothed gradient warm-up.

Key improvements:
- Cutting-plane LP for faster direction finding
- Dual certificate for optimality verification and early stopping
- Multi-length exploration with smart interpolation
- Adaptive time allocation based on convergence
- Power-sum smoothing as alternative to log-sum-exp

import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build sparse constraint matrix A where A[k,j] = f[j] if j<=k<=j+n-1."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        fj = f[j]
        if fj < 1e-15:
            continue
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        count = k_end - k_start
        if count <= 0:
            continue
        rows.extend(range(k_start, k_end))
        cols.extend([j] * count)
        vals.extend([fj] * count)
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(f, M, n, A=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g_star, sum_g) or None."""
    if A is None:
        A = build_lp_matrix(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={"presolve": True, "time_limit": 30})
    if res.status == 0:
        return res.x, res.x.sum()
    elif res.status == 1:
        # Time limit hit but we have a solution
        if res.x is not None and res.x.sum() > 0:
            return res.x, res.x.sum()
    return None


def solve_lp_dual(f, M, n):
    """Solve dual LP: min M*sum(y) s.t. B^T y >= 1, y >= 0.
    Returns (optimal_value, y_star) or None."""
    A = build_lp_matrix(f, n)
    # Dual: min M * 1^T y s.t. A^T y >= 1, y >= 0
    # In scipy: min c^T y s.t. A_ub y <= b_ub, bounds
    c_dual = np.full(2 * n - 1, M)
    A_ub_dual = -A.T.tocsr()
    b_ub_dual = -np.ones(n)
    bounds_dual = [(0, None)] * (2 * n - 1)
    res = linprog(c_dual, A_ub=A_ub_dual, b_ub=b_ub_dual, bounds=bounds_dual,
                  method='highs', options={"presolve": True, "time_limit": 30})
    if res.status == 0:
        return res.fun, res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with dual certificate for stopping."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    best_dual_bound = M  # We'll try to get a lower bound from dual

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    A_cache = build_lp_matrix(f, n)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve primal LP for direction
        result = solve_lp_direction(f, M, n, A_cache)
        if result is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break
        g_star, sum_g = result

        if sum_g < 1e-12:
            if verbose:
                print(f"    LP gives zero at iter {it}")
            break

        # Check convergence: at optimum, sum_g = 1
        gain = sum_g - 1.0
        g_norm = g_star / sum_g
        dist = np.max(np.abs(g_norm - f))

        if verbose and it % 5 == 0:
            print(f"    LP-FW iter {it}: M={M:.12f}, sum_g={sum_g:.10f}, "
                  f"gain={gain:.2e}, dist={dist:.2e}")

        # Convergence check
        if abs(gain) < 1e-10 and dist < 1e-8:
            no_improve += 1
            if verbose:
                print(f"    Converged! sum_g={sum_g:.15f}")
            # Try to verify with dual

                # Solve dual for certificate
                dual_result = solve_lp_dual(f, M, n)
                if dual_result is not None:
                    dual_obj, y_star = dual_result
                    # Dual gives lower bound: for any f', max(f'*f') >= 1/sum(y*)
                    # If dual_obj = M * sum(y*) and this equals sum_g = 1,
                    # then M is optimal
                    if verbose:
                        print(f"    Dual certificate: obj={dual_obj:.12f}")
                    best_dual_bound = min(best_dual_bound, 1.0 / max(y_star.sum(), 1e-15))
                break
            # Perturb to check if truly optimal
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(f + rng.standard_normal(n) * 0.0005)
            c = fast_convolve(f)
            M = c.max()
            A_cache = build_lp_matrix(f, n)
            continue
        else:
            no_improve = 0

        # Line search
        direction = g_norm - f
        best_alpha = 0.0
        best_val = M

        # Golden section search for efficiency
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_lo, a_hi = 0.0, 1.0
        x1 = a_hi - gr * (a_hi - a_lo)
        x2 = a_lo + gr * (a_hi - a_lo)

        def eval_alpha(alpha):
            ft = f + alpha * direction
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                return float('inf')
            ft = ft / st
            return fast_convolve(ft).max()

        f1 = eval_alpha(x1)
        f2 = eval_alpha(x2)
        if f1 < best_val:
            best_val = f1
            best_alpha = x1
        if f2 < best_val:
            best_val = f2
            best_alpha = x2

        for _ in range(25):
            if f1 < f2:
                a_hi = x2
                x2 = x1
                f2 = f1
                x1 = a_hi - gr * (a_hi - a_lo)
                f1 = eval_alpha(x1)
            else:
                a_lo = x1
                x1 = x2
                f1 = f2
                x2 = a_lo + gr * (a_hi - a_lo)
                f2 = eval_alpha(x2)
            if f1 < best_val:
                best_val = f1
                best_alpha = x1
            if f2 < best_val:
                best_val = f2
                best_alpha = x2
            if a_hi - a_lo < 1e-6:
                break

        if best_val < M:
            best_alpha_check = best_alpha
        else:
            best_alpha_check = 0.0

        if best_alpha_check > 1e-10:
            f_new = f + best_alpha_check * direction
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            c_new = fast_convolve(f_new)
            M = c_new.max()
            f = f_new
            A_cache = build_lp_matrix(f, n)

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
        else:
            no_improve += 1
            if no_improve > 30:
                break

    if verbose:
        print(f"    LP-FW done: {it} iters, best_obj={best_obj:.12f}")

    return best_f, best_obj


def smoothed_gradient_descent(f0, time_limit, verbose=False):
    """Power-sum smoothed gradient descent with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    t0 = time.monotonic()
    momentum = np.zeros(n)
    mu = 0.85

    # Power-sum parameter schedule
    p_schedule = [2, 4, 8, 16, 32, 64, 128]

    for p in p_schedule:
        if time.monotonic() - t0 > time_limit * 0.9:
            break

        stage_start = time.monotonic()
        stage_time = min(60, (time_limit - (time.monotonic() - t0)) / (len(p_schedule) - p_schedule.index(p) + 1))
        no_improve = 0
        lr = 0.5 / (n * p)
        it = 0

        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.9:
            it += 1
            c = fast_convolve(f)
            max_c = c.max()
            if max_c < 1e-15:
                break

            # Power-sum weights
            c_norm = c / max_c
            w = c_norm ** (p - 1)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.ones_like(c) / len(c)
            else:
                w = w / w_sum

            # Gradient: g_j = 2 * sum_i w_i * f_{i-j} (cross-correlation)
            w_rev = w[::-1].copy()
            conv_len = n + len(w_rev) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]

            # Momentum step
            momentum = mu * momentum - lr * g
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)

            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()

            if obj_trial <= c.max() + 1e-12:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                f_trial = f + 0.5 * momentum
                f_trial = project_simplex(f_trial)
                c_trial = fast_convolve(f_trial)
                obj_trial = c_trial.max()
                if obj_trial <= c.max() + 1e-12:
                    f = f_trial
                else:
                    momentum = momentum * 0.5
                no_improve += 1
                lr *= 0.9

            if no_improve > 100:
                rng = np.random.default_rng(it * 17 + p * 3)
                noise_scale = 0.003 / np.sqrt(p / 4.0)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                momentum = np.zeros(n)
                lr = 0.5 / (n * p)
                no_improve = 0

        if verbose:
            print(f"    SGD p={p}: best={best_obj:.10f}, time={time.monotonic()-t0:.1f}s")

    return best_f, best_obj


def make_hann(n):
    """Hann window normalized to sum 1."""
    if n <= 1:
        return np.ones(max(n, 1))
    x = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window normalized to sum 1."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, flat_frac=0.6):
    """Flat-top window with cosine transition."""
    x = np.arange(n) - (n - 1) / 2.0
    half = (n - 1) / 2.0
    t = np.abs(x) / half
    a = np.ones(n)
    mask = t > flat_frac
    if mask.any():
        t_trans = (t[mask] - flat_frac) / (1.0 - flat_frac)
        a[mask] = 0.5 * (1.0 + np.cos(np.pi * np.minimum(t_trans, 1.0)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_nuttall(n):
    """Nuttall window (optimal for spectral leakage)."""
    a0, a1, a2, a3 = 0.3635819, 0.4891775, 0.1365995, 0.0106411
    x = np.arange(n)
    t = x / (n - 1) if n > 1 else np.zeros(1)
    a = (a0 - a1 * np.cos(2 * np.pi * t) + a2 * np.cos(4 * np.pi * t) - a3 * np.cos(6 * np.pi * t))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length with light smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Mild Gaussian smoothing
    sigma = max(0.5, new_n * 0.005)
    ks = int(4 * sigma) | 1
    kx = np.arange(ks) - ks // 2
    kernel = np.exp(-0.5 * (kx / sigma) ** 2)
    kernel /= kernel.sum()
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0.0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def objective_normalized(f):
    """Compute 2*n*max(f*f) for normalized f (sum=1)."""
    n = len(f)
    c = fast_convolve(f)
    return 2.0 * n * c.max()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining smoothed gradient descent and
    LP-based Frank-Wolfe with dual certificates.
    Explores multiple sequence lengths for better optimization landscape.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_hann(768)

    n_parent = len(parent)

    # === Phase 1: Screen initializations at parent length ===
    print("=== Phase 1: Initialization Screening ===")
    t1_start = time.monotonic()

    shape_fns = [
        ("hann", make_hann),
        ("gauss_0.25", lambda nn: make_gaussian(nn, 0.25)),
        ("gauss_0.3", lambda nn: make_gaussian(nn, 0.3)),
        ("gauss_0.35", lambda nn: make_gaussian(nn, 0.35)),
        ("flat_top", make_flat_top),
        ("nuttall", make_nuttall),
    ]

    best_init = None
    best_init_obj = float('inf')

    # Parent
    obj_p = objective_normalized(parent)
    if obj_p < best_init_obj:
        best_init_obj = obj_p
        best_init = parent.copy()
    try_seq(parent, "init_parent")

    # Theoretical shapes
    for name, fn in shape_fns:
        a = fn(n_parent)
        obj = objective_normalized(a)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a.copy()
        try_seq(a, f"init_{name}")

    # Small random perturbations
    for trial in range(8):
        noise = rng.standard_normal(n_parent) * (0.02 / (trial + 1))
        a_pert = np.maximum(parent + noise, 0)
        a_pert = a_pert / a_pert.sum()
        obj = objective_normalized(a_pert)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_pert.copy()
        try_seq(a_pert, f"init_pert{trial}")

    print(f"  Best init: obj={best_init_obj:.10f}")
    print(f"  Phase 1 time: {time.monotonic()-t1_start:.1f}s")

    # === Phase 2: Smoothed gradient descent warm-up ===
    print(f"\n=== Phase 2: Smoothed GD Warm-up (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    t_alloc = max(30, min(rem, 200))
    print(f"  Allocation: {t_alloc:.0f}s")

    a_opt, obj = smoothed_gradient_descent(best_init, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, "sgd_warm")

    # === Phase 3: LP-FW at parent length ===
    print(f"\n=== Phase 3: LP-FW (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 500
    t_alloc = max(60, min(rem, 400))
    print(f"  Allocation: {t_alloc:.0f}s")

    # Start from the best of sgd result and best_init
    candidates = [best_init, a_opt]
    if best_seq is not None and len(best_seq) == n_parent:
        candidates.append(best_seq)
    start = min(candidates, key=objective_normalized)

    a_opt, obj = lp_fw_optimize(start, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, "lpfw_parent")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    base = best_seq if best_seq is not None else a_opt

    # Choose target lengths
    target_lengths = set()
    for factor in [0.75, 1.0, 1.25, 1.5, 2.0, 2.5, 3.0]:
        tn = int(n_parent * factor)
        tn = max(128, (tn // 64) * 64)
        if tn <= 4096:
            target_lengths.add(tn)
    target_lengths = sorted(target_lengths)

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 100
        if rem < 80:
            break

        if target_n == n_parent:
            # Already done, just verify
            continue

        print(f"  --- n={target_n}, remaining={rem:.0f}s ---")

        # Interpolate and try theoretical shapes
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")

        f_hann = make_hann(target_n)
        try_seq(f_hann, f"hann_n{target_n}")

        f_gauss = make_gaussian(target_n, 0.3)
        try_seq(f_gauss, f"gauss_n{target_n}")

        f_nuttall = make_nuttall(target_n)
        try_seq(f_nuttall, f"nuttall_n{target_n}")

        # Pick best starting point
        cands = [f_interp, f_hann, f_gauss, f_nuttall]
        if target_n == len(base):
            cands.append(base)
        start_n = min(cands, key=objective_normalized)

        # Quick SGD warm-up
        rem2 = deadline - time.monotonic() - 60
        if rem2 > 40:
            t_sgd = min(rem2 * 0.2, 60)
            a_sgd, _ = smoothed_gradient_descent(start_n, time_limit=t_sgd)
            if objective_normalized(a_sgd) < objective_normalized(start_n):
                start_n = a_sgd

        # LP-FW
        rem3 = deadline - time.monotonic() - 40
        if rem3 > 30:
            t_lpfw = min(rem3 * 0.6, 180)
            a_opt_n, _ = lp_fw_optimize(start_n, time_limit=t_lpfw, verbose=True)
            try_seq(a_opt_n, f"lpfw_n{target_n}")

    # === Phase 5: Final polish on global best ===
    print(f"\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # More LP-FW iterations on the best
        rem = deadline - time.monotonic() - 30
        if rem > 40:
            t_alloc = min(rem, 200)
            print(f"  LP-FW polish (n={n_best}): t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "polish_lpfw")

        # Try perturbed versions
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            for p in range(2):
                rem2 = deadline - time.monotonic() - 15
                if rem2 < 20:
                    break
                noise = rng.standard_normal(n_best) * (0.002 / (p + 1))
                f_pert = project_simplex(best_seq + noise)
                t_pert = min(rem2 / 2, 80)
                a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_pert)
                try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = make_hann(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
