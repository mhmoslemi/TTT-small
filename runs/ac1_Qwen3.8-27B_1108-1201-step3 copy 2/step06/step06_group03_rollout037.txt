"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive multiplicative weights
with top-peak subgradients, LP-Frank-Wolfe refinement, and multi-strategy
initialization. Key insight: asymmetric sequences can achieve score < 2 by
shifting the convolution peak away from the center.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, peaks, weights):
    """Compute weighted subgradient of max(conv) at given peaks."""
    n = len(a)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            idx = np.arange(j_min, j_max + 1)
            sg[j_min:j_max + 1] += 2.0 * weights[i] * a[k - idx]
    return sg


def multiplicative_weights(a0, time_limit=60.0, verbose=False,
                           lr_init=0.05, top_k=30, restart_every=80):
    """
    Multiplicative weights (mirror descent) for minimizing max(conv(a,a))
    subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    tau = 1.0  # temperature for softmax

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Select top-k peaks
        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]

        # Softmax weights (adaptive temperature)
        tau = M / (3.0 + 0.05 * it)
        if tau < 1e-10:
            tau = 1e-10
        scaled = pv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        # Compute subgradient
        sg = compute_subgradient(a, peaks, w)

        # Multiplicative update
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        # Evaluate
        new_M = fast_convolve(a_new).max()

        if new_M < M * (1 - 1e-12):
            a = a_new
            if new_M < best_obj - 1e-14:
                best_obj = new_M
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.5)
            else:
                no_improve += 1
            if verbose and it % 50 == 0:
                print(f"    MW iter {it}: obj={best_obj:.10f}, lr={lr:.4f}")
        else:
            no_improve += 1
            lr *= 0.8

        # Restart
        if no_improve > restart_every:
            rng = np.random.default_rng(it * 13 + 7)
            noise = rng.standard_normal(n) * 0.005
            a = np.maximum(best_a + noise, 1e-15)
            a = a / a.sum()
            lr = lr_init
            no_improve = 0

        if lr < 1e-7:
            break

    return best_a, best_obj


def lp_frank_wolfe(a0, time_limit=60.0, verbose=False):
    """
    LP-based Frank-Wolfe: at each step, solve LP to find direction g
    that maximizes sum(g) subject to (a*g)[k] <= M for all k, g >= 0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M

"""

(mirror descent) with top-peak subgradients, LP-Frank-Wolfe refinement,
coordinate descent polish, and multi-strategy initialization.
Key insight: asymmetric sequences can achieve score < 2 by shaping the
auto-convolution to be flatter than the triangular shape of uniform sequences.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, peaks, weights):
    """Compute weighted subgradient: sum_i w_i * 2*a[k_i - j] for valid j."""
    n = len(a)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        wi = weights[i]
        if wi < 1e-12:
            continue
        k = int(peaks[i])
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            idx = np.arange(j_min, j_max + 1)
            sg[j_min:j_max + 1] += 2.0 * wi * a[k - idx]
    return sg

    n = len(a)
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return a, best_obj
            old = a[j]
            if old < 1e-14:
                continue
            best_local_obj = best_obj
            best_local_val = old

            for frac in [0.6, 0.8, 0.9, 1.1, 1.2, 1.4, 0.4]:
                new_val = old * frac
                if new_val <= 0 or new_val > 0.8:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-15:
                    a[j] = old
                    continue
                a = a / s
                val = fast_convolve(a).max()
                if val < best_local_obj - 1e-13:
                    best_local_obj = val
                    best_local_val = new_val

            # Set to best local
            a[j] = best_local_val
            a = a / a.sum()
            if best_local_obj < best_obj - 1e-13:
                best_obj = best_local_obj
                improved = True

        if not improved:
            break

    return a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv)/sum^2.
    Uses diverse initializations, multiplicative weights, LP-FW, and CD.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Try to use parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print("=== Phase 1: Initializations ===")
    test_lengths = [640, 768, 1024, 1280]
    init_candidates = []

    for n_try in test_lengths:
        for v in range(14):
            a_init = make_init(n_try, variant=v, seed=seed)
            init_candidates.append((a_init, f"v{v}_n{n_try}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))

    # Score initializations by fast objective
    scored = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored.append((obj, a_norm, label))

    scored.sort(key=lambda x: x[0])
    top_inits = scored[:6]
    print(f"  Top 6 inits: {[(l, f'{o:.4f}') for o, _, l in top_inits]}")

    # === Phase 2: Multiplicative weights on top inits ===
    print("\n=== Phase 2: Multiplicative Weights ===")
    for idx, (obj_init, a_init, label) in enumerate(top_inits[:4]):
        rem = deadline - time.monotonic() - 500
        if rem < 20:
            break
        t_alloc = min(rem / max(1, 4 - idx), 250)
        if t_alloc < 15:
            t_alloc = 15
        print(f"  MW on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, obj = multiplicative_weights(a_init, time_limit=t_alloc,
                                            verbose=True, lr_init=0.04,
                                            top_k=40, restart_every=60)
        try_seq(a_opt, f"mw_{label}")

    # === Phase 3: Continued MW on best ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"\n=== Phase 3: MW Continue (n={len(best_seq)}) ===")
            a_opt, obj = multiplicative_weights(best_seq, time_limit=t_alloc,
                                                verbose=True, lr_init=0.02,
                                                top_k=50, restart_every=40)
            try_seq(a_opt, "mw_continue")

    # === Phase 4: LP-Frank-Wolfe ===
    print("\n=== Phase 4: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  LP-FW (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_frank_wolfe(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length ===")
    if best_seq is not None:
        for target_n in [512, 896, 1024, 1536]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            if len(best_seq) == target_n:
                continue
            # Interpolate
            old_n = len(best_seq)
            old_x = np.linspace(0, 1, old_n)
            new_x = np.linspace(0, 1, target_n)
            f_interp = np.interp(new_x, old_x, best_seq)
            f_interp = np.maximum(f_interp, 0)
            ss = f_interp.sum()
            if ss > 1e-15:
                f_interp = f_interp / ss
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.3, 60)
                if t_alloc > 15:
                    a_opt, obj = multiplicative_weights(f_interp, time_limit=t_alloc,
                                                        lr_init=0.03, top_k=30)
                    try_seq(a_opt, f"mw_n{target_n}")

    # === Phase 6: Random restarts ===
    print("\n=== Phase 6: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(5):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem / max(1, 5 - r), 100)
            if t_alloc < 15:
                t_alloc = 15

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.008
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            elif r == 1:
                # Blend with asymmetric
                asym = make_init(n_best, variant=int(rng.integers(0, 14)), seed=seed + r * 7)
                f_init = 0.6 * best_seq + 0.4 * asym
                f_init = f_init / f_init.sum()
            elif r == 2:
                # Mirror blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif r == 3:
                # Larger perturbation
                noise = rng.standard_normal(n_best) * 0.02
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            else:
                # Random re-init with structure
                f_init = make_init(n_best, variant=int(rng.integers(0, 14)), seed=seed + 100 + r)
                f_init = 0.5 * best_seq + 0.5 * f_init
                f_init = f_init / f_init.sum()

            print(f"  Restart {r}: t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_weights(f_init, time_limit=t_alloc,
                                                lr_init=0.03, top_k=35, restart_every=50)
            try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Coordinate descent polish ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(rem, 80)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 8: Final LP-FW squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 40)
            print(f"\n=== Phase 8: Final LP-FW ===")
            a_opt, obj = lp_frank_wolfe(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

    # === Phase 9: One more MW pass ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 25)
            a_opt, obj = multiplicative_weights(best_seq, time_limit=t_alloc,
                                                lr_init=0.01, top_k=50,
                                                restart_every=30)
            try_seq(a_opt, "final_mw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
