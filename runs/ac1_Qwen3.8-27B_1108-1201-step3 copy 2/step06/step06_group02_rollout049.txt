"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe with away steps,
adaptive M-relaxation, bounded LP directions, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def build_conv_matrix(f, n):
    """Build the convolution constraint matrix A where (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_fw_lp(f, M, n, upper_bound=None):
    """Solve the FW LP: max sum(g) s.t. (f*g)[k] <= M, 0 <= g <= ub."""
    c = -np.ones(n)
    A = build_conv_matrix(f, n)
    b = np.full(2 * n - 1, M)
    if upper_bound is not None:
        bounds = [(0, upper_bound)] * n
    else:
        bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def ibp_line_search(f, s, t_max=1.0, n_evals=15):
    """
    Inexact bisection (IBP) line search for FW.
    Finds smallest t such that <s-f, f_t - f> / ||s-f||^2 <= gamma
    Simplified: use golden section to minimize the objective along the path.
    """
    # Golden section search
    lo, hi = 0.0, t_max
    golden = (np.sqrt(5) - 1) / 2  # ~0.618
    
    # Initial evaluations
    t1 = hi - golden * (hi - lo)
    t2 = lo + golden * (hi - lo)
    
    f_t1 = np.maximum(f + t1 * (s - f), 0)
    f_t1 = f_t1 / f_t1.sum()
    obj1 = fast_convolve(f_t1).max()
    
    f_t2 = np.maximum(f + t2 * (s - f), 0)
    f_t2 = f_t2 / f_t2.sum()
    obj2 = fast_convolve(f_t2).max()
    
    best_t = t1 if obj1 < obj2 else t2
    best_obj = min(obj1, obj2)
    
    for _ in range(n_evals - 2):
        if obj1 < obj2:
            hi = t2
            t2 = t1
            obj2 = obj1
            t1 = hi - golden * (hi - lo)
            f_t1 = np.maximum(f + t1 * (s - f), 0)
            f_t1 = f_t1 / f_t1.sum()
            obj1 = fast_convolve(f_t1).max()
        else:
            lo = t1
            t1 = t2
            obj1 = obj2
            t2 = lo + golden * (hi - lo)
            f_t2 = np.maximum(f + t2 * (s - f), 0)
            f_t2 = f_t2 / f_t2.sum()
            obj2 = fast_convolve(f_t2).max()
        
        if obj1 < best_obj:
            best_obj = obj1
            best_t = t1
        if obj2 < best_obj:
            best_obj = obj2
            best_t = t2
    
    return best_t, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A(f, n):
    """Build convolution matrix: (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, n, ub=None):
    """LP: max sum(g) s.t. (f*g)[k] <= M, 0 <= g <= ub."""
    c = -np.ones(n)
    A = build_A(f, n)
    b = np.full(2 * n - 1, M)
    if ub is not None:
        bounds = [(0, ub)] * n
    else:
        bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0)
    except:
        pass
    return None


def line_search(f, direction, n_evals=12):
    """Golden section line search along f + t*direction (projected to simplex)."""
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5) - 1) / 2
    
    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)
    
    def eval_at(t):
        ft = f + t * direction
        ft = np.maximum(ft, 0)
        st = ft.sum()
        if st < 1e-15:
            return np.inf
        ft = ft / st
        return fast_convolve(ft).max()
    
    o1 = eval_at(t1)
    o2 = eval_at(t2)
    best_t = t1 if o1 < o2 else t2
    best_o = min(o1, o2)
    
    for _ in range(n_evals - 2):
        if o1 < o2:
            hi = t2
            t2, o2 = t1, o1
            t1 = hi - gr * (hi - lo)

        else:

            t1, o1 = t2, o2
            t2 = lo + gr * (hi - lo)
            o2 = eval_at(t2)
        
        if o1 < best_o:
            best_o, best_t = o1, t1
        if o2 < best_o:
            best_o, best_t = o2, t2
    
    return best_t, best_o


def away_line_search(f, s, n_evals=8):

    # Away direction is (f - s), valid as long as f + t*(f-s) >= 0

def peak_descent(f, time_limit, verbose=False):
    """Local search to reduce the maximum peak."""
    f = normalize(np.maximum(f, 0))
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 100:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        # Contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = np.arange(js, je + 1)
        
        if len(contrib) == 0:
            break
        
        # Find the most contributing pair
        products = f[contrib] * f[peak_k - contrib]
        max_idx = contrib[np.argmax(products)]
        partner = peak_k - max_idx
        
        # Try reducing the larger of the pair
        src = max_idx if f[max_idx] >= f[partner] else partner
        
        # Try different reduction amounts
        found = False
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            delta = f[src] * frac
            if delta < 1e-15:
                continue
            trial = f.copy()
            trial[src] -= delta
            # Redistribute to non-contributing coordinates
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib] = False
            if non_contrib.sum() < 1:
                non_contrib[:] = True
                non_contrib[src] = False
            nc_vals = trial[non_contrib]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                continue
            trial[non_contrib] += delta * nc_vals / nc_sum
            trial = np.maximum(trial, 0)
            trial = normalize(trial)
            obj = fast_convolve(trial).max()
            if obj < best_obj - 1e-14:
                f = trial
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                found = True
                break
        
        if not found:
            # Try a smaller peak
            top_peaks = np.argsort(conv)[-3:]
            found_any = False
            for pk in top_peaks:
                pk = int(pk)
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                c2 = np.arange(js2, je2 + 1)
                prods2 = f[c2] * f[pk - c2]
                if len(c2) == 0:
                    continue

                for frac2 in [0.1, 0.2, 0.3]:
                    delta2 = f[src2] * frac2

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        sg = np.zeros(n)
        for i in range(js, je + 1):
            sg[i] += 2.0 * f[peak_k - i]
        
        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay learning rate
        lr *= 0.999
        
        if no_improve > 500:
            # Restart
            rng = np.random.default_rng(time.monotonic() * 1000)
            f = normalize(np.maximum(best_f + rng.standard_normal(n) * 0.01, 0))
            lr = 0.005
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-Frank-Wolfe with away steps and adaptive M-relaxation
- Bounded LP directions to prevent degenerate solutions
- Multi-length exploration with interpolation
- Peak-targeted coordinate descent for final polish
- Multiplicative descent as an alternative optimizer
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, nonnegative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(f, n):
    """Build matrix A where (A@g)[k] = (f*g)[k] for linear convolution."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_fw_lp(f, M, n, ub=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, 0 <= g [<= ub]."""
    c = -np.ones(n)
    A = build_conv_matrix(f, n)
    b = np.full(2 * n - 1, M)
    if ub is not None:
        bounds = [(0.0, float(ub))] * n
    else:
        bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def golden_line_search(f, direction, t_max=1.0, n_evals=12):
    """Golden section line search: minimize max(conv) along f + t*direction."""
    lo, hi = 0.0, t_max
    gr = (np.sqrt(5) - 1) / 2.0
    
    def eval_at(t):
        ft = f + t * direction
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return np.inf
        ft = ft / s
        return fast_convolve(ft).max()
    
    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)
    o1 = eval_at(t1)
    o2 = eval_at(t2)
    best_t = t1 if o1 <= o2 else t2
    best_o = o1 if o1 <= o2 else o2
    
    for _ in range(max(0, n_evals - 2)):
        if o1 <= o2:
            hi = t2
            t2, o2 = t1, o1
            t1 = hi - gr * (hi - lo)
            o1 = eval_at(t1)
        else:
            lo = t1
            t1, o1 = t2, o2
            t2 = lo + gr * (hi - lo)
            o2 = eval_at(t2)
        if o1 < best_o:
            best_o, best_t = o1, t1
        if o2 < best_o:
            best_o, best_t = o2, t2
    
    return best_t, best_o


def away_step_search(f, s, n_evals=8):
    """Search for away step: minimize max(conv) along f + t*(f-s), t>=0, f+t(f-s)>=0."""
    diff = f - s
    # Find max t such that f + t*diff >= 0
    neg = diff < -1e-15
    if neg.any():
        t_max = (f[neg] / (-diff[neg])).min()
    else:
        t_max = 1.0
    t_max = min(t_max, 1.0)
    
    if t_max < 1e-10:
        return 0.0, fast_convolve(f).max()
    
    lo, hi = 0.0, t_max
    gr = (np.sqrt(5) - 1) / 2.0
    
    def eval_at(t):
        ft = f + t * diff
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return np.inf
        ft = ft / s
        return fast_convolve(ft).max()
    
    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)
    o1 = eval_at(t1)
    o2 = eval_at(t2)
    best_t = t1 if o1 <= o2 else t2
    best_o = o1 if o1 <= o2 else o2
    
    for _ in range(max(0, n_evals - 2)):
        if o1 <= o2:
            hi = t2
            t2, o2 = t1, o1
            t1 = hi - gr * (hi - lo)
            o1 = eval_at(t1)
        else:
            lo = t1
            t1, o1 = t2, o2
            t2 = lo + gr * (hi - lo)
            o2 = eval_at(t2)
        if o1 < best_o:
            best_o, best_t = o1, t1
        if o2 < best_o:
            best_o, best_t = o2, t2
    
    return best_t, best_o


def lp_fw_optimize(f0, time_limit, ub_factor=3.0, verbose=False):
    """
    LP-Frank-Wolfe with away steps.
    f0: initial sequence (will be normalized)
    time_limit: max seconds
    ub_factor: upper bound for LP variables = ub_factor / n
    """
    f = normalize(f0.astype(np.float64))
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    m_factors = [1.0, 1.02, 1.05, 1.1, 1.2, 1.4, 1.7, 2.2, 3.0, 5.0]
    m_idx = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3.0:
            break
        
        # Solve LP with escalating M
        g = None
        for level in range(m_idx, len(m_factors)):
            M_try = M * m_factors[level]
            ub = (ub_factor / n) if level >= 2 else None
            g = solve_fw_lp(f, M_try, n, ub=ub)
            if g is not None and g.sum() > 1e-12:
                m_idx = level
                break
            m_idx = min(level + 1, len(m_factors))
        
        if g is None or g.sum() < 1e-12:
            # Perturb and continue
            rng = np.random.default_rng(it * 131 + 17)
            f = normalize(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_idx = 0
            no_improve += 1
            if no_improve > 150:
                break
            continue
        
        g = normalize(g)
        
        # Check if direction is meaningful
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_factors) - 1)
            if no_improve > 100:
                break
            continue
        
        no_improve = 0
        
        # Forward line search
        t_fwd, obj_fwd = golden_line_search(f, g - f, t_max=1.0, n_evals=10)
        
        # Away step
        t_away, obj_away = away_step_search(f, g, n_evals=6)
        
        # Choose best step
        if obj_away < obj_fwd and t_away > 0:
            direction = f - g
            t_step = t_away
            step_obj = obj_away
        else:
            direction = g - f
            t_step = t_fwd
            step_obj = obj_fwd
        
        if t_step < 1e-10 or step_obj >= M - 1e-14:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_factors) - 1)
            if no_improve > 80:
                break
            continue
        
        # Apply step
        f = f + t_step * direction
        f = normalize(f)
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            m_idx = 0
            if verbose and it % 100 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}, n={n}")
        elif M > best_obj * 1.01:
            # Got worse, revert
            f = best_f.copy()
            conv = fast_convolve(f)
            M = conv.max()
    
    return best_f, best_obj
