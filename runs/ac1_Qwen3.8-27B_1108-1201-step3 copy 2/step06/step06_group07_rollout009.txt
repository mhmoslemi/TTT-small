def solve_lp(f, M, n):
    """Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    # c = -1 for all variables (minimize -sum(g))
    c = -np.ones(n)
    
    # A: (2n-1) x n matrix where A[k, j] = f[k-j] if 0 <= k-j < n

    for k in range(2*n-1):

            idx = k - j
            if 0 <= idx < n:

    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return f.copy()  # fallback

LP-guided iterative descent for minimizing 2n*max(auto-conv)/(sum)^2.
Uses linear programming to find optimal descent directions, combined with
mirror descent and coordinate descent for polishing.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    # Clip small negative values from FFT numerical error
    result = np.maximum(result, 0)
    return result

def build_conv_matrix(f, n):
    """Build A where A[k,j] = f[k-j] if 0 <= k-j < n."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def solve_lp_direction(f, M, n, max_time=5.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    c = -np.ones(n)
    A = build_conv_matrix(f, n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    options = {'presolve': True, 'time_limit': max_time}
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options=options)
    
    if res.status == 0:
        return res.x
    return None

def lp_iterate(f, time_limit, verbose=False):
    """Main LP-guided iteration."""
    n = len(f)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()

    best_M = fast_convolve(f).max()

    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Solve LP
        g = solve_lp_direction(f, M, n)
        if g is None:
            # LP failed, use subgradient direction
            g = f.copy()
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            g_sum = 1.0
        g_norm = g / g_sum
        
        # Backtracking line search
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            f_trial = (1-t)*f + t*g_norm
            f_trial = np.maximum(f_trial, 1e-15)
            f_trial = f_trial / f_trial.sum()
            new_M = fast_convolve(f_trial).max()
            if new_M < M - 1e-14:
                f = f_trial
                improved = True
                break
        
        if not improved:

            if no_improve > 50:

                noise = rng.standard_normal(n) * 0.001

                f = f / f.sum()
                no_improve = 0
            elif no_improve > 20:

                f_sym = 0.5 * (f + f[::-1])
                f_sym = np.maximum(f_sym, 1e-15)
                f_sym = f_sym / f_sym.sum()
                sym_M = fast_convolve(f_sym).max()
                if sym_M < M:
                    f = f_sym
                    if sym_M < best_M:
                        best_M = sym_M
                        best_f = f.copy()
                    no_improve = 0
                else:
                    break  # Truly stuck
            else:

                for t in [0.005, 0.001, 0.0005]:
                    f_trial = (1-t)*f + t*g_norm
                    f_trial = np.maximum(f_trial, 1e-15)

                    new_M = fast_convolve(f_trial).max()
                    if new_M < M - 1e-15:

                        break
