The constraint q >= 0 means that P is the DFT of a non-negative sequence. This is a strong constraint. By the uncertainty principle, a non-negative sequence that's concentrated in time (small support) has a spread-out spectrum, and vice versa.

Our sequence has full support (all n positions), so the spectrum is "concentrated" in the sense that it's the DFT of an n-length signal. The degree of freedom is in how we shape the spectrum.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initialization with parameterized families
2. FISTA (accelerated projected gradient) on log-sum-exp surrogate
   with adaptive temperature proportional to peak gap
3. L-BFGS polishing on the smooth surrogate
4. Nesterov subgradient descent with iterate averaging
5. Multi-length exploration
Key innovations: adaptive temperature based on top-peak gap,
FISTA with backtracking for fast convergence, and spectral
diagnostics to guide the optimization.

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective_and_grad(f, tau):
    """
    Compute F(f) = log(sum exp((f*f)/tau)) and its gradient.
    Returns (F_value, gradient, conv).
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    F_val = np.log(exp_vals.sum()) + scaled.max()  # = log(sum exp(conv/tau))
    
    # Gradient: d/df_j F = (2/tau) * sum_k w[k] * f[k-j] (with appropriate indexing)
    # (f*f)[k] = sum_j f[j] * f[k-j]
    # d(f*f)[k]/df[j] = f[k-j] + f[k-j] (if k-j != j) or 2*f[j] (if k-j=j)
    # Actually d(f*f)[k]/df[j] = f[k-j] (treating f as the first arg) + f[k-j] (treating as second)
    # Wait, (f*f)[k] = sum_i f[i]*f[k-i]. d/d(f[j]) = f[k-j] (from i=j) + (if k-j is in range and k-j != j) f[k-j]
    # Actually: d/d(f[j]) sum_i f[i]*f[k-i] = f[k-j] + sum_i delta_{j,k-i} f[i] = f[k-j] + f[k-j] if j is in range
    # Hmm, let me be more careful.
    # (f*f)[k] = sum_{i=0}^{n-1} f[i] * f[k-i] where f[m] = 0 for m not in [0, n-1]
    # d(f*f)[k]/df[j] = f[k-j] + f[k-j] = 2*f[k-j] (with f[m]=0 outside range)
    # Wait no. d/d(f[j]) of sum_i f[i]*f[k-i]:
    #   = sum_i (delta_{i,j} * f[k-i] + f[i] * delta_{k-i,j})
    #   = f[k-j] + f[k-j]  (the second term: k-i=j means i=k-j)
    #   = 2*f[k-j] (with f=0 outside [0,n-1])
    
    # So gradient[j] = (2/tau) * sum_k w[k] * 2 * f[k-j]... wait that's not right either.
    # Let me redo. F = sum_k w[k] * (f*f)[k] (approximately, when tau is small)
    # Actually F = log(sum_k exp((f*f)[k]/tau)), so dF/d(f[j]) = sum_k w[k] * (f*f)[k]_j / tau
    # where (f*f)[k]_j = d(f*f)[k]/df[j] = 2*f[k-j] (with appropriate boundary)
    
    # So gradient[j] = (2/tau) * sum_k w[k] * f[k-j]
    # This is exactly (2/tau) * (w * f)[j] where (w*f) is the cross-correlation
    # (w*f)[j] = sum_k w[k] * f[k-j]
    
    # Compute via FFT:
    # We want h[j] = sum_k w[k] * f[k-j] for j = 0, ..., n-1
    # This is the cross-correlation of w and f, evaluated at lags 0 to n-1.
    # h = ifft(fft(w) * conj(fft(f)))... let me think.
    # Actually, cross-correlation: (w ⊛ f)[j] = sum_k w[k] * f[k+j]
    # We want sum_k w[k] * f[k-j] = (w ⊛ f)[-j]... 
    # Let me just use: h[j] = sum_k w[k] * f[k-j]
    # Let rev_f = f[::-1], then h[j] = sum_k w[k] * f[k-j]
    # = conv(w, rev_f)[n-1-j]... 
    # Actually, let me use the convolution formula:
    # (w * rev_f)[m] = sum_k w[k] * rev_f[m-k] = sum_k w[k] * f[n-1-(m-k)]
    # = sum_k w[k] * f[n-1-m+k]
    # Let j = n-1-m, so m = n-1-j:
    # (w * rev_f)[n-1-j] = sum_k w[k] * f[n-1-(n-1-j)+k] = sum_k w[k] * f[j+k]
    # Hmm, that gives sum_k w[k] * f[j+k], not sum_k w[k] * f[k-j].
    
    # Let me try differently. Define:
    # h[j] = sum_k w[k] * f[k-j]
    # = sum_k w[k] * f[k-j]
    # Let l = k-j, so k = l+j:
    # = sum_l w[l+j] * f[l]
    # This is the cross-correlation of w (shifted by j) with f.
    # = (w_{j} ⊛ f)[0] where w_{j}[l] = w[l+j]
    
    # Alternatively: h[j] = sum_l f[l] * w[l+j]
    # This is the (j-th) cross-correlation coefficient.
    # In FFT terms: h = ifft(fft(w, N) * conj(fft(f, N))) ... no, that's not right either.
    
    # Let me just use the direct formula with FFT:
    # h[j] = sum_{l=0}^{n-1} f[l] * w[l+j] (where w is defined for l+j in [0, 2n-2])
    # This is the cross-correlation (f ⊛ w) at lag j.
    # Cross-correlation: (f ⊛ w)[j] = sum_l f[l] * w[l+j]
    # In FFT: (f ⊛ w) = ifft(fft(f) * conj(fft(w)))  [for the full cross-correlation]
    # We need the first n values of this.
    
    N = 2 * n - 1
    fft_size = 1 << (N - 1).bit_length()
    F = np.fft.fft(f, fft_size)
    W = np.fft.fft(w, fft_size)
    cross = np.fft.ifft(F * np.conj(W)).real
    h = cross[:n]  # cross-correlation at lags 0, 1, ..., n-1
    
    grad = (2.0 / tau) * h
    return F_val, grad, conv

def fista_smooth(f0, time_limit, tau_init=None, verbose=False):
    """
    FISTA on the log-sum-exp smooth surrogate, projected onto simplex.
    Adaptive temperature based on top-peak gap.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.08 * conv.max()
    
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    L = 1.0  # Lipschitz estimate
    
    # FISTA variables
    x = f.copy()
    y = f.copy()
    t_k = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive temperature based on peak gap
        conv_y = fast_convolve(y)
        top2 = np.sort(conv_y)[-2:]
        gap = top2[1] - top2[0]
        tau = max(1e-8, 0.05 * gap + 0.01 * conv_y.max())
        
        # Compute gradient at y
        F_y, grad_y, conv_y = smooth_objective_and_grad(y, tau)
        
        # Estimate Lipschitz constant
        # ||grad||_inf gives a rough bound
        L = max(L * 0.99, np.max(np.abs(grad_y)) * 1.1)
        
        # Proximal step
        x_new = project_simplex(y - grad_y / L)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        y_new = x_new + momentum * (x_new - x)
        
        x = x_new
        y = project_simplex(y_new)  # Ensure y stays on simplex
        t_k = t_new
        
        # Exact evaluation
        conv_x = fast_convolve(x)
        obj_x = conv_x.max()
        
        if obj_x < best_obj - 1e-14:
            best_obj = obj_x
            best_f = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 200 == 0:
            score = 2 * n * best_obj
            print(f"    FISTA iter {it}: score={score:.8f}, tau={tau:.2e}, L={L:.2e}, no_improve={no_improve}")
        
        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7 + 13)
            noise_scale = 0.002 + 0.003 * min(no_improve / 500.0, 2.0)
            x = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            y = x.copy()
            t_k = 1.0
            no_improve = 0
        
        if no_improve > 1000:
            break
    
    return best_f, 2 * n * best_obj

def lbfgs_smooth(f0, time_limit, verbose=False):
    """
    L-BFGS-B optimization on the smooth surrogate.
    Uses reduced parameterization: x[0..n-2] are free, x[n-1] = 1 - sum(x[0..n-2]).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    tau = 0.03 * fast_convolve(f).max()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    def obj_and_grad(z):
        # Reconstruct f from z (first n-1 elements are z, last is 1-sum(z))
        f_full = np.zeros(n)
        f_full[:n-1] = z
        f_full[n-1] = 1.0 - np.sum(z)
        f_full = np.maximum(f_full, 0.0)
        fs = f_full.sum()
        if fs < 1e-15:
            return 1e10, np.zeros(n-1)
        f_full = f_full / fs
        
        F_val, grad, conv = smooth_objective_and_grad(f_full, tau)
        
        # Chain rule: dF/dz[j] = dF/df[j] * df_full[j]/dz[j] - (dF/df[n-1] + sum_j dF/df[j] * df_full[j]/dz[j]) * (something)
        # This is complicated due to the normalization. Let me use a simpler approach.
        # Actually, let's just use the gradient directly on f and handle the constraint.
        # For L-BFGS-B, let's use bounds [0, 1] on all elements and handle sum constraint approximately.
        
        return F_val, grad
    
    # Actually, let me use a different approach: optimize directly on f with bounds,
    # and use the normalization as a soft constraint.
    
    def obj_f(f_vec):
        f_vec = np.maximum(f_vec, 0.0)
        fs = f_vec.sum()
        if fs < 1e-15:
            return 1e10
        f_n = f_vec / fs
        F_val, _, _ = smooth_objective_and_grad(f_n, tau)
        return F_val
    
    def grad_f(f_vec):
        f_vec = np.maximum(f_vec, 0.0)
        fs = f_vec.sum()
        if fs < 1e-15:
            return np.zeros(len(f_vec))
        f_n = f_vec / fs
        _, grad_n, _ = smooth_objective_and_grad(f_n, tau)
        # chain rule through normalization
        # f_n = f_vec / fs
        # df_n[i]/df_vec[j] = (delta_{ij} * fs - f_vec[i]) / fs^2 = (delta_{ij} - f_n[i]) / fs
        # dF/df_vec[j] = sum_i grad_n[i] * (delta_{ij} - f_n[i]) / fs
        # = (grad_n[j] - sum_i grad_n[i] * f_n[i]) / fs
        avg_grad = np.dot(grad_n, f_n)
        grad_f_vec = (grad_n - avg_grad) / fs
        return grad_f_vec
    
    x0 = f.copy()
    bounds = [(0.0, None)] * n
    
    result = minimize(
        obj_f, x0, jac=grad_f, method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
    )
    
    if result.success or result.fun < obj_f(x0):
        f_opt = np.maximum(result.x, 0.0)
        f_opt = f_opt / f_opt.sum()
        obj = fast_convolve(f_opt).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt.copy()
    
    if verbose:
        print(f"    L-BFGS: score={2*n*best_obj:.8f}")
    
    return best_f, 2 * n * best_obj

def nesterov_subgrad(f0, time_limit, verbose=False):
    """
    Nesterov-accelerated subgradient descent on the exact max objective.
    With iterate averaging for convergence.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Nesterov variables
    x = f.copy()
    y = f.copy()
    t_k = 1.0
    step = 1.0 / np.sqrt(n)
    avg_x = np.zeros(n)
    avg_count = 0
    
    # Warm start averaging
    avg_x = f.copy()
    avg_count = 1
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_max = int(np.argmax(conv_y))
        
        # Subgradient: d/d(f[j]) of (f*f)[k_max] = 2 * f[k_max - j] (with boundary)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        subgrad = np.zeros(n)
        subgrad[js:je+1] = 2.0 * f[k_max - je:k_max - js + 1][::-1]
        # Wait, let me recompute.
        # (f*f)[k] = sum_{j=0}^{n-1} f[j] * f[k-j] (with f[m]=0 outside [0,n-1])
        # d/d(f[i]) (f*f)[k] = f[k-i] + f[k-i] = 2*f[k-i] (with f=0 outside)
        # So subgrad[i] = 2*f[k_max - i] for i where 0 <= k_max-i <= n-1
        # i.e., k_max-(n-1) <= i <= k_max, and 0 <= i <= n-1
        # So i in [max(0, k_max-n+1), min(n-1, k_max)]
        # subgrad[i] = 2 * f[k_max - i]
        
        subgrad = np.zeros(n)
        for i in range(js, je + 1):
            subgrad[i] = 2.0 * f[k_max - i]
        
        # Step
        step = 1.0 / (np.sqrt(n) * np.sqrt(it + 1))
        x_new = project_simplex(y - step * subgrad)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        y_new = x_new + momentum * (x_new - x)
        
        x = x_new
        y = project_simplex(y_new)
        t_k = t_new
        
        # Averaging
        avg_x = (avg_count * avg_x + x) / (avg_count + 1)
        avg_count += 1
        
        # Evaluate
        conv_avg = fast_convolve(avg_x)
        obj_avg = conv_avg.max()
        
        if obj_avg < best_obj - 1e-14:
            best_obj = obj_avg
            best_f = avg_x.copy()
            if verbose and it % 100 == 0:
                print(f"    Subgrad iter {it}: score={2*n*best_obj:.8f}")
    
    return best_f, 2 * n * best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    """Asymmetric Gaussian."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_ramped(n, alpha=1.0):
    """Monotone ramp: p_j ∝ (j+1)^alpha."""
    a = (np.arange(n) + 1.0) ** alpha
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_exponential(n, rate=0.5):
    """Exponential: p_j ∝ exp(-rate * j / n)."""
    a = np.exp(-rate * np.arange(n) / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width_frac=0.5):
    """Flat-topped bell."""
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_two_bump(n, c1=0.3, c2=0.7, sigma=0.15, w1=0.6):
    """Two-bump Gaussian mixture."""
    x = np.arange(n) / n
    a = w1 * np.exp(-0.5 * ((x - c1) / sigma) ** 2) + (1-w1) * np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
