"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Starts from the parent sequence (n=768) and nearby lengths
2. Uses soft-max smoothed L-BFGS in log-space with multiple temperatures
3. Applies coordinate descent with pairwise moves
4. Uses simulated annealing for local exploration
5. Peak equalization heuristic to balance top peaks
The key difference from the previous approach: focuses on high-precision
local optimization with coordinate-level moves and peak equalization
rather than the LP-based fictitious play.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    # Gradient w.r.t. a: 2n * (w * a) convolved (cross-correlation)
    # Actually: d(max_k c_k)/da_i = 2 * sum_k w_k * a_{k-i}
    # Cross-correlation of w with a
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    # Project gradient to log-space (variance-preserving)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
        # Also check the z itself gave improvement
        a_check = softmax_from_z(res.x)
        sc = compute_score(a_check)
        if sc < result['score']:
            result['a'] = a_check
            result['score'] = sc
    except Exception as e:
        pass

    result['seconds'] = time.monotonic() - t0
    return result


def coordinate_descent(a0, time_limit, n_pairs_per_iter=200, verbose=False):
    """
    Coordinate descent using pairwise moves.
    For each pair (i,j), find the optimal transfer t that minimizes the score.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0

    # Precompute random pairs (biased towards center where action happens)
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        # Generate random pairs
        pairs = []
        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-12 and a[j] > 1e-12:
                pairs.append((i, j))
            elif i != j and a[i] > 1e-12:
                pairs.append((i, j))

        for (i, j) in pairs:
            if a[i] < 1e-14:
                continue

            # Try transferring a fraction of a[i] to a[j]
            # Binary/ternary search for optimal t
            # t > 0: move from i to j; t < 0: move from j to i
            best_t = 0.0
            best_local_score = best_score

            # Try several transfer sizes
            base = min(a[i], a[j]) if a[j] > 0 else a[i]
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 0.01, 0.05]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5

                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local_score - 1e-14:
                    best_local_score = sc
                    best_t = t

                # Also try negative transfer
                if a[j] > 1e-15:
                    t2 = frac * a[j]
                    a_trial2 = a.copy()
                    a_trial2[j] -= t2
                    a_trial2[i] += t2
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial2 = a_trial2 / ss2
                    sc2 = compute_score(a_trial2)
                    if sc2 < best_local_score - 1e-14:
                        best_local_score = sc2
                        best_t = -t2

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                ss = a.sum()
                a = a / ss
                best_score = best_local_score
                best_a = a.copy()
                improved = True

        if not improved and iters > 50:
            # Perturb to escape local minimum
            noise = rng.standard_normal(n) * 0.001
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 50 == 0:
            print(f"  CD iter={iters}, score={best_score:.8f}")

    return best_a, best_score


def simulated_annealing(a0, time_limit, n_moves_per_iter=100,
                        t_start=0.01, t_end=0.0001, verbose=False):
    """Simulated annealing with local moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    t0 = time.monotonic()
    total_time = time_limit
    rng = np.random.default_rng(999)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Temperature schedule (exponential cooling)
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        temp = t_start * (t_end / t_start) ** progress

        improved_this_iter = False
        for _ in range(n_moves_per_iter):
            if time.monotonic() - t0 > time_limit:
                break

            # Choose move type
            move_type = rng.random()
            a_trial = a.copy()

            if move_type < 0.4:
                # Single coordinate: scale a[i] and redistribute
                i = rng.integers(0, n)
                if a[i] < 1e-14:
                    continue
                factor = 1.0 + rng.standard_normal() * 0.1
                new_val = max(0, a[i] * factor)
                diff = a[i] - new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[i] = new_val
                a_trial[mask] = a[mask] + diff * a[mask] / others_sum

            elif move_type < 0.7:
                # Pairwise swap
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                t = rng.standard_normal() * 0.01 * max(a[i], a[j], 1e-6)
                a_trial[i] -= t
                a_trial[j] += t

            elif move_type < 0.85:
                # Block move: modify a contiguous block
                block_len = min(n, rng.integers(3, max(4, n // 50)))
                start = rng.integers(0, n - block_len + 1)
                idx = np.arange(start, start + block_len)
                noise = rng.standard_normal(block_len) * 0.01
                a_trial[idx] += noise
            else:
                # Two-point: adjust two coordinates
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                t1 = rng.standard_normal() * 0.02 * a[i]
                t2 = rng.standard_normal() * 0.02 * a[j] if a[j] > 0 else 0
                a_trial[i] -= t1
                a_trial[j] -= t2
                # Redistribute to keep sum
                redist = t1 + t2
                if abs(redist) > 1e-18:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    others = a[mask]
                    os = others.sum()
                    if os > 1e-15:
                        a_trial[mask] = others + redist * others / os

            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            trial_score = compute_score(a_trial)

            # Accept or reject
            delta = trial_score - current_score
            if delta < 0 or rng.random() < np.exp(-delta / temp):
                a = a_trial
                current_score = trial_score
                if trial_score < best_score - 1e-14:
                    best_score = trial_score
                    best_a = a_trial.copy()
                    improved_this_iter = True

        if verbose and iters % 100 == 0:
            print(f"  SA iter={iters}, temp={temp:.6f}, best={best_score:.8f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=10, verbose=False):

        min_peak = peak_values.min()

        max_idx = int(np.argmax(conv))

        je = min(n - 1, max_idx)

        improved = False

            i = idx[ci]
            j = max_idx - i
            if a[i] < 1e-14:
                continue

            # Try reducing a[i] and redistributing
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta

                others = a_trial[taski]
                os = others.sum()
                if os < 1e-15:

                a_trial[taski] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()

                break

        if not improved:

            if peak_positions:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def nesterov_gradient(a0, time_limit, T=300, lr_init=0.05, momentum=0.9):
    """Nesterov accelerated gradient descent in log-space."""
    n = len(a0)
    z = z_from_a(a0)
    v = np.zeros(n)
    best_z = z.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    lr = lr_init
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Nesterov lookahead
        z_lookahead = z - momentum * v
        a_la = softmax_from_z(z_lookahead)
        grad = smoothed_gradient(z_lookahead, T, n)

        v = momentum * v + lr * grad
        z_new = z - v

        # Project: ensure z doesn't go to extreme values
        z_new = np.clip(z_new, -20, 20)

        a_new = softmax_from_z(z_new)
        score = compute_score(a_new)

        if score < best_score - 1e-14:
            best_score = score
            best_z = z_new.copy()

        z = z_new
        lr *= 0.9998

    a_out = softmax_from_z(best_z)
    return a_out, best_score

        a_sa, sc_sa = simulated_annealing(best_seq, t_alloc, n_moves_per_iter=80,
                                          t_start=0.005, t_end=0.00005, verbose=True)
        best_seq, best_score = check_and_update(a_sa, "sim_anneal", best_seq, best_score)

    # === Phase 7: Try nearby lengths ===
    print(f"\n=== Phase 7: Nearby Lengths (best={best_score:.8f}) ===")
    nearby_ns = [n_parent - 1, n_parent + 1, n_parent - 2, n_parent + 2,
                 n_parent - 4, n_parent + 4]
    for alt_n in nearby_ns[:4]:
        if alt_n < 64 or alt_n > 2048:
            continue
        rem = deadline - time.monotonic() - 80
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 80.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  L-BFGS n={alt_n}, budget={t_alloc:.0f}s")
        res = lbgfs_optimize(init_alt, t_alloc, T=500, maxiter=3000)
        best_seq, best_score = check_and_update(res['a'], f"lbgfs_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final peak equalization + CD ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_alloc = min(rem * 0.4, 60.0)
        a_pe2, _ = peak_equalization(best_seq, t_alloc, n_peaks=20)
        best_seq, best_score = check_and_update(a_pe2, "final_pe", best_seq, best_score)

    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_alloc = min(rem * 0.5, 40.0)
        a_cd2, _ = coordinate_descent(best_seq, t_alloc, n_pairs_per_iter=500)
        best_seq, best_score = check_and_update(a_cd2, "final_cd", best_seq, best_score)

    # === Phase 9: Final L-BFGS ===
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem * 0.7, 30.0)
        res = lbgfs_optimize(best_seq, t_alloc, T=3000, maxiter=3000)
        best_seq, best_score = check_and_update(res['a'], "final_lbgfs", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
