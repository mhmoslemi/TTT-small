def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def symmetrize(a):
    """Make sequence palindromic and normalize to sum=1."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g

def subspace_peak_reduction(a0, time_limit, verbose=False, label=""):

    redistribution subproblem within that subspace.

        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False
        
        # Find top peaks

        peak_indices = np.argsort(conv)[-K:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            partners = k - idx
            products = a[idx] * a[partners]
            total = products.sum()
            if total < 1e-15:
                continue
            
            # Try reducing the top-contributing pair and redistributing
            top_pairs = np.argsort(products)[-3:]
            
            for pi in top_pairs:
                j = int(idx[pi])
                p = int(partners[pi])
                old_j = a[j]
                old_p = a[p]
                
                if old_j < 1e-12 and old_p < 1e-12:
                    continue
                
                # Try reducing both j and p
                for frac in [0.8, 0.9, 0.95]:
                    new_j = old_j * frac
                    new_p = old_p * frac
                    diff = (old_j - new_j) + (old_p - new_p)
                    if diff < 1e-14:
                        continue
                    
                    trial = a.copy()
                    trial[j] = new_j
                    if p != j:
                        trial[p] = new_p
                    else:
                        trial[j] = max(new_j, 0)
                    
                    # Redistribute to non-contributing positions
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[idx] = False
                    if non_contrib.sum() > 0:
                        nc_vals = trial[non_contrib]
                        nc_sum = nc_vals.sum()
                        if nc_sum > 1e-15:
                            trial[non_contrib] += diff * nc_vals / nc_sum
                    
                    trial = np.maximum(trial, 0.0)
                    ss = trial.sum()
                    if ss < 1e-10:
                        continue
                    trial = trial / ss
                    
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        a = trial
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Try broader approach: scale down all peak contributors
                k = int(np.argmax(conv))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx_range = np.arange(js, je + 1)
                
                for scale in [0.95, 0.9, 0.85, 0.8]:
                    trial = a.copy()
                    trial[idx_range] *= scale
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[idx_range] = False
                    if non_contrib.sum() > 0:
                        nc = trial[non_contrib]
                        nc_sum = nc.sum()
                        if nc_sum > 1e-15:

                            if deficit > 0:

                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                
                if not improved:
                    break
        else:
            no_improve = 0
    
    return best_a, best_obj

        # So a[j] contributes to conv[j:j+n]

    while time.monotonic() - t0 < time_limit:

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_raised_cosine(n):
    """Raised cosine window."""
    x = np.arange(n) - n / 2.0
    t = np.abs(x) / (n / 2.0)
    a = np.cos(np.pi * t / 2.0) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_optimal_shape(n, width=0.4):

def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        rounds += 1
        improved_any = False

            # Search range: [0, min(0.5, old_val * 3)]

def make_gaussian(n, sigma_frac=0.3):

    t = np.abs(x) / (n / 2.0)
    a = np.cos(np.pi * t / 2.0) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_optimal_parametric(n, width_frac=0.4, power=3):
    """Parametric family: (1 - (2x/n - 1)^2/w^2)^power for |2x/n-1| < w."""
    x = np.arange(n) - n / 2.0
    t = np.abs(x) / (n * width_frac / 2.0)
    a = np.zeros(n)
    mask = t < 1.0
    a[mask] = (1.0 - t[mask] ** 2) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, a)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        a = np.asarray(a, dtype=np.float64)

        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_gaussian(768, 0.3)
