           # Rebuild constraint matrix (a has changed)
           A = build_A(a)
           
           # Solve LP: max sum(b) s.t. A@b <= M, b >= 0
           c = -np.ones(n)
           b_ub = np.full(2 * n - 1, M)
           try:
               res = linprog(c, A_ub=A, b_ub=b_ub,
                            bounds=[(0, None)] * n, method='highs',
                            options={'presolve': True, 'time_limit': 10})
           except Exception:
               break
           
           if not res.success:
               break
           
           b = res.x
           b_sum = float(b.sum())
           
           if b_sum <= 1.0 + 1e-9:
               # At (or near) local optimum
               break
           
           g = b / b_sum
           g = np.maximum(g, 0.0)
           g_sum = g.sum()
           if g_sum < 1e-15:
               break
           g = g / g_sum
           
           # Line search
           best_t = 0.0
           best_trial = M
           for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
               trial = (1 - t) * a + t * g
               trial = np.maximum(trial, 0.0)
               ts = trial.sum()
               if ts < 1e-15:
                   continue
               trial = trial / ts
               trial_obj = fast_convolve(trial).max()
               if trial_obj < best_trial:
                   best_trial = trial_obj
                   best_t = t
           
           if best_t == 0.0:
               no_improve += 5
               continue
           
           a = (1 - best_t) * a + best_t * g
           a = np.maximum(a, 0.0)
           a = a / a.sum()
       
       return best_a, best_obj
   
   
   def subgradient_polish(a_init, time_limit, top_k=20, verbose=False, label=""):
       """Subgradient-based polish for fine-tuning."""
       a = np.maximum(a_init.astype(np.float64), 1e-15)
       a = a / a.sum()
       n = len(a)
       
       conv = fast_convolve(a)
       best_obj = conv.max()
       best_a = a.copy()
       
       t0 = time.monotonic()
       lr = 0.01
       no_improve = 0
       it = 0
       
       while time.monotonic() - t0 < time_limit:
           it += 1
           conv = fast_convolve(a)
           M = conv.max()
           
           if M < best_obj - 1e-15:
               best_obj = M
               best_a = a.copy()
               no_improve = 0
           else:
               no_improve += 1
           
           if no_improve > 500:
               break
           
           # Find top peaks
           K = min(top_k, len(conv))
           if K < len(conv):
               peak_idx = np.argpartition(conv, -K)[-K:]
           else:
               peak_idx = np.arange(len(conv))
           
           # Softmax weights
           pv = conv[peak_idx]
           scale = 0.001
           scaled = pv / scale - np.max(pv) / scale
           w = np.exp(scaled)
           w /= w.sum()
           
           # Compute subgradient
           g = np.zeros(n)
           for idx_i in range(len(peak_idx)):
               k = int(peak_idx[idx_i])
               wi = w[idx_i]
               if wi < 1e-15:
                   continue
               js = max(0, k - n + 1)
               je = min(n - 1, k)
               if je < js:
                   continue
               g[js:je + 1] += 2.0 * wi * a[k - np.arange(js, je + 1)]
           
           # Multiplicative update
           log_update = -lr * g
           log_update = np.clip(log_update, -50, 50)
           a_new = a * np.exp(log_update)
           a_new = np.maximum(a_new, 1e-15)
           a_new = a_new / a_new.sum()
           
           # Check improvement
           new_obj = fast_convolve(a_new).max()
           if new_obj < M:
               a = a_new
           else:
               # Try smaller step
               for lr_try in [lr * 0.5, lr * 0.1]:
                   log_update = -lr_try * g
                   log_update = np.clip(log_update, -50, 50)
                   a_try = a * np.exp(log_update)
                   a_try = np.maximum(a_try, 1e-15)
                   a_try = a_try / a_try.sum()
                   if fast_convolve(a_try).max() < M:
                       a = a_try
                       lr = lr_try
                       break
               else:
                   lr *= 0.5
           
           if no_improve > 100:
               lr = max(lr * 0.5, 1e-6)
               
           if lr < 1e-8:
               break
       
       return best_a, best_obj
   
   
   def make_gaussian(n, sigma_frac=0.3):
       x = np.arange(n) - n / 2.0
       sigma = n * sigma_frac
       a = np.exp(-0.5 * (x / sigma) ** 2)
       a = np.maximum(a, 0)
       s = a.sum()
       if s > 1e-15:
           a /= s
       return a
   
   
   def make_tukey(n, r=0.5):
       """Tukey (tapered cosinus) window."""
       a = np.ones(n)
       m = int(n * r / 2)
       if m > 0:
           x = np.arange(m) / m
           taper = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
           a[:m] = taper
           a[-m:] = taper[::-1]
       s = a.sum()
       if s > 1e-15:
           a /= s
       return a
   
   
   def make_bell(n, width_frac=0.6):
       x = np.arange(n) - n / 2.0
       half_width = n * width_frac / 2.0
       t = np.clip(np.abs(x) / half_width, 0, 1.0)
       a = 1.0 - t ** 4
       a = np.maximum(a, 0)
       s = a.sum()
       if s > 1e-15:
           a /= s
       return a
   
   
   def symmetrize(a):
       a_sym = 0.5 * (a + a[::-1])
       a_sym = np.maximum(a_sym, 0.0)
       ss = a_sym.sum()
       if ss > 1e-15:
           a_sym /= ss
       return a_sym
