"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
with active-set constraints, adaptive step sizes, diverse initializations,
multi-length exploration, and final polishing via coordinate descent.

The core idea: at each iteration, solve an LP to find the best non-negative
direction d such that the cross-convolution (a*d) stays below the current
peak M. Taking a step in this direction guarantees (for small enough step)
that the peak of the auto-convolution does not increase. The active-set
variant only constrains peaks near the maximum, allowing larger steps.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective value 2n*max(conv)/(sum)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (S * S))


def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix M such that M @ b = convolve(a, b)."""
    n = len(a)
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    M = np.zeros((2 * n - 1, n))
    M[mask] = a[idx[mask]]
    return M


def solve_direction_lp(a, M, active_indices=None):
    """
    Solve LP: max sum(d) s.t. d >= 0, (a*d)(k) <= M for k in active set.
    Returns the optimal direction d (sum-normalized) or None.
    """
    n = len(a)
    if active_indices is None:
        active_indices = np.arange(2 * n - 1)
    
    num_constraints = len(active_indices)
    if num_constraints == 0:
        return np.ones(n) / n
    
    # Build constraint matrix for only active indices
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    M_mat = np.zeros((num_constraints, n))
    for i, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            M_mat[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    
    try:
        res = linprog(c_obj, A_ub=M_mat, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            d = np.maximum(res.x, 0)
            d_sum = d.sum()
            if d_sum > 1e-12:
                return d / d_sum
    except Exception:
        pass
    return None

    else:
        a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Process in mini-batches for efficiency
        batch = min(20, max(1, int(time_limit * 50)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            c = fast_convolve(a)
            M = c.max()
            peak_idx = c.argmax()
            
            # Subgradient: cross-correlation at peak
            g = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            g[lo:hi + 1] = a[peak_idx - hi:peak_idx - lo + 1][::-1]
            
            # Nesterov momentum step
            v = momentum * v - cur_lr * g
            a_new = np.maximum(a + v, 0)
            a_new = a_new / a_new.sum()
            
            M_new = fast_convolve(a_new).max()
            if M_new < best_M - 1e-14:
                best_M = M_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        
        if no_improve > 500:
            cur_lr *= 0.8
            if no_improve > 2000:
                a = np.maximum(best_a + rng.standard_normal(n) * 0.005, 0)
                a = a / a.sum()
                v = np.zeros(n)
                no_improve = 0
                cur_lr = lr_init
    
    return best_a, best_M


def make_gaussian_init(n, sigma=0.3, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bimodal_init(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.25) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.75) / 0.15) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_triangle_init(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - 2.0 * np.abs(x - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_polish(a0, time_limit):
    """Coordinate descent: try scaling individual elements."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.6])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(0, n, max(1, n // 500)):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_M
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                c_trial = fast_convolve(a_trial)
                if c_trial.max() < best_j_obj - 1e-14:
                    best_j_obj = c_trial.max()
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_j_obj < best_M - 1e-14:
                    best_M = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_M


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization using LP-based direction finding with
    active-set constraints, diverse initializations, and multi-length search.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(12345)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(1152) / 1152
        print("No parent, using uniform n=1152")
        p_score = float(evaluate_sequence(parent.tolist()))
    
    n_parent = len(parent)
    
    # === Phase 1: LP optimization from parent (most of the time) ===
    print(f"\n=== Phase 1: LP optimization from parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 500.0)
        print(f"  Allocating {t_alloc:.0f}s")
        a_lp, _ = lp_optimize(parent, time_limit=t_alloc,
                              active_thresh=0.02, seed=42)
        try_seq(a_lp, f"lp_parent_n{n_parent}")
    
    # === Phase 2: More LP with tighter active set ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 20:
            t_alloc = min(rem, 200.0)
            print(f"\n=== Phase 2: Tighter LP (n={len(best_seq)}) ===")
            print(f"  Allocating {t_alloc:.0f}s")
            a_lp2, _ = lp_optimize(best_seq, time_limit=t_alloc,
                                   active_thresh=0.005, seed=99)
            try_seq(a_lp2, f"lp_tight_n{len(best_seq)}")
    
    # === Phase 3: Diverse initializations ===
    print(f"\n=== Phase 3: Diverse initializations (best={best_score:.6f}) ===")
    n_base = len(best_seq) if best_seq is not None else n_parent
    
    initializations = {}
    initializations["gauss_0.2"] = make_gaussian_init(n_base, 0.2)
    initializations["gauss_0.35"] = make_gaussian_init(n_base, 0.35)
    initializations["gauss_0.5"] = make_gaussian_init(n_base, 0.5)
    initializations["cosine"] = make_cosine_init(n_base)
    initializations["bimodal"] = make_bimodal_init(n_base)
    initializations["triangle"] = make_triangle_init(n_base)
    initializations["uniform"] = make_uniform_init(n_base)
    
    for name, init_a in initializations.items():
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(rem * 0.12, 35.0)
        if t_alloc < 5:
            continue
        print(f"  {name}: {t_alloc:.0f}s")
        a_opt, _ = lp_optimize(init_a, time_limit=t_alloc,
                               active_thresh=0.03, seed=hash(name) % 10000)
        try_seq(a_opt, f"diverse_{name}")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration (best={best_score:.6f}) ===")
    base_good = best_seq if best_seq is not None else parent
    target_lengths = []
    for factor in [0.75, 1.0, 1.25, 1.5, 0.6, 2.0]:
        tl = int(round(n_parent * factor))
        tl = max(256, min(4096, tl))
        if tl != n_parent and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_parent))
    
    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        f_interp = interpolate_sequence(base_good, target_n)
        t_alloc = min(rem * 0.25, 50.0)

                               active_thresh=0.02, seed=target_n % 10000)

    print(f"\n=== Phase 5: Deep LP on best (score={best_score:.6f}) ===")

    if best_seq is not None:

        if rem > 10:

            print(f"\n=== Phase 7: LP after SG ===")
            a_lp3, _ = lp_optimize(best_seq, time_limit=t_alloc,
                                   active_thresh=0.003, seed=88888)
            try_seq(a_lp3, "lp_after_sg")
    
    # === Phase 8: Coordinate descent ===
    print(f"\n=== Phase 8: Coordinate descent (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd = min(rem, 45.0)
            print(f"  CD: {t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd_polish")
    
    # === Phase 9: Final micro LP ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
