    """Compute 2*n*max_c for a normalized sequence (sum=1)."""
    n = len(f)
    c = fast_convolve(f)
    return 2.0 * n * c.max()


def get_peak_gradient(f, top_k, temperature):
    """Compute smoothed gradient of max using top-k peaks with softmax weighting."""
    n = len(f)
    c = fast_convolve(f)
    len_c = len(c)
    k_top = min(top_k, len_c)
    peaks = np.argpartition(c, -k_top)[-k_top:]
    pv = c[peaks]
    scaled = temperature * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        grad[js:je + 1] += w[i] * contrib
    return grad, c


def mirror_descent_improved(f0, time_limit, top_k=20, tau_init=10.0,
                            tau_max=500.0, lr_init=0.005, lr_min=1e-7,
                            lr_decay=0.9999, tau_decay=1.005,
                            restart_every=150, verbose=False):
    """
    Improved mirror descent with backtracking line search and adaptive
    temperature. Works in log-space via multiplicative updates.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0
    consecutive_rejects = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()

        # Compute smoothed gradient
        grad, c = get_peak_gradient(f, top_k, tau)

        # Multiplicative update
        f_new = f * np.exp(-lr * grad)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        # Backtracking line search
        obj_new = fast_convolve(f_new).max()
        accepted = False
        if obj_new < M:
            f = f_new
            accepted = True
        else:
            # Try damped versions
            for damp in [0.5, 0.3, 0.15, 0.07]:
                f_trial = (1.0 - damp) * f + damp * f_new
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f = f_trial
                    f_new = f_trial
                    obj_new = obj_trial
                    accepted = True
                    break
            if not accepted:
                lr *= 0.5
                consecutive_rejects += 1
                if consecutive_rejects > 20:
                    # Perturb and restart
                    rng = np.random.default_rng(it * 13 + 7)
                    f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
                    f = f / f.sum()
                    lr = lr_init * 0.5
                    tau = max(tau_init, tau * 0.5)
                    consecutive_rejects = 0
                continue
        consecutive_rejects = 0

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1

        # Adapt parameters
        tau = min(tau * tau_decay, tau_max)
        lr = max(lr * lr_decay, lr_min)

        # Periodic restart
        if no_improve > restart_every:
            rng = np.random.default_rng(it * 31 + 17)
            noise_scale = 0.002 + 0.003 * min(no_improve / restart_every, 2.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = max(lr_init * 0.4, lr_min)
            tau = tau_init
            no_improve = 0

        if lr < lr_min:
            break

    return best_f, best_obj


def solve_lp_direction_tight(f, M, n, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_improved(f0, time_limit, verbose=False):
    """
    Improved LP-Frank-Wolfe with tighter constraints and better line search.
    Uses M = current_max * (1 + epsilon) with small epsilon for tighter LP.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    epsilon = 0.01  # Tight constraint: M * (1 + epsilon)

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()

        # Solve LP with tight constraint
        M_lp = M * (1.0 + epsilon)
        g = solve_lp_direction_tight(f, M_lp, n)

        if g is None:
            epsilon *= 1.5
            if epsilon > 5.0:
                rng = np.random.default_rng(it * 19 + 3)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 0.0)
                f = f / f.sum()
                epsilon = 0.01
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            epsilon *= 1.2
            continue
        g = g / g_sum
        epsilon = max(epsilon * 0.9, 0.005)

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        else:
            no_improve = 0

        # Goldstein line search
        best_alpha = 0.0
        best_val = M
        alphas = np.concatenate([np.linspace(0.05, 0.5, 10),
                                 np.linspace(0.5, 1.0, 11)])
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Fine goldstein section
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, eps={epsilon:.4f}")
        else:
            no_improve += 1
            epsilon = min(epsilon * 1.2, 5.0)

        if no_improve > 100:
            break

    return best_f, best_obj


def direct_peak_shave(f0, time_limit, verbose=False):
    """
    Aggressively target the single worst peak. Compute exact gradient at
    the peak position and take large multiplicative steps.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        k_star = int(np.argmax(c))

        # Gradient at the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[js:je + 1] = f[k_star - idx] + f[idx]

        # Also add small weight to 2nd and 3rd peaks
        sorted_peaks = np.argsort(c)[-4:]
        for pk in sorted_peaks[1:]:
            pk = int(pk)
            js2 = max(0, pk - n + 1)
            je2 = min(n - 1, pk)
            w = c[pk] / M * 0.3
            idx2 = np.arange(js2, je2 + 1)
            grad[js2:je2 + 1] += w * (f[pk - idx2] + f[idx2])

        # Multiplicative step with line search
        f_new = f * np.exp(-lr * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    DPS iter {it}: obj={best_obj:.10f}, lr={lr:.5f}")
            lr = min(lr * 1.03, 0.08)
        else:
            # Try smaller step
            accepted = False
            for factor in [0.5, 0.25, 0.1]:
                f_trial = f * np.exp(-lr * factor * grad)
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f = f_trial
                    obj_new = obj_trial
                    accepted = True
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_f = f.copy()
                        no_improve = 0
                    break
            if not accepted:
                lr *= 0.7
                no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 43 + 23)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.02
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, try to find a better value
    by 1D optimization. Uses the fact that changing one coordinate
    affects only a subset of convolution values.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        rng = np.random.default_rng(it * 1000 + 42)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-14:
                continue

            # Try a range of multipliers
            best_j_obj = best_obj
            best_j_mult = 1.0

            for mult in [0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 3.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 2.0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-12:
                    best_j_obj = val
                    best_j_mult = mult

            if best_j_mult != 1.0:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    new_val = old_val * best_j_mult
                    scale = (1.0 - new_val) / denom
                    f[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            # Try adding mass to zero coordinates
            zero_coords = np.where(f < 1e-10)[0]
            if len(zero_coords) > 0:
                for j in zero_coords[:20]:
                    for small_val in [0.001, 0.005, 0.01]:
                        f_trial = f.copy()
                        f_trial[j] = small_val
                        scale = (1.0 - small_val) / max(f.sum() - small_val, 1e-15)
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = np.maximum(f_trial, 0)
                        ss = f_trial.sum()
                        if ss < 1e-15:
                            continue
                        f_trial = f_trial / ss
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-12:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            improved = True
                            break
            if not improved:
                break

    return best_f, best_f_obj if False else best_obj


def pair_swap_optimize(f0, time_limit, verbose=False):
    """
    Try swapping mass between pairs of coordinates. This can find
    improvements that single-coordinate changes miss.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(min(5000, max(100, int((time_limit - (time.monotonic() - t0)) * 50)))):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            # Pick random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if f[i] < 1e-10 or f[j] < 1e-10:
                continue

            # Try transferring mass from i to j (or vice versa)
            for direction in [1, -1]:
                for frac in [0.1, 0.25, 0.5]:
                    delta = direction * f[min(i, j)] * frac
                    new_i = f[i] + (delta if direction == 1 else 0)
                    new_j = f[j] - (delta if direction == 1 else 0) if direction == 1 else f[j] + delta

                    # Simpler: just try symmetric swap
                    t = f[i] * frac
                    f_trial = f.copy()
                    f_trial[i] = f[i] - t
                    f_trial[j] = f[j] + t
                    if f_trial[i] < 0 or f_trial[j] < 0:
                        continue
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-12:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break

                if improved:
                    break

        if not improved:
            break

    return best_f, best_obj


def make_init_asymmetric(n, variant=0, seed=42):
    """Generate various asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 137)
    x = np.arange(n) / n

    if variant == 0:
        a = np.exp(-x / 0.1)
    elif variant == 1:
        a = np.exp(-(1 - x) / 0.1)
    elif variant == 2:
        a = np.exp(-x / 0.15) * (1 + 0.3 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif variant == 3:
        a = np.exp(-x / 0.2)
    elif variant == 4:
        a = (1 - x) ** 2
    elif variant == 5:
        a = np.exp(-x / 0.08)
    elif variant == 6:
        a = np.exp(-x / 0.25)
    elif variant == 7:
        a = (1 - x) ** 3
    elif variant == 8:
        a = np.exp(-x / 0.12) + 0.1 * np.exp(-(1 - x) / 0.3)
    elif variant == 9:
        # Smooth bump off-center
        a = np.exp(-20 * (x - 0.2) ** 2) + 0.3 * np.exp(-20 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Evaluate parent and generate diverse initializations
    2. Run improved mirror descent on best candidates (multiple lengths)
    3. LP-FW refinement
    4. Direct peak shaving
    5. Pair swap optimization
    6. Coordinate descent polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Phase 1: Generate and evaluate initializations ===
    print("\n=== Phase 1: Initializations ===")
    test_lengths = [512, 640, 768, 896, 1024]
    init_pool = []

    for n_try in test_lengths:
        for v in range(10):
            a_init = make_init_asymmetric(n_try, variant=v, seed=seed)
            obj = compute_objective_norm(a_init)
            init_pool.append((obj, a_init, f"init_{v}_n{n_try}"))

    # Also add parent
    if parent is not None:
        obj_p = compute_objective_norm(parent)
        init_pool.append((obj_p, parent, "parent"))

    init_pool.sort(key=lambda x: x[0])
    print(f"  Top 5 inits: {['%s:%.4f' % (l, o) for o, _, l in init_pool[:5]]}")

    # === Phase 2: Mirror descent on top candidates at multiple lengths ===
    print("\n=== Phase 2: Mirror Descent ===")
    # Select top candidates ensuring diversity in length
    selected = []
    selected_lengths = set()
    for obj, a, label in init_pool:
        n = len(a)
        if n not in selected_lengths:
            selected.append((obj, a, label))
            selected_lengths.add(n)
        if len(selected) >= 4:
            break
    # Also always include the best overall
    if init_pool[0] not in selected:
        selected.insert(0, init_pool[0])

    for idx, (obj_init, a_init, label) in enumerate(selected[:4]):
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        t_alloc = min(rem / max(1, len(selected) - idx - 1), 180)
        t_alloc = max(t_alloc, 20)
        print(f"  MD on {label} (n={len(a_init)}, obj_init={obj_init:.6f}): t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent_improved(a_init, time_limit=t_alloc,
                                             top_k=20, tau_init=8.0,
                                             lr_init=0.004, verbose=True)
        try_seq(a_opt, f"md_{label}")

    # === Phase 3: Direct peak shaving on current best ===
    print("\n=== Phase 3: Direct Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  DPS on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = direct_peak_shave(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "dps")

    # === Phase 4: LP-FW refinement ===
    print("\n=== Phase 4: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 40:
            t_alloc = min(rem, 150)
            print(f"  LP-FW on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_improved(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        base_n = len(base)
        target_lengths = [512, 640, 768, 896, 1024, 1152]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 200
            if rem < 50:
                break
            if target_n == base_n:
                continue
            f_interp = interpolate_seq(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.3, 60)
                if t_alloc < 15:
                    t_alloc = 15
                print(f"  MD n={target_n}: t={t_alloc:.0f}s")
                a_opt, obj = mirror_descent_improved(f_interp, time_limit=t_alloc,
                                                     top_k=15, tau_init=5.0,
                                                     lr_init=0.003)
                try_seq(a_opt, f"md_n{target_n}")

    # === Phase 6: Second round of MD with different parameters ===
    print("\n=== Phase 6: MD Round 2 (different params) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  MD2 (low tau, high lr): t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_improved(best_seq, time_limit=t_alloc,
                                                 top_k=10, tau_init=30.0,
                                                 tau_max=200.0,
                                                 lr_init=0.008,
                                                 restart_every=80)
            try_seq(a_opt, "md2")

    # === Phase 7: Peak shaving round 2 ===
    print("\n=== Phase 7: DPS Round 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 80)
            print(f"  DPS2: t={t_alloc:.0f}s")
            a_opt, obj = direct_peak_shave(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "dps2")

    # === Phase 8: Pair swap optimization ===
    print("\n=== Phase 8: Pair Swap ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Pair swap: t={t_alloc:.0f}s")
            a_opt, obj = pair_swap_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "swap")

    # === Phase 9: LP-FW final ===
    print("\n=== Phase 9: Final LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 50)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_improved(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

    # === Phase 10: Final MD squeeze ===
    print("\n=== Phase 10: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem, 30)
            print(f"  Final MD: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_improved(best_seq, time_limit=t_alloc,
                                                 top_k=5, tau_init=100.0,
                                                 tau_max=500.0,
                                                 lr_init=0.002,
                                                 restart_every=50)
            try_seq(a_opt, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
