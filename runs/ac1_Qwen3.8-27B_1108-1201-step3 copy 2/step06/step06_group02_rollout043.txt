```
    if n is None: n=len(f)

    if active is None:

    m=len(active)

    A = np.zeros((m, n+1), dtype=np.float64)
    A[:,0] = -1.0
    for r,k in enumerate(active):
        idx1, part = info[int(k)]
        if idx1.size>0:

    A_eq = np.zeros((1,n+1)); A_eq[0,1:] = 1.0

    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options=options)
```

```python
"""
Searches for low-peak linear self-convolution nonnegative sequences.

The score is
    2 * n * max_k (a * a)[k] / (sum a)^2
with a >= 0.

The optimizer works on the simplex sum(a)=1 and uses:
  * fast FFT convolution for objective evaluations,
  * an LP-Frank-Wolfe style step: for fixed a, solve the linear program
        min t  s.t.  (a*g)[k] <= t for all (or a large active subset of all) k,
                     sum(g)=1, g>=0,
  * a smooth log-sum-exp subgradient mirror-descent step,
  * random pairwise mass moves and support perturbations for escaping local minima,
  * occasional multi-start and length exploration.
"""

import time
import numpy as np
from scipy.optimize import linprog

_FFT_SIZE_CACHE = {}


def fft_size_for_convolution(n):
    key = int(n)
    if key not in _FFT_SIZE_CACHE:
        needed = 2 * key - 1
        _FFT_SIZE_CACHE[key] = 1 << (max(1, needed - 1)).bit_length()
    return _FFT_SIZE_CACHE[key]


def fast_convolve(a):
    """Linear self-convolution by FFT, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    out_len = 2 * n - 1
    size = fft_size_for_convolution(n)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)[:out_len]
    # FFT roundoff can create tiny negative values.
    np.maximum(c, 0.0, out=c)
    return c


def fast_convolution_max(a):
    """Only max coefficient of linear self-convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    out_len = 2 * n - 1
    size = fft_size_for_convolution(n)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)[:out_len]
    # max already ignores tiny negative values, but ensure finite.
    if not np.all(np.isfinite(c)):
        return float("inf")
    return float(np.max(c))


def objective_normalized(a):
    """Score for a nonnegative sequence with sum exactly 1."""
    a = np.asarray(a, dtype=np.float64)
    if len(a) == 0:
        return float("inf")
    m = fast_convolution_max(a)
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * len(a) * m)


def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return np.array([1.0])
    a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return out / s


def softmax_peak_weights(conv, beta):
    """Weights peaked at largest convolution coefficients."""
    conv = np.asarray(conv, dtype=np.float64)
    if conv.size == 0:
        return np.array([])
    M = float(np.max(conv))
    if M <= 0.0 or not np.isfinite(M):
        w = np.zeros(conv.size, dtype=np.float64)
        w[int(np.argmax(conv))] = 1.0
        return w
    x = -float(beta) * (M - conv)
    # If beta is too large, many exponentials underflow; that is okay.
    w = np.exp(x)
    if not np.all(np.isfinite(w)):
        w = np.where(np.isfinite(w) & (w > 0.0), w, 0.0)
    ws = float(np.sum(w))
    if ws <= 1e-300:
        w = np.zeros(conv.size, dtype=np.float64)
        w[int(np.argmax(conv))] = 1.0
        return w
    return w / ws


def weighted_subgradient(f, weights, n):
    """
    For weights w_k on convolution coefficients, compute
        grad_i = 2 * sum_k w_k f_{k-i}
    using one FFT-based cross-correlation.
    """
    f = np.asarray(f, dtype=np.float64)
    weights = np.asarray(weights, dtype=np.float64)
    L = 2 * n - 1
    if weights.size != L:
        # Pad or truncate defensively.
        tmp = np.zeros(L, dtype=np.float64)
        m = min(L, weights.size)
        tmp[:m] = weights[:m]
        weights = tmp
    if float(np.sum(weights)) <= 1e-300:
        return np.zeros(n, dtype=np.float64)

    # R[m] = weights[L-1-m]
    R = weights[::-1]
    full_len = n + L - 1
    size = 1 << (max(1, full_len - 1)).bit_length()
    F = np.fft.rfft(f, size)
    RF = np.fft.rfft(R, size)
    C = np.fft.irfft(F * RF, size)[:full_len]

    # Need C[L-n : L] reversed.
    seg = C[L - n:L]
    sg = 2.0 * seg[::-1]
    np.maximum(sg, 0.0, out=sg)
    return sg


def build_convolve_rows_info(n):
    """
    Precompute, for each convolution index k, the variable indices
    (offset by 1 because t is variable 0 in LP) and f indices.
    """
    info = []
    L = 2 * n - 1
    for k in range(L):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1, dtype=np.int32)
        part = np.asarray(k - idx, dtype=np.int32)
        # Variable indices for g are 1-based because variable 0 is t.
        info.append((idx + 1, part))
    return info


def solve_lp_direction(f, info, active=None, n=None, lp_time=1.0):
    """
    Solve min t s.t. (f*g)[k] <= t for selected k, sum(g)=1, g>=0, t>=0.
    Returns (g, t). g is normalized to sum 1.
    """
    f = np.asarray(f, dtype=np.float64)
    if n is None:
        n = len(f)
    L = 2 * n - 1
    if active is None:
        active = np.arange(L, dtype=np.int64)
    active = np.asarray(active, dtype=np.int64)
    m = int(active.size)
    if m == 0:
        return None, float("inf")

    A_ub = np.zeros((m, n + 1), dtype=np.float64)
    A_ub[:, 0] = -1.0
    b_ub = np.zeros(m, dtype=np.float64)

    for r, k in enumerate(active):
        k = int(k)
        if 0 <= k < L:
            idx1, part = info[k]
            if idx1.size > 0:
                A_ub[r, idx1] = f[part]

    c = np.zeros(n + 1, dtype=np.float64)
    c[0] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * (n + 1)
    options = {"time_limit": float(max(0.2, min(5.0, lp_time)))}

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options=options,
        )
    except Exception:
        return None, float("inf")

    if not res.success:
        return None, float("inf")

    x = np.asarray(res.x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return None, float("inf")

    t = float(x[0])
    g = x[1:]
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    s = float(np.sum(g))
    if s <= 1e-15:
        return None, float("inf")
    g = g / s
    return g, t


def trial_from_alpha(f, d, a):
    trial = f + a * d
    if not np.all(np.isfinite(trial)):
        return None, float("inf")
    trial = np.maximum(trial, 0.0)
    s = float(np.sum(trial))
    if s <= 1e-15:
        return None, float("inf")
    trial = trial / s
    return trial, objective_normalized(trial)


def trial_from_log(f, logf, sg, step):
    trial = np.exp(logf - step * sg)
    if not np.all(np.isfinite(trial)):
        return None, float("inf")
    trial = np.maximum(trial, 0.0)
    s = float(np.sum(trial))
    if s <= 1e-15:
        return None, float("inf")
    trial = trial / s
    return trial, objective_normalized(trial)


def line_search_direction(f, g, current_best=None, max_golden=10):
    """Line search on f + alpha*(g-f), alpha in [0,1]."""
    f = np.asarray(f, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    if current_best is None:
        current_best = objective_normalized(f)

    d = g - f
    if not np.all(np.isfinite(d)) or float(np.max(np.abs(d))) < 1e-12:
        return f.copy(), current_best

    best_score = float(current_best)
    best_trial = f.copy()
    best_alpha = 0.0
    best_grid_idx = -1

    grid = [0.01, 0.03, 0.06, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for i, a in enumerate(grid):
        trial, sc = trial_from_alpha(f, d, a)
        if trial is None:
            continue
        if sc < best_score:
            best_score = sc
            best_trial = trial.copy()
            best_alpha = a
            best_grid_idx = i

    if best_grid_idx >= 0:
        if best_grid_idx == 0:
            lo, hi = 0.0, grid[1]
        elif best_grid_idx == len(grid) - 1:
            lo, hi = grid[-2], 1.0
        else:
            lo, hi = grid[best_grid_idx - 1], grid[best_grid_idx + 1]

        if hi > lo + 1e-15:
            gr = 0.618
            for _ in range(max_golden):
                a1 = hi - gr * (hi - lo)
                a2 = lo + gr * (hi - lo)
                if a1 <= lo + 1e-14 or a2 >= hi - 1e-14:
                    break
                t1, s1 = trial_from_alpha(f, d, a1)
                t2, s2 = trial_from_alpha(f, d, a2)
                if t1 is not None and s1 < best_score:
                    best_score = s1
                    best_trial = t1.copy()
                if t2 is not None and s2 < best_score:
                    best_score = s2
                    best_trial = t2.copy()
                if s1 < s2:
                    hi = a2
                else:
                    lo = a1

            mid = 0.5 * (lo + hi)
            tm, sm = trial_from_alpha(f, d, mid)
            if tm is not None and sm < best_score:
                best_score = sm
                best_trial = tm.copy()

    return best_trial, best_score


def line_search_log(f, sg, current_best=None, max_golden=8):
    """Log-space multiplicative line search."""
    f = np.asarray(f, dtype=np.float64)
    sg = np.asarray(sg, dtype=np.float64)
    if current_best is None:
        current_best = objective_normalized(f)

    if sg.size == 0 or not np.all(np.isfinite(sg)):
        return f.copy(), current_best
    mx = float(np.max(sg))
    if mx <= 1e-300:
        return f.copy(), current_best
    sg = sg / mx

    logf = np.log(np.maximum(f, 1e-15))
    best_score = float(current_best)
    best_trial = f.copy()
    best_step = 0.0
    best_grid_idx = -1

    grid = [1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 1.0, 3.0]
    for i, st in enumerate(grid):
        trial, sc = trial_from_log(f, logf, sg, st)
        if trial is None:
            continue
        if sc < best_score:
            best_score = sc
            best_trial = trial.copy()
            best_step = st
            best_grid_idx = i

    if best_grid_idx >= 0:
        if best_grid_idx == 0:
            lo, hi = 0.0, grid[1]
        elif best_grid_idx == len(grid) - 1:
            lo, hi = grid[-2], grid[-1] * 2.0
        else:
            lo, hi = grid[best_grid_idx - 1], grid[best_grid_idx + 1]

        if hi > lo + 1e-15:
            gr = 0.618
            for _ in range(max_golden):
                s1 = hi - gr * (hi - lo)
                s2 = lo + gr * (hi - lo)
                if s1 <= lo + 1e-14 or s2 >= hi - 1e-14:
                    break
                t1, v1 = trial_from_log(f, logf, sg, s1)
                t2, v2 = trial_from_log(f, logf, sg, s2)
                if t1 is not None and v1 < best_score:
                    best_score = v1
                    best_trial = t1.copy()
                if t2 is not None and v2 < best_score:
                    best_score = v2
                    best_trial = t2.copy()
                if v1 < v2:
                    hi = s2
                else:
                    lo = s1

            mid = 0.5 * (lo + hi)
            tm, vm = trial_from_log(f, logf, sg, mid)
            if tm is not None and vm < best_score:
                best_score = vm
                best_trial = tm.copy()

    return best_trial, best_score


def select_active_indices(conv, M, max_active, margin):
    """Select high convolution indices for LP active constraint set."""
    conv = np.asarray(conv, dtype=np.float64)
    L = conv.size
    if L == 0:
        return np.array([], dtype=np.int64)
    if M is None:
        M = float(np.max(conv))
    if M <= 0.0 or not np.isfinite(M):
        return np.array([int(np.argmax(conv))], dtype=np.int64)

    lower = M * (1.0 - margin)
    cand = np.where(conv >= lower)[0]
    if cand.size == 0:
        cand = np.array([int(np.argmax(conv))], dtype=np.int64)

    if cand.size > max_active:
        vals = conv[cand]
        order = np.argsort(vals)
        top = cand[order[-max_active:]]
        # Always keep exact (numerical) maxima.
        exact = np.where(conv >= M * (1.0 - 1e-7))[0]
        top = np.unique(np.concatenate((top, exact)))
        if top.size > max(int(2 * max_active), 32):
            vals2 = conv[top]
            order2 = np.argsort(vals2)
            top = top[order2[-max_active:]]
    return top.astype(np.int64)


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = max(1e-6, n * sigma_frac)
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return normalize_nonnegative(a)


def make_bell(n, width_frac=0.55):
    x = np.arange(n) - n / 2.0
    half = max(1e-6, n * width_frac / 2.0)
    t = np.clip(np.abs(x) / half, 0.0, 1.0)
    a = 1.0 - t ** 4
    return normalize_nonnegative(a)


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / float(n))
    return normalize_nonnegative(a)


def make_random_binary(rng, n, p):
    v = (rng.random(n) < p).astype(np.float64)
    return normalize_nonnegative(v)


def random_simplex(rng, n):
    v = rng.random(n)
    return normalize_nonnegative(v)


def random_perturb_simplex(f, rng, amp, mode=0):
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    m = int(mode)
    if m == 0:
        v = f + rng.standard_normal(n) * amp
    elif m == 1:
        shift = int(rng.integers(1, max(2, n // 8 + 1)))
        v = np.roll(f, shift)
    elif m == 2:
        mix = 0.1 + 0.1 * (m % 3)
        v = (1.0 - mix) * f + mix * (np.ones(n) / n)
    elif m == 3:
        # zero a random subset of coordinates
        frac = 0.05 + 0.05 * (m % 4)
        nzero = int(round(n * frac))
        v = f.copy()
        if 0 < nzero < n:
            idx = rng.choice(n, size=nzero, replace=False)
            v[idx] = 0.0
        elif nzero >= n:
            v[:] = 0.0
    else:
        # blend with random binary
        rb = make_random_binary(rng, n, 0.5)
        mix = 0.1 + 0.1 * (m % 5)
        v = (1.0 - mix) * f + mix * rb
    return project_simplex(np.maximum(v, 0.0))


def random_pair_move(f, rng, max_evals=10, current_best=None):
    """Random mass-transfer moves i -> j with small line search."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    if n < 2:
        return f.copy(), current_best if current_best is not None else objective_normalized(f)
    best_score = (
        float(current_best)
        if current_best is not None and np.isfinite(current_best)
        else objective_normalized(f)
    )
    best_trial = f.copy()

    s = float(np.sum(f))
    if s <= 1e-15:
        return best_trial, best_score
    p = f / s

    for _ in range(max_evals):
        i = int(rng.choice(n, p=p))
        j = int(rng.integers(n))
        if i == j:
            continue
        base = best_trial.copy()
        for frac in (0.1, 0.2, 0.3, 0.5, 0.7):
            delta = frac * base[i]
            if delta <= 0.0:
                continue
            trial = base.copy()
            trial[i] -= delta
            trial[j] += delta
            if float(np.min(trial)) < -1e-12:
                continue
            trial = np.maximum(trial, 0.0)
            ts = float(np.sum(trial))
            if ts <= 1e-15:
                continue
            trial = trial / ts
            sc = objective_normalized(trial)
            if sc < best_score - 1e-14:
                best_score = sc
                best_trial = trial.copy()
                base = trial.copy()
                p = base / base.sum()
    return best_trial, best_score


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if old_n == new_n:
        return normalize_nonnegative(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    v = np.interp(new_x, old_x, f)
    return normalize_nonnegative(v)


def optimize_sequence(f0, budget, seed=0, verbose=False):
    """
    Main local optimizer for one length. Returns (best_sequence, best_internal_score).
    """
    f = normalize_nonnegative(f0)
    n = len(f)
    if n < 2:
        return f.copy(), objective_normalized(f)

    info = build_convolve_rows_info(n)
    best_f = f.copy()
    best_score = objective_normalized(f)

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget))
    rng = np.random.default_rng(seed)

    it = 0
    no_improve = 0
    beta = 150.0
    margin = 0.02

    while time.monotonic() < deadline:
        it += 1
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break

        conv = fast_convolve(f)
        M = float(np.max(conv))
        current_score = float(2.0 * n * M)
        if current_score < best_score - 1e-24:
            best_score = current_score
            best_f = f.copy()

        improved = False

        # --- LP-Frank-Wolfe direction ---
        if rem > 3.0:
            # Use full constraints early and periodically; use active subset often.
            use_full = (
                it < 80
                or no_improve > 8
                or (it % 30 == 0)
            )
            lp_time = min(1.0, max(0.25, 0.05 * rem))
            if use_full:
                g, _ = solve_lp_direction(f, info, active=None, n=n, lp_time=lp_time)
            else:
                max_active = 80
                if no_improve > 6:
                    max_active = 140
                active = select_active_indices(conv, M, max_active=max_active, margin=margin)
                if active.size < 8:
                    active = None
                g, _ = solve_lp_direction(f, info, active=active, n=n, lp_time=lp_time)

            if g is not None:
                if float(np.max(np.abs(g - f))) > 1e-9:
                    trial, sc = line_search_direction(f, g, current_best=best_score)
                    if sc < best_score - 1e-14:
                        f = trial.copy()
                        best_score = sc
                        best_f = f.copy()
                        improved = True

        # --- Smooth log-sum-exp mirror descent ---
        if not improved and rem > 1.0:
            w = softmax_peak_weights(conv, beta)
            sg = weighted_subgradient(f, w, n)
            mx = float(np.max(sg))
            if mx > 1e-300:
                trial, sc = line_search_log(f, sg, current_best=best_score)
                if sc < best_score - 1e-14:
                    f = trial.copy()
                    best_score = sc
                    best_f = f.copy()
                    improved = True

        # --- Random pairwise moves ---
        if not improved and rem > 0.8:
            trial, sc = random_pair_move(f, rng, max_evals=8, current_best=best_score)
            if sc < best_score - 1e-14:
                f = trial.copy()
                best_score = sc
                best_f = f.copy()
                improved = True

        if improved:
            no_improve = 0
            beta = min(5000.0, beta * 1.06)
            margin = max(0.01, margin * 0.85)
        else:
            no_improve += 1
            margin = min(0.45, margin + 0.015)
            beta = max(30.0, beta * 0.90)

            if no_improve == 5 or no_improve % 20 == 0:
                mode = no_improve % 4
                amp = 0.003 + 0.004 * min(1.0, float(no_improve) / 30.0)
                f = random_perturb_simplex(best_f, rng, amp, mode)
                no_improve = 0
                margin = max(0.02, margin * 0.7)
                beta = max(60.0, beta * 0.7)
            elif no_improve > 120:
                # More aggressive restart around incumbent.
                f = 0.7 * best_f + 0.3 * random_simplex(rng, n)
                f = project_simplex(f)
                no_improve = 50

        if verbose and (it % 25 == 0 or (improved and it % 5 == 0)):
            print(f"    optimize iter={it} best={best_score:.10f} no_improve={no_improve}")

    return best_f.copy(), best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Searches for a nonnegative sequence minimizing
    evaluate_sequence.
    """
    t_start = time.monotonic()
    total_budget = max(2.0, float(budget_s) - 12.0)
    deadline = t_start + total_budget
    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    # Load parent if available, applying the same clipping idea.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and float(np.sum(p)) > 1e-12:
            parent = normalize_nonnegative(p)
    except Exception:
        parent = None

    if parent is None or len(parent) == 0:
        parent = make_gaussian(768, 0.32)
    parent = normalize_nonnegative(parent)

    # If enormous, downsample to a manageable length first.
    if len(parent) > 2048:
        parent = interpolate_sequence(parent, 1536)
    n = len(parent)

    # Build diverse starting candidates.
    candidates = []
    candidates.append(("parent", parent))
    candidates.append(("uniform", np.ones(n) / n))
    candidates.append(("gauss", make_gaussian(n, 0.30)))
    candidates.append(("bell", make_bell(n, 0.55)))
    candidates.append(("cos1", make_cosine_mod(n, 1, 0.30)))
    candidates.append(("cos3", make_cosine_mod(n, 3, 0.18)))
    candidates.append(("bin50", make_random_binary(rng, n, 0.50)))
    candidates.append(("bin30", make_random_binary(rng, n, 0.30)))
    candidates.append(("bin70", make_random_binary(rng, n, 0.70)))

    # Rank by internal objective quickly.
    scored = []
    for label, a in candidates:
        a = normalize_nonnegative(a)
        sc = objective_normalized(a)
        if np.isfinite(sc):
            scored.append((sc, label, a))

    if not scored:
        seq = np.ones(768) / 768
        return [float(x) for x in seq]

    scored.sort()

    # Verify top candidates with the real evaluator and choose start.
    best_start = scored[0][2].copy()
    best_start_label = scored[0][1]
    best_actual = objective_normalized(best_start)

    top_start = best_start.copy()
    for sc, label, a in scored[:3]:
        try:
            actual = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            actual = float("inf")
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_start = a.copy()
            best_start_label = label
            top_start = a.copy()

    # Keep the actual incumbent.
    incumbent = top_start.copy()
    incumbent_score = float(evaluate_sequence([float(x) for x in incumbent]))
    if not np.isfinite(incumbent_score):
        incumbent = np.ones(768) / 768
        incumbent_score = float(evaluate_sequence([float(x) for x in incumbent]))

    rem = deadline - time.monotonic()
    if rem > 30.0:
        t_alloc = min(rem * 0.68, max(20.0, total_budget * 0.68))
        f_opt, int_score = optimize_sequence(
            top_start, budget=t_alloc, seed=int(seed) + 1, verbose=True
        )
        try:
            act = float(evaluate_sequence([float(x) for x in f_opt]))
        except Exception:
            act = float("inf")
        if np.isfinite(act) and act < incumbent_score:
            incumbent = f_opt.copy()
            incumbent_score = act
            print(f"  Main optimizer improved actual score to {incumbent_score:.10f}")

    # Multi-start from other promising initializations if time remains.
    done_labels = {best_start_label}
    rem = deadline - time.monotonic()
    for sc, label, a in scored:
        if rem < 80.0:
            break
        if label in done_labels:
            continue
        done_labels.add(label)
        t_alloc = min(70.0, rem * 0.25)
        f_opt, _ = optimize_sequence(a, budget=t_alloc, seed=int(seed) + 100 + len(done_labels), verbose=False)
        try:
            act = float(evaluate_sequence([float(x) for x in f_opt]))
        except Exception:
            act = float("inf")
        if np.isfinite(act) and act < incumbent_score:
            incumbent = f_opt.copy()
            incumbent_score = act
            print(f"  Secondary start {label} improved actual score to {incumbent_score:.10f}")
        rem = deadline - time.monotonic()

    # Try a modest length increase if the incumbent is not too long and time remains.
    rem = deadline - time.monotonic()
    L = len(incumbent)
    if rem > 120.0 and L < 1024:
        target_n = 1024
        interp = interpolate_sequence(incumbent, target_n)
        t_alloc = min(90.0, rem * 0.45)
        f_opt, _ = optimize_sequence(interp, budget=t_alloc, seed=int(seed) + 1000, verbose=False)
        try:
            act = float(evaluate_sequence([float(x) for x in f_opt]))
        except Exception:
            act = float("inf")
        if np.isfinite(act) and act < incumbent_score:
            incumbent = f_opt.copy()
            incumbent_score = act
            print(f"  Length {target_n} improved actual score to {incumbent_score:.10f}")

    # Final small polish if time remains.
    rem = deadline - time.monotonic()
    if rem > 25.0:
        f_opt, _ = optimize_sequence(incumbent, budget=min(20.0, rem * 0.7), seed=int(seed) + 2000, verbose=False)
        try:
            act = float(evaluate_sequence([float(x) for x in f_opt]))
        except Exception:
            act = float("inf")
        if np.isfinite(act) and act < incumbent_score:
            incumbent = f_opt.copy()
            incumbent_score = act

    # Ensure returned sequence is valid and finite.
    incumbent = normalize_nonnegative(incumbent)
    incumbent = np.clip(incumbent, 0.0, 1000.0)
    try:
        final_score = float(evaluate_sequence([float(x) for x in incumbent]))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        fallback = np.ones(768) / 768
        return [float(x) for x in fallback]

    print(f"FINAL actual score: {final_score:.10f} length={len(incumbent)}")
    return [float(x) for x in incumbent]
```
