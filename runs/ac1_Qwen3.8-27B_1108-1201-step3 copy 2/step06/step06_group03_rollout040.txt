                relax = 1.0
                no_improve = 0
            continue
        
        no_improve = 0
        
        # Line search: find alpha that minimizes max of the quadratic
        # (f_new * f_new)(k) = (1-α)^2(f*f)(k) + 2α(1-α)(f*g)(k) + α^2(g*g)(k)
        conv_f = fast_convolve(f)
        conv_fg = fast_convolve(f)  # placeholder, will compute properly
        # Actually: (f*g)(k) = sum_j f(j)g(k-j)
        # We can compute this via FFT too
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        F = np.fft.rfft(f, fft_size)
        G = np.fft.rfft(g, fft_size)
        conv_f = np.fft.irfft(F * F, fft_size)[:size]
        conv_fg = np.fft.irfft(F * G, fft_size)[:size]
        conv_g = np.fft.irfft(G * G, fft_size)[:size]
        
        best_alpha = 0
        best_val = M
        # Coarse search
        for alpha in np.linspace(0, 1, 21):
            val = (1-alpha)**2 * conv_f + 2*alpha*(1-alpha) * conv_fg + alpha**2 * conv_g
            val_max = val.max()
            if val_max < best_val:
                best_val = val_max
                best_alpha = alpha
        
        # Fine search
        if best_alpha > 0:
            lo = max(0, best_alpha - 0.05)
            hi = min(1, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2
                val = (1-mid)**2 * conv_f + 2*mid*(1-mid) * conv_fg + mid**2 * conv_g
                val_max = val.max()
                if val_max < best_val:
                    best_val = val_max
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha < 1e-6:
            continue
        
        # Update
        f = (1 - best_alpha) * f + best_alpha * g
        f = np.maximum(f, 0)
        f_s = f.sum()
        if f_s < 1e-15:
            f = best_f.copy()
        else:
            f = f / f_s
        
        M = fast_convolve(f).max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            # Tighten relaxation
            relax = max(1.0, relax * 0.95)
            if verbose and total_iter % 20 == 0:
                print(f"      LP-FW iter {total_iter}: obj={best_obj:.10f}, "
                      f"g_sum={g_sum:.4f}, alpha={best_alpha:.3f}")
        
        if no_improve > 200:
            break
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()
            
            # Try several perturbations
            for frac in [0.5, 0.7, 0.85, 0.95, 1.1, 1.3, 1.5, 2.0, 0.3, 0.1]:
                new_val = old_val * frac
                if new_val < 0 or new_val > 0.9:
                    continue
                # Renormalize: keep sum = 1
                # f[j] changes from old_val to new_val
                # Other elements scale by (1-new_val)/(1-old_val)
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def smoothed_descent(f0, time_limit, verbose=False):
    """
    Smoothed subgradient method targeting top peaks with softmax weighting.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.02
    tau = 10.0
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Get top peaks
        k = max(5, len(conv) // 20)
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        
        # Softmax weights
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(k)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k):
            if w[i] < 1e-12:
                continue
            ki = int(peaks[i])
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            sg[js:je+1] += w[i] * (f[ki - idx] + f[idx])
        
        # Multiplicative update
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < M:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      SMD iter {it}: obj={best_obj:.10f}")
            lr = min(lr * 1.01, 0.08)
        else:
            lr *= 0.8
            no_improve += 1
        
        tau *= 1.002
        
        if no_improve > 80:
            noise = rng.standard_normal(n) * 0.005
            f = np.maximum(best_f + noise, 1e-15)
            f = f / f.sum()
            lr = 0.02
            tau = 10.0
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_f, best_obj
