import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(f, g):
    """Compute cross-convolution f * g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def objective(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1.0)
    rho_val = np.nonzero(cond)[0][-1] if cond.any() else n-1
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M):
    """Solve LP to find best direction g."""
    n = len(f)
    # Build constraint matrix A where A[k,i] = f[i] if valid
    # Constraint: A @ g <= M * ones(2n-1)
    # Maximize: ones(n) @ g
    # Subject to: g >= 0
    
    # A is (2n-1) x n, banded
    # A[k,i] = f[i] if max(0, k-n+1) <= i <= min(n-1, k)
    
    # For efficiency, build A as sparse or dense

    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):

        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[lo:hi+1]
    
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:

    return None

def line_search(f, g, t_max=1.0, n_points=50):

        obj = objective(f_trial)
        if obj < best_obj:

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based direction finding,
subgradient descent with momentum, L-BFGS soft-max polishing, and coordinate
descent. The LP approach finds the optimal descent direction at each point,
giving monotone improvement and typically faster convergence than raw
subgradient methods.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve_fft(f, g):
    """Cross-convolution via FFT: result[k] = sum_i f[i]*g[k-i]"""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def objective(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build sparse constraint matrix
    # A[k, i] = f[i] if max(0, k-n+1) <= i <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method=method)
    if res.success and res.x is not None:
        return res.x
    return None

def line_search(f, g, t_max=1.0, n_coarse=20):
    """Find best t for f_new = (1-t)*f + t*g."""
    best_t = 0.0
    best_obj = objective(f)
    
    # Coarse search
    ts = np.linspace(0, t_max, n_coarse)
    objs = np.array([objective((1-t)*f + t*g) for t in ts])
    min_idx = np.argmin(objs)
    best_obj = objs[min_idx]
    best_t = ts[min_idx]
    
    # Fine search around best
    if 0 < min_idx < n_coarse - 1:
        t_lo = ts[min_idx - 1]
        t_hi = ts[min_idx + 1]
        for _ in range(20):
            t_mid1 = t_lo + (t_hi - t_lo) / 3
            t_mid2 = t_hi - (t_hi - t_lo) / 3
            obj1 = objective((1-t_mid1)*f + t_mid1*g)
            obj2 = objective((1-t_mid2)*f + t_mid2*g)
            if obj1 < obj2:
                t_hi = t_mid2
            else:
                t_lo = t_mid1
            t_mid = (t_lo + t_hi) / 2
            obj_mid = objective((1-t_mid)*f + t_mid*g)
            if obj_mid < best_obj:
                best_obj = obj_mid
                best_t = t_mid
    
    return best_t, best_obj

def lp_optimize(f0, time_limit, max_iters=500):
    """LP-based optimization with line search."""
    t0 = time.monotonic()
    f = f0.copy().astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_obj = objective(f)
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(f)
        M = b.max()
        
        g = lp_direction(f, M)
        if g is None or g.sum() < 1e-10:

        g = g / g.sum()
        
        t, obj = line_search(f, g, t_max=1.0, n_coarse=15)
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 5:
                # Try smaller step
                t, obj = line_search(f, g, t_max=0.1, n_coarse=10)
                f_new = (1 - t) * f + t * g
                f_new = np.maximum(f_new, 0)

                obj_new = objective(f_new)

                    best_obj = obj_new
                    best_f = f_new.copy()
                    f = f_new.copy()
                    stagnant = 0
                else:
                    break
    
    return best_f, best_obj

    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def objective(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        L = hi - lo + 1
        rows.append(np.full(L, k))
        cols.append(np.arange(lo, hi + 1))
        vals.append(f[lo:hi + 1])
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_optimize(f0, time_limit, max_iters=2000):
    t0 = time.monotonic()
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    f = f / f.sum()
    best_f = f.copy()
    best_obj = objective(f)
    stagnant = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit and iter_count < max_iters:
        iter_count += 1
        b = fast_convolve(f)
        M = b.max()
        n = len(f)

        A = build_lp_matrix(f)
        b_ub = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search
        best_t = 0.0
        best_trial_obj = best_obj
        # Try several t values
        for t in [0.5, 0.3, 0.7, 0.2, 0.1, 0.9, 0.4, 0.6]:
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            ss = f_trial.sum()
            if ss < 1e-10:
                continue
            f_trial = f_trial / ss
            obj_trial = objective(f_trial)
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_t = t

        if best_t == 0.0:
            # Try very small steps
            for t in [0.05, 0.02, 0.01]:
                f_trial = (1 - t) * f + t * g
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-10:
                    continue
                f_trial = f_trial / ss
                obj_trial = objective(f_trial)
                if obj_trial < best_obj:
                    best_trial_obj = obj_trial
                    best_t = t

        if best_t > 0 and best_trial_obj < best_obj - 1e-14:
            f_new = (1 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            best_obj = objective(f_new)
            best_f = f_new.copy()
            f = f_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 10:
                break

    return best_f, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.001, seed=42):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        # Batch of 30 iterations
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            it_count += 1
            b = fast_convolve(a)
            b_max = b.max()

            # Temperature-weighted peak selection
            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum

            # Compute subgradient via FFT cross-correlation
            # g[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
            # This is 2 * cross_corr(w, a) restricted to [0, n)
            size = n + len(w) - 1
            fft_size = 1 << (size - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            cg = irfft(np.conj(W) * A, fft_size)[:n]
            g = 2.0 * cg

            # Nesterov step
            v = momentum * v - cur_lr * g
            a_new = project_simplex(a + v)

            b_new = fast_convolve(a_new)
            obj_new = b_new.max()

            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            a = a_new

        # Adjustments
        if no_improve > 2000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > 1000:
            cur_lr *= 0.95

        cur_temp *= 1.02
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def soft_max_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    # Gradient via cross-correlation
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=5000):
    t0 = time.monotonic()
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    best_obj = soft_max_obj_grad(z0, T, n)[0]
    best_z = z0.copy()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass

    return softmax_from_z(best_z), best_obj


def coordinate_polish(a0, time_limit):
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    improved = True

    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    return best_a, best_obj


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bimodal_init(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = objective(a_norm)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = objective(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = make_cosine_init(1152)
        print("No parent, using cosine n=1152")

    n_parent = len(parent)

    # === Phase 1: LP optimization on parent ===
    print(f"\n=== Phase 1: LP optimization (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 900
    if rem > 60:
        t_alloc = min(rem, 600.0)
        print(f"  LP optimize, t={t_alloc:.0f}s")
        a_lp, obj_lp = lp_optimize(parent, time_limit=t_alloc)
        try_seq(a_lp, f"lp_n{n_parent}")

    # === Phase 2: Subgradient from multiple diverse starts ===
    print(f"\n=== Phase 2: Subgradient diverse starts ===")
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = n_parent

    initializations = {
        "parent": parent if len(parent) == n_base else interpolate_sequence(best_seq if best_seq is not None else parent, n_base),
        "gauss_0.2": make_gaussian_init(n_base, 0.2),
        "gauss_0.4": make_gaussian_init(n_base, 0.4),
        "cosine": make_cosine_init(n_base),
        "tukey": make_tukey_init(n_base, 0.5),
        "bimodal": make_bimodal_init(n_base),
        "uniform": np.ones(n_base) / n_base,
    }

    if best_seq is not None:
        initializations["best_perturb"] = project_simplex(
            best_seq + rng.standard_normal(n_base) * 0.01)

    n_inits = len(initializations)
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        t_per_init = min(rem / n_inits, 80.0)
        for name, init_a in initializations.items():
            rem = deadline - time.monotonic() - 300
            if rem < 10:
                break
            t_alloc = min(t_per_init, rem * 0.9)
            if t_alloc < 5:
                continue
            print(f"  SG {name} (t={t_alloc:.0f}s)")
            a_opt, _ = subgradient_optimize(
                init_a, time_limit=t_alloc,
                lr_init=0.005, momentum=0.9,
                temp_init=0.001,
                seed=seed + hash(name) % 10000
            )
            try_seq(a_opt, f"sg_{name}")

    # === Phase 3: LP optimization on best (second pass) ===
    print(f"\n=== Phase 3: LP second pass (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 60:
            t_alloc = min(rem, 300.0)
            print(f"  LP pass 2, n={len(best_seq)}, t={t_alloc:.0f}s")
            a_lp2, _ = lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_lp2, "lp_pass2")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration ===")
    if best_seq is not None:
        target_lengths = []
        for mult in [0.75, 1.25, 1.5, 2.0]:
            tl = int(n_parent * mult)
            tl = max(256, min(3000, tl))
            if tl not in target_lengths and tl != n_parent:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:3]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 80.0)
            if t_alloc < 10:
                continue
            print(f"  Length n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(
                f_interp, time_limit=t_alloc,
                lr_init=0.005, momentum=0.9,
                temp_init=0.001,
                seed=seed + target_n
            )
            try_seq(a_opt, f"len_n{target_n}")

            # Quick LP on this length
            rem = deadline - time.monotonic() - 100
            if rem > 30 and best_seq is not None and len(best_seq) == target_n:
                t_lp = min(rem * 0.5, 60.0)
                a_lp3, _ = lp_optimize(best_seq, time_limit=t_lp)
                try_seq(a_lp3, f"lp_n{target_n}")

    # === Phase 5: L-BFGS polishing ===
    print(f"\n=== Phase 5: L-BFGS polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for T in [30, 80, 150]:
            rem = deadline - time.monotonic() - 60
            if rem < 10:
                break
            t_alloc = min(rem * 0.4, 60.0)
            if t_alloc < 5:
                continue
            print(f"  L-BFGS T={T}, n={n_best}, t={t_alloc:.0f}s")
            a_lb, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=3000)
            try_seq(a_lb, f"lbgfs_T{T}")

    # === Phase 6: Another LP pass on the polished best ===
    print(f"\n=== Phase 6: Final LP pass (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(rem, 150.0)
            print(f"  Final LP, t={t_alloc:.0f}s")
            a_fin, _ = lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_fin, "lp_final")

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: CD polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd = min(rem, 40.0)
            print(f"  CD, t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd")

    # === Phase 8: Final micro subgradient ===
    print(f"\n=== Phase 8: Micro SG (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_ms = min(rem, 20.0)
            a_micro, _ = subgradient_optimize(
                best_seq, time_limit=t_ms,
                lr_init=0.001, momentum=0.95,
                temp_init=0.0003,
                seed=seed + 99999
            )
            try_seq(a_micro, "micro")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = objective(best_seq)

    # Final verification with evaluate_sequence
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
