Let me also think about a completely different approach:

**Direct minimization using the "mirror descent" framework:**

The problem is: minimize f(a) = max_k (Ca)_k over the simplex.

Mirror descent with exponential weights (i.e., on the probability simplex with entropy regularization) gives:
- a_t = exp(-eta * g_{t-1}) / Z, where g is the subgradient

This is equivalent to multiplicative weight updates. The convergence rate is O(sqrt(log(K)/T)) where K is the number of constraints.

For our problem, K = 2n-1 ≈ 1535. So the rate is O(sqrt(log(1535)/T)) ≈ O(2.3/sqrt(T)). To get to within epsilon of optimal, we need T ≈ (2.3/epsilon)^2 iterations.

If the optimal value is ~0.75/n and the initial is 1/n, the gap is 0.25/n. For n=768, gap ≈ 0.000326. To get to within half of this, we need T ≈ (2.3/0.000163)^2 ≈ 200 million. That's way too many!

So mirror descent alone won't work. We need something faster.

**Second-order methods (L-BFGS) should be much faster** for this problem, especially with good smoothing. The previous code already uses this.

Let me think about what specific improvement to make:

**Key insight**: The previous code uses a single L-BFGS call with a fixed temperature T. The issue is that:
- Low T: accurate gradient but noisy (the soft-max doesn't smooth enough)
- High T: smooth gradient but biased (the soft-max distorts the objective)

The solution is to **anneal T**: start high (for fast progress) and decrease (for accuracy). This is like simulated annealing but for the smoothing parameter.

**Another key insight**: The previous code evaluates the score using the exact max, but optimizes using the soft-max approximation. There's a gap between the two. To close this gap, we should:
1. Optimize with soft-max (fast, smooth)
2. Periodically evaluate with exact max
3. If the exact max is at a specific lag, add a penalty term for that lag in the soft-max objective

**Yet another idea: Active set exploitation**

Near the optimum, only a few lags are "active" (achieve the max). We can:
1. Identify the active set
2. Solve a QP: minimize `sum_{k in active} (Ca)_k` subject to `(Ca)_k ≤ M for all k` and sum(a)=1, a≥0
3. Update the active set
4. Repeat

This is the "constraint generation" or "cutting plane" approach.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive soft-max L-BFGS optimization
combined with LP-based fictitious play (polynomial averaging), peak shaving, and
multi-length exploration. Key improvements over previous approach:
1. Adaptive temperature annealing in L-BFGS for better gradient quality
2. Polynomial (Favre) averaging in fictitious play for faster convergence
3. Active-set-aware peak shaving
4. Broader length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def cross_corr(w, a):
    """Cross-correlation via FFT: result[i] = sum_k w[k] * a[k+i] (circular, length n)."""
    n = len(a)
    size = n
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def lbgfs_soft_max(a0, time_limit, T_start=1000, T_end=50, maxiter=5000, n_restarts=3):
    """
    L-BFGS refinement in log-space with soft-max smoothing and temperature annealing.
    Runs multiple restarts with different initial temperatures.
    """
    n = len(a0)
    best_a = np.maximum(a0.copy(), 1e-15)
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Different initial z for each restart
        if restart == 0:
            z0 = z_from_a(best_a)
        else:
            rng = np.random.default_rng(42 + restart)
            z0 = z_from_a(best_a) + rng.standard_normal(n) * 0.1
        
        # Time allocation for this restart
        t_alloc = (time_limit - (time.monotonic() - t0)) / (n_restarts - restart)
        
        # Temperature annealing: T starts high, ends low
        T_traj = np.geomspace(T_start, T_end, maxiter)
        
        # We'll use a custom callback to update T
        state = {'T_idx': 0, 'n_calls': 0}
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            state['n_calls'] += 1
            T = T_traj[min(state['T_idx'], maxiter - 1)]
            state['T_idx'] += 1
            a = softmax_from_z(z)
            b = fast_convolve(a)
            # Soft-max objective
            max_b = b.max()
            exp_b = np.exp(b / T - max_b / T)
            log_sum_exp = max_b + T * np.log(exp_b.sum() / len(b))
            obj = 2.0 * n * log_sum_exp
            return obj
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            T = T_traj[min(state['T_idx'], maxiter - 1)]
            state['T_idx'] += 1
            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b / T - max_b / T)
            w = exp_b / exp_b.sum()
            # Gradient of soft-max objective w.r.t. a
            grad_a = 2.0 * n * cross_corr(w, a)
            # Chain rule through softmax
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - a * S)
            return grad_z
        
        try:
            res = minimize(
                fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxfun': maxiter * 20, 'disp': False}
            )
            a_out = softmax_from_z(res.x)
            a_out = np.maximum(a_out, 0)
            s = a_out.sum()
            if s > 1e-10:
                a_out = a_out / s
                score = 2.0 * n * fast_convolve(a_out).max()
                if score < best_score:
                    best_score = score
                    best_a = a_out.copy()
                    print(f"    L-BFGS restart {restart}: score={best_score:.8f}")
        except Exception as e:
            print(f"    L-BFGS restart {restart} failed: {e}")
    
    return best_a, best_score


def fictitious_play_with_averaging(n, time_limit, initial=None, solver=None, verbose=True):
    """
    Fictitious play with polynomial (Favre) averaging for faster convergence.
    """
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
        t = np.linspace(0, 2 * np.pi, n)
        a = a + 0.02 * np.cos(t)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    # Polynomial averaging: keep a history
    history = [a.copy()]
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Best response for b given a
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        b = b_new
        history.append(b.copy())
        
        # Best response for a given b
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is None:
            continue
        a = a_new
        history.append(a.copy())
        
        # Polynomial (Favre) averaging with degree 2
        # Weight: w_t = (t+1)^2 for the t-th iterate
        weights = np.array([(i + 1) ** 2 for i in range(len(history))], dtype=np.float64)
        weights /= weights.sum()
        a_avg = np.zeros(n)
        for i, h in enumerate(history):
            a_avg += weights[i] * h
        
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-15:
            a_avg = a_avg / s
        
        # Evaluate candidates
        candidates = [a, b, a_avg, 0.5 * (a + b)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            max_conv = fast_convolve(cand).max()
            score = 2.0 * n * max_conv
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=5000):
    """
    Aggressive peak shaving: directly reduce the contribution of top peaks.
    Uses a smarter approach: identify which pairs of indices contribute most
    to each peak and adjust them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find top peaks (excluding endpoints which are always small)
        b_work = b.copy()
        if n > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]
        
        if peak_val <= best_score / (2.0 * n) + 1e-15:
            break
        
        # Find top contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Sort by contribution
        order = np.argsort(contribs)[::-1]
        
        improved = False
        for ci in order[:5]:
            i = int(idx[ci])
            j = int(peak_k - i)
            old_val = a[i]
            if old_val < 1e-14:
                continue
            
            # Try reducing a[i] and redistributing
            for factor in [0.97, 0.94, 0.9, 0.85]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to other elements proportional to their values
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing both i and j simultaneously
            for ci in order[:3]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s < 1e-15:
                            continue
                        a_trial = a_trial / s
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if not improved:
                break
    
    return best_a, best_score


def multiplicative_weight_update(a0, time_limit, n_iters=None):
    """
    Multiplicative weight / mirror descent approach.
    a_{t+1}[i] = a_t[i] * exp(-eta * g_t[i]) / Z
    where g_t is the subgradient of max_k conv(a,a) at a_t.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if n_iters is None:
        n_iters = 100000
    
    # Adaptive learning rate
    eta = 0.5
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find peaks with some softness
        peak = b.max()
        # Soft weighting: give more weight to peaks close to max
        threshold = peak * 0.99
        active = b >= threshold
        if not active.any():
            active = np.zeros(len(b), dtype=bool)
            active[int(np.argmax(b))] = True
        
        # Compute subgradient (average over active peaks)
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        
        grad /= active.sum()
        # Note: the actual gradient of g_k(a) = sum_j a[j]*a[k-j] w.r.t. a is 2*(T_k a)
        # But we use the "asymmetric" version for multiplicative updates
        # grad[i] = sum_{k active} a[k-i] / |active|
        
        # Multiplicative update
        max_grad = np.abs(grad).max()
        if max_grad < 1e-15:
            break
        grad_normed = grad / max_grad
        
        a_new = a * np.exp(-eta * grad_normed)
        a_new = np.maximum(a_new, 1e-300)
        s = a_new.sum()
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Decay eta
        eta *= 0.999
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.4f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def damped_lp_refine(a0, time_limit, solver=None):
    """Damped alternating LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        
        # Damped update
        alpha = min(1.0, 3.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score
