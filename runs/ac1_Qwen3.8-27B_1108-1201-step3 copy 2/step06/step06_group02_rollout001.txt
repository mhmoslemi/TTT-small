def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent,
subgradient optimization, and peak equalization strategies.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_normalized(a):
    """2*n*max(conv) for a with sum=1."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def obj_raw(a):
    """Full objective without assuming normalization."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)

def coord_descent_sweep(a, num_sweeps=10, trial_ratios=None, verbose=False):
    """Coordinate descent: for each coordinate, try several values and keep the best."""
    if trial_ratios is None:
        trial_ratios = [0.3, 0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.5, 2.0, 0.0]
    
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s  # Normalize
    n = len(a)
    best_obj = obj_normalized(a)
    
    for sweep in range(num_sweeps):
        improved = False
        indices = np.random.permutation(n)
        for i in indices:
            old_val = a[i]
            if old_val < 1e-15 and i not in [0, n-1]:
                # Skip zero coordinates sometimes (they might be optimal at 0)
                if np.random.random() > 0.1:
                    continue
            
            best_trial_obj = best_obj
            best_trial_val = old_val
            
            for ratio in trial_ratios:
                trial_val = old_val * ratio if old_val > 0 else 0.0
                # Also try absolute small values
                if old_val < 1e-6:
                    trial_val = max(0, old_val + (ratio - 1.0) * 0.01 / n)
                
                a[i] = max(0.0, trial_val)
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = old_val
                    continue
                trial_obj = 2.0 * n * fast_convolve(a).max() / (s_new * s_new)
                if trial_obj < best_trial_obj - 1e-15:
                    best_trial_obj = trial_obj
                    best_trial_val = max(0.0, trial_val)
            
            a[i] = best_trial_val
            if best_trial_val != old_val:
                improved = True
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                    best_obj = obj_normalized(a)
        
        if not improved and verbose:
            print(f"    Coordinate descent converged at sweep {sweep}")
            break
    
    return a
