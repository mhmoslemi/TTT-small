```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_1d(A, B, D_idx, n, S):
    """Compute 2n * max_k(A[k] + B[k]*v + D*v^2 if k==D_idx) / (S+v)^2
    D is 1 at D_idx, 0 elsewhere."""
    def g(v):
        vals = A + B * v
        vals[D_idx] += v * v  # D_{D_idx} = 1
        return 2.0 * n * vals.max() / (S + v) ** 2
    return g

def coord_descent(a_init, time_limit=60, verbose=False):
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return a, np.inf
    
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max() / S**2
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_v = a[j]
            a[j] = 0.0
            A = fast_convolve(a)  # conv with a_j = 0
            S_other = S - old_v
            
            # B_k = 2*a[k-j] for valid k
            B = np.zeros(2*n-1)
            k_lo = j
            k_hi = j + n - 1
            B[k_lo:k_hi+1] = 2.0 * a  # a[0], a[1], ..., a[n-1] (with a[j]=0)
            # At k=2j, B should be 0 (quadratic term handles it)
            B[2*j] = 0.0
            
            D_idx = 2*j  # where the v^2 term is
            
            # Optimize v >= 0
            g = objective_1d(A, B, D_idx, n, S_other)
            
            # Find v_max: the function goes to 0 as v→∞ (numerator ~ v, denominator ~ v^2)
            # Actually numerator ~ max(B_k)*v for large v (B_k > 0), denominator ~ v^2
            # So g(v) ~ 2n*max(B_k)/v → 0 as v → ∞
            # But we want v in a reasonable range. Let's use [0, 5*max(a)]
            v_max = max(5.0 * np.max(a), 0.1)
            
            res = minimize_scalar(g, bounds=(0, v_max), method='bounded', 
                                  options={'xatol': 1e-8})
            v_new = res.x
            a[j] = v_new
            S = S_other + v_new
            
            # Check improvement
            c = A + B * v_new
            c[2*j] += v_new * v_new
            obj = 2.0 * n * c.max() / S**2
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                any_improved = True
        
        if verbose and sweep % 5 == 0:
            print(f"    Sweep {sweep}: obj={best_obj:.10f}")
        if not any_improved:
            break
    
    return best_a, best_obj
```

def coord_descent_fast(a_init, time_limit=60, verbose=False):
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    S = a.sum()
    c = fast_convolve(a)  # current convolution
    best_obj = 2.0 * n * c.max() / S**2
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_v = a[j]
            S_other = S - old_v
            
            # B_k = 2*a[k-j] for valid k (excluding the quadratic term)
            B = np.zeros(2*n-1)
            k_lo = j
            k_hi = j + n - 1
            # a[k-j] for k in [j, j+n-1] gives a[0] to a[n-1]
            B[k_lo:k_hi+1] = 2.0 * a
            B[2*j] = 0.0  # quadratic term handled separately
            
            # A_k = c_k - old_v * B_k - old_v^2 * D_k
            # D is 1 at position 2j
            A = c - old_v * B - old_v**2 * (np.arange(2*n-1) == 2*j).astype(float)
            # Actually simpler: A = c - old_v*B; A[2j] -= old_v**2
            A = c - old_v * B
            A[2*j] -= old_v**2
            
            # Now c_k(v) = A[k] + B[k]*v + v^2 * delta_{k,2j}
            # Optimize: 2n * max_k(A[k] + B[k]*v + v^2*delta_{k,2j}) / (S_other + v)^2
            
            D_idx = 2*j
            
            def g(v):
                vals = A + B * v
                vals[D_idx] += v * v
                mx = vals.max()
                return 2.0 * n * mx / (S_other + v) ** 2
            
            v_max = max(3.0 * np.max(a), 0.1)
            res = minimize_scalar(g, bounds=(0, v_max), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = max(res.x, 0.0)
            
            if abs(v_new - old_v) < 1e-12:
                continue
            
            # Update
            a[j] = v_new
            S = S_other + v_new
            # Update convolution incrementally
            c = A + B * v_new
            c[2*j] += v_new * v_new
            
            obj = 2.0 * n * c.max() / S**2
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                any_improved = True
        
        if verbose and sweep % 5 == 0:
            print(f"    Sweep {sweep}: obj={best_obj:.10f}")
        if not any_improved and sweep > 1:
            break
    
    return best_a, best_obj

            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweep % 20 == 0:
            print(f"      CD sweep {sweep}: obj={best_obj:.10f}")

    return best_a, best_obj


def coord_descent_pair(a_init, time_limit=60.0, verbose=False):
    """Coordinate descent that optimizes pairs of adjacent coordinates."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return a, np.inf
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for iteration in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a random pair of nearby indices
        j = rng.integers(0, n - 1)
        k = j + 1

        v1_old = a[j]
        v2_old = a[k]
        S_other = S - v1_old - v2_old

        # Build combined B vectors for both j and k
        B_j = np.zeros(2 * n - 1)
        B_j[j:j + n]:= 2.0 * a
        B_j[2 * j] = 0.0

        B_k = np.zeros(2 * n - 1)
        B_k[k:k + n] = 2.0 * a
        B_k[2 * k] = 0.0

        # A = c - v1*B_j - v1^2*delta_{2j} - v2*B_k - v2^2*delta_{2k}
        # But careful: B_j at position 2k and B_k at position 2j may overlap
        A = c.copy()
        A -= v1_old * B_j
        A[2 * j] -= v1_old * v1_old
        A -= v2_old * B_k
        A[2 * k] -= v2_old * v2_old

        D1 = 2 * j
        D2 = 2 * k

        def g2(v1, v2):
            vals = A + v1 * B_j + v2 * B_k
            vals[D1] += v1 * v1
            vals[D2] += v2 * v2
            denom = (S_other + v1 + v2)
            if denom < 1e-15:
                return 1e10
            return 2.0 * n * vals.max() / (denom * denom)

        # Nelder-Mead on the 2D problem
        from scipy.optimize import minimize as sp_minimize
        x0 = np.array([v1_old, v2_old])
        res = sp_minimize(
            lambda x: g2(max(x[0], 0), max(x[1], 0)),
            x0, method='Nelder-Mead',
            options={'xatol': 1e-12, 'fatol': 1e-12, 'maxiter': 200,
                     'adaptive': True}
        )
        v1_new = max(res.x[0], 0.0)
        v2_new = max(res.x[1], 0.0)

        if abs(v1_new - v1_old) + abs(v2_new - v2_old) < 1e-14:
            continue

        a[j] = v1_new
        a[k] = v2_new
        S = S_other + v1_new + v2_new
        c = A + v1_new * B_j + v2_new * B_k
        c[D1] += v1_new * v1_new
        c[D2] += v2_new * v2_new

        obj = 2.0 * n * c.max() / (S * S)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        if verbose and iteration % 100 == 0:
            print(f"      PairCD iter {iteration}: obj={best_obj:.10f}")

    return best_a, best_obj


def multi_start_cd(a_init, time_limit=60.0, n_restarts=3, seed=42):
    """Run coordinate descent with random restarts."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    best_a, best_obj = coord_descent(a, time_limit=time_limit / (n_restarts + 1))

    rng = np.random.default_rng(seed)
    for r in range(n_restarts):
        rem = time_limit - (time_limit / (n_restarts + 1)) * (r + 1)
        if rem < 5:
            break
        # Perturb
        if r == 0:
            noise = rng.standard_normal(n) * 0.01
            a_start = np.maximum(best_a + noise, 0)
        elif r == 1:
            # Blend with fresh init
            fresh = make_init(n, rng.integers(0, 13), seed + r)
            a_start = 0.8 * best_a + 0.2 * fresh
            a_start = np.maximum(a_start, 0)
        else:
            # Mirror blend
            a_start = 0.6 * best_a + 0.4 * best_a[::-1]
            noise = rng.standard_normal(n) * 0.005
            a_start = np.maximum(a_start + noise, 0)
        a_start = a_start / a_start.sum()

        a_opt, obj = coord_descent(a_start, time_limit=rem)
        if obj < best_obj:
            best_a = a_opt
            best_obj = obj

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization using exact coordinate descent.
    Explores diverse asymmetric initializations, multiple sequence lengths,
    and uses restarts to escape local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        # Scale to reasonable magnitude (objective is scale-invariant)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initialization screening ===
    print("=== Phase 1: Initialization Screening ===")
    lengths = [384, 512, 640, 768, 896, 1024]
    candidates = []

    for n_try in lengths:
        for kind in range(13):
            a_init = make_init(n_try, kind, seed)
            # Quick objective evaluation
            obj_quick = 2.0 * n_try * fast_convolve(a_init).max()
            candidates.append((obj_quick, a_init, f"init_k{kind}_n{n_try}"))

    if parent is not None:
        for n_try in [512, 768, 896, 1024]:
            a_p = interpolate_seq(parent, n_try)
            obj_quick = 2.0 * n_try * fast_convolve(a_p).max()
            candidates.append((obj_quick, a_p, f"parent_n{n_try}"))

    candidates.sort(key=lambda x: x[0])
    print(f"  Total candidates: {len(candidates)}, best quick obj: {candidates[0][0]:.4f}")

    # Screen top candidates with short coordinate descent
    screened = []
    for obj_q, a_c, label in candidates[:15]:
        rem = deadline - time.monotonic() - 600
        if rem < 10:
            break
        t_lim = min(8.0, rem / max(1, 15 - len(screened)))
        a_opt, obj = coord_descent(a_c, time_limit=t_lim)
        scored = 2.0 * len(a_opt) * fast_convolve(a_opt).max()
        screened.append((scored, a_opt, label))
        print(f"    {label}: quick={obj_q:.4f} -> cd={scored:.4f}")

    screened.sort(key=lambda x: x[0])

    # Try the best screened candidates with evaluate_sequence
    for obj_s, a_s, label in screened[:5]:
        try_seq(a_s, f"screen_{label}")

    if best_seq is not None:
        # Use the actual best found so far
        top_candidate = best_seq
    elif screened:
        top_candidate = screened[0][1]
    else:
        top_candidate = make_init(768, 1, seed)
        try_seq(top_candidate, "fallback_init")

    # === Phase 2: Extended coordinate descent on best ===
    print("\n=== Phase 2: Extended CD ===")
    t_allocs = [180, 120, 60, 30]
    for ta in t_allocs:
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_use = min(ta, rem)
        print(f"  CD t={t_use:.0f}s, n={len(top_candidate)}")
        a_opt, obj = coord_descent(top_candidate, time_limit=t_use, verbose=True)
        sc = try_seq(a_opt, f"cd_ext")
        if np.isfinite(sc) and sc <= best_score + 1e-6:
            top_candidate = a_opt

    # === Phase 3: Multi-length exploration ===
    print("\n=== Phase 3: Multi-length ===")
    for target_n in [512, 640, 768, 896, 1024, 1280]:
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        if len(top_candidate) == target_n:
            continue
        a_interp = interpolate_seq(top_candidate, target_n)
        sc = try_seq(a_interp, f"interp_n{target_n}")
        if sc < best_score + 0.02:
            t_use = min(40, rem * 0.4)
            a_opt, obj = coord_descent(a_interp, time_limit=t_use, verbose=False)
            sc2 = try_seq(a_opt, f"cd_interp_n{target_n}")

    # === Phase 4: Pair coordinate descent ===
    print("\n=== Phase 4: Pair CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_use = min(rem * 0.5, 80)
            print(f"  PairCD t={t_use:.0f}s")
            a_opt, obj = coord_descent_pair(best_seq, time_limit=t_use, verbose=True)
            try_seq(a_opt, "pair_cd")
            # Also try pair CD on the result
            if obj < 2 * len(best_seq) * fast_convolve(best_seq).max():
                top_candidate = a_opt
                rem2 = deadline - time.monotonic() - 60
                if rem2 > 15:
                    a_opt2, obj2 = coord_descent(a_opt, time_limit=min(rem2, 30))
                    try_seq(a_opt2, "cd_after_pair")

    # === Phase 5: Random restarts ===
    print("\n=== Phase 5: Restarts ===")
    n_best = len(best_seq) if best_seq is not None else 768
    for r in range(6):
        rem = deadline - time.monotonic() - 50
        if rem < 15:
            break
        t_use = min(20, rem * 0.35)

        if r == 0:
            noise = rng.standard_normal(n_best) * 0.01
            a_start = np.maximum(best_seq + noise, 0)
            a_start /= a_start.sum()
        elif r == 1:
            fresh = make_init(n_best, rng.integers(0, 13), seed + 100 + r)
            a_start = 0.85 * best_seq + 0.15 * fresh
            a_start = np.maximum(a_start, 0)
            a_start /= a_start.sum()
        elif r == 2:
            a_start = 0.5 * best_seq + 0.5 * best_seq[::-1]
            a_start /= a_start.sum()
        elif r == 3:
            shift = rng.integers(1, max(2, n_best // 8))
            a_shift = np.roll(best_seq, shift)
            a_start = 0.7 * best_seq + 0.3 * a_shift
            a_start = np.maximum(a_start, 0)
            a_start /= a_start.sum()
        elif r == 4:
            # Gaussian window perturbation
            center = rng.integers(n_best // 4, 3 * n_best // 4)
            sigma = n_best / 8
            pert = np.exp(-((np.arange(n_best) - center) ** 2) / (2 * sigma ** 2)) * 0.05
            a_start = np.maximum(best_seq + pert, 0)
            a_start /= a_start.sum()
        else:
            # Larger perturbation
            noise = rng.standard_normal(n_best) * 0.03
            a_start = np.maximum(best_seq + noise, 0)
            a_start /= a_start.sum()

        a_opt, obj = coord_descent(a_start, time_limit=t_use)
        try_seq(a_opt, f"restart_{r}")

    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_use = min(rem, 40)
            print(f"  Final CD: t={t_use:.0f}s")
            a_opt, obj = coord_descent(best_seq, time_limit=t_use, verbose=True)
            try_seq(a_opt, "final_cd")

            # One more pair pass
            rem2 = deadline - time.monotonic() - 10
            if rem2 > 10:
                a_opt2, obj2 = coord_descent_pair(best_seq, time_limit=min(rem2, 15))
                try_seq(a_opt2, "final_pair")

    # Fallback
    if best_seq is None:
        best_seq = make_init(768, 1, seed)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
