    Returns (g, sum_g) or (None, 0) if infeasible.
    """
    n = len(f)
    c = -np.ones(n)
    A = build_lp_matrix(n, f)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x, res.x.sum()
    return None, 0.0


def golden_section_line_search(f, g, max_iter=40):
    """Find t in [0,1] minimizing max(fast_convolve((1-t)*f + t*g)).
    Returns (t_opt, obj_opt).
    """
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    
    def eval_obj(t):
        ft = (1.0 - t) * f + t * g
        return fast_convolve(ft).max()
    
    a = 0.0
    b = 1.0
    c_pt = b - inv_phi * (b - a)
    d_pt = a + inv_phi * (b - a)
    fc = eval_obj(c_pt)
    fd = eval_obj(d_pt)
    
    for _ in range(max_iter):
        if abs(b - a) < 1e-14:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - inv_phi * (b - a)
            fc = eval_obj(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + inv_phi * (b - a)
            fd = eval_obj(d_pt)
    
    t_opt = (a + b) / 2.0
    obj_opt = eval_obj(t_opt)
    return t_opt, obj_opt


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with exact golden-section line search.
    Minimizes max_k (f*f)_k subject to f >= 0, sum(f) = 1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c_conv = fast_convolve(f)
    M = c_conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c_conv = fast_convolve(f)
        M = c_conv.max()
        
        g, S = solve_lp_direction(f, M)
        if g is None:
            break
        
        if S < 1e-10:
            break
        
        g_norm = g / S
        
        # Convergence check
        diff = np.max(np.abs(g_norm - f))
        if diff < 1e-14:
            if verbose:
                print(f"    LP-FW: converged at iter {it}, M={M:.12f}, diff={diff:.2e}")
            break
        
        # Line search
        t_opt, obj_opt = golden_section_line_search(f, g_norm)
        
        f_new = (1.0 - t_opt) * f + t_opt * g_norm
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        
        f = f_new
        new_M = fast_convolve(f).max()
        
        if new_M < best_obj - 1e-16:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    LP-FW iter {it}: M={best_obj:.12f}, t={t_opt:.4f}, S={S:.6f}")
        else:
            no_improve += 1
        
        # Stall detection
        if no_improve > 50:
            rng = np.random.default_rng(it * 12345 + 67890)
            noise = rng.standard_normal(n) * (0.005 / np.sqrt(no_improve / 50.0))
            f = np.maximum(best_f + noise, 0.0)
            f = f / f.sum()
            no_improve = 0
            best_obj = fast_convolve(f).max()
    
    if verbose:
        print(f"    LP-FW done: {it} iters, best M={best_obj:.12f}, time={time.monotonic()-t0:.1f}s")
    
    return best_f, best_obj


def make_raised_cosine(n):
    """Raised cosine (Hann) window, normalized."""
    x = np.linspace(-1.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window, normalized."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.7, power=6):
    """Bell-shaped window, normalized."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0.0, 1.0)
    a = 1.0 - t ** power
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_to_length(f, new_n):
    """Interpolate sequence f to new length new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Light smoothing
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0.0)
    s = f_smooth.sum()
    if s > 1e-15:
        f_smooth /= s
    return f_smooth


def lp_fw_optimize_multi_restart(f0, time_limit, n_restarts=3, verbose=False):
    """Run LP-FW with multiple restarts from perturbed initial points."""
    n = len(f0)
    rng = np.random.default_rng(42)
    
    best_f = f0.copy()
    best_obj = fast_convolve(f0).max()
    
    t0 = time.monotonic()
    
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        if r == 0:
            f_init = f0.copy()
        else:
            noise_scale = 0.01 * (r + 1)
            noise = rng.standard_normal(n) * noise_scale
            f_init = np.maximum(f0 + noise, 0.0)
            f_init = f_init / f_init.sum()
        
        t_rem = time_limit - (time.monotonic() - t0)
        t_alloc = max(10.0, t_rem * 0.6 / (n_restarts - r))
        
        a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=verbose)
        if obj < best_obj:
            best_obj = obj
            best_f = a_opt.copy()
    
    return best_f, best_obj


def coordinate_polish(f0, time_limit):
    """Coordinate descent polish: adjust individual coordinates to reduce max convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.4:
                    continue
                
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization and quick screening
    Phase 2: LP-FW with exact line search (main workhorse)
    Phase 3: Multi-length exploration
    Phase 4: Final coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
