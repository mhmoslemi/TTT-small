"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding 
with adaptive blending, exploiting the crucial insight that optimal sequences 
must be asymmetric. Combines LP iterations with coordinate descent for fine-tuning.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_lp_matrices(a):
    """Build the LP matrices for the direction-finding problem."""
    n = len(a)
    num_vars = n + 1  # n components of g + 1 for M
    num_ineq = 2 * n - 1
    
    # Build A_ub: for each k, sum_i g_i * a_{k-i} - M <= 0
    # Use the banded structure for efficiency
    A_ub = np.zeros((num_ineq, num_vars))
    
    for k in range(num_ineq):
        # g_i appears with coefficient a_{k-i} where max(0,k-n+1) <= i <= min(n-1,k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # For each i in [i_start, i_end], A_ub[k, i] = a[k - i]
        A_ub[k, i_start:i_end+1] = a[k - i_start : k - i_end : -1] if i_start != i_end else a[k - i_start]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(num_ineq)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c = np.zeros(num_vars)
    c[n] = 1.0
    
    bounds = [(0, None)] * n + [(0, None)]
    
    return c, A_ub, b_ub, A_eq, b_eq, bounds


def lp_direction(a):
    """
    Find g >= 0, sum(g)=1 that minimizes max_k (a*g)_k.
    Returns (g, M_opt) or (None, None) on failure.
    """
    n = len(a)
    c, A_ub, b_ub, A_eq, b_eq, bounds = build_lp_matrices(a)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    g = result.x[:n]
    g = np.maximum(g, 0)
    ss = g.sum()
    if ss < 1e-15:
        return None, None
    g = g / ss
    return g, result.x[n]


def coordinate_descent(a, time_limit, max_rounds=10, verbose=False):
    """
    Coordinate descent: for each coordinate j, find the best value
    by trying a range of multipliers.
    """
    a = np.array(a, dtype=np.float64)
    n = len(a)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return a
    a = a / ss
    
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for round_num in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try several multipliers
            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    improved = True
        
        if not improved and round_num > 0:
            break
    
    return a


def make_asymmetric_inits(n, rng):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding with 
adaptive blending. Key insight: optimal sequences must be asymmetric (symmetric 
sequences have lower bound 2). Uses scipy HiGHS solver for fast LP solves, 
combined with coordinate descent for fine-tuning and multi-restart strategy.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lp_direction(a):
    """
    Given current a (sum=1, a>=0), find g >= 0, sum(g)=1
    that minimizes max_k (a*g)_k.
    Returns (g, M_opt) or (None, None) on failure.
    """
    n = len(a)
    num_vars = n + 1
    num_ineq = 2 * n - 1

    A_ub = np.zeros((num_ineq, num_vars))
    for k in range(num_ineq):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, i] = a[k - i]
        A_ub[k, n] = -1.0

    b_ub = np.zeros(num_ineq)
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(num_vars)
    c[n] = 1.0
    bounds = [(0, None)] * n + [(0, None)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')

    if result.status != 0:
        return None, None

    g = result.x[:n]
    g = np.maximum(g, 0)
    ss = g.sum()
    if ss < 1e-15:
        return None, None
    g = g / ss
    return g, result.x[n]

def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with exact 1D search per coordinate."""
    a = np.array(a0, dtype=np.float64)
    n = len(a)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return a
    a = a / ss

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    for round_num in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 0.3]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and round_num > 0:
            break

    if verbose:
        print(f"    {label} CD done: obj={best_obj:.10f}")

    return best_a


def make_asymmetric_inits(n, rng):
    """Create various asymmetric initializations."""
    inits = []

    a = np.exp(-np.linspace(0, 2, n))
    inits.append((a / a.sum(), 'exp_decay'))

    a = np.exp(np.linspace(-2, 0, n))
    inits.append((a / a.sum(), 'exp_growth'))

    a = np.power(np.linspace(0.01, 1, n), 3)
    inits.append((a / a.sum(), 'power3'))

    a = np.power(np.linspace(0.01, 1, n), 7)
    inits.append((a / a.sum(), 'power7'))

    x = np.linspace(0, np.pi, n)
    a = np.sin(x) * np.exp(-0.5 * x)
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        inits.append((a / a.sum(), 'sine_exp'))

    a = np.ones(n)
    for _ in range(5):
        center = rng.integers(0, n)
        width = max(1, n // 4)
        a += np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0)
    inits.append((a / a.sum(), 'random_smooth'))

    mid = n * 0.3
    a = np.where(np.arange(n) <= mid,
                 (np.arange(n) + 1) / (mid + 1),
                 np.maximum(n - np.arange(n), 0) / (n - mid + 1))
    a = np.maximum(a, 0)
    inits.append((a / a.sum(), 'asym_triangle'))

    # Super-exponential
    a = np.exp(-np.linspace(0, 4, n) ** 2)
    inits.append((a / a.sum(), 'gauss_one_sided'))

    return inits

            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    n_base = 768
    if parent is not None:
        n_base = len(parent)

    # === Phase 1: LP from parent (main workhorse) ===
    print("\n=== Phase 1: LP from parent ===")
    if parent is not None:
        a = parent / parent.sum()
        remaining = deadline - time.monotonic()
        t_lp = remaining * 0.35
        print(f"  Time: {t_lp:.0f}s, n={n_base}")
        a_opt = lp_optimize(a, time_limit=t_lp, verbose=True, label="LP_parent")
        try_seq(a_opt, "LP_parent")

    # === Phase 2: CD refinement ===
    print("\n=== Phase 2: CD refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = min(remaining * 0.12, 120)
        print(f"  Time: {t_cd:.0f}s")
        a_opt = coordinate_descent(best_seq, time_limit=t_cd, verbose=True, label="CD1")
        try_seq(a_opt, "CD1")

    # === Phase 3: LP from diverse asymmetric inits ===
    print("\n=== Phase 3: LP from diverse inits ===")
    remaining = deadline - time.monotonic()
    init_list = []
    if parent is not None:
        a_p = parent / parent.sum()
        init_list.append((a_p, "parent"))
        noise = rng.standard_normal(n_base) * 0.01
        a_pert = np.maximum(a_p + noise, 0)
        a_pert = a_pert / a_pert.sum()
        init_list.append((a_pert, "parent_pert"))

    for a_init, label in make_asymmetric_inits(n_base, rng):
        init_list.append((a_init, label))

    n_inits = len(init_list)
    t_budget = remaining * 0.30
    t_per = min(t_budget / max(n_inits, 1), 25)

    for a_init, label in init_list:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        t_alloc = min(t_per, remaining - 10)
        if t_alloc < 3:
            break
        print(f"  LP from {label}: {t_alloc:.1f}s")
        a_opt = lp_optimize(a_init, time_limit=t_alloc, verbose=False, label=label)
        try_seq(a_opt, f"LP_{label}")

    # === Phase 4: LP refinement from current best ===
    print("\n=== Phase 4: LP refinement from best ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_lp2 = remaining * 0.25
            print(f"  Time: {t_lp2:.0f}s")
            a = best_seq / best_seq.sum()
            a_opt = lp_optimize(a, time_limit=t_lp2, verbose=True, label="LP_refine")
            try_seq(a_opt, "LP_refine")

    # === Phase 5: CD again ===
    print("\n=== Phase 5: Final CD ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_cd2 = min(remaining * 0.15, 60)
            print(f"  Time: {t_cd2:.0f}s")
            a_opt = coordinate_descent(best_seq, time_limit=t_cd2, verbose=True, label="CD2")
            try_seq(a_opt, "CD2")

    # === Phase 6: Try different sequence lengths ===
    print("\n=== Phase 6: Different lengths ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            for n_try in [n_base + 64, n_base - 64, n_base + 128, n_base - 128]:
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                if n_try < 64:
                    continue
                old_n = len(best_seq)
                old_x = np.linspace(0, 1, old_n)
                new_x = np.linspace(0, 1, n_try)
                a_interp = np.interp(new_x, old_x, best_seq / best_seq.sum())
                a_interp = np.maximum(a_interp, 0)
                ss = a_interp.sum()
                if ss < 1e-10:
                    continue
                a_interp = a_interp / ss

                t_alloc = min(20, remaining - 10)
                if t_alloc < 3:
                    break
                a_opt = lp_optimize(a_interp, time_limit=t_alloc, verbose=False, label=f"n{n_try}")
                try_seq(a_opt, f"LP_n{n_try}")

    # === Phase 7: Final short LP + CD polish ===
    print("\n=== Phase 7: Final polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a = best_seq / best_seq.sum()
            t_final = min(remaining - 5, 30)
            a_opt = lp_optimize(a, time_limit=t_final * 0.5, verbose=False, label="final_LP")
            try_seq(a_opt, "final_LP")
            a_opt = coordinate_descent(a_opt, time_limit=t_final * 0.5, verbose=False, label="final_CD")
            try_seq(a_opt, "final_CD")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
