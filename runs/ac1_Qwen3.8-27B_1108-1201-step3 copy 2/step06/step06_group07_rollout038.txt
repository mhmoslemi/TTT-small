def project_simplex(v):
    n=len(v)
    if n==1: return np.array([1.0])
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.arange(1,n+1)
    cond = (u - (cssv - 1.0)/rho) > 0
    if not cond.any(): return np.ones(n)/n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0)/(r+1.0)
    return np.maximum(v - theta, 0.0)
```

def lp_direction(f, M, max_time=5.0):
    if not HAVE_SCIPY: return None
    f = positive_normalize(f)
    n=len(f); m=2*n-1
    if M <= 0: return None
    try:
       data=[]; offsets=[]
       for d in range(n):
           data.append(np.full(n-d, f[d], dtype=np.float64))
           offsets.append(-d)
       A = diags(data, offsets, shape=(m,n), format='csc')
       c = -np.ones(n, dtype=np.float64)
       bub = np.full(m, M, dtype=np.float64)
       res = None
       try:
           res = linprog(c, A_ub=A, b_ub=bub, bounds=(0.0, None), method='highs', options={'time_limit': max_time})
       except Exception:
           res = None
       if res is None or not res.success or res.x is None:
           try:
               res2 = linprog(c, A_ub=A, b_ub=bub, bounds=(0.0, None), method='highs')
               if res2.success and res2.x is not None:
                   res = res2
           except Exception:
               pass
       if res is not None and res.success and res.x is not None:
           g = np.asarray(res.x, dtype=np.float64)
           g = np.maximum(g, 0.0)
           s = g.sum()
           if s > 1e-12:
               g /= s
               return g
    except Exception:
       return None
    return None

```python
"""
Search algorithm for minimizing:
    2*n*max(conv(a,a)) / (sum(a))^2
with a_i >= 0.

The program uses a portfolio strategy:
  * start from the previous best sequence when available,
  * interpolate to several larger lengths,
  * apply p-norm / smooth-maximum mirror descent with full FFT-based gradients,
  * periodically use the previous LP-direction idea from the state-of-the-art note,
  * finish with targeted active-peak descent and, for small n, coordinate descent.

All major functions are top-level; no lambdas or nested closures are used.
"""

import numpy as np
import time
import math

try:
    from scipy.sparse import diags
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def pow2ceil(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def positive_normalize(a, floor=1e-18):
    """Nonnegative normalization to sum 1."""
    v = np.asarray(a, dtype=np.float64).copy()
    if v.size == 0:
        return v
    v = np.maximum(v, floor)
    s = v.sum()
    if not np.isfinite(s) or s <= 1e-300:
        v = np.ones(v.size, dtype=np.float64)
        s = float(v.size)
    v /= s
    return v


def symmetrize(a):
    """Palindromic symmetrization, normalized."""
    a = positive_normalize(a)
    if len(a) <= 1:
        return a
    return positive_normalize(0.5 * (a + a[::-1]))


def interpolate_sequence(f, new_n):
    """Linear interpolation/resampling on [0,1], normalized."""
    f = positive_normalize(f)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    return positive_normalize(out)


def make_gaussian(n, sigma_frac=0.22, center_frac=0.5):
    """Gaussian-like sequence."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    x = t - center_frac
    a = np.exp(-0.5 * (x / sigma_frac) ** 2)
    return positive_normalize(a)


def make_exponential(n, rate=3.0):
    """One-sided exponential, asymmetric initializer."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = np.exp(-rate * t)
    return positive_normalize(a)


def make_random_ripple(n, rng, num=3, max_amp=0.25):
    """Nonnegative base-plus-cosine ripple initializer."""
    base = np.ones(n, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    for _ in range(num):
        freq = int(rng.integers(1, 6))
        amp = float(rng.uniform(0.05, max_amp))
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        base += amp * np.cos(2.0 * math.pi * freq * t + phase)
    return positive_normalize(base)


def approx_score(a):
    """Fast score estimate, assuming scale invariance."""
    a = positive_normalize(a)
    if len(a) == 0:
        return float("inf")
    b = fast_convolve(a)
    return float(2.0 * len(a) * b.max())


def score_of_sequence(a):
    """Call the global evaluator, used for final candidate acceptance."""
    try:
        vals = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(vals)
        if np.isfinite(sc):
            return float(sc)
    except Exception:
        return float("inf")
    return float("inf")


def candidate_key(item):
    """Tuple key for sorting candidate tuples by score."""
    return item[0]


def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    size = pow2ceil(L)
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:L]
    return np.maximum(b, 0.0)


def fast_convolve2(x, y):
    """Linear convolution of two real sequences via FFT."""
    n = len(x)
    m = len(y)
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    L = n + m - 1
    size = pow2ceil(L)
    return np.maximum(np.fft.irfft(np.fft.rfft(x, size) * np.fft.rfft(y, size), size)[:L], 0.0)


def weights_p(b, p):
    """Normalized weights proportional to b^(p-1); p=1 gives uniform."""
    L = len(b)
    if L == 0:
        return np.array([], dtype=np.float64)

    if p is None or p <= 1.01:
        s = b.sum()
        if s > 1e-300:
            return b / s
        w = np.zeros(L, dtype=np.float64)
        w[L // 2] = 1.0
        return w

    bmax = float(b.max())
    if bmax <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[np.argmax(b)] = 1.0
        return w

    w = np.zeros(L, dtype=np.float64)
    pos = b > 1e-300
    if not pos.any():
        w[np.argmax(b)] = 1.0
        return w

    log_b = np.log(b[pos])
    log_max = math.log(bmax)
    z = (float(p) - 1.0) * (log_b - log_max)
    z = np.where(z < -700.0, -700.0, z)
    w[pos] = np.exp(z)
    s = w.sum()
    if s <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[np.argmax(b)] = 1.0
    else:
        w /= s
    return w


def weights_lse(b, tau):
    """Softmax weights over peaks, with tau as fractional temperature."""
    L = len(b)
    if L == 0:
        return np.array([], dtype=np.float64)
    M = float(b.max())
    T = max(float(tau) * M, 1e-300)
    shifted = (b - M) / T
    mask = b >= M - 50.0 * T
    w = np.zeros(L, dtype=np.float64)
    vals = np.where(shifted[mask] < -700.0, -700.0, shifted[mask])
    w[mask] = np.exp(vals)
    idx = int(np.argmax(b))
    if w[idx] <= 0.0:
        w[idx] = 1.0
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w = np.zeros(L, dtype=np.float64)
        w[idx] = 1.0
    return w


def weights_active(b, tol=1e-4):
    """Uniform weights over peaks within tol of the maximum."""
    L = len(b)
    if L == 0:
        return np.array([], dtype=np.float64)
    M = float(b.max())
    threshold = M - max(float(tol) * M, 1e-12)
    mask = b >= threshold
    if not mask.any():
        mask = np.zeros(L, dtype=bool)
        mask[int(np.argmax(b))] = True
    w = np.zeros(L, dtype=np.float64)
    w[mask] = 1.0
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w[int(np.argmax(b))] = 1.0
    return w


def compute_b_and_grad(a, weight_mode="p", p=8.0, tau=0.1, active_tol=1e-4):
    """
    Compute b=conv(a,a) and normalized gradient of a weighted sum of b.

    For weights w_k,
        d/(da_j) sum_k w_k b_k = 2 sum_{r=0}^{n-1} w_{j+r} a_r.
    """
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64), np.array([], dtype=np.float64)

    L = 2 * n - 1
    # One FFT size large enough for both autocomvolution and cross-correlation.
    size = pow2ceil(3 * n - 2)

    F = np.fft.rfft(a, size)
    b = np.real(np.fft.irfft(F * F, size))[:L]
    b = np.maximum(b, 0.0)

    if weight_mode == "p":
        w = weights_p(b, p)
    elif weight_mode == "lse":
        w = weights_lse(b, tau)
    else:
        w = weights_active(b, active_tol)

    wr = w[::-1]
    Fw = np.fft.rfft(wr, size)
    cfull = np.real(np.fft.irfft(F * Fw, size))[:max(1, 3 * n - 2)]

    idx = L - 1 - np.arange(n, dtype=np.int64)
    c = cfull[idx]
    g = 2.0 * c

    gnorm = float(np.max(np.abs(g)))
    if gnorm > 1e-300:
        g /= gnorm
    else:
        g = np.zeros(n, dtype=np.float64)

    return b, g


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0], dtype=np.float64)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    r = int(np.nonzero(cond)[0][-1])
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def trial_update(a, g, lr, use_add):
    """One mirror or projected gradient trial step."""
    n = len(a)
    if n == 0:
        return a.copy()

    if use_add:
        gc = g - g.mean()
        gc_norm = float(np.max(np.abs(gc)))
        if gc_norm > 1e-300:
            gc = gc / gc_norm
        return project_simplex(a - lr * gc)

    logu = -lr * g
    logu = np.where(logu < -50.0, -50.0, logu)
    v = a * np.exp(logu)
    v = np.maximum(v, 1e-18)
    s = v.sum()
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return v / s


def log_norm(b, p):
    """Stable log Lp-norm of b. For p <= 1 use max."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return 0.0
    if p <= 1.1:
        return float(np.max(b))

    pos = b > 1e-300
    if not pos.any():
        return -1000.0

    lb = np.log(b[pos])
    mx = float(lb.max())
    z = float(p) * (lb - mx)
    z = np.where(z < -700.0, -700.0, z)
    ls = math.log(float(np.sum(np.exp(z))))
    return mx + ls / float(p)


def smooth_descent(a0, time_limit, weight_mode="p", p=8.0, lr_init=0.1,
                   upd_mode="mult", active_tol=1e-4, tau=0.1,
                   verbose=False, label=""):
    """
    Fast descent on p-norm / smooth-maximum surrogates.

    Returns:
        best_a, best_M, final_a, final_M
    """
    a = positive_normalize(a0)
    n = len(a)
    if n == 0:
        return a, 0.0, a, 0.0

    best_a = a.copy()
    b0 = fast_convolve(a)
    best_M = float(b0.max())
    final_M = best_M

    t0 = time.monotonic()
    lr = float(lr_init)
    no_improve = 0
    it = 0

    if n <= 1200:
        trial_factors = [1.0, 0.5, 0.25, 0.1, 0.05]
    else:
        trial_factors = [1.0, 0.5, 0.25]

    while time.monotonic() - t0 < time_limit:
        it += 1

        b, g = compute_b_and_grad(a, weight_mode=weight_mode, p=p,
                                  tau=tau, active_tol=active_tol)
        cur_M = float(b.max())
        if weight_mode == "active" or p <= 1.01:
            cur_val = cur_M
        else:
            cur_val = log_norm(b, p)

        use_add = False
        if upd_mode == "add":
            use_add = True
        elif upd_mode == "alt":
            use_add = (it % 20 == 0)

        best_trial_a = a
        best_trial_M = cur_M
        best_trial_val = cur_val
        found = False

        for fac in trial_factors:
            lr_t = lr * fac
            if lr_t <= 1e-12:
                continue

            a_t = trial_update(a, g, lr_t, use_add)
            b_t = fast_convolve(a_t)
            M_t = float(b_t.max())
            if weight_mode == "active" or p <= 1.01:
                val_t = M_t
            else:
                val_t = log_norm(b_t, p)

            if (val_t < best_trial_val - 1e-14) or \
               (abs(val_t - best_trial_val) <= 1e-14 and M_t < best_trial_M - 1e-15):
                best_trial_val = val_t
                best_trial_M = M_t
                best_trial_a = a_t

            if val_t < cur_val - 1e-10:
                found = True
                break

        a = best_trial_a
        M = best_trial_M
        val = best_trial_val
        final_M = M

        # Keep the trajectory from wandering too far away from the best max.
        if M > best_M * 1.25:
            a = best_a.copy()
            M = best_M
            val = cur_M if weight_mode == "active" or p <= 1.01 else log_norm(fast_convolve(a), p)
            final_M = M
            lr = max(lr * 0.5, 1e-5)
            no_improve += 50

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        elif val < cur_val - 1e-9:
            no_improve = 0
        else:
            no_improve += 1

        if val < cur_val - 1e-9:
            lr = min(lr * 1.05, 0.5)
        elif val > cur_val + 1e-8:
            lr = max(lr * 0.7, 1e-5)

        if verbose and it % 200 == 0:
            print(f"  {label} smooth {weight_mode} p={p:g} it={it} "
                  f"bestM={best_M:.10f} curM={M:.10f} lr={lr:.6f}")

        # Occasionally untie symmetric/asymmetric basins by symmetrization.
        if it % 250 == 0 and n > 1:
            a_sym = symmetrize(a)
            b_sym = fast_convolve(a_sym)
            M_sym = float(b_sym.max())
            if M_sym < best_M - 1e-14:
                best_M = M_sym
                best_a = a_sym.copy()
                final_M = M_sym
                no_improve = 0
            if M_sym < M * 0.99:
                a = a_sym.copy()
                final_M = M_sym

        if not found and no_improve > 100:
            lr = max(lr * 0.5, 1e-5)
            no_improve = 0

        if no_improve > 500:
            seed = (it * 1000003 + 17) % 2147483647
            rng = np.random.default_rng(seed)
            scale = 0.01 + 0.03 * min(1.0, no_improve / 2000.0)
            noise = rng.standard_normal(n)
            v = best_a * (1.0 + scale * noise)
            if n > 1:
                v = 0.95 * v + 0.05 * (np.ones(n, dtype=np.float64) / n)
            a = positive_normalize(v)
            lr = min(0.5, max(lr_init * 1.2, 1e-5))
            no_improve = 0

        if lr < 1e-6:
            if no_improve < 100:
                no_improve = 100
            else:
                break

    return best_a, best_M, a, final_M


def cascade_optimization(a0, time_limit, verbose=False, label=""):
    """
    Multi-stage p-norm / active-peak descent.

    Returns:
        best_a, best_M
    """
    a = positive_normalize(a0)
    overall_a = a.copy()
    overall_M = float(fast_convolve(a).max())

    stages = [
        # p, weight_mode, fraction, lr_init, update_mode
        (2.0, "p", 0.10, 0.18, "mult"),
        (4.0, "p", 0.15, 0.15, "mult"),
        (8.0, "p", 0.20, 0.12, "mult"),
        (16.0, "p", 0.20, 0.10, "alt"),
        (32.0, "p", 0.15, 0.08, "alt"),
        (64.0, "p", 0.10, 0.06, "alt"),
        (0.0, "active", 0.10, 0.04, "alt"),
    ]

    t_start = time.monotonic()
    for p, wmode, frac, lr, upd in stages:
        elapsed = time.monotonic() - t_start
        remaining = float(time_limit) - elapsed
        if remaining <= 2.0:
            break

        phase_time = float(time_limit) * frac
        if phase_time > remaining:
            phase_time = remaining
        if phase_time <= 0.0:
            continue

        best_phase_a, M_phase, final_a, final_M = smooth_descent(
            a,
            phase_time,
            weight_mode=wmode,
            p=p,
            lr_init=lr,
            upd_mode=upd,
            verbose=verbose,
            label=label
        )

        if M_phase < overall_M - 1e-14:
            overall_M = M_phase
            overall_a = best_phase_a.copy()

        if final_M <= max(overall_M, M_phase) * 1.03:
            a = final_a.copy()
        elif M_phase <= overall_M:
            a = best_phase_a.copy()
        else:
            a = overall_a.copy()

    return overall_a, overall_M


def lp_direction(f, M, max_time=5.0):
    """
    LP direction from the state-of-the-art note:
    maximize sum(g) subject to g >= 0 and (f*g)_k <= M.
    The normalized optimizer g/sum(g) minimizes max(f*h) among h>=0, sum(h)=1.
    """
    if not HAVE_SCIPY:
        return None

    f = positive_normalize(f)
    n = len(f)
    if n == 0 or M <= 1e-300:
        return None

    # A small floor improves LP conditioning and avoids useless huge directions.
    f = np.maximum(f, 1e-6)
    m = 2 * n - 1

    try:
        data = []
        offsets = []
        for d in range(n):
            data.append(np.full(n - d, f[d], dtype=np.float64))
            offsets.append(-d)

        A = diags(data, offsets, shape=(m, n), format="csc")
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(m, M, dtype=np.float64)

        res = None
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                          method="highs", options={"time_limit": max_time})
        except Exception:
            res = None

        if res is None or not res.success or res.x is None:
            try:
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                              method="highs")
            except Exception:
                res = None

        if res is not None and res.success and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        return None

    return None


def lp_descent(a0, time_limit, max_calls=20, verbose=False, label=""):
    """
    Iterate the LP-direction convex combination update.
    """
    a = positive_normalize(a0)
    best_a = a.copy()
    b0 = fast_convolve(a)
    best_M = float(b0.max())

    t0 = time.monotonic()
    calls = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and calls < max_calls:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break

        b = fast_convolve(a)
        M = float(b.max())
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()

        max_time = min(5.0, max(1.0, remaining - 5.0))
        g = lp_direction(a, M, max_time=max_time)
        if g is None:
            break

        calls += 1
        best_combo = a.copy()
        best_combo_M = M

        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            combo = (1.0 - t) * a + t * g
            combo = positive_normalize(combo)
            b_t = fast_convolve(combo)
            M_t = float(b_t.max())

            if M_t < best_combo_M - 1e-14:
                best_combo_M = M_t
                best_combo = combo.copy()

            if M_t < M * 0.995:
                break

        if best_combo_M < M - 1e-13:
            a = best_combo.copy()
            no_improve = 0
            if best_combo_M < best_M - 1e-14:
                best_M = best_combo_M
                best_a = a.copy()
                if verbose:
                    print(f"  {label} LP iter={calls} M={best_M:.10f}")
        else:
            no_improve += 1
            if no_improve == 1:
                sym = symmetrize(a)
                M_sym = float(fast_convolve(sym).max())
                if M_sym < best_M - 1e-14:
                    best_M = M_sym
                    best_a = sym.copy()
                    a = sym.copy()
                    no_improve = 0
            if no_improve > 3:
                break

    return best_a, best_M


def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """
    Small-n coordinate polish.
    For coordinate j, fix other ratios and try several mass allocations.
    """
    a = positive_normalize(a0)
    n = len(a)
    if n == 0:
        return a, 0.0

    best_a = a.copy()
    best_M = float(fast_convolve(a).max())
    t0 = time.monotonic()
    rounds = 0

    mults = [0.0, 0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.2, 1.5]

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old = float(a[j])
            if old <= 1e-12:
                continue

            denom = 1.0 - old
            if denom <= 1e-12:
                continue

            best_j_M = best_M
            best_j_x = old
            best_j_a = a

            for mult in mults:
                x = old * mult
                if x < 0.0 or x > 0.999:
                    continue
                if abs(x - old) <= 1e-15:
                    continue

                s = (1.0 - x) / denom
                trial = a * s
                trial[j] = x

                M_t = float(fast_convolve(trial).max())
                if M_t < best_j_M - 1e-15:
                    best_j_M = M_t
                    best_j_x = x
                    best_j_a = trial.copy()

            if best_j_x != old:
                a = best_j_a
                if best_j_M < best_M - 1e-15:
                    best_M = best_j_M
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

        if verbose and rounds % 5 == 0:
            print(f"  {label} CD round={rounds} M={best_M:.10f}")

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for the best nonnegative sequence within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(30.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float("inf")

    # Try to use the previous construction.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = p.ravel()
        p = np.maximum(p, 0.0)
        if p.size > 0 and np.isfinite(p).all() and float(p.sum()) > 1e-12:
            parent = p / float(p.sum())
    except Exception:
        parent = None

    if parent is not None:
        sc = score_of_sequence(parent)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = parent.copy()
            print(f"Parent sequence score: {sc:.6f}, n={len(parent)}")

    # Build a small candidate pool.
    cands = []

    if parent is not None:
        cands.append((approx_score(parent), parent.copy(), "parent"))
        parent_n = len(parent)
        for target in [1024, 2048, 4096, 8192]:
            if target > parent_n:
                a = interpolate_sequence(parent, target)
                cands.append((approx_score(a), a, f"parent_{target}"))

    # Asymmetric and ripple initializers.
    for n_try in [2048, 4096]:
        a = make_exponential(n_try, 3.0)
        cands.append((approx_score(a), a, f"exp3_{n_try}"))

        a = make_exponential(n_try, 6.0)
        cands.append((approx_score(a), a, f"exp6_{n_try}"))

        a = make_gaussian(n_try, 0.12, 0.25)
        cands.append((approx_score(a), a, f"gaussL_{n_try}"))

        a = make_gaussian(n_try, 0.12, 0.75)
        cands.append((approx_score(a), a, f"gaussR_{n_try}"))

        a = make_random_ripple(n_try, rng, 3, 0.25)
        cands.append((approx_score(a), a, f"rippleR_{n_try}"))

    if parent is None:
        for n_try in [1024, 2048]:
            a = make_exponential(n_try, 4.0)
            cands.append((approx_score(a), a, f"nomp_exp_{n_try}"))

    # Remove nonfinite and sort.
    good_cands = []
    for item in cands:
        if np.isfinite(item[0]):
            good_cands.append(item)
    cands = good_cands
    cands.sort(key=candidate_key)

    if len(cands) > 0 and best_seq is None:
        sc = score_of_sequence(cands[0][1])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cands[0][1].copy()

    # Select up to three diverse initial candidates, preferably different lengths.
    selected = []
    selected_labels = set()
    selected_ns = set()

    for item in cands:
        n_len = len(item[1])
        lab = item[2]
        if lab not in selected_labels and n_len not in selected_ns:
            selected.append(item)
            selected_labels.add(lab)
            selected_ns.add(n_len)
        if len(selected) >= 3:
            break

    for item in cands:
        if len(selected) >= 3:
            break
        if item[2] not in selected_labels:
            selected.append(item)
            selected_labels.add(item[2])
            selected_ns.add(len(item[1]))

    # Stage 1: optimize a few initial candidates.
    if time.monotonic() < deadline - 90:
        remaining = deadline - time.monotonic()
        init_budget = min(remaining * 0.55, 650.0)
        budgets = [init_budget * 0.50, init_budget * 0.30, init_budget * 0.20]

        for idx, item in enumerate(selected):
            remaining = deadline - time.monotonic()
            if remaining < 90:
                break
            alloc = budgets[idx] if idx < len(budgets) else 60.0
            alloc = min(alloc, remaining - 90)
            if alloc < 20:
                continue

            print(f"\nOptimizing initial candidate {item[2]}, n={len(item[1])}, time={alloc:.0f}s")
            a_opt, M_opt = cascade_optimization(item[1], alloc, verbose=True, label=item[2])
            sc = score_of_sequence(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.asarray(a_opt, dtype=np.float64).copy()
                print(f"  New best from {item[2]}: {sc:.6f}, n={len(a_opt)}")

    if best_seq is None:
        best_seq = make_gaussian(2048, 0.25)
        best_score = score_of_sequence(best_seq)

    # Stage 2: upscale the current global best in steps.
    for target in [2048, 4096, 8192]:
        remaining = deadline - time.monotonic()
        if remaining < 150:
            break
        if len(best_seq) < target:
            alloc = min((remaining - 120) * 0.30, 350.0)
            if alloc < 45:
                continue

            print(f"\nUpscaling best to n={target}, time={alloc:.0f}s")
            a0 = interpolate_sequence(best_seq, target)
            a_opt, M_opt = cascade_optimization(a0, alloc, verbose=True, label=f"upscale_{target}")
            sc = score_of_sequence(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.asarray(a_opt, dtype=np.float64).copy()
                print(f"  New best after upscaling to {target}: {sc:.6f}, n={len(a_opt)}")

    # Stage 3: LP-direction polish for moderate length.
    if best_seq is not None and len(best_seq) <= 2048:
        remaining = deadline - time.monotonic()
        if remaining > 80:
            alloc = min(remaining * 0.20, 200.0)
            if alloc > 30:
                print(f"\nLP-directory phase, n={len(best_seq)}, time={alloc:.0f}s")
                a_opt, M_opt = lp_descent(best_seq, alloc, max_calls=20,
                                          verbose=True, label="LP")
                sc = score_of_sequence(a_opt)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = np.asarray(a_opt, dtype=np.float64).copy()
                    print(f"  New best from LP: {sc:.6f}, n={len(a_opt)}")

    # Stage 4: final active-peak smoothing.
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 70:
            alloc = min(remaining * 0.35, 250.0)
            if alloc > 20:
                print(f"\nFinal active-peak smoothing, n={len(best_seq)}, time={alloc:.0f}s")
                a_opt, M_opt, a_end, M_end = smooth_descent(
                    best_seq,
                    alloc,
                    weight_mode="active",
                    p=0.0,
                    lr_init=0.045,
                    upd_mode="alt",
                    verbose=True,
                    label="final_active"
                )
                # Score both the phase-best and the endpoint; endpoint can be better in p-norm.
                for cand in [a_opt, a_end]:
                    sc = score_of_sequence(cand)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = np.asarray(cand, dtype=np.float64).copy()
                        print(f"  New best from final active: {sc:.6f}, n={len(best_seq)}")

    # Stage 5: coordinate descent for smaller sequences.
    if best_seq is not None and len(best_seq) <= 1024:
        remaining = deadline - time.monotonic()
        if remaining > 80:
            alloc = min(remaining * 0.40, 250.0)
            if alloc > 30:
                print(f"\nCoordinate polish, n={len(best_seq)}, time={alloc:.0f}s")
                a_opt, M_opt = coordinate_descent(best_seq, alloc,
                                                  verbose=True, label="CD")
                sc = score_of_sequence(a_opt)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = np.asarray(a_opt, dtype=np.float64).copy()
                    print(f"  New best from CD: {sc:.6f}, n={len(a_opt)}")

    # Final sanity and return.
    if best_seq is None:
        best_seq = np.ones(2048, dtype=np.float64) / 2048.0

    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    s = float(best_seq.sum())
    if s > 1e-12:
        best_seq = best_seq / s

    final_score = score_of_sequence(best_seq)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
