    """LP-based refinement step."""
    t0 = time.monotonic()

    # Constraint: A @ b <= M (element-wise)
    # This is a Toeplitz matrix
    
    # For efficiency, build the matrix
    m = 2 * n - 1  # number of constraints
    A = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[hi:lo-1:-1] if lo < hi else a[lo:hi+1]

    # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i]

    A = np.zeros((m, n))
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    
    # LP: maximize sum(b) subject to A @ b <= M, b >= 0
    c = -np.ones(n)  # minimize -sum(b)

        b_opt = res.x
        b_sum = b_opt.sum()
        if b_sum > 1e-10:

        # Top-k peak selection with softmax weighting
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        
        # Weight by value (higher peaks get more weight)
        top_vals = b[top_idx]
        w_top = top_vals / top_vals.sum()
        
        # Build full weight vector
        w = np.zeros(len(b))
        w[top_idx] = w_top
        
        # FFT-based subgradient
        g = cross_corr_fft(w, a)
        
        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        
        # Adaptive adjustments
        if no_improve > kick_every:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_every // 2:
            cur_lr *= 0.95
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ b)[k] = (a*b)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi+1] = a[k - j_hi:k - j_lo:-1]
    return A


def lp_refine(a, max_iter=5, time_limit=60.0):
    """LP-based refinement: find b that maximizes sum(b) subject to (a*b)[k] <= M."""
    t0 = time.monotonic()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    M = fast_convolve(a).max()
    A = build_conv_matrix(a)
    
    for _ in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: max sum(b) s.t. A @ b <= M, b >= 0
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=[(0, None)]*n, method='highs',
                         options={'time_limit': time_limit/3})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b_opt = res.x
        b_sum = b_opt.sum()
        if b_sum < 1e-10:
            break
        
        b_norm = b_opt / b_sum
        
        # Check if b_norm is better
        obj_b = fast_convolve(b_norm).max()
        
        # Try convex combinations
        best_combo = None
        best_combo_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1-t)*a + t*b_norm
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_combo_obj:
                best_combo_obj = obj_trial
                best_combo = a_trial
        
        if best_combo is not None:
            a = best_combo
            if best_combo_obj < best_obj:
                best_obj = best_combo_obj
                best_a = a.copy()
                M = fast_convolve(a).max()
                A = build_conv_matrix(a)
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, n_multipliers=10):
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.95:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial /= a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a /= a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj
