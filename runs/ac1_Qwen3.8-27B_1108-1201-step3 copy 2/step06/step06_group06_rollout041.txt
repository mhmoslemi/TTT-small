from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_obj_and_grad(b, tau):
    """Objective and gradient in softmax parameterization."""
    a = np.exp(b - b.max())
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    obj = tau * (np.log(w.sum()) + scaled.max())  # log-sum-exp
    
    # Gradient w.r.t. a
    # g_j = (2/tau) * sum_k w_k * a_{k-j}
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    h = np.fft.irfft(A * W, fft_size)
    g = (2.0 / tau) * h[n-1:2*n-1][::-1]
    
    # Gradient w.r.t. b: grad_b_j = a_j * (g_j - sum_i g_i * a_i)
    grad_b = a * (g - np.dot(g, a))
    
    return obj, grad_b

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(b):
    b_shifted = b - b.max()
    e = np.exp(b_shifted)
    return e / e.sum()


def smooth_objective(b, tau):
    """Compute smooth objective value (log-sum-exp of conv/tau)."""
    a = softmax(b)
    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = (conv - max_c) / tau
    return tau * (np.log(np.exp(scaled).sum()) + max_c / tau)


def smooth_grad_b(b, tau):
    """Gradient of smooth objective w.r.t. b (softmax parameterization)."""
    a = softmax(b)
    n = len(a)
    
    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = (conv - max_c) / tau
    w = np.exp(scaled)
    w /= w.sum()
    
    # g_j = (2/tau) * (w_rev * a)_j where (w_rev * a) is cross-correlation
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    h = np.fft.irfft(A * W, fft_size)
    g = (2.0 / tau) * h[n-1:2*n-1][::-1]
    
    # Chain rule through softmax
    grad_b = a * (g - np.dot(g, a))
    return grad_b


def smooth_full(b, tau):
    """Objective + gradient for scipy optimizer."""
    a = softmax(b)
    n = len(a)
    
    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = (conv - max_c) / tau
    obj = tau * (scaled[0] * 0 + np.log(np.exp(scaled).sum()))  # log-sum-exp relative to 0
    # Actually: logsumexp(conv/tau) = max_c/tau + log(sum(exp((conv-max_c)/tau)))
    obj = max_c / tau + np.log(np.exp(scaled).sum())
    # But we want the actual logsumexp, not the scaled version
    # obj = log(sum(exp(conv_k / tau))) = max_c/tau + log(sum(exp((conv_k - max_c)/tau)))
    
    w = np.exp(scaled)
    w /= w.sum()
    
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    h = np.fft.irfft(A * W, fft_size)
    g = (2.0 / tau) * h[n-1:2*n-1][::-1]
    
    grad_b = a * (g - np.dot(g, a))
    return obj, grad_b

            return smooth_full(b, tau)

            fun_and_grad, b, method='L-BFGS-B',
            options={'maxiter': 500, 'gtol': 1e-6, 'maxfun': 2000}
        )
        
        a_new = softmax(result.x)
        conv = fast_convolve(a_new)

    return best_a, 2 * n * best_obj


def peak_refine(a0, time_limit):
    """Refine using exact subgradient on the max-convolution."""
    n = len(a0)
    a = a0 / a0.sum()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find near-peak positions (within 1% of max)
        threshold = M * 0.99
        peak_mask = conv >= threshold
        p = peak_mask.astype(float)
        p /= p.sum()
        
        # Subgradient: g_j = 2 * sum_k p_k * a_{k-j}
        p_rev = p[::-1]
        total_len = n + len(p) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.fft.rfft(p_rev, fft_size)
        h = np.fft.irfft(A * P, fft_size)
        g = 2.0 * h[n-1:2*n-1][::-1]
        
        # Step size (decreasing)
        lr = 0.1 / (1 + 0.01 * it)
        
        a_new = a - lr * g
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        if conv_new.max() < best_obj - 1e-15:
            best_obj = conv_new.max()
            best_a = a_new.copy()
        
        a = a_new
        
        # Periodic restart
        if it % 200 == 0:
            a = project_simplex(best_a + np.random.default_rng(it).standard_normal(n) * 0.001)
    
    return best_a, 2 * n * best_obj


def coordinate_descent_refine(a0, time_limit):
    """Fine-grained coordinate descent on peak-reducing moves."""
    n = len(a0)
    a = a0 / a0.sum()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        peak_idx = int(np.argmax(conv))
        
        # Find the pair that contributes most to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        products = a[idx] * a[peak_idx - idx]
        j_star = idx[np.argmax(products)]
        k_star = peak_idx - j_star
        
        if j_star == k_star:
            # Diagonal: reduce a[j_star]
            for frac in [0.9, 0.8, 0.7, 0.6]:
                trial = a.copy()
                diff = trial[j_star] * (1 - frac)
                trial[j_star] *= frac
                # Redistribute to positions that don't contribute to the peak
                mask = np.ones(n, dtype=bool)
                mask[j_star] = False
                # Avoid positions that would increase other peaks
                for dk in range(-2, 3):
                    pos = j_star + dk
                    if 0 <= pos < n:
                        if abs(peak_idx - pos) <= 1:  # Close to peak
                            mask[pos] = False
                if mask.sum() > 0 and a[mask].sum() > 1e-15:
                    trial[mask] += diff * a[mask] / a[mask].sum()
                trial = np.maximum(trial, 0)
                s = trial.sum()
                if s > 1e-15:
                    trial /= s
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-15:
                    a = trial
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                    break
        else:
            # Off-diagonal: reduce one of the pair
            for target in [j_star, k_star]:
                for frac in [0.9, 0.8, 0.7]:
                    trial = a.copy()
                    diff = trial[target] * (1 - frac)
                    trial[target] *= frac
                    # Redistribute to "safe" positions
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    # Avoid positions near the peak
                    other = peak_idx - target
                    mask[max(0, other-1):min(n, other+2)] = False
                    if mask.sum() > 0 and a[mask].sum() > 1e-15:
                        trial[mask] += diff * a[mask] / a[mask].sum()
                    trial = np.maximum(trial, 0)
                    s = trial.sum()
                    if s > 1e-15:
                        trial /= s
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-15:
                        a = trial
                        best_obj = obj
                        best_a = a.copy()
                        no_improve = 0
                        break
                if no_improve == 0:
                    break
        
        no_improve += 1
        if no_improve > 50:
            # Perturb
            rng = np.random.default_rng(int(time.monotonic() * 10000))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
    
    return best_a, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. L-BFGS on softmax-parameterized log-sum-exp surrogate with multi-temperature annealing
2. Exact subgradient refinement on the true max-convolution

import numpy as np
from scipy.optimize import minimize as scipy_minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(b):
    """Stable softmax."""
    b_shifted = b - b.max()
    e = np.exp(b_shifted)
    return e / e.sum()


def compute_smooth_grad_wrt_b(b, tau):
    """
    Compute gradient of log-sum-exp(conv(a,a)/tau) w.r.t. b
    where a = softmax(b). Returns (objective_value, gradient_vector).
    """
    a = softmax_fn(b)
    n = len(a)

    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = (conv - max_c) / tau
    obj = max_c / tau + np.log(np.exp(scaled).sum())

    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum

    # g_j = (2/tau) * sum_k w_k * a_{k-j} (cross-correlation)
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    h = np.fft.irfft(A * W, fft_size)
    g = (2.0 / tau) * h[n - 1:2 * n - 1][::-1]

    # Chain rule through softmax: grad_b_j = a_j * (g_j - sum_i g_i * a_i)
    ga = np.dot(g, a)
    grad_b = a * (g - ga)

    return obj, grad_b


def subgradient_max_conv(a):
    """
    Compute a subgradient of max_k conv(a,a)_k with respect to a.
    Uses uniform distribution over near-max positions.
    Returns the subgradient vector.
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()

    # Use positions within 0.5% of max for a "fatter" subgradient
    threshold = M * 0.995
    peak_mask = conv >= threshold
    p = peak_mask.astype(np.float64)
    p /= p.sum()

    # Subgradient: g_j = 2 * sum_k p_k * a_{k-j}
    p_rev = p[::-1]
    total_len = n + len(p) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_rev, fft_size)
    h = np.fft.irfft(A * P, fft_size)
    g = 2.0 * h[n - 1:2 * n - 1][::-1]

    return g, M, conv


def lbfgs_anneal(a0, time_limit):
    """
    Multi-temperature L-BFGS annealing on softmax-parameterized smooth surrogate.
    """
    n = len(a0)
    a = a0 / a0.sum()
    b = np.log(a + 1e-30)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    # Temperature schedule: start high, decrease geometrically
    initial_conv = fast_convolve(a)
    tau = 0.1 * initial_conv.max()
    tau_min = 1e-5 * initial_conv.max()

    stage = 0
    while time.monotonic() - t0 < time_limit:
        stage += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed

        # Allocate time for this L-BFGS stage
        stage_time = min(remaining, 20 + 5 * stage)
        if stage_time < 5:
            break

        max_iter = int(200 * (stage_time / 20.0))
        max_iter = max(50, min(max_iter, 3000))

        def fun_and_grad(b_inner):
            return compute_smooth_grad_wrt_b(b_inner, tau)

        try:
            result = scipy_minimize(
                fun_and_grad, b, method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-8,
                         'maxfun': max_iter * 3, 'iprint': -1}
            )
            a_new = softmax_fn(result.x)
            conv_new = fast_convolve(a_new)
            if conv_new.max() < best_obj:
                best_obj = conv_new.max()
                best_a = a_new.copy()
                b = result.x.copy()
        except Exception as e:
            # If L-BFGS fails, continue with current b
            pass

        # Decrease temperature
        tau *= 0.5
        if tau < tau_min:
            tau = tau_min

        if stage % 3 == 0:
            print(f"    L-BFGS stage {stage}: tau={tau:.2e}, obj={2*n*best_obj:.8f}, time={elapsed:.0f}s")

    return best_a, 2 * n * best_obj


def exact_subgradient_refine(a0, time_limit):
    """
    Refine using exact subgradient of the true max-convolution.
    Uses decreasing step sizes and periodic restarts.
    """
    n = len(a0)
    a = a0 / a0.sum()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g, M, conv = subgradient_max_conv(a)

        # Adaptive step size
        base_lr = 0.5 / (n * (1 + 0.005 * it))
        a_new = a - base_lr * g
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Try a larger step occasionally
        if it % 10 == 0 and no_improve > 5:
            big_lr = 1.0 / (n * (1 + 0.005 * it))
            a_big = a - big_lr * g
            a_big = project_simplex(a_big)
            obj_big = fast_convolve(a_big).max()
            if obj_big < best_obj - 1e-15:
                best_obj = obj_big
                best_a = a_big.copy()
                a = a_big
                no_improve = 0

        # Periodic perturbation to escape local minima
        if no_improve > 100:
            rng = np.random.default_rng(it * 7 + 3)
            noise_scale = 0.001 * min(no_improve / 50.0, 3.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            no_improve = 0

        if it % 200 == 0:
            print(f"    Subgrad iter {it}: obj={2*n*best_obj:.8f}, lr={base_lr:.2e}")

    return best_a, 2 * n * best_obj


def targeted_pair_swap(a0, time_limit):
    """
    Targeted improvement: find the main peak, identify the largest contributing pair,
    and try to swap mass to reduce it.
    """
    n = len(a0)
    a = a0 / a0.sum()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find the top contributing pair to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        products = a[idx] * a[peak_k - idx]
        top_pairs = np.argsort(products)[::-1][:3]

        improved_this_round = False

        for pi in range(min(3, len(top_pairs))):
            j1 = int(idx[top_pairs[pi]])
            j2 = int(peak_k - j1)

            if j1 == j2:
                # Diagonal: reduce a[j1]
                for frac in [0.95, 0.85, 0.75, 0.6]:
                    trial = a.copy()
                    diff = trial[j1] * (1 - frac)
                    trial[j1] *= frac
                    # Redistribute to positions far from the peak
                    mask = np.ones(n, dtype=bool)
                    # Avoid positions that would increase the same peak
                    for dk in range(-2, 3):
                        pos = j1 + dk
                        if 0 <= pos < n:
                            if abs((peak_k - pos)) <= 1:
                                mask[pos] = False
                    mask[j1] = False
                    if mask.sum() > 1 and a[mask].sum() > 1e-15:
                        trial[mask] += diff * a[mask] / a[mask].sum()
                    trial = np.maximum(trial, 0)
                    s = trial.sum()
                    if s > 1e-15:
                        trial /= s
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-15:
                        a = trial
                        best_obj = obj
                        best_a = a.copy()
                        no_improve = 0
                        improved_this_round = True
                        break
            else:
                # Off-diagonal: reduce one of the pair
                for target in [j1, j2]:
                    for frac in [0.95, 0.85, 0.75]:
                        trial = a.copy()
                        diff = trial[target] * (1 - frac)
                        trial[target] *= frac
                        # Redistribute
                        mask = np.ones(n, dtype=bool)
                        mask[target] = False
                        other = peak_k - target
                        if 0 <= other < n:
                            mask[max(0, other - 1):min(n, other + 2)] = False
                        if mask.sum() > 1 and a[mask].sum() > 1e-15:
                            trial[mask] += diff * a[mask] / a[mask].sum()
                        trial = np.maximum(trial, 0)
                        s = trial.sum()
                        if s > 1e-15:
                            trial /= s
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-15:
                            a = trial
                            best_obj = obj
                            best_a = a.copy()
                            no_improve = 0
                            improved_this_round = True
                            break
                    if improved_this_round:
                        break

            if improved_this_round:
                break

        if not improved_this_round:
            no_improve += 1
            if no_improve > 30:
                # Try random pair swaps
                rng = np.random.default_rng(int(time.monotonic() * 100000))
                for _ in range(20):
                    i, j = rng.integers(0, n, size=2)
                    if i == j:
                        continue
                    trial = a.copy()
                    amt = min(a[i], a[j]) * 0.1
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0)
                    s = trial.sum()
                    if s > 1e-15:
                        trial /= s
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-15:
                        a = trial
                        best_obj = obj
                        best_a = a.copy()
                        no_improve = 0
                        break
                if no_improve > 80:
                    # Larger perturbation
                    noise_scale = 0.003 * min(no_improve / 30.0, 5.0)
                    a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                    no_improve = 0

    return best_a, 2 * n * best_obj


def local_3point_cd(a0, time_limit):
    """
    Local 3-point coordinate descent: for each position i, try moving
    mass to i-1 and i+1 in various proportions.
    """
    n = len(a0)
    a = a0 / a0.sum()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        for i in range(1, n - 1):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            # Try moving fractions of a[i] to neighbors
            for frac in [0.1, 0.3, 0.5]:
                for direction in [0, 1]:  # 0: left, 1: right
                    j = i - 1 if direction == 0 else i + 1
                    if j < 0 or j >= n:
                        continue
                    trial = a.copy()
                    amt = a[i] * frac
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0)
                    s = trial.sum()
                    if s > 1e-15:
                        trial /= s
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-15:
                        a = trial
                        best_obj = obj
                        best_a = a.copy()
                        no_improve = 0

        no_improve += 1
        if no_improve > 5:
            break

    return best_a, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    """Asymmetric Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Flat-topped bell (raised cosine) initialization."""
    x = np.arange(n) - (n - 1) / 2.0
    half = n * width_frac / 2.0
    t = np.abs(x) / half
    a = np.where(t <= 1.0, 0.5 * (1 + np.cos(np.pi * t)), 0.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

            a_opt, obj = exact_subgradient_refine(best_state['seq'], time_limit=t9)
            try_seq(a_opt, "final_subgrad")

    # Fallback
    if best_state['seq'] is None:
        best_state['seq'] = np.ones(768) / 768
        best_state['score'] = float(evaluate_sequence(best_state['seq'].tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_state['seq'].tolist()))
    if np.isfinite(final_score):
        best_state['score'] = final_score

    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])})")
    return [float(x) for x in best_state['seq']]
