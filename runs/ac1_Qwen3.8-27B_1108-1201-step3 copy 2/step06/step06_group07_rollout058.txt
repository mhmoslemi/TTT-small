Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B optimization with

mirror descent polishing, multiple sequence lengths, and diverse initializations.

def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)

def compute_grad_a_fft(a, w):
    """Compute 2 * cross_correlation(a, w) using FFT."""
    n = len(a)
    total_len = n + len(w) - 1
    N = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(A * np.conj(W), N)
    result = np.zeros(n)
    result[0] = cross[0]
    if n > 1:

def soft_objective_and_grad(x, beta):
    """Compute smooth approximation of 2n*max(conv) and its gradient.
    x parameterizes a = softmax(x), so sum(a)=1 and a>0 automatically."""
    a = np.exp(x - x.max())
    a_sum = a.sum()
    a = a / a_sum
    n = len(a)
    
    conv = fast_convolve(a)
    
    # Softmax weights
    scaled = beta * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    w = exp_scaled / sum_exp
    
    # Smooth objective: 2n * (1/beta) * log(sum(exp(beta*conv)))
    obj = 2.0 * n * np.log(sum_exp) / beta
    
    # Gradient w.r.t. a
    grad_a = compute_grad_a_fft(a, w)
    
    # Chain rule through softmax: d(softmax(x))/dx = diag(a) - a*a^T
    # grad_x = a * (grad_a - dot(a, grad_a))
    dot_ag = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot_ag)
    
    return obj, grad_x

def lbfgs_optimize(a_init, time_limit, beta=500.0, max_iter=2000):

    # Convert to x parameterization: a = softmax(x), so x = log(a) (approximately)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    x0 = np.log(a)  # log parameterization

    result = minimize(
        soft_objective_and_grad,
        x0,
        args=(beta,),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'gtol': 1e-6, 'maxfun': 5000}
    )
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with softmax parameterization,
soft-max smoothed max, mirror descent polishing, multi-scale search, and diverse
initializations. Key improvement: uses second-order (quasi-Newton) optimization
via scipy L-BFGS-B which is much faster per iteration than first-order methods.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_a_fft(a, w):
    """Compute 2 * sum_k w[k] * a[k-j] for j=0..n-1 using FFT cross-correlation."""
    n = len(a)
    total_len = n + len(w) - 1
    N = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(A * np.conj(W), N)
    result = np.zeros(n)
    result[0] = cross[0]
    if n > 1:
        result[1:] = cross[N-1 : N-n : -1]
    return 2.0 * result


def soft_obj_grad(x, beta):
    """
    Smooth objective 2n/beta * log(sum exp(beta*conv(softmax(x))))
    and its gradient w.r.t. x.
    """
    a = np.exp(x - x.max())
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    
    scaled = beta * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    w = exp_scaled / sum_exp
    
    obj = 2.0 * n * np.log(sum_exp) / beta
    
    grad_a = compute_grad_a_fft(a, w)
    dot_ag = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot_ag)
    
    return obj, grad_x


def lbfgs_run(a_init, beta=500.0, max_iter=2000, gtol=1e-7):
    """Run L-BFGS-B from initialization a_init (must be non-negative, sum=1)."""
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    x0 = np.log(a)
    
    result = minimize(
        soft_obj_grad,
        x0,
        args=(beta,),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'gtol': gtol, 'maxfun': 5000, 'ftol': 1e-10}
    )
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt

def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
