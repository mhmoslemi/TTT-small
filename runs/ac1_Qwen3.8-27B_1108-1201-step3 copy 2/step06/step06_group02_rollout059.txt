def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)

    return np.maximum(v - theta, 0.0)

    Select target positions j that minimize interference with existing peaks.

    # interference(j) = sum over peaks k of f[k-j]^2 (clamped to valid range)
    
    # Efficient computation using cross-correlation
    # f[k-j] for k in peak_indices is a gather operation
    # For all j, this is like a cross-correlation of f with a sparse kernel

        k = int(k)

        # Valid range: 0 <= k-j < n, i.e., k-n+1 <= j <= k

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)

    if ss > 1e-15:

def spectral_flatten(f, n_iterations=1000, verbose=False):
    n = len(f)
    fft_size = 1 << (2*n - 2).bit_length()
    a = f.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()

        A = np.fft.rfft(a, fft_size)
        S = np.abs(A) ** 2
        
        # Compute convolution (inverse DFT of S / fft_size)
        conv = np.fft.irfft(A * A, fft_size)[:2*n-1]
        
        # Find peak
        peak_k = np.argmax(conv)
        
        # The peak is at position peak_k in the convolution
        # In the frequency domain, the peak at position k is:
        # conv[k] = (1/fft_size) * sum_m S[m] * exp(2*pi*i*m*k/fft_size)
        # To reduce conv[peak_k], we want to reduce S[m] weighted by exp(2*pi*i*m*peak_k/fft_size)
        
        # Simple approach: reduce S at frequencies that constructively interfere at peak_k
        # weight[m] = exp(2*pi*i*m*peak_k/fft_size) (real part for the contribution to conv[peak_k])
        
        # Actually, conv[peak_k] = sum_m |A[m]|^2 * exp(2*pi*i*m*peak_k/fft_size) / fft_size
        # The real part of exp(2*pi*i*m*peak_k/fft_size) = cos(2*pi*m*peak_k/fft_size)
        # So frequencies where cos is positive contribute positively to the peak
        
        # Reduce S where cos is positive (constructive interference at peak)
        m = np.arange(len(S))
        cos_vals = np.cos(2 * np.pi * m * peak_k / fft_size)

    if wsum < 1e-30:

def interference_score(f, peak_indices):
    """
    For each position j, compute how much adding mass at j would
    increase the top peaks. Low score = good target for receiving mass.
    """
    n = len(f)
    scores = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # Adding delta at j changes conv[k] by delta * f[k-j]
        # So score[j] += f[k-j]^2 for valid j
        idx = np.arange(k - je, k - js + 1)
        scores[js:je + 1] += f[idx] ** 2
    return scores

def golden_section_1d(func, lo, hi, n_iter=15):
    """Golden section minimization of func on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = lo
    b = hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(n_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = func(d)
    return (a + b) / 2.0, min(fc, fd)

                        delta = f[top_src] * frac
                        if delta <= 0 or delta >= f[top_src]:
                            return best_obj + 1.0
                        trial = f.copy()
                        trial[top_src] -= delta
                        trial[target] += delta
                        trial = np.maximum(trial, 0)
                        ts = trial.sum()
                        if ts < 1e-15:
                            return best_obj + 1.0
                        trial = trial / ts
                        return fast_convolve(trial).max()

                    opt_frac, opt_val = golden_section_1d(obj_frac, 0.0, 0.85, n_iter=12)

                    if opt_val < M - 1e-13:
                        delta = f[top_src] * opt_frac
                        f = f.copy()
                        f[top_src] -= delta
                        f[target] += delta
                        f = np.maximum(f, 0)
                        f = f / f.sum()
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if found:
            co = fast_convolve(f).max()
            if co < best_obj - 1e-13:
                best_obj = co
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"      FP iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                rng = np.random.default_rng(it * 13 + 7)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 0)
                f = f / f.sum()

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Bilinear interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2*n*max(conv)/(sum)^2.
    Phases: Init -> Mirror Descent -> Pair Descent -> Fine Polish -> Length Explore -> Squeeze
    """
    budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + budget - 10.0

    best_seq = None
    best_score = float('inf')

    def _try(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768)
        _try(parent, "default_gauss")

    n_base = len(parent)
    t_start = time.monotonic()

    # === Phase 1: Diverse initialization (25s) ===
    print("\n=== Phase 1: Initialization ===")
    init_list = []
    for n_try in [n_base]:
        init_list.append(("gauss", make_gaussian(n_try, 0.22)))
        init_list.append(("gauss_w", make_gaussian(n_try, 0.35)))
        init_list.append(("bell", make_bell(n_try, 0.55)))
        init_list.append(("cos1", make_cosine_mod(n_try, 1, 0.35)))
        init_list.append(("cos2", make_cosine_mod(n_try, 2, 0.28)))
        init_list.append(("cos3", make_cosine_mod(n_try, 3, 0.2)))
        init_list.append(("multi3", make_multi_cosine(n_try, 3, 0.25, seed=42)))
        init_list.append(("multi5", make_multi_cosine(n_try, 5, 0.2, seed=99)))
        init_list.append(("edge", make_edge_concentrated(n_try, 1.6)))
        init_list.append(("arcosh", make_arcosh(n_try, 1.2)))
        init_list.append(("wigner", make_wigner(n_try, seed=7)))
        init_list.append(("flat", np.ones(n_try) / n_try))
        init_list.append(("parent", parent))

    scored_inits = []
    for label, a_init in init_list:
        sc = objective_normalized(a_init)
        scored_inits.append((sc, label, a_init))
        if sc < 2.1:
            _try(a_init, f"init_{label}")

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Top 5: {[(lbl, f'{sc:.5f}') for sc, lbl, _ in scored_inits[:5]]}")
    print(f"  Phase 1 time: {time.monotonic()-t_start:.1f}s")

    # === Phase 2: Mirror descent on best initializations (700s) ===
    print("\n=== Phase 2: Mirror Descent ===")
    for trial_idx in range(min(3, len(scored_inits))):
        rem = deadline - time.monotonic() - 700
        if rem < 40:
            break
        sc, label, a_init = scored_inits[trial_idx]
        if len(a_init) != n_base:
            a_init = interpolate_sequence(a_init, n_base)
        t_alloc = min(rem * 0.45, 280)
        if t_alloc < 20:
            break
        print(f"  MD [{label}] alloc={t_alloc:.0f}s (init_obj={sc:.6f})")
        a_opt, obj = mirror_descent_mp(a_init, time_limit=t_alloc, n_peaks=5, verbose=True, lr_init=0.007)
        _try(a_opt, f"md_{label}")

    # Also run MD on the current best if it improved
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t_alloc = min(rem * 0.5, 200)
            print(f"  MD [best] alloc={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_mp(best_seq, time_limit=t_alloc, n_peaks=7, verbose=True, lr_init=0.005)
            _try(a_opt, "md_best")

    # === Phase 3: Interference-aware pair descent (450s) ===
    print("\n=== Phase 3: Pair Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem * 0.8, 350)
            print(f"  PSD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = pair_descent_smart(best_seq, time_limit=t_alloc, n_peaks=5, verbose=True)
            _try(a_opt, "psd")

    # === Phase 4: Fine golden-section polish (350s) ===
    print("\n=== Phase 4: Fine Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.85, 300)
            print(f"  FP: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = fine_polish_golden(best_seq, time_limit=t_alloc, verbose=True)
            _try(a_opt, "fp")

    # === Phase 5: Multi-length exploration (150s) ===
    print("\n=== Phase 5: Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_lengths = []
        for factor in [0.8, 0.9, 1.1, 1.25]:
            tn = int(base_n * factor)
            if tn >= 256 and tn <= 2048 and tn != base_n and tn not in target_lengths:
                target_lengths.append(tn)
        # Also try some absolute values
        for tn in [640, 896, 1024, 1280]:
            if tn != base_n and tn not in target_lengths and 256 <= tn <= 2048:
                target_lengths.append(tn)

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = _try(f_interp, f"interp_n{target_n}")

            if sc < best_score + 0.03 and rem > 40:
                t_alloc = min(rem * 0.35, 50)
                if t_alloc > 15:
                    print(f"  MD n={target_n}: alloc={t_alloc:.0f}s")
                    a_opt, obj = mirror_descent_mp(f_interp, time_limit=t_alloc, n_peaks=5)
                    _try(a_opt, f"md_n{target_n}")

                    rem2 = deadline - time.monotonic() - 20
                    if rem2 > 20:
                        t_alloc2 = min(rem2 * 0.5, 30)
                        a_opt2, obj2 = fine_polish_golden(a_opt, time_limit=t_alloc2)
                        _try(a_opt2, f"fp_n{target_n}")

    # === Phase 6: Final squeeze (remaining time) ===
    print("\n=== Phase 6: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 3
        if rem > 15:
            t_alloc = min(rem, 40)
            print(f"  Final FP: alloc={t_alloc:.0f}s")
            a_opt, obj = fine_polish_golden(best_seq, time_limit=t_alloc)
            _try(a_opt, "final_fp")

        rem = deadline - time.monotonic() - 2
        if rem > 10:
            t_alloc = min(rem, 20)
            print(f"  Final MD: alloc={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_mp(best_seq, time_limit=t_alloc, n_peaks=8, lr_init=0.003)
            _try(a_opt, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {total_time:.1f}s")

    return [float(x) for x in best_seq]
