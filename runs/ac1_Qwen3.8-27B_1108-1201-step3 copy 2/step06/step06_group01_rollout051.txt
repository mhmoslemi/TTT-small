"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Smooth-max projected gradient descent (log-sum-exp smoothing)
2. LP-based best response (fictitious play) for global structure
3. Peak-shaving local search for final polish
4. Multi-length exploration

The key innovation over the previous approach is the use of a direct
gradient-based optimizer on the smooth approximation of the max auto-convolution,
which can capture local improvements that the LP/fictitious play might miss.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_grad(a, tau):
    """
    Compute the smooth max of conv(a,a) and its gradient with respect to a.
    
    F(a) = tau * log(sum_k exp(conv(a,a)_k / tau))
    
    Returns (F, grad) where grad = dF/da.
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    
    # Log-sum-exp computation
    b_max = b.max()
    log_sum_exp = b_max / tau + np.log(np.sum(np.exp((b - b_max) / tau)))
    F = tau * log_sum_exp
    
    # Softmax weights
    w = np.exp((b - b_max) / tau)
    w /= w.sum()
    
    # Gradient: grad_i = 2 * sum_k w_k * a_{k-i}
    # This is the cross-correlation of w (length 2n-1) and a (length n).
    # Use FFT: cross_corr = IFFT(conj(FFT(w)) * FFT(a)) with proper padding.
    
    # Linear cross-correlation: (w cross_corr a)[i] = sum_k w[k] * a[k-i]
    # = (reverse(w) * a)[len(w)-1-i] (linear convolution)
    
    rev_w = w[::-1]
    conv_size = len(rev_w) + n - 1  # = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (conv_size - 1).bit_length()
    
    RW = rfft(rev_w, fft_size)
    A = rfft(a, fft_size)
    conv_result = irfft(RW * A, fft_size)[:conv_size]
    
    # (rev_w * a)[m] for m = 0, ..., conv_size-1
    # We want (rev_w * a)[(2n-2)-i] for i = 0, ..., n-1
    # That is indices (2n-2), (2n-3), ..., (n-1)
    # Which is conv_result[n-1 : 2n-1] reversed
    
    cross_corr = conv_result[n-1 : 2*n-1][::-1]
    grad = 2.0 * cross_corr
    
    return F, grad

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:

import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient.
    F(a) = tau * log(sum_k exp(conv(a,a)_k / tau))
    Returns (value, gradient).
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    b_max = b.max()
    exp_terms = np.exp((b - b_max) / tau)
    w = exp_terms / exp_terms.sum()
    F = tau * (b_max / tau + np.log(exp_terms.sum()))
    
    # Gradient: grad_i = 2 * sum_k w_k * a_{k-i}
    # = 2 * (rev_w * a)_{(2n-2)-i} where * is linear convolution
    rev_w = w[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    RW = rfft(rev_w, fft_size)
    A = rfft(a, fft_size)
    conv_res = irfft(RW * A, fft_size)[:conv_len]
    cross_corr = conv_res[n - 1: 2 * n - 1][::-1]
    grad = 2.0 * cross_corr
    return F, grad

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T
