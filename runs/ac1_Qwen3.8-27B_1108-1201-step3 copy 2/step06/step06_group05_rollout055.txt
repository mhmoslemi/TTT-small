from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient_fft(f, c, beta):
    """Compute the smoothed subgradient using log-sum-exp."""
    n = len(f)
    scaled = beta * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    w_rev = w[::-1].copy()
    conv_len = n + (2*n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2*n-2, 2*n-2-n, -1)
    return 2.0 * conv[indices]

def mirror_descent(f0, time_limit, beta_schedule=None, verbose=False):
    """Fast mirror descent with adaptive beta."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    if beta_schedule is None:
        beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512]
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        
        lr = 0.1
        no_improve = 0
        it = 0
        stage_start = time.monotonic()
        
        while time.monotonic() - stage_start < (time_limit * 0.07) and time.monotonic() - t0 < time_limit * 0.9:
            it += 1
            c = fast_convolve(f)
            M = c.max()

                best_f = f.copy()
                no_improve = 0
            
            g = compute_subgradient_fft(f, c, beta)
            
            # Multiplicative update
            f_new = f * np.exp(-lr * g)
            f_new = np.maximum(f_new, 1e-15)
            f_new = f_new / f_new.sum()
            
            c_new = fast_convolve(f_new)
            obj_new = c_new.max()
            
            if obj_new < M + 1e-14:
                f = f_new
                no_improve = 0
            else:

                lr *= 0.8
            
            if verbose and it % 200 == 0:
                print(f"    MD beta={beta} it={it}: obj={M:.10f}, best={best_obj:.10f}, lr={lr:.4f}")

        if verbose:

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast multiplicative weight
updates with FFT-based subgradients, multi-length exploration, and LP-FW polish.
Key innovations: (1) soft-max weighted subgradients for stable peak targeting,
(2) adaptive learning rate with geometric decay, (3) diverse asymmetric
initializations, (4) efficient LP-FW using structured constraints.
"""
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_weighted_gradient(f, c, beta):
    """Compute gradient of soft-max smoothed max of c, where c = conv(f,f)."""
    n = len(f)
    scaled = beta * (c - c.max())
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient: grad_j = 2 * sum_k w[k] * f[k-j] for valid k-j
    # Use FFT: grad = 2 * (f convolved with w_reversed)[specific indices]
    w_rev = w[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    return 2.0 * conv[indices]

def multiplicative_update(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()

        it = 0

            M = c.max()

            g = compute_weighted_gradient(f, c, beta)
