import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def correct_gradient(f, peak_k):
    """Compute the correct gradient of conv[f,f][peak_k] w.r.t. f."""
    n = len(f)
    js = max(0, peak_k - n + 1)
    je = min(n - 1, peak_k)
    if je < js:
        return np.zeros(n)
    grad = np.zeros(n)
    # For j in [js, je]: d g_k / da[j] = 2 * f[k-j]
    idx = np.arange(js, je + 1)
    grad[js:je+1] = 2.0 * f[peak_k - idx]
    return grad


def compute_weighted_gradient(f, conv, top_frac=0.1, sharpness=10.0):
    """Compute weighted subgradient of max(conv) using correct gradients."""
    n = len(f)
    len_c = len(conv)
    k_top = max(5, int(len_c * top_frac))
    
    # Get top peaks
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]
    
    # Sharp weighting (closer to argmax)
    scaled = sharpness * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Sum weighted gradients
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-10:
            continue
        k = int(peaks[i])
        grad += w[i] * correct_gradient(f, k)
    
    return grad

def solve_lp_direction(f, M, n):

def make_init(n, variant, seed=42):

    rng = np.random.default_rng(seed + variant)

        a = (np.arange(n) + 1) ** (-0.7)

        a = np.exp(-30.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-30.0 * (x - 0.6) ** 2)
    elif variant == 7:

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with 1D optimal step."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000 + 1)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for ratio in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1, 3.0]:
                new_val = old_val * ratio
                if new_val < 0 or new_val > 0.5:
                    continue
                if (1.0 - old_val) < 1e-10:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial[mask] *= (1.0 - new_val) / (1.0 - old_val)
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[j] = best_j_val
                f[mask] *= (1.0 - best_j_val) / (1.0 - old_val)
                f = np.maximum(f, 0)
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def make_init(n, variant, seed=42):

        a = np.maximum(a, 0)
    elif variant == 8:

    s = a.sum()

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
