def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return c


def compute_subgradient(f, c, top_k=20):
    """Compute weighted subgradient focusing on top peaks."""
    n = len(f)
    len_c = len(c)
    # Select top peaks
    if top_k >= len_c:
        peaks = np.arange(len_c)
        weights = np.ones(len_c) / len_c
    else:
        # Use softmax weighting on top peaks
        k = min(top_k, len_c)
        idx = np.argpartition(c, -k)[-k:]
        peaks = idx
        pv = c[peaks]
        # Softmax with temperature
        temp = c.max() - c.min()
        if temp < 1e-15:
            weights = np.ones(k) / k
        else:
            scaled = (pv - pv.max()) / (temp * 0.1)
            weights = np.exp(scaled)
            weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-12:
            continue
        k_pos = int(peaks[i])
        js = max(0, k_pos - n + 1)
        je = min(n - 1, k_pos)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k_pos - idx] + f[idx]
        # Handle the diagonal case (k_pos - idx == idx, i.e., idx = k_pos/2)
        # In that case, contrib should be 2*f[idx] not f[idx] + f[idx] = 2*f[idx]
        # Actually f[k_pos - idx] + f[idx] already gives 2*f[idx] when k_pos - idx == idx
        sg[js:je + 1] += weights[i] * contrib
    return sg

def lp_direction(f, M, n):
    """
    Solve LP: max sum(g) s.t. (f*g_conv)[k] <= M for all k, g >= 0.
    Here (f*g_conv)[k] = sum_j f[j]*g[k-j] (cross-convolution).
    """
    # The constraint matrix A: A[k, j] = f[max(0, k-j)] if k-j < n else 0
    # Actually: (f * g)[k] = sum_j f[j] * g[k-j] where g[m] = 0 for m outside [0, n-1]
    # So (f*g)[k] = sum_{j: 0<=k-j<n} f[j] * g[k-j]
    #            = sum_{j: max(0, k-n+1) <= j <= min(n-1, k)} f[j] * g[k-j]
    # Let m = k-j, then j = k-m, and the constraint is:
    # sum_m f[k-m] * g[m] <= M
    # where m ranges over max(0, k-n+1) <= k-m <= min(n-1, k), i.e., max(0, 2k-n+1) <= m <= min(k, n-1)
    # Hmm, this is getting complicated. Let me just build the matrix directly.
    
    # A is (2n-1) x n
    # A[k, m] = coefficient of g[m] in (f*g)[k]
    # (f*g)[k] = sum_j f[j] * g[k-j]
    # For a fixed m (index of g), g[m] appears in (f*g)[k] when k-j = m, i.e., j = k-m.
    # So A[k, m] = f[k-m] if 0 <= k-m < n, else 0.
    
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        # g[m] appears in (f*g)[k] for k = m, m+1, ..., m+n-1 (where j = k-m ranges 0 to n-1)
        for k in range(m, m + n):
            j = k - m
            if 0 <= j < n:
                A[k, m] = f[j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10.0})
    if res.success:
        return res.x
    return None

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Cyclic coordinate descent with 1D line search per coordinate.
    For each coordinate j, we optimize a[j] while scaling the rest to maintain sum=1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try several candidate values
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 0.85, old_val * 1.15,
                         old_val * 1.3, old_val * 1.5, old_val * 2.0, 0.0]
            # Add some fine perturbations
            candidates.append(old_val * 0.95)
            candidates.append(old_val * 1.05)
            
            for new_val in candidates:
                if new_val < 0 or new_val >= 1.0:
                    continue
                # Scale rest to maintain sum = 1
                rest_sum = 1.0 - new_val
                if rest_sum < 1e-12:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] = f[mask] * (rest_sum / (1.0 - old_val)) if (1.0 - old_val) > 1e-12 else f[mask]
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-12:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                rest_sum = 1.0 - best_j_val
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] = f[mask] * (rest_sum / (1.0 - old_val)) if (1.0 - old_val) > 1e-12 else f[mask]
                f = np.maximum(f, 0)
                f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            if sweep >= 2:
                break
    
    return best_f, best_obj


def make_init(n, variant, seed=42):
    """Generate diverse asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 1000)
    x = np.arange(n) / n
    
    if variant == 0:
        # Exponential decay (front-loaded)
        a = np.exp(-x / 0.15)
    elif variant == 1:
        # Exponential decay (back-loaded)
        a = np.exp(-(1 - x) / 0.15)
    elif variant == 2:
        # Gentle ramp
        a = 0.3 + 0.7 * x
    elif variant == 3:
        # Sine-modulated
        a = np.exp(-x / 0.3) * (1 + 0.3 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif variant == 4:
        # Two-bump asymmetric
        a = np.exp(-100 * (x - 0.25) ** 2) + 0.4 * np.exp(-100 * (x - 0.75) ** 2)
        a = np.maximum(a, 1e-4)
    elif variant == 5:
        # Power law
        a = (x + 0.01) ** (-0.3)
    elif variant == 6:
        # Near-uniform with slight tilt
        a = 1.0 + 0.2 * x + rng.standard_normal(n) * 0.01
        a = np.maximum(a, 0.1)
    elif variant == 7:
        # Cosine window (front-heavy)
        a = np.cos(np.pi * x / 2) ** 2 + 0.1
    elif variant == 8:
        # Logarithmic
        a = np.log1p(x * 10 + 1)
    elif variant == 9:
        # Random smooth
        a = np.ones(n)
        for _ in range(5):
            center = rng.random()
            width = 0.1 + 0.3 * rng.random()
            amp = 0.5 + rng.random()
            a += amp * np.exp(-((x - center) / width) ** 2)
        a = np.maximum(a, 0.01)
    elif variant == 10:
        # Inverse power (back-loaded)
        a = (1 - x + 0.01) ** (-0.3)
    elif variant == 11:
        # Very gentle asymmetry
        a = 1.0 - 0.1 * x + 0.05 * x**2
        a = np.maximum(a, 0.1)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Linear interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def objective_normalized(a):
    """Compute the objective for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()
