    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""

import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        if v[0] >= target_sum:
            return np.array([target_sum])
        return np.array([max(v[0], 0.0)])
    v = np.maximum(v, -1e10)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - target_sum) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def lse_gradient_fft(f, beta):
    """
    Compute gradient of LSE(beta, conv(f)) = (1/beta)*log(sum exp(beta*conv_k)).
    Uses FFT-based cross-correlation for O(n log n) computation.
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    # grad[m] = sum_k p[k] * 2*f[k-m] (valid indices)
    # = 2 * sum_j f[j] * p[m+j]
    # = 2 * (f_rev * p)[m + n - 1] where f_rev = f[::-1]
    conv_len = 3 * n - 2
    size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f[::-1], size)
    P = np.fft.rfft(p, size)
    conv_result = np.fft.irfft(F * P, size)
    g = conv_result[n - 1:2 * n - 1]
    return 2.0 * g

    scaled -= scaled.max()

def build_lp_matrix(f, n):
    """Build constraint matrix for LP: rows k, col j = f[k-j] if valid."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            ft = ft / st
            v = objective_normalized(ft)
            if v < best_val:
                best_val = v
                best_alpha = a

        if best_alpha > 1e-8:
            # Golden section refinement
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                c_pt = hi - gr * (hi - lo)
                d_pt = lo + gr * (hi - lo)
                fc = objective_normalized(np.maximum(f + c_pt * (g - f), 0) /
                                          np.maximum(f + c_pt * (g - f), 0).sum())
                fd = objective_normalized(np.maximum(f + d_pt * (g - f), 0) /
                                          np.maximum(f + d_pt * (g - f), 0).sum())
                if fc < fd:
                    hi = d_pt
                else:
                    lo = c_pt
                if abs(hi - lo) < 1e-5:
                    break
            mid = (lo + hi) / 2.0
            ft = f + mid * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st > 1e-15:
                ft = ft / st
                v = objective_normalized(ft)
                if v < best_val:
                    best_val = v
                    best_alpha = mid

            if best_alpha > 1e-8 and best_val < best_obj:
                f_new = f + best_alpha * (g - f)
                f_new = np.maximum(f_new, 0.0)
                st = f_new.sum()
                f_new = f_new / max(st, 1e-15)
                f = f_new
                best_obj = best_val
                best_f = f.copy()
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 60:
            break

    return best_f, best_obj


def peak_cd_optimize(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent with multi-scale adjustments.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = objective_normalized(f)

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 60:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Target top peaks
        top_k = min(16, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(5, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.4, 0.6, 0.75, 0.85, 1.2, 1.4, 1.6]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val

                    # Redistribute mass
                    if abs(d) > 1e-15:
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = trial[non_peak]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum

                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = objective_normalized(trial)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True

        if not improved:
            no_improve += 1
            # Aggressive: reduce all peak contributors uniformly
            if no_improve > 10:
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)

                trial = f.copy()
                trial[js:je + 1] *= 0.9
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc_sum = trial[non_contrib].sum()
                    if nc_sum > 1e-15:
                        trial[non_contrib] += d * trial[non_contrib] / nc_sum
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = objective_normalized(trial)
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    # Try increasing some non-peak elements
                    rng = np.random.default_rng(no_improve * 7)
                    trial = f.copy()
                    trial[js:je + 1] *= 0.95
                    d = 1.0 - trial.sum()
                    if d > 0:
                        # Add to random non-peak positions
                        candidates = np.where(non_contrib)[0]
                        if len(candidates) > 0:
                            chosen = rng.choice(candidates, size=min(20, len(candidates)), replace=False)
                            trial[chosen] += d / len(chosen)
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()
                    obj = objective_normalized(trial)
                    if obj < best_obj:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        no_improve = 0
                    else:
                        break
        else:
            no_improve = 0

        if verbose and it % 10 == 0:
            print(f"    Peak CD: obj={best_obj:.10f}")
        it = it + 1 if 'it' in dir() else 1

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, verbose=False):
    """
    Multiplicative mirror descent with peak-weighted updates.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = objective_normalized(f)

    t0 = time.monotonic()
    lr = 0.003
    temperature = 0.003
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Weight peaks by softmax
        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws

        # Compute gradient-like quantity
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        # Multiplicative update
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj = objective_normalized(f_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new  # still update to explore
            no_improve += 1

        lr *= 0.9998
        temperature *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = max(0.003, lr)
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_symmetric_bump(n, width=0.6, power=2.0):
    """Symmetric bump with adjustable width and shape."""
    x = np.arange(n) - (n - 1) / 2.0
    half_w = n * width / 2.0
    t = np.abs(x) / half_w
    a = np.where(t < 1.0, (1.0 - t) ** power, 0.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_plateau(n, flat_frac=0.3, edge=2.0):
    """Plateau with smooth edges."""
    x = np.arange(n) - (n - 1) / 2.0
    half_flat = n * flat_frac / 2.0
    half_total = half_flat + n * 0.2
    t = np.abs(x) / half_total
    a = np.where(t < 1.0, (1.0 - t) ** edge, 0.0)
    # Create plateau in center
    center_mask = np.abs(x) < half_flat
    a[center_mask] = a[center_mask].max() if center_mask.any() else 0
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_optimal_shape(n, alpha=1.5):
    """
    Try a shape that balances concentration and spread.
    Based on the idea that the optimal reduces the autocorrelation peak
    by spreading mass non-uniformly.
    """
    x = np.arange(n) / (n - 1)  # [0, 1]
    # Use a shape that's slightly broader than Gaussian
    a = (1.0 + alpha * (x - 0.5) ** 2) * np.exp(-3.0 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def try_evaluate(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid multi-phase optimization combining smooth gradient, LP-FW,
    peak CD, and mirror descent with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq, best_score = try_evaluate(parent, best_seq, best_score, "default")

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Diverse Initializations ===")
    n_parent = len(parent)

    init_candidates = [parent.copy()]
    for n_try in [512, 640, 768, 896, 1024]:
        if n_try == n_parent:
            init_candidates.append(parent.copy())
        else:
            init_candidates.append(interpolate_sequence(parent, n_try))
        init_candidates.append(make_gaussian(n_try, 0.2))
        init_candidates.append(make_gaussian(n_try, 0.3))
        init_candidates.append(make_symmetric_bump(n_try, 0.5, 2.0))
        init_candidates.append(make_symmetric_bump(n_try, 0.7, 3.0))
        init_candidates.append(make_plateau(n_try, 0.3, 2.0))
        init_candidates.append(make_optimal_shape(n_try, 1.5))
        init_candidates.append(make_optimal_shape(n_try, 2.5))

    best_init = None
    best_init_obj = float('inf')
    for ic in init_candidates:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = objective_normalized(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_evaluate(a_norm, best_seq, best_score, f"init_n{len(a_norm)}")

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Smooth gradient optimization ===
    print(f"\n=== Phase 2: Smooth Gradient (n={len(best_init)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, obj = smooth_gradient_optimize(best_init, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_evaluate(a_opt, best_seq, best_score, "smooth_grad")

    # === Phase 3: LP-FW refinement ===
    if best_seq is not None:
        print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.3, 400)
        if t_alloc > 30:
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_evaluate(a_opt, best_seq, best_score, "lpfw_main")

    # === Phase 4: Peak CD ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Peak CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 250)
        if t_alloc > 20:
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = peak_cd_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_evaluate(a_opt, best_seq, best_score, "peak_cd")

    # === Phase 5: Mirror Descent ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Mirror Descent (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.15, 200)
        if t_alloc > 15:
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_evaluate(a_opt, best_seq, best_score, "mirror_desc")

    # === Phase 6: Multi-length exploration ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 6: Multi-length ===")
    for target_n in [640, 896, 1024, 1280]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = try_evaluate(f_interp, best_seq, best_score, f"interp_n{target_n}")

        # Quick LP-FW on interpolated
        t_alloc = min(rem * 0.25, 100)
        if t_alloc > 20:
            a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc)
            best_seq, best_score = try_evaluate(a_opt, best_seq, best_score, f"lpfw_n{target_n}")

    # === Phase 7: Perturbed restarts ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Perturbed LP-FW ===")
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_alloc = min(rem / (4 - p + 1), 100)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise, 1.0)
            elif p == 1:
                # Blend with uniform
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif p == 2:
                # Reverse blend
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Larger noise
                noise = rng.standard_normal(n_best) * 0.01
                f_init = project_simplex(best_seq + noise, 1.0)

            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc)
            best_seq, best_score = try_evaluate(a_opt, best_seq, best_score, f"perturb_{p}")

    # === Phase 8: Final polish ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Polish ===")
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            # Peak CD
            t1 = min(rem * 0.3, 60)
            if t1 > 10:
                a_opt, obj = peak_cd_optimize(best_seq, time_limit=t1)
                best_seq, best_score = try_evaluate(a_opt, best_seq, best_score, "fin_peak")

            rem = deadline - time.monotonic()
            if rem > 20:
                # Smooth gradient (short)
                t2 = min(rem * 0.3, 50)
                if t2 > 10:
                    a_opt, obj = smooth_gradient_optimize(best_seq, time_limit=t2)
                    best_seq, best_score = try_evaluate(a_opt, best_seq, best_score, "fin_smooth")

            rem = deadline - time.monotonic()
            if rem > 20:
                # LP-FW final
                t3 = min(rem * 0.4, 80)
                if t3 > 10:
                    a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3)
                    best_seq, best_score = try_evaluate(a_opt, best_seq, best_score, "fin_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
