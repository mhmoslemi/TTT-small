    import cvxpy as cp
    import time
    
    t0 = time.monotonic()
    a_var = cp.Variable(n)
    t_var = cp.Variable()
    
    base_constraints = [a_var >= 0, cp.sum(a_var) == 1]
    
    if initial_lags is None:
        initial_lags = [0, n-1, n//2, n-1-n//2]
    
    active_lags = set(initial_lags)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build constraints
        constraints = list(base_constraints)
        for k in sorted(active_lags):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            terms = [a_var[i] * a_var[k-i] for i in range(lo, hi)]

            constraints.append(conv_k <= t_var)
        
        problem = cp.Problem(cp.Minimize(2*n*t_var), constraints)
        
        try:
            problem.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                problem.solve(solver=cp.SCS)
            except Exception:
                break
        
        if a_var.value is None:
            break

        b = fast_convolve(a_val)

        threshold = max_b * 0.99

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. CVXPY cutting-plane method (exploits convexity)
2. Mirror descent with multiplicative updates
3. L-BFGS refinement in log-space
4. Coordinate descent polishing
5. Multi-length exploration

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def conv_cvxpy(a_var, k, n):
    """Build the CVXPY expression for conv(a,a)[k]."""
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    terms = [a_var[i] * a_var[k - i] for i in range(lo, hi)]
    return cp.sum(terms)

def mirror_descent(a0, time_limit=300, lr_init=0.01, n_top=50, seed=42):
    """Mirror descent (multiplicative weights) on the simplex."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        if b_max < best_obj - 1e-15:
            best_obj = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient using top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        w = b[top_idx].copy()
        w -= w.max()
        w = np.exp(w * 5)  # sharpness factor
        w /= w.sum()
        
        # Subgradient: g[j] = sum_k w_k * 2*a[k-j] (for valid k-j)
        g = np.zeros(n)
        for idx_k in range(len(top_idx)):
            k = top_idx[idx_k]
            if w[idx_k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                g[j] += w[idx_k] * 2 * a[k - j]
        
        # Multiplicative update: a_new[j] = a[j] * exp(-lr * g[j])
        log_a = np.log(np.maximum(a, 1e-300)) - lr * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a = a_new
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
        elif no_improve > 100:
            lr *= 0.9
        
        if no_improve > 1000:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using convex optimization:
1. CVXPY cutting-plane method (exploits convexity directly)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.stats import beta as beta_dist

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return a.astype(np.float64).copy()
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))

    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    result = irfft(np.conj(A) * W, fft_size)
    return result[:n]


def softmax_vec(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbgfs_refine(a0, T, time_limit=120, maxiter=5000):
    """
    L-BFGS in log-space with soft-max smoothed objective.
    Uses analytical gradient via FFT cross-correlation.
    """
    t0 = time.monotonic()
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    a_init = softmax_vec(z0)
    best_obj = eval_local(a_init)
    best_z = z0.copy()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_vec(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        lse = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T) * lse

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_vec(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # g[j] = (4n/T) * sum_k w[k] * a[k-j] * I[0<=k-j<n]
        # = (4n/T) * cross_correlation_fft(a, w)[j]
        cross = cross_correlation_fft(a, w)
        g = (4.0 * n / T) * cross

        # Gradient w.r.t. z for softmax parameterization:
        # grad_z[j] = a[j] * (g[j] - sum_i a[i]*g[i])
        S_grad = np.dot(a, g)
        grad_z = a * (g - a * S_grad)

        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 30}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x.copy()
            best_obj = final_obj
    except Exception as e:
        print(f"    L-BFGS exception: {e}")

    return softmax_vec(best_z), best_obj


def mirror_descent(a0, time_limit=300, lr_init=0.05, sharpness=10.0,
                   kick_interval=500, kick_size=0.003, seed=42):
    """
    Mirror descent (multiplicative weights) on probability simplex.
    Uses KL geometry instead of Euclidean, which is better for simplex.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        b_max = b.max()

        if b_max < best_obj - 1e-15:
            best_obj = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Softmax weights on peaks
        scaled = b * sharpness
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        # Subgradient via cross-correlation
        # g[j] = 2 * sum_k w[k] * a[k-j] * I[0<=k-j<n]
        cross = cross_correlation_fft(a, w)
        g = 2.0 * cross

        # Multiplicative update: a_new = a * exp(-lr * g), renormalize
        log_a = np.log(np.maximum(a, 1e-300)) - lr * g
        a = softmax_vec(log_a)

        # Adaptive learning rate
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            lr = lr_init * max(0.3, 1.0 - 0.1 * (it // 5000))
            no_improve = 0
        elif no_improve > kick_interval // 2:
            lr *= 0.9

    return best_a, best_obj


def subgradient_optimize(a0, time_limit=300, lr_init=0.005, momentum=0.9,
                         temp_init=0.001, temp_growth=1.02,
                         kick_interval=1000, kick_size=0.003,
                         n_top=None, seed=42):
    """
    Projected subgradient with Nesterov momentum and temperature annealing.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)

        # Temperature-weighted peaks
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            w_sel /= w_sel.sum()
            w = np.zeros(len(b))
            w[top_idx] = w_sel

        # Subgradient via cross-correlation
        cross = cross_correlation_fft(a, w)
        g = 2.0 * cross

        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95

        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def coordinate_polish(a0, time_limit=60, multipliers=None):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.5])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def solve_cvxpy_cutting_plane(n, time_limit=300, max_iters=100):
    """
    Solve min 2n*t s.t. conv(a,a)[k] <= t for active k, sum(a)=1, a>=0
    using iterative constraint generation (cutting plane).
    """
    t0 = time.monotonic()
    a_var = cp.Variable(n)
    t_var = cp.Variable()

    base_constraints = [a_var >= 0, cp.sum(a_var) == 1]

    # Initial active set: edges and center
    initial_lags = [0, n - 1, n // 2, (n - 1) // 2, 1, n - 2]
    active_lags = sorted(set(l for l in initial_lags if 0 <= l < 2 * n - 1))

    best_a = None
    best_obj = float('inf')

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        constraints = list(base_constraints)
        for k in active_lags:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            terms = [a_var[i] * a_var[k - i] for i in range(lo, hi)]
            conv_k = cp.sum(terms)
            constraints.append(conv_k <= t_var)

        problem = cp.Problem(cp.Minimize(2 * n * t_var), constraints)

        solved = False
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                problem.solve(solver=solver)
                if a_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        a_val = np.maximum(np.array(a_var.value).flatten(), 0)
        S = a_val.sum()
        if S < 1e-10:
            break
        a_val = a_val / S

        b = fast_convolve(a_val)
        max_b = b.max()
        obj = 2.0 * n * max_b

        if obj < best_obj:
            best_obj = obj
            best_a = a_val.copy()

        # Find near-maximum lags to add
        threshold = max_b * 0.99
        near_max = np.where(b >= threshold)[0]
        new_lags = set(int(k) for k in near_max) - set(active_lags)

        if not new_lags:
            print(f"    CVXPY CP converged at iter {it}, obj={obj:.8f}")
            break

        # Limit number of new constraints per iteration
        if len(new_lags) > 5:
            sorted_lags = sorted(new_lags, key=lambda k: -b[k])
            new_lags = set(sorted_lags[:5])

        active_lags = sorted(set(active_lags) | new_lags)
        # Cap total constraints
        if len(active_lags) > 30:
            # Keep only the most important (highest b values)
            b_vals = [b[k] for k in active_lags]
            sorted_pairs = sorted(zip(active_lags, b_vals), key=lambda x: -x[1])
            active_lags = [p[0] for p in sorted_pairs[:30]]

        if it % 5 == 0:
            print(f"    CVXPY iter {it}: {len(active_lags)} constraints, obj={obj:.8f}")

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3, center=0.5):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine window."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    """Tukey window."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_u_shaped_init(n):
    """U-shaped (edges high, center low) initialization."""
    x = np.linspace(0, 1, n)
    a = (x - 0.5) ** 2 + 0.1
    return a / a.sum()


def make_beta_init(n, alpha, beta_p):
    """Beta distribution shaped initialization."""
    x = np.linspace(0.001, 0.999, n)
    a = beta_dist.pdf(x, alpha, beta_p)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization exploiting convexity of the problem.
    Key improvements: CVXPY cutting plane, mirror descent, proper gradients.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_cosine_init(768)
        best_seq = parent.copy()
        best_score = eval_local(parent)
        print(f"No parent, using cosine n=768, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: CVXPY cutting plane (exploits convexity) ===
    print(f"\n=== Phase 1: CVXPY cutting plane (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem * 0.5, 240.0)
        print(f"  CVXPY, t={t_alloc:.0f}s")
        try:
            a_cvx, obj_cvx = solve_cvxpy_cutting_plane(n_parent, time_limit=t_alloc)
            if a_cvx is not None:
                try_seq(a_cvx, f"cvxpy_n{n_parent}")
        except Exception as e:
            print(f"  CVXPY failed: {e}")

    # === Phase 2: Mirror descent on best ===
    print(f"\n=== Phase 2: Mirror descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem * 0.4, 120.0)
            print(f"  Mirror descent n={n_best}, t={t_alloc:.0f}s")
            a_md, _ = mirror_descent(best_seq, time_limit=t_alloc,
                                     lr_init=0.03, sharpness=8.0, seed=seed)
            try_seq(a_md, f"mirror_n{n_best}")

    # === Phase 3: L-BFGS refinement ===
    print(f"\n=== Phase 3: L-BFGS refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for T in [30, 80, 200]:
            rem = deadline - time.monotonic() - 400
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 80.0)
            if t_alloc < 10:
                continue
            print(f"  L-BFGS T={T}, n={n_best}, t={t_alloc:.0f}s")
            a_lb, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=5000)
            try_seq(a_lb, f"lbgfs_T{T}")

    # === Phase 4: Diverse initializations with subgradient ===
    print(f"\n=== Phase 4: Diverse initializations (best={best_score:.6f}) ===")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {
        "gauss_0.2": make_gaussian_init(n_base, 0.2),
        "gauss_0.4": make_gaussian_init(n_base, 0.4),
        "cosine": make_cosine_init(n_base),
        "tukey_0.3": make_tukey_init(n_base, 0.3),
        "tukey_0.7": make_tukey_init(n_base, 0.7),
        "u_shaped": make_u_shaped_init(n_base),
        "beta_3_1": make_beta_init(n_base, 3.0, 1.0),
        "beta_1_3": make_beta_init(n_base, 1.0, 3.0),
        "beta_0.5_3": make_beta_init(n_base, 0.5, 3.0),
        "asym_gauss": make_gaussian_init(n_base, 0.3, 0.3),
    }

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(rem * 0.12, 30.0)
        if t_alloc < 5:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_opt, _ = subgradient_optimize(
            init_a, time_limit=t_alloc,
            lr_init=0.01, momentum=0.9,
            temp_init=0.002, temp_growth=1.03,
            kick_interval=500, kick_size=0.005,
            n_top=max(10, n_base // 20),
            seed=seed + hash(name) % 10000
        )
        try_seq(a_opt, f"diverse_{name}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break

            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.25, 60.0)
            if t_alloc < 8:
                continue
            print(f"  Interpolating to n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(
                f_interp, time_limit=t_alloc,
                lr_init=0.005, momentum=0.92,
                temp_init=0.001, temp_growth=1.02,
                kick_interval=800, kick_size=0.003,
                n_top=max(10, target_n // 20),
                seed=seed + target_n % 10000
            )
            try_seq(a_opt, f"length_n{target_n}")

            # Quick L-BFGS on this
            if best_seq is not None and len(best_seq) == target_n:
                rem = deadline - time.monotonic() - 150
                if rem > 10:
                    t_lb = min(rem * 0.3, 30.0)
                    a_lb, _ = lbgfs_refine(best_seq, T=50, time_limit=t_lb, maxiter=3000)
                    try_seq(a_lb, f"lbgfs_n{target_n}")

    # === Phase 6: Deep mirror descent on best ===
    print(f"\n=== Phase 6: Deep mirror descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem, 120.0)
            print(f"  Deep mirror n={n_best}, t={t_alloc:.0f}s")
            a_deep, _ = mirror_descent(
                best_seq, time_limit=t_alloc,
                lr_init=0.01, sharpness=15.0,
                kick_interval=300, kick_size=0.001,
                seed=seed + 99999
            )
            try_seq(a_deep, "deep_mirror")

    # === Phase 7: Final L-BFGS + coordinate polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_lb = min(rem * 0.4, 60.0)
            print(f"  L-BFGS T=100, t={t_lb:.0f}s")
            a_fin, _ = lbgfs_refine(best_seq, T=100, time_limit=t_lb, maxiter=5000)
            try_seq(a_fin, "final_lbgfs")

        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd = min(rem, 45.0)
            print(f"  CD polish, t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd_polish")

    # === Phase 8: Micro-optimization ===
    print(f"\n=== Phase 8: Micro-opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_alloc = min(rem * 0.5, 20.0)
            a_micro, _ = subgradient_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=0.001, momentum=0.95,
                temp_init=0.0005, temp_growth=1.005,
                kick_interval=2000, kick_size=0.001,
                n_top=max(20, n_best // 5),
                seed=seed + 777777
            )
            try_seq(a_micro, "micro_sg")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = eval_local(best_seq)

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
