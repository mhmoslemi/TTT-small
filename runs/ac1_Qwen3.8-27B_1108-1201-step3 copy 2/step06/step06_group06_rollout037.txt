def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w = w / w.sum()
    
    # h[i] = sum_l f[l] * w[l+i]
    f_rev = f[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    F = np.fft.rfft(f_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_r = np.fft.irfft(F * W, fft_size)
    h = conv_r[n - 1: 2 * n - 1]
    
    grad = (2.0 / tau) * h
    return grad, conv


def adam_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.004, reheat=True):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0
    
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    tau_base = tau
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_smooth_grad(f, tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Temperature: exponential decay with periodic reheating
        tau *= 0.99985
        if tau < 5e-8:
            tau = 5e-8
        
        if reheat and no_improve > 300 and (t % 1000 < 200):
            # Periodic reheat
            tau = 0.04 * fast_convolve(f).max()
            lr = lr_init * 1.2
            no_improve = 0
        
        if no_improve > 800:
            rng = np.random.default_rng(t * 7 + 3)
            noise = 0.002 + 0.002 * min(no_improve / 2000.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)

def subgradient_heavy_ball(f0, time_limit):
    """Subgradient method with heavy ball momentum for the max-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Heavy ball parameters
    momentum = 0.9
    lr = 0.01
    velocity = np.zeros(n)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find active peaks
        tol = 1e-6 * M
        active = np.where(conv >= M - tol)[0]
        
        # Compute subgradient: weighted average of 2*flip_k(f) for active k
        # To be more robust, use all peaks with weight proportional to conv value
        weights = conv.copy()
        weights[weights < M * 0.5] = 0  # only use peaks near the max
        if weights.sum() < 1e-15:
            weights[active] = 1.0
        weights = weights / weights.sum()
        
        # grad[i] = sum_k weights[k] * 2 * f[k-i] (if k-i in [0, n-1])
        # This is 2 * (f * weights)[cross-correlation]
        # Same computation as smooth grad but with discrete weights
        f_rev = f[::-1]
        total_len = n + len(weights) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        F = np.fft.rfft(f_rev, fft_size)
        W = np.fft.rfft(weights, fft_size)
        conv_r = np.fft.irfft(F * W, fft_size)
        h = conv_r[n - 1: 2 * n - 1]
        
        grad = 2.0 * h
        
        # Heavy ball update
        velocity = momentum * velocity - lr * grad
        f_new = f + velocity
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Adaptive learning rate

            velocity = np.zeros(n)
            no_improve = 0
            lr = 0.005
            momentum = 0.9
    
    return best_f, 2 * n * best_obj


def mirror_descent(f0, time_limit):
    """Mirror descent (multiplicative weight updates) for the max-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    # Ensure no zeros for multiplicative updates
    f = np.maximum(f, 1e-10)
    f = f / f.sum()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    eta = 0.05
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute subgradient
        tol = 1e-5 * M
        weights = conv.copy()
        weights[weights < M * 0.3] = 0
        if weights.sum() < 1e-15:
            weights[np.argmax(conv)] = 1.0
        weights = weights / weights.sum()
        
        f_rev = f[::-1]
        total_len = n + len(weights) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        F = np.fft.rfft(f_rev, fft_size)
        W = np.fft.rfft(weights, fft_size)
        conv_r = np.fft.irfft(F * W, fft_size)
        h = conv_r[n - 1: 2 * n - 1]
        grad = 2.0 * h
        
        # Mirror descent: f_i <- f_i * exp(-eta * grad_i)
        # Normalize grad to avoid overflow
        grad_centered = grad - grad.mean()
        f_new = f * np.exp(-eta * grad_centered)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Adaptive eta
        if no_improve > 200:
            eta *= 0.9
        if no_improve > 600:
            rng = np.random.default_rng(it * 13 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            f = np.maximum(f, 1e-10)
            f = f / f.sum()
            no_improve = 0
            eta = 0.05
    
    return best_f, 2 * n * best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_init_gaussian(n, center=0.5, sigma=0.25):
    x = np.arange(n) / n - center
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_cosine(n, power=2.0):
    x = np.arange(n) / (n - 1) * 2 - 1  # [-1, 1]
    a = np.cos(np.pi * x / 2) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_random(n, seed=42):

    a = a ** 2  # make it smoother
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
