def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)

def subgradient_step(a, lr, n_top=50):
    """One subgradient step on the simplex."""
    b = fast_convolve(a)
    b_max = b.max()
    
    # Weight the top peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_vals = b[top_idx] - b_max + 1e-10  # small values for non-peaks
        w_vals = np.maximum(w_vals, 0)
        w_sum = w_vals.sum()
        if w_sum < 1e-30:
            w_full = np.zeros(len(b))
            w_full[b.argmax()] = 1.0
        else:
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_vals / w_sum
    else:
        w_full = np.ones(len(b)) / len(b)
    
    # Compute subgradient
    n = len(a)
    g = np.zeros(n)
    for k in range(len(b)):
        if w_full[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w_full[k] * a[k - idx]
    
    return project_simplex(a - lr * g), b_max

def lp_based_optimize(a0, time_limit, verbose=True):
    """Optimize using LP-based steepest descent with adaptive line search."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy()
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_obj = eval_obj(a)
    it = 0
    stall = 0
    
    if verbose:
        print(f"  LP optimization start: n={n}, obj={best_obj:.8f}")
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        
        # Solve LP
        g = solve_lp_direction(a, M)
        if g is None:
            # Fallback to subgradient
            a, _ = subgradient_step(a, lr=0.01, n_top=max(20, n//10))
            a = a / a.sum()
        else:
            g = np.maximum(g, 0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g = g / g_sum  # normalize to simplex
            
            # Line search: try different mixing parameters
            best_t = 0.3
            best_mix_obj = best_obj
            for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
                a_mix = (1 - t) * a + t * g
                a_mix = np.maximum(a_mix, 0)
                a_mix_sum = a_mix.sum()
                if a_mix_sum < 1e-10:
                    continue
                a_mix = a_mix / a_mix_sum
                obj_mix = eval_obj(a_mix)

                    best_mix_obj = obj_mix

            a = np.maximum(a, 0)
            a = a / a.sum()

            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

            stall = 0

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
                a_mix = (1.0 - t) * a + t * g
                a_mix = np.maximum(a_mix, 0.0)
                sm = a_mix.sum()
                if sm < 1e-10:
                    continue
                a_mix = a_mix / sm
                obj_mix = eval_obj(a_mix)
                if obj_mix < best_mix_obj:
                    best_mix_obj = obj_mix
                    best_t = t
            
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
        
        obj = eval_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {it}, obj={best_obj:.10f}, t={elapsed:.0f}s, stall={stall}")
        
        if stall > 100:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            a = a / a.sum()
            stall = 0
    
    if verbose:
        print(f"    LP done: best_obj={best_obj:.10f}, iters={it}")
    return best_a, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_skewed_init(n, alpha=3.0):
    """Skewed initialization: high on one side."""
    x = np.linspace(0.01, 0.99, n)
    a = x ** alpha
    return a / a.sum()


def update_best(a, best_seq, best_score, label=""):
    """Try to update the best sequence/score. Returns (best_seq, best_score)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_seq = a_norm.copy()
        best_score = score
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

            n_top=max(30, n_parent // 10), seed=seed,

    return [float(x) for x in best_seq]
