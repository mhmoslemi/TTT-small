Multi-phase optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
Uses Nesterov accelerated subgradient, LP-based Frank-Wolfe with active
set reduction, and local refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def subgradient_at_peak(a, k, n):
    """Compute subgradient of n*conv[k] at a."""
    # ∂conv[k]/∂a_i = 2*a[k-i] if k-i in [0,n-1], else 0
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for i in range(js, je + 1):
        sg[i] = 2.0 * a[k - i]
    return n * sg

def multi_peak_subgradient(a, k_peaks, weights, n):
    """Weighted sum of subgradients at multiple peaks."""
    sg = np.zeros(n)
    for idx, k in enumerate(k_peaks):
        if weights[idx] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            sg[i] += 2.0 * weights[idx] * a[k - i]
    return n * sg

    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    
    # Nesterov momentum
    f_prev = f.copy()
    momentum = 0.0
    lr = 1.0 / np.sqrt(n)
    
    for restart in range(restarts):
        f_prev = f.copy()
        momentum = 0.0
        lr = 1.0 / np.sqrt(n)
        
        for _ in range(5000):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            
            conv = fast_convolve(f)
            M = conv.max()

                best_f = f.copy()
            
            # Find top K peaks with softmax weights
            k_actual = min(K, len(conv))
            peaks = np.argpartition(conv, -k_actual)[-k_actual:]

            w = np.exp(scaled)

            if ws < 1e-15:

"""
Multi-phase optimization minimizing 2*n*max(auto_conv)/(sum)^2.

with adaptive active-set constraint reduction, peak-targeted pairwise

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] ** 2])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def solve_lp_active(f, M, n, max_constraints=200, multiplier=1.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M*mult for active constraints, g >= 0.
    Only includes constraints where (f*g)[k] > 0.5*M to reduce LP size."""
    fg = fast_convolve(f)
    active = fg > 0.5 * M
    # Always include the max constraint
    max_idx = np.argmax(fg)
    active[max_idx] = True
    # Cap number of constraints
    if active.sum() > max_constraints:
        active_indices = np.where(active)[0]
        vals = fg[active_indices]
        top = np.argpartition(vals, -max_constraints)[-max_constraints:]
        active = np.zeros(len(fg), dtype=bool)
        active[active_indices[top]] = True

    idx = np.where(active)[0]
    if len(idx) == 0:
        return None

    # Build constraint matrix: A[idx, j] = f[idx - j] if valid
    A = np.zeros((len(idx), n))
    for row, k in enumerate(idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[row, j] = f[k - j]

    b = np.full(len(idx), M * multiplier)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_full(f, M, n, multiplier=1.0):
    """Solve full LP with all constraints."""
    c = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    b = np.full(2 * n - 1, M * multiplier)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def nesterov_optimize(f0, time_limit, K=30, verbose=False):
    """Nesterov accelerated projected subgradient with multi-peak softmax."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    fprev = f.copy()
    momentum = 0.0
    lr = 1.0 / (2.0 * np.sqrt(n))
    temperature = 0.005

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 50 == 0:
                print(f"    Nest iter {it}: obj={best_obj:.10f}")

        # Top-K peaks with softmax
        k_actual = min(K, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / (temperature * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                sg[i] += 2.0 * w[idx] * a[k - i] if False else 2.0 * w[idx] * f[k - i]

        sg = n * sg

        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum ** 2)) / 2.0
        f_old = f.copy()
        f = f + momentum * (f - fprev)
        fprev = f_old

        # Projected step
        f = project_simplex(f - lr * sg)

        lr *= 0.99995
        temperature *= 1.001

        # Periodic restart
        if it % 500 == 0:
            f = 0.8 * best_f + 0.2 * f
            f = project_simplex(f)
            fprev = f.copy()
            momentum = 0.0
            lr = 1.0 / (2.0 * np.sqrt(n)) * 0.5

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6, power=4):

    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
