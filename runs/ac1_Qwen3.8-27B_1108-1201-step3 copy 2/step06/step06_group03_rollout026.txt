"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using ADAM optimization with
softmax parameterization, temperature-annealed log-sum-exp smoothing,
peak equalization, and multi-length exploration.
Key improvements over previous approach:
- ADAM optimizer with per-coordinate adaptive learning rates
- Softer initial smoothing for better exploration
- Peak equalization via iterative reweighting
- Longer sequence lengths for better lower bounds

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def compute_smooth_grad(f, conv, tau, top_frac=0.3, min_k=30):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv[k])) w.r.t. f."""
    n = len(f)
    len_c = len(conv)
    k_top = max(min_k, int(len_c * top_frac))
    
    # Get top-k peaks
    if k_top >= len_c:
        peaks = np.arange(len_c)
        weights = np.ones(len_c) / len_c
    else:
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        weights = np.exp(scaled)
        ws = weights.sum()
        if ws < 1e-30:
            weights = np.zeros(len(peaks))
            weights[np.argmax(pv)] = 1.0
        else:
            weights /= ws
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        grad[js:je + 1] += (2.0 / tau) * weights[i] * contrib
    
    return grad * 2 * n  # Scale by 2n for the full objective


def adam_optimize(f0, time_limit, tau_init=10.0, tau_final=200.0,
                  lr_init=0.01, lr_final=0.0001,
                  beta1=0.9, beta2=0.999, eps=1e-8,
                  restart_every=200, verbose=False):
    """ADAM optimization on softmax-parameterized simplex."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    
    # Initialize z so that softmax(z) = f
    z = np.log(f + 1e-30)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    
    t0 = time.monotonic()
    no_improve = 0
    tau = tau_init
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal tau and lr
        tau = tau_init + (tau_final - tau_init) * progress
        lr = lr_init * (lr_final / lr_init) ** progress
        
        # Current f
        f = softmax(z)
        
        # Compute objective and gradient
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()
        
        # Smooth gradient
        grad_f = compute_smooth_grad(f, conv, tau)
        
        # Transform to z-gradient (softmax chain rule)
        # d softmax(z)_i / dz_j = f_i * (delta_ij - f_j)
        # So grad_z = grad_f / f * (I - f f^T) * f = grad_f - f * (f @ grad_f)
        grad_z = grad_f - f * np.dot(f, grad_f)
        
        # ADAM update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check improvement
        f_new = softmax(z)
        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and t % 50 == 0:
                print(f"      ADAM iter {t}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > restart_every:
            rng = np.random.default_rng(t * 7 + 13)
            noise_scale = 0.005 + 0.005 * (no_improve / restart_every)
            z = z + rng.standard_normal(n) * noise_scale
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            if verbose:
                print(f"      ADAM restart at iter {t}")
        
        if lr < 1e-8:
            break
    
    return best_f, best_obj / (2.0 * n)

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[js:je + 1] += f[k - idx] + f[idx]
        
        # Reduce high-importance coordinates, increase low-importance ones
        mean_f = f.mean()
        d = np.zeros(n)
        high_mask = importance > np.median(importance)
        low_mask = importance < np.median(importance)
        
        # Move mass from high to low
        mass_to_move = f[high_mask].sum() * 0.05
        if mass_to_move < 1e-15:
            break
        d[high_mask] = -f[high_mask] * 0.05
        # Distribute to low coordinates proportionally
        if low_mask.sum() > 0:
            d[low_mask] = mass_to_move * f[low_mask] / f[low_mask].sum()
        
        # Ensure sum d = 0 (approximately)
        d = d - d.mean()
        d = d - f.min() + np.abs(d.min())  # Ensure f + d >= 0 (rough)
        d = d - d.mean()
        
        # Try the step
        f_trial = f + d
        f_trial = np.maximum(f_trial, 0)
        f_trial = f_trial / f_trial.sum()
        
        obj_trial = fast_convolve(f_trial).max()
        if obj_trial < M - 1e-14:
            f = f_trial
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"      PEQ iter {it}: obj={best_obj:.8f}")
        else:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with proper normalization handling."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int((time.monotonic() - t0) * 1000) % 100000 + 1)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            if f[j] < 1e-10:
                continue
            
            old = f[j]
            best_j_obj = best_obj
            best_j_new = old
            
            # Try several values
            for frac in [0.3, 0.5, 0.7, 0.85, 1.2, 1.5, 2.0, 0.0]:
                new = old * frac if frac > 0 else 0.0
                if new < 0 or new > 0.5:
                    continue
                denom = 1.0 - old
                if denom < 1e-10:
                    continue
                scale = (1.0 - new) / denom
                f_trial = f.copy()
                f_trial[j] = new
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_new = new
            
            if best_j_new != old:
                denom = 1.0 - old
                if denom > 1e-10:
                    scale = (1.0 - best_j_new) / denom
                    f[j] = best_j_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def make_init(n, variant, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed + variant * 1000)
    
    if variant == 0:
        a = np.exp(-np.arange(n) / (n * 0.1))
    elif variant == 1:
        a = np.exp(-np.arange(n) / (n * 0.2))
    elif variant == 2:
        a = np.exp(-np.arange(n) / (n * 0.3))
    elif variant == 3:
        x = np.arange(n) / n
        a = np.exp(-20 * (x - 0.2) ** 2) + 0.3 * np.exp(-20 * (x - 0.8) ** 2)
    elif variant == 4:
        x = np.arange(n) / n
        a = np.exp(-10 * x ** 2)
    elif variant == 5:
        a = np.ones(n)
        a[:n//4] *= 2.0
        a[3*n//4:] *= 0.3
    elif variant == 6:
        a = (np.arange(n) + 1) ** (-0.3)
    elif variant == 7:
        # Two peaks
        x = np.arange(n) / n
        a = np.exp(-50*(x-0.15)**2) + 0.5*np.exp(-50*(x-0.85)**2)
    elif variant == 8:
        # Smooth random
        a = np.ones(n) + 0.3 * np.sin(4 * np.pi * np.arange(n) / n) + 0.2 * np.sin(7 * np.pi * np.arange(n) / n)
        a = np.maximum(a, 0.01)
    else:
        a = rng.random(n) + 0.01
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result
