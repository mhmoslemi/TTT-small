import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score(a):
    """Compute the objective value for a sequence (assumed non-negative, sum > 0)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    a_norm = a / s
    max_conv = fast_convolve(a_norm).max()
    return 2.0 * n * max_conv


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_val = np.nonzero(cond)[0][-1] + 1
        theta = (cssv[rho_val - 1] - 1.0) / rho_val
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def lbgfs_smooth(a0, T, maxiter=3000, verbose=False):

def peak_coordinate_descent(a0, time_limit=30.0, max_rounds=5000):
    """
    Coordinate descent that targets the peak of the auto-convolution.
    At each step, find the peak, identify top contributors, and adjust them.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    for round_num in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_val = b.max()
        if peak_val <= best_score / (2.0 * n) + 1e-15:
            break
        
        # Find top peaks
        num_peaks = min(5, len(b))
        peak_indices = np.argsort(b)[::-1][:num_peaks]
        
        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            pk = int(pk)
            # Contributors: a[i] * a[pk-i] for valid i
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            
            # Find top 5 contributors
            top_k = min(5, len(idx))
            top_idx = idx[np.argsort(contribs)[::-1][:top_k]]
            
            for i in top_idx:
                if a[i] < 1e-14:
                    continue
                # Try reducing a[i] slightly and redistributing
                for frac in [0.95, 0.9, 0.85, 0.8]:
                    new_val = a[i] * frac
                    diff = a[i] - new_val
                    if diff < 1e-15:
                        continue
                    
                    # Redistribute to positions that don't contribute to the peak
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    
                    # Find positions where reducing won't hurt the peak
                    # Simply add to positions far from the peak
                    mask = np.ones(n, dtype=bool)
                    mask[max(0, pk-i-5):min(n, pk-i+6)] = False  # Near peak contribution zone
                    if mask.sum() < 1:
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        others_sum = a_trial[mask].sum()
                        if others_sum < 1e-15:
                            continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    
                    sc = score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try pair adjustments
            b = fast_convolve(a)
            peak_indices = np.argsort(b)[::-1][:3]
            pair_improved = False
            for pk in peak_indices:
                pk = int(pk)
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[pk - idx]
                top_k = min(4, len(idx))
                top_idx = idx[np.argsort(contribs)[::-1][:top_k]]
                
                for ii in range(len(top_idx)):
                    for jj in range(ii+1, len(top_idx)):
                        i1, i2 = top_idx[ii], top_idx[jj]
                        if i1 == i2 or a[i1] < 1e-14 or a[i2] < 1e-14:
                            continue
                        for f1 in [0.9, 0.8]:
                            for f2 in [0.9, 0.8]:
                                a_trial = a.copy()
                                a_trial[i1] *= f1
                                a_trial[i2] *= f2
                                a_trial = np.maximum(a_trial, 0)
                                ss = a_trial.sum()
                                if ss < 1e-15:
                                    continue
                                a_trial /= ss
                                sc = score(a_trial)
                                if sc < best_score - 1e-14:
                                    a = a_trial
                                    best_score = sc
                                    best_a = a.copy()
                                    pair_improved = True
                                    break
                            if pair_improved:
                                break
                        if pair_improved:
                            break
                    if pair_improved:
                        break
                if pair_improved:
                    break
            if not pair_improved:
                break
    
    return best_a


def simulated_annealing(a0, time_limit=60.0):
    """Simulated annealing on the simplex to escape local optima."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)

def multi_length_interpolate(seqs, new_n):
    """Interpolate between multiple sequences to a new length."""
    # Weight by inverse score
    scores = [score(s) for s in seqs]
    valid = [i for i in range(len(seqs)) if np.isfinite(scores[i])]
    if not valid:
        return np.ones(new_n) / new_n
    
    weights = np.array([1.0 / max(scores[i], 1e-10) for i in valid])
    weights /= weights.sum()
    
    # Interpolate each to new_n and combine
    combined = np.zeros(new_n)
    for w, i in zip(weights, valid):
        s = seqs[i]
        s_norm = s / s.sum()
        x_old = np.linspace(0, 1, len(s))
        x_new = np.linspace(0, 1, new_n)
        interp = np.interp(x_new, x_old, s_norm)
        combined += w * interp
    
    combined = np.maximum(combined, 0)
    ss = combined.sum()
    if ss > 1e-15:
        combined /= ss
    return combined

def structured_initializations(n, parent=None):
    """Generate diverse structured initial sequences of length n."""
    seqs = []
    
    # Uniform (baseline)
    seqs.append(np.ones(n) / n)
    
    # Cosine window
    x = np.linspace(0, 1, n)
    seqs.append(np.cos(np.pi * x) ** 2)
    seqs[-1] = np.maximum(seqs[-1], 0) / seqs[-1].sum()
    
    # Gaussian
    centers = [n * 0.5]
    sigmas = [n * 0.25, n * 0.15, n * 0.35]
    for c in centers:
        for sig in sigmas:
            seq = np.exp(-0.5 * ((np.arange(n) - c) / sig) ** 2)
            seq = np.maximum(seq, 0)
            if seq.sum() > 0:
                seqs.append(seq / seq.sum())
    
    # Power law decay
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        seq = (np.arange(1, n+1) ** (-alpha))
        seqs.append(seq / seq.sum())
    
    # Two-sided power law
    for alpha in [0.5, 1.0, 2.0]:
        c = n // 2
        seq = np.abs(np.arange(n) - c) ** (-alpha)
        seq[c] = seq.max() * 2
        seq = np.maximum(seq, 0)
        if seq.sum() > 0:
            seqs.append(seq / seq.sum())
    
    # Modulated cosine
    for freq in [1, 2, 3, 5, 8]:
        seq = 1.0 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n)
        seq = np.maximum(seq, 0)
        if seq.sum() > 0:
            seqs.append(seq / seq.sum())
    
    # Parent interpolation
    if parent is not None:
        if len(parent) == n:
            seqs.append(parent.copy())
        else:
            p = parent / parent.sum()
            x_old = np.linspace(0, 1, len(p))
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, p)
            interp = np.maximum(interp, 0)
            if interp.sum() > 0:
                seqs.append(interp / interp.sum())
    
    # Random (with seed for reproducibility)
    rng = np.random.default_rng(123)
    for _ in range(3):
        seq = rng.random(n)
        seqs.append(seq / seq.sum())
    
    return seqs

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1)
    return np.maximum(v - theta, 0.0)
