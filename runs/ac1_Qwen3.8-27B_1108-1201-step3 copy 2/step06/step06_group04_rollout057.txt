"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using block LP-Frank-Wolfe with
cyclic coordinate blocks, momentum-assisted direction selection, adaptive
constraint tightening, spectral-aware initialization, and multi-length search.
Key innovations: fast block LP solving with constraint selection, two-stage
(coarse then fine) block optimization, and dual-guided perturbations for
escaping local minima.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def make_bspline_like(n, order=8):
    """Create a B-spline-like smooth window of length n."""
    # Use repeated convolution of box functions to approximate a B-spline
    kernel = np.ones(order) / order
    for _ in range(4):
        kernel = np.convolve(kernel, np.ones(order) / order, 'same')
    kernel = np.maximum(kernel, 0)
    # Resample to length n
    x_old = np.linspace(0, 1, len(kernel))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, kernel)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric_taper(n, peak_pos=0.35):
    """Asymmetric sequence with peak away from center."""
    x = np.arange(n) / n
    # Two-sided exponential with different rates on each side
    left_rate = 4.0
    right_rate = 2.5
    a = np.zeros(n)
    for i in range(n):
        if x[i] <= peak_pos:
            a[i] = np.exp(-left_rate * (peak_pos - x[i]) / peak_pos)
        else:
            a[i] = np.exp(-right_rate * (x[i] - peak_pos) / (1 - peak_pos))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flattop(n, width=0.6):
    """Flat-top window with smooth edges."""
    x = np.arange(n) / n
    half_w = width / 2.0
    center = 0.45
    t = np.abs(x - center) / half_w
    a = np.ones(n)
    mask = t > 1.0
    a[mask] = (1.0 + np.cos(np.pi * t[mask])) / 2.0
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_block_lp(f, M, bs, be, n):
    """Solve LP for variables f[bs:be] fixed others, maximizing sum of block."""
    B = be - bs
    n_constraints = 2 * n - 1

    # Build constraint matrix: A[k, i] = f[k - (bs+i)] for valid indices
    A = np.zeros((n_constraints, B))
    for i in range(B):
        j = bs + i
        # g[j] appears in conv[f,g][k] for k in [j, j+n-1] with coeff f[k-j]
        k_start = j
        k_end = min(j + n - 1, n_constraints - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, i] = f[:length]

    # Compute constant from fixed variables
    conv_ff = fast_convolve(f)
    block_contrib = A @ f[bs:be]
    C = conv_ff - block_contrib
    b_ub = M - C

    # Handle infeasibility: if b_ub < 0 anywhere, g_block=0 is infeasible
    # This shouldn't happen if M >= conv_ff.max(), but add safety
    b_ub = np.maximum(b_ub, 0.0)

    c = -np.ones(B)
    bounds = [(0, None)] * B
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_refine': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

    n_blocks = (n + block_size - 1) // block_size
    no_improve = 0

                m_relax = min(m_relax * 1.1, 3.0)
                continue

            g_sum = g_block.sum()
            if g_sum < 1e-10:
                continue

            # Line search over alpha
            best_alpha = 0.0
            best_cmax = M
            alphas = [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]
            for alpha in alphas:
                trial = f.copy()
                trial[bs:be] = (1 - alpha) * f[bs:be] + alpha * g_block
                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum
                cmax = fast_convolve(trial).max()
                if cmax < best_cmax - 1e-12:
                    best_cmax = cmax
                    best_alpha = alpha

            if best_alpha > 0:
                f[bs:be] = (1 - best_alpha) * f[bs:be] + best_alpha * g_block
                f = np.maximum(f, 0)
                f = f / f.sum()
                cycle_improved = True

                conv = fast_convolve(f)
                new_M = conv.max()
                M = new_M
                m_relax = max(1.0, m_relax * 0.95)

                obj = 2.0 * n * M
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    momentum = 0.0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if not cycle_improved:
            # Stuck: apply perturbation
            if no_improve % 5 == 0:
                scale = 0.001 * min(m_relax, 3.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0

            if no_improve > 50:
                # More aggressive: swap some mass between peak and non-peak regions
                conv_cur = fast_convolve(best_f)
                k_peak = int(np.argmax(conv_cur))
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                f = best_f.copy()
                f[js:je + 1] *= 0.9
                diff = 1.0 - f.sum()
                mask = np.ones(n, dtype=bool)
                mask[js:je + 1] = False
                if mask.sum() > 0 and diff > 0:
                    w = f[mask]
                    ws = w.sum()
                    if ws > 1e-15:
                        f[mask] += diff * w / ws
                f = np.maximum(f, 0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                m_relax = 1.0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Block LP-FW iter {it}: obj={best_obj:.10f}, M={M:.8f}, "
                  f"m_relax={m_relax:.2f}, t={elapsed:.0f}s")

        if no_improve > 200:
            break

    return best_f, best_obj


def full_lpfw(f_init, time_limit, verbose=False, seed=0):
    """Full (non-block) LP-Frank-Wolfe for final polishing."""
    f = np.maximum(f_init.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    rng = np.random.default_rng(seed)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        # Build full constraint matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            length = k_end - k_start + 1
            A[k_start:k_end + 1, j] = f[:length]

        b_ub = np.full(2 * n - 1, M * m_exp)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})

        if not res.success or res.x is None:
            m_exp = min(m_exp * 1.1, 5.0)
            if m_exp > 5.0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        g = g / g_sum

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            trial = (1 - alpha) * f + alpha * g
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            cmax = fast_convolve(trial).max()
            if cmax < best_val:
                best_val = cmax
                best_alpha = alpha

        if best_alpha < 1e-8:
            no_improve += 1
            if no_improve > 20:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        f = (1 - best_alpha) * f + best_alpha * g
        f = np.maximum(f, 0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()
        m_exp = 1.0

        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"    Full LP-FW iter {it}: obj={best_obj:.10f}")

        if no_improve > 80:
            break

    return best_f, best_obj


def coord_descent_polish(f_init, time_limit, verbose=False, seed=0):
    """
    Coordinate descent targeting convolution peaks.
    Identifies top contributing pairs and adjusts them.
    """
    f = np.maximum(f_init.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    rng = np.random.default_rng(seed)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 60:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Focus on the top few peaks
        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try adjusting top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(6, len(order))):
                j_max = int(idx[order[oi]])
                if j_max < 0 or j_max >= n:
                    continue
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.4, 0.6, 0.8, 1.3, 1.5, 1.8]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute to non-contributing positions
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = trial[non_peak]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                            else:
                                trial[non_peak] += d / cnt
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        no_improve = 0
                        break

            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Perturb and restart
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0

    return best_f, best_obj


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization across lengths
    Phase 2: Block LP-FW (coarse blocks)
    Phase 3: Block LP-FW (fine blocks)
    Phase 4: Full LP-FW polish
    Phase 5: Coordinate descent
    Phase 6: Multi-length exploration
    Phase 7: Final deep optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_bspline_like(768, order=8)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")

    n_parent = len(parent)
    print(f"Starting with parent n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Diverse initialization (30s) ===
    print("\n=== Phase 1: Initialization ===")
    t_phase1_end = time.monotonic() + 30

    # Try various initializations at parent length and nearby lengths
    lengths_to_try = []
    for dn in [-32, -16, -8, 0, 8, 16, 32, 48, 64]:
        nl = n_parent + dn
        if nl >= 128 and nl <= 2000:
            lengths_to_try.append(nl)
    # Also try some fixed good lengths
    for nl in [640, 720, 768, 800, 850, 896, 960]:
        if nl not in lengths_to_try:
            lengths_to_try.append(nl)

    init_results = {}
    for nl in lengths_to_try:
        if time.monotonic() > t_phase1_end:
            break

        # Multiple initialization types
        inits = {
            'interp': interpolate_sequence(parent, nl),
            'bspline': make_bspline_like(nl, order=8),
            'asym': make_asymmetric_taper(nl, 0.35),
            'flattop': make_flattop(nl, 0.55),
        }
        if nl != n_parent:
            inits['interp2'] = interpolate_sequence(parent, nl)

        for name, a in inits.items():
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                obj = obj_normalized(a)
                key = (nl, name)
                if key not in init_results or obj < init_results[key][1]:
                    init_results[key] = (a, obj)
                best_seq, best_score = try_sequence(a, best_seq, best_score,
                                                     f"init_{name}_n{nl}")

    # Sort by objective
    sorted_inits = sorted(init_results.items(), key=lambda x: x[1][1])
    print(f"  Top 5 inits: " + ", ".join(
        f"({k[0]},{k[1]})={v[1]:.4f}" for k, v in sorted_inits[:5]))

    # === Phase 2: Block LP-FW with coarse blocks (400s) ===
    print("\n=== Phase 2: Block LP-FW (coarse, block=128) ===")
    top_candidates = sorted_inits[:4]
    t_alloc_coarse = min(400, (deadline - time.monotonic()) * 0.35)
    t_per_cand = t_alloc_coarse / max(1, len(top_candidates))

    phase2_best = None
    for i, (key, (a_init, obj_init)) in enumerate(top_candidates):
        remaining = deadline - time.monotonic() - 500
        if remaining < 30:
            break
        t_cand = min(t_per_cand, remaining)
        if t_cand < 20:
            break
        nl = len(a_init)
        print(f"  Coarse block LP-FW: n={nl}, init={key[1]}, t={t_cand:.0f}s")
        a_opt, obj = block_lpfw(a_init, time_limit=t_cand, block_size=128,
                                verbose=False, seed=seed + i)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                             f"coarse_n{nl}_{key[1]}")
        if phase2_best is None or obj < phase2_best[1]:
            phase2_best = (a_opt, obj, nl)

    if best_seq is not None:
        print(f"  After Phase 2: best={best_score:.6f}")

    # === Phase 3: Block LP-FW with fine blocks (400s) ===
    print("\n=== Phase 3: Block LP-FW (fine, block=64) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t_alloc_fine = min(400, (deadline - time.monotonic()) * 0.3)
        if t_alloc_fine > 30:
            print(f"  Fine block LP-FW: n={n_best}, t={t_alloc_fine:.0f}s")
            a_opt, obj = block_lpfw(best_seq, time_limit=t_alloc_fine, block_size=64,
                                    verbose=True, seed=seed + 100)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"fine_n{n_best}")

    # Also try fine blocks on the second-best length
    if best_seq is not None:
        for nl_try in [n_parent - 16, n_parent + 16]:
            if nl_try == len(best_seq):
                continue
            remaining = deadline - time.monotonic() - 300
            if remaining < 40:
                break
            interp = interpolate_sequence(best_seq, nl_try)
            t_cand = min(120, remaining * 0.5)
            if t_cand < 20:
                break
            print(f"  Fine block LP-FW: n={nl_try}, t={t_cand:.0f}s")
            a_opt, obj = block_lpfw(interp, time_limit=t_cand, block_size=64,
                                    verbose=False, seed=seed + 200)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"fine_n{nl_try}")

    # === Phase 4: Full LP-FW polish (200s) ===
    print("\n=== Phase 4: Full LP-FW ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t_alloc_full = min(200, (deadline - time.monotonic()) * 0.2)
        if t_alloc_full > 20:
            print(f"  Full LP-FW: n={n_best}, t={t_alloc_full:.0f}s")
            a_opt, obj = full_lpfw(best_seq, time_limit=t_alloc_full,
                                   verbose=False, seed=seed + 300)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"full_n{n_best}")

    # === Phase 5: Coordinate descent (150s) ===
    print("\n=== Phase 5: Coordinate Descent ===")
    if best_seq is not None:
        t_alloc_cd = min(150, (deadline - time.monotonic()) * 0.15)
        if t_alloc_cd > 15:
            print(f"  Coord descent: t={t_alloc_cd:.0f}s")
            a_opt, obj = coord_descent_polish(best_seq, time_limit=t_alloc_cd,
                                              verbose=False, seed=seed + 400)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"cd_n{len(a_opt)}")

    # === Phase 6: More block LP-FW with different block sizes ===
    print("\n=== Phase 6: Adaptive block sizes ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        for bs in [32, 48, 96]:
            if remaining < 30:
                break
            t_cand = min(80, remaining * 0.4)
            if t_cand < 15:
                break
            # Perturb slightly for diversity
            perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.001)
            a_opt, obj = block_lpfw(perturbed, time_limit=t_cand, block_size=bs,
                                    verbose=False, seed=seed + 500 + bs)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"adaptive_bs{bs}")
            remaining = deadline - time.monotonic() - 60

    # === Phase 7: Final deep optimization ===
    print("\n=== Phase 7: Final Deep Optimization ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        # Alternate between block LP-FW and coord descent
        for round_i in range(3):
            remaining = deadline - time.monotonic() - 10
            if remaining < 20:
                break
            t_each = remaining / (3 - round_i) / 2
            if t_each < 10:
                break

            # Block LP-FW
            a_opt, obj = block_lpfw(best_seq, time_limit=t_each, block_size=64,
                                    verbose=False, seed=seed + 600 + round_i)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"deep_block_r{round_i}")

            remaining = deadline - time.monotonic() - 10
            if remaining < 10 or best_seq is None:
                break

            # Coord descent
            t_cd = min(t_each, remaining - 5)
            if t_cd > 5:
                a_opt, obj = coord_descent_polish(best_seq, time_limit=t_cd,
                                                  verbose=False, seed=seed + 700 + round_i)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                     f"deep_cd_r{round_i}")

    # === Final: try a few more lengths ===
    print("\n=== Final Length Check ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for dl in [-8, -4, 4, 8]:
            nl = n_best + dl
            if nl < 128:
                continue
            remaining = deadline - time.monotonic() - 5
            if remaining < 5:
                break
            interp = interpolate_sequence(best_seq, nl)
            # Quick block LP-FW
            t_q = min(30, remaining * 0.3)
            if t_q < 5:
                break
            a_opt, obj = block_lpfw(interp, time_limit=t_q, block_size=64,
                                    verbose=False, seed=seed + 800)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"final_n{nl}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
