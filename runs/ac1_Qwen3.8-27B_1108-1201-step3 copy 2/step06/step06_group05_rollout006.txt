    Compute gradient direction of p-norm of c.
    Returns g where g[j] = 2 * sum_k w[k] * f[k-j] (proper range).
    w[k] proportional to c[k]^(p-1).
    """
    n = len(f)
    max_c = np.max(c)
    if max_c < 1e-15:
        return np.zeros(n)
    c_pos = np.maximum(c, 0)
    w = c_pos ** (p - 1)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    
    # corr[j] = sum_k w[k] * f[k-j] for j=0..n-1
    N = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.fft(w, N)
    F = np.fft.fft(f, N)
    corr = np.fft.ifft(W * np.conj(F), N)[:n].real
    g = 2.0 * corr
    return g


def compute_topk_subgrad(f, c, k):
    """
    Compute subgradient using top-k peaks.
    g[j] = 2/|active| * sum_{k in active} f[k-j] (proper range)
    """
    n = len(f)
    m = len(c)
    if k >= m:
        k = m
    top_idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(m)
    # Weight by excess above the k-th largest
    threshold = c[top_idx].min()
    excess = np.maximum(c - threshold, 0)
    if excess.sum() < 1e-30:
        w[top_idx] = 1.0 / k
    else:
        w = excess / excess.sum()
    
    N = 1 << (m + n - 1 - 1).bit_length()
    W = np.fft.fft(w, N)
    F = np.fft.fft(f, N)
    corr = np.fft.ifft(W * np.conj(F), N)[:n].real
    g = 2.0 * corr
    return g


def pnorm_grad_descent(f0, time_limit, verbose=False):
    """
    p-norm smoothed gradient descent with adaptive step size.
    Uses increasing p for progressively sharper approximation to max.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    # p schedule
    p_schedule = [4, 8, 16, 32, 64, 128]
    
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit * 0.85:
            break
        
        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.12, (time_limit - (time.monotonic() - t0)) * 0.2)
        
        it = 0
        no_improve = 0
        prev_g = None
        prev_f = None
        t = 1.0 / n  # initial step size
        
        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.85:
            it += 1
            c = fast_convolve(f)
            g = compute_pnorm_grad(f, c, p)
            
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            # Barzilai-Borwein step size
            if prev_g is not None:
                s_vec = f - prev_f
                y_vec = g - prev_g
                sy = s_vec @ y_vec
                yy = y_vec @ y_vec
                ss = s_vec @ s_vec
                if yy > 1e-20:
                    t_bb = sy / yy
                    t = max(1e-10, min(t_bb, 2.0 / g_norm))
            elif ss > 0:
                t_ss = ss / sy
                t = max(1e-10, min(t_ss, 2.0 / g_norm))
            
            # Gradient step with projection
            f_trial = f - t * g
            f_trial = project_simplex(f_trial)
            
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            
            if obj_trial <= best_obj + 1e-12:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # Backtrack
                t *= 0.5
                f_trial = f - t * g
                f_trial = project_simplex(f_trial)
                c_trial = fast_convolve(f_trial)
                obj_trial = c_trial.max()
                if obj_trial <= c.max() + 1e-10:
                    f = f_trial
                    if obj_trial < best_obj - 1e-13:
                        best_obj = obj_trial
                        best_f = f.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
                    t *= 0.5
            
            prev_g = g.copy()
            prev_f = f.copy()
            
            if no_improve > 60:
                # Restart
                rng = np.random.default_rng(it * 7 + p * 3)
                noise = 0.003 * rng.standard_normal(n)
                f = project_simplex(best_f + noise)
                t = 1.0 / n
                no_improve = 0
                prev_g = None
                prev_f = None
            
            if verbose and it % 200 == 0:
                print(f"    p={p} iter={it} obj={c.max():.10f} best={best_obj:.10f} t={t:.2e}")
        
        if verbose:
            print(f"  p={p} done: best={best_obj:.10f} elapsed={time.monotonic()-t0:.1f}s")
    
    return best_f, best_obj


def topk_subgrad_descent(f0, time_limit, k=10, verbose=False):
    """
    Subgradient method using top-k peak averaging.
    More robust than p-norm for very sharp peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    t = 5.0 / n  # step size
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        g = compute_topk_subgrad(f, c, k)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Decreasing step size: t_k = c / sqrt(k)
        t = min(5.0 / np.sqrt(it), 2.0 / g_norm)
        
        f_trial = f - t * g
        f_trial = project_simplex(f_trial)
        
        c_trial = fast_convolve(f_trial)
        obj_trial = c_trial.max()
        
        if obj_trial <= best_obj + 1e-12:
            f = f_trial
            if obj_trial < best_obj - 1e-13:
                best_obj = obj_trial
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            t *= 0.8
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 13 + 5)
            noise = 0.002 * rng.standard_normal(n)
            f = project_simplex(best_f + noise)
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    topk iter={it} obj={c.max():.10f} best={best_obj:.10f}")
    
    return best_f, best_obj
