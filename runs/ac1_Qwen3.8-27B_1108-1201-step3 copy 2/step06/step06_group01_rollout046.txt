import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring high-accuracy solvers."""
    global _lp_solver_found
    if '_lp_solver_found' in globals():
        return _lp_solver_found
    priority = [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]
    for solver_name in priority:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_found = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_found = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. Returns (b, M).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def frequency_domain_refine(a0, time_limit, n_iterations=5000):
    """
    Gerchberg-Saxton style frequency domain refinement.
    Iteratively:
    1. Compute auto-correlation
    2. Identify peaks
    3. In frequency domain, reduce spectral components contributing to peaks
    4. Invert to time domain and project to non-negative orthant
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current auto-correlation
        c = fast_convolve(a)

        # Find peaks (exclude endpoints which are just a[0]^2 and a[n-1]^2)
        c_work = c.copy()
        c_work[0] = -np.inf
        c_work[-1] = -np.inf
        peak_idx = np.argsort(c_work)[::-1][:min(10, len(c_work) - 2)]
        peak_vals = c[peak_idx]
        max_peak = c[peak_idx[0]]

        if max_peak <= best_score / (2.0 * n) * 1.001:
            # No improvement possible
            break

        # Get FFT of current signal
        A = fft(a, fft_size)

        # For each peak, compute the "error" signal in frequency domain
        # The contribution of a to peak k is: sum_i a[i]*a[k-i]
        # We want to reduce the spectral energy at frequencies that contribute to peaks
        # Simple approach: compute gradient direction and subtract in freq domain

        # Compute subgradient at the dominant peak
        k_star = int(peak_idx[0])
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        grad = np.zeros(n)
        for j in range(js, je + 1):
            grad[j] = 2.0 * a[k_star - j]

        # Compute FFT of gradient
        G = fft(grad, fft_size)

        # Newton-like step in frequency domain: reduce |A|^2 in directions that increase peaks
        # A_new = A - alpha * G / (2*A) (complex Newton for reducing |A|^2 in peak direction)
        # Safeguard: where A is small, use steepest descent instead
        eps = 1e-10
        A_safe = np.where(np.abs(A) > eps, A, eps)

        # Adaptive step size
        step_size = 0.5 * max_peak / (np.abs(A_safe).max() + eps)
        step_size = min(step_size, 0.1)

        A_new = A - step_size * G / (2.0 * A_safe)

        # Invert to time domain
        a_new = ifft(A_new, fft_size)[:n].real

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                A_try = A - frac * step_size * G / (2.0 * A_safe)
                a_try = ifft(A_try, fft_size)[:n].real
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                score_try = 2.0 * n * fast_convolve(a_try).max()
                if score_try < best_score - 1e-12:
                    best_score = score_try
                    best_a = a_try.copy()
                    a = a_try
                    break

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FreqDom iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def gerchberg_saxton(a0, time_limit):
    """
    GS-style: alternate between time-domain (non-negativity) and frequency-domain
    (flat spectrum) constraints.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    damping = 0.5

    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Compute spectrum
        A = fft(a, fft_size)
        power = np.abs(A) ** 2

        # Step 2: Flatten the spectrum (reduce peaks, increase valleys)
        # Target: uniform power = total_power / fft_size
        total_power = power.sum()
        target_power = total_power / fft_size

        # Damped flattening
        A_flat = np.zeros_like(A, dtype=complex)
        for i in range(fft_size):
            if power[i] > 0:
                phase = A[i] / np.abs(A[i])
                new_mag = damping * np.sqrt(target_power) + (1 - damping) * np.sqrt(power[i])
                A_flat[i] = phase * new_mag
            else:
                A_flat[i] = 0

        # Step 3: Invert to time domain
        a_time = ifft(A_flat, fft_size)[:n].real

        # Step 4: Enforce non-negativity
        a_time = np.maximum(a_time, 0)
        s_time = a_time.sum()
        if s_time < 1e-15:
            break
        a_time = a_time / s_time

        score = 2.0 * n * fast_convolve(a_time).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_time.copy()
        a = a_time

        # Gradually increase damping for stability
        damping = min(0.9, 0.3 + 0.0005 * it)

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T=200, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    from scipy.optimize import minimize
    n = len(a0)
    a_safe = np.maximum(a0, 1e-300)
    z0 = np.log(a_safe / a_safe.sum())
    t0 = time.monotonic()
    stopped = [False]

    def softmax_from_z(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        return exp_z / exp_z.sum()

    def soft_max_obj(z, T_param):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b - max_b)
        log_sum_exp = max_b + np.log(exp_b.sum())
        return (2.0 * n / T_param) * log_sum_exp

    def soft_max_grad(z, T_param):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b - max_b)
        w = exp_b / exp_b.sum()

        # Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a
        # = sum_k w[k] * 2 * T_k @ a (where T_k @ a = conv(a,a)[k] as function of a)
        # Actually conv(a,a)[k] = sum_i a[i]*a[k-i], d/da[j] = 2*a[k-j]
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad_a[j] += 2.0 * w[k] * a[k - j]

        # Convert from a-gradient to z-gradient (softmax parameterization)
        # a = softmax(z), da/dz = a*(I - 1*a^T)
        # grad_z = a * (grad_a - sum(grad_a * a))
        correction = np.dot(grad_a, a)
        grad_z = a * (grad_a - correction)
        return grad_z / T_param

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        return soft_max_obj(z, T)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        return soft_max_grad(z, T)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 50, 'disp': False, 'maxls': 50}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0

    return a_out

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Fallback initialization
        n_init = 768
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi, n_init)
        parent = parent + 0.02 * np.cos(2 * t)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Initialized fallback: n={len(parent)}, score={best_score:.6f}")

    # === Phase 1: Fictitious Play with Momentum ===
    print(f"\n=== Phase 1: FP with Momentum (best={best_score:.6f}) ===")
    n_parent = len(parent)
    fp_time_budget = min(500.0, (deadline - time.monotonic()) * 0.35)
    print(f"  FP budget: {fp_time_budget:.0f}s, n={n_parent}")

    a_fp, score_fp, iters = fictitious_play_momentum(
        n_parent, fp_time_budget, initial=parent, solver=solver, verbose=True
    )
    best_seq, best_score = check_and_update(a_fp, f"FP_mom_n{n_parent}_it{iters}", best_seq, best_score)

    # === Phase 2: Frequency Domain Refinement ===
    print(f"\n=== Phase 2: Frequency Domain (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_fd = min(rem * 0.4, 120.0)
            print(f"  FreqDom n={len(best_seq)}, budget={t_fd:.0f}s")
            a_fd, _ = frequency_domain_refine(best_seq, t_fd)
            best_seq, best_score = check_and_update(a_fd, "freqdom", best_seq, best_score)

    # === Phase 3: Gerchberg-Saxton ===
    print(f"\n=== Phase 3: Gerchberg-Saxton (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_gs = min(rem * 0.3, 80.0)
            print(f"  GS n={len(best_seq)}, budget={t_gs:.0f}s")
            a_gs, _ = gerchberg_saxton(best_seq, t_gs)
            best_seq, best_score = check_and_update(a_gs, "GS", best_seq, best_score)

    # === Phase 4: L-BFGS Polish ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for T_val in [100, 300, 800]:
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 60.0)
            print(f"  L-BFGS T={T_val}, n={n_best}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_refine(best_seq, t_alloc, T=T_val)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_ps = min(rem * 0.5, 60.0)
            a_ps, _ = peak_shave_local(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 6: Second FP round with refined start ===
    print(f"\n=== Phase 6: FP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80
        if rem > 40:
            t_alloc = min(rem * 0.5, 200.0)
            print(f"  FP round 2: n={n_best}, budget={t_alloc:.0f}s")
            a_fp2, _, iters2 = fictitious_play_momentum(
                n_best, t_alloc, initial=best_seq, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_fp2, f"FP2_n{n_best}", best_seq, best_score)

            # Quick L-BFGS after FP2
            rem = deadline - time.monotonic() - 40
            if rem > 10 and best_seq is not None:
                a_lb2 = lbgfs_refine(best_seq, min(rem * 0.6, 40.0), T=300)
                best_seq, best_score = check_and_update(a_lb2, "lbgfs_after_fp2", best_seq, best_score)

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5]:
            alt_n = int(n_best * factor)
            if 128 <= alt_n <= 4096 and alt_n != n_best:
                alt_lengths.append(alt_n)
        # Also try some fixed values
        for fixed_n in [512, 1024, 1536]:
            if fixed_n not in alt_lengths and 128 <= fixed_n <= 4096:
                alt_lengths.append(fixed_n)

        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic() - 30
            if rem < 25:
                break
            t_alloc = min(rem * 0.4, 100.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = fictitious_play_momentum(
                alt_n, t_alloc, initial=init_alt, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_alt, f"FP_alt_n{alt_n}", best_seq, best_score)

            if best_seq is not None and len(best_seq) == alt_n:
                rem = deadline - time.monotonic() - 20
                if rem > 10:
                    a_ps_alt, _ = peak_shave_local(best_seq, min(rem * 0.6, 30.0))
                    best_seq, best_score = check_and_update(a_ps_alt, f"ps_alt_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_final = min(rem * 0.4, 30.0)
            a_final = lbgfs_refine(best_seq, t_final, T=500, maxiter=3000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_ps_final, _ = peak_shave_local(best_seq, min(rem * 0.6, 20.0))
            best_seq, best_score = check_and_update(a_ps_final, "final_ps", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent if parent is not None else np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
