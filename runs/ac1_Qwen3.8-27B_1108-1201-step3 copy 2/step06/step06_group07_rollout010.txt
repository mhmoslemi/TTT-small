def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_at_peaks(a, peak_indices, weights):
    """Compute weighted subgradient of sum_k w_k * (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g

def lbfgsb_polish(a0, time_limit, temp=0.001, max_iter=500, verbose=False, label=""):
    """
    L-BFGS-B polish using softmax reparameterization.
    a = softmax(z), minimize smoothed max of conv(a,a).
    """
    a_init = np.maximum(a0.astype(np.float64), 1e-15)
    a_init = a_init / a_init.sum()
    n = len(a_init)

    # Initialize z from a: z = log(a) (since softmax(log(a)/a.sum()) ≈ a when a is close to uniform)
    # Better: z = log(a) - log(n) so that softmax(z) ≈ a
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 -= z0.mean()  # center for numerical stability

    best_obj = fast_convolve(a_init).max()
    best_a = a_init.copy()
    t0 = time.monotonic()

    def objective_and_grad(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        # Smoothed max
        scaled = conv / temp
        scaled -= scaled.max()
        log_sum_exp = np.log(np.sum(np.exp(scaled))) + conv.max() / temp
        obj = temp * log_sum_exp

        # Gradient: d(obj)/d(a) * da/d(z)
        weights = np.exp(scaled) / np.sum(np.exp(scaled))
        g_a = compute_gradient_at_peaks(a, np.arange(len(conv)), weights)
        # Gradient through softmax: grad_z = a * (g_a - <g_a, a>)
        grad_z = a * (g_a - np.dot(g_a, a))

        return obj, grad_z

    # Run L-BFGS-B
    result = scipy_minimize(
        objective_and_grad, z0, method='L-BFGS-B',
        jac=True, options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8}
    )

    if result.success or result.fun < objective_and_grad(z0)[0]:
        z_opt = result.x
        a_opt = np.exp(z_opt - z_opt.max())
        a_opt = a_opt / a_opt.sum()
        obj_opt = fast_convolve(a_opt).max()
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt
            if verbose:
                print(f"    {label} L-BFGS-B: obj={best_obj:.10f} (iters={result.nit})")

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    {label} L-BFGS-B done in {elapsed:.1f}s")

    return best_a, best_obj


def spectral_init(n, iterations=5000, verbose=False):
    """
    Frequency-domain inspired initialization.
    Start with random phase spectrum, iteratively project to non-negativity
    and adjust spectrum to flatten autocorrelation.
    """
    rng = np.random.default_rng(12345)
    fft_size = 1 << (2 * n - 1).bit_length()

    # Start with random phases
    rfft_len = fft_size // 2 + 1
    magnitude = np.ones(rfft_len)
    phase = rng.standard_normal(rfft_len)
    phase[0] = 0  # DC has no phase

    spectrum = magnitude * np.exp(1j * phase)

    for it in range(iterations):
        # Inverse FFT to get time domain
        a_extended = np.fft.irfft(spectrum, fft_size)
        a = a_extended[:n]

        # Project to non-negative and normalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        # Compute current autocorrelation
        conv = fast_convolve(a)

        # Compute target spectrum: we want conv to be flat
        # Flat conv of length 2n-1 → spectrum with only DC
        # So we want |A(f)|^2 ∝ delta_DC, meaning A(f) ≈ constant
        # But we also need a >= 0, so we can't achieve perfect flat
        # Instead: nudge the spectrum toward flat magnitude

        current_mag = np.abs(np.fft.rfft(a, fft_size))
        # Target: proportional to sqrt(2n-1) at DC, 0 elsewhere (ideal)
        # Practical: decay away from DC
        freqs = np.arange(rfft_len)
        target_mag = np.exp(-0.5 * (freqs / (fft_size * 0.3)) ** 2)

        # Blend: move toward target magnitude, keep current phase
        blend = 0.01
        new_mag = (1 - blend) * current_mag + blend * target_mag
        current_phase = np.angle(np.fft.rfft(a, fft_size))
        spectrum = new_mag * np.exp(1j * current_phase)

    return a


def weighted_iterative_method(a0, time_limit, verbose=False, label=""):
    """
    Weighted iterative method: iteratively adjust a to equalize the top peaks.
    For each peak, compute its contribution pattern and adjust a to reduce it.
    Uses a "water-filling" inspired approach.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    mu = 0.01  # relaxation parameter

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False

        # Find top peaks
        K = min(12, len(conv))
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k in peak_indices:
            k = int(k)
            if conv[k] < M * 0.999:
                continue

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            partners = k - idx
            products = a[idx] * a[partners]

            # Find the pair contributing most to this peak
            j_max_local = np.argmax(products)
            j_max = idx[j_max_local]
            k_max = partners[j_max_local]

            # Try reducing the larger of the two
            if a[j_max] >= a[k_max]:
                reduce_idx = j_max
                protect_idx = k_max
            else:
                reduce_idx = k_max
                protect_idx = j_max

            # Reduce and redistribute
            for frac in [0.85, 0.9, 0.95]:
                diff = a[reduce_idx] * (1 - frac)
                if diff < 1e-15:
                    continue
                trial = a.copy()
                trial[reduce_idx] *= frac

                # Redistribute to positions that DON'T contribute to this peak
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib[reduce_idx]:
                    non_contrib[reduce_idx] = False
                if non_contrib.sum() > 0:
                    nc_idx = np.where(non_contrib)[0]
                    nc_weights = a[nc_idx]
                    nc_sum = nc_weights.sum()
                    if nc_sum > 1e-15:

                        trial = np.maximum(trial, 0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-14:
                            a = trial
                            best_obj = obj
                            best_a = trial.copy()
                            improved = True
                            break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        # Symmetrize periodically

            if fast_convolve(a_sym).max() < best_obj:
                best_obj = fast_convolve(a_sym).max()
                best_a = a_sym.copy()
                a = a_sym.copy()

    return best_a, best_obj


def fine_coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with binary search on each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj

            old_val = a[j]
            if old_val < 1e-13:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.4:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 6
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, center1_frac=0.3, center2_frac=0.7, sigma=0.08):
    """Bimodal Gaussian."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center1_frac) / sigma) ** 2) + \
        np.exp(-0.5 * ((x - center2_frac) / sigma) ** 2)
    a = a / a.sum()
    return a


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Generate diverse initializations ===
    print("\n=== Phase 1: Initializations ===")
    n_values = [768, 896, 1024]
    if parent is not None:
        n_values.insert(0, len(parent))

    init_pool = []

    # Parent-based
    if parent is not None:
        init_pool.append((parent, "parent"))
        init_pool.append((symmetrize(parent), "parent_sym"))
        # Interpolate parent to other sizes
        for n in [896, 1024]:
            if n != len(parent):
                init_pool.append((interpolate_sequence(parent, n), f"parent_interp_n{n}"))

    # Analytical initializations
    for n in n_values:
        init_pool.append((make_gaussian(n, 0.25), f"gauss25_n{n}"))
        init_pool.append((make_gaussian(n, 0.35), f"gauss35_n{n}"))
        init_pool.append((make_bell(n, 0.5), f"bell_n{n}"))
        init_pool.append((make_bimodal(n), f"bimodal_n{n}"))

    # Quick eval all inits
    best_inits = []
    for a_init, lbl in init_pool:
        a_norm = a_init / a_init.sum()
        obj = 2 * len(a_norm) * fast_convolve(a_norm).max()
        best_inits.append((obj, a_norm, lbl))
        sc = try_seq(a_norm, f"init_{lbl}")

    best_inits.sort(key=lambda x: x[0])
    print(f"  Top 3 inits: {[(l, f'{o:.6f}') for _, _, l in best_inits[:3]]}")

    # === Phase 2: Active-set mirror descent on top candidates ===
    print("\n=== Phase 2: Active-Set Mirror Descent ===")
    remaining = deadline - time.monotonic()
    top_candidates = best_inits[:4]
    if best_seq is not None and best_seq not in [c[1] for c in top_candidates]:
        top_candidates.insert(0, (best_score / 2, best_seq, "best_so_far"))

    for idx, (obj, a_cand, lbl) in enumerate(top_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = remaining * (0.5 / max(1, len(top_candidates) - idx))
        t_alloc = min(t_alloc, remaining - 120)
        if t_alloc < 30:
            t_alloc = remaining * 0.3

        print(f"  Running ASD on '{lbl}' for {t_alloc:.0f}s")
        a_opt, obj_opt = active_set_mirror_descent(
            a_cand, time_limit=t_alloc, band_frac=0.005,
            lr_init=0.008, verbose=True, label=f"ASD_{lbl}"
        )
        try_seq(a_opt, f"ASD_{lbl}")

        # Symmetrized version
        a_sym = symmetrize(a_opt)
        try_seq(a_sym, f"ASD_{lbl}_sym")

    # === Phase 3: L-BFGS-B polish ===
    print("\n=== Phase 3: L-BFGS-B Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_lbfgs = min(remaining * 0.2, 180)
            print(f"  L-BFGS-B time: {t_lbfgs:.0f}s")
            a_opt, obj_opt = lbfgsb_polish(
                best_seq, time_limit=t_lbfgs, temp=0.0005,
                max_iter=300, verbose=True, label="LBFGS"
            )
            try_seq(a_opt, "LBFGS")

            # Try with different temperature
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_opt2, _ = lbfgsb_polish(
                    best_seq, time_limit=min(remaining * 0.3, 60), temp=0.002,
                    max_iter=200, verbose=True, label="LBFGS_warm"
                )
                try_seq(a_opt2, "LBFGS_warm")

    # === Phase 4: Weighted iterative method ===
    print("\n=== Phase 4: Weighted Iterative ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_wim = min(remaining * 0.15, 120)
            a_opt, obj_opt = weighted_iterative_method(
                best_seq, time_limit=t_wim, verbose=True, label="WIM"
            )
            try_seq(a_opt, "WIM")

    # === Phase 5: Fine coordinate descent ===
    print("\n=== Phase 5: Fine Coordinate Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_cd = min(remaining * 0.15, 120)
            a_opt, obj_opt = fine_coordinate_descent(
                best_seq, time_limit=t_cd, verbose=True, label="CD"
            )
            try_seq(a_opt, "CD")

    # === Phase 6: Second round ASD with different params ===
    print("\n=== Phase 6: ASD Round 2 (tighter) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_asd2 = min(remaining * 0.25, 200)
            # Start from symmetrized best
            a_sym = symmetrize(best_seq)
            a_opt, obj_opt = active_set_mirror_descent(
                a_sym, time_limit=t_asd2, band_frac=0.002,
                lr_init=0.004, verbose=True, label="ASD2_sym"
            )
            try_seq(a_opt, "ASD2_sym")

            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_opt2, _ = active_set_mirror_descent(
                    best_seq, time_limit=min(remaining * 0.4, 80), band_frac=0.003,
                    lr_init=0.005, verbose=True, label="ASD2_direct"
                )
                try_seq(a_opt2, "ASD2_direct")

    # === Phase 7: Multi-restart with noise ===
    print("\n=== Phase 7: Multi-Restart ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_restarts = 3
        t_per = min(remaining * 0.15 / n_restarts, 60)
        if t_per > 15:
            for r in range(n_restarts):
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                if r == 0:
                    noise = rng.standard_normal(len(best_seq)) * 0.001
                    f_init = np.maximum(best_seq + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                elif r == 1:
                    f_init = symmetrize(best_seq)
                    noise = rng.standard_normal(len(f_init)) * 0.0005
                    f_init = np.maximum(f_init + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                else:
                    # Blend with a different shape
                    alt = make_gaussian(len(best_seq), 0.3)
                    f_init = 0.95 * best_seq + 0.05 * alt
                    f_init = f_init / f_init.sum()

                t_alloc = min(t_per, remaining - 25)
                if t_alloc < 10:
                    break
                a_opt, _ = active_set_mirror_descent(
                    f_init, time_limit=t_alloc, band_frac=0.004,
                    lr_init=0.006, label=f"restart_{r}"
                )
                try_seq(a_opt, f"restart_{r}")

    # === Phase 8: Final micro-polish ===
    print("\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_polish = min(remaining - 10, 60)
            a_opt, _ = fine_coordinate_descent(
                best_seq, time_limit=t_polish, label="polish"
            )
            try_seq(a_opt, "polish")

            # Try symmetrized final
            a_s = symmetrize(best_seq)
            try_seq(a_s, "final_sym")

            # One more ASD pass with very tight band
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_opt, _ = active_set_mirror_descent(
                    best_seq, time_limit=min(remaining - 10, 40),
                    band_frac=0.001, lr_init=0.002, label="ultra_fine"
                )
                try_seq(a_opt, "ultra_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
