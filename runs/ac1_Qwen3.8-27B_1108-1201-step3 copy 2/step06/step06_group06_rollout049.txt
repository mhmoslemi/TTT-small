    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    prev_grad = None
    prev_a = None
    lr = 1.0 / n
    
    no_improve = 0
    it = 0

        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
            
        # Find top-k peaks
        top_k = min(3, 2*n-1)
        peak_indices = np.argsort(conv)[-top_k:]
        
        # Compute weighted subgradient
        # Weight peaks by their excess over the mean
        mean_c = conv.mean()
        excess = np.maximum(conv - mean_c, 0)

        w = excess / (excess.sum() + 1e-15)

"""
Multi-scale subgradient optimization with multi-peak tracking for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses coarse-to-fine resolution, Barzilai-Borwein
step sizes, and Adam polishing.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def subgrad_peak(a, k):
    """Subgradient of c[k] (full convolution) w.r.t. a, where c = a * a."""
    n = len(a)
    m = np.arange(n)
    idx1 = k - m
    idx2 = m - k
    grad = np.zeros(n)
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask1] += a[idx1[mask1]]
    grad[mask2] += a[idx2[mask2]]
    return grad

def subgrad_topk(a, conv, k_peaks=3):
    """Weighted subgradient using top-k peaks."""
    peak_vals = conv.copy()
    # Softmax-like weighting
    max_val = conv.max()
    if max_val < 1e-30:
        return np.zeros(len(a))
    # Weight peaks by exp((c[k] - max)/tau) for small tau
    tau = 0.1 * max_val
    w = np.exp((conv - max_val) / tau)
    w /= w.sum()
    # Only use top-k
    top_idx = np.argsort(conv)[-k_peaks:]
    w_masked = np.zeros_like(w)
    w_masked[top_idx] = w[top_idx]
    w_masked /= w_masked.sum()
    
    g = np.zeros(len(a))
    for i in top_idx:
        g += w_masked[i] * subgrad_peak(a, int(i))
    return g

def normalize_simplex(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s
