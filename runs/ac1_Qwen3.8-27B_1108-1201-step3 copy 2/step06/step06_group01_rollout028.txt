import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = rfft(a, fs)
    return irfft(A * A, fs)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fs = 1 << (size - 1).bit_length()
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A * B, fs)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _find_lp_solver():
    """Find a working LP solver."""
    global _cached_solver
    if '_cached_solver' in globals() and _cached_solver is not None:
        return _cached_solver
    candidates = [cp.MOSEK, cp.GUROBI, cp.CBC, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in candidates:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = solver
                return solver
        except Exception:
            continue
    _cached_solver = cp.ECOS
    return cp.ECOS


def lp_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. Returns (b, M).
    """
    if solver is None:
        solver = _find_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= t_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(t_var.value)


def compute_smooth_gradient(a, T_smooth):
    """Compute gradient of (1/T)log(sum exp(T*c[m])) w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    Tc = T_smooth * c
    Tc_shifted = Tc - Tc.max()
    w = np.exp(Tc_shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    # gradient[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = conv(flip(w), a)[2n-2-i]
    fw = w[::-1]
    fs = 1 << (3 * n - 1).bit_length()
    FW = rfft(fw, fs)
    A = rfft(a, fs)
    conv_result = irfft(FW * A, fs)
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return grad, w


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / ind > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_smoothed(a0, time_limit, T_init=100.0, T_min=20.0, max_iters=5000):
    """Nesterov-accelerated gradient descent on smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    T_cur = T_init
    t_nest = 1.0
    a_prev = a.copy()
    a_ymm = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Compute gradient
        grad, w = compute_smooth_gradient(a, T_cur)
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        y = a + beta * (a - a_prev)
        # Gradient step with adaptive learning rate
        lr = T_cur / (50.0 * t_new)
        a_candidate = y - lr * grad
        a_new = project_simplex(a_candidate)
        a_new = a_new / a_new.sum()
        # Evaluate
        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
        a_prev = a
        a = a_new
        t_nest = t_new
        # Decrease temperature
        T_cur = max(T_min, T_cur * 0.999)
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov it={it+1} ({elapsed:.0f}s): score={best_score:.6f} T={T_cur:.1f}")
    return best_a, best_score


def lbfgs_polish(a0, time_limit, T=300, maxiter=3000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-300)
    z0 = np.log(a_safe / a_safe.sum())
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-30:
            return 1e15
        a = a / s
        c = fast_conv(a)
        Tc = T * c
        Tc_shifted = Tc - Tc.max()
        lse = Tc.max() + np.log(np.exp(Tc_shifted).sum())
        return (2.0 * n / T) * lse

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-30:
            return np.zeros_like(z)
        a = a / s
        grad, _ = compute_smooth_gradient(a, T)
        # Transform gradient to z-space
        S = np.dot(grad, a)
        grad_z = a * (grad / a - S / a)
        return grad_z

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0.copy()
    return a_out
