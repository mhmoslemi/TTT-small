LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the linear program approach: at each step, find the best direction g
that maximizes sum(g) subject to a*g <= M pointwise, then take a convex
combination. Combined with mirror descent polishing for local refinement.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x>=0, sum(x)=S}."""
    S = v.sum()
    if S <= 0:
        return np.zeros_like(v)
    v = v / S  # project to sum=1 first
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_with_a(a, g):
    """Compute a * g (linear convolution) efficiently using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(A * G, fft_size)[:size]

    Returns g normalized to sum(g) = 1, or None if failed.
    """
    n = len(a)
    size = 2 * n - 1

    # C[k, j] = a[j] if 0 <= k-j < n, else 0

    cols = []

            cols.extend(range(js, je + 1))
            vals.extend(a[js:je+1].tolist())

    b_ub = np.full(size, M)

def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01,
                   lr_decay=0.9999, verbose=False, label=""):
    """Fast multiplicative weight mirror descent on the simplex (sum=1)."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))
        
        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        g = compute_subgradient(a, peak_indices, w)
        
        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-10:
                continue
            log_update = -trial_lr * g
            log_update = np.clip(log_update, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break
        
        log_update = -best_lr * g
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        new_obj = fast_convolve(a).max()
        
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * lr_decay, 1e-8)
        
        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0
        
        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            if fast_convolve(a_sym).max() < best_obj:
                best_obj = fast_convolve(a_sym).max()
                best_a = a_sym.copy()
                a = a_sym.copy()
        
        if lr < 1e-9:
            break
    
    return best_a, best_obj


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_uni(n):
    return np.ones(n) / n


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym

"""
LP-based optimization for step function coefficient minimization.
Primary method: Linear programming to find best improvement direction,
combined with mirror descent for local polishing. The LP at each step
finds g >= 0 maximizing sum(g) subject to (a*g)_k <= M for all k,
then a line search determines the optimal convex combination step.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    v = v / s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve_direction(a, M):
    """
    Solve: max sum(g) s.t. g >= 0, (a*g)_k <= M for all k = 0,...,2n-2.
    Returns g normalized to sum(g) = 1, or None.
    """
    n = len(a)
    size = 2 * n - 1
    
    rows = []
    cols = []
    vals = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
            vals.extend(a[js:je + 1].tolist())
    
    C = csr_matrix((vals, (rows, cols)), shape=(size, n))
    c_obj = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def lp_line_search(a, g, M):
    """Find best t minimizing max(((1-t)*a + t*g) * ((1-t)*a + t*g))."""
    best_t = 0.0
    best_obj = M
    
    ts = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7]
    for t in ts:
        a_new = (1.0 - t) * a + t * g
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    if best_t > 0:
        idx = ts.index(best_t)
        if idx > 0:
            lo = ts[idx - 1]
        else:
            lo = 0.0
        if idx < len(ts) - 1:
            hi = ts[idx + 1]
        else:
            hi = 1.0
        for t in np.linspace(lo, hi, 25):
            a_new = (1.0 - t) * a + t * g
            obj = fast_convolve(a_new).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t
    
    return best_t, best_obj
