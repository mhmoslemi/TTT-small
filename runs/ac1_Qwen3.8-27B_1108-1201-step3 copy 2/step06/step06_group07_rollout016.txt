def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent(f0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.005,
                   verbose=False, label=""):

    a = np.maximum(f0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

        conv = fast_convolve(a)
        M = conv.max()

        scaled = pv / temp
        scaled -= scaled.max()

            w[np.argmax(pv)] = 1.0

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:

            if trial_lr < 1e-10:
                continue
            log_update = np.clip(-trial_lr * g, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = np.clip(-best_lr * g, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temp = min(temp * 1.002, 0.05)
        if no_improve > 80:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * 0.9999, 1e-9)

        if no_improve > 400:
            rng = np.random.default_rng(it * 7919 + 13)
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(best_a + noise, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if lr < 1e-10:
            break

    return best_a, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False, label=""):
    """Fine coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, c1=0.3, c2=0.7, sigma=0.08):
    """Bimodal Gaussian."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = a / a.sum()
    return a


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP-based optimization (main workhorse)
    Phase 2: Mirror descent from best LP result
    Phase 3: Multi-start with diverse initializations and lengths
    Phase 4: Coordinate descent polish
    Phase 5: Final symmetrization and verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_base = 768 if parent is None else len(parent)

    # === Phase 1: LP-based optimization from parent (main) ===
    print("\n=== Phase 1: LP Optimization (main) ===")
    if parent is not None:
        remaining = deadline - time.monotonic()
        t_lp = min(remaining * 0.35, 700)
        print(f"  Time: {t_lp:.0f}s, n={n_base}")
        f_opt, obj = lp_optimize(parent, time_limit=t_lp, verbose=True, label="LP_main")
        try_seq(f_opt, "LP_main")

    # === Phase 2: LP from symmetrized parent ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print("\n=== Phase 2: LP from symmetrized ===")
            t_lp2 = min(remaining * 0.15, 250)
            f_sym = symmetrize(best_seq)
            f_opt, obj = lp_optimize(f_sym, time_limit=t_lp2, verbose=True, label="LP_sym")
            try_seq(f_opt, "LP_sym")

    # === Phase 3: Mirror descent from best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print("\n=== Phase 3: Mirror Descent ===")
            t_md = min(remaining * 0.15, 250)
            f_opt, obj = mirror_descent(best_seq, time_limit=t_md, top_k=30,
                                        temp_init=0.0003, lr_init=0.005,
                                        verbose=True, label="MD")
            try_seq(f_opt, "MD")

    # === Phase 4: Multi-start with different initializations/lengths ===
    print("\n=== Phase 4: Multi-start ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        init_list = []
        # Different lengths
        for n_try in [n_base, 512, 1024]:
            init_list.append((make_gaussian(n_try, 0.25), f"gauss_{n_try}"))
            init_list.append((make_gaussian(n_try, 0.35), f"gaussW_{n_try}"))
            init_list.append((make_bell(n_try, 0.5), f"bell_{n_try}"))

        # From best with noise
        if best_seq is not None:
            noise = rng.standard_normal(n_base) * 0.005
            f_noisy = np.maximum(best_seq + noise, 1e-15)
            f_noisy = f_noisy / f_noisy.sum()
            init_list.append((f_noisy, "noisy_best"))

        # Pure LP on diverse inits with limited time each
        t_per_init = min(remaining * 0.25 / max(len(init_list), 1), 40)
        for f_init, label in init_list:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            t_alloc = min(t_per_init, remaining - 20)
            if t_alloc < 10:
                break
            print(f"  Trying {label} (n={len(f_init)}), time={t_alloc:.0f}s")
            f_opt, obj = lp_optimize(f_init, time_limit=t_alloc, verbose=False, label=label)
            try_seq(f_opt, f"LP_{label}")

    # === Phase 5: Coordinate descent polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 5: Coordinate Descent Polish ===")
            t_cd = min(remaining * 0.15, 150)
            f_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_cd,
                                                   verbose=True, label="CD")
            try_seq(f_opt, "CD_polish")

    # === Phase 6: Final LP refinement on best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 6: Final LP Refinement ===")
            t_final = min(remaining * 0.3, 200)
            f_opt, obj = lp_optimize(best_seq, time_limit=t_final, verbose=True, label="LP_final")
            try_seq(f_opt, "LP_final")

            # Also try MD from the refined result
            remaining = deadline - time.monotonic()
            if remaining > 20:
                t_md2 = min(remaining * 0.4, 100)
                f_opt, obj = mirror_descent(f_opt, time_limit=t_md2, top_k=20,
                                            temp_init=0.0001, lr_init=0.003,
                                            verbose=False, label="MD_final")
                try_seq(f_opt, "MD_final")

    # === Phase 7: Symmetrization and final check ===
    if best_seq is not None:
        print("\n=== Phase 7: Final Symmetrization ===")
        a_s = symmetrize(best_seq)
        try_seq(a_s, "final_sym")

        # Try small perturbations of the symmetric version
        for noise_scale in [0.0005, 0.001, 0.002]:
            noise = rng.standard_normal(len(best_seq)) * noise_scale
            f_pert = np.maximum(a_s + noise, 1e-15)
            f_pert = f_pert / f_pert.sum()
            try_seq(f_pert, f"pert_{noise_scale}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
