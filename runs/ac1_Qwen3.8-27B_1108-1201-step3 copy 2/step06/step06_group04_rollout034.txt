    n = len(f)
    conv = fast_convolve(f)
    m = conv.max()
    weights = np.exp((conv - m) / temperature)
    weights /= weights.sum()
    # Hankel matrix
    idx = np.arange(n)
    H = weights[idx[:, None] + idx[None, :]]
    grad = 2.0 * (H @ f)
    return grad

def adam_smoothed_optimize(f0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, 
                            temp_init=0.01, temp_end=0.001, verbose=False):
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    t0 = time.monotonic()
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    no_improve = 0
    
    # Anneal temperature
    n_expected_steps = 1000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, t / n_expected_steps)
        temp = temp_init * (temp_end / temp_init) ** progress
        temp = max(temp, 1e-6)
        
        grad = gradient_smoothed(f, temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        f_new = f - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s_new
        
        # Check improvement
        obj_new = 2.0 * n * fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Periodic perturbation
        if no_improve > 50:
            rng = np.random.default_rng(t * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamSmoothed: t={t}, obj={best_obj:.8f}, temp={temp:.6f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_obj

def peak_directed_gradient(f, top_k=10):
    """Get gradient focusing only on top-k peaks."""
    n = len(f)
    conv = fast_convolve(f)
    # Get top-k peak indices
    peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    peak_weights = conv[peak_idx]
    peak_weights = peak_weights / peak_weights.sum()
    
    # For each peak at position k, the gradient contribution is 2*f[k-i] for coordinate i
    # Weighted sum: grad[i] = 2 * sum_{k in peaks} w_k * f[k-i]
    grad = np.zeros(n)
    for k, w in zip(peak_idx, peak_weights):
        k = int(k)
        # grad[i] += 2 * w * f[k-i] for i where 0 <= k-i < n
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            grad[i] += 2.0 * w * f[k - i]
    return grad

subject to f >= 0, 1^T f = 1

1^T f = 1
f >= 0

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-Frank-Wolfe, Adam with log-sum-exp smoothing, peak-directed
coordinate descent, and multi-restart perturbation.
"""
import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_val(f):
    """Objective for sum-normalized f: 2*n*max(conv(f,f))."""
    n = len(f)
    if n == 0:
        return np.inf
    return 2.0 * n * fast_convolve(f).max()

def build_lp_matrix(f, n):
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub

def solve_lp(f, M, n, A_ub):
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

def lp_fw_step(f, M, n, A_ub):
    """Single LP-FW step: find direction g, take step towards g."""
    g = solve_lp(f, M, n, A_ub)
    if g is None:
        return None, None
    gs = g.sum()
    if gs < 1e-15:
        return None, None
    g = g / gs
    return g

def golden_search(f, g, n, max_iter=10):
    """Golden section search for step size."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    f_max_cur = fast_convolve(f).max()
    
    def eval_a(a):
        ft = f + a * (g - f)
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return f_max_cur
        return fast_convolve(ft / s).max()
    
    # Coarse grid
    best_a, best_v = 0.0, f_max_cur
    for a in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        v = eval_a(a)
        if v < best_v:
            best_v = v
            best_a = a
    
    # Golden section
    span = 0.2
    lo = max(0.0, best_a - span)
    hi = min(1.0, best_a + span)
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    c_val = eval_a(c_pt)
    d_val = eval_a(d_pt)
    
    for _ in range(max_iter):
        if hi - lo < 1e-5:
            break
        if c_val < d_val:
            hi, d_pt = d_pt, c_pt
            d_val = c_val
            c_pt = hi - gr * (hi - lo)
            c_val = eval_a(c_pt)
        else:
            lo, c_pt = c_pt, d_pt
            c_val = d_val
            d_pt = lo + gr * (hi - lo)
            d_val = eval_a(d_pt)
        mid = (lo + hi) / 2.0
        mid_v = eval_a(mid)
        if mid_v < best_v:
            best_v = mid_v
            best_a = mid
    
    return best_a

def lp_fw_optimize(f0, time_limit, verbose=False):

    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    m_relax = 1.0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 >= time_limit:
            break
        
        A_ub = build_lp_matrix(f, n)
        g = solve_lp(f, M * m_relax, n, A_ub)
        
        if g is None:
            m_relax = min(m_relax * 1.2, 8.0)
            if m_relax >= 8.0:
                # Perturb
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max()
                m_relax = 1.0
                no_improve = 0
            continue
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            m_relax *= 1.1
            continue
        g = g / gs
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 2
        elif diff < 1e-6:
            no_improve += 1
        else:
            no_improve = 0
        
        # Line search
        alpha = golden_search(f, g, n, max_iter=10)
        if alpha < 1e-6:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)
        else:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn > 1e-15:
                f_new = f_new / sn
                f = f_new
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                cur_obj = 2.0 * n * M

        if no_improve > 80:
            # Structured perturbation
            rng = np.random.default_rng(it * 37 + 11)
            scale = 0.003 * min(m_relax, 4.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            m_relax = 1.0
            no_improve = 0
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW it={it}: obj={best_obj:.8f}, relax={m_relax:.2f}, t={elapsed:.1f}s")
    
    return best_f, best_obj

def adam_smoothed(f0, time_limit, lr_init=0.05, temp_init=0.02, verbose=False):
    """Adam optimizer with log-sum-exp smoothing."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 >= time_limit:
            break
        
        # Anneal temperature
        progress = min(1.0, t / 2000.0)
        temp = temp_init * (0.05) ** progress
        temp = max(temp, 0.0005)
        
        # Compute smoothed gradient
        conv = fast_convolve(f)
        m_val = conv.max()
        weights = np.exp((conv - m_val) / temp)
        weights /= weights.sum()
        
        # Hankel matrix for gradient
        idx = np.arange(n)
        H = weights[idx[:, None] + idx[None, :]]
        grad = 2.0 * (H @ f)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.maximum(f_new, 0)
        sn = f_new.sum()
        if sn < 1e-15:
            continue
        f_new = f_new / sn
        f = f_new
        
        # Check
        cur_obj = 2.0 * n * fast_convolve(f).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay learning rate
        lr *= 0.9995
        
        # Perturbation if stuck
        if no_improve > 80:
            rng = np.random.default_rng(t * 23 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t}: obj={best_obj:.8f}, temp={temp:.5f}, t={elapsed:.1f}s")
        
        if lr < 1e-10:
            break
    
    return best_f, best_obj

def peak_cd(f0, time_limit, verbose=False):
    """Peak-directed coordinate descent."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 60:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        # Top peaks
        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue
            
            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(5, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                
                for mult in [0.6, 0.75, 0.85, 1.2, 1.4]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    
                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute to non-peak area
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = f[non_peak]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj_trial = 2.0 * n * fast_convolve(trial).max()
                    if obj_trial < best_obj - 1e-14:
                        best_obj = obj_trial
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD it={it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        best_seq = a.copy()

    return best_seq, best_score

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-Frank-Wolfe, Adam with log-sum-exp smoothing, peak-directed
coordinate descent, pairwise exchange, and multi-restart perturbation.
Key improvements over previous: better local minimum escape via annealed
perturbations, frequency-domain perturbations, and adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(f):
    """Objective for sum-normalized f: 2*n*max(conv(f,f))."""
    n = len(f)
    if n == 0:
        return np.inf
    return 2.0 * n * fast_convolve(f).max()


def build_lp_matrix(f, n):
    """Build constraint matrix for LP: rows are convolution equations."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp(f, M, n, A_ub):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def golden_search(f, g, n, max_iter=10):
    """Golden section line search for step size alpha in [0,1]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    f_max_cur = fast_convolve(f).max()

    def eval_a(a):
        ft = f + a * (g - f)
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return f_max_cur
        return fast_convolve(ft / s).max()

    best_a, best_v = 0.0, f_max_cur
    for a in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        v = eval_a(a)
        if v < best_v:
            best_v = v
            best_a = a

    span = 0.2
    lo = max(0.0, best_a - span)
    hi = min(1.0, best_a + span)
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    c_val = eval_a(c_pt)
    d_val = eval_a(d_pt)

    for _ in range(max_iter):
        if hi - lo < 1e-5:
            break
        if c_val < d_val:
            hi, d_pt = d_pt, c_pt
            d_val = c_val
            c_pt = hi - gr * (hi - lo)
            c_val = eval_a(c_pt)
        else:
            lo, c_pt = c_pt, d_pt
            c_val = d_val
            d_pt = lo + gr * (hi - lo)
            d_val = eval_a(d_pt)
        mid = (lo + hi) / 2.0
        mid_v = eval_a(mid)
        if mid_v < best_v:
            best_v = mid_v
            best_a = mid

    return best_a


def lp_fw_optimize(f0, time_limit, verbose=False):

    f = f0.astype(np.float64).copy()

    n = len(f)

    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    no_improve = 0
    m_relax = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 >= time_limit:
            break

        g = solve_lp(f, M * m_relax, n, A_ub)

                m_relax = 1.0
                no_improve = 0
            continue

        gs = g.sum()
        if gs < 1e-15:
            m_relax = min(m_relax * 1.1, 8.0)
            continue
        g = g / gs

            no_improve += 2

            no_improve += 1
        else:
            no_improve = 0

def adam_smoothed(f0, time_limit, lr_init=0.05, temp_init=0.02, verbose=False):
    """Adam optimizer with log-sum-exp smoothing of the max objective."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 >= time_limit:
            break

        progress = min(1.0, t / 3000.0)
        temp = temp_init * (0.03) ** progress
        temp = max(temp, 0.0003)

        conv = fast_convolve(f)
        m_val = conv.max()
        weights = np.exp((conv - m_val) / temp)
        weights /= weights.sum()

        idx = np.arange(n)
        H = weights[idx[:, None] + idx[None, :]]
        grad = 2.0 * (H @ f)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.maximum(f_new, 0)
        sn = f_new.sum()
        if sn < 1e-15:
            continue
        f_new = f_new / sn
        f = f_new

        cur_obj = 2.0 * n * fast_convolve(f).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997

        if no_improve > 60:
            rng = np.random.default_rng(t * 23 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t}: obj={best_obj:.8f}, temp={temp:.5f}, t={elapsed:.1f}s")

        if lr < 1e-10:
            break

    return best_f, best_obj


def peak_cd(f0, time_limit, verbose=False):

    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            order = np.argsort(products)[::-1]
            for oi in range(min(5, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.6, 0.75, 0.85, 1.2, 1.4]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = f[non_peak]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj_trial = 2.0 * n * fast_convolve(trial).max()
                    if obj_trial < best_obj - 1e-14:
                        best_obj = obj_trial
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break

        if not improved:
            no_improve += 1
            # Aggressive: reduce all peak contributors
            if no_improve > 10:
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = f.copy()
                trial[js:je + 1] *= 0.93
                d = f.sum() - trial.sum()
                nc = np.ones(n, dtype=bool)
                nc[js:je + 1] = False
                if nc.sum() > 0:
                    nc_w = f[nc]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[nc] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj_t = 2.0 * n * fast_convolve(trial).max()
                if obj_t < best_obj:
                    f = trial
                    best_obj = obj_t
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD it={it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_f, best_obj


def pairwise_exchange(f, n_trials=200, seed=None):
    """Random pairwise mass exchange to escape local minima."""
    n = len(f)
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    if seed is None:
        seed = int(time.monotonic() * 10000) % 1000000
    rng = np.random.default_rng(seed)

    for _ in range(n_trials):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * 0.15
        amount = frac * f[i]
        if amount < 1e-8:
            continue

        trial = f.copy()
        trial[i] -= amount
        trial[j] += amount
        if trial[i] < 0:
            continue
        trial_sum = trial.sum()
        if trial_sum < 1e-15:
            continue
        trial = trial / trial_sum
        obj = 2.0 * n * fast_convolve(trial).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = trial.copy()

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_seq(a, best_seq, best_score, label=""):
    """Evaluate and potentially update the best sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: LP-Frank-Wolfe, Adam with log-sum-exp
    smoothing, peak-directed coordinate descent, pairwise exchange, and
    multi-restart perturbation with frequency-domain and structured mutations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        n_def = 768
        x = np.arange(n_def) - n_def / 2.0
        sigma = n_def * 0.25
        parent = np.exp(-0.5 * (x / sigma) ** 2)
        parent = parent / parent.sum()
        best_seq, best_score = try_seq(parent, best_seq, best_score, "default")

    n_parent = len(parent)

    # === Phase 1: LP-FW from parent ===
    print(f"\n=== Phase 1: LP-FW (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.12, 130)
    if t1 > 20:
        print(f"  Time: {t1:.0f}s")
        a_opt, obj = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "lpfw_p1")

    # === Phase 2: Adam smoothed ===
    if best_seq is not None:
        print(f"\n=== Phase 2: Adam Smoothed (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t2 = min(rem * 0.12, 130)
        if t2 > 20:
            print(f"  Time: {t2:.0f}s")
            a_opt, obj = adam_smoothed(best_seq, time_limit=t2, verbose=True)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, "adam_p2")

    # === Phase 3: Peak CD ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Peak CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t3 = min(rem * 0.08, 80)
        if t3 > 15:
            print(f"  Time: {t3:.0f}s")
            a_opt, obj = peak_cd(best_seq, time_limit=t3, verbose=True)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, "peakcd_p3")

    # === Phase 4: Pairwise exchange ===
    if best_seq is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 4: Pairwise Exchange ===")
        a_opt, obj = pairwise_exchange(best_seq, n_trials=500, seed=seed)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "pairwise_p4")

    # === Phase 5: Multi-restart perturbation ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Multi-Restart Perturbation ===")
        n_best = len(best_seq)
        n_restarts = 10
        rem = deadline - time.monotonic()
        t_total_p5 = min(rem * 0.28, 500)
        t_per = t_total_p5 / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            rem_r = deadline - time.monotonic()
            if rem_r < 15:
                break
            t_alloc = min(t_per, rem_r * 0.25)

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif r == 1:
                f_init = 0.92 * best_seq + 0.08 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif r == 2:
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif r == 3:
                conv = fast_convolve(best_seq)
                k_peak = int(np.argmax(conv))
                js, je = max(0, k_peak - n_best + 1), min(n_best - 1, k_peak)
                f_init = best_seq.copy()
                f_init[js:je + 1] *= 0.82
                d = 1.0 - f_init.sum()
                nc = np.ones(n_best, dtype=bool)
                nc[js:je + 1] = False
                if nc.sum() > 0:
                    f_init[nc] += d * f_init[nc] / max(f_init[nc].sum(), 1e-15)
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            elif r == 4:
                noise = rng.standard_normal(n_best) * 0.008
                f_init = project_simplex(best_seq + noise)
            elif r == 5:
                shift = int(rng.integers(1, max(2, n_best // 12)))
                f_init = np.roll(best_seq, shift)
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            elif r == 6:
                F = np.fft.rfft(best_seq)
                F[1:-1] *= (1.0 + rng.standard_normal(len(F[1:-1])) * 0.02)
                f_init = np.fft.irfft(F, n_best)
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            elif r == 7:
                s1 = int(rng.integers(0, n_best // 3))
                s2 = int(rng.integers(s1 + n_best // 6, n_best - 1))
                f_init = best_seq.copy()
                section = f_init[s1:s2].copy()
                f_init[s1:s2] = section[::-1]
                f_init = f_init / f_init.sum()
            elif r == 8:
                # Blend of two random perturbations
                n1 = rng.standard_normal(n_best) * 0.005
                n2 = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + n1)
                f_init2 = project_simplex(best_seq + n2)
                f_init = 0.5 * f_init + 0.5 * f_init2
                f_init = f_init / f_init.sum()
            else:
                # Peak anti-correlation: reduce where peaks are, increase elsewhere
                conv = fast_convolve(best_seq)
                k_top5 = np.argpartition(conv, -5)[-5:]
                f_init = best_seq.copy()
                for kk in k_top5:
                    kk = int(kk)
                    js, je = max(0, kk - n_best + 1), min(n_best - 1, kk)
                    f_init[js:je + 1] *= 0.9
                d = 1.0 - f_init.sum()
                f_init = f_init + d / n_best
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()

            label = f"restart_{r}"
            t_use = min(t_alloc, 45)
            if t_use > 10:
                a_opt, obj = adam_smoothed(f_init, time_limit=t_use, verbose=False)
                best_seq, best_score = try_seq(a_opt, best_seq, best_score, label)

                # Follow up with short LP-FW if promising
                rem_r2 = deadline - time.monotonic()
                if rem_r2 > 20:
                    t_lpfw = min(rem_r2 * 0.2, 25)
                    if t_lpfw > 8:
                        a_opt2, obj2 = lp_fw_optimize(a_opt, time_limit=t_lpfw, verbose=False)
                        best_seq, best_score = try_seq(a_opt2, best_seq, best_score, label + "_lp")

    # === Phase 6: Multi-length exploration ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Multi-Length ===")
        n_best = len(best_seq)
        for target_n in [n_best - 48, n_best + 48, n_best + 96]:
            if target_n < 100:
                continue
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            best_seq, best_score = try_seq(f_interp, best_seq, best_score, f"interp_{target_n}")

            t_alloc = min(rem * 0.15, 50)
            if t_alloc > 15:
                a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
                best_seq, best_score = try_seq(a_opt, best_seq, best_score, f"lpfw_{target_n}")

                # Also try Adam on the interpolated
                rem2 = deadline - time.monotonic()
                if rem2 > 15:
                    t_ad = min(rem2 * 0.1, 30)
                    if t_ad > 10:
                        a_opt2, obj2 = adam_smoothed(a_opt, time_limit=t_ad, verbose=False)
                        best_seq, best_score = try_seq(a_opt2, best_seq, best_score, f"adam_{target_n}")

    # === Phase 7: Deep LP-FW on best ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Deep LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_alloc = min(rem, 180)
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, "deep_p7")

    # === Phase 8: Final Adam polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            print(f"\n=== Phase 8: Final Adam ===")
            t_alloc = min(rem, 70)
            a_opt, obj = adam_smoothed(best_seq, time_limit=t_alloc, lr_init=0.015, verbose=True)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, "final_adam")

    # === Phase 9: Final peak CD + pairwise ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            print(f"\n=== Phase 9: Final Polish ===")
            t_pc = min(rem * 0.6, 35)
            if t_pc > 8:
                a_opt, obj = peak_cd(best_seq, time_limit=t_pc, verbose=False)
                best_seq, best_score = try_seq(a_opt, best_seq, best_score, "fin_peak")

            rem2 = deadline - time.monotonic() - 5
            if rem2 > 3:
                a_opt, obj = pairwise_exchange(best_seq, n_trials=300, seed=seed + 999)
                best_seq, best_score = try_seq(a_opt, best_seq, best_score, "fin_pair")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
