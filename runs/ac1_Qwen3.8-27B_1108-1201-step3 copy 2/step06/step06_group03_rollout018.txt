"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-directed optimization.
Key innovation: solve a linear program at each step to find the direction
that most reduces the maximum convolution peak, rather than using gradient
approximations. Combined with periodic coordinate descent and multi-length
exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_conv_max(a):
    """Compute max of auto-convolution."""
    conv = fast_convolve(a)
    return conv


def get_score(a):
    """Compute the objective score directly."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_lp_direction(a, peaks, D_max):
    """
    Solve LP to find direction d minimizing max first-order change at given peaks.
    LP: min t
        s.t. (a*d)[k] <= t for k in peaks
             sum(d) = 0
             |d[j]| <= D_max
    Returns (d, t) or (None, None) if infeasible.
    """
    n = len(a)
    num_peaks = len(peaks)
    
    # Variables: [t, d_0, ..., d_{n-1}], total n+1
    c = np.zeros(n + 1)
    c[0] = 1.0  # minimize t
    
    # Inequality constraints: (a*d)[k] - t <= 0 for each peak k
    A_ub = np.zeros((num_peaks, n + 1))
    b_ub = np.zeros(num_peaks)
    
    for idx, k in enumerate(peaks):
        A_ub[idx, 0] = -1.0  # -t
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[idx, j + 1] = a[k - j]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(None, None)] + [(-D_max, D_max)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        d = res.x[1:]
        t = res.x[0]
        return d, float(t)
    return None, None

        # Solve LP
        d, t = solve_lp_direction(a, peaks, D_max)
        
        if d is not None and t < -1e-10:
            # Take step
            eps, new_max = line_search_1d(a, d, current_max)
            if eps > 1e-10 and new_max < current_max - 1e-12:
                a = np.maximum(a + eps * d, 0)
                a = a / a.sum()
                current_max = fast_convolve(a).max()
                if current_max < best_max - 1e-14:
                    best_max = current_max
                    best_a = a.copy()
                    no_improve = 0
                    if verbose and it % 50 == 0:
                        score = 2.0 * n * best_max
                        print(f"    LP opt iter {it}: max_conv={best_max:.10f}, score~{score:.6f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 50:
            rng = np.random.default_rng(it * 137 + 42)
            noise = rng.standard_normal(n) * (0.001 + 0.002 * no_improve / 50)
            a = np.maximum(best_a + noise, 1e-15)
            a = a / a.sum()
            current_max = fast_convolve(a).max()
            no_improve = 0
            # Also try different D_max
            D_max = 1.0 / n * (1.0 + 0.5 * (it % 3))
        
        if it % 500 == 0:
            print(f"    LP opt: iter={it}, elapsed={elapsed:.1f}s, best={best_max:.10f}")
    
    return best_a, best_max * 2.0 * n  # Return score approximation


def coordinate_descent_refine(a_init, time_limit, verbose=False):
    """Fine coordinate descent optimization."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a_init)) / len(a_init)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-10:
                continue
            
            old_val = a[j]
            best_j_val = old_val
            best_j_max = best_max
            
            # Try multiplicative perturbations
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                # Adjust others to maintain sum
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                conv_trial = fast_convolve(a_trial)
                val = conv_trial.max()
                if val < best_j_max - 1e-13:
                    best_j_max = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                if best_j_max < best_max - 1e-14:
                    best_max = best_j_max
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Try different random order
            break
        if verbose and pass_count % 5 == 0:
            score = 2.0 * n * best_max
            print(f"    CD pass {pass_count}: score~{score:.6f}")
    
    return best_a, best_max


def smoothed_descent(a_init, time_limit, verbose=False, lr=0.005, decay=0.9995):
    """Smoothed mirror descent as backup optimizer."""
    a = np.maximum(a_init.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Compute subgradient at top peaks
        top_k = max(5, len(conv) // 20)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Softmax weights
        pv = conv[peaks]
        temp = 0.005
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[k - idx] + a[idx]
            sg[js:je + 1] += w[i] * contrib
        
        # Multiplicative step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = fast_convolve(a_new).max()
        if obj_new < M:
            a = a_new
            if obj_new < best_max - 1e-14:
                best_max = obj_new
                best_a = a.copy()
                no_improve = 0
        else:
            no_improve += 1
        
        lr *= decay
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-15)
            a = a / a.sum()
            lr = 0.005
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_a, best_max


def make_asymmetric_init(n, variant=0, seed=42):
    """Generate asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 100)
    
    if variant == 0:
        a = np.exp(-np.arange(n) / (n * 0.15))
    elif variant == 1:
        a = np.exp(-np.arange(n) / (n * 0.25))
    elif variant == 2:
        a = np.exp(-np.arange(n) / (n * 0.1))
    elif variant == 3:
        x = np.arange(n) / n
        a = np.exp(-30 * (x - 0.2) ** 2) + 0.3 * np.exp(-30 * (x - 0.6) ** 2)
    elif variant == 4:
        a = np.linspace(1.0, 0.01, n)
    elif variant == 5:
        a = (np.arange(n) + 1) ** (-0.8)
    elif variant == 6:
        a = np.exp(-np.arange(n) / (n * 0.2)) * (1 + 0.3 * np.sin(4 * np.pi * np.arange(n) / n))
        a = np.maximum(a, 0)
    elif variant == 7:
        a = np.exp(-0.5 * (np.arange(n) / (n * 0.1) - 1) ** 2)
    elif variant == 8:
        a = np.sin(np.pi * np.arange(n) / n) ** 2
        a = a * np.exp(-np.arange(n) / (n * 0.5))
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
