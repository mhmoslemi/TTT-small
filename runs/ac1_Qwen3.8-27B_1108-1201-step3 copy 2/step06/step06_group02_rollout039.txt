"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated mirror descent,
improved LP-Frank-Wolfe, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

    Nesterov-accelerated mirror descent for minimizing max(conv(a,a))
    subject to a on the probability simplex.
    
    Uses multiplicative updates (exponentiated gradient) with Nesterov momentum.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated multiplicative (mirror) descent with adaptive step size
2. Improved LP-Frank-Wolfe with warm-starting
3. Multi-length interpolation and re-optimization
4. Peak-targeted coordinate descent for final polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def mirror_descent(f0, time_limit, verbose=False, top_k=3, lr_init=0.005, 
                   momentum_beta=0.9, min_lr=1e-10, restart_interval=300):
    """
    Accelerated mirror descent (exponentiated gradient) with momentum.
    Minimizes max(conv(a,a)) subject to a on probability simplex.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    # Momentum buffer
    v = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # Weight: emphasize the highest peaks more
        pv = conv[peaks]
        # Use exponential weighting (sharper focus on top peaks)
        temp = max((pv.max() - pv.min()) * 0.1, 1e-10)
        weights = np.exp((pv - pv.max()) / temp)
        weights /= weights.sum()
        
        # Compute subgradient (weighted sum of reflected sequences)
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * f[k - idx]
        
        # Momentum update
        v = momentum_beta * v + (1.0 - momentum_beta) * sg
        
        # Multiplicative update
        f = f * np.exp(-lr * v)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        # Evaluate
        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate: decrease when no improvement
        if no_improve > 100:
            lr *= 0.85
            no_improve = 0
            if lr < min_lr:
                break
        
        # Periodic restart to escape local minima
        if it % restart_interval == 0 and no_improve > 50:
            rng = np.random.default_rng(it * 13 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-12)
            f = f / f.sum()
            v = np.zeros(n)
            lr = max(lr, lr_init * 0.3)
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"      MD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, no_imp={no_improve}")
    
    return best_f, best_obj

def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        if f[j] > 1e-15:
            A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def peak_descent(f0, time_limit, verbose=False):
    """Peak-targeted coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 100:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        
        if len(contrib_idx) == 0:
            break
        
        products = f[contrib_idx] * f[peak_k - contrib_idx]
        
        # Find the pair with maximum product
        max_i = int(np.argmax(products))
        src = int(contrib_idx[max_i])
        dst = int(peak_k - src)
        
        # Try reducing src and redistributing
        found = False
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = f[src] * frac
            if diff < 1e-15:
                continue
            
            trial = f.copy()
            trial[src] -= diff
            
            # Redistribute to non-contributing positions
            mask = np.ones(n, dtype=bool)
            mask[contrib_idx] = False
            mask[src] = False  # Don't add back to source
            
            if mask.sum() < 1:
                continue
            
            nc_vals = trial[mask]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                # Distribute uniformly
                trial[mask] += diff / mask.sum()
            else:
                trial[mask] += diff * nc_vals / nc_sum
            
            trial = np.maximum(trial, 0.0)
            trial = trial / trial.sum()
            
            obj = fast_convolve(trial).max()
            if obj < best_obj - 1e-15:
                f = trial
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                found = True
                break
        
        if not found:
            no_improve += 1
            # Try a random perturbation
            if no_improve % 20 == 0:
                rng = np.random.default_rng(hash(f.tobytes()) % (2**31))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"      PD iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated mirror descent (exponentiated gradient) with sharp peak targeting and annealing
2. LP-Frank-Wolfe with adaptive M escalation and warm-starting
3. Peak-targeted coordinate descent with swap operations
4. Multi-length interpolation with re-optimization
5. Simulated annealing style restarts to escape local minima

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 0:
        return np.array([0.0])
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def compute_subgradient(f, conv, top_k):
    """Compute weighted subgradient at top peaks."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    # Sharp weighting: emphasize the maximum peak
    pv_max = pv.max()
    pv_min = pv.min()
    if pv_max - pv_min < 1e-15:
        weights = np.ones(k_actual) / k_actual
    else:
        # Exponential weighting with sharp temperature
        temp = (pv_max - pv_min) * 0.05
        weights = np.exp((pv - pv_max) / temp)
        weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += weights[i] * f[k - idx]
    
    return sg


def mirror_descent(f0, time_limit, verbose=False, top_k=1, lr_init=0.01,
                   momentum=0.85, min_lr=1e-12, anneal_start=500,
                   anneal_rate=0.999, accept_prob_init=0.3):
    """
    Accelerated mirror descent with momentum and simulated annealing.
    Minimizes max(conv(a,a)) subject to a on probability simplex.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    # Momentum buffer
    v = np.zeros(n)
    
    # Annealing
    temperature = 0.01
    accept_prob = accept_prob_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute subgradient
        sg = compute_subgradient(f, conv, top_k)
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * sg
        
        # Multiplicative update
        f_new = f * np.exp(-lr * v)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        # Evaluate new point
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        # Annealing: accept worse moves with some probability
        if obj_new < best_obj:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Accept with probability based on temperature
            delta = obj_new - best_obj
            if delta < temperature * best_obj:
                rng_step = np.random.default_rng(it)
                if rng_step.random() < accept_prob * np.exp(-delta / (temperature * max(best_obj, 1e-10))):
                    f = f_new
                    no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 80:
            lr *= 0.8
            no_improve = 0
            if lr < min_lr:
                break
        
        # Annealing schedule
        if it > anneal_start:
            temperature *= anneal_rate
            accept_prob *= 0.9995
        
        # Periodic restart to escape local minima
        if no_improve > 250:
            rng = np.random.default_rng(it * 17 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-12)
            f = f / f.sum()
            v = np.zeros(n)
            lr = max(lr, lr_init * 0.4)
            no_improve = 0
            temperature = 0.01
            accept_prob = accept_prob_init
        
        if verbose and it % 200 == 0:
            print(f"      MD iter {it}: best={best_obj:.10f}, cur={conv.max():.10f}, lr={lr:.6f}, T={temperature:.4f}")
    
    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with adaptive M escalation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_mult = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break
        
        # Solve LP direction
        M_target = conv.max() * m_mult
        g = solve_lp_direction_fast(f, M_target, n)
        
        if g is None or g.sum() < 1e-12:
            m_mult *= 1.08
            if m_mult > 4.0:
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                m_mult = 1.0
                no_improve += 1
                if no_improve > 60:
                    break
            continue
        
        g = g / g.sum()
        
        # Check if direction is meaningful
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            m_mult = max(m_mult * 1.02, 1.0)
            if no_improve > 50:
                rng = np.random.default_rng(it * 23 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                m_mult = 1.0
                no_improve = 0
            continue
        
        no_improve = 0
        m_mult = 1.0
        
        # Line search: coarse + fine
        best_alpha = 0.0
        best_val = conv.max()
        
        alphas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        vals = []
        for alpha in alphas:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                vals.append(best_val)
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            vals.append(val)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine bilevel search around best
        if 0 < best_alpha < 1.0:
            idx = alphas.index(best_alpha)
            lo = alphas[max(0, idx - 1)]
            hi = alphas[min(len(alphas) - 1, idx + 1)]
            # Bisection
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                ft = ft / ft.sum()
                vm = fast_convolve(ft).max()
                idx_mid = alphas.index(best_alpha)
                val_lo = vals[max(0, idx - 1)]
                val_hi = vals[min(len(alphas) - 1, idx + 1)]
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid
                if vm <= val_lo:
                    hi = mid
                else:
                    lo = mid
        
        if best_alpha < 1e-10:
            no_improve += 1
            continue
        
        f = np.maximum(f + best_alpha * (g - f), 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
        
        if verbose and it % 10 == 0:
            print(f"      LP-FW iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.4f}")
    
    return best_f, best_obj


def solve_lp_direction_fast(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    
    # Build sparse constraint matrix
    from scipy.sparse import lil_matrix
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        fj = f[j]
        if fj > 1e-14:
            for i in range(2 * n - 1):
                row = i
                if row >= 0 and row < 2 * n - 1:
                    A[i, j] = fj if (j <= i < j + n) else 0.0
    # Actually let's do it more carefully
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        fj = f[j]
        if fj > 1e-14:
            for i in range(j, min(2 * n - 1, j + n)):
                A[i, j] = fj
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def peak_descent(f0, time_limit, verbose=False):
    """Peak-targeted coordinate descent with swap."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 150:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        
        if len(contrib_idx) == 0:
            break
        
        products = f[contrib_idx] * f[peak_k - contrib_idx]
        max_i = int(np.argmax(products))
        src = int(contrib_idx[max_i])
        dst = int(peak_k - src)
        
        found = False
        for frac in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4]:
            diff = f[src] * frac
            if diff < 1e-15:
                continue
            
            trial = f.copy()
            trial[src] -= diff
            
            # Find non-contributing positions
            mask = np.ones(n, dtype=bool)
            mask[contrib_idx] = False
            mask[src] = False
            
            if mask.sum() < 1:
                continue
            
            # Distribute proportionally to existing values (or uniformly if all zero)
            nc_vals = trial[mask]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                trial[mask] += diff / mask.sum()
            else:
                trial[mask] += diff * nc_vals / nc_sum
            
            trial = np.maximum(trial, 0.0)
            trial = trial / trial.sum()
            
            obj = fast_convolve(trial).max()
            if obj < best_obj - 1e-15:
                f = trial
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                found = True
                break
        
        if not found:
            no_improve += 1
            # Try swap: move mass from src to a random non-contributing position
            if no_improve % 10 == 0:
                rng = np.random.default_rng(it * 7 + 1)
                non_contrib = np.where(mask if 'mask' in dir() else np.ones(n, dtype=bool))[0]
                if len(non_contrib) > 0:
                    target = non_contrib[rng.integers(len(non_contrib))]
                    for frac in [0.05, 0.1, 0.2]:
                        diff = f[src] * frac
                        trial = f.copy()
                        trial[src] -= diff
                        trial[target] += diff
                        trial = np.maximum(trial, 0.0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-15:
                            f = trial
                            best_obj = obj
                            best_f = f.copy()
                            no_improve = 0
                            break
        
        if verbose and it % 100 == 0:
            print(f"      PD iter {it}: obj={best_obj:.10f}, no_imp={no_improve}")
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

            t_alloc = min(rem * 0.25, 400)
            print(f"\n=== Phase 2: LP-Frank-Wolfe (n={len(best_seq)}, t={t_alloc:.0f}s) ===")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")
    
    # Phase 3: Mirror descent with wider peaks and stronger annealing
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            t_alloc = min(rem * 0.2, 300)
            print(f"\n=== Phase 3: MD Wide Peaks + Annealing (t={t_alloc:.0f}s) ===")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc, verbose=True,
                                         top_k=5, lr_init=0.004, momentum=0.92,
                                         anneal_start=100, anneal_rate=0.998,
                                         accept_prob_init=0.5)
            try_seq(a_opt, "md_wide")
    
    # Phase 4: Multi-length exploration
    if best_seq is not None:
        rem = deadline - time.monotonic()
        print(f"\n=== Phase 4: Multi-Length ===")
        for target_n in [1024, 896, 640, 512, 1152]:
            if target_n == len(best_seq) or rem < 30:
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            
            if sc < best_score + 0.005:
                t_alloc = min(rem * 0.12, 90)
                if t_alloc > 10:
                    print(f"  Optimizing n={target_n} (t={t_alloc:.0f}s)")
                    a_opt, obj = mirror_descent(f_interp, time_limit=t_alloc,
                                                verbose=False, top_k=1,
                                                lr_init=0.008, momentum=0.85)
                    try_seq(a_opt, f"md_n{target_n}")
                rem = deadline - time.monotonic()
    
    # Phase 5: Peak-targeted descent
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.2, 200)
            print(f"\n=== Phase 5: Peak Descent (t={t_alloc:.0f}s) ===")
            a_opt, obj = peak_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "peak_desc")
    
    # Phase 6: Final polishing (very small steps, top_k=1)
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.4, 120)
            print(f"\n=== Phase 6: Final Polish (t={t_alloc:.0f}s) ===")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc, verbose=True,
                                         top_k=1, lr_init=0.001, momentum=0.95,
                                         min_lr=1e-14)
            try_seq(a_opt, "final_polish")
    
    # Phase 7: One more LP-FW squeeze if time permits
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.5, 60)
            print(f"\n=== Phase 7: Final LP-FW (t={t_alloc:.0f}s) ===")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_lpfw")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    elapsed = time.monotonic() - total_start
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {elapsed:.1f}s")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
