import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def compute_subgradient(a, conv, top_k=10, temperature=0.001):
    """
    Compute a weighted subgradient of max(conv) w.r.t. a.
    Uses softmax weighting over the top peaks.
    Returns gradient of shape (n,).
    """
    n = len(a)
    conv_len = len(conv)
    
    # Select top-k peaks
    if top_k >= conv_len:
        peaks = np.arange(conv_len)
        weights = np.ones(conv_len) / conv_len
    else:
        # Use softmax over all peaks, but emphasize the top ones
        # Scale by temperature
        scaled = conv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        # Zero out very small weights for efficiency
        w[w < 1e-10] = 0
        w /= w.sum()
        peaks = np.where(w > 1e-10)[0]
        weights = w[peaks]
    
    # Compute subgradient
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        k = int(peaks[idx])
        wt = weights[idx]
        if wt < 1e-12:
            continue
        # conv[k] = sum_i a[i] * a[k-i]
        # d(conv[k])/d(a[j]) = a[k-j] (for 0 <= k-j <= n-1)
        # So d(max_conv)/d(a[j]) = sum_k w[k] * a[k-j]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            sg[j] += wt * a[k - j]
    
    return sg


def softmax_mirror_descent(a0, time_limit, lr_init=0.01, temp_init=0.001, 
                           temp_growth=1.01, lr_decay=0.999, 
                           patience=100, verbose=False):
    """
    Mirror descent with softmax weighting over peaks.
    Uses multiplicative updates (exponential family).
    """
    a = np.maximum(a0.copy(), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break
        
        # Compute subgradient
        sg = compute_subgradient(a, conv, top_k=20, temperature=temp)
        
        # Multiplicative update
        a = a * np.exp(-lr * 2.0 * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Evaluate
        conv = fast_convolve(a)
        curr_obj = conv.max()
        
        if curr_obj < best_obj - 1e-12:
            best_obj = curr_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 20 == 0:

            # Products

                    if other_sum < 1e-15:
                        continue
                    trial[other_mask] += reduction * trial[other_mask] / other_sum
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts
                    
                    obj = fast_convolve(trial).max()
                    if obj < best_new_obj - 1e-12:

                        best_new_a = trial

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_random_smooth(n, seed=0, smoothness=3):
    """Generate a random smooth non-negative sequence."""
    rng = np.random.default_rng(seed)
    a = rng.standard_normal(n)
    # Apply smoothing (low-pass filter)
    for _ in range(smoothness):
        a = 0.5 * a + 0.25 * np.roll(a, 1) + 0.25 * np.roll(a, -1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def compute_weighted_subgradient(a, conv, top_k=20, temperature=0.001):
    """
    Compute weighted subgradient of max(conv) using top-K softmax weighting.
    Returns gradient array of shape (n,).
    """
    n = len(a)
    conv_len = len(conv)

    # Softmax over all lags with temperature
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    total = w.sum()
    if total < 1e-30:
        w = np.zeros(conv_len)
        w[np.argmax(conv)] = 1.0
    else:
        w /= total

    # Focus on top-K peaks for efficiency
    if top_k < conv_len:
        thresh_idx = np.argpartition(w, -top_k)[-top_k]
        min_thresh = w[thresh_idx[0]]
        w[w < min_thresh * 0.5] = 0.0
        w_sum = w.sum()
        if w_sum > 1e-30:
            w /= w_sum
        else:
            w = np.zeros(conv_len)
            w[np.argmax(conv)] = 1.0

    # Compute subgradient: d(conv[k])/d(a[j]) = a[k-j] for valid j
    sg = np.zeros(n)
    active = np.where(w > 1e-12)[0]

        je = min(n - 1, k)
        if js > je:

    return sg

                phase = np.angle(A)
                phase += rng.standard_normal(len(phase)) * 0.1
                A_new = amp * np.exp(1j * phase)
                a = np.fft.irfft(A_new, 2 * n - 1)[:n]
                a = np.maximum(a, 1e-15)
            a = a / a.sum()
            conv = fast_convolve(a)
            lr = lr_init * 0.5
            temp = temp_init * 2
            no_improve = 0

        if lr < 1e-10:
            break

    return best_a, best_obj


def peak_reduction(a0, time_limit, verbose=False):
    """
    Targeted peak reduction: find worst peak, compute sensitivity,
    apply best single or dual coordinate adjustment.
    """
    a = np.maximum(a0.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.3:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Find top peaks (avoid lag 0 which is sum(a^2))
        conv_ordered = conv.copy()
        conv_ordered[0] = -1  # deprioritize lag 0
        top_peaks = np.argsort(conv_ordered)[-5:]
        improved = False

        for pk in top_peaks:
            pk = int(pk)
            if pk == 0:
                continue

            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contrib = np.arange(js, je + 1)
            if len(contrib) == 0:
                continue

            prods = a[contrib] * a[pk - contrib]
            if prods.max() < 1e-15:
                continue

            # Try reducing top contributors
            top_contrib_idx = contrib[np.argsort(prods)[-min(8, len(contrib)):]]

            best_new_obj = M
            best_new_a = None

            for j in top_contrib_idx:
                j = int(j)
                for frac in [0.08, 0.15, 0.25, 0.4, 0.6]:
                    reduction = a[j] * frac
                    if reduction < 1e-15:
                        continue

                    trial = a.copy()
                    trial[j] -= reduction

                    # Redistribute to elements NOT contributing to this peak
                    nc_mask = np.ones(n, dtype=bool)
                    nc_mask[js:je + 1] = False
                    nc_vals = trial[nc_mask]
                    nc_sum = nc_vals.sum()

                    if nc_sum < 1e-12:
                        # Redistribute to low-contributing elements within the peak
                        low_thresh = prods.max() * 0.3
                        low_mask = (prods < low_thresh)
                        if low_mask.sum() < 1:
                            continue
                        low_idx = contrib[low_mask]
                        nc_vals = trial[low_idx]
                        nc_sum = nc_vals.sum()
                        if nc_sum < 1e-12:
                            continue
                        trial[low_idx] += reduction * nc_vals / nc_sum
                    else:
                        trial[nc_mask] += reduction * nc_vals / nc_sum

                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-12:
                        continue
                    trial = trial / ts

                    obj = fast_convolve(trial).max()
                    if obj < best_new_obj - 1e-13:
                        best_new_obj = obj
                        best_new_a = trial

            if best_new_a is not None and best_new_obj < best_obj - 1e-13:
                a = best_new_a
                best_obj = best_new_obj
                best_a = a.copy()
                no_improve = 0
                improved = True
                if verbose:
                    print(f"      PRD: peak={pk}, obj={best_obj:.10f}")
                break

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0 and no_improve > 30:
                rng = np.random.default_rng(hash((int(time.monotonic() * 1000), n)) % (2**31))
                a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-15)
                a = a / a.sum()

    return best_a, best_obj


def alternating_optimize(a0, time_limit, verbose=False):
    """
    Alternate between mirror descent (smooth) and peak reduction (targeted).
    Coarse-to-fine: start with high LR/temp, gradually decrease.
    """
    a = np.maximum(a0.copy().astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    # Phase A: Coarse (exploration)
    coarse_time = min(time_limit * 0.4, time_limit - 5)
    a_coarse, obj_coarse = mirror_descent(
        a, time_limit=coarse_time,
        lr_init=0.015, temp_init=0.002,
        temp_growth=1.012, lr_decay=0.999,
        patience=60, top_k=30, verbose=verbose
    )
    if obj_coarse < best_obj:
        best_obj = obj_coarse
        best_a = a_coarse.copy()
    a = a_coarse.copy()

    # Phase B: Peak reduction
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        t2 = min(rem * 0.3, 60)
        a_pr, obj_pr = peak_reduction(a, time_limit=t2, verbose=verbose)
        if obj_pr < best_obj:
            best_obj = obj_pr
            best_a = a_pr.copy()
        a = a_pr.copy()

    # Phase C: Fine mirror descent (exploitation)
    rem = time_limit - (time.monotonic() - t0)
    if rem > 3:
        t3 = min(rem * 0.5, 80)
        a_fine, obj_fine = mirror_descent(
            a, time_limit=t3,
            lr_init=0.005, temp_init=0.0005,
            temp_growth=1.006, lr_decay=0.9995,
            patience=100, top_k=15, verbose=verbose
        )
        if obj_fine < best_obj:
            best_obj = obj_fine
            best_a = a_fine.copy()

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=5, max_amp=0.25, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.02, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_random_smooth(n, seed=0, smoothness=3):
    rng = np.random.default_rng(seed)
    a = rng.standard_normal(n)
    for _ in range(smoothness):
        a = 0.5 * a + 0.25 * np.roll(a, 1) + 0.25 * np.roll(a, -1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def make_flat_spectrum(n, seed=0, n_bands=8):
    """Generate sequence with approximately flat power spectrum in middle bands."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.zeros(fft_size // 2 + 1, dtype=complex)
    # Set random phase, flat amplitude in middle frequency band
    freqs = np.arange(fft_size // 2 + 1)
    mid_start = fft_size // (2 * n_bands)
    mid_end = fft_size // 2 - fft_size // (2 * n_bands)
    A[mid_start:mid_end + 1] = np.exp(1j * rng.uniform(0, 2 * np.pi, mid_end - mid_start + 1))
    a = np.fft.irfft(A, fft_size)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Diverse initialization at multiple lengths
    2. Alternating optimization (mirror descent + peak reduction)
    3. Multi-length exploration with interpolation
    4. Final intensive polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        try_seq(parent, "default_parent")

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()

    init_candidates = []
    n_values = [512, 640, 768, 896, 1024]

    for n_try in n_values:
        init_candidates.append((f'g2_{n_try}', make_gaussian(n_try, 0.2)))
        init_candidates.append((f'g3_{n_try}', make_gaussian(n_try, 0.3)))
        init_candidates.append((f'bell_{n_try}', make_bell(n_try, 0.55)))
        init_candidates.append((f'cos1_{n_try}', make_cosine_mod(n_try, 1, 0.12)))
        init_candidates.append((f'cos3_{n_try}', make_cosine_mod(n_try, 3, 0.08)))
        init_candidates.append((f'mc4_{n_try}', make_multi_cosine(n_try, 4, 0.2, seed=n_try)))
        init_candidates.append((f'mc6_{n_try}', make_multi_cosine(n_try, 6, 0.12, seed=n_try + 200)))
        init_candidates.append((f'sr_{n_try}', make_random_smooth(n_try, seed=n_try, smoothness=4)))
        init_candidates.append((f'fs_{n_try}', make_flat_spectrum(n_try, seed=n_try)))

    if parent is not None:
        init_candidates.append(('parent', parent.copy()))

    scored_inits = []
    for label, a_init in init_candidates:
        sc = objective_normalized(a_init)
        scored_inits.append((sc, a_init, label))
        if sc < 1.8:
            try_seq(a_init, f"init_{label}")

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Top 5 inits:")
    for i in range(min(5, len(scored_inits))):
        print(f"    {scored_inits[i][2]}: obj={scored_inits[i][0]:.6f}")
    print(f"  Phase 1 time: {time.monotonic() - t0:.1f}s")

    # === Phase 2: Alternating optimization on top candidates ===
    print("\n=== Phase 2: Alternating Optimization ===")
    top_k = min(5, len(scored_inits))
    for trial_idx in range(top_k):
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        sc_init, a_init, label = scored_inits[trial_idx]
        t_alloc = min(rem * 0.35, 200)
        print(f"  ALT [{label}]: alloc={t_alloc:.0f}s (init_obj={sc_init:.6f})")
        a_opt, obj = alternating_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"alt_{label}")

    # === Phase 3: Intensive optimization on current best ===
    print("\n=== Phase 3: Intensive Optimization ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.6, 150)
            print(f"  Intensive MD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(
                best_seq, time_limit=t_alloc,
                lr_init=0.008, temp_init=0.0008,
                temp_growth=1.007, lr_decay=0.9993,
                patience=120, top_k=25, verbose=True
            )
            try_seq(a_opt, "intensive_md")

        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.5, 80)
            print(f"  Intensive PRD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = peak_reduction(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "intensive_prd")

    # === Phase 4: Multi-length exploration ===
    print("\n=== Phase 4: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_ns = [512, 640, 768, 896, 1024, 1280]
        if base_n not in target_ns:
            target_ns.append(base_n)

        for target_n in target_ns:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            if sc < best_score + 0.05:
                t_alloc = min(rem * 0.3, 50)
                if t_alloc > 10:
                    a_opt, obj = alternating_optimize(f_interp, time_limit=t_alloc, verbose=False)
                    try_seq(a_opt, f"alt_interp_n{target_n}")

    # === Phase 5: Structured perturbations + re-optimization ===
    print("\n=== Phase 5: Structured Perturbations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        perturbations = []

        # Shifts
        for shift in [1, 2, 3, 5, 8]:
            perturbations.append((f'shift{shift}', np.roll(best_seq, shift)))

        # Reflection
        perturbations.append(('reflect', best_seq[::-1]))

        # Blends with uniform
        for mix in [0.05, 0.1, 0.15, 0.25]:
            perturbations.append((f'uniform{mix}',
                                  (1 - mix) * best_seq + mix * (np.ones(n_best) / n_best)))

        # Frequency-domain phase perturbation
        for phase_amp in [0.05, 0.1, 0.2]:
            A = np.fft.rfft(best_seq)
            phase = np.angle(A)
            rng_p = np.random.default_rng(seed + int(phase_amp * 1000))
            phase += rng_p.standard_normal(len(phase)) * phase_amp
            a_new = np.fft.irfft(np.abs(A) * np.exp(1j * phase), 2 * n_best - 1)[:n_best]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                perturbations.append((f'phase{phase_amp}', a_new / s))

        for label, f_pert in perturbations:
            rem = deadline - time.monotonic() - 40
            if rem < 15:
                break
            f_pert = np.maximum(f_pert, 0)
            ps = f_pert.sum()
            if ps < 1e-10:
                continue
            f_pert = f_pert / ps
            sc_pert = objective_normalized(f_pert)
            if sc_pert < 1.8:
                try_seq(f_pert, f"pert_{label}")
            t_alloc = min(rem * 0.25, 35)
            if t_alloc > 8 and sc_pert < best_score + 0.1:
                a_opt, obj = mirror_descent(
                    f_pert, time_limit=t_alloc,
                    lr_init=0.006, temp_init=0.001,
                    temp_growth=1.008, lr_decay=0.999,
                    patience=50, top_k=20, verbose=False
                )
                try_seq(a_opt, f"md_pert_{label}")

    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_alloc = min(rem * 0.5, 30)
            a_opt, obj = peak_reduction(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_prd")

        rem = deadline - time.monotonic() - 8
        if rem > 8:
            t_alloc = min(rem * 0.7, 15)
            a_opt, obj = mirror_descent(
                best_seq, time_limit=t_alloc,
                lr_init=0.003, temp_init=0.003,
                temp_growth=1.015, lr_decay=0.999,
                patience=40, top_k=10, verbose=False
            )
            try_seq(a_opt, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
