Actually, let me reconsider the time budget. The previous program has 8 phases and takes the full 1800 seconds. I need to be strategic about time allocation.

Key insight: mirror descent is MUCH faster per iteration than the previous subgradient method because:
- Subgradient: O(n * |active|) for the gradient computation
- Mirror descent: O(n + |active| * (n)) for the gradient (but the gradient is just a slice of a, so it's O(n + |active|))

Actually, the gradient computation for mirror descent is:
- For each active peak k, add a shifted/reversed slice of a to g
- This is O(n) per active peak, so O(n * |active|) total

But in the "typical" case where there's 1-3 active peaks, this is O(n).

The convolution is O(n log n) via FFT.

So each iteration is O(n log n + n * |active|) ≈ O(n log n).

For n=1152, this is very fast - probably 10-50 microseconds per FFT, so we can do millions of iterations in 1800 seconds.

Let me now write the full program:

"""
Mirror descent with adaptive peak selection, multi-length exploration,
and coordinate descent polishing for minimizing the autocorrelation ratio.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_local(a):
    """Compute objective locally (assumes a is on simplex)."""
    n = len(a)
    b = fast_convolve(a)
    return float(b.max())

def md_subgradient(a, active_peaks):
    """Compute subgradient of max_k c_k(a) at point a, given active peaks."""
    n = len(a)
    m = len(active_peaks)
    g = np.zeros(n)
    for k in active_peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # g[lo:hi+1] += a[k-hi : k-lo+1][::-1]
        g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    g *= 1.0 / m
    return g

def mirror_descent_optimize(a0, time_limit, eta_init=0.1, peak_frac=0.999,
                            kick_interval=200, kick_size=0.001, seed=42):
    """
    Mirror descent (entropy-regularized) optimization on the simplex.
    Minimizes max_k c_k(a) where c_k = sum_j a[j]*a[k-j].
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    b = fast_convolve(a)
    best_a = a.copy()
    best_obj = b.max()
    eta = eta_init
    no_improve = 0
    it = 0
    consecutive_bad = 0
    
    while time.monotonic() - t0 < time_limit:
        # Process in mini-batches of 10 for efficient time checking
        for _ in range(10):
            if time.monotonic() - t0 >= time_limit:
                break
                
            it += 1
            k_max = b.argmax()
            
            # Find active peaks
            b_max = b[k_max]
            if b_max > 0:
                threshold = b_max * peak_frac
                active = np.where(b >= threshold)[0].tolist()
            else:
                active = [k_max]
            
            # Compute subgradient
            g = md_subgradient(a, active)
            
            # Mirror descent step
            # a_new[j] = a[j] * exp(-eta * g[j]) / Z
            log_weights = np.log(np.maximum(a, 1e-300)) - eta * g
            log_weights -= log_weights.max()
            a_new = np.exp(log_weights)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-300:
                # Numerical failure, skip
                a_new = a.copy()
            else:
                a_new /= a_new_sum
            
            # Evaluate
            b = fast_convolve(a_new)
            obj = b.max()
            
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a_new.copy()
                no_improve = 0
                consecutive_bad = 0
            else:
                no_improve += 1
                if obj > b_max + 1e-12:
                    consecutive_bad += 1
                else:
                    consecutive_bad = 0
            
            a = a_new
            
            # Adaptive step size
            if consecutive_bad > 5:
                eta *= 0.5
                consecutive_bad = 0
            elif no_improve > kick_interval:
                eta *= 0.8
            elif no_improve == 0:
                # Slightly increase eta when making progress
                eta = min(eta * 1.01, 5.0)
        
        # Periodic kick to escape local minima
        if no_improve > kick_interval * 2:
            a = best_a + rng.standard_normal(n) * kick_size
            a = np.maximum(a, 0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a /= a_sum
            else:
                a = best_a.copy()
            b = fast_convolve(a)
            no_improve = 0
            eta = eta_init
    
    return best_a, best_obj

def coordinate_polish(a0, time_limit, n_scales=12):
    """
    Coordinate descent: for each element, try scaling it by various factors.
    """
    t0 = time.monotonic()

    n = len(a)
    best_a = a.copy()

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 >= time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:

                    a_trial = a.copy()
                    a_trial[j] = s / n
                    a_trial /= a_trial.sum()
                    obj = fast_convolve(a_trial).max()
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        a = a_trial
                        best_a = a.copy()
                        improved = True

            best_j_val = old_val
            
            for s in scales:

                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale_others = (1.0 - new_val) / denom
                if scale_others < 0 or scale_others > 10:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale_others
                a_trial /= a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale_others = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale_others
                a /= a.sum()
                best_obj = best_j_obj
                best_a = a.copy()
                improved = True
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
