Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive peak-shaving with
multiple strategies: subgradient mirror descent, LP-Frank-Wolfe, and

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient
descent, reduced-constraint LP-Frank-Wolfe, spectral shaping, and multi-length
exploration. Key innovations: momentum-based subgradient for faster convergence,
LP with only top-K peak constraints for speed, and iterative spectrum flattening.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tapered(n, power=2.0):
    """Tapered edges sequence."""
    x = np.arange(n) / (n - 1.0)
    a = 1.0 - np.abs(2.0 * x - 1.0) ** power
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def nesterov_subgradient(f0, time_limit, top_k=10, verbose=False):
    """Nesterov-accelerated mirror descent targeting top peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    eta = 0.008
    it = 0

    # Nesterov momentum
    y = f.copy()
    momentum = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Nesterov lookahead
        conv_y = fast_convolve(y)
        M_y = conv_y.max()

        # Find top-k peaks
        k_actual = min(top_k, len(conv_y))
        peak_indices = np.argpartition(conv_y, -k_actual)[-k_actual:]
        peak_values = conv_y[peak_indices]

        # Temperature-scaled weights
        temp = max(peak_values.min(), 1e-15) * 0.5
        scaled = (peak_values - peak_values.min()) / temp
        weights = np.exp(scaled)
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.zeros(len(peak_indices))
            weights[np.argmax(peak_values)] = 1.0
        else:
            weights /= ws

        # Compute subgradient at y
        sg = np.zeros(n)
        for i in range(len(peak_indices)):
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * weights[i] * y[k - idx]

        # Mirror descent step from y
        f_new = y * np.exp(-eta * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        # Nesterov update for y
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = (f_new + momentum * (f_new - f)) / (1.0 + momentum)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        f = f_new

        co = fast_convolve(f).max()
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    NSG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}")
        else:
            no_improve += 1
            eta *= 0.997

        if no_improve > 200:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            y = f.copy()
            eta = 0.008
            no_improve = 0

        if eta < 1e-7:
            break

    return best_f, best_obj


def solve_lp_reduced(f, M, n, peak_indices):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in peak_indices, g >= 0."""
    K = len(peak_indices)
    c = -np.ones(n)
    A_ub = np.zeros((K, n))
    for i, k in enumerate(peak_indices):
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        if m_start <= m_end:
            A_ub[i, m_start:m_end + 1] = f[k - np.arange(m_start, m_end + 1)]
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.fun > -1e10:
        return res.x
    return None


def lp_fw_reduced(f0, time_limit, n_constraints=50, verbose=False):
    """LP-Frank-Wolfe with reduced constraint set (only top peaks)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factors = [1.0, 1.01, 1.03, 1.06, 1.12, 1.25, 1.5]

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        # Get top peaks
        k_actual = min(n_constraints, len(conv))
        top_peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        top_peaks_sorted = top_peaks[np.argsort(conv[top_peaks])[::-1]]

        g = None
        for mfac in m_factors:
            M_try = conv.max() * mfac
            g = solve_lp_reduced(f, M_try, n, top_peaks_sorted)
            if g is not None and g.sum() > 1e-12:
                break

        if g is None or g.sum() < 1e-12:
            rng = np.random.default_rng(it * 23 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            no_improve += 1
            if no_improve > 40:
                break
            continue

        g = g / g.sum()

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            no_improve += 1
            continue

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        M = conv.max()
        f = f_new

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 5 == 0:
                print(f"    LP-FW-R iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def spectral_shaping(f0, time_limit, verbose=False):
    """Alternate nonnegativity projection with spectrum flattening."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    alpha = 0.3

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Compute DFT
        A = np.fft.rfft(f, 2 * n - 1)
        power = np.abs(A) ** 2
        target = power.sum() / len(power)

        # Flatten: reduce peaks, boost valleys
        ratio = power / max(target, 1e-30)
        new_power = target * (1.0 + alpha * (ratio - 1.0))
        new_power = np.maximum(new_power, target * 0.05)

        # Preserve phase
        phase = np.angle(A)
        new_A = np.sqrt(new_power) * np.exp(1j * phase)

        # IFT
        f_new = np.fft.irfft(new_A, 2 * n - 1)[:n]
        f_new = np.maximum(f_new, 1e-12)
        sn = f_new.sum()
        if sn < 1e-15:
            no_improve += 1
            alpha *= 0.9
            continue
        f_new = f_new / sn

        co = fast_convolve(f_new).max()
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            alpha *= 0.98

        if no_improve > 200 or alpha < 0.01:
            break

    return best_f, best_obj


def peak_direct_descent(f0, time_limit, verbose=False):
    """Direct peak reduction via targeted coordinate moves."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        if len(contrib_idx) == 0:
            break

        products = f[contrib_idx] * f[peak_k - contrib_idx]

        # Try reducing top contributors
        top_contrib = np.argsort(products)[::-1][:5]
        found_improve = False

        for tc in top_contrib:
            src = contrib_idx[tc]
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                diff = f[src] * frac
                if diff < 1e-15:
                    continue
                trial = f.copy()
                trial[src] -= diff
                # Redistribute to non-contributing positions
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[contrib_idx] = False
                if non_contrib.sum() < 2:
                    non_contrib[:] = True
                    non_contrib[src] = False
                nc_vals = trial[non_contrib]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-15:
                    continue
                trial[non_contrib] += diff * nc_vals / nc_sum
                trial = np.maximum(trial, 0.0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-13:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    found_improve = True
                    if verbose and it % 30 == 0:
                        print(f"    PDD iter {it}: obj={best_obj:.10f}")
                    break
            if found_improve:
                break

        if not found_improve:
            # Try other peaks
            top_peaks = np.argsort(conv)[-6:]
            for pk in top_peaks:
                pk = int(pk)
                if pk == peak_k:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                cidx2 = np.arange(js2, je2 + 1)
                if len(cidx2) == 0:
                    continue
                prods2 = f[cidx2] * f[pk - cidx2]
                src2 = cidx2[np.argmax(prods2)]
                for frac2 in [0.08, 0.15, 0.25]:
                    diff2 = f[src2] * frac2
                    trial = f.copy()
                    trial[src2] -= diff2
                    nc2 = np.ones(n, dtype=bool)
                    nc2[cidx2] = False
                    nc2_vals = trial[nc2]
                    nc2_sum = nc2_vals.sum()
                    if nc2_sum < 1e-15:
                        continue
                    trial[nc2] += diff2 * nc2_vals / nc2_sum
                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()
                    obj2 = fast_convolve(trial).max()
                    if obj2 < best_obj - 1e-13:
                        f = trial
                        best_obj = obj2
                        best_f = f.copy()
                        no_improve = 0
                        found_improve = True
                        break
                if found_improve:
                    break

            if not found_improve:
                no_improve += 1

        if no_improve > 60:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimizer: Nesterov subgradient, reduced LP-FW,
    spectral shaping, direct peak descent, multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_score = p_score
            best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.3)
        try_seq(parent, "default")

    # === Phase 1: Rich initialization ===
    print("\n=== Phase 1: Initialization ===")
    t0 = time.monotonic()

    n_values = [640, 768, 896, 1024, 1152, 1280]
    init_candidates = []

    for n_try in n_values:
        init_candidates.append(('gauss25', n_try, make_gaussian(n_try, 0.25)))
        init_candidates.append(('gauss35', n_try, make_gaussian(n_try, 0.35)))
        init_candidates.append(('bell50', n_try, make_bell(n_try, 0.5)))
        init_candidates.append(('taper2', n_try, make_tapered(n_try, 2.0)))
        init_candidates.append(('taper4', n_try, make_tapered(n_try, 4.0)))
        init_candidates.append(('cos1_10', n_try, make_cosine_mod(n_try, 1, 0.10)))
        init_candidates.append(('cos2_08', n_try, make_cosine_mod(n_try, 2, 0.08)))
        init_candidates.append(('mc3', n_try, make_multi_cosine(n_try, 3, 0.2, seed=n_try)))
        init_candidates.append(('mc5', n_try, make_multi_cosine(n_try, 5, 0.15, seed=n_try + 7)))

    if parent is not None:
        init_candidates.append(('parent', len(parent), parent))
        for n_try in n_values:
            if n_try != len(parent):
                init_candidates.append((f'pint{n_try}', n_try, interpolate_sequence(parent, n_try)))

    # Quick score all
    scored = []
    for label, n_try, a_init in init_candidates:
        sc = objective_normalized(a_init)
        scored.append((sc, a_init, label))
    scored.sort(key=lambda x: x[0])

    # Evaluate top initializations with full evaluate_sequence
    for sc, a_init, label in scored[:8]:
        try_seq(a_init, f"init_{label}")

    print(f"  Best init obj: {scored[0][0]:.8f} ({scored[0][2]})")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Nesterov subgradient on top candidates ===
    print("\n=== Phase 2: Nesterov Subgradient ===")
    for trial_idx in range(min(3, len(scored))):
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        sc, a_init, label = scored[trial_idx]
        t_alloc = min(rem * 0.5, 180)
        print(f"  NSG [{label}] n={len(a_init)}: {t_alloc:.0f}s (init={sc:.6f})")
        a_opt, obj = nesterov_subgradient(a_init, time_limit=t_alloc, top_k=12, verbose=True)
        try_seq(a_opt, f"nsg_{label}")

    # === Phase 3: Reduced LP-FW on best ===
    print("\n=== Phase 3: LP-FW (Reduced) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_alloc = min(rem * 0.5, 150)
            print(f"  LP-FW-R n={len(best_seq)}: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_reduced(best_seq, time_limit=t_alloc, n_constraints=40, verbose=True)
            try_seq(a_opt, "lpfw_r")

    # === Phase 4: Spectral shaping ===
    print("\n=== Phase 4: Spectral Shaping ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 15:
            t_alloc = min(rem * 0.4, 80)
            print(f"  Spectral n={len(best_seq)}: {t_alloc:.0f}s")
            a_opt, obj = spectral_shaping(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "spectral")

    # === Phase 5: Direct peak descent ===
    print("\n=== Phase 5: Direct Peak Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 15:
            t_alloc = min(rem * 0.5, 100)
            print(f"  PDD n={len(best_seq)}: {t_alloc:.0f}s")
            a_opt, obj = peak_direct_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "pdd")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-length ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [1024, 896, 1152, 640, 1280]:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 10:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp{target_n}")
            if sc < best_score + 0.03:
                t_alloc = min(rem * 0.3, 40)
                if t_alloc > 8:
                    a_opt, obj = nesterov_subgradient(f_interp, time_limit=t_alloc, top_k=12)
                    try_seq(a_opt, f"nsg_int{target_n}")

    # === Phase 7: Perturbation + re-optimize ===
    print("\n=== Phase 7: Perturbations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        perts = []
        for sh in [1, 2, 3, 5, 8]:
            perts.append((f'sh{sh}', np.roll(best_seq, sh)))
        for mx in [0.05, 0.1, 0.15]:
            perts.append((f'mx{mx}', (1 - mx) * best_seq + mx * np.ones(n_best) / n_best))
        for am in [0.002, 0.005, 0.008]:
            nz = np.random.default_rng(seed + int(am * 10000)).standard_normal(n_best) * am
            perts.append((f'nz{am}', np.maximum(best_seq + nz, 0)))

        for label, f_p in perts:
            rem = deadline - time.monotonic() - 60
            if rem < 8:
                break
            f_p = project_simplex(f_p)
            t_alloc = min(rem * 0.15, 25)
            if t_alloc < 5:
                break
            a_opt, obj = nesterov_subgradient(f_p, time_limit=t_alloc, top_k=12)
            try_seq(a_opt, f"pert_{label}")

    # === Phase 8: Final combined polish ===
    print("\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem * 0.5, 30)
            a_opt, obj = nesterov_subgradient(best_seq, time_limit=t_alloc, top_k=20, verbose=True)
            try_seq(a_opt, "final_nsg")

        rem = deadline - time.monotonic() - 8
        if rem > 5:
            t_alloc = min(rem, 15)
            a_opt, obj = lp_fw_reduced(best_seq, time_limit=t_alloc, n_constraints=30)
            try_seq(a_opt, "final_lpfw")

        rem = deadline - time.monotonic() - 3
        if rem > 3:
            t_alloc = min(rem, 8)
            a_opt, obj = peak_direct_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_pdd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    total_time = time.monotonic() - (deadline - max(1.0, float(budget_s) - 5.0))
    print(f"Total time: {total_time:.1f}s")
    return [float(x) for x in best_seq]
