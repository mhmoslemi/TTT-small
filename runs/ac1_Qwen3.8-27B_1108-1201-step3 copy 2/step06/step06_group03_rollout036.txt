Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a KKT-based Newton active-set method

The key insight is that at the optimum, the KKT conditions yield a linear system
that can be solved to get a Newton direction, which is then used with a line search.

import numpy as np
import scipy.sparse as sp
import scipy.sparse.linalg as spla
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalized_obj(f):
    """Compute the objective for a normalized (sum=1) sequence."""
    n = len(f)
    c = fast_convolve(f)
    return 2.0 * n * c.max()


def solve_kkt(n, S, w, reg=1e-10):
    """
    Solve the KKT linear system.
    S: list of active constraint indices
    w: dict mapping active index to weight
    Returns (a, lambda) or None if solve fails.
    """
    # Build sparse matrix A (n x n)
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        # Diagonal: 1 + sum of w[k] for k in S where k-j = j (i.e., k = 2j)
        diag = 1.0 + reg
        for k in S:
            m = k - j
            if 0 <= m <= n - 1:
                if m == j:
                    diag += w[k]
                else:
                    rows.append(j)
                    cols.append(m)
                    vals.append(w[k])
        rows.append(j)
        cols.append(j)
        vals.append(diag)
    
    A_mat = sp.csr_matrix((vals, (rows, cols)), shape=(n, n))
    
    # Full KKT: [A  -1; 1^T  0] [a; lambda] = [0; 1]
    e_vec = sp.csr_matrix(-np.ones(n).reshape(n, 1))
    one_row = sp.csr_matrix(np.ones((1, n)))
    zero_scalar = sp.csr_matrix([[0.0]])
    
    top = sp.hstack([A_mat, e_vec])
    bottom = sp.hstack([one_row, zero_scalar])
    KKT = sp.vstack([top, bottom])
    
    rhs = np.zeros(n + 1)
    rhs[-1] = 1.0
    
    try:
        sol = spla.spsolve(KKT, rhs)
        if np.any(np.isnan(sol)) or np.any(np.isinf(sol)):
            return None
        return sol[:n], sol[n]
    except Exception:
        return None


def newton_active_set(f0, time_limit, verbose=False, S_frac=0.15, max_iter=5000):

    Newton method based on KKT active set conditions.
    f0: initial sequence (will be normalized)

        f = f / total

        c = fast_convolve(f)

            no_improve += 1

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a KKT-based Newton active-set method
combined with diverse asymmetric initializations, smoothed mirror descent for quick
local optimization, and multi-length exploration. The KKT approach solves a linear
system derived from the first-order optimality conditions at each iteration, providing
a powerful Newton-like direction that is then used with a line search.
"""

import numpy as np
import scipy.sparse as sp
import scipy.sparse.linalg as spla
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalized_obj(f):
    """Compute the objective for a normalized (sum=1) sequence."""
    n = len(f)
    c = fast_convolve(f)
    return 2.0 * n * c.max()


def solve_kkt_system(n, S_list, w_arr, reg=1e-8):
    """
    Solve the KKT linear system:
      A * a - lambda * 1 = 0
      1^T * a = 1
    where (A*a)[j] = a[j] + sum_{k in S} w[k] * a[k-j]
    
    Returns (a, lambda) or None.
    """
    rows = np.empty(n + n * len(S_list) + n, dtype=np.int32)
    cols = np.empty(n + n * len(S_list) + n, dtype=np.int32)
    vals = np.empty(n + n * len(S_list) + n, dtype=np.float64)
    
    idx = 0
    for j in range(n):
        diag = 1.0 + reg
        for ii in range(len(S_list)):
            k = S_list[ii]
            m = k - j
            if 0 <= m <= n - 1:
                if m == j:
                    diag += w_arr[ii]
                else:
                    rows[idx] = j
                    cols[idx] = m
                    vals[idx] = w_arr[ii]
                    idx += 1
        rows[idx] = j
        cols[idx] = j
        vals[idx] = diag
        idx += 1
    
    nnz = idx
    A_mat = sp.csr_matrix((vals[:nnz], (rows[:nnz], cols[:nnz])), shape=(n, n))
    
    # Build full KKT matrix
    e_vec = sp.csr_matrix(-np.ones(n).reshape(n, 1))
    one_row = sp.csr_matrix(np.ones((1, n)))
    zero_scalar = sp.csr_matrix([[0.0]])
    
    top = sp.hstack([A_mat, e_vec])
    bottom = sp.hstack([one_row, zero_scalar])
    KKT = sp.vstack([top, bottom])
    
    rhs = np.zeros(n + 1)
    rhs[-1] = 1.0
    
    try:
        sol = spla.spsolve(KKT, rhs)
        if np.any(np.isnan(sol)) or np.any(np.isinf(sol)):
            return None
        return sol[:n], sol[n]
    except Exception:
        return None

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _try(parent, "parent")
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    # === Phase 1: Diverse Initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    lengths = [512, 640, 768, 896, 1024]
    candidates = []
    
    for n_try in lengths:
        for v in range(14):
            a_init = make_init(n_try, v, seed=seed)
            sc = normalized_obj(a_init)
            candidates.append((sc, a_init, f"v{v}_n{n_try}"))
    
    if parent is not None:
        sc = normalized_obj(parent)
        candidates.append((sc, parent, "parent"))
    
    candidates.sort(key=lambda x: x[0])
    top_n = min(8, len(candidates))
    print(f"  Top-3: {[f'{l}:{o:.4f}' for o, _, l in candidates[:3]]}")
    
    # Quick SMD on top candidates
    for idx in range(top_n):
        sc, a_init, label = candidates[idx]
        rem = deadline - time.monotonic() - 700
        if rem < 8:
            break
        t_alloc = min(rem / max(1, top_n - idx), 25)
        if t_alloc < 4:
            t_alloc = 4
        a_opt, M = smoothed_mirror_descent(a_init, time_limit=t_alloc)
        _try(a_opt, f"smd_{label}")
    
    # === Phase 2: Newton Active Set on best ===
    print("\n=== Phase 2: Newton Active Set ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_alloc = min(rem, 350)
            print(f"  NAS on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, M = newton_active_set(best_seq, time_limit=t_alloc, S_top=12, verbose=True)
            _try(a_opt, "nas_main")
    
    # NAS on second-best length (different from best)
    if best_seq is not None:
        best_n = len(best_seq)
        for idx in range(top_n):
            sc, a_init, label = candidates[idx]
            if len(a_init) == best_n:
                continue
            rem = deadline - time.monotonic() - 300
            if rem < 20:
                break
            t_alloc = min(rem * 0.5, 100)
            if t_alloc < 10:
                t_alloc = 10
            print(f"  NAS on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
            a_opt, M = newton_active_set(a_init, time_limit=t_alloc, S_top=10)
            _try(a_opt, f"nas_{label}")
            break  # Just one different-length NAS
    
    # === Phase 3: Multi-length Interpolation ===
    print("\n=== Phase 3: Multi-Length Exploration ===")
    if best_seq is not None:
        target_lengths = [640, 896, 1024, 1152, 1280]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 250
            if rem < 25:
                break
            if len(best_seq) == target_n:
                continue
            f_interp = interpolate_seq(best_seq, target_n)
            t_alloc = min(rem * 0.35, 70)
            if t_alloc < 12:
                t_alloc = 12
            print(f"  NAS interp n={target_n}: t={t_alloc:.0f}s")
            a_opt, M = newton_active_set(f_interp, time_limit=t_alloc, S_top=10)
            _try(a_opt, f"nas_interp_n{target_n}")
    
    # === Phase 4: Combined Polish ===
    print("\n=== Phase 4: Combined Polish ===")
    if best_seq is not None:
        # Peak-targeted refine first
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t1 = min(rem * 0.3, 50)
            a_opt, M = peak_targeted_refine(best_seq, time_limit=t1)
            _try(a_opt, "ptd")
        
        # Then NAS with fewer active constraints (for fine-tuning)
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t2 = min(rem * 0.5, 80)
            a_opt, M = newton_active_set(best_seq, time_limit=t2, S_top=6, verbose=True)
            _try(a_opt, "nas_fine")
        
        # SMD with very low learning rate
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t3 = min(rem * 0.6, 30)
            a_opt, M = smoothed_mirror_descent(best_seq, time_limit=t3, lr_init=0.003)
            _try(a_opt, "smd_fine")
    
    # === Phase 5: Random Restarts ===
    print("\n=== Phase 5: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(4):
            rem = deadline - time.monotonic() - 40
            if rem < 15:
                break
            t_alloc = min(rem / max(1, 4 - r), 50)
            if t_alloc < 8:
                t_alloc = 8
            
            if r == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = np.maximum(best_seq + noise, 1e-10)
                f_init = f_init / f_init.sum()
            elif r == 1:
                asym = make_init(n_best, variant=int(rng.integers(0, 14)), seed=seed + 200 + r)
                f_init = 0.85 * best_seq + 0.15 * asym
                f_init = f_init / f_init.sum()
            elif r == 2:
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Mix with a different-length interpolation
                new_n = n_best + int(rng.integers(-64, 65))
                new_n = max(256, new_n)
                f_interp = interpolate_seq(best_seq, new_n)
                f_init = f_interp
                # Adjust back to original length
                f_init = interpolate_seq(f_init, n_best)
            
            a_opt, M = newton_active_set(f_init, time_limit=t_alloc, S_top=8)
            _try(a_opt, f"restart_{r}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = _try(best_seq, "fallback")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
