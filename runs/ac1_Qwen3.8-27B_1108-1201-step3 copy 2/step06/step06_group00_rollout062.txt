```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating minimization

coordinate descent for final polishing.

The LP alternation is based on the state-of-the-art approach: fix f, solve LP

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute the objective value for a non-negative sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M):
    """
    Solve the LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g vector.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build the constraint matrix A where A[k, j] = f[j] if valid
    # (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # For row k, valid j are: max(0, k-n+1) <= j <= min(n-1, k)
    
    # Use sparse matrix for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    b = np.full(num_constraints, M)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0)
        else:
            return None
    except Exception as e:
        print(f"  LP error: {e}")
        return None

    best_obj = fast_convolve(f).max()

        M = b.max()

        if g is None:

        # Normalize g to sum=1
        g_sum = g.sum()
        if g_sum < 1e-10:

        g = g / g_sum
        
        # Line search for blending parameter
        best_t = 0
        best_trial_obj = best_obj
        for trial_t in [t, t*0.5, t*0.7, t*0.3, t*1.2]:

            if trial_t < 0.001:
                continue
            f_trial = (1 - trial_t) * f + trial_t * g
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 0.01:
                continue
            f_trial = f_trial / s
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_t = trial_t
        
        if best_t > 0:
            f_new = (1 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            f_new = f_new / s
            obj_new = fast_convolve(f_new).max()
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f_new.copy()
            
            f = f_new
        else:
            # No improvement, try smaller t
            f_new = (1 - min(t*0.1, 0.01)) * f + min(t*0.1, 0.01) * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            f = f_new
        
        # Decay t
        t = max(t * t_decay, min_t)
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter {it}: obj={best_obj:.8f}, t={t:.4f}, time={elapsed:.0f}s")
    
    return best_f, best_obj


def multiplicative_weights(f0, time_limit, eta_init=0.1, eta_decay=0.999,
                           max_iters=10000, n_top=None, seed=42):
    """
    Multiplicative weights / mirror descent on the simplex.
    Update: a[i] <- a[i] * exp(-eta * g[i]) / Z
    where g is the subgradient of the objective.
    """
    t0 = time.monotonic()
    n = len(f0)
    a = f0.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    
    if n_top is None:
        n_top = max(10, n // 10)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        b = fast_convolve(a)
        b_max = b.max()
        
        # Find top peaks
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        
        # Compute subgradient as average over top peaks
        # g[m] = (1/|top|) * sum_{k in top} d/dam (a*a)[k]
        g = np.zeros(n)
        count = 0
        for k in top_idx:
            # d/dam (a*a)[k] = 2*a[k-m] if 0<=k-m<n and 2m!=k, a[m] if 2m=k and valid
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                j = k - m
                if 2 * m == k:
                    g[m] += a[m]
                else:
                    g[m] += 2 * a[j]
            count += 1
        
        if count > 0:
            g /= count
        
        # Subgradient of 2n*max(conv) is 2n*g (but we work on simplex, so normalize)
        # Actually, the gradient of max(conv) with respect to a is g as computed above
        # (up to the weighting over active peaks)
        
        # Multiplicative weights update
        # a_new[i] = a[i] * exp(-eta * g[i]) / sum_j a[j] * exp(-eta * g[j])
        log_weights = -eta * g
        log_weights -= log_weights.max()  # numerical stability
        weights = np.exp(log_weights)
        a_new = a * weights
        a_sum = a_new.sum()
        if a_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_sum
        
        # Compute new objective
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        eta = max(eta * eta_decay, 1e-6)
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MW iter {it}: obj={best_obj:.8f}, eta={eta:.2e}, time={elapsed:.0f}s")
    
    return best_a, best_obj


def coordinate_descent(f0, time_limit, n_grid=15):
    """
    Cyclic coordinate descent with grid search per coordinate.
    """
    t0 = time.monotonic()
    n = len(f0)
    a = f0.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Grid of multipliers for each coordinate
    multipliers = np.linspace(0.3, 2.0, n_grid)
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 10:
                    continue
                
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                best_obj = best_j_obj
                best_a = a.copy()
                improved = True
    
    return best_a, best_obj


def paren_subgradient(f0, time_limit, n_top=None, eta_init=0.01, seed=42):
    """
    Subgradient with paren-based updates for escaping flat regions.
    Uses block coordinate updates on groups of elements.
    """
    t0 = time.monotonic()
    n = len(f0)
    a = f0.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    rng = np.random.default_rng(seed)
    
    if n_top is None:
        n_top = max(10, n // 5)
    
    block_size = max(10, n // 100)
    n_blocks = n // block_size
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        
        # Select a random block to update
        block_idx = rng.integers(0, n_blocks)
        start = block_idx * block_size
        end = start + block_size
        
        # Compute subgradient for the block
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        g_block = np.zeros(block_size)
        for k in top_idx:
            for m in range(start, end):
                j = k - m
                if 0 <= j < n:
                    if 2 * m == k:
                        g_block[m - start] += a[m]
                    else:
                        g_block[m - start] += 2 * a[j]
        
        g_block /= len(top_idx)
        
        # Update the block using projected gradient
        eta = eta_init / (1 + it * 0.001)
        a_block = a[start:end]
        a_block_new = np.maximum(a_block - eta * g_block, 0)
        
        # Rescale to maintain sum
        rest_sum = 1.0 - a_block.sum()
        block_sum = a_block_new.sum()
        if block_sum > rest_sum:
            a_block_new *= rest_sum / block_sum
        a[start:end] = a_block_new
        a = a / a.sum()
        
        obj_new = fast_convolve(a).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a.copy()
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  BlockSG iter {it}: obj={best_obj:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

**Approach: Direct optimization of the peak using the active set**

At any point `a`, the active set is `A = {k : (a*a)[k] = max}`. If we can characterize the active set, we can do a more targeted optimization.

The key insight: at a local minimum, the gradient of the active peaks should be "balanced" - i.e., the weighted average of the gradients of active peaks should be constant (on the interior of the simplex).

This gives us a system of equations:
`sum_{k in A} w_k * grad_k(a) = c * 1` (constant vector)

for some `w_k >= 0`, `sum w_k = 1`, and `c > 0`.

This is a non-linear system (because `grad_k(a)` depends on `a`). But we can solve it iteratively:
1. Start with `w_k = 1/|A|` for all active k
2. Compute `g = sum w_k * grad_k`
3. Find the direction that makes `g` as constant as possible
4. Update `a` in that direction
5. Update the active set
6. Repeat

This is essentially what the subgradient method does, but with the added step of optimizing the weights `w_k`.

Actually, the optimal `w_k` at a local minimum can be found by solving:
minimize `||sum w_k * grad_k||^2` subject to `w_k >= 0`, `sum w_k = 1`

This is a quadratic program (QP) in the weights. For a small active set (say |A| <= 20), this is very fast.

```python
def optimal_subgradient_step(a, n_top=50):
    """
    Compute the optimal subgradient (minimum norm element of the subdifferential).
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    
    # Find active set (all k where b[k] is close to max)
    tol = 1e-10
    active = np.where(b >= b_max - tol)[0]
    
    # If active set is too large, keep only the top ones
    if len(active) > n_top:
        active = np.argpartition(b, -n_top)[-n_top:]
    
    # Compute gradients for each active peak
    # grad_k[m] = 2*a[k-m] if 0<=k-m<n and 2m!=k, a[m] if 2m=k
    G = np.zeros((len(active), n))
    for i, k in enumerate(active):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                if 2 * m == k:
                    G[i, m] = a[m]
                else:
                    G[i, m] = 2 * a[j]
    
    # Solve QP: min ||G^T w||^2 s.t. w >= 0, sum(w) = 1
    # This is equivalent to: min w^T (G G^T) w s.t. w >= 0, 1^T w = 1
    # (since ||G^T w||^2 = w^T G G^T w)
    
    from scipy.optimize import minimize
    
    m = len(active)
    GGT = G @ G.T
    
    def qp_obj(w):
        return w @ GGT @ w
    
    def qp_grad(w):
        return 2 * GGT @ w
    
    constraints = [{'type': 'eq', 'fun': lambda w: w.sum() - 1, 'jac': lambda w: np.ones(m)}]
    bounds = [(0, 1)] * m
    w0 = np.ones(m) / m
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
```

```python
2. Optimal subgradient (min-norm subdifferential element) for refinement
3. Coordinate descent for final polishing

then blends the solution back. The optimal subgradient finds the best direction

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_alternate(f0, time_limit, t_init=0.3, t_decay=0.998, min_t=0.005,
                 max_iters=5000, lp_rows=None, lp_cols=None, num_con=None):
    """LP-based alternating minimization with adaptive step size."""
    t0 = time.monotonic()
    n = len(f0)
    f = f0.copy().astype(np.float64)
    f = f / f.sum()

    best_f = f.copy()
    best_bmax = fast_convolve(f).max()

    if lp_rows is None:
        lp_rows, lp_cols, num_con = precompute_lp_sparsity(n)

    it = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        M = b.max()

        g = solve_lp_direction(f, M, lp_rows, lp_cols, num_con, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check convergence
        if np.abs(g - f).max() < 1e-8:
            break

        # Line search over several step sizes
        best_t = 0.0
        best_trial = best_bmax
        for trial_t in [t_init, t_init * 0.5, t_init * 0.7, t_init * 0.3, t_init * 1.5]:
            trial_t = np.clip(trial_t, 0.001, 1.0)
            f_trial = (1.0 - trial_t) * f + trial_t * g
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_trial:
                best_trial = obj_trial
                best_t = trial_t

        if best_t > 0:
            f_new = (1.0 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            obj_new = fast_convolve(f_new).max()
            if obj_new < best_bmax:
                best_bmax = obj_new
                best_f = f_new.copy()
            f = f_new
        it += 1
        t_init = max(t_init * t_decay, min_t)

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: bmax={best_bmax:.10f}, t={t_init:.5f}, time={elapsed:.0f}s")

    return best_f, best_bmax


def compute_active_gradients(a, b, n_top=50, tol=1e-8):
    """Compute gradients for active peaks. Returns (gradients_matrix, active_indices)."""
    n = len(a)
    b_max = b.max()
    active = np.where(b >= b_max - b_max * tol)[0]
    if len(active) > n_top:
        active = np.argpartition(b, -n_top)[-n_top:]
    
    m = len(active)
    G = np.zeros((m, n))
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for jj in range(lo, hi + 1):
            if 2 * jj == k:
                G[i, jj] += a[jj]
            else:
                G[i, jj] += 2.0 * a[k - jj]
    return G, active

def solve_qp_weights(G):
    """Solve min ||G^T w||^2 s.t. w >= 0, sum(w) = 1."""
    m = G.shape[0]
    if m == 1:

    GGT = G @ G.T
    w0 = np.ones(m) / m
    
    def obj_func(w):
        return w @ GGT @ w
    
    def grad_func(w):
        return 2.0 * GGT @ w
    
    def constraint_fun(w):
        return w.sum() - 1.0
    
    def constraint_jac(w):
        return np.ones(m)

            s = w_opt.sum()
            if s > 1e-15:

def optimal_subgradient_optimize(f0, time_limit, n_top=30, eta_init=0.02,

    a = f0.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_bmax = fast_convolve(a).max()
    eta = eta_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        G, active = compute_active_gradients(a, b, n_top=n_top)
        
        if G.shape[0] == 0:
            break
        
        w_opt = solve_qp_weights(G)
        g_opt = G.T @ w_opt
        
        # Projected gradient step on simplex
        a_new = project_simplex(a - eta * g_opt)
        
        # Also try a few different step sizes
        best_local = best_bmax
        best_local_a = a
        for trial_eta in [eta, eta * 0.5, eta * 2.0, eta * 0.1]:
            a_trial = project_simplex(a - trial_eta * g_opt)
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_local:
                best_local = obj_trial
                best_local_a = a_trial
        
        if best_local < best_bmax:
            best_bmax = best_local
            best_a = best_local_a.copy()
        
        a = best_local_a
        eta = max(eta * eta_decay, 1e-8)
        it += 1
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    OS iter {it}: bmax={best_bmax:.10f}, eta={eta:.2e}, time={elapsed:.0f}s")
    
    return best_a, best_bmax


def coordinate_descent(f0, time_limit, n_grid=15):
    """Cyclic coordinate descent with grid search."""
    t0 = time.monotonic()
    n = len(f0)
    a = f0.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.linspace(0.4, 1.8, n_grid)
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 5:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                best_obj = best_j_obj
                best_a = a.copy()
                improved_global = True
    
    return best_a, best_obj


def mirror_descent_optimize(f0, time_limit, eta_init=0.05, eta_decay=0.999,
                            n_top=50, max_iters=100000):
    """Mirror descent (multiplicative weights) with active set subgradient."""
    t0 = time.monotonic()
    n = len(f0)
    a = f0.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_bmax = fast_convolve(a).max()
    eta = eta_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        
        # Uniform subgradient over top peaks
        g = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                if 2 * m == k:
                    g[m] += a[m]
                else:
                    g[m] += 2.0 * a[k - m]
        g /= len(top_idx)
        
        # Multiplicative weights update
        log_w = -eta * g
        log_w -= log_w.max()
        w = np.exp(log_w)
        a_new = a * w
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_bmax:
            best_bmax = obj_new
            best_a = a_new.copy()
        
        a = a_new
        eta = max(eta * eta_decay, 1e-8)
        it += 1
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}: bmax={best_bmax:.10f}, eta={eta:.2e}, time={elapsed:.0f}s")
    
    return best_a, best_bmax
```
