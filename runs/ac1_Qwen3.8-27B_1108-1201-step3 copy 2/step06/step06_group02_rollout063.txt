```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using parametric power-law initialization
and subgradient optimization targeting peak reduction.

Key insight: non-symmetric power-law sequences a_i = (i+1)^p minimize the
central auto-convolution peak by "anti-correlating" a_i with a_{n-1-i}.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using power-law (sawtooth) initialization
with subgradient optimization.

Key insight: Non-symmetric monotone sequences (sawtooth a_i = (i+1)^p) minimize
the central auto-convolution peak by pairing large values with small ones.
The sawtooth (p=1) gives objective ≈ 4/3 for large n, far below the previous 1.505.
Higher powers p give even lower asymptotic objectives but must satisfy n > (2p+1)!/(p!)^2
for the center to dominate the edge peaks.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def safe_objective(a):
    """Compute 2*n*max(conv)/sum^2, return inf if invalid."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_power_law(n, p=1.0):
    """Generate a_i = (i+1)^p normalized to sum=1."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def make_scaled(a, max_val=1000.0):
    """Scale sequence so max = max_val (for avoiding clipping)."""
    m = a.max()
    if m > max_val:
        a = a * (max_val / m)
    return a


def subgradient_optimize(a0, time_limit, lr_init=0.003, decay=0.9997, verbose=False):
    """
    Projected subgradient descent to minimize max(conv) on the simplex.
    a is kept normalized to sum=1; objective is 2*n*max(conv).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_val = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        cur_val = conv.max()

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SD iter {it}: max_conv={best_val:.8f}, obj={2*n*best_val:.6f}")
        else:
            no_improve += 1

        # Find the argmax (and nearby peaks within 1%)
        peak_k = int(np.argmax(conv))
        threshold = conv.max() * 0.995
        peak_indices = np.where(conv >= threshold)[0]

        # Compute subgradient: d(conv[k])/d(a[m]) = 2*a[k-m] for valid m
        sg = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            partners = k - idx
            sg[idx] += 2.0 * a[partners]
        if len(peak_indices) > 0:
            sg /= len(peak_indices)

        # Gradient step with projection
        a_new = project_simplex(a - lr * sg)
        a = a_new

        lr *= decay
        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7919)
            pert = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + pert)
            lr = lr_init
            no_improve = 0

        if lr < 1e-12:
            break

    return best_a, best_val


def peak_shaving_optimize(a0, time_limit, verbose=False):
    """
    Targeted peak reduction: identify the dominant peak, find the
    most contributing pair, and redistribute mass to reduce it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_val = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_val - 1e-14:
            best_val = M
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    PS iter {it}: max_conv={best_val:.8f}, obj={2*n*best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > 100:
            break

        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        partners = peak_k - idx

        # Products contributing to the peak
        products = a[idx] * a[partners]
        total = products.sum()
        if total < 1e-15:
            break

        # Find the largest contributing pair
        max_ci = np.argmax(products)
        src = idx[max_ci]
        partner = partners[max_ci]

        # Strategy: reduce the larger of the pair, move mass to a
        # coordinate that doesn't contribute to this peak
        if a[src] >= a[partner]:
            source = int(src)
        else:
            source = int(partner)

        # Find non-contributing coordinates
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[idx] = False
        if non_contrib.sum() < 1:
            non_contrib[:] = True
            non_contrib[source] = False

        # Try several reduction fractions
        found_improve = False
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = a[source] * frac
            if diff < 1e-15:
                continue
            trial = a.copy()
            trial[source] -= diff
            nc_mask = non_contrib.copy()
            nc_vals = trial[nc_mask]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                continue
            # Distribute proportionally to existing values (or uniformly)
            if np.any(nc_vals > 1e-12):
                trial[nc_mask] += diff * nc_vals / nc_sum
            else:
                nc_idx = np.where(nc_mask)[0]
                trial[nc_idx] += diff / len(nc_idx)
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < M - 1e-14:
                a = trial
                found_improve = True
                break

        if not found_improve:
            # Try a smaller adjustment or different peak
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_val


def coordinate_descent(fini, time_limit, verbose=False, max_iter=50000):
    """
    Cyclic coordinate descent: for each coordinate i, find the value
    that minimizes the peak of the convolution.
    """
    a = np.maximum(fini.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_val = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for i in range(n):
            # Try several multipliers for a[i]
            orig = a[i]
            if orig < 1e-15:
                # Try small positive values
                for trial_val in [orig + 1e-4, orig + 1e-3]:
                    a[i] = trial_val
                    a = a / a.sum()
                    val = fast_convolve(a).max()
                    if val < best_val - 1e-12:
                        best_val = val
                        best_a = a.copy()
                        improved = True
                        break
                    a = a / a.sum()
                    a[i] = orig
                continue

            for factor in [0.7, 0.85, 1.15, 1.3]:
                a[i] = orig * factor
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                val = fast_convolve(a).max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                a[i] = orig
                a = a / a.sum()

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
            a = best_a.copy()

        if verbose and it % 10 == 0:
            print(f"    CD iter {it}: max_conv={best_val:.8f}, obj={2*n*best_val:.6f}")

    return best_a, best_val

    print("\n=== Phase 5: Coordinate Descent Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.5, 60)
            n_cd = len(best_seq)
            # For large n, do partial CD on a subset
            if n_cd > 1024:
                # Work on a smaller version
                n_small = 512
                old_x = np.linspace(0, 1, n_cd)
                new_x = np.linspace(0, 1, n_small)
                a_small = np.interp(new_x, old_x, best_seq)
                a_small = np.maximum(a_small, 0.0)
                a_small = make_scaled(a_small, 1000.0)
            else:
                a_small = best_seq.copy()

            print(f"  CD (n={len(a_small)}): alloc={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(a_small, time_limit=t_alloc, verbose=True)
            a_opt = make_scaled(a_opt, 1000.0)
            try_seq(a_opt, "cd_polish")

    # === Phase 6: Final round of subgradient with different initializations ===
    print("\n=== Phase 6: Final SD rounds ===")
    final_inits = []
    # Best sequence perturbed
    if best_seq is not None:
        for amp in [0.001, 0.005, 0.01]:
            rng = np.random.default_rng(seed + int(amp * 10000))
            pert = best_seq + rng.standard_normal(len(best_seq)) * amp * best_seq.max()
            pert = np.maximum(pert, 0.0)
            final_inits.append((f"pert{amp}", pert))

    # Fresh sawtooth with slightly different parameters
    for n_f in [1024, 1536, 2048]:
        for p_f in [0.95, 1.0, 1.05, 1.1]:
            a_f = make_power_law(n_f, p_f)
            a_f = make_scaled(a_f, 1000.0)
            final_inits.append((f"fresh_n{n_f}_p{p_f}", a_f))

    for label, a_init in final_inits:
        rem = deadline - time.monotonic() - 30
        if rem < 15:
            break
        t_alloc = min(rem * 0.25, 40)
        if t_alloc < 5:
            break
        a_opt, obj = subgradient_optimize(a_init, time_limit=t_alloc)
        a_opt = make_scaled(a_opt, 1000.0)
        try_seq(a_opt, f"final_{label}")

    # === Phase 7: Peak-shaving on final best ===
    print("\n=== Phase 7: Final Peak-Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem, 15)
            a_opt, obj = peak_shaving_optimize(best_seq, time_limit=t_alloc)
            a_opt = make_scaled(a_opt, 1000.0)
            try_seq(a_opt, "final_ps")

    # Fallback
    if best_seq is None:
        best_seq = make_scaled(make_power_law(768, 1.0), 1000.0)
        best_score = safe_objective(best_seq)

    # Final verification
    final_score = safe_objective(best_seq)
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    total_time = time.monotonic() - (deadline - max(1.0, float(budget_s) - 30.0))
    print(f"Total time: {total_time:.1f}s")

    return [float(x) for x in best_seq]
```
