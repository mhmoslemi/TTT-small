def lp_direction_minimax(f, M, active_frac=1.0):
    """
    Solve LP: minimize t s.t. A@g <= t, sum(g)=1, g>=0.
    Returns (g, t_min) where g is the optimal direction and t_min = min max_k (f*g)_k.
    """
    n = len(f)
    conv = fast_convolve(f)
    
    # Active set
    if active_frac < 1.0:
        threshold = conv.min() + active_frac * (conv.max() - conv.min())
        active_mask = conv >= threshold
    else:
        active_mask = np.ones(2*n-1, dtype=bool)
    
    A_full = build_constraint_matrix(f)
    A = A_full[active_mask, :]
    m = A.shape[0]
    
    # LP: minimize t
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.concatenate([np.zeros(n), [1.0]])
    
    # Inequality: A@g - t <= 0  =>  [A, -1] @ [g; t] <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.hstack([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = res.x[:n]
        t_min = res.x[n]
        return g, t_min
    return None, float('inf')

"""
LP-based mirror descent with minimax direction finding for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses linear programming to find optimal
descent directions (minimizing max cross-correlation), combined with
subgradient polish, active set reduction, and multi-start diversity.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_constraint_matrix(f):
    """Build Toeplitz matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    first_col = np.concatenate([f, np.zeros(n - 1)])
    first_row = np.concatenate([[f[0]], np.zeros(n - 1)])
    return toeplitz(first_col, first_row)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_minimax_lp(f, active_frac=1.0, verbose=False):
    """
    Solve LP: minimize t s.t. A@g <= t, sum(g)=1, g>=0.
    Returns (g, t_min) or (None, inf).
    """
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()

    if active_frac < 1.0:
        threshold = conv.min() + active_frac * (M - conv.min())
        active_mask = conv >= threshold
    else:
        active_mask = np.ones(2 * n - 1, dtype=bool)

    A_full = build_constraint_matrix(f)
    A = A_full[active_mask, :]
    m = A.shape[0]

    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.hstack([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        gs = g.sum()
        if gs < 1e-15:
            return None, float('inf')
        g = g / gs
        t_min = res.x[n]
        return g, t_min
    return None, float('inf')


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g

def coordinate_descent(f_init, time_limit, verbose=False, label=""):
    """Fine coordinate descent for local polish."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exponential(n, rate=5.0):
    """Exponentially decaying sequence."""
    a = np.exp(-rate * np.arange(n) / n)
    a = a / a.sum()
    return a


def make_bimodal(n, c1=0.3, c2=0.7, sigma=0.08):
    """Bimodal Gaussian."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = a / a.sum()
    return a


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym = a_sym / ss
    return a_sym

        if t_per > 20:
            for r in range(n_restarts):
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                t_alloc = min(t_per, remaining - 10)
                if t_alloc < 15:
                    break
                if r == 0:
                    f_init = best_seq.copy()
                elif r == 1:
                    noise = rng.standard_normal(len(best_seq)) * 0.003
                    f_init = np.maximum(best_seq + noise, 0)
                    f_init = f_init / f_init.sum()
                else:
                    f_init = 0.8 * best_seq + 0.2 * make_exponential(len(best_seq), 3.0)
                    f_init = f_init / f_init.sum()
                print(f"  Restart {r}, time={t_alloc:.0f}s")
                a_opt, obj = lp_optimize(f_init, time_limit=t_alloc, verbose=(r == 0),
                                         label=f"restart_{r}")
                try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Subgradient + LP alternating ===
    print(f"\n=== Phase 7: Alternating SG+LP ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_total = min(remaining - 20, 200)
            n_rounds = 4
            t_round = t_total / n_rounds
            for rr in range(n_rounds):
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                t_r = min(t_round, remaining - 10)
                if t_r < 10:
                    break
                # Subgradient first
                a_opt, obj = subgradient_optimize(best_seq, time_limit=t_r * 0.5,
                                                  top_k=20, temp_init=0.0002,
                                                  lr_init=0.005, label=f"alt_SG_{rr}")
                try_seq(a_opt, f"alt_SG_{rr}")
                # Then LP
                remaining = deadline - time.monotonic()
                if remaining > 15:
                    a_opt, obj = lp_optimize(best_seq, time_limit=min(t_r * 0.5, remaining - 10),
                                             verbose=False, label=f"alt_LP_{rr}")
                    try_seq(a_opt, f"alt_LP_{rr}")

    # === Phase 8: Coordinate Descent Polish ===
    print(f"\n=== Phase 8: CD Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_cd = min(remaining - 10, 120)
            print(f"  Time: {t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd, verbose=True, label="CD")
            try_seq(a_opt, "CD")

    # === Phase 9: Final LP polish ===
    print(f"\n=== Phase 9: Final LP Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_final = min(remaining - 10, 100)
            print(f"  Time: {t_final:.0f}s")
            a_opt, obj = lp_optimize(best_seq, time_limit=t_final, verbose=True, label="final_LP")
            try_seq(a_opt, "final_LP")

            # Try symmetrized
            a_s = symmetrize(best_seq)
            try_seq(a_s, "final_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
