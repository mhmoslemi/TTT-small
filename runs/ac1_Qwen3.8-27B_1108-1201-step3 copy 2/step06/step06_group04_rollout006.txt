"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using dual ascent with projected gradient,
momentum-based optimization, peak equalization, and multi-length exploration.
Uses a different approach than LP-Frank-Wolfe: instead of solving an LP for the
direction, it uses a min-max duality to concentrate optimization on the peaks.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient with Nesterov
momentum on a smoothed max-convolution objective, combined with adaptive beta
scheduling, multi-length exploration, peak-targeted coordinate adjustments,
and structured random restarts. Differs from LP-Frank-Wolfe by using second-order
like dynamics (momentum) directly on the smoothed objective rather than solving
an LP for each direction.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_and_grad(a, beta):
    """Compute smoothed max of conv(a,a) and its gradient w.r.t. a."""
    conv = fast_convolve(a)
    n = len(a)
    m = conv.max()
    shifted = beta * (conv - m)
    # Numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    w = exp_shifted / sum_exp  # softmax weights
    
    obj = m + np.log(sum_exp) / beta
    
    # Gradient: d(obj)/d(a[j]) = beta * sum_k w[k] * d(conv[k])/d(a[j])
    # d(conv[k])/d(a[j]) = a[k-j] + a[k-j] (factor 2 except at diagonal)
    # Simplification: use 2*a[k-j] for all (small error at diagonal)
    grad = np.zeros(n)
    # Efficient computation: grad[j] = beta * sum_k w[k] * 2*a[k-j] (for valid k,j)
    # = 2 * beta * sum_k w[k] * a[k-j]
    # For each j: sum over k such that 0<=k-j<n, i.e., j<=k<=j+n-1
    # This is a "sliding correlation" of w with a
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            other = k - j
            if other < n:
                if j == other:
                    grad[j] += beta * w[k] * a[j]
                else:
                    grad[j] += beta * w[k] * a[other]
    return obj, grad


def smoothed_max_and_grad_fast(a, beta):
    """Faster version using the fact that grad[j] = beta * (w * a_flipped correlate)."""
    conv = fast_convolve(a)
    n = len(a)
    m = conv.max()
    shifted = beta * (conv - m)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    w = exp_shifted / sum_exp
    
    obj = m + np.log(sum_exp) / beta
    
    # grad[j] = beta * sum_k w[k] * d(conv[k])/d(a[j])
    # d(conv[k])/d(a[j]) ≈ 2*a[k-j] (for j≠k-j) or a[j] (for j=k-j)
    # Approximation: grad[j] ≈ 2*beta * sum_k w[k]*a[k-j]
    # This is 2*beta * (cross-correlation of w and a)
    # Specifically: grad[j] ≈ 2*beta * sum_{k} w[k] * a[k-j] (summed over valid k,j)
    
    # Compute using FFT: cross-correlation
    # correlate(w, a, 'full')[k] = sum_j w[k-j]*a[j] (numpy convention)
    # We want: sum_k w[k]*a[k-j] for each j
    # = correlate(a, w, 'full')[j + len(w) - 1 - (len(a)-1)]... 
    
    # Simpler: just use direct for w (which has structure - concentrated at peaks)
    # For large beta, w is concentrated on few peaks, so the loop is fast
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = beta * w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        others = k - idx
        contrib = 2.0 * a[others]
        # Fix diagonal: where idx == others (i.e., 2*idx = k)
        diag_mask = idx == others
        if diag_mask.any():
            contrib[diag_mask] = a[others[diag_mask]]
        grad[idx] += wk * contrib
    
    return obj, grad


def momentum_optimize(a_init, time_limit, beta_init=20.0, beta_final=200.0,
                      momentum_init=0.8, verbose=True):
    """
    Projected gradient with Nesterov momentum on smoothed max convolution.
    Adaptive beta schedule: start low (smooth), increase (sharp).
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_obj_raw = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum velocity
    lr = 1.0 / (n * 0.5)
    beta = beta_init
    
    no_improve = 0
    it = 0
    momentum = momentum_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adapt beta: gradually increase
        frac = min(1.0, it / max(1, int(0.7 * time_limit / 0.05)))
        beta = beta_init * (beta_final / beta_init) ** frac
        
        # Nesterov lookahead
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0)
        sl = a_look.sum()
        if sl > 1e-15:
            a_look = a_look / sl
        
        obj, grad = smoothed_max_and_grad_fast(a_look, beta)
        
        # Update momentum
        v = momentum * v - lr * grad
        v -= v.mean()  # Keep zero-mean (tangent to simplex)
        
        # Step
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Check with actual (unsmoothed) objective
        actual_obj = 2.0 * n * fast_convolve(a_new).max()
        a = a_new
        
        if actual_obj < best_obj_raw - 1e-13:
            best_obj_raw = actual_obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 2.0 / n)
            if verbose and it % 100 == 0:
                print(f"    Momentum iter {it}: obj={best_obj_raw:.10f}, beta={beta:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve > 30:
                lr = lr * 0.7
            if no_improve > 100:
                # Reset momentum, try smaller beta
                v = np.zeros(n)
                beta = max(beta_init, beta * 0.5)
                no_improve = 0
            if no_improve > 300:
                break
        
        # Decay momentum slowly
        momentum = max(0.5, momentum * 0.9999)
    
    return best_a, best_obj_raw


def peak_targeted_cd(a_init, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: for each top peak, try reducing
    the contributing elements and redistributing to non-contributing ones.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 60:
        conv = fast_convolve(a)
        M = conv.max()
        improved = False
        
        # Get top peaks
        n_peaks = min(8, 2 * n - 1)
        peak_indices = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            # Contributors to peak k
            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue
            
            # Non-contributors
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je + 1] = False
            nc_count = non_contrib.sum()
            if nc_count < 1:
                continue
            
            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(6, len(order))):
                j_max = int(idx[order[oi]])
                old_val = a[j_max]
                if old_val < 1e-14:
                    continue
                
                for frac in [0.3, 0.5, 0.7, 0.85]:
                    new_val = old_val * frac
                    delta = old_val - new_val
                    if delta < 1e-15:
                        continue
                    
                    trial = a.copy()
                    trial[j_max] = new_val
                    # Redistribute delta to non-contributors proportionally
                    nc_vals = a[non_contrib]
                    nc_sum = nc_vals.sum()
                    if nc_sum < 1e-15:
                        # Uniform redistribution
                        trial[non_contrib] += delta / nc_count
                    else:
                        trial[non_contrib] += delta * nc_vals / nc_sum
                    
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts
                    
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = trial.copy()
                        a = trial
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Aggressive: reduce all peak contributors together
                k = int(np.argmax(fast_convolve(a)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = a.copy()
                trial[js:je + 1] *= 0.9
                delta = a.sum() - trial.sum()
                if delta > 1e-15:
                    nc = np.ones(n, dtype=bool)
                    nc[js:je + 1] = False
                    if nc.sum() > 0:
                        nw = a[nc].sum()
                        if nw > 1e-15:
                            trial[nc] += delta * a[nc] / nw
                        else:
                            trial[nc] += delta / nc.sum()
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    a = trial
                    best_obj = obj
                    best_a = trial.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Projected gradient with Nesterov momentum on smoothed max-conv
    2. Peak-targeted coordinate descent  
    3. Multi-length exploration
    4. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        n = 768
        x = np.arange(n) - n / 2.0
        sigma = n * 0.28
        parent = np.exp(-0.5 * (x / sigma) ** 2)
        parent = parent / parent.sum()
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default_gauss")
    
    n_parent = len(parent)
    
    # === Phase 1: Momentum optimization on parent ===
    print(f"\n=== Phase 1: Momentum Opt (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 600.0)
    print(f"  Time: {t1:.0f}s")
    a_opt, obj = momentum_optimize(parent, time_limit=t1, beta_init=15.0,
                                   beta_final=300.0, momentum_init=0.85, verbose=True)
    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "momentum_main")
    
    # === Phase 2: Peak CD on best ===
    if best_seq is not None:
        print(f"\n=== Phase 2: Peak CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t2 = min(rem * 0.2, 200.0)
        print(f"  Time: {t2:.0f}s")
        a_opt, obj = peak_targeted_cd(best_seq, time_limit=t2, verbose=False)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "peak_cd")
    
    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length ===")
    base = best_seq if best_seq is not None else parent
    for target_n in [n_parent - 16, n_parent - 8, n_parent + 8, n_parent + 16,
                     n_parent - 32, n_parent + 32]:
        if target_n < 100:
            continue
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score,
                                             f"interp_n{target_n}")
        # Short momentum optimization on this length
        t_alloc = min(rem * 0.3, 80.0)
        if t_alloc > 30:
            print(f"  Momentum n={target_n}: t={t_alloc:.0f}s")
            a_opt, obj = momentum_optimize(f_interp, time_limit=t_alloc,
                                           beta_init=15.0, beta_final=200.0,
                                           momentum_init=0.85, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"mom_n{target_n}")
    
    # === Phase 4: Second momentum pass with different params ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Momentum v2 (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        if rem > 60:
            t4 = min(rem * 0.3, 250.0)
            print(f"  Time: {t4:.0f}s")
            # Use different momentum and beta schedule
            a_opt, obj = momentum_optimize(best_seq, time_limit=t4,
                                           beta_init=50.0, beta_final=500.0,
                                           momentum_init=0.9, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "momentum_v2")
    
    # === Phase 5: Perturbed restarts ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Perturbed ===")
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem / (5 - p), 100.0)
            if t_alloc < 20:
                break
            
            if p == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                # Blend with uniform
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif p == 2:
                # Reverse blend
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Sharper Gaussian blend
                x = np.arange(n_best) - n_best / 2.0
                sigma = n_best * 0.15
                gauss = np.exp(-0.5 * (x / sigma) ** 2)
                gauss = gauss / gauss.sum()
                f_init = 0.6 * best_seq + 0.4 * gauss
                f_init = f_init / f_init.sum()
            
            label = f"perturb_{p}"
            best_seq, best_score = try_sequence(f_init, best_seq, best_score, "init_" + label)
            a_opt, obj = momentum_optimize(f_init, time_limit=t_alloc,
                                           beta_init=20.0, beta_final=300.0,
                                           momentum_init=0.85, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, label)
    
    # === Phase 6: Final peak CD polish ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Final Peak CD ===")
        rem = deadline - time.monotonic()
        if rem > 30:
            t6 = min(rem * 0.5, 100.0)
            a_opt, obj = peak_targeted_cd(best_seq, time_limit=t6, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "final_cd")
    
    # === Phase 7: Final micro-momentum ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t7 = min(rem * 0.6, 60.0)
            print(f"\n=== Phase 7: Micro Momentum (t={t7:.0f}s) ===")
            a_opt, obj = momentum_optimize(best_seq, time_limit=t7,
                                           beta_init=100.0, beta_final=500.0,
                                           momentum_init=0.7, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "micro_mom")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
