"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set LP Frank-Wolfe with warm-cashing and constraint management
2. Analytic gradient descent on the active peak
3. Pair-unbalancing coordinate descent
4. Spectral (Patterson-CHEN) method
5. Multi-length exploration with diverse asymmetric initializations
6. Momentum-based perturbation for escaping local minima

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def objective_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def build_constraint_row(f, k, n):
    """Build row k of the constraint matrix: A[k,j] = f[j] for valid j."""
    row = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    row[js:je + 1] = f[js:je + 1]
    return row


def solve_lp_active(f, M, n, active_indices, relax=1.0):
    """Solve LP with only active constraints. Returns direction g or None."""
    if len(active_indices) == 0:
        return np.ones(n) / n

    c = -np.ones(n)
    A_ub = np.zeros((len(active_indices), n))
    b_ub = np.full(len(active_indices), M * relax)

    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je + 1] = f[js:je + 1]

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 10, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def full_lp_solve(f, M, n, relax=1.0):
    """Solve the full LP with all constraints."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[k, js:je + 1] = f[js:je + 1]
    b_ub = np.full(2 * n - 1, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 15, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def gradient_descent_on_peak(f, time_limit, lr_init=0.01, lr_decay=0.999,
                             top_k=5, verbose=False):
    """Gradient descent targeting the top peaks of the convolution."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    momentum = np.zeros(n)
    beta = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        peak_vals = peak_vals / peak_vals.max()

        grad = np.zeros(n)
        for ii in range(len(peaks)):
            k = int(peaks[ii])
            w = peak_vals[ii]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                m = k - j
                if j == m:
                    grad[j] += 2.0 * w * f[j]
                else:
                    grad[j] += 2.0 * w * f[m]

        # Normalize gradient to have zero mean over support
        support = f > 1e-12
        if support.sum() > 0:
            grad_mean = grad[support].mean()
            grad[support] -= grad_mean

        # Momentum update
        momentum = beta * momentum + grad
        f_new = f - lr * momentum
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            break
        f_new = f_new / sn

        obj_new = 2.0 * n * fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Restart with perturbation
                rng = np.random.default_rng(it * 17 + 3)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.005, 0.0)
                f = f / f.sum()
                lr = lr_init * 0.5
                momentum = np.zeros(n)
                no_improve = 0

        lr *= lr_decay
        if lr < 1e-8:
            break

        if verbose and it % 50 == 0:
            print(f"    GD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def pair_unbalance_descent(f, time_limit, verbose=False):
    """Coordinate descent that unbalances pairs at peak indices."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        peak = int(np.argmax(conv))
        js = max(0, peak - n + 1)
        je = min(n - 1, peak)

        improved = False
        # Try unbalancing pairs at the peak
        processed = set()
        for j in range(js, je + 1):
            m = peak - j
            if m < 0 or m >= n or m in processed:
                continue
            processed.add(j)
            processed.add(m)
            if j == m:
                continue

            sj = f[j] + f[m]
            if sj < 1e-12:
                continue

            cur_ratio = f[j] / sj
            # Try making more unbalanced
            for new_r in [0.0, 0.1, 0.3, 0.7, 0.9, 1.0, 0.05, 0.95, 0.2, 0.8]:
                if abs(new_r - cur_ratio) < 0.05:
                    continue
                trial = f.copy()
                trial[j] = sj * new_r
                trial[m] = sj * (1.0 - new_r)
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj_val = 2.0 * n * fast_convolve(trial).max()
                if obj_val < best_obj - 1e-13:
                    best_obj = obj_val
                    best_f = trial.copy()
                    f = trial
                    improved = True
                    break

        if improved:
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    Pair iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Try a second peak
            if no_improve > 20:
                conv2 = fast_convolve(f)
                peak2 = int(np.argmax(conv2))
                if peak2 != peak:
                    js2 = max(0, peak2 - n + 1)
                    je2 = min(n - 1, peak2)
                    for j in range(js2, je2 + 1):
                        m = peak2 - j
                        if m < 0 or m >= n or m <= j:
                            continue
                        sj = f[j] + f[m]
                        if sj < 1e-12:
                            continue
                        for new_r in [0.0, 0.2, 0.8, 1.0]:
                            trial = f.copy()
                            trial[j] = sj * new_r
                            trial[m] = sj * (1.0 - new_r)
                            trial = np.maximum(trial, 0.0)
                            ts = trial.sum()
                            if ts < 1e-15:
                                continue
                            trial = trial / ts
                            obj_val = 2.0 * n * fast_convolve(trial).max()
                            if obj_val < best_obj - 1e-13:
                                best_obj = obj_val
                                best_f = trial.copy()
                                f = trial
                                no_improve = 0
                                break
                        if no_improve == 0:
                            break

            if no_improve > 100:
                # Aggressive perturbation
                rng = np.random.default_rng(it * 31 + 7)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.01, 0.0)
                f = f / f.sum()
                no_improve = 0
                it = 0

    return best_f, best_obj


def spectral_method(f, time_limit, threshold_margin=0.05, verbose=False):
    """Patterson-CHEN style spectral method for reducing autocorrelation peaks."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 500:
        it += 1
        # Compute DFT
        F = np.fft.fft(f)
        # Periodic autocorrelation
        r = np.fft.ifft(np.abs(F) ** 2).real

        # Target: reduce peaks above threshold
        M = r.max()
        threshold = M * (1.0 - threshold_margin)

        # Reduce DFT magnitudes where periodic autocorrelation is high
        for k in range(n):
            if r[k % n] > threshold:
                # Scale down |F[k]| and |F[n-k]|
                reduction = np.sqrt(threshold / r[k % n]) if r[k % n] > 0 else 1.0
                reduction = max(0.5, min(1.0, reduction))
                F[k] *= reduction
                if k != 0 and k != n // 2:
                    F[n - k] *= reduction

        # Inverse FFT
        f_new = np.real(np.fft.ifft(F))
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            break
        f_new = f_new / sn

        obj_new = 2.0 * n * fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 20 == 0:
            print(f"    Spectral iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def lpfw_optimize(f0, time_limit, verbose=False, use_full_lp_every=50):
    """
    LP Frank-Wolfe with active set management.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    top_k_constraints = min(80, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Determine active constraints
        conv_cur = fast_convolve(f)
        M = conv_cur.max()
        active_indices = np.argpartition(conv_cur, -top_k_constraints)[-top_k_constraints:]
        active_indices = active_indices[np.argsort(conv_cur[active_indices])[::-1]]

        # Solve LP with active constraints
        g = solve_lp_active(f, M, n, active_indices, relax=m_relax)

        if g is None:
            # Try full LP
            if m_relax < 2.0:
                m_relax *= 1.2
            else:
                g = full_lp_solve(f, M * m_relax, n)
                if g is None:
                    # Perturbation
                    rng = np.random.default_rng(it * 23 + 5)
                    f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 0.0)
                    f = f / f.sum()
                    m_relax = 1.0
                    no_improve = 0
                    continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_relax *= 1.1
            continue
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-13:
            no_improve += 1
            if no_improve > 30:
                m_relax = min(m_relax * 1.05, 5.0)
            if no_improve > 150:
                # Structured perturbation
                rng = np.random.default_rng(it * 41 + 13)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 0.0)
                f = f / f.sum()
                m_relax = 1.0
                no_improve = 0
                it = max(0, it - use_full_lp_every)
            continue

        if diff > 1e-5:
            no_improve = 0
        elif diff > 1e-8:
            no_improve += 1
        else:
            no_improve += 5

        # Line search: try several alphas
        best_alpha = 0.0
        best_alpha_val = 2.0 * n * fast_convolve(f).max()
        alphas_to_try = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        if diff < 0.05:
            alphas_to_try = [0.3, 0.5, 0.7, 0.9, 1.0]

        for alpha in alphas_to_try:
            f_trial = f + alpha * (g - f)
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            v = 2.0 * n * fast_convolve(f_trial).max()
            if v < best_alpha_val:
                best_alpha_val = v
                best_alpha = alpha

        if best_alpha < 1e-8:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 3.0)
            continue

        f = f + best_alpha * (g - f)
        f = np.maximum(f, 0.0)
        sn = f.sum()
        if sn < 1e-15:
            break
        f = f / sn

        conv_new = fast_convolve(f)
        M_new = conv_new.max()
        current_obj = 2.0 * n * M_new

        if current_obj < best_obj - 1e-15:
            best_obj = current_obj
            best_f = f.copy()
            m_relax = 1.0
        else:
            no_improve += 1

        if verbose and it % 20 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, diff={diff:.2e}, relax={m_relax:.2f}")

        if no_improve > 300:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.25, center_frac=0.5):
    """Gaussian initial sequence."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exponential(n, decay=3.0, direction=1):
    """Asymmetric exponential decay."""
    if direction == 1:
        x = np.arange(n)
    else:
        x = n - 1 - np.arange(n)
    a = np.exp(-decay * x / n)
    a = a / a.sum()
    return a


def make_u_shape(n, sharpness=2.0):
    """U-shaped sequence (mass at edges)."""
    x = np.arange(n)
    t = 2.0 * x / (n - 1) - 1.0
    a = 1.0 + sharpness * t ** 2
    a = a / a.sum()
    return a


def make_offset_gaussian(n, sigma=0.2, offset=0.3):
    """Offset Gaussian (asymmetric center)."""
    x = np.arange(n) - n * (0.5 + offset)
    sigma_abs = n * sigma
    a = np.exp(-0.5 * (x / sigma_abs) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_peaks(n, pos1=0.2, pos2=0.8, width=0.15, weight1=0.6):
    """Two-peaked sequence."""
    x = np.arange(n)
    a = weight1 * np.exp(-0.5 * ((x - n * pos1) / (n * width)) ** 2) + \
        (1 - weight1) * np.exp(-0.5 * ((x - n * pos2) / (n * width)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining LP-Frank-Wolfe, gradient descent,
    pair unbalancing, and spectral methods with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_offset_gaussian(768, 0.2, 0.2)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")

    n_parent = len(parent)

    # === Phase 1: Diverse initialization (60s) ===
    print("=== Phase 1: Diverse Initializations ===")
    t_phase1_end = time.monotonic() + min(60.0, (dl - time.monotonic()) * 0.08)

    lengths_to_try = [512, 640, 768, 896, 1024]
    if n_parent not in lengths_to_try:
        lengths_to_try.append(n_parent)
    lengths_to_try = sorted(set(lengths_to_try))

    for n_try in lengths_to_try:
        if time.monotonic() > t_phase1_end:
            break
        # Various initializations
        inits = [
            ("gauss_c", make_gaussian(n_try, 0.25, 0.5)),
            ("gauss_off_l", make_offset_gaussian(n_try, 0.2, -0.15)),
            ("gauss_off_r", make_offset_gaussian(n_try, 0.2, 0.15)),
            ("gauss_off_r2", make_offset_gaussian(n_try, 0.25, 0.25)),
            ("exp_l", make_exponential(n_try, 3.0, 1)),
            ("exp_r", make_exponential(n_try, 3.0, -1)),
            ("exp_l2", make_exponential(n_try, 5.0, 1)),
            ("u_shape", make_u_shape(n_try, 2.0)),
            ("two_peak", make_two_peaks(n_try, 0.15, 0.85, 0.2, 0.5)),
        ]
        if n_try == n_parent:
            inits.insert(0, ("parent", parent.copy()))
        else:
            inits.insert(0, ("parent_interp", interpolate_sequence(parent, n_try)))

        for name, init in inits:
            obj = objective_normalized(init)
            best_seq, best_score = try_sequence(init, best_seq, best_score,
                                                f"init_{name}_n{n_try}")

    print(f"  Best after init: {best_score:.6f} (n={len(best_seq)})")

    # === Phase 2: Quick LP-FW on top candidates (200s) ===
    print("=== Phase 2: Quick LP-FW ===")
    t_phase2_end = time.monotonic() + min(200.0, (dl - time.monotonic()) * 0.25)

    # Collect top candidates by normalized objective
    candidates = []
    n_best = len(best_seq)
    seen_ns = set()
    for n_try in lengths_to_try:
        if n_try in seen_ns:
            continue
        if n_try == n_best:
            candidates.append((best_seq.copy(), objective_normalized(best_seq)))
        else:
            f_interp = interpolate_sequence(best_seq, n_try)
            candidates.append((f_interp, objective_normalized(f_interp)))
        seen_ns.add(n_try)

    candidates.sort(key=lambda x: x[1])
    candidates = candidates[:4]  # Top 4

    for ci, (cand, cand_obj) in enumerate(candidates):
        if time.monotonic() > t_phase2_end:
            break
        rem = t_phase2_end - time.monotonic()
        alloc = min(rem / (len(candidates) - ci), 80.0)
        if alloc < 20:
            alloc = rem
        print(f"  Quick LP-FW on candidate {ci} (n={len(cand)}): t={alloc:.0f}s")
        a_opt, obj = lpfw_optimize(cand, time_limit=alloc, verbose=False)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                            f"quick_lpfw_{ci}")

    # === Phase 3: Deep LP-FW on best (500s) ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Deep LP-FW (n={len(best_seq)}) ===")
        t_phase3 = min(500.0, (dl - time.monotonic()) * 0.45)
        print(f"  Allocating {t_phase3:.0f}s")
        a_opt, obj = lpfw_optimize(best_seq, time_limit=t_phase3, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "deep_lpfw")
        print(f"  After deep LP-FW: {best_score:.6f}")

    # === Phase 4: Gradient refinement (150s) ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Gradient Descent ===")
        t_phase4 = min(150.0, (dl - time.monotonic()) * 0.2)
        print(f"  Allocating {t_phase4:.0f}s")
        a_opt, obj = gradient_descent_on_peak(best_seq, time_limit=t_phase4,
                                              top_k=8, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "grad_desc")

    # === Phase 5: Pair unbalancing (100s) ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Pair Unbalancing ===")
        t_phase5 = min(100.0, (dl - time.monotonic()) * 0.15)
        print(f"  Allocating {t_phase5:.0f}s")
        a_opt, obj = pair_unbalance_descent(best_seq, time_limit=t_phase5, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "pair_desc")

    # === Phase 6: Spectral method (80s) ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Spectral Method ===")
        t_phase6 = min(80.0, (dl - time.monotonic()) * 0.1)
        print(f"  Allocating {t_phase6:.0f}s")
        a_opt, obj = spectral_method(best_seq, time_limit=t_phase6, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "spectral")

    # === Phase 7: Multi-length + perturbation (remaining) ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Multi-length + Perturbation ===")
        n_best = len(best_seq)

        # Try neighboring lengths
        for target_n in sorted(set([n_best - 64, n_best - 32, n_best + 32,
                                     n_best + 64, n_best + 128])):
            if target_n < 128 or target_n > 2048:
                continue
            rem = dl - time.monotonic() - 30
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            best_seq, best_score = try_sequence(f_interp, best_seq, best_score,
                                                f"interp_n{target_n}")
            alloc = min(rem * 0.25, 60.0)
            if alloc < 15:
                continue
            print(f"  LP-FW n={target_n}: t={alloc:.0f}s")
            a_opt, obj = lpfw_optimize(f_interp, time_limit=alloc, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                f"ml_lpfw_n{target_n}")

        # Perturbed restarts
        for p in range(3):
            rem = dl - time.monotonic() - 20
            if rem < 25:
                break
            alloc = min(rem / (3 - p), 70.0)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = np.maximum(best_seq + noise, 0.0)
                f_init = f_init / f_init.sum()
            elif p == 1:
                # Blend with uniform
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                # Reverse blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()

            print(f"  Perturbed LP-FW variant {p}: t={alloc:.0f}s")
            a_opt, obj = lpfw_optimize(f_init, time_limit=alloc, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                f"perturb_{p}")

    # === Phase 8: Final polish ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Polish ===")
        rem = dl - time.monotonic() - 10
        if rem > 20:
            # Short LP-FW
            alloc1 = min(rem * 0.4, 60.0)
            if alloc1 > 10:
                print(f"  Final LP-FW: t={alloc1:.0f}s")
                a_opt, obj = lpfw_optimize(best_seq, time_limit=alloc1)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_lpfw")

            # Short gradient
            rem2 = dl - time.monotonic() - 5
            if rem2 > 10:
                alloc2 = min(rem2 * 0.5, 30.0)
                if alloc2 > 5:
                    print(f"  Final GD: t={alloc2:.0f}s")
                    a_opt, obj = gradient_descent_on_peak(best_seq, time_limit=alloc2,
                                                          top_k=4)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                        "fin_gd")

            # Final pair descent
            rem3 = dl - time.monotonic() - 3
            if rem3 > 5:
                alloc3 = min(rem3, 20.0)
                if alloc3 > 3:
                    print(f"  Final Pair: t={alloc3:.0f}s")
                    a_opt, obj = pair_unbalance_descent(best_seq, time_limit=alloc3)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                        "fin_pair")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
