"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using adaptive subgradient descent
with soft-max peak weighting, backtracking line search, peak-pair coordinate
descent, and multi-length exploration. Key differences from prior approach:
- No LP solving (faster iterations)
- Soft-max subgradient with adaptive temperature
- Backtracking line search for reliable step sizes
- Peak-pair 1D line search for exact local optimization
- Parametric family initialization for diverse starting points

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def soft_max_subgradient(f, n, top_k=8, temperature=0.001):
    """Compute soft-max weighted subgradient of max convolution."""
    conv = fast_convolve(f)
    # Get top_k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peak_indices]
        # Soft-max weighting
        scaled = pv / temperature
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        kk = int(peak_indices[i])
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        idx = np.arange(js, je + 1)
        sg[idx] += weights[i] * f[kk - idx]
    
    return conv, sg


def backtracking_line_search(f, direction, n, max_obj, max_backtrack=20, c1=1e-4):
    """Backtracking line search for step size alpha."""
    alpha = 1.0
    f0 = f
    obj0 = max_obj
    for _ in range(max_backtrack):
        trial = f0 + alpha * direction
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        trial = trial / s
        obj = 2.0 * n * fast_convolve(trial).max()
        if obj <= obj0 - c1 * alpha * 0:  # Since we can't easily compute directional derivative, just check improvement
            return alpha, trial
        alpha *= 0.5
    return alpha, trial

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    temperature = 0.0005
    top_k = 8
    base_lr = 0.01
    lr = base_lr
    no_improve = 0
    patience = 100
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv, sg = soft_max_subgradient(f, n, top_k=top_k, temperature=temperature)
        
        # Normalize subgradient
        sg_norm = sg.sum()
        if sg_norm < 1e-15:
            break
        sg_dir = -sg / sg_norm  # Direction to decrease the peak
        
        # Project onto feasible direction (maintain sum = 1)
        # The direction should have mean 0 (or we adjust)
        sg_dir = sg_dir - sg_dir.mean()
        
        # Step
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            trial = f + lr_try * sg_dir
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = 2.0 * n * fast_convolve(trial).max()
            if obj < best_obj:
                best_obj = obj
                best_f = trial.copy()
                no_improve = 0
                f = trial
                break
            elif obj < 2.0 * n * conv.max():
                # Accept even if not better than best (progress)
                f = trial
                break
        else:
            no_improve += 1
        
        # Adaptive temperature
        temperature *= 1.005
        if temperature > 0.01:
            temperature = 0.001
        
        # Learn rate adjustment
        if no_improve > 0:
            lr *= 0.995
        if lr < 1e-6:
            lr = 1e-6
        
        # Restart
        if no_improve > patience:
            rng = np.random.default_rng(it * 17 + 3)
            noise_level = 0.003 + 0.002 * (no_improve // patience)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_level)
            conv = fast_convolve(f)
            temperature = 0.0005
            lr = base_lr
            no_improve = 0
            if 2.0 * n * conv.max() < best_obj:
                best_obj = 2.0 * n * conv.max()
                best_f = f.copy()
        
        if verbose and it % 100 == 0:
            print(f"      SubGrad iter {it}: obj={best_obj:.10f}, temp={temperature:.6f}, lr={lr:.6f}")
    
    return best_f, best_obj

def parametric_family(n, mu, sigma, skew):
    """Generate a parametric sequence: skewed Gaussian."""
    x = np.arange(n) / n  # [0, 1)
    # Skewed Gaussian
    z = (x - mu) / sigma
    if skew != 0:
        # Apply skew transformation
        z = z + skew * z * np.abs(z)
    a = np.exp(-0.5 * z * z)
    # Ensure non-negative and finite
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

            print(f"      SubGrad iter {it}: obj={best_obj:.10f}, temp={temperature:.6f}, lr={lr:.6f}")

    return best_f, best_obj


def peak_pair_cd(f0, time_limit, verbose=False):
    """Peak-pair coordinate descent targeting top contributing pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 60:
        it += 1
        conv = fast_convolve(f)
        improved = False

        top_k = min(6, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]

            order = np.argsort(products)[::-1]
            for oi in range(min(4, len(order))):
                j1 = int(idx[order[oi]])
                j2 = k - j1
                if j2 < 0 or j2 >= n or j2 == j1:
                    continue

                total = f[j1] + f[j2]
                for frac in [0.25, 0.35, 0.45, 0.55, 0.65, 0.75]:
                    trial = f.copy()
                    trial[j1] = total * frac
                    trial[j2] = total * (1.0 - frac)
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve > 8:
                k_max = int(np.argmax(conv))
                js = max(0, k_max - n + 1)
                je = min(n - 1, k_max)
                trial = f.copy()
                trial[js:je + 1] *= 0.94
                d = f.sum() - trial.sum()
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.sum() > 0:
                    w = f[non_peak]
                    ws = w.sum()
                    if ws > 1e-15:
                        trial[non_peak] += d * w / ws
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                elif no_improve > 30:
                    break
        else:
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"      PeakPair iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def multiplicative_cd(f0, time_limit, verbose=False):
    """Enhanced multiplicative coordinate descent with peak targeting."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temp = 0.0008
    base_lr = 0.005

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        k_top = min(12, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        pressure = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            pressure[idx] += w[i] * f[kk - idx]

        max_p = pressure.max()
        if max_p < 1e-15:
            break

        lr = base_lr * (1.0 + 0.2 * np.sin(0.05 * it))
        f = f * np.exp(-lr * pressure / max_p)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        obj = 2.0 * n * fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp *= 1.008
        if temp > 0.008:
            temp = 0.0008

        if no_improve > 60:
            rng = np.random.default_rng(it * 29 + 7)
            noise = rng.standard_normal(n) * (0.003 + 0.002 * (no_improve // 60))
            f = np.maximum(best_f + noise, 1e-15)
            f = f / f.sum()
            no_improve = 0
            temp = 0.0008

        if verbose and it % 200 == 0:
            print(f"      MulCD iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def parametric_family(n, mu, sigma, skew):
    """Generate a parametric skewed Gaussian sequence."""
    x = np.arange(n) / n
    z = (x - mu) / sigma
    if abs(skew) > 1e-6:
        z = z + skew * 0.5 * (z * np.abs(z) - np.abs(z))
    a = np.exp(-0.5 * z * z)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def bimodal_family(n, sep, width, asym):
    """Generate a bimodal sequence."""
    x = np.arange(n) / n
    center1 = 0.5 - sep
    center2 = 0.5 + sep
    s1 = 1.0
    s2 = asym
    a = s1 * np.exp(-0.5 * ((x - center1) / width) ** 2) + \
        s2 * np.exp(-0.5 * ((x - center2) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def parametric_search(time_limit, verbose=False):
    """Grid search over parametric families at multiple n values."""
    best_f = None
    best_obj = float('inf')
    t0 = time.monotonic()

    n_values = [384, 512, 640, 768, 800, 896, 1024]

    for n in n_values:
        if time.monotonic() - t0 > time_limit - 20:
            break
        # Unimodal search
        for mu in [0.35, 0.4, 0.45, 0.5, 0.55, 0.6]:
            for sigma in [0.18, 0.22, 0.26, 0.30, 0.34]:
                for skew in [-0.4, 0.0, 0.4]:
                    a = parametric_family(n, mu, sigma, skew)
                    obj = obj_normalized(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_f = a.copy()

        # Bimodal search
        for sep in [0.08, 0.12, 0.16, 0.20]:
            for width in [0.08, 0.10, 0.12, 0.15]:
                for asym in [0.5, 0.7, 1.0, 1.3]:
                    a = bimodal_family(n, sep, width, asym)
                    obj = obj_normalized(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_f = a.copy()

        if verbose:
            print(f"      Parametric n={n}: best={best_obj:.8f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining parametric search, adaptive subgradient,
    peak-pair coordinate descent, multiplicative CD, and multi-length exploration.
    Uses bimodal shape exploration and soft-max subgradients to escape local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Parametric family search ===
    print("\n=== Phase 1: Parametric Search ===")
    t1 = min(150, (deadline - time.monotonic()) * 0.12)
    if t1 > 15:
        f_param, obj_param = parametric_search(t1, verbose=True)
        if f_param is not None:
            best_seq, best_score = try_sequence(f_param, best_seq, best_score, "parametric")

    # Choose best starting point
    if best_seq is not None:
        start_f = best_seq / best_seq.sum()
    else:
        start_f = parametric_family(768, 0.5, 0.25, 0.0)
        best_seq, best_score = try_sequence(start_f, best_seq, best_score, "default")
        start_f = best_seq / best_seq.sum()

    n_cur = len(start_f)
    print(f"  Starting: n={n_cur}, obj={obj_normalized(start_f):.8f}")

    # === Phase 2: Subgradient optimization ===
    print(f"\n=== Phase 2: Subgradient (n={n_cur}) ===")
    rem = deadline - time.monotonic()
    t2 = min(500, rem * 0.30)
    if t2 > 30:
        print(f"  Time budget: {t2:.0f}s")
        f_opt, obj = subgradient_optimize(start_f, time_limit=t2, verbose=True)
        best_seq, best_score = try_sequence(f_opt, best_seq, best_score, "subgrad_main")

    # === Phase 3: Peak-pair CD ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Peak-Pair CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t3 = min(300, rem * 0.18)
        if t3 > 20:
            print(f"  Time budget: {t3:.0f}s")
            f_opt, obj = peak_pair_cd(best_seq, time_limit=t3, verbose=True)
            best_seq, best_score = try_sequence(f_opt, best_seq, best_score, "peakpair")

    # === Phase 4: Multiplicative CD ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Multiplicative CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t4 = min(300, rem * 0.18)
        if t4 > 20:
            print(f"  Time budget: {t4:.0f}s")
            f_opt, obj = multiplicative_cd(best_seq, time_limit=t4, verbose=True)
            best_seq, best_score = try_sequence(f_opt, best_seq, best_score, "mulcd")

    # === Phase 5: Multi-length exploration ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Multi-Length ===")
        n_best = len(best_seq)
        for delta in [-32, -16, -8, 8, 16, 32, 64]:
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            rem = deadline - time.monotonic()
            t5 = min(80, rem * 0.12)
            if t5 < 15:
                break
            f_interp = interpolate_sequence(best_seq, n_try)
            best_seq, best_score = try_sequence(f_interp, best_seq, best_score, f"interp_n{n_try}")

            t5b = min(50, rem * 0.08)
            if t5b > 10:
                f_opt, obj = subgradient_optimize(f_interp, time_limit=t5b)
                best_seq, best_score = try_sequence(f_opt, best_seq, best_score, f"subgrad_n{n_try}")

    # === Phase 6: Bimodal exploration ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Bimodal Exploration ===")
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 30
        if rem > 40:
            t6 = min(80, rem * 0.15)
            # Try bimodal starting points and optimize
            for sep in [0.10, 0.15]:
                for width in [0.10, 0.13]:
                    a_bi = bimodal_family(n_best, sep, width, 1.0)
                    # Blend with current best
                    blend = 0.3 * a_bi + 0.7 * best_seq
                    blend = blend / blend.sum()
                    best_seq, best_score = try_sequence(blend, best_seq, best_score, f"bimodal_{sep}_{width}")
                    t6b = min(30, (deadline - time.monotonic()) - 30)
                    if t6b > 15:
                        f_opt, obj = subgradient_optimize(blend, time_limit=t6b)
                        best_seq, best_score = try_sequence(f_opt, best_seq, best_score, f"bi_opt_{sep}_{width}")

    # === Phase 7: Final refinement cycle ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Final Refinement ===")
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            n_fin = len(best_seq)
            t_each = rem / 3.0
            methods = [
                ("peakpair_fin", peak_pair_cd),
                ("subgrad_fin", subgradient_optimize),
                ("mulcd_fin", multiplicative_cd),
            ]
            for m_name, m_fn in methods:
                rem = deadline - time.monotonic() - 10
                if rem < 10:
                    break
                t_alloc = min(t_each, rem)
                if t_alloc < 5:
                    break
                print(f"  {m_name}: t={t_alloc:.0f}s")
                f_opt, obj = m_fn(best_seq, time_limit=t_alloc)
                best_seq, best_score = try_sequence(f_opt, best_seq, best_score, m_name)

    # === Phase 8: Random restarts ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Random Restarts ===")
        rem = deadline - time.monotonic() - 10
        n_best = len(best_seq)
        restart_count = 0
        while rem > 25 and restart_count < 6:
            t_alloc = min(40, rem * 0.25)
            if t_alloc < 8:
                break
            restart_count += 1
            if restart_count % 4 == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            elif restart_count % 4 == 2:
                frac = 0.03 + 0.02 * restart_count
                f_init = (1.0 - frac) * best_seq + frac * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif restart_count % 4 == 3:
                f_init = 0.75 * best_seq + 0.25 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Structural change: dampen center, boost edges
                f_init = best_seq.copy()
                center = n_best // 2
                w = max(10, n_best // 8)
                f_init[center - w:center + w] *= 0.85
                d = best_seq.sum() - f_init.sum()
                if d > 0:
                    edges = np.concatenate([f_init[:w], f_init[-w:]])
                    es = edges.sum()
                    if es > 1e-15:
                        f_init[:w] += d * 0.5 * f_init[:w] / (f_init[:w].sum() + 1e-15)
                        f_init[-w:] += d * 0.5 * f_init[-w:] / (f_init[-w:].sum() + 1e-15)
                    else:
                        f_init[:w] += d * 0.5 / w
                        f_init[-w:] += d * 0.5 / w
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()

            print(f"  Restart {restart_count}: t={t_alloc:.0f}s, obj_init={obj_normalized(f_init):.8f}")
            f_opt, obj = subgradient_optimize(f_init, time_limit=t_alloc)
            best_seq, best_score = try_sequence(f_opt, best_seq, best_score, f"restart_{restart_count}")
            rem = deadline - time.monotonic() - 10

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
