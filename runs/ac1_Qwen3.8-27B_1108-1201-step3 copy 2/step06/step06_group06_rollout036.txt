"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a peak-chasing gradient method
with adaptive temperature, multi-resolution initialization, and LP-based refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def optimize_smooth(f0, time_limit, tau_init=None, lr_init=0.003, top_k_weight=5, verbose=False):
    """
    Optimize using log-sum-exp smooth approximation with Adam.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.05 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv, M = compute_peak_gradient(f, top_k=top_k_weight, tau=tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature decay
        tau *= 0.9995
        if tau < 5e-6:
            tau = 5e-6

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.7
        if no_improve > 500:
            # Restart
            rng = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.001 + 0.002 * min(no_improve / 1000.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.5
            tau = 0.04 * fast_convolve(f).max()

    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP direction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.3
            if m_exp > 10.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.2
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.15, 8.0)
            if no_improve > 100:
                break
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:

def peak_redundancy(f, n_peaks=3):
    """
    Identify peak-contributing element pairs and suggest redistribution.
    Returns a modified sequence with mass moved away from peak contributors.
    """
    f = f.copy()
    n = len(f)
    conv = fast_convolve(f)
    
    # Get top peaks
    peak_indices = np.argsort(conv)[-n_peaks:]
    
    # For each peak, find the dominant contributing pair
    # and reduce those elements
    for pk in sorted(peak_indices, reverse=True):
        pk = int(pk)
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        if je < js:
            continue
        
        idx = np.arange(js, je + 1)
        products = f[idx] * f[pk - idx]
        j_max = idx[np.argmax(products)]
        
        # Reduce f[j_max] and redistribute
        old_val = f[j_max]
        new_val = old_val * 0.7  # Reduce by 30%
        diff = old_val - new_val
        
        # Redistribute to elements that don't contribute to top peaks
        contrib = np.zeros(n, dtype=bool)
        for pk2 in peak_indices[:n_peaks]:
            pk2 = int(pk2)
            js2 = max(0, pk2 - n + 1)
            je2 = min(n - 1, pk2)
            contrib[js2:je2+1] = True
        contrib[j_max] = True
        
        non_contrib = ~contrib
        if non_contrib.sum() > 0 and f[non_contrib].sum() > 1e-15:
            f[non_contrib] += diff * f[non_contrib] / f[non_contrib].sum()
            f[j_max] = new_val
        else:
            # Fallback: redistribute uniformly
            f[j_max] = new_val
            f[:] += diff / n
    
    f = np.maximum(f, 0)
    f /= f.sum()
    return f


def multi_scale_optimize(f0, time_limit, verbose=False):
    """
    Multi-scale: downsample, optimize, upsample, optimize.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f = f / f.sum()
    
    scales = [max(32, n // 8), max(64, n // 4), max(128, n // 2), n]
    t0 = time.monotonic()
    
    for target_n in scales:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        if target_n >= n:
            # Full resolution
            a, obj = optimize_smooth(f, time_limit=rem * 0.8)
            return a, obj
        else:
            # Downsample
            old_x = np.linspace(0, 1, len(f))
            new_x = np.linspace(0, 1, target_n)
            f_down = np.interp(new_x, old_x, f)
            f_down = np.maximum(f_down, 0)
            f_down /= f_down.sum()
            
            # Optimize at lower resolution
            t_sub = rem * 0.4
            a_down, obj = optimize_smooth(f_down, time_limit=t_sub)
            
            # Upsample back
            # Keep the current resolution result (don't downsample back up)
            # Actually, use the lower-res result to guide: blend
            f_up = np.interp(old_x, new_x, a_down)
            f_up = np.maximum(f_up, 0)
            f_up /= f_up.sum()
            
            # Blend with current
            f = 0.5 * f + 0.5 * f_up
            f = project_simplex(f)
    
    return f, fast_convolve(f).max()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smooth gradient optimization
with top-k peak focusing, LP-Frank-Wolfe refinement, peak-targeted redistribution,
and multi-scale exploration. Key innovations: adaptive temperature with dual-rate

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, tau):
    """
    Compute gradient of log-sum-exp smooth approximation of max(conv(f,f)).
    Uses FFT-based computation. Returns (gradient, conv, max_conv).
    """
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()

    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # h[j] = sum_{l=0}^{n-1} w[j+l] * f[l]
    # Computed via convolution with reversed w
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    grad = (2.0 / tau) * h
    return grad, conv, M


def compute_topk_gradient(f, top_k, tau):
    """
    Compute gradient focusing on top-k peaks with softmax weighting.
    Returns (gradient, conv, max_conv).
    """
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()

    # Focus on top-k: use a sharper softmax
    effective_tau = tau * max(1.0, n / top_k)
    scaled = conv / effective_tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    grad = (2.0 / effective_tau) * h
    return grad, conv, M


def smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.003, top_k=10, verbose=False):
    """
    Smooth gradient optimization with adaptive temperature and restarts.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    tau_decay = 0.9996
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        if top_k <= 10:
            grad, conv, M = compute_gradient(f, tau)
        else:
            grad, conv, M = compute_topk_gradient(f, top_k, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature: slow decay, with occasional bumps
        tau *= tau_decay
        if tau < 3e-6:
            tau = 3e-6

        # Learning rate schedule
        if no_improve > 300:
            lr *= 0.8
        if no_improve > 800:
            # Strategic restart: perturb towards filling valleys
            rng = np.random.default_rng(t * 7 + restart_count * 101)
            restart_count += 1
            noise_scale = 0.001 + 0.003 * min(restart_count / 5.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * (0.5 + 0.1 * restart_count)
            tau = 0.04 * fast_convolve(f).max()
            # Speed up decay after restarts
            tau_decay = min(0.999, 0.9996 - 0.0001 * restart_count)

        if verbose and t % 2000 == 0:
            print(f"      iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}, restarts={restart_count}")

    return best_f, best_obj


def lp_solve_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP-based direction finding and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = lp_solve_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.4
            if m_exp > 15.0:
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.2
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                m_exp = min(m_exp * 1.15, 12.0)
            if no_improve > 150:
                break
            continue
        else:
            no_improve = 0

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            M = best_val
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 5 == 0:
                    print(f"      FW iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, best_obj


def peak_rebalance(f, n_peaks=4, strength=0.15):
    """
    Redistribute mass from peak-contributing elements to valley-filling elements.
    """
    f = f.copy()
    n = len(f)
    conv = fast_convolve(f)

    # Get top peaks
    peak_indices = np.argsort(conv)[-n_peaks:]

    # Identify elements that heavily contribute to peaks
    contrib_weight = np.zeros(n)
    for pk in peak_indices:
        pk = int(pk)
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        products = f[idx] * f[pk - idx]
        total = products.sum()
        if total > 1e-15:
            contrib_weight[idx] += products / total

    # Find valley positions (low convolution)
    valley_indices = np.argsort(conv)[:n_peaks]

    # Elements that contribute to valleys (should be increased)
    valley_weight = np.zeros(n)
    for vk in valley_indices:
        vk = int(vk)
        js = max(0, vk - n + 1)
        je = min(n - 1, vk)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        valley_weight[idx] += 1.0

    # Redistribution: move mass from high-contrib to high-valley elements
    high_contrib = contrib_weight > np.percentile(contrib_weight, 70)
    high_valley = valley_weight > 0

    # Combined: reduce high-contrib, increase high-valley (if not also high-contrib)
    reduce_mask = high_contrib & ~high_valley
    increase_mask = high_valley & ~high_contrib

    if reduce_mask.sum() > 0 and increase_mask.sum() > 0:
        reduce_total = f[reduce_mask].sum() * strength
        if reduce_total > 1e-10:
            f[reduce_mask] -= reduce_total * f[reduce_mask] / f[reduce_mask].sum()
            f[increase_mask] += reduce_total * f[increase_mask] / f[increase_mask].sum()

    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f


def simulated_annealing_step(f, conv_at_best, best_obj, temperature, rng):
    """
    Propose a random perturbation and accept based on SA criterion.
    """
    n = len(f)
    scale = 0.005 + 0.01 * temperature
    f_trial = project_simplex(f + rng.standard_normal(n) * scale)
    conv_trial = fast_convolve(f_trial)
    obj_trial = conv_trial.max()

    delta = obj_trial - best_obj
    if delta < 0:
        return f_trial, obj_trial, True
    else:
        # Accept with probability exp(-delta/temperature)
        if temperature > 1e-8:
            if rng.random() < np.exp(-delta / (temperature * best_obj)):
                return f_trial, obj_trial, True
    return f, best_obj, False


def diverse_init(n):
    """Generate diverse initializations for length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussians (various centers and widths)
    for center in [0.25, 0.35, 0.45, 0.5]:
        for sigma in [0.12, 0.18, 0.25, 0.35]:
            x = np.arange(n) - n * center
            a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                inits.append(a / s)

    # Cosine window variants
    for power in [0.5, 1.0, 2.0]:
        a = np.cos(np.pi * np.arange(n) / (2.0 * n)) ** (2 * power)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)

    # Edge-concentrated (U-shape)
    for depth in [0.3, 0.5, 0.7]:
        x = np.linspace(-1, 1, n)
        a = 1.0 - depth * np.exp(-x ** 2 / 0.3)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)

    # Two-hump
    for gap in [0.2, 0.3, 0.4]:
        x = np.arange(n)
        a = np.exp(-((x - n * (0.5 - gap / 2)) / (n * 0.12)) ** 2) + \
            np.exp(-((x - n * (0.5 + gap / 2)) / (n * 0.12)) ** 2)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)

    # Exponential decay
    for rate in [0.5, 1.0, 2.0]:
        a = np.exp(-np.arange(n) * rate / n)
        inits.append(a / a.sum())

    # Triangle
    a = 1.0 - 2.0 * np.abs(np.arange(n) / n - 0.5)
    inits.append(a / a.sum())

    # Flat-top
    half = n * 0.3
    x = np.arange(n) - n / 2.0
    a = np.where(np.abs(x) < half, 1.0, 0.0)
    # Smooth edges
    a = np.maximum(0, 1.0 - (np.abs(x) / (half * 1.5)) ** 8)
    s = a.sum()
    if s > 1e-10:
        inits.append(a / s)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Diverse initialization with scoring
    2. Smooth gradient optimization (main workhorse)
    3. LP-Frank-Wolfe refinement
    4. Peak-targeted rebalancing
    5. Simulated annealing exploration
    6. Final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Diverse Initialization ===
    print("=== Phase 1: Diverse Initialization ===")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"  Parent loaded: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = diverse_init(768)[2]
        try_seq(parent, "fallback_init")

    # Score diverse initializations
    work_n = len(parent)
    candidates = []

    # From parent
    candidates.append((best_score if best_score < float('inf') else 99.0, best_seq if best_seq is not None else parent))

    # Diverse inits at parent's length
    inits = diverse_init(work_n)
    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 1600:
            break
        conv = fast_convolve(a_init)
        approx_score = 2 * work_n * conv.max()
        candidates.append((approx_score, a_init))

    # Also try at other lengths
    for target_n in [1024, 1280, 1536, 512]:
        if time.monotonic() > deadline - 1550:
            break
        inits_n = diverse_init(target_n)
        for a_init in inits_n[:5]:
            conv = fast_convolve(a_init)
            approx_score = 2 * target_n * conv.max()
            candidates.append((approx_score, a_init))

    # Sort by approximate score and keep top 5
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in candidates[:5]]

    print(f"  Top candidate scores: {[f'{c[0]:.4f}' for c in candidates[:5]]}")

    # === Phase 2: Smooth Gradient Optimization ===
    print("\n=== Phase 2: Smooth Gradient Optimization ===")

    # Optimize the best candidate first (longest time)
    opt_budget = min(500, deadline - time.monotonic() - 600)
    if opt_budget > 60:
        print(f"  Optimizing best candidate (n={len(top_candidates[0])}), time={opt_budget:.0f}s")
        a_opt, obj = smooth_optimize(top_candidates[0], time_limit=opt_budget,
                                     lr_init=0.004, top_k=10, verbose=True)
        try_seq(a_opt, "smooth_best")

    # Optimize second-best candidate
    if time.monotonic() < deadline - 500 and len(top_candidates) > 1:
        rem = min(200, deadline - time.monotonic() - 500)
        if rem > 40:
            print(f"  Optimizing 2nd candidate (n={len(top_candidates[1])}), time={rem:.0f}s")
            a_opt, obj = smooth_optimize(top_candidates[1], time_limit=rem,
                                         lr_init=0.003, top_k=15)
            try_seq(a_opt, "smooth_2nd")

    # Second pass on best with lower temperature
    if time.monotonic() < deadline - 400:
        rem = min(150, deadline - time.monotonic() - 400)
        if rem > 30:
            tau_fine = 0.025 * fast_convolve(best_seq).max()
            a_opt, obj = smooth_optimize(best_seq, time_limit=rem,
                                         tau_init=tau_fine, lr_init=0.002, top_k=5)
            try_seq(a_opt, "smooth_fine")

    # === Phase 3: LP-Frank-Wolfe ===
    print("\n=== Phase 3: LP-Frank-Wolfe ===")
    if best_seq is not None:
        t3 = min(350, deadline - time.monotonic() - 250)
        if t3 > 50:
            print(f"  Time: {t3:.0f}s")
            a_opt, obj = fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lfw_main")

            # Perturbed restart
            if time.monotonic() < deadline - 200:
                rem = min(120, deadline - time.monotonic() - 200)
                if rem > 30:
                    perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.005)
                    a_opt, obj = fw_optimize(perturbed, time_limit=rem)
                    try_seq(a_opt, "lfw_pert")

    # === Phase 4: Peak Rebalancing ===
    print("\n=== Phase 4: Peak Rebalancing ===")
    if best_seq is not None:
        t4 = min(120, deadline - time.monotonic() - 100)
        if t4 > 20:
            f = best_seq.copy()
            for it in range(30):
                if time.monotonic() > deadline - 100:
                    break
                for nk in [3, 5, 7]:
                    for strength in [0.1, 0.2, 0.3]:
                        f_trial = peak_rebalance(f, n_peaks=nk, strength=strength)
                        sc = try_seq(f_trial, f"rebal_k{nk}_s{strength}")
                        if sc < best_score:
                            f = f_trial
                            break
                    if sc < best_score:
                        break
                else:
                    # Try SA step
                    temperature = 0.01 * (1 - it / 30.0)
                    f_trial, obj_trial, accepted = simulated_annealing_step(
                        f, fast_convolve(f), fast_convolve(f).max(), temperature, rng)
                    if accepted:
                        f = f_trial
                        try_seq(f, f"sa_{it}")
                    else:
                        break

    # === Phase 5: Multi-length Exploration ===
    print("\n=== Phase 5: Multi-length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [len(base) + 128, len(base) - 128, len(base) + 256]:
            if target_n < 64 or target_n > 4096:
                continue
            if time.monotonic() > deadline - 60:
                break
            old_x = np.linspace(0, 1, len(base))
            new_x = np.linspace(0, 1, target_n)
            f_interp = np.interp(new_x, old_x, base)
            f_interp = np.maximum(f_interp, 0)
            f_interp /= f_interp.sum()
            try_seq(f_interp, f"interp_n{target_n}")

            # Quick optimization on interpolated
            rem = min(40, deadline - time.monotonic() - 40)
            if rem > 15:
                a_opt, obj = smooth_optimize(f_interp, time_limit=rem, lr_init=0.003)
                try_seq(a_opt, f"smooth_n{target_n}")

    # === Phase 6: Final Polish ===
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        t6 = min(80, deadline - time.monotonic() - 10)
        if t6 > 15:
            tau_final = 0.015 * fast_convolve(best_seq).max()
            a_opt, obj = smooth_optimize(best_seq, time_limit=t6 * 0.6,
                                         tau_init=tau_final, lr_init=0.001, top_k=3)
            try_seq(a_opt, "polish_smooth")

        t6b = min(30, deadline - time.monotonic() - 5)
        if t6b > 10:
            a_opt, obj = fw_optimize(best_seq, time_limit=t6b)
            try_seq(a_opt, "polish_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
