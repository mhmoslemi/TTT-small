"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using accelerated projected
subgradient (FISTA) with peak condensing, LP-Frank-Wolfe polishing,
targeted coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_peak(f, k, n):
    """Compute subgradient of (f*f)_k with respect to f."""
    g = np.zeros(n)
    idx_start = max(0, k - n + 1)
    idx_end = min(n - 1, k)
    for i in range(idx_start, idx_end + 1):
        g[i] = 2.0 * f[k - i]
    return g


def condensed_subgradient(f, num_peaks=5, temperature=0.01):
    """Weighted average of subgradients from top peaks."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    
    # Get top peaks
    k = min(num_peaks, len(conv))
    peak_vals, peak_indices = np.sort(conv)[-k:][::-1], np.argsort(conv)[-k:][::-1]
    
    # Weight by softmax of peak values
    weights = np.exp((peak_vals - M) / (M * temperature + 1e-15))
    weights /= weights.sum()
    
    # Weighted average of subgradients
    g = np.zeros(n)
    for w, pi in zip(weights, peak_indices):
        if w < 1e-10:
            continue
        pi = int(pi)
        idx_start = max(0, pi - n + 1)
        idx_end = min(n - 1, pi)
        for i in range(idx_start, idx_end + 1):
            g[i] += w * 2.0 * f[pi - i]
    
    return g, M, conv


def fista_optimize(f0, time_limit, verbose=False, num_peaks=5, temperature=0.01):

    t = 1.0
    alpha = 1.0  # initial step size
    v = np.zeros(n)  # momentum velocity
    beta = 0.9  # momentum coefficient

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    alpha_history = []

    while time.monotonic() - t0 < time_limit:

        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()

    """LP-based Frank-Wolfe for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: max sum(g) s.t. (f*g)_k <= M*m_exp, g >= 0
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            A_ub[k_start:k_end + 1, j] = f[:k_end - k_start + 1]

        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5})
        
        if not res.success or res.x is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                break
            continue
        
        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            m_exp *= 1.05
            if m_exp > 3.0:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = 2.0 * n * fast_convolve(f).max()
        for a in [0.1, 0.3, 0.5, 0.7, 0.9]:
            ft = f + a * (g - f)
            ft = np.maximum(ft, 0.0)
            ft_sum = ft.sum()
            if ft_sum < 1e-15:

            val = 2.0 * n * fast_convolve(ft).max()

                best_alpha = a

            continue
        
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)

        f = f_new

            best_f = f.copy()

                new_val = old_val * mult

                trial[j_max] = new_val
                diff = old_val - new_val
                
                if abs(diff) > 1e-15:
                    # Redistribute to non-peak contributors
                    mask = np.ones(n, dtype=bool)
                    mask[js:je + 1] = False
                    if mask.sum() > 0:
                        w = trial[mask]
                        ws = w.sum()
                        if ws > 1e-15:
                            trial[mask] += diff * w / ws

The objective function (for normalized f) is `Phi(f) = max_k (f*f)_k`. This is Lipschitz with constant L = 2*max(f) (roughly). For f on the simplex, max(f) <= 1, so L <= 2.

The step size for subgradient method is typically `alpha_k = c / sqrt(k)` for non-smooth, or `alpha_k = 1/(L*k)` for smooth (with acceleration).

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using accelerated projected
subgradient (FISTA) with peak condensing, LP-Frank-Wolfe polishing,
targeted coordinate descent, and multi-length exploration.
Key improvements: O(n log n) subgradient via FFT, Nesterov momentum,
adaptive step size, multi-peak soft condensing, diverse initialization,
and iterative restarts to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_condensed_grad(y, num_peaks, temperature, n):
    """Compute condensed subgradient from top peaks of conv(y,y)."""
    conv_y = fast_convolve(y)
    M_y = conv_y.max()
    k_count = min(num_peaks, len(conv_y))
    peak_indices = np.argsort(conv_y)[-k_count:]
    peak_vals = conv_y[peak_indices]
    w = np.exp((peak_vals - M_y) / (M_y * temperature + 1e-15))
    w /= w.sum()
    g = np.zeros(n)
    for wi, pi in zip(w, peak_indices):
        if wi < 1e-10:
            continue
        pi = int(pi)
        low = max(0, pi - n + 1)
        high = min(n - 1, pi)
        seg = y[pi - high:pi - low + 1]
        g[low:high + 1] += wi * 2.0 * seg
    return g, M_y, conv_y


def fista_optimize(f0, time_limit, verbose=False, num_peaks=5, temperature=0.05):
    """FISTA accelerated projected subgradient with peak condensing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    y = f.copy()
    t = 1.0
    alpha = 0.3
    no_improve = 0
    iter_count = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        g, M_y, _ = compute_condensed_grad(y, num_peaks, temperature, n)

        f_new = project_simplex(y - alpha * g)
        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            alpha = min(alpha * 1.005, 2.0)
        else:
            no_improve += 1
            alpha = max(alpha * 0.92, 0.005)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = f_new + momentum * (f_new - f)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        f = f_new
        t = t_new

        if no_improve > 60:
            f = best_f.copy()
            y = f.copy()
            t = 1.0
            alpha = 0.2
            no_improve = 0
            rng = np.random.default_rng(iter_count * 17 + 3)
            scale = 0.0005 + 0.0003 * min(no_improve, 20)
            f = project_simplex(f + rng.standard_normal(n) * scale)
            y = f.copy()

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={iter_count}, obj={best_obj:.10f}, alpha={alpha:.4f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lp_fw_polish(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    m_exp = 1.0

    # Prebuild A_ub template (structure doesn't change, only values do)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Update A_ub (f changed)
        for j in range(n):
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            length = k_end - k_start + 1
            A_ub[k_start:k_end + 1, j] = f[:length]

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 10})

        if not res.success or res.x is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                break
            continue

        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            m_exp *= 1.05
            if m_exp > 3.0:
                break
            continue
        if diff > 1e-5:
            m_exp = 1.0

        # Coarse + fine line search
        best_alpha = 0.0
        best_val = 2.0 * n * fast_convolve(f).max()
        for a in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + a * (g - f)
            ft = np.maximum(ft, 0.0)
            ft_sum = ft.sum()
            if ft_sum < 1e-15:
                continue
            ft = ft / ft_sum
            val = 2.0 * n * fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = a

        if best_alpha < 1e-8:
            m_exp *= 1.1
            if m_exp > 3.0:
                break
            continue

        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        f = f_new
        m_exp = 1.0

        if best_val < best_obj:
            best_obj = best_val
            best_f = f.copy()

        if verbose and it % 3 == 0:
            print(f"    LP-FW it={it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coord_descent_peak(f0, time_limit, verbose=False):
    """Targeted coordinate descent on peak contributors."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        it += 1
        conv = fast_convolve(f)
        improved = False

        top_k = min(10, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je - js + 1 < 2:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            order = np.argsort(products)[::-1]

            for oi in range(min(5, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.6, 0.75, 0.85, 0.95, 1.05, 1.15, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue
                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        mask = np.ones(n, dtype=bool)
                        mask[js:je + 1] = False
                        if mask.sum() > 0:
                            w = trial[mask]
                            ws = w.sum()
                            if ws > 1e-15:
                                trial[mask] += d * w / ws
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Aggressive: scale down all peak region
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = f.copy()
                trial[js:je + 1] *= 0.93
                d = f.sum() - trial.sum()
                mc = np.ones(n, dtype=bool)
                mc[js:je + 1] = False
                if mc.sum() > 0:
                    nc_w = f[mc]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[mc] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"    CD it={it}: obj={best_obj:.10f}, no_improve={no_improve}")

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tukey(n, r=0.3):
    """Tukey (tapered cosine) window."""
    a = np.ones(n)
    m = int(n * r / 2)
    if m > 0:
        for i in range(m):
            taper = 0.5 * (1 + np.cos(np.pi * (2 * i / m - 1)))
            a[i] = taper
            a[n - 1 - i] = taper
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            t_alloc = min(rem / (3 - p + 0.5), 150)

                break
            if p == 0:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            elif p == 1:
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                # Swap symmetric halves
                f_init = np.zeros(n_best)
                f_init[:n_best//2] = best_seq[n_best//2:]
                f_init[n_best//2:] = best_seq[:n_best//2]
                f_init = f_init / f_init.sum()
            print(f"  Restart {p}: t={t_alloc:.0f}s")
            a_opt, obj = fista_optimize(f_init, time_limit=t_alloc, verbose=False,
                                        num_peaks=5, temperature=0.05)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"fista_r{p}")
            rem = deadline - time.monotonic()

    # === Phase 5: LP-FW polishing ===
    if best_seq is not None:
        print(f"\n=== Phase 5: LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.15, 180)
        if t_lp > 15:
            print(f"  Budget: {t_lp:.0f}s")
            a_opt, obj = lp_fw_polish(best_seq, time_limit=t_lp, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw")

    # === Phase 6: Coordinate descent ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Coord Descent (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.12, 120)
        if t_cd > 10:
            print(f"  Budget: {t_cd:.0f}s")
            a_opt, obj = coord_descent_peak(best_seq, time_limit=t_cd, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "coord_cd")

    # === Phase 7: Multi-length exploration ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Multi-length ===")
        base = best_seq
        for target_n in [768, 832, 896, 960, 1024]:
            if len(base) == target_n:
                continue
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            f_interp = interpolate_sequence(base, target_n)
            best_seq, best_score = try_sequence(f_interp, best_seq, best_score, f"interp_n{target_n}")
            t_sub = min(rem * 0.15, 60)
            if t_sub > 10:
                print(f"  FISTA n={target_n}: t={t_sub:.0f}s")
                a_opt, obj = fista_optimize(f_interp, time_limit=t_sub, verbose=False,
                                            num_peaks=5, temperature=0.05)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"fista_n{target_n}")

    # === Phase 8: Final combined polish ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Polish ===")
        rem = deadline - time.monotonic()
        if rem > 30:
            # FISTA with very low temperature (near-exact peak targeting)
            t1 = min(rem * 0.35, 80)
            if t1 > 10:
                print(f"  FISTA final: t={t1:.0f}s")
                a_opt, obj = fista_optimize(best_seq, time_limit=t1, verbose=False,
                                            num_peaks=2, temperature=0.01)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_fista")

            rem = deadline - time.monotonic()
            if rem > 20:
                t2 = min(rem * 0.3, 50)
                if t2 > 8:
                    print(f"  CD final: t={t2:.0f}s")
                    a_opt, obj = coord_descent_peak(best_seq, time_limit=t2, verbose=False)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_cd")

            rem = deadline - time.monotonic()
            if rem > 15:
                t3 = min(rem * 0.3, 40)
                if t3 > 8:
                    print(f"  LP-FW final: t={t3:.0f}s")
                    a_opt, obj = lp_fw_polish(best_seq, time_limit=t3, verbose=False)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_lpfw")

            # One more CD pass
            rem = deadline - time.monotonic()
            if rem > 10:
                t4 = min(rem * 0.5, 20)
                if t4 > 5:
                    a_opt, obj = coord_descent_peak(best_seq, time_limit=t4, verbose=False)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
