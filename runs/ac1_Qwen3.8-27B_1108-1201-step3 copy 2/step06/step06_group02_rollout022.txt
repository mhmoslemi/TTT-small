Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max smoothed mirror descent
with adaptive temperature, peak surgery, LP-FW refinement, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    conv = fast_convolve(a)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def pair_optimize(f, n_trials=200, time_limit=60.0, verbose=False):
    """
    Pair-wise coordinate optimization: for promising pairs (i,j),
    find optimal mass redistribution via 1D convex minimization.
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        n_top = min(20, len(conv))
        top_peaks = np.argpartition(conv, -n_top)[-n_top:]

        # Build list of candidate pairs
        candidates = []
        for k in top_peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if js > je:
                continue
            idx_range = np.arange(js, je + 1)
            products = f[idx_range] * f[k - idx_range]
            # Top contributing indices
            top_contrib = idx_range[np.argsort(products)[-10:]]
            for i in top_contrib:
                j = k - i
                if i != j and 0 <= j < n:
                    candidates.append((int(i), int(j)))
            # Also add random pairs for diversity
            if it % 5 == 0:
                ri = rng.integers(0, n)
                rj = rng.integers(0, n)
                if ri != rj:
                    candidates.append((int(ri), int(rj)))

        # Deduplicate
        seen = set()
        unique_candidates = []
        for c in candidates:
            key = (min(c[0], c[1]), max(c[0], c[1]))
            if key not in seen:
                seen.add(key)
                unique_candidates.append(c)

        improved = False
        for i, j in unique_candidates[:n_trials]:
            if time.monotonic() - t0 > time_limit:
                break

            # Compute slope for each k: s_k = 2*(a[k-i] - a[k-j])
            # New conv[k] = conv[k] + delta * s_k (exact for bilinear)
            # But we need to handle the case where i or j appear in both roles
            # For exactness, let's compute directly for the affected conv values

            # Affected k values: k = i+j (where both i and j contribute)
            # Actually, changing a[i] affects conv[k] for k = i+m (all valid m)
            # and changing a[j] affects conv[k] for k = j+m (all valid m)
            # So affected k range is [i, i+n-1] union [j, j+n-1]

            k_affected = np.unique(np.concatenate([
                np.arange(max(0, i), min(2*n-1, i+n-1)+1),
                np.arange(max(0, j), min(2*n-1, j+n-1)+1)

            # delta_conv[k] = delta * (2*a[k-i] - 2*a[k-j]) + cross_terms
            # Cross term: if k-i == j or k-j == i, there's a -delta^2 term

            slopes = np.zeros(len(k_affected))
            has_quadratic = False
            for idx, k in enumerate(k_affected):
                k = int(k)
                # d conv[k]/d a[i] = 2*a[k-i] if valid
                coeff_i = 0.0
                if 0 <= k - i < n:
                    coeff_i = 2.0 * f[k - i]
                # d conv[k]/d a[j] = 2*a[k-j] if valid
                coeff_j = 0.0
                if 0 <= k - j < n:
                    coeff_j = 2.0 * f[k - j]
                # Cross term: if k-i == j AND k-j == i (i.e., i+j==k),
                # then there's a -delta^2 term (from (a[i]+d)(a[j]-d) instead of a[i]a[j])
                if k - i == j and k - j == i:
                    has_quadratic = True
                    # The exact change: (a[i]+d)(a[j]-d) - a[i]*a[j] = d*(a[j]-a[i]) - d^2
                    # So the linear part is d*(a[j]-a[i]) and quadratic is -d^2
                    # But we already have 2*a[k-i]-2*a[k-j] = 2*a[j]-2*a[i]
                    # The exact change for this specific term is d*(a[j]-a[i]) - d^2
                    # So we need to subtract d^2, meaning the slope for this k is actually
                    # (a[j]-a[i]) instead of 2*(a[j]-a[i]), plus a -d^2 term
                    # Let's just handle this by using a smaller step and verifying
                    slopes[idx] = 2.0 * (f[k - i] - f[k - j])
                else:
                    slopes[idx] = coeff_i - coeff_j

            # Now minimize max_k (conv[k] + delta * slopes[k]) over valid delta range
            # Valid range: a[i]+delta >= 0 and a[j]-delta >= 0
            delta_lo = -f[i] * 0.95
            delta_hi = f[j] * 0.95

            if delta_lo >= delta_hi:
                continue

            # Ternary search on the convex function
            lo = delta_lo
            hi = delta_hi
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                # Compute max with quadratic correction if needed
                val_m1 = 0.0
                val_m2 = 0.0
                for idx, k in enumerate(k_affected):
                    base = conv[int(k)]
                    v1 = base + m1 * slopes[idx]
                    v2 = base + m2 * slopes[idx]
                    if has_quadratic and int(k) == i + j:
                        v1 -= m1 * m1
                        v2 -= m2 * m2
                    val_m1 = max(val_m1, v1)
                    val_m2 = max(val_m2, v2)
                # Also check non-affected peaks (they don't change)
                val_m1 = max(val_m1, np.max(conv[~np.isin(np.arange(len(conv)), k_affected)]))
                val_m2 = max(val_m2, np.max(conv[~np.isin(np.arange(len(conv)), k_affected)]))

                if val_m1 < val_m2:
                    hi = m2
                else:
                    lo = m1

            delta_opt = (lo + hi) / 2.0

            # Apply and verify
            f_new = f.copy()
            f_new[i] += delta_opt
            f_new[j] -= delta_opt
            f_new = np.maximum(f_new, 0.0)
            ss = f_new.sum()
            if ss < 1e-15:
                continue
            f_new = f_new / ss
            obj_new = fast_convolve(f_new).max()

            if obj_new < best_obj - 1e-14:
                f = f_new
                best_obj = obj_new
                best_f = f.copy()
                improved = True
                if verbose:
                    print(f"    PairOpt it{it}: pair=({i},{j}), delta={delta_opt:.6f}, obj={best_obj:.10f}")

        if not improved:
            break

    return best_f


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with adaptive M-relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        conv = fast_convolve(f)
        M_current = conv.max()

        # Try increasing M relaxation
        g = None
        for mult in [1.0, 1.02, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0]:
            M_try = M_current * mult
            g = solve_lp_direction(f, M_try, n)
            if g is not None and g.sum() > 1e-12:
                break

        if g is None or g.sum() < 1e-12:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve += 1
            if no_improve > 100:
                break
            continue

        g = g / g.sum()

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 80:
                break
            continue
        no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            f = f_new

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def peak_surgery(f, time_limit, verbose=False):
    """Targeted peak reduction with conflict-aware mass redistribution."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            break

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        if len(contrib_idx) == 0:
            break

        products = f[contrib_idx] * f[peak_k - contrib_idx]
        max_prod_pos = np.argmax(products)
        source = int(contrib_idx[max_prod_pos])
        partner = int(peak_k - source)

        # Compute conflict score for each non-contributing position
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[contrib_idx] = False
        non_contrib[source] = False

        if non_contrib.sum() < 1:
            no_improve += 1
            break

        # Conflict: for each candidate target position t,
        # measure how much it contributes to other top peaks
        top_peaks = np.argpartition(conv, -10)[-10:]
        targets = np.nonzero(non_contrib)[0]

        conflict_scores = np.zeros(len(targets))
        for t_idx, t in enumerate(targets):
            for pk in top_peaks:
                pk = int(pk)
                if 0 <= pk - t < n and t != pk - t:
                    conflict_scores[t_idx] += conv[pk] * f[pk - t]
                elif 0 <= pk - t < n and t == pk - t:
                    conflict_scores[t_idx] += conv[pk] * f[t] * 0.5

        # Pick target with lowest conflict
        best_target = int(targets[np.argmin(conflict_scores)])

        # Try different fractions
        best_trial = None
        best_trial_obj = M
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = f[source] * frac
            if diff < 1e-15:
                continue
            trial = f.copy()
            trial[source] -= diff
            trial[best_target] += diff
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_trial = trial

        if best_trial is not None:
            f = best_trial
            best_obj = best_trial_obj
            best_f = f.copy()
            no_improve = 0
            if verbose:
                print(f"    PS: peak={peak_k}, src={source}, tgt={best_target}, obj={best_obj:.10f}")
        else:
            # Try random perturbation
            no_improve += 1
            if no_improve > 10:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0

    return best_f


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.25, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n, dtype=np.float64)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.03, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """

    pair-wise coordinate optimization, LP-FW, and peak surgery.

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score) and p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        try_seq(parent, "default_parent")

    # === Phase 1: Quick initialization scan ===
    print("\n=== Phase 1: Initialization Scan ===")
    t0 = time.monotonic()

    n_base = len(parent)
    init_candidates = [('parent', parent)]

    for n_try in [n_base, 1024, 640]:
        init_candidates.append((f'cos1_{n_try}', make_cosine_mod(n_try, 1, 0.15)))

    # Score initializations
