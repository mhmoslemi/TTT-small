**Better idea: Use the "smooth approximation" of the max**

Instead of using the exact max (which is non-smooth), use a smooth approximation:
soft_max(conv, T) = T * log(sum(exp(conv[k]/T)))

This is smooth and its gradient is:
∂/∂a[i] soft_max = sum_k (conv[k]/T * exp(conv[k]/T) / sum_j exp(conv[j]/T)) * 2*a[k-i]
= 2 * sum_k p_k * a[k-i] where p_k = exp(conv[k]/T) / sum_j exp(conv[j]/T)

As T -> 0, this approaches the exact subgradient. For finite T, we get a smooth approximation.

The nice thing is: the gradient is smooth, so we can use larger step sizes and faster convergence.

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-15)
    return f_interp / f_interp.sum()

def interpolate_sequence(f, new_n):

    return f_interp / f_interp.sum()

                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_base = 768
    if parent is None:
        parent = make_gaussian(n_base, 0.3)
        n_base = len(parent)
        try_seq(parent, "default")

    # === Phase 1: Parametric exploration ===
    print("\n=== Phase 1: Parametric Exploration ===")
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Try parent and various parametric shapes
    candidates = [(parent, "parent")]

    # Grid over parametric shapes
    for alpha in [0.0, -0.1, -0.2, -0.3, 0.1, 0.2]:
        for beta in [0.0, 0.05, -0.05, 0.1]:
            for gamma in [0.0, 0.03, -0.03]:
                a_p = make_parametric(n_base, alpha, beta, gamma)
                candidates.append((a_p, f"param_{alpha:.1f}_{beta:.2f}_{gamma:.2f}"))

    # Inverted cosine variants
    for pw in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((make_inverted_cosine(n_base, pw), f"invcos_{pw}"))

    # Gaussian variants
    for sf in [0.2, 0.25, 0.3, 0.35, 0.4, 0.5]:
        candidates.append((make_gaussian(n_base, sf), f"gauss_{sf}"))

    # Perturbed parent
    for eps in [0.0005, 0.001, 0.002, 0.005, 0.01]:
        rng_p = np.random.default_rng(seed + int(eps * 100000))
        pert = parent + rng_p.standard_normal(n_base) * eps
        pert = np.maximum(pert, 1e-15)
        pert = pert / pert.sum()
        candidates.append((pert, f"pert_{eps}"))

    # Evaluate all
    for a_cand, label in candidates:
        sc = try_seq(a_cand, f"p1_{label}")

    print(f"  Phase 1 done: {time.monotonic()-t0:.1f}s, best={best_score:.8f}")

    # === Phase 2: Main smooth mirror descent ===
    print("\n=== Phase 2: Smooth Mirror Descent (Main) ===")
    a_main = best_seq / best_seq.sum() if best_seq is not None else parent
    rem = deadline - time.monotonic() - 180
    if rem > 60:
        t_alloc = min(rem, 900)
        print(f"  n={len(a_main)}, alloc={t_alloc:.0f}s")
        a_opt, obj = smooth_mirror_descent(a_main, time_limit=t_alloc, K_init=15, T_init=0.002, lr_base=2.5, restart_period=4000, verbose=True)
        try_seq(a_opt, "smd_main")

    # === Phase 3: Second MD round with different params ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 40:
            t_alloc = min(rem, 400)
            print(f"\n=== Phase 3: MD Round 2 ===")
            print(f"  alloc={t_alloc:.0f}s")
            a_opt, obj = smooth_mirror_descent(best_seq, time_limit=t_alloc, K_init=30, T_init=0.0005, lr_base=1.5, restart_period=2000, verbose=True)
            try_seq(a_opt, "smd_r2")

    # === Phase 4: Basin hopping ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            n_best = len(best_seq)
            t_alloc = min(rem, 200)
            print(f"\n=== Phase 4: Basin Hopping ===")
            n_hops = max(3, t_alloc // 15)
            per_hop = t_alloc / n_hops
            rng_h = np.random.default_rng(seed + 999)

            for hop in range(n_hops):
                rem_hop = deadline - time.monotonic() - 30
                if rem_hop < 10:
                    break
                scale = rng_h.choice([0.0003, 0.0005, 0.001, 0.0015, 0.003])
                pert = best_seq + rng_h.standard_normal(n_best) * scale
                pert = np.maximum(pert, 1e-15)
                pert = pert / pert.sum()
                # Also try structured perturbations
                if hop % 3 == 1:
                    shift = rng_h.integers(1, 5)
                    pert = np.roll(best_seq, int(shift))
                    pert = pert / pert.sum()
                elif hop % 3 == 2:
                    mix = rng_h.uniform(0.01, 0.05)
                    pert = (1 - mix) * best_seq + mix * make_gaussian(n_best, 0.3)
                    pert = pert / pert.sum()

                try_seq(pert, f"hop_{hop}")
                a_opt, obj = smooth_mirror_descent(pert, time_limit=min(per_hop, rem_hop - 5), K_init=20, T_init=0.001, lr_base=2.0, restart_period=1500, verbose=False)
                try_seq(a_opt, f"hop_opt_{hop}")

    # === Phase 5: Coordinate descent polishing ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(rem, 100)
            print(f"\n=== Phase 5: Coordinate Descent ===")
            a_opt, obj = coord_descent_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd_polish")

    # === Phase 6: Multi-length exploration ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Multi-Length ===")
        base_n = len(best_seq)
        targets = sorted(set([base_n - 96, base_n - 48, base_n + 48, base_n + 96, base_n + 128, 512, 640, 896, 1024]))
        targets = [x for x in targets if 256 <= x <= 2048 and x != base_n]

        for tn in targets[:5]:
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            f_interp = interpolate_sequence(best_seq, tn)
            sc = try_seq(f_interp, f"interp_{tn}")
            if sc < best_score + 0.005:
                t_alloc = min(rem - 5, 25)
                if t_alloc > 8:
                    a_opt, obj = smooth_mirror_descent(f_interp, time_limit=t_alloc, K_init=20, T_init=0.001, lr_base=2.0, restart_period=1000, verbose=False)
                    try_seq(a_opt, f"md_{tn}")

    # === Phase 7: Final polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_alloc = min(rem, 20)
            print(f"\n=== Phase 7: Final Polish ===")
            a_opt, obj = coord_descent_polish(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_cd")
            rem2 = deadline - time.monotonic() - 2
            if rem2 > 5:
                a_opt2, obj2 = smooth_mirror_descent(best_seq, time_limit=min(rem2, 10), K_init=40, T_init=0.0003, lr_base=1.0, restart_period=500, verbose=False)
                try_seq(a_opt2, "final_smd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
