def evaluate_sequence(sequence: list[float]) -> float:
    """

    Returns np.inf if the input is invalid.

    # Verify that the input is a list

    for x in sequence:

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    sequence = [max(0, x) for x in sequence]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """
    Compute gradient of F(a) = log(sum_k exp((a*a)[k]/tau)).
    Returns (gradient vector, convolution).
    Uses FFT for O(n log n) computation.
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # h[j] = sum_{l=0}^{n-1} w[j+l] * f[l]
    # Computed via: h = conv(f, w_rev)[n-1:2n-1][::-1]
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    grad = (2.0 / tau) * h
    return grad, conv

def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def minimax_multiplicative_weights(f0, time_limit, verbose=False):
    """
    Minimax approach: alternate between inner QP (projected gradient)
    and outer multiplicative weight update.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    # Initialize p (outer variable)
    p = np.ones(2 * n - 1) / (2 * n - 1)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    lr_p = 0.15
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Inner: solve min a^T Q(p) a via projected gradient (few iterations)
        a_inner = f.copy()
        inner_iters = min(30, max(10, 2000 // (it + 1)))
        inner_lr = 2.0 / (n * 1.1)

        for _ in range(inner_iters):
            # Gradient of a^T Q(p) a = 2 * (Q(p) * a)
            # (Q(p)*a)[j] = sum_l p[j+l] * a[l]
            p_rev = p[::-1]
            total_len = n + len(p) - 1
            fft_size = 1 << (total_len - 1).bit_length()
            A = np.fft.rfft(a_inner, fft_size)
            P = np.fft.rfft(p_rev, fft_size)
            conv_r = np.fft.irfft(A * P, fft_size)
            grad = conv_r[n - 1:2 * n - 1][::-1]  # This is (Q*p_a)[j]
            grad = 2.0 * grad

            a_inner = a_inner - inner_lr * grad
            a_inner = project_simplex(a_inner)

        f = a_inner
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 50 == 0:
                print(f"    Minimax iter {it}: obj={2*n*best_obj:.8f}")

        # Outer: multiplicative weight update on p
        # Weight positions proportional to their convolution value
        scaled = conv / (0.02 * M + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        p = (1 - lr_p) * p + lr_p * w
        p = p / p.sum()

        # Adaptive lr_p
        if it % 100 == 0:
            lr_p = max(0.03, lr_p * 0.95)

    return best_f, 2 * n * best_obj


def peak_targeted_cd(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: identifies peak-contributing pairs
    and redistributes mass to reduce the peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if j_max == j_other:
                # Diagonal element: just reduce f[j_max]
                old_val = f[j_max]
                if old_val < 1e-10:
                    continue
                for factor in [0.5, 0.7, 0.85]:
                    trial = f.copy()
                    trial[j_max] *= factor
                    diff = old_val - trial[j_max]
                    # Redistribute to non-peak-contributing elements
                    contrib_mask = np.ones(n, dtype=bool)
                    # Elements contributing to top peaks
                    for pk2 in peak_indices[:3]:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        contrib_mask[js2:je2 + 1] = False
                    if contrib_mask.sum() > 0:
                        trial[contrib_mask] += diff * f[contrib_mask] / f[contrib_mask].sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: reduce the product
                old_jm = f[j_max]
                old_jo = f[j_other]
                for scale_jm in [0.6, 0.8]:
                    new_jm = old_jm * scale_jm
                    diff = old_jm - new_jm
                    trial = f.copy()
                    trial[j_max] = new_jm
                    # Redistribute to j_other's "opposite" (non-contributing)
                    mask = np.ones(n, dtype=bool)
                    mask[j_max] = False
                    # Also avoid adding to elements that contribute to the same peak
                    mask[j_other] = False
                    if mask.sum() > 0:
                        trial[mask] += diff * f[mask] / f[mask].sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Try a larger perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    """Asymmetric Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    """Flat-topped bell initialization."""
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
