def build_conv_matrix(f):
    """Build matrix C of shape (2n-1, n) such that (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    k_idx = np.arange(2 * n - 1)[:, None]
    i_idx = np.arange(n)[None, :]
    j = k_idx - i_idx

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the state-of-the-art LP direction-finding approach with adaptive step sizes,
diverse initializations, multi-length exploration, subgradient backup, and
L-BFGS/coordinate-descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix C of shape (2n-1, n) such that (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    k_idx = np.arange(2 * n - 1)[:, None]
    i_idx = np.arange(n)[None, :]
    j = k_idx - i_idx
    valid = (j >= 0) & (j < n)
    C = np.zeros((2 * n - 1, n))
    C[valid] = f[j[valid]]
    return C


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def lp_descent(a, time_limit, verbose=True, seed=42, tight_factor=1.0):
    """
    LP-based descent: at each step, solve an LP to find the best direction,
    then take a step in that direction with adaptive line search.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    a = a / S

    best_a = a.copy()
    best_M = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    step_sizes = [0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = fast_convolve(a).max()

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 200 == 1:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: M={M:.10f}, best={best_M:.10f}, t={elapsed:.1f}s, no_imp={no_improve}")

        # If stuck for too long, perturb
        if no_improve > 500:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v_reset = True
            no_improve = 0
            continue
        elif no_improve > 200:
            a = project_simplex(a + rng.standard_normal(n) * 0.001)
            no_improve = 0
            continue

        # Build convolution matrix
        C = build_conv_matrix(a)

        # Solve LP: max sum(g) s.t. C@g <= M * tight_factor, g >= 0
        bound = M * tight_factor
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, bound),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 10, 'presolve': True})
        except Exception:
            try:
                res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, bound),
                              bounds=[(0, None)] * n, method='highs-ds')
            except Exception:
                break

        if not res.success:
            if no_improve > 100:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        g = np.maximum(res.x, 0)
        S_g = g.sum()
        if S_g < 1e-10:
            break
        g = g / S_g

        # Line search: find best step size
        best_t = 0
        best_new_M = M
        for t in step_sizes:
            a_new = (1 - t) * a + t * g
            M_new = fast_convolve(a_new).max()
            if M_new < best_new_M:
                best_new_M = M_new
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
        else:
            # Tiny step to escape
            a = 0.99 * a + 0.01 * g

    return best_a, best_M


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.001, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    """
    Projected subgradient method with momentum.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        # Temperature-weighted peak selection
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx]
            w_sel_sum = w_sel.sum()
            if w_sel_sum > 1e-30:
                w_sel /= w_sel_sum
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel
            w = w_full

        # Subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]

        # Momentum step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.97

        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def cross_corr_fft(w, a):
    """Cross-correlation via FFT: result[i] = sum_k w[k] * a[k-i] (valid)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:size]


def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    # Gradient wrt a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0 * n
    # Chain rule through softmax
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=3000):
    """L-BFGS in log-space with soft-max objective."""
    t0 = time.monotonic()
    n = len(a0)
    z0 = z_from_a(a0)
    best_obj = soft_max_obj_grad(z0, T, n)[0]
    best_z = z0.copy()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass

    return softmax_from_z(best_z), best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = np.array([0.5, 0.7, 0.85, 0.9, 1.0, 1.1, 1.2, 1.4, 1.8, 2.5])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                s = a.sum()
                a /= s
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine window initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    """Tukey (tapered cosine) window."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_highs_first_init(n, decay=0.7):
    """Decaying sequence (high values first) - exploits asymmetry benefit."""
    a = np.array([decay ** i for i in range(n)])
    return a / a.sum()


def make_rising_init(n, decay=0.7):
    """Rising sequence (high values last)."""
    a = np.array([decay ** (n - 1 - i) for i in range(n)])
    return a / a.sum()


def make_bimodal_init(n, gap=0.3):

    print(f"\n=== Phase 7: LP pass 3 (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem, 100.0)
            print(f"  LP descent t={t_alloc:.0f}s")
            a_lp3, _ = lp_descent(best_seq, time_limit=t_alloc, seed=seed + 555,
                                  tight_factor=0.9995, verbose=False)
            try_seq(a_lp3, "lp_final")

    # === Phase 8: Coordinate descent polish ===
    print(f"\n=== Phase 8: Coordinate descent (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_cd = min(rem, 60.0)
            print(f"  CD polish, t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd_polish")

    # === Phase 9: Final micro-optimization ===
    print(f"\n=== Phase 9: Final micro-opt (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            # Short LP
            t_alloc = min(rem * 0.4, 30.0)
            a_micro, _ = lp_descent(best_seq, time_limit=t_alloc,
                                    seed=seed + 777777, verbose=False,
                                    tight_factor=0.999)
            try_seq(a_micro, "micro_lp")

            # Perturbation + re-optimize
            rem = deadline - time.monotonic() - 10
            if rem > 8:
                for attempt in range(3):
                    rem = deadline - time.monotonic() - 5
                    if rem < 5:
                        break
                    noise = rng.standard_normal(n_best) * 0.002
                    a_pert = project_simplex(best_seq + noise)
                    t_alloc = min(rem * 0.4, 15.0)
                    a_p, _ = lp_descent(a_pert, time_limit=t_alloc,
                                        seed=seed + attempt * 31337, verbose=False)
                    try_seq(a_p, f"perturb_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
