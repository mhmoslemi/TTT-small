"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. L-BFGS in log-space with soft-max smoothing and temperature annealing
2. Nesterov-accelerated subgradient with peak weighting
3. Targeted peak shaving via coordinate moves
4. Multi-length exploration with interpolation
Key innovations: adaptive temperature schedule, Nesterov momentum, 
spectral-domain initialization, and iterative dual-guided reweighting.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 efficiently."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    max_conv = fast_convolve(a_norm).max()
    return 2.0 * n * max_conv


def softmax_z(z):
    """Convert log-domain vector to probability via softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-30))


def softmax_cross_corr(w, a):
    """Compute cross-correlation sum_j w[i+j]*a[j] for each i, via FFT."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_corr[i] = sum_j w[i+j] * a[j]
    # This is the 'full' cross-correlation
    result = irfft(np.conj(W) * A, fft_size)[:n]
    return result


def soft_max_obj_and_grad(z, T, n):
    """
    Soft-max smoothed objective and gradient in log-space.
    f_T(a) = (1/T) * log(sum_k exp(T * (a*a)[k]))
    """
    a = softmax_z(z)
    b = fast_convolve(a)  # length 2n-1
    max_b = b.max()
    scaled = T * (b - max_b)
    exp_s = np.exp(scaled)
    log_sum_exp = max_b + np.log(exp_s.sum()) / T
    obj = log_sum_exp  # This is (1/T)*log(sum exp(T*b_k))
    
    # Weights for each lag
    w = exp_s / exp_s.sum()
    
    # Gradient w.r.t. a: grad_a[i] = (2/T) * sum_k w_k * 2 * a_{k-i} (if valid)
    # = (4/T) * sum_{j=0}^{n-1} w[i+j] * a[j]
    # But we need to scale by 2*n for the actual score
    # Actual score = 2n * max_b / (sum a)^2, with sum a = 1, so score = 2n * max_b
    # Smoothed: score_T = 2n * (1/T) * log(sum exp(T*b_k))
    # grad_score_T[i] = 2n * (2/T) * sum_k w_k * 2 * a_{k-i} * [valid]
    #                  = 2n * (4/T) * sum_j w[i+j] * a[j]
    
    grad_a = (4.0 * n / T) * softmax_cross_corr(w, a)
    
    # Transform to z-space: grad_z[i] = a_i * (grad_a[i] - sum_j a_j * grad_a[j])
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return 2.0 * obj, grad_z


def lbgfs_refine(a0, T, time_budget, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_budget:
            stopped[0] = True
            return 1e15
        obj, _ = soft_max_obj_and_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_budget:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_and_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_out = res.x
    except Exception:
        z_out = z0

    a_out = softmax_z(z_out)
    a_out = a_out / a_out.sum()
    return a_out


def nesterov_subgrad(a0, time_budget, lr_init=0.003, momentum=0.9):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    t0 = time.monotonic()
    lr = lr_init
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        # Nesterov step: evaluate gradient at y
        b = fast_convolve(y)
        # Subgradient: peak is at argmax(b)
        # For soft version, use top-k peaks
        top_k = min(20, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        # Weight peaks by excess over mean
        b_mean = b.mean()
        b_excess = np.maximum(b[peaks] - b_mean, 0)
        w_sum = b_excess.sum()
        if w_sum < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = b_excess / w_sum

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_arr = np.arange(js, je + 1)
            sg[j_arr] += w[idx] * 2.0 * a[k - j_arr]

        # Update momentum
        v = momentum * v - lr * sg

        # Nesterov extrapolation
        beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum**2)) / 2.0
        y = a + momentum * beta * v

        # Project y onto simplex
        a_new = project_simplex(y)

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        step += 1

        # Adapt learning rate
        lr *= 0.999
        if no_improve > 200:
            # Random perturbation to escape
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            v = np.zeros(n)
            lr = max(lr * 2, 0.001)
            no_improve = 0

    return best_a, best_score


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def peak_shave(a0, time_budget, n_steps=5000):
    """Targeted peak shaving: reduce the top contributing pair at each peak."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        # Find the peak (exclude endpoints which are always small)
        b_mid = b.copy()
        b_mid[:2] = -np.inf
        b_mid[-2:] = -np.inf
        peak_k = int(np.argmax(b_mid))
        peak_val = b[peak_k]
        threshold = best_score / (2.0 * n)
        if peak_val <= threshold + 1e-15:
            break

        # Find top contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_ci = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_ci:
            i = int(idx[ci])
            j = peak_k - i
            if j < 0 or j >= n or i == j:
                continue
            if a[i] < 1e-12 or a[j] < 1e-12:
                continue
            # Try reducing both i and j
            for f1 in [0.95, 0.9, 0.85, 0.8]:
                for f2 in [0.95, 0.9, 0.85, 0.8]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    # Redistribute to maintain sum
                    old_sum = a.sum()
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try single-element reduction
            for ci in top_ci:
                i = int(idx[ci])
                if a[i] < 1e-12:
                    continue
                for factor in [0.95, 0.9, 0.85]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1 - factor)
                    a_trial[i] *= factor
                    # Distribute to others
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    if others.sum() > 1e-15:
                        a_trial[mask] = others + diff * others / others.sum()
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try 3-element moves
            if len(top_ci) >= 3:
                for c1 in range(len(top_ci)):
                    for c2 in range(c1 + 1, len(top_ci)):
                        i1 = int(idx[top_ci[c1]])
                        j1 = peak_k - i1
                        i2 = int(idx[top_ci[c2]])
                        j2 = peak_k - i2
                        if j1 < 0 or j1 >= n or j2 < 0 or j2 >= n:
                            continue
                        for f in [0.9, 0.8]:
                            a_trial = a.copy()
                            a_trial[i1] *= f
                            a_trial[i2] *= f
                            a_trial = a_trial / a_trial.sum()
                            score = compute_score(a_trial)
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break

        if not improved:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_init(n, mode='cosine', rng=None):
    """Create spectrally-motivated initial sequences."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    if mode == 'cosine':
        a = np.cos(np.pi * x) ** 2
    elif mode == 'gaussian':
        a = np.exp(-20 * (x - 0.5) ** 2)
    elif mode == 'edge':
        a = np.exp(-5 * np.abs(x - 0.5))
    elif mode == 'raised_cosine':
        t = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(t) + 0.2 * np.cos(2 * t)
    elif mode == 'random_smooth':
        a = np.ones(n)
        for freq in rng.choice([2, 3, 4, 5, 7, 8], size=3, replace=False):
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.2 * np.cos(2 * np.pi * freq * x + phase)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def dual_guided_refine(a0, time_budget, n_outer=50):
    """
    Iterative method using dual variables to guide updates.
    At each outer iteration, compute the "dual" weights from the auto-correlation,
    then solve a weighted quadratic to update a.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        # Soft max weights
        T = max(10.0, 100.0 * (1.0 - outer / n_outer))
        b_shift = b - b.max()
        w = np.exp(T * b_shift)
        w = w / w.sum()

        # Compute gradient direction: want to reduce peaks
        # grad[i] = sum_k w_k * 2 * a[k-i] (for valid k-i)
        grad = 2.0 * softmax_cross_corr(w, a)

        # Take a step: a_new = a - eta * grad, then project
        eta = 0.1 / (np.max(grad) + 1e-10)
        a_cand = a - eta * grad
        a_cand = np.maximum(a_cand, 0)
        s = a_cand.sum()
        if s < 1e-15:
            break
        a_cand = a_cand / s

        # Line search
        best_eta = eta
        best_s = compute_score(a_cand)
        for trial_factor in [0.5, 0.25, 2.0, 4.0]:
            trial_eta = eta * trial_factor
            a_trial = a - trial_eta * grad
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            s_score = compute_score(a_trial)
            if s_score < best_s:
                best_s = s_score
                best_eta = trial_eta
                a_cand = a_trial

        if best_s < best_score - 1e-14:
            best_score = best_s
            best_a = a_cand.copy()
            a = a_cand
        else:
            # Try smaller step
            a_trial = a - 0.01 * eta * grad
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
                s_score = compute_score(a_trial)
                if s_score < best_score:
                    best_score = s_score
                    best_a = a_trial.copy()
                    a = a_trial

    return best_a, best_score

    print(f"Initial: n={n}, score={best_score:.8f}")

    rng = np.random.default_rng(12345)

    # === Phase 1: Spectral initialization + L-BFGS at parent length ===
    print("\n=== Phase 1: Multi-init L-BFGS at n={} ===".format(n))
    remaining = deadline - time.monotonic()
    init_budget = min(remaining * 0.25, 300.0)
    per_init_budget = init_budget / 8

    inits = [parent.copy()]
    for mode in ['cosine', 'gaussian', 'edge', 'raised_cosine', 'random_smooth']:
        inits.append(spectral_init(n, mode, rng))
    # Perturbed parent
    inits.append(np.maximum(parent + rng.standard_normal(n) * 0.05, 0) / parent.sum())
    inits.append(np.maximum(parent + rng.standard_normal(n) * 0.1, 0) / parent.sum())

    for idx, init_a in enumerate(inits):
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(per_init_budget, (deadline - time.monotonic()) - 600)
        if t_alloc < 10:
            break

        # Temperature annealing
        temps = [500, 200, 80, 30]
        a_curr = init_a.copy()
        for T in temps:
            if time.monotonic() > deadline - 500:
                break
            t_t = min(t_alloc / len(temps) * 1.5, (deadline - time.monotonic()) - 400)
            if t_t < 5:
                break
            a_new = lbgfs_refine(a_curr, T, t_t, maxiter=3000)
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"  Init {idx}, T={T}: score={sc:.8f}")
            a_curr = a_new

    print(f"  After Phase 1: best={best_score:.8f}")

    # === Phase 2: Focused L-BFGS with fine temperature annealing ===
    print("\n=== Phase 2: Fine L-BFGS annealing ===")
    temps_fine = [150, 100, 60, 40, 25, 15, 10, 7, 5]
    for T in temps_fine:
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(60.0, (deadline - time.monotonic()) - 250)
        if t_alloc < 10:
            break
        a_new = lbgfs_refine(best_a, T, t_alloc, maxiter=5000)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"  T={T}: score={sc:.8f}")
        else:
            # Also try from a slightly perturbed version
            a_pert = np.maximum(best_a + rng.standard_normal(n) * 0.005, 0)
            a_pert = a_pert / a_pert.sum()
            a_new2 = lbgfs_refine(a_pert, T, t_alloc * 0.5, maxiter=2000)
            sc2 = compute_score(a_new2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()
                print(f"  T={T} (perturbed): score={sc2:.8f}")

    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: Dual-guided refinement ===
    print("\n=== Phase 3: Dual-guided refinement ===")
    if time.monotonic() < deadline - 200:
        t_alloc = min(120.0, (deadline - time.monotonic()) - 180)
        a_new, sc_new = dual_guided_refine(best_a, t_alloc, n_outer=100)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            print(f"  Dual-guided: score={sc_new:.8f}")

        # Also try from perturbed
        a_pert = np.maximum(best_a + rng.standard_normal(n) * 0.003, 0)
        a_pert = a_pert / a_pert.sum()
        a_new2, sc_new2 = dual_guided_refine(a_pert, t_alloc * 0.5, n_outer=50)
        if sc_new2 < best_score:
            best_score = sc_new2
            best_a = a_new2.copy()
            print(f"  Dual-guided (pert): score={sc_new2:.8f}")

    print(f"  After Phase 3: best={best_score:.8f}")

    # === Phase 4: Nesterov subgradient ===
    print("\n=== Phase 4: Nesterov subgradient ===")
    if time.monotonic() < deadline - 150:
        t_alloc = min(80.0, (deadline - time.monotonic()) - 130)
        a_new, sc_new = nesterov_subgrad(best_a, t_alloc, lr_init=0.002)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            print(f"  Nesterov: score={sc_new:.8f}")

    print(f"  After Phase 4: best={best_score:.8f}")

    # === Phase 5: Peak shaving ===
    print("\n=== Phase 5: Peak shaving ===")
    if time.monotonic() < deadline - 80:
        t_alloc = min(60.0, (deadline - time.monotonic()) - 70)
        a_new, sc_new = peak_shave(best_a, t_alloc)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            print(f"  Peak shave: score={sc_new:.8f}")

    print(f"  After Phase 5: best={best_score:.8f}")

    # === Phase 6: Try slightly different lengths ===
    print("\n=== Phase 6: Multi-length exploration ===")
    alt_lengths = [n + 32, n + 64, n + 128, n - 32]
    alt_lengths = [x for x in alt_lengths if 256 <= x <= 4096 and x != n]

    for alt_n in alt_lengths[:3]:
        if time.monotonic() > deadline - 60:
            break
        t_total = min(80.0, (deadline - time.monotonic()) - 50)
        if t_total < 20:
            break

        a_interp = interpolate_seq(best_a, alt_n)
        # Quick L-BFGS at this length
        for T in [100, 40, 15]:
            if time.monotonic() > deadline - 40:
                break
            t_t = min(t_total / 3, (deadline - time.monotonic()) - 35)
            if t_t < 5:
                break
            a_new = lbgfs_refine(a_interp, T, t_t, maxiter=2000)
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"  n={alt_n}, T={T}: score={sc:.8f}")
            a_interp = a_new

    print(f"  After Phase 6: best={best_score:.8f}")

    # === Phase 7: Final polish cycle ===
    print("\n=== Phase 7: Final polish ===")
    if time.monotonic() < deadline - 20:
        # One more round of L-BFGS + peak shave
        n_final = len(best_a)
        t1 = min(30.0, (deadline - time.monotonic()) - 15)
        if t1 > 5:
            a_new = lbgfs_refine(best_a, 10, t1, maxiter=3000)
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        t2 = min(15.0, (deadline - time.monotonic()) - 5)
        if t2 > 3:
            a_new, sc = peak_shave(best_a, t2)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

    # Final verification with evaluate_sequence
    fs = float(evaluate_sequence([float(max(0, min(1000, x))) for x in best_a]))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_a)}")
    return [float(max(0.0, min(1000.0, x))) for x in best_a]
