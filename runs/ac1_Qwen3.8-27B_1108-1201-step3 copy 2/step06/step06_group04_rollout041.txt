    n = len(a)
    conv = fast_convolve(a)
    # Find top T peaks
    peak_idx = np.argpartition(conv, -T)[-T:]
    # Compute subgradient
    subgrad = np.zeros(n)

        for j in range(n):
            kj = k - j
            if 0 <= kj < n:

    subgrad /= T

    a_new = a - lr * subgrad
    # Project
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:

    a_new /= s
    return a_new

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

        M = conv.max()
        current_obj = 2.0 * n * M
        
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top T peaks
        if len(conv) <= T_peaks:
            peak_idx = np.arange(len(conv))
        else:
            peak_idx = np.argpartition(conv, -T_peaks)[-T_peaks:]
        
        # Weight peaks by their value (emphasize larger peaks)
        peak_vals = conv[peak_idx]
        peak_vals = np.maximum(peak_vals - 0.9 * M, 1e-10)  # relative weighting
        weights = peak_vals / peak_vals.sum()
        
        # Compute weighted subgradient
        subgrad = np.zeros(n)
        for i, k in enumerate(peak_idx):
            k = int(k)
            w = weights[i]
            if w < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            subgrad[idx] += 2.0 * w * f[k - idx]
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * subgrad
        
        # Adaptive step size (backtracking)
        for attempt in range(8):
            f_trial = f - lr * momentum
            f_trial = np.maximum(f_trial, 0.0)
            tr = f_trial.sum()
            if tr < 1e-15:
                lr *= 0.5
                continue
            f_trial = f_trial / tr
            trial_obj = 2.0 * n * fast_convolve(f_trial).max()
            if trial_obj < current_obj or attempt >= 7:
                break
            lr *= 0.5
        
        f = f_trial
        lr = max(lr * 1.01, lr_init * 0.3)  # slowly increase lr, floor at 30% of init
        
        if no_improve > 200:
            # Perturb to escape
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            momentum = np.zeros(n)
            lr = lr_init
            no_improve = 0
        
        if it % 5000 == 0 and verbose:
            print(f"    SGM iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        
        if no_improve > 500:
            break
    
    return best_f, best_obj

def spectral_flattening(f0, time_limit, verbose=False):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively adjust the spectrum to be flat while maintaining non-negativity in time domain.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Use FFT size that's a power of 2, at least 2n-1
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    # Target: flatten the non-DC part of the power spectrum
    # We want |A[m]|^2 to be as uniform as possible for m != 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Forward FFT
        A = np.fft.rfft(f, fft_size)
        
        # Current power spectrum (magnitude squared)
        P = np.abs(A) ** 2
        
        # DC bin is fixed: P[0] = (sum f)^2 = 1
        # Target: make P[1:] as flat as possible
        # Use water-filling: try to equalize P[1:]
        # But we can't change A[0] (it's determined by sum(f)=1)
        
        # Compute target amplitude for non-DC bins
        # Mean of P[1:]
        if len(P) > 1:
            target_level = P[1:].mean()
            # For the flattest spectrum, all non-DC bins should be at target_level
            # But we need to ensure the total energy is preserved
            # Total energy = sum(P) = fft_size * sum(f^2) (by Parseval)
            # So we can't just set all to target_level
            # Instead, use a soft flattening: 
            # |A_new[m]| = sqrt(alpha * P[m] + (1-alpha) * target_level) for m > 0
            
            alpha = 0.3  # mixing parameter (tune this)
            new_mag = np.abs(A).copy()
            new_mag[1:] = np.sqrt(alpha * P[1:] + (1 - alpha) * target_level)
            
            # Reconstruct A with new magnitudes and old phases
            phases = np.angle(A)
            A_new = new_mag * np.exp(1j * phases)
            # Keep DC fixed
            A_new[0] = A[0]
            
            # Inverse FFT
            f_new = np.fft.irfft(A_new, fft_size)[:n]
            
            # Enforce non-negativity
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                no_improve += 1
                alpha = min(alpha * 1.2, 0.9)
                continue
            f_new = f_new / sn
            
            obj = 2.0 * n * fast_convolve(f_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
            else:
                no_improve += 1
                # Reduce aggressiveness
                alpha = max(alpha * 0.95, 0.05)
        
        if no_improve > 100:
            # Reset with perturbation
            if no_improve > 300:
                break
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            alpha = 0.3
            no_improve = 0
        
        if it % 1000 == 0 and verbose:
            print(f"    Spectral iter {it}: obj={best_obj:.10f}, alpha={alpha:.3f}")
    
    return best_f, best_obj

def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                conv_cur = fast_convolve(best_f)
                k_peak = int(np.argmax(conv_cur))
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                f = best_f.copy()
                f[js:je+1] *= 0.8
                diff = 1.0 - f.sum()
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je+1] = False
                if non_peak.sum() > 0:
                    w = f[non_peak]
                    wsum = w.sum()
                    if wsum > 1e-15:
                        f[non_peak] += diff * w / wsum
                f = np.maximum(f, 0)
                s2 = f.sum()
                if s2 > 1e-15:
                    f = f / s2
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.08, 6.0)
            if no_improve > 80:
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.002 * min(m_exp, 3.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            if diff > 1e-5:
                no_improve = 0
            elif diff > 1e-7:
                no_improve += 1
            else:
                no_improve += 3
        
        # Line search
        best_alpha = 0.0
        best_alpha_val = M
        for a in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_new = f + a * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            val = fast_convolve(f_new).max()
            if val < best_alpha_val:
                best_alpha_val = val
                best_alpha = a
        
        if best_alpha < 1e-8:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)
        else:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            sn = f.sum()
            if sn > 1e-15:
                f = f / sn
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            current_obj = 2.0 * n * M
            if current_obj < best_obj - 1e-15:
                best_obj = current_obj
                best_f = f.copy()
        
        if no_improve > 100:
            break
        
        if verbose and it % 10 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, verbose=False):
    """Peak-targeted coordinate descent for final polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 60:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue
            
            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                
                for mult in [0.6, 0.8, 0.9, 1.1, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue
                    
                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = f[non_peak]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = f.copy()
                trial[js:je + 1] *= 0.93
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc_w = f[non_contrib]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts > 1e-15:
                    trial = trial / ts
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Fast subgradient descent with momentum and adaptive step size
2. Novel spectral flattening via iterative GS-style magnitude equalization
3. LP-based Frank-Wolfe for precision refinement
4. Peak-targeted coordinate descent for final polish
5. Multi-length exploration with diverse initializations
Key innovation: spectral domain optimization targeting fourth-moment minimization
of the DFT magnitude to reduce auto-convolution peaks.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0):
    """Asymmetric exponential decay initial sequence."""
    x = np.arange(n)
    a = np.exp(-decay * x / n)
    a = a / a.sum()
    return a


def make_bell(n, width=0.7):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def subgradient_descent(f0, time_limit, T_peaks=20, lr_init=0.02, verbose=False):
    """Fast subgradient descent with momentum and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.5
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()
        current_obj = 2.0 * n * M

        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top T peaks with soft weighting
        if len(conv) <= T_peaks:
            peak_idx = np.arange(len(conv))
        else:
            peak_idx = np.argpartition(conv, -T_peaks)[-T_peaks:]

        peak_vals = conv[peak_idx]
        # Weight by exponential of scaled peak value (emphasize larger peaks)
        scaled = peak_vals / (M + 1e-15)
        weights = np.exp(3.0 * (scaled - 1.0))
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.ones(len(peak_idx)) / len(peak_idx)
        else:
            weights /= ws

        # Compute weighted subgradient
        subgrad = np.zeros(n)
        for i in range(len(peak_idx)):
            k = int(peak_idx[i])
            w = weights[i]
            if w < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            subgrad[idx] += 2.0 * w * f[k - idx]

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * subgrad

        # Adaptive step size with backtracking
        best_trial = None
        for attempt in range(6):
            f_trial = f - lr * momentum
            f_trial = np.maximum(f_trial, 0.0)
            tr = f_trial.sum()
            if tr < 1e-15:
                lr *= 0.5
                continue
            f_trial = f_trial / tr
            trial_obj = 2.0 * n * fast_convolve(f_trial).max()
            if trial_obj <= current_obj * (1.0 + 1e-10):
                best_trial = f_trial
                break
            lr *= 0.5

        if best_trial is not None:
            f = best_trial
            lr = min(lr * 1.02, lr_init)
        else:
            # All attempts failed, try very small step
            f_trial = f - 0.001 * momentum
            f_trial = np.maximum(f_trial, 0.0)
            tr = f_trial.sum()
            if tr > 1e-15:
                f = f_trial / tr

        # Reset momentum occasionally
        if it % 500 == 0:
            momentum *= 0.5

        if no_improve > 300:
            # Perturb to escape local minimum
            rng = np.random.default_rng(it * 7 + 3)
            scale = 0.002 + 0.001 * (no_improve // 300)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            momentum = np.zeros(n)
            lr = lr_init
            no_improve = 0

        if it % 10000 == 0 and verbose:
            print(f"    SGM iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")

        if no_improve > 800:
            break

    return best_f, best_obj


def spectral_flattening(f0, time_limit, alpha_init=0.2, verbose=False):
    """
    GS-style spectral flattening: iteratively equalize |FFT(a)[m]| for m!=0
    to minimize the peak of the auto-convolution.
    Key insight: min max(conv) <=> min sum_{m!=0}|A[m]|^4 (fourth moment).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    n_freq = fft_size // 2 + 1

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    alpha = alpha_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Forward FFT (real)
        A = np.fft.rfft(f, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: equalize magnitudes for m >= 1
        if n_freq > 1:
            target_level = mag[1:].mean()
            if target_level < 1e-15:
                target_level = 1e-10

            # Soft flattening: blend current magnitude with target
            new_mag = mag.copy()
            new_mag[1:] = np.sqrt(alpha * mag[1:]**2 + (1.0 - alpha) * target_level**2)

            # Reconstruct with new magnitudes, old phases
            A_new = new_mag * np.exp(1j * phase)
            A_new[0] = A[0]  # Keep DC fixed

            # Inverse FFT
            f_new = np.fft.irfft(A_new, fft_size)[:n]

            # Enforce non-negativity
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                alpha = max(alpha * 0.8, 0.02)
                no_improve += 1
                continue
            f_new = f_new / sn

            obj = 2.0 * n * fast_convolve(f_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
                # Gradually increase alpha (more aggressive flattening)
                alpha = min(alpha * 1.005, 0.6)
            else:
                no_improve += 1
                alpha = max(alpha * 0.98, 0.05)

        if no_improve > 80:
            # Perturbed restart
            if no_improve > 250:
                break
            rng = np.random.default_rng(it * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            alpha = alpha_init
            no_improve = 0

        if it % 2000 == 0 and verbose:
            print(f"    Spectral iter {it}: obj={best_obj:.10f}, alpha={alpha:.3f}")

    return best_f, best_obj


def spectral_flattening_v2(f0, time_limit, verbose=False):
    """
    Enhanced spectral flattening with adaptive target and phase adjustment.
    Uses the insight that we want A^2 (not |A|^2) to be flat in frequency.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    n_freq = fft_size // 2 + 1

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    gamma = 0.15  # mixing parameter

    while time.monotonic() - t0 < time_limit:
        it += 1

        A = np.fft.rfft(f, fft_size)
        mag = np.abs(A)

        # Compute B = A^2 (squared, not magnitude squared)
        # We want |B[m]| = |A[m]|^2 to be flat for m != 0
        # This means |A[m]| should be flat for m != 0
        b_mag = mag ** 2

        if n_freq > 1:
            target_b = b_mag[1:].mean()
            if target_b < 1e-15:
                target_b = 1e-10

            # Want |A_new[m]|^2 = gamma * |A[m]|^2 + (1-gamma) * target_b
            # So |A_new[m]| = sqrt(gamma * |A[m]|^2 + (1-gamma) * target_b)
            new_mag = mag.copy()
            new_mag[1:] = np.sqrt(gamma * b_mag[1:] + (1.0 - gamma) * target_b)

            phase = np.angle(A)
            A_new = new_mag * np.exp(1j * phase)
            A_new[0] = A[0]

            f_new = np.fft.irfft(A_new, fft_size)[:n]
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                gamma = max(gamma * 0.85, 0.03)
                no_improve += 1
                continue
            f_new = f_new / sn

            obj = 2.0 * n * fast_convolve(f_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
                gamma = min(gamma * 1.003, 0.5)
            else:
                no_improve += 1
                gamma = max(gamma * 0.97, 0.05)

        if no_improve > 60:
            if no_improve > 200:
                break
            rng = np.random.default_rng(it * 17 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            gamma = 0.15
            no_improve = 0

        if it % 5000 == 0 and verbose:
            print(f"    SpecV2 iter {it}: obj={best_obj:.10f}, gamma={gamma:.3f}")

    return best_f, best_obj


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP: (f*g)[k] for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                # Directional perturbation
                conv_cur = fast_convolve(best_f)
                k_peak = int(np.argmax(conv_cur))
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                f = best_f.copy()
                f[js:je + 1] *= 0.85
                diff = 1.0 - f.sum()
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.sum() > 0:
                    w = f[non_peak]
                    wsum = w.sum()
                    if wsum > 1e-15:
                        f[non_peak] += diff * w / wsum
                f = np.maximum(f, 0)
                s2 = f.sum()
                if s2 > 1e-15:
                    f = f / s2
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 60:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            if diff > 1e-5:
                no_improve = 0
            elif diff > 1e-7:
                no_improve += 1
            else:
                no_improve += 2

        # Coarse line search
        best_alpha = 0.0
        best_alpha_val = M
        for a in [0.1, 0.25, 0.4, 0.6, 0.8, 1.0]:
            f_new = f + a * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            val = fast_convolve(f_new).max()
            if val < best_alpha_val:
                best_alpha_val = val
                best_alpha = a

        if best_alpha < 1e-8:
            no_improve += 2
            m_exp = min(m_exp * 1.05, 4.0)
        else:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            sn = f.sum()
            if sn > 1e-15:
                f = f / sn
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            current_obj = 2.0 * n * M
            if current_obj < best_obj - 1e-15:
                best_obj = current_obj
                best_f = f.copy()

        if no_improve > 80:
            break

        if verbose and it % 5 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, m_exp={m_exp:.2f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Peak-targeted coordinate descent for final polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        top_k = min(6, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.6, 0.75, 0.9, 1.1, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = f[non_peak]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve > 8:
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = f.copy()
                trial[js:je + 1] *= 0.91
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc_w = f[non_contrib]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts > 1e-15:
                    trial = trial / ts
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

        if verbose and it % 10 == 0:
            print(f"    CD iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def multiplicative_descent(f0, time_limit, lr_init=0.003, verbose=False):
    """Mirror descent with multiplicative weight updates focused on peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        k_actual = min(20, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-10)
        f = f / f.sum()

        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temperature *= 1.01

        if no_improve > 100:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-10)
            f = f / f.sum()
            lr = max(lr_init * 0.7, 1e-6)
            no_improve = 0

        if lr < 1e-8:
            break

        if it % 5000 == 0 and verbose:
            print(f"    MD iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: subgradient descent, spectral flattening,
    LP-Frank-Wolfe, coordinate descent, multiplicative descent, and
    multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_asymmetric(768, 3.0)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")

    n_parent = len(parent)

    # === Phase 1: Diverse initialization and quick evaluation ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = [parent]
    for n_try in [640, 720, 768, 832, 896]:
        if n_try == n_parent:
            init_candidates.append(parent.copy())
        else:
            init_candidates.append(interpolate_sequence(parent, n_try))
        init_candidates.append(make_gaussian(n_try, 0.2))
        init_candidates.append(make_gaussian(n_try, 0.35))
        init_candidates.append(make_asymmetric(n_try, 2.0))
        init_candidates.append(make_asymmetric(n_try, 4.0))
        init_candidates.append(make_asymmetric(n_try, 6.0))
        init_candidates.append(make_bell(n_try, 0.5))
        init_candidates.append(make_bell(n_try, 0.8))

    best_init = None
    best_init_obj = float('inf')
    for ic in init_candidates:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = obj_normalized(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_sequence(a_norm, best_seq, best_score,
                                             f"init_n{len(a_norm)}")

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")
    base_f = best_init

    # === Phase 2: Fast subgradient descent ===
    print(f"\n=== Phase 2: Subgradient Descent (n={len(base_f)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 250)
        print(f"  SGM: t={t_alloc:.0f}s")
        a_opt, obj = subgradient_descent(base_f, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "sgm_main")

    # === Phase 3: Spectral flattening (novel) ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Spectral Flattening (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 500
        if rem > 40:
            t_alloc = min(rem, 200)
            t1 = t_alloc * 0.5
            t2 = t_alloc * 0.5
            print(f"  SpecV1: t={t1:.0f}s")
            a_opt, obj = spectral_flattening(best_seq, time_limit=t1, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "spec1")

            rem2 = deadline - time.monotonic() - 400
            if rem2 > 20 and best_seq is not None:
                t2_actual = min(t2, rem2)
                if t2_actual > 10:
                    print(f"  SpecV2: t={t2_actual:.0f}s")
                    a_opt, obj = spectral_flattening_v2(best_seq, time_limit=t2_actual, verbose=True)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "spec2")

    # === Phase 4: LP-Frank-Wolfe refinement ===
    if best_seq is not None:
        print(f"\n=== Phase 4: LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 350
        if rem > 40:
            t_alloc = min(rem, 180)
            print(f"  LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw")

    # === Phase 5: Multi-length exploration ===
    base = best_seq if best_seq is not None else base_f
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    for target_n in [704, 768, 832, 896]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score,
                                             f"interp_n{target_n}")

        # Quick SGM at new length
        t_alloc = min(rem * 0.3, 60)
        if t_alloc > 20:
            a_opt, obj = subgradient_descent(f_interp, time_limit=t_alloc)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"sgm_n{target_n}")

        # Quick spectral at new length
        rem2 = deadline - time.monotonic() - 150
        if rem2 > 20:
            t_sig = min(rem2 * 0.3, 40)
            if t_sig > 10:
                a_opt, obj = spectral_flattening(best_seq, time_limit=t_sig)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                     f"spec_n{target_n}")

    # === Phase 6: Deep optimization on best ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Deep Optimization (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 120
        if rem > 50:
            t_alloc = min(rem, 150)
            t1 = t_alloc * 0.4
            t2 = t_alloc * 0.35
            t3 = t_alloc * 0.25

            print(f"  SGM deep: t={t1:.0f}s")
            a_opt, obj = subgradient_descent(best_seq, time_limit=t1, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "sgm_deep")

            rem2 = deadline - time.monotonic() - 80
            if rem2 > 20 and best_seq is not None:
                t2_actual = min(t2, rem2)
                if t2_actual > 10:
                    print(f"  Spec deep: t={t2_actual:.0f}s")
                    a_opt, obj = spectral_flattening(best_seq, time_limit=t2_actual)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "spec_deep")

            rem3 = deadline - time.monotonic() - 50
            if rem3 > 20 and best_seq is not None:
                t3_actual = min(t3, rem3)
                if t3_actual > 10:
                    print(f"  MD: t={t3_actual:.0f}s")
                    a_opt, obj = multiplicative_descent(best_seq, time_limit=t3_actual)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "md")

    # === Phase 7: Perturbed LP-FW variants ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Perturbed LP-FW ===")
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 50
            if rem < 25:
                break
            t_alloc = min(rem / (3 - p), 60)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.001
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                f_init = 0.95 * best_seq + 0.05 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()

            print(f"  Variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"perturb_{p}")

    # === Phase 8: Final coordinate descent polish ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Polish ===")
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t1 = min(rem * 0.5, 40)
            t2 = min(rem * 0.5, 40)

            print(f"  CD: t={t1:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t1)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "cd")

            rem2 = deadline - time.monotonic() - 5
            if rem2 > 10 and best_seq is not None:
                print(f"  Final SGM: t={rem2 - 5:.0f}s")
                a_opt, obj = subgradient_descent(best_seq, time_limit=max(rem2 - 5, 1))
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "final_sgm")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
