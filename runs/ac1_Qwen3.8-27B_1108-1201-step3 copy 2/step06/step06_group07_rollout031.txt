import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(n, f):
    """Build matrix A such that A @ g = f * g (linear convolution)."""
    A = np.zeros((2*n-1, n))
    for i in range(n):
        end = min(i + n, 2*n-1)
        A[i:end, i] = f[:end-i]
    return A

def solve_lp(f, M, max_time=5.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    t0 = time.monotonic()
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': max_time})
    
    if res.status != 0 or time.monotonic() - t0 > max_time:
        return None
    return res.x

                        print(f"    {label} LP iter {it}: M={M:.10f}, t={t:.3f}")
                break

        if improved:
            no_improve = 0
            # Adapt step size
            if best_t is not None and best_t > t_init * 0.5:
                t_init = min(t_init * 1.05, 0.7)
        else:
            no_improve += 1
            t_init = max(t_init * 0.7, 0.02)

            if no_improve > 40:
                # Perturbation restart
                noise = rng.standard_normal(n) * 0.002
                f = best_f + noise
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                M = fast_convolve(f).max()
                no_improve = 0
                t_init = 0.3

                if M < best_M:
                    best_M = M
                    best_f = f.copy()

    return best_f, best_M


def coordinate_descent(f, time_limit, verbose=False, label=""):
    """Coordinate descent: adjust one element at a time."""
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_M = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    round_count = 0

    while time.monotonic() - t0 < time_limit:
        round_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]

            if old_val < 1e-14:
                # Try adding a small value at position j
                for new_val in [0.005/n, 0.01/n, 0.02/n]:
                    trial = f.copy()
                    trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others_sum = trial[mask].sum()
                    if others_sum > 1e-15:
                        trial[mask] *= (1.0 - new_val) / others_sum
                    trial = np.maximum(trial, 0.0)
                    ss = trial.sum()
                    if ss < 1e-15:
                        continue
                    trial = trial / ss
                    M_trial = fast_convolve(trial).max()
                    if M_trial < best_M - 1e-13:
                        f = trial
                        best_M = M_trial
                        best_f = f.copy()
                        improved = True
                continue

            # Try modifying f[j]
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                trial = f.copy()
                trial[j] = new_val
                diff = old_val - new_val
                if abs(diff) < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                others = trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                trial[mask] += diff * others / others_sum
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                M_trial = fast_convolve(trial).max()
                if M_trial < best_M - 1e-13:
                    f = trial
                    best_M = M_trial
                    best_f = f.copy()
                    improved = True
                    break

        if not improved:
            # Try a few random coordinate perturbations
            n_tries = min(20, n // 10)
            for _ in range(n_tries):
                if time.monotonic() - t0 > time_limit:
                    break
                j = np.random.randint(n)
                old_val = f[j]
                if old_val < 1e-14:
                    continue
                mult = np.random.uniform(0.5, 1.5)
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                trial = f.copy()
                trial[j] = new_val
                diff = old_val - new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                others = trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                trial[mask] += diff * others / others_sum
                trial = np.maximum(trial, 0.0)
                trial = trial / trial.sum()
                M_trial = fast_convolve(trial).max()
                if M_trial < best_M - 1e-13:
                    f = trial
                    best_M = M_trial
                    best_f = f.copy()
                    improved = True

            if not improved:
                break

        if verbose and round_count % 5 == 0:
            print(f"    {label} CD round {round_count}: M={best_M:.10f}")

    return best_f, best_M


def make_initialization(n, kind, rng):
    """Create various initializations, all normalized to sum=1."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'gaussian':
        x = np.arange(n) - n / 2.0
        sigma = n * 0.28
        a = np.exp(-0.5 * (x / sigma) ** 2)
        return a / a.sum()
    elif kind == 'gaussian_wide':
        x = np.arange(n) - n / 2.0
        sigma = n * 0.4
        a = np.exp(-0.5 * (x / sigma) ** 2)
        return a / a.sum()
    elif kind == 'random':
        a = rng.random(n) + 0.05
        return a / a.sum()
    elif kind == 'bimodal':
        x = np.arange(n) / max(n - 1, 1)
        a = np.exp(-0.5 * ((x - 0.3) / 0.07) ** 2) + \
            np.exp(-0.5 * ((x - 0.7) / 0.07) ** 2)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif kind == 'tapered':
        x = np.arange(n) / max(n - 1, 1) * 2 - 1
        a = (1 - np.abs(x) ** 3) ** 2
        a = np.maximum(a, 0)
        return a / a.sum()
    elif kind == 'sparse':
        m = max(2, n // 4)
        a = np.zeros(n)
        indices = rng.choice(n, size=m, replace=False)
        a[indices] = 1.0 / m
        return a
    elif kind == 'flat_center':
        a = np.zeros(n)
        center = n // 2
        width = n // 3
        a[center - width:center + width + 1] = 1.0 / (2 * width + 1)
        return a
    else:
        return np.ones(n) / n


def symmetrize(a):
    """Make a sequence palindromic (symmetric)."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_to_n(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp /= ss
    return a_interp


def weighted_average_flow(f, time_limit, t_init=0.2, verbose=False, label=""):
    """
    Alternative flow: instead of LP, use weighted average of top-peak
    reduction directions. Faster per iteration but less exact.
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_M = M

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        K = min(20, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        # Compute subgradient of max autocorrelation
        g = np.zeros(n)
        pv = conv[peak_indices]
        # Weight peaks by their value
        w = pv / pv.sum() if pv.sum() > 0 else np.ones(K) / K

        for idx in range(len(peak_indices)):
            k = int(peak_indices[idx])
            wi = w[idx]
            if wi < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            for j in range(js, je + 1):
                g[j] += 2.0 * wi * f[k - j]

        # Project gradient to tangent space of simplex
        g_centered = g - g.mean()

        # Try step sizes
        improved = False
        for t in [t_init, t_init * 0.5, t_init * 0.25, t_init * 0.1]:
            if t < 1e-8:
                continue
            f_trial = f - t * g_centered
            f_trial = np.maximum(f_trial, 0.0)
            ss = f_trial.sum()
            if ss < 1e-15:
                continue
            f_trial = f_trial / ss
            M_trial = fast_convolve(f_trial).max()
            if M_trial < M - 1e-13:
                f = f_trial
                M = M_trial
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    if verbose and it % 50 == 0:
                        print(f"    {label} WAF iter {it}: M={M:.10f}")
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            t_init = max(t_init * 0.8, 1e-6)
            if no_improve > 100:
                noise = rng.standard_normal(n) * 0.003
                f = best_f + noise
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                M = fast_convolve(f).max()
                no_improve = 0
                t_init = 0.01

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: LP-based flow (primary), weighted average flow,
    coordinate descent, across multiple sequence lengths with diverse
    initializations. Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Try parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: LP Flow from Parent (highest priority) ===
    print("\n=== Phase 1: LP Flow from Parent ===")
    if parent is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.30, 500)
        print(f"  n={len(parent)}, time={t_alloc:.0f}s")

        a_opt, M = lp_flow(parent, time_limit=t_alloc, t_init=0.3,
                           verbose=True, label="LP_parent")
        try_seq(a_opt, "LP_parent")

        # Symmetrized parent
        remaining = deadline - time.monotonic()
        if remaining > 60:
            a_sym = symmetrize(parent)
            t_alloc2 = min(remaining * 0.15, 200)
            a_opt, M = lp_flow(a_sym, time_limit=t_alloc2, t_init=0.25,
                               verbose=True, label="LP_parent_sym")
            try_seq(a_opt, "LP_parent_sym")

    # === Phase 2: LP Flow from Diverse Initializations ===
    print("\n=== Phase 2: LP Flow from Diverse Inits ===")
    n_values = [512, 768, 1024, 1280, 1536, 2048]
    if parent is not None:
        if len(parent) not in n_values:
            n_values.insert(0, len(parent))

    # Prioritize n values near parent
    init_kinds = ['uniform', 'gaussian', 'gaussian_wide', 'random', 'tapered']

    candidates = []
    for n in n_values[:5]:
        for kind in init_kinds[:4]:
            candidates.append((n, kind))

    # Limit to keep time manageable
    max_cands = 10
    candidates = candidates[:max_cands]

    for ci, (n, kind) in enumerate(candidates):
        remaining = deadline - time.monotonic()
        if remaining < 90:
            break
        t_alloc = min(remaining * 0.12, 150)
        if t_alloc < 40:
            break

        label = f"n{n}_{kind}"
        a_init = make_initialization(n, kind, rng)
        print(f"  {label}: t={t_alloc:.0f}s")
        a_opt, M = lp_flow(a_init, time_limit=t_alloc, t_init=0.3,
                           verbose=(ci < 2), label=label)
        try_seq(a_opt, label)

    # === Phase 3: Weighted Average Flow (faster per iteration) ===
    print("\n=== Phase 3: Weighted Average Flow ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.12, 180)
        if t_alloc > 30:
            print(f"  n={len(best_seq)}, time={t_alloc:.0f}s")
            a_opt, M = weighted_average_flow(best_seq, time_limit=t_alloc,
                                             t_init=0.01, verbose=True,
                                             label="WAF")
            try_seq(a_opt, "WAF")

    # === Phase 4: Coordinate Descent ===
    print("\n=== Phase 4: Coordinate Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.10, 150)
        if t_alloc > 20:
            print(f"  n={len(best_seq)}, time={t_alloc:.0f}s")
            a_opt, M = coordinate_descent(best_seq, time_limit=t_alloc,
                                          verbose=True, label="CD")
            try_seq(a_opt, "CD")

    # === Phase 5: LP Flow Round 2 (from best, with perturbation) ===
    print("\n=== Phase 5: LP Flow Round 2 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 100:
            t_alloc = min(remaining * 0.20, 300)
            print(f"  n={len(best_seq)}, time={t_alloc:.0f}s")

            # Add small perturbation to escape potential local minimum
            n = len(best_seq)
            noise = rng.standard_normal(n) * 0.001
            f_pert = best_seq + noise
            f_pert = np.maximum(f_pert, 0.0)
            f_pert = f_pert / f_pert.sum()

            a_opt, M = lp_flow(f_pert, time_limit=t_alloc * 0.5, t_init=0.2,
                               verbose=True, label="LP_r2_pert")
            try_seq(a_opt, "LP_r2_pert")

            remaining = deadline - time.monotonic()
            if remaining > 60:
                a_opt, M = lp_flow(best_seq, time_limit=min(remaining * 0.3, 150),
                                   t_init=0.15, verbose=True, label="LP_r2")
                try_seq(a_opt, "LP_r2")

    # === Phase 6: Try Interpolation to Different n ===
    print("\n=== Phase 6: Interpolation to Other n ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        target_ns = [512, 768, 1024, 1280, 1536, 2048, 2560]
        # Filter out n values close to current and too large
        cur_n = len(best_seq)
        target_ns = [n for n in target_ns if abs(n - cur_n) > cur_n * 0.15 and n <= 2560]

        for new_n in target_ns[:3]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            t_alloc = min(remaining * 0.15, 120)
            if t_alloc < 30:
                break

            a_interp = interpolate_to_n(best_seq, new_n)
            label = f"interp_n{new_n}"
            print(f"  {label}: time={t_alloc:.0f}s")
            a_opt, M = lp_flow(a_interp, time_limit=t_alloc, t_init=0.2,
                               verbose=False, label=label)
            try_seq(a_opt, label)

    # === Phase 7: Final CD + LP Polish ===
    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_cd = min(remaining * 0.4, 120)
            if t_cd > 15:
                a_opt, M = coordinate_descent(best_seq, time_limit=t_cd,
                                              verbose=False, label="CD_final")
                try_seq(a_opt, "CD_final")

            remaining = deadline - time.monotonic()
            if remaining > 40:
                t_lp = min(remaining * 0.5, 120)
                if t_lp > 15:
                    a_opt, M = lp_flow(best_seq, time_limit=t_lp, t_init=0.1,
                                       verbose=False, label="LP_final")
                    try_seq(a_opt, "LP_final")

        # Try symmetrized final
        if best_seq is not None:
            a_sym = symmetrize(best_seq)
            try_seq(a_sym, "final_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback used: score={best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
