Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on LSE-smoothed objective with softmax parameterization (continuation method)
2. LP-based Frank-Wolfe for high-precision polishing
3. Multi-length exploration
4. Final coordinate descent polish
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    x_max = x.max()
    e = np.exp(x - x_max)
    return e / e.sum()


def lse_obj_grad(x, beta):
    """
    LSE-smoothed max of auto-convolution, with softmax parameterization.
    Returns (objective, gradient w.r.t. x).
    """
    a = softmax(x)
    n = len(a)
    c = fast_convolve(a)
    
    # LSE
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    log_sum = bc_max + np.log(exp_bc.sum())
    obj = log_sum / beta
    
    # Gradient w.r.t. a
    w = exp_bc / exp_bc.sum()  # softmax(beta * c)
    
    # g_a = 2 * cross_correlate(w, a)
    # Using FFT: correlate(w, a)_j = sum_i w[i] * a[i+j] ... 
    # Actually, let me use the same formula as the previous solution
    # g_a[j] = 2 * sum_i a[i] * w[j+i] (with bounds)
    # = 2 * convolve(a, w_rev)[2n-2-j] (as in previous solution)
    
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g_a = 2.0 * conv[indices]
    
    # Chain rule: grad_x = a * (g_a - dot(g_a, a))
    dot = np.dot(g_a, a)
    grad_x = a * (g_a - dot)
    
    return obj, grad_x


def lbfqs_optimize(f0, time_limit, betas=None):
    """
    L-BFGS optimization with LSE smoothing and continuation.
    f0: initial point on the simplex
    Returns: best_f, best_unsmoothed_obj
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if betas is None:
        betas = [8, 16, 32, 64, 128, 256, 512, 1024]
    
    # Initialize x from f: x = log(f) + small constant (inverse of softmax)
    # Since f = softmax(x), x = log(f) + const (the const doesn't matter for softmax)
    # Use x = log(f + eps) as initial guess
    eps = 1e-300
    x = np.log(f + eps)
    
    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    
    t0 = time.monotonic()
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        
        stage_t0 = time.monotonic()
        stage_time = min(120, (time_limit - (time.monotonic() - t0)) * 0.2)
        
        result = minimize(
            lse_obj_grad,
            x,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': 5000,
                'maxfun': 5000,
                'ftol': 1e-15,
                'gtol': 1e-12,
                'maxls': 50,
            }
        )
        
        x = result.x
        f_new = softmax(x)
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            print(f"    L-BFGS beta={beta}: obj={obj_new:.10f} (improved)")
        else:

        # Update x for next beta (re-parameterize from best_f)
        # Actually, keep x as is (it should be close)
        f = f_new
    
    return best_f, best_obj

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
