    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_gradient(conv, temperature):
    """Compute soft weights and gradient of smooth max of conv."""
    shifted = conv / temperature
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    return w

    Compute subgradient of smooth_max(conv(f,f)) using FFT.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated projected
gradient descent with FFT-based subgradient computation, smooth-max annealing,
targeted coordinate descent on peak contributors, multi-length exploration,
and a short LP-FW polish phase. Key improvements over prior LP-FW: no LP
solves in the main loop (O(n log n) per iteration), Nesterov momentum for
accelerated convergence, adaptive temperature, and multi-scale initialization.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()


def smooth_weights(conv, temperature):
    """Soft-max weights over convolution values."""
    shifted = conv / temperature
    shifted -= shifted.max()
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def subgradient_fft(f, w):
    """
    Compute subgradient of smooth_max(conv(f,f)):
    SG[m] = 2 * sum_k w[k] * f[k-m]  (f[x]=0 outside [0,n-1]).
    This is 2 * cross_correlation(f, w)[m].
    """
    n = len(f)
    conv_len = 2 * n - 1
    fft_size = 1 << (n + conv_len - 1 + 1).bit_length()
    F = np.fft.fft(f, fft_size)
    W = np.fft.fft(w, fft_size)
    corr = np.fft.ifft(np.conj(F) * W).real
    return 2.0 * corr[:n]


def nesterov_smooth_descent(f0, time_limit, temp_init=0.003, lr_init=0.08,
                             momentum=0.85, verbose=False):
    """
    Nesterov-accelerated projected gradient descent with smooth-max objective.
    Uses FFT-based subgradient for O(n log n) per iteration.
    """
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    x = f.copy()
    v = np.zeros(n)

    best_obj = obj_val(f)
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    it = 0
    no_improve = 0
    best_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv_x = fast_convolve(x)
        w = smooth_weights(conv_x, temp)
        grad = subgradient_fft(x, w)

        # Heavy ball momentum step
        v = momentum * v - lr * grad
        x_new = project_simplex(x + v)

        obj_new = obj_val(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = x_new.copy()
            best_temp = temp
            no_improve = 0
        else:
            no_improve += 1

        x = x_new

        # Adaptive learning rate and temperature
        if no_improve > 80:
            lr *= 0.93
        if no_improve > 40:
            temp *= 1.03
        # Anneal temperature slowly
        if it % 500 == 0:
            temp *= 1.08

        if no_improve > 400:
            # Restart with noise around best
            rng = np.random.default_rng(it * 13 + 7)
            noise_scale = 0.003 * min(1.0, no_improve / 800.0)
            x = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = max(lr_init * 0.4, lr * 0.5)
            no_improve = 0

        if lr < 1e-9:
            lr = 1e-9

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NSD iter {it}: obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"temp={temp:.5f}, no_imp={no_improve}, t={elapsed:.1f}s")

    return best_f, best_obj


def coordinate_descent_peak(f0, time_limit, verbose=False):
    """
    Targeted coordinate descent: identifies peak contributors and adjusts them.
    """
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

    n = len(f)

    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 60:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Find top peaks
        top_k = min(15, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try adjusting top contributors
            order = np.argsort(products)[::-1]
            n_try = min(6, len(order))

            for oi in range(n_try):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.4, 0.6, 0.75, 0.85, 1.2, 1.4, 1.6]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.4:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute to non-contributing positions
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w_red = f[non_peak]
                            wsum = w_red.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w_red / wsum
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = obj_val(trial)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break  # break mult loop for this j_max
                if improved and no_improve == 0:
                    # Continue to next peak
                    pass

        if not improved:
            no_improve += 1
            # Aggressive: scale down all peak contributors
            if no_improve > 10:
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = f.copy()
                trial[js:je + 1] *= 0.93
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc_w = f[non_contrib]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = obj_val(trial)
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDescent iter {it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_f, best_obj


def build_lp_matrix(f, n):
    """Build constraint matrix for LP: (f*g)[k] for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def lp_fw_steps(f0, time_limit, max_iter=30, verbose=False):
    """
    Short LP-Frank-Wolfe polish: solve LP direction and take a step.
    Used only for final refinement.
    """
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    m_exp = 1.0

    A_ub = build_lp_matrix(f, n)

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1

        A_ub = build_lp_matrix(f, n)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': max(1, time_limit - (time.monotonic() - t0))})
        except Exception:
            break

        if not res.success or res.x is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                break
            continue

        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        # Try several step sizes
        best_alpha = 0.0
        best_alpha_obj = best_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            obj_new = obj_val(f_new)
            if obj_new < best_alpha_obj:
                best_alpha_obj = obj_new
                best_alpha = alpha

        if best_alpha < 1e-8:
            break

        f = f + best_alpha * (g - f)
        f = np.maximum(f, 0)
        f = f / f.sum()

        obj_cur = obj_val(f)
        if obj_cur < best_obj:
            best_obj = obj_cur
            best_f = f.copy()
            m_exp = 1.0
            conv = fast_convolve(f)
            M = conv.max()
        else:
            m_exp *= 1.1

        if verbose:
            print(f"      LP-FW step {it}: obj={best_obj:.8f}, alpha={best_alpha:.3f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width=0.8):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_core(n, core_frac=0.7):
    """Flat core with smooth edges."""
    x = np.abs(np.arange(n) - n / 2.0) / (n / 2.0)
    core_end = core_frac / 2.0
    edge = 1.0 - core_frac
    a = np.ones(n)
    mask = x > core_end
    if edge > 0:
        a[mask] = (1.0 - (x[mask] - core_end) / edge) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

        if rem > 20:
            t_alloc = min(rem, 80.0)
            print(f"  LP-FW: t={t_alloc:.0f}s, max_iter=15")
            a_opt, obj = lp_fw_steps(best_seq, time_limit=t_alloc, max_iter=15, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_polish")

    # === Phase 6: Final coordinate descent ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Final Coord Descent ===")
        rem = deadline - time.monotonic() - 20.0
        if rem > 10:
            t_alloc = min(rem, 60.0)
            a_opt, obj = coordinate_descent_peak(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "final_cd")

    # === Phase 7: One more Nesterov with very low temperature ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5.0
        if rem > 15:
            t_alloc = min(rem, 40.0)
            print(f"\n=== Phase 7: Final Nesterov (t={t_alloc:.0f}s) ===")
            a_opt, obj = nesterov_smooth_descent(best_seq, time_limit=t_alloc,
                                                 temp_init=0.0005, lr_init=0.02,
                                                 momentum=0.8, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "final_nesterov")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
