Let me try the **virtual gradient** approach:

The LP subproblem is:
`max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0`

The Lagrangian is:
`L(g, λ) = sum(g) - sum_k λ_k * ((f*g)[k] - M)`

At the optimum, the complementary slackness conditions give:
- For each `k` with `λ_k > 0`: `(f*g)[k] = M` (active constraint)
- The gradient condition: `1 - sum_k λ_k * (f shifted by k) = 0` for free variables

So the "virtual gradient" is `sum_k λ_k * (f shifted by k)`, which points in the direction that would reduce the objective if we move `f` in that direction.

This is essentially the subgradient of the max function, weighted by the dual variables!

So the algorithm becomes:
1. Solve the LP to get `g` and the dual variables `λ`.
2. Compute the "sharp gradient" `s[i] = sum_k λ_k * (f[i+k] + f[i-k])` (with boundary handling).
3. Take a step in the direction of `-s` (projected onto the simplex).

This should be more effective than the standard Frank-Wolfe step because it uses information from all active constraints simultaneously.

Multi-strategy optimizer: LP-Frank-Wolfe with adaptive relaxation,
fast multi-peak coordinate descent, diversified restarts, and multi-n exploration.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(f):
    """Objective for sum-normalized sequence."""
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

    """Build LP constraint matrix (dense)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n - 1, 2 * n - 2)
        A[j:end+1, j] = f[:end-j+1]
    return A

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def multi_peak_cd(f0, time_limit, n_peaks=10, n_movers=6, verbose=False):
    """Multi-peak coordinated coordinate descent."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = obj_norm(a)
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 300:
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        # Get top peaks
        k_peaks = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]
        
        # Compute weighted scores
        mean_c = 1.0 / (2*n - 1)
        weights = np.maximum(conv[peaks] - mean_c, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            weights = weights / weights.sum()
        
        scores = np.zeros(n)
        for pi in range(len(peaks)):
            k = int(peaks[pi])
            w = weights[pi]
            # Contribution of each index i to peak k
            for i in range(n):
                v = 0.0
                if 0 <= k - i < n:
                    v += a[k - i]
                if 0 <= i - k < n:
                    v += a[i - k]
                scores[i] += w * v
        
        # Try moves: reduce top scorers, increase bottom scorers
        top = np.argsort(scores)[::-1][:n_movers]
        bot = np.argsort(scores)[:n_movers]
        
        improved = False
        best_trial = None
        best_trial_obj = cur_obj
        
        for i in top:
            if a[i] < 1e-10:
                continue
            for j in bot:
                if j == i:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.4]:
                    t = a[i] * frac
                    if t < 1e-10:
                        continue
                    trial = a.copy()
                    trial[i] -= t
                    trial[j] += t
                    t_obj = 2.0 * n * fast_convolve(trial).max()
                    if t_obj < best_trial_obj - 1e-14:
                        best_trial_obj = t_obj
                        best_trial = trial.copy()
        
        if best_trial is not None:
            a = best_trial
            if best_trial_obj < best_obj:
                best_obj = best_trial_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                scale = 0.003 * min(1.0, no_improve / 50.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                no_improve = 0
        
        if verbose and (it if 'it' in dir() else 0) % 100 == 0:
            print(f"    MP-CD: obj={best_obj:.10f}, no_improve={no_improve}")
    
    return best_a, best_obj

Let me optimize the score computation using convolution:

The score for index i is: `sum_k w_k * (a[k-i] + a[i-k])` (with boundary handling).

This is essentially a weighted sum of "shifted" versions of a. We can compute it using convolution:

`scores[i] = sum_k w_k * (a[k-i] + a[i-k])`

Let me define `W[k] = w_k` for `k` in the peaks. Then:
- `sum_k w_k * a[k-i]` is the cross-correlation of `w` and `a` at shift `i`.
- `sum_k w_k * a[i-k]` is similar.

Actually, let me think more carefully. For a fixed peak at position `k`:
- The contribution of `a[i]` to `conv[k]` is: `a[i] * a[k-i]` (if `0 <= k-i < n`) plus `a[i] * a[i-k]`... 

Wait, no. The gradient of `conv[k]` with respect to `a[i]` is `a[k-i] + a[i-k]` (with boundary). So the "score" is:
`scores[i] = sum_k w_k * (a[k-i] + a[i-k]) * a[i]`

No wait, that's not right either. Let me re-derive.

`conv[k] = sum_m a[m] * a[k-m]`

`∂conv[k]/∂a[i] = a[k-i] + a[i-k]` (with a[x] = 0 for x outside [0,n-1])

So if we want to find which `a[i]` to change to reduce `conv[k]`, we look at `a[i] * ∂conv[k]/∂a[i]` = the actual contribution of `a[i]` to `conv[k]`.

The total contribution of `a[i]` to all top peaks is:
`scores[i] = sum_k w_k * a[i] * (a[k-i] + a[i-k])`

This is `a[i] * (sum_k w_k * a[k-i] + sum_k w_k * a[i-k])`

The first term `sum_k w_k * a[k-i]` is the cross-correlation of `w` and `a` evaluated at `i`.
The second term `sum_k w_k * a[i-k]` is the same (by symmetry of cross-correlation for real signals).

So `scores[i] = 2 * a[i] * (w ⊛ a)[i]` where ⊛ denotes cross-correlation.

We can compute this efficiently using FFT!

`(w ⊛ a)[i] = sum_k w[k] * a[k-i]` (with appropriate boundary handling)

Using FFT: `cross_corr = ifft(fft(w) * conj(fft(a)))`

But we need to be careful about the linear vs circular convolution distinction. Since `w` is sparse (only has values at the peak positions), we can just do a sparse computation.

Actually, for n=768 and at most 10 peaks, the sparse computation is only 10*768 = 7680 operations, which is very fast. Let me just do it directly:

For peak at position `k`:
- `∂conv[k]/∂a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)`

So the "responsibility score" for index i (how much reducing a[i] would help reduce peak k) is:
`a[i] * (a[k-i] + a[i-k])` (with zeros for out-of-bounds)

And the total score across all peaks is:
`scores[i] = sum_k w_k * a[i] * (a[k-i] + a[i-k])`

    # a[k-i] is non-zero when 0 <= k-i < n, i.e., k-n+1 <= i <= k
    # a[i-k] is non-zero when 0 <= i-k < n, i.e., k <= i <= k+n-1
    # Combined: max(0, k-n+1) <= i <= min(n-1, k+n-1)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k + n - 1)
    i_range = np.arange(i_lo, i_hi + 1)
    term1 = np.zeros(len(i_range))
    term2 = np.zeros(len(i_range))

    ki = k - i_range
    valid1 = (ki >= 0) & (ki < n)

    # a[i - k] for i in i_range
    ik = i_range - k
    valid2 = (ik >= 0) & (ik < n)

Now, the key question is: what moves to try?

**Move strategy**: 
- Pick the index `i` with the highest score (most responsible for peaks).
- Pick the index `j` with the lowest score (least responsible).
- Move mass from `i` to `j`.
- Find the optimal amount to move using a 1D line search.

But we also need to consider that moving mass from `i` to `j` might increase OTHER peaks. So we should evaluate the full objective after the move.

**Line search for the move amount t**:
The objective after moving `t` from `i` to `j` is:
`F(t) = 2*n*max_k conv_t[k]` where `conv_t` is the auto-convolution of the modified sequence.

This is a 1D non-smooth function (max of linear functions). We can use a ternary search or golden section search on `[0, a[i]]`.

Actually, since `F(t)` is the max of linear functions in `t` (each `conv_t[k]` is quadratic in `t` but approximately linear for small `t`), the function is approximately convex in `t` near 0. So a simple grid search + refinement should work.

def fast_cd(a0, time_limit, verbose=False):
    a = normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 500:
        conv = fast_convolve(a)
        # Get top peaks
        k_peaks = min(8, len(conv))
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]
        
        # Weights
        mean_c = 1.0 / (2*n-1)
        w = np.maximum(conv[peaks] - mean_c, 0)
        if w.sum() < 1e-15:
            w = np.ones(len(peaks))/len(peaks)
        else:
            w = w / w.sum()
        
        # Scores
        scores = np.zeros(n)
        for pi in range(len(peaks)):
            k = int(peaks[pi])
            wi = w[pi]
            i_lo = max(0, k-n+1)
            i_hi = min(n-1, k+n-1)
            i_range = np.arange(i_lo, i_hi+1)
            ki = k - i_range
            ik = i_range - k
            t1 = np.zeros(len(i_range))
            t2 = np.zeros(len(i_range))
            v1 = (ki >= 0) & (ki < n)
            v2 = (ik >= 0) & (ik < n)
            t1[v1] = a[ki[v1]]
            t2[v2] = a[ik[v2]]
            scores[i_range] += wi * a[i_range] * (t1 + t2)
        
        # Best move
        top_i = np.argmax(scores)
        # Find best j to move to
        # Exclude top_i's neighborhood
        candidate_scores = scores.copy()
        candidate_scores[top_i] = -1  # exclude source
        bot_j = np.argmin(candidate_scores)
        
        # Line search for t
        best_t = 0
        best_trial_obj = obj_norm(a)
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            t = a[top_i] * frac
            if t < 1e-12:
                break
            trial = a.copy()
            trial[top_i] -= t
            trial[bot_j] += t
            t_obj = 2.0 * n * fast_convolve(trial).max()
            if t_obj < best_trial_obj:
                best_trial_obj = t_obj
                best_t = t
        
        if best_t > 0:
            a[top_i] -= best_t
            a[bot_j] += best_t
            a = np.maximum(a, 0)
            cur = obj_norm(a)
            if cur < best_obj:
                best_obj = cur
                best_a = a.copy()
            no_improve = 0

                scale = 0.01 * min(1.0, no_improve/100.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                no_improve = 0
    
    return best_a, best_obj
