from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def objective_and_grad(z, T, n):
    """Compute smoothed objective and gradient with respect to z."""
    a = softmax(z)
    conv = fast_convolve(a)
    # Log-sum-exp of T * conv
    Tconv = T * conv
    max_Tconv = Tconv.max()
    log_sum_exp = max_Tconv + np.log(np.exp(Tconv - max_Tconv).sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Weights w_k
    w = np.exp(Tconv - max_Tconv)
    w /= w.sum()
    
    # Gradient: d f_T / d z_l = (4n/T) * a_l * (sum_k w_k * a_{k-l} - sum_k w_k * conv[k])
    # cross_corr(w, a)_l = sum_k w_k * a_{k-l}
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_corr(w, a)[l] = sum_k w_k * a_{k-l}
    # This is the cross-correlation, which is ifft(conj(W) * A)
    cross = irfft(np.conj(W) * A, fft_size)[:n]

    grad = (4.0 * n / T) * a * (cross - weighted_conv_sum)
    
    return obj, grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent in log-space
with adaptive log-sum-exp temperature, combined with coordinate descent local search
and multi-length exploration. Uses FFT-based convolution and cross-correlation for
efficient gradient computation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def cross_corr_w_a(w, a, fft_size):
    """Compute cross_l = sum_k w_k * a_{k-l} for l=0,...,n-1."""
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(W * np.conj(A), fft_size)[:len(a)]

def compute_grad(z, T, n, fft_size):
    """Compute gradient of smoothed objective with respect to z."""
    a = softmax(z)
    conv = fast_convolve(a)
    Tconv = T * conv
    max_Tconv = Tconv.max()
    exp_Tconv = np.exp(Tconv - max_Tconv)
    w = exp_Tconv / exp_Tconv.sum()
    
    cross = cross_corr_w_a(w, a, fft_size)
    weighted_c = np.dot(w, conv)
    
    grad = 4.0 * n * a * (cross - weighted_c)
    return a, conv, grad

def adam_step(z, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v

def smoothed_grad_descent(a_init, time_limit, T_start=20.0, T_end=2000.0, lr_init=0.05):
    """Main optimization loop using smoothed gradient descent with Adam."""
    n = len(a_init)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Initialize z from a_init
    a_safe = np.maximum(a_init, 1e-30)
    z = np.log(a_safe)
    z = z - z.mean()  # Center for numerical stability
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1
    lr = lr_init
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_convolve(a_init).max()
    
    t0 = time.monotonic()
    no_improve = 0
    total_steps = 0
    
    while time.monotonic() - t0 < time_limit:
        # Determine T based on progress
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute gradient
        a, conv, grad = compute_grad(z, T, n, fft_size)
        
        # Adam update
        z, m, v = adam_step(z, grad, m, v, t, lr)
        z = z - z.mean()  # Re-center
        
        a = softmax(z)
        score = 2.0 * n * fast_convolve(a).max()
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        total_steps += 1
        t += 1
        
        # Learning rate decay
        lr = lr_init / (1 + 0.005 * total_steps)
        
        # Periodic reheat
        if no_improve > 500:
            # Reheat: reduce T, increase lr
            T = T_start * 2
            lr = lr_init * 0.5
            no_improve = 0
            # Add small noise
            rng = np.random.default_rng(total_steps)
            z += rng.standard_normal(n) * 0.01
            z = z - z.mean()
        
        if total_steps % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD: t={total_steps} ({elapsed:.0f}s) T={T:.1f} lr={lr:.4f} score={best_score:.6f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent in log-space (softmax parameterization)
   with adaptive log-sum-exp temperature and Adam optimizer.
2. Coordinate descent with golden-section search for local refinement.
3. Peak-aware local search (identifying and reducing peak-contributing coordinates).
4. Multi-length exploration.

Key mathematical insight: The gradient of the smoothed objective
f_T(z) = (2n/T) * logsumexp(T * conv(softmax(z), softmax(z)))
with respect to the log-space variable z is:
  grad_l = 4n * a_l * (cross_l - weighted_c)
where cross_l = sum_k w_k * a_{k-l} (cross-correlation of peak weights w
with the sequence a) and weighted_c = sum_k w_k * conv[k] (weighted mean
of the convolution). This can be computed in O(n log n) via FFT.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def compute_smoothed_obj_grad(z, T, n, fft_size):
    """
    Compute smoothed objective and gradient.
    Returns (objective, a, gradient).
    """
    a = softmax_fn(z)
    conv = fast_convolve(a)
    
    # Log-sum-exp of T * conv
    Tconv = T * conv
    max_Tconv = Tconv.max()
    exp_Tconv = np.exp(Tconv - max_Tconv)
    log_sum_exp = max_Tconv + np.log(exp_Tconv.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Peak weights
    w = exp_Tconv / exp_Tconv.sum()
    
    # Cross-correlation: cross_l = sum_k w_k * a_{k-l}
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    
    # Weighted mean of convolution
    weighted_c = np.dot(w, conv)
    
    # Gradient
    grad = 4.0 * n * a * (cross - weighted_c)
    
    return obj, a, grad


def smoothed_grad_descent(a_init, time_limit, T_start=30.0, T_end=3000.0, lr_init=0.03):
    """
    Smoothed gradient descent with Adam optimizer and adaptive temperature.
    """
    n = len(a_init)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Initialize z from a_init
    a_safe = np.maximum(a_init, 1e-30)
    z = np.log(a_safe)
    z = z - z.mean()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 1
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_convolve(a_init).max()
    
    t0 = time.monotonic()
    no_improve = 0
    step = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        # Adaptive temperature: increase over time
        progress = min(1.0, step / max(1, int(time_limit * 10)))  # ~10 steps per second
        T = T_start * (T_end / T_start) ** (0.7 * progress)
        
        # Compute gradient
        obj, a, grad = compute_smoothed_obj_grad(z, T, n, fft_size)
        
        # Normalize gradient for numerical stability
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e10:
            grad = grad / grad_norm * 1e10
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t_adam)
        v_hat = v / (1 - beta2**t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.mean()  # Re-center for numerical stability
        
        t_adam += 1
        step += 1
        
        # Evaluate
        a_new = softmax_fn(z)
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Learning rate schedule
        lr = lr_init / (1.0 + 0.002 * step)
        
        # Periodic reheat to escape local minima
        if no_improve > 800:

            z += rng.standard_normal(n) * noise_scale
            z = z - z.mean()
            no_improve = 0
        
        # Progress reporting
        if step % 2000 == 0:

    return best_a, best_score


def coordinate_descent(a_init, time_limit, n_golden_evals=25):
    """
    Coordinate descent with golden-section search.
    For each coordinate i, find the optimal value of a_i
    (with others rescaled to maintain sum=1).
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, 2.0 * n * fast_convolve(a_init).max()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    pass_num = 0
    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            old_ai = a[i]
            others_sum = 1.0 - old_ai
            if others_sum < 1e-12:
                continue
            
            # Golden section search
            lo = 0.0
            hi = 1.0 - 1e-12
            best_t = old_ai
            best_val = np.inf
            
            for _ in range(n_golden_evals):
                m1 = lo + 0.381966 * (hi - lo)
                m2 = hi - 0.381966 * (hi - lo)
                
                # Evaluate at m1
                a_trial = a.copy()
                a_trial[i] = m1
                if m1 < 1.0 - 1e-14:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a_trial[mask] = a[mask] * (1.0 - m1) / others_sum
                val1 = fast_convolve(a_trial).max()
                if val1 < best_val:
                    best_val = val1
                    best_t = m1
                
                # Evaluate at m2
                a_trial[i] = m2
                if m2 < 1.0 - 1e-14:
                    a_trial[mask] = a[mask] * (1.0 - m2) / others_sum
                else:
                    a_trial[mask] = 0.0
                val2 = fast_convolve(a_trial).max()
                if val2 < best_val:
                    best_val = val2
                    best_t = m2
                
                if val1 <= val2:
                    hi = m2
                else:
                    lo = m1
            
            # Apply update
            if abs(best_t - old_ai) > 1e-10:
                a[i] = best_t
                if best_t < 1.0 - 1e-14:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a[mask] = a[mask] * (1.0 - best_t) / others_sum
                else:
                    a[mask] = 0.0
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
        if pass_num % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_num} ({elapsed:.0f}s) score={best_score:.8f}")
    
    return best_a, best_score


def peak_shave(a_init, time_limit):
    """

    t0 = time.monotonic()

            break
        
        conv = fast_convolve(a)
        # Find peak (exclude first and last which are trivially small)
        conv_mid = conv.copy()
        conv_mid[0] = -np.inf
        conv_mid[-1] = -np.inf
        peak_k = int(np.argmax(conv_mid))

            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors

            old_val = a[i]

                continue
            
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                a_trial = a.copy()

                diff = old_val * (1 - factor)
                if diff < 1e-15:
                    continue

                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()

                    break

            quiet_mask[idx] = False

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
