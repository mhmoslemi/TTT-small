def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_topk(f, c, k, temperature=0.0):
    """
    Compute subgradient using top-k peaks.
    If temperature > 0, uses softmax weights; otherwise equal weights.
    """
    n = len(f)
    conv_len = 2 * n - 1
    
    if temperature > 0:
        top_k = np.argpartition(c, -k)[-k:]
        vals = c[top_k]
        vals = vals - vals.max()
        w = np.exp(vals / temperature)
        w /= w.sum()
    else:
        top_k = np.argpartition(c, -k)[-k:]
        w = np.ones(k) / k
    
    # g_j = 2 * sum_{k in top_k} w_idx(k) * f[k-j] * [0 <= k-j < n]
    g = np.zeros(n)
    for idx in range(len(top_k)):
        if w[idx] < 1e-12:
            continue
        k = int(top_k[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        indices = np.arange(js, je + 1)
        g[indices] += w[idx] * 2.0 * f[k - indices]
    
    return g

def nesterov_subgradient(f0, time_limit, k=10, verbose=False):
    """
    Nesterov-accelerated projected subgradient method with Polyak step size.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = float(c.max())
    best_f = f.copy()
    
    # Nesterov momentum
    y = f.copy()
    t = 1.0
    
    # Polyak step size parameters
    # We need a lower bound on the optimal value. Use the trivial bound: 1/(2n-1)
    # Actually, for sum(a)=1, min max(conv(a,a)) >= 1/(2n-1) (by averaging)
    # But a better bound: since (a*a)[n-1] = ||a||^2 >= 1/n, we have max >= 1/n
    # So the optimal value is in [1/n, 1] (roughly)
    obj_lb = 1.0 / n  # lower bound
    obj_ub = best_obj  # upper bound (current best)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t = it  # Nesterov momentum parameter
        
        c = fast_convolve(y)
        g = subgradient_topk(y, c, k)
        g_norm = np.linalg.norm(g)
        
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        step = (c.max() - obj_lb) / (g_norm ** 2 + 1e-15)
        step = min(step, 1.0)  # cap
        
        # Nesterov step
        y_new = y - step * g / g_norm  # normalize step direction
        y_new = project_simplex(y_new)
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        f_new = y_new + beta * (y_new - f)
        f_new = project_simplex(f_new)
        
        f = f_new
        y = y_new
        t = t_new
        
        c = fast_convolve(f)
        obj = c.max()
        
        if obj < best_obj - 1e-14:
            best_obj = float(obj)
            best_f = f.copy()
            no_improve = 0
            obj_ub = best_obj
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            y = f.copy()
            t = 1.0
            no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    NS iter {it}: obj={obj:.10f}, best={best_obj:.10f}")
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

N_simplex(a) = {v : v_j <= 0 for j with a_j = 0, and the "free" components sum to 0}

This is getting complex. Let me just implement a practical version:

**Practical Active-Set Method:**
1. Start with a good initial point
2. Compute c = conv(a,a), M = max(c)
3. Active set S = {k : c[k] >= M - epsilon}
4. Solve QP: min a^T Q_S a s.t. sum(a) = 1, a >= 0
   where Q_S = sum_{k in S} (1/|S|) B_k
5. If the solution a_new has max(conv(a_new)) <= M + delta, accept and update S
6. If there's a new peak k* with c_{k*}(a_new) > M, add k* to S and go to step 4
7. If a peak in S is no longer active (c_k(a_new) < M - 2*delta), remove it from S
8. Repeat until convergence

The QP in step 4 can be solved by the active-set method for QP:
- Start with all n variables "free"
- Solve Q_S * a = mu * 1 (find mu so sum(a) = 1)
- If any a_j < 0, fix a_j = 0 and re-solve with fewer variables
- Continue until all free variables have a_j >= 0

For the linear system Q_S * a_free = mu * 1_free:
- Q_S is symmetric PSD
- We need to solve (Q_S_reduced) * a_free = mu * 1_free where Q_S_reduced is the submatrix for free variables
- mu is determined by the constraint sum(a_free) = 1

We can compute mu as: mu = 1 / (1^T * Q_S_reduced^{-1} * 1) (if Q_S_reduced is PD)
Or more generally, solve the augmented system.

For efficiency, I'll use numpy's solve for the linear system (n is at most ~2048, so this is fine).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an active-set Newton method
with multi-scale exploration. The key innovation is a second-order method
that solves equality-constrained QPs on the active set of peaks, achieving
much faster local convergence than first-order methods like LP-FW.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_qp_simplex(Q, n):
    """
    Solve min a^T Q a s.t. sum(a) = 1, a >= 0.
    Q is a function that computes Q @ v for a given v (to avoid storing Q explicitly).
    Returns the solution a.
    """
    # Active set method for QP
    free = np.ones(n, dtype=bool)
    a = np.ones(n) / n
    
    for outer in range(n):  # at most n iterations
        n_free = int(free.sum())
        if n_free <= 1:
            # Only one free variable, it must be 1
            a = np.zeros(n)
            a[np.nonzero(free)[0][0]] = 1.0
            return a
        
        # Build the reduced Q matrix for free variables
        idx = np.nonzero(free)[0]
        Q_reduced = np.zeros((n_free, n_free))
        for i in range(n_free):
            e_i = np.zeros(n)
            e_i[idx[i]] = 1.0
            Qv = Q(e_i)
            for j in range(i, n_free):
                val = Qv[idx[j]]
                Q_reduced[i, j] = val
                Q_reduced[j, i] = val
        
        # Solve [Q_r  1; 1'  0] [a_r; mu] = [0; 1]
        # Or equivalently: Q_r * a_r = mu * 1, sum(a_r) = 1
        # From the first: a_r = mu * Q_r^{-1} * 1
        # From the second: mu * 1^T * Q_r^{-1} * 1 = 1, so mu = 1 / (1^T Q_r^{-1} 1)
        
        one = np.ones(n_free)
        try:
            # Solve Q_r * x = 1
            x = np.linalg.solve(Q_reduced + 1e-12 * np.eye(n_free), one)
            mu = 1.0 / np.dot(one, x)
            a_r = mu * x
        except np.linalg.LinAlgError:
            # Fallback: use pinv
            x = np.linalg.pinv(Q_reduced + 1e-10 * np.eye(n_free)) @ one
            mu = 1.0 / np.dot(one, x)
            a_r = mu * x
        
        # Check if all free variables are non-negative
        if np.all(a_r >= -1e-10):
            a = np.zeros(n)
            a[idx] = np.maximum(a_r, 0)
            a = a / a.sum()  # renormalize
            return a
        
        # Remove the most negative variable from the free set
        min_idx = idx[np.argmin(a_r)]
        free[min_idx] = False
        a[min_idx] = 0.0
    
    # Fallback
    a = np.maximum(a_r, 0) if n_free > 0 else np.ones(n) / n
    a_full = np.zeros(n)
    a_full[idx] = a
    a_full = np.maximum(a_full, 0)
    s = a_full.sum()
    if s > 1e-15:
        a_full /= s
    else:
        a_full = np.ones(n) / n
    return a_full

B_k is defined so that for any vector v:
(B_k @ v)[j] = v[k-j] if 0 <= k-j < n, else 0

So (B_k @ v)[j] = v[k-j] for j in [max(0, k-n+1), min(n-1, k)]

Let me verify: (a*a)[k] = sum_j a[j] * a[k-j]. In matrix form, this is a^T B_k a where (B_k)[j, i] = 1 if j + i = k (and both j, i in [0, n-1]).

So B_k[j, i] = 1 if i = k - j and 0 <= k-j < n.
(B_k @ a)[j] = sum_i B_k[j, i] * a[i] = a[k-j] if 0 <= k-j < n.
a^T B_k a = sum_j a[j] * (B_k @ a)[j] = sum_j a[j] * a[k-j]. ✓

So for the QP, Q = sum_k lambda_k * B_k, and:
Q @ a = sum_k lambda_k * B_k @ a = sum_k lambda_k * (shift of a by k)

Specifically: (Q @ a)[j] = sum_k lambda_k * a[k-j] * [0 <= k-j < n]

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set QP method with FFT-based matvec for fast local convergence
2. Nesterov-accelerated projected subgradient for global exploration
3. LP-FW with adaptive constraint reduction
4. Multi-scale (multi-n) search with interpolation
5. Diverse restarts to escape local minima
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def Q_matvec(v, active_peaks, weights, n):
    """Compute Q @ v where Q = sum_k weights[k] * B_k."""
    Qv = np.zeros(n)
    for idx in range(len(active_peaks)):
        k = int(active_peaks[idx])
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        Qv[js:je + 1] += w * v[k - (np.arange(js, je + 1))]
    return Qv


def solve_qp_active_set(f0, active_peaks, weights, n, tol=1e-10, max_iter=500):
    """
    Solve min a^T Q a s.t. sum(a)=1, a>=0 using active-set + CG.
    Q = sum_k weights[k] * B_k (given via active_peaks and weights).
    Returns solution a and the QP objective value.
    """
    free = np.ones(n, dtype=bool)
    a = np.ones(n) / n
    
    for outer in range(max_iter):
        n_free = int(free.sum())
        if n_free <= 1:
            a = np.zeros(n)
            fn_idx = np.nonzero(free)[0]
            if len(fn_idx) > 0:
                a[fn_idx[0]] = 1.0
            return a, 0.0
        
        idx = np.nonzero(free)[0]
        
        # Solve Q_r * a_r = mu * 1 using CG on the free subspace
        # We need: Q_r @ a_r = mu * 1, and 1' @ a_r = 1
        # => mu = 1 / (1' @ Q_r^{-1} @ 1)
        # => a_r = mu * Q_r^{-1} @ 1
        
        # Use CG to solve Q_r @ x = 1
        one_free = np.ones(n_free)
        
        # CG on the free subspace
        def cg_solve(b, max_cg=200):
            """Solve Q_r @ x = b where Q_r is the restriction of Q to free indices."""
            x = np.zeros(n_free)
            r = b.copy()
            p = r.copy()
            rs = np.dot(r, r)
            
            for cg_it in range(max_cg):
                # Q_r @ p = (Q @ p_full)[free]
                p_full = np.zeros(n)
                p_full[idx] = p
                Qp = Q_matvec(p_full, active_peaks, weights, n)
                Qp_r = Qp[idx]
                
                alpha = rs / (np.dot(p, Qp_r) + 1e-15)
                x += alpha * p
                r -= alpha * Qp_r
                rs_new = np.dot(r, r)
                if np.sqrt(rs_new) < tol * (np.dot(b, b) + 1e-15):
                    break
                p = r + (rs_new / (rs + 1e-15)) * p
                rs = rs_new
            
            return x
        
        x = cg_solve(one_free)
        denom = np.dot(one_free, x)
        
        if abs(denom) < 1e-20:
            # Q is singular on the free set, use regularization
            a = np.zeros(n)
            a[idx] = one_free / n_free
            return a, 0.0
        
        mu = 1.0 / denom
        a_r = mu * x
        
        # Check non-negativity
        min_val = np.min(a_r)
        if min_val >= -1e-10:
            a = np.zeros(n)
            a[idx] = np.maximum(a_r, 0)
            s = a.sum()
            if s > 1e-15:
                a /= s
            return a, float(np.dot(a, Q_matvec(a, active_peaks, weights, n)))
        
        # Remove the most negative variable
        worst = idx[np.argmin(a_r)]
        free[worst] = False
    
    # Fallback: return uniform on free set
    a = np.zeros(n)
    idx = np.nonzero(free)[0]
    if len(idx) > 0:
        a[idx] = 1.0 / len(idx)
    else:
        a = np.ones(n) / n
    return a, float(np.dot(a, Q_matvec(a, active_peaks, weights, n)))


def active_set_newton(f0, time_limit, verbose=False):
    """
    Active-set Newton method for min max_k (a*a)[k] s.t. a>=0, sum(a)=1.
    Iteratively solves QPs on the active set of peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = float(c.max())
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    
    # Initial active set: top peaks
    M = c.max()
    k_peaks = min(8, 2 * n - 1)
    active = np.argpartition(c, -k_peaks)[-k_peaks:]
    active = np.sort(active)
    weights = np.ones(len(active)) / len(active)
    
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Update active set based on current solution
        c = fast_convolve(f)
        M = c.max()
        gap = M - np.median(c)
        
        # Active peaks: those within a fraction of the max
        threshold = M - max(0.05 * gap, 1e-6)
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 20:
            active = np.where(c >= M - 0.02 * np.ptp(c))[0]
            if len(active) > 20:
                active = np.argpartition(c, -20)[-20:]
        
        active = np.sort(active)
        
        # Equal weights (or proportional to peak values)
        peak_vals = c[active]
        weights = peak_vals / peak_vals.sum()
        
        # Solve QP: min a^T Q a s.t. sum(a)=1, a>=0
        a_qp, qp_obj = solve_qp_active_set(f, active, weights, n)
        
        if a_qp is None or not np.all(np.isfinite(a_qp)):
            stagnation += 1
            if stagnation > 30:
                break
            continue
        
        # Check if the QP solution actually reduces the true objective
        c_qp = fast_convolve(a_qp)
        obj_qp = float(c_qp.max())
        
        # Try a line search between f and a_qp
        best_alpha = 0.0
        best_trial_obj = M
        
        for alpha in [0.1, 0.25, 0.5, 0.75, 1.0]:
            trial = (1 - alpha) * f + alpha * a_qp
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            trial_c = fast_convolve(trial)
            trial_obj = float(trial_c.max())
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = (1 - best_alpha) * f + best_alpha * a_qp
            f = np.maximum(f, 0)
            f = f / f.sum()
            
            c = fast_convolve(f)
            new_obj = float(c.max())
            
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1
        
        if verbose and it % 5 == 0:
            print(f"    ASN iter {it}: obj={float(c.max()):.10f}, best={best_obj:.10f}, |active|={len(active)}")
        
        # Check convergence
        if stagnation > 80:
            # Perturb and restart
            rng = np.random.default_rng(it * 37 + 13)
            noise = rng.standard_normal(n) * 0.003
                f = project_simplex(best_f + noise)
            stagnation = 0
        
        if it > 5000:
            break
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, k_peaks=8, verbose=False):
    """
    Projected subgradient with Polyak-like step size and Nesterov momentum.
    Uses top-k peaks for the subgradient.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = float(c.max())
    best_f = f.copy()
    
    # Nesterov momentum
    y = f.copy()
    t_param = 1.0
    
    # Lower bound: 1/n (from ||a||^2 >= 1/n at the center peak)
    obj_lb = 1.0 / n
    obj_gap = best_obj - obj_lb
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_param += 1.0
        
        c = fast_convolve(y)
        M = c.max()
        
        # Subgradient from top-k peaks
        top_k = np.argpartition(c, -k_peaks)[-k_peaks:]
        peak_vals = c[top_k]
        # Softmax weights
        max_pv = peak_vals.max()
        w = np.exp((peak_vals - max_pv) / (max_pv * 0.1 + 1e-10))
        w /= w.sum()
        
        g = np.zeros(n)
        for idx in range(len(top_k)):
            k = int(top_k[idx])
            if w[idx] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            indices = np.arange(js, je + 1)
            g[indices] += w[idx] * 2.0 * f[k - indices]
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        step = (M - obj_lb) / (g_norm ** 2 + 1e-15)
        step = min(step, 5.0 / g_norm)
        
        # Nesterov update
        y_new = y - step * g
        y_new = project_simplex(y_new)
        
        beta = (t_param - 1.0) / ((1.0 + np.sqrt(1.0 + 4.0 * t_param ** 2)) / 2.0)
        f_new = y_new + beta * (y_new - f)
        f_new = project_simplex(f_new)
        
        f = f_new
        y = y_new
        
        c = fast_convolve(f)
        obj = float(c.max())
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 41 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            y = f.copy()
            t_param = 1.0
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    Subgrad iter {it}: obj={obj:.10f}, best={best_obj:.10f}")
    
    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, n_constraints=None, verbose=False):
    """
    LP-based Frank-Wolfe with optional constraint reduction.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = float(M)
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Determine constraint set
        if n_constraints is not None and n_constraints < conv_len:
            active_mask = np.zeros(conv_len, dtype=bool)
            top_idx = np.argpartition(c, -n_constraints)[-n_constraints:]
            active_mask[top_idx] = True
        else:
            active_mask = np.ones(conv_len, dtype=bool)
        
        active_indices = np.where(active_mask)[0]
        n_cons = len(active_indices)
        
        # Build LP: max 1'g s.t. A_ub @ g <= b_ub, g >= 0
        # Row i of A_ub corresponds to constraint k=active_indices[i]:
        # (f*g)[k] = sum_j f[j]*g[k-j] <= M
        # For variable g[i], it appears in constraint k with coefficient f[k-i]
        A_ub = np.zeros((n_cons, n))
        for ci, k in enumerate(active_indices):
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            A_ub[ci, i_min:i_max + 1] = f[k - np.arange(i_min, i_max + 1)]
        
        b_ub = np.full(n_cons, M)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            g = res.x if res.success else None
        except:
            g = None
        
        if g is None or g.sum() < 1e-15:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                c = fast_convolve(f)
                M = c.max()
                no_improve = 0
            continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 80:
                break
            continue
        else:
            no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.05, 1.0, 10)
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0)
            f = f / f.sum()
            c = fast_convolve(f)
            M = float(c.max())
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()

                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}")
        
        if no_improve > 150:
            break
    
    return best_f, best_obj


def make_shape(n, shape_type="raised_cos"):
    """Create various initial shapes."""
    if shape_type == "gaussian":
        x = np.arange(n) - n / 2.0
        a = np.exp(-0.5 * (x / (n * 0.25)) ** 2)
    elif shape_type == "bell":
        x = np.arange(n) - n / 2.0
        hw = n * 0.35
        t = np.clip(np.abs(x) / hw, 0, 1.0)
        a = 1.0 - t ** 4
    elif shape_type == "raised_cos":
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif shape_type == "flat_top":
        x = np.linspace(-1, 1, n)
        a = np.where(np.abs(x) < 0.5, 1.0, 0.5 * (1 + np.cos(2 * np.pi * (np.abs(x) - 0.5))))
    elif shape_type == "sqrt_cos":
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length with light smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Light smoothing kernel
    kernel = np.array([0.05, 0.2, 0.5, 0.2, 0.05])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    s = f_smooth.sum()
    if s > 1e-15:
        f_smooth /= s
    return f_smooth
