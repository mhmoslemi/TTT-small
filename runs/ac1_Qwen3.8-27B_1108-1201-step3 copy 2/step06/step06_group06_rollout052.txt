import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def linear_ramp(n):
    """Linearly increasing sequence, normalized to sum=1."""
    a = np.arange(1, n + 1, dtype=np.float64)

def ramp_with_dip(n, dip_strength=0.3):
    """Linear ramp with a quadratic dip in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) - dip_strength * i * (n - 1 - i) / (n * (n + 1) / 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def ramp_with_oscillation(n, freq=5.0, amp=0.1, phase=0.0):
    """Linear ramp with sinusoidal modulation."""
    i = np.arange(n, dtype=np.float64)
    base = (i + 1)
    mod = 1.0 + amp * np.sin(2 * np.pi * freq * i / n + phase)
    a = base * mod
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:

    return a

def compute_smooth_grad(f, tau):

    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()

def compute_subgrad(f):

    n = len(f)
