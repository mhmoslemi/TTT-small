def fft_convolve(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lp_fw_step(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)

    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def interpolate_seq(a, new_n):
    """Linearly interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    out = np.interp(new_x, old_x, a)
    out = np.maximum(out, 0)
    s = out.sum()
    if s > 1e-15:
        out /= s
    return out


def make_gaussian(n, sigma_frac=0.3):

    t0 = time.monotonic()

    eta = 0.15
    momentum_coeff = 0.5
    v = np.zeros(n)
    K = 5
    it = 0
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.1:
            break

        conv = fft_convolve(a)
        M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      SGD it={it}: obj={2*n*M:.8f}")
        else:
            no_improve += 1

        # Top-K peaks with softmax weighting
        k_actual = min(K, len(conv))
        peak_idx = np.argpartition(conv, -k_actual)[-k_actual:]
        pvals = conv[peak_idx]
        pmax = pvals.max()
        temp = 0.05 * pmax + 1e-15
        weights = np.exp((pvals - pmax) / temp)
        weights /= weights.sum()

        # Subgradient
        sg = np.zeros(n)
        for w, k in zip(weights, peak_idx):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            indices = np.arange(js, je + 1)
            sg[indices] += w * a[k - indices]

        # Momentum
        v = momentum_coeff * v + (1.0 - momentum_coeff) * sg
        v_mean = v.mean()
        if v_mean > 1e-15:
            v = v / v_mean

        # Multiplicative update
        a = a * np.exp(-eta * v)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # Adapt parameters
        eta *= 0.9998
        if it % 1000 == 0:
            K = min(K + 1, 30)
        momentum_coeff = min(momentum_coeff + 0.002, 0.85)

        # Random perturbation if stuck
        if no_improve > 800:
            a = a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            eta = 0.08
            no_improve = 0
            v = np.zeros(n)

        if no_improve > 3000:
            break

    return best_a, best_obj


def peak_targeted_cd(a0, time_limit, verbose=False, seed=0):
    """
    Peak-targeted coordinate descent: reduces the dominant peak by
    moving mass from contributing to non-contributing coordinates.
    """
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(a.size) / a.size
    else:
        a /= s
    n = len(a)

    conv = fft_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 200:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        conv = fft_convolve(a)
        M = conv.max()

        # Pick a peak (not always the max, to escape local minima)
        if it % 10 < 5:
            peak_k = int(np.argmax(conv))
        else:
            top_peaks = np.argsort(conv)[::-1][:5]
            peak_k = int(top_peaks[rng.integers(len(top_peaks))])

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            no_improve += 1
            continue
        contrib = np.arange(js, je + 1)
        products = a[contrib] * a[peak_k - contrib]
        total = products.sum()
        if total < 1e-15:
            no_improve += 1
            continue

        # Find most contributing coordinate
        max_local = np.argmax(products)
        src = contrib[max_local]

        # Try reducing src and redistributing
        best_trial = None
        best_trial_obj = M

        fracs = [0.05, 0.1, 0.15, 0.25, 0.4, 0.6]
        for frac in fracs:
            diff = a[src] * frac
            if diff < 1e-15:
                continue
            trial = a.copy()
            trial[src] -= diff

            # Target: coordinates not contributing to this peak
            nc_mask = np.ones(n, dtype=bool)
            nc_mask[contrib] = False
            nc_mask[src] = False

            nc_indices = np.where(nc_mask)[0]
            if len(nc_indices) < 1:
                nc_mask[:] = True
                nc_mask[src] = False
                nc_indices = np.where(nc_mask)[0]

            if len(nc_indices) == 0:
                continue

            # Distribute: preferentially to low-valued coords
            nc_vals = trial[nc_indices]
            spread = nc_vals.max() - nc_vals.min()
            if spread > 1e-15:
                weights = (nc_vals.max() - nc_vals) + 0.1 * nc_vals.max()
            else:
                weights = np.ones(len(nc_indices))
            weights = np.maximum(weights, 1e-10)
            weights /= weights.sum()
            trial[nc_indices] += diff * weights

            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial /= ts
            obj = fft_convolve(trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_trial = trial

        if best_trial is not None:
            a = best_trial
            no_improve = 0
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_a = a.copy()
                if verbose and it % 100 == 0:
                    print(f"      PTC it={it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1

    return best_a, best_obj


def lp_fw_step(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    # Build Toeplitz-like structure
    for j in range(n):
        cols = np.arange(j, j + n)
        rows = np.arange(0, n)
        # f*g at lag k: sum_j f[j] * g[k-j]
        # For fixed j, g[k-j] appears at row k where k = j + col_idx
        # A_ub[k, j] = f[j] if 0 <= k-j < n, i.e., k in [j, j+n-1]
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fft_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_levels = [1.0, 1.01, 1.03, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0]
    m_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3.0:
            break

        conv = fft_convolve(f)
        M = conv.max()

        g = None
        for level_try in range(m_idx, len(m_levels)):
            M_try = M * m_levels[level_try]
            g = lp_fw_step(f, M_try, n)
            if g is not None and g.sum() > 1e-12:
                m_idx = level_try
                break
            m_idx = min(level_try + 1, len(m_levels) - 1)

        if g is None or g.sum() < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            # Perturb and retry
            rng = np.random.default_rng(it * 7 + 3)
            f = f + rng.standard_normal(n) * 0.005
            f = np.maximum(f, 0)
            f /= f.sum()
            m_idx = 0
            continue

        g = g / g.sum()

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fft_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 20:
                break
            continue

        f = np.maximum(f + best_alpha * (g - f), 0)
        f /= f.sum()
        no_improve = 0
        m_idx = 0

        conv = fft_convolve(f)
        M = conv.max()
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 5 == 0:
                print(f"      LP-FW it={it}: obj={2*n*best_obj:.8f}")

    return best_f, best_obj


def fine_polish(a0, time_limit, verbose=False, seed=0):
    """
    Fine-tuning with very small steps: combines subgradient with
    targeted coordinate moves.
    """
    a = np.maximum(a0.copy(), 1e-10)
    a /= a.sum()
    n = len(a)

    conv = fft_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.1:
            break
        if no_improve > 500:
            break

        conv = fft_convolve(a)
        M = conv.max()

        if it % 3 == 0:
            # Subgradient step
            k_actual = min(3, len(conv))
            peak_idx = np.argpartition(conv, -k_actual)[-k_actual:]
            pvals = conv[peak_idx]
            pmax = pvals.max()
            weights = np.exp((pvals - pmax) / (0.02 * pmax + 1e-15))
            weights /= weights.sum()

            sg = np.zeros(n)
            for w, k in zip(weights, peak_idx):
                k = int(k)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                if js <= je:
                    indices = np.arange(js, je + 1)
                    sg[indices] += w * a[k - indices]

            eta = 0.01
            a = a * np.exp(-eta * sg)
            a = np.maximum(a, 1e-10)
            a /= a.sum()
        else:
            # Coordinate move: pick random coordinate, try to adjust
            i = int(rng.integers(n))
            for delta in [0.01, -0.01, 0.005, -0.005]:
                trial = a.copy()
                trial[i] = max(0, trial[i] + delta * a.mean())
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                obj = fft_convolve(trial).max()
                if obj < M - 1e-15:
                    a = trial
                    break

        new_M = fft_convolve(a).max()
        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"      Polish it={it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization for minimizing 2n*max(conv)/(sum)^2.
    Combines subgradient descent, peak-targeted CD, LP-FW, and
    multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s)) - 25.0

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            ps = try_seq(parent, "parent")
            print(f"  Parent loaded: n={len(parent)}, score={ps:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(1024, 0.3)
        try_seq(parent, "default_gauss")

    print(f"\n=== Phase 1: Initialization & Screening ===")
    t_phase1 = time.monotonic()

    # Generate diverse initializations at multiple n values
    n_values = [768, 1024, 1536, 2048, 4096]
    initializations = []

    for n in n_values:
        cands = []
        cands.append((f"gauss_{n}", make_gaussian(n, 0.25)))
        cands.append((f"gauss_w_{n}", make_gaussian(n, 0.4)))
        cands.append((f"asym_{n}", make_asymmetric(n, 0.5)))
        cands.append((f"cos1_{n}", make_cosine_mod(n, 1, 0.15)))
        cands.append((f"cos2_{n}", make_cosine_mod(n, 2, 0.1)))
        cands.append((f"ramp_{n}", make_ramped(n, 1.0)))
        cands.append((f"ramp2_{n}", make_ramped(n, 2.0)))
        cands.append((f"bimod_{n}", make_bimodal(n)))
        cands.append((f"flat_env_{n}", make_flat_envelope(n)))
        cands.append((f"flat_{n}", np.ones(n) / n))

        # Interpolated parent
        if parent is not None and n != len(parent):
            cands.append((f"par_interp_{n}", interpolate_seq(parent, n)))

        for label, a in cands:
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            initializations.append((label, a))

    # Also add parent itself
    if parent is not None:
        initializations.append(("parent", parent.copy()))

    # Score all initializations quickly
    scored = []
    for label, a in initializations:
        sc = obj_norm(a)
        scored.append((sc, label, a))
    scored.sort(key=lambda x: x[0])

    print(f"  Tested {len(scored)} initializations")
    print(f"  Top 5:")
    for i, (sc, label, a) in enumerate(scored[:5]):
        print(f"    {label}: obj={sc:.6f}")

    # Try top initializations with evaluate_sequence
    for sc, label, a in scored[:10]:
        try_seq(a, f"init_{label}")

    print(f"  Phase 1 time: {time.monotonic()-t_phase1:.1f}s")

    # === Phase 2: Subgradient descent on top candidates ===
    print(f"\n=== Phase 2: Subgradient Descent ===")
    t_phase2 = time.monotonic()

    max_phase2_time = 350
    for i, (sc, label, a) in enumerate(scored[:4]):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_alloc = min(rem * 0.5, max_phase2_time / max(1, 4 - i))
        if t_alloc < 20:
            break
        print(f"  SGD [{label}] n={len(a)}: alloc={t_alloc:.0f}s (init_obj={sc:.6f})")
        a_opt, obj = subgradient_descent(a, time_limit=t_alloc, verbose=True, seed=seed + i)
        try_seq(a_opt, f"sgd_{label}")

    print(f"  Phase 2 time: {time.monotonic()-t_phase2:.1f}s")

    # === Phase 3: Peak-targeted CD on best ===
    print(f"\n=== Phase 3: Peak-Targeted CD ===")
    t_phase3 = time.monotonic()

    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  PTC: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = peak_targeted_cd(best_seq, time_limit=t_alloc, verbose=True, seed=seed+10)
            try_seq(a_opt, "ptc_best")

        # Also try PTC on a different candidate
        rem = deadline - time.monotonic() - 250
        if rem > 20 and len(scored) > 1:
            sc2, label2, a2 = scored[1]
            t_alloc = min(rem, 80)
            print(f"  PTC [{label2}]: alloc={t_alloc:.0f}s")
            a_opt2, obj2 = peak_targeted_cd(a2, time_limit=t_alloc, verbose=False, seed=seed+20)
            try_seq(a_opt2, f"ptc_{label2}")

    print(f"  Phase 3 time: {time.monotonic()-t_phase3:.1f}s")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    t_phase4 = time.monotonic()

    if best_seq is not None:
        base_n = len(best_seq)
        target_ns = [n for n in [1024, 1536, 2048, 3072, 4096] if n != base_n]
        target_ns.sort(key=lambda x: abs(x - base_n))

        for target_n in target_ns[:3]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            f_interp = interpolate_seq(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.4, 80)
                if t_alloc > 15:
                    print(f"  SGD n={target_n}: alloc={t_alloc:.0f}s")
                    a_opt, obj = subgradient_descent(f_interp, time_limit=t_alloc, verbose=False, seed=seed+target_n%100)
                    try_seq(a_opt, f"sgd_interp_n{target_n}")

    print(f"  Phase 4 time: {time.monotonic()-t_phase4:.1f}s")

    # === Phase 5: LP-FW polish ===
    print(f"\n=== Phase 5: LP-FW Polish ===")
    t_phase5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # LP is expensive for large n; cap at 2048
        if n_best <= 2048:
            rem = deadline - time.monotonic() - 100
            if rem > 30:
                t_alloc = min(rem, 120)
                print(f"  LP-FW: n={n_best}, alloc={t_alloc:.0f}s")
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
                try_seq(a_opt, "lpfw_polish")
        else:
            # For large n, use subgradient + PTC instead
            rem = deadline - time.monotonic() - 100
            if rem > 30:
                t_alloc = min(rem * 0.6, 80)
                print(f"  SGD (large n): n={n_best}, alloc={t_alloc:.0f}s")
                a_opt, obj = subgradient_descent(best_seq, time_limit=t_alloc, verbose=True, seed=seed+50)
                try_seq(a_opt, "sgd_largen")

    print(f"  Phase 5 time: {time.monotonic()-t_phase5:.1f}s")

    # === Phase 6: Fine polish ===
    print(f"\n=== Phase 6: Fine Polish ===")
    t_phase6 = time.monotonic()

    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 60)
            print(f"  Fine polish: alloc={t_alloc:.0f}s")
            a_opt, obj = fine_polish(best_seq, time_limit=t_alloc, verbose=True, seed=seed+99)
            try_seq(a_opt, "fine_polish")

    print(f"  Phase 6 time: {time.monotonic()-t_phase6:.1f}s")

    # === Phase 7: Final squeeze with multiple short runs ===
    print(f"\n=== Phase 7: Final Squeeze ===")

    if best_seq is not None:
        for run in range(3):
            rem = deadline - time.monotonic() - 5
            if rem < 5:
                break
            t_alloc = min(rem * 0.5, 15)
            if t_alloc < 3:
                break
            # Alternate between strategies
            if run % 2 == 0:
                a_opt, obj = peak_targeted_cd(best_seq, time_limit=t_alloc, verbose=False, seed=seed + 100 + run)
            else:
                a_opt, obj = subgradient_descent(best_seq, time_limit=t_alloc, verbose=False, seed=seed + 200 + run)
            try_seq(a_opt, f"squeeze_{run}")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Ensure valid output
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score):
        best_seq = np.ones(512) / 512
        final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {final_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() + 25.0 - (deadline):.1f}s / 1800s budget")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
