    # = sum_j w[j] * a[n-1-m+j]
    # We want sum_k w[k] * a[k-i]. Let k=j, then sum_j w[j] * a[j-i].
    # From conv: n-1-m+j = j-i => n-1-m = -i => m = n-1+i

def fast_convolve(a):

    n = len(a)

    A = np.fft.rfft(a, fft_size)

    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)

    return e / s

    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""

        b = res.x
        b_sum = b.sum()
        if b_sum < 1e-10:
            no_improve += 1
            if no_improve > 50:
                m_factor = min(m_factor + 0.005, 1.01)
            continue
        b = b / b_sum
        no_improve = 0
        
        diff = np.max(np.abs(b - a))
        if diff < 1e-12:
            no_improve = 1
            continue
        
        # Line search
        best_t = 0.0
        best_val = M
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
            a_t = (1 - t) * a + t * b
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            if s_t < 1e-15:
                continue
            a_t = a_t / s_t
            val = fast_convolve(a_t).max()
            if val < best_val:
                best_val = val
                best_t = t
        
        if best_t > 1e-10:
            a = (1 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            a = a / a.sum()
            c = fast_convolve(a)
            M = c.max()
            m_factor = max(m_factor - 0.001, 0.99)
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                if verbose and it % 200 == 0:
                    print(f"    LP-FW-tight iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            if no_improve > 80:
                break
    
    return best_a, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """Coordinate descent: adjust individual coordinates to reduce max convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    # Sort indices by value (optimize large coordinates first)
    idx_order = np.argsort(f)[::-1]
    
    pass_count = 0
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        c = fast_convolve(f)
        improved = False
        
        for j in idx_order[:min(200, n)]:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try perturbing this coordinate
            for delta_frac in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]:
                new_val = old_val * (1 + delta_frac)
                if new_val < 0:
                    continue
                
                f_trial = f.copy()
                f_trial[j] = new_val
                ss = f_trial.sum()
                if ss < 1e-10:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                f[j] = best_j_val
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and pass_count > 1:
            break
    
    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    """Raised cosine (Hann window)."""
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. Initialization and quick screening
    2. LSE-smoothed L-BFGS optimization
    3. LP-FW generalized (main workhorse)
    4. LP-FW tight (precision polishing)
    5. Multi-length exploration
    6. Coordinate descent final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # === Phase 0: Load parent ===
    print("=== Phase 0: Initialization ===")
    t0 = time.monotonic()
    
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = make_gaussian(768, 0.3)
        try_seq(parent, "default_gauss")
    
    n_parent = len(parent)
    print(f"  Phase 0 time: {time.monotonic()-t0:.1f}s")
    
    # === Phase 1: Quick screening ===
    print("\n=== Phase 1: Quick Screening ===")
    t1 = time.monotonic()
    
    init_candidates = {}
    init_candidates["parent"] = parent.copy()
    init_candidates["gauss"] = make_gaussian(n_parent, 0.25)
    init_candidates["gauss_wide"] = make_gaussian(n_parent, 0.35)
    init_candidates["raised_cos"] = make_raised_cosine(n_parent)
    
    # Add perturbed versions
    for i in range(3):
        noise = rng.standard_normal(n_parent) * 0.02
        a_pert = np.maximum(parent + noise, 0)
        a_pert = a_pert / a_pert.sum()
        init_candidates[f"pert{i}"] = a_pert
    
    best_init = None
    best_init_obj = float('inf')
    for name, a in init_candidates.items():
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        c = fast_convolve(a)
        obj = 2 * len(a) * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a
        try_seq(a, f"screen_{name}")
    
    print(f"  Best init obj: {best_init_obj:.8f}")
    print(f"  Phase 1 time: {time.monotonic()-t1:.1f}s")
    
    # === Phase 2: LSE L-BFGS ===
    print(f"\n=== Phase 2: LSE L-BFGS (n={len(best_init)}) ===")
    t2 = time.monotonic()
    rem = deadline - time.monotonic()
    t_alloc_lse = min(rem * 0.12, 180)
    print(f"  Allocation: {t_alloc_lse:.0f}s")
    
    if t_alloc_lse > 20:
        beta_schedule = [8, 16, 32, 64, 128]
        a_opt, obj = lse_lbfgs_optimize(best_init, time_limit=t_alloc_lse,
                                         beta_schedule=beta_schedule, verbose=True)
        try_seq(a_opt, "lse_lbfgs")
    
    # Also try from parent if it was different
    if not np.allclose(best_init, parent, atol=0.01):
        rem = deadline - time.monotonic()
        if rem > 300:
            t_alloc2 = min(rem * 0.05, 60)
            if t_alloc2 > 15:
                a_opt2, obj2 = lse_lbfgs_optimize(parent, time_limit=t_alloc2,
                                                   beta_schedule=[16, 32, 64], verbose=False)
                try_seq(a_opt2, "lse_parent")
    
    print(f"  Phase 2 time: {time.monotonic()-t2:.1f}s")
    
    # === Phase 3: LP-FW Generalized (main workhorse) ===
    print(f"\n=== Phase 3: LP-FW Generalized ===")
    t3 = time.monotonic()
    base = best_seq if best_seq is not None else best_init
    n_base = len(base)
    
    rem = deadline - time.monotonic()
    t_alloc_lpfw = min(rem * 0.45, 720)
    print(f"  Allocation: {t_alloc_lpfw:.0f}s, n={n_base}")
    
    if t_alloc_lpfw > 30:
        a_opt, obj = lp_fw_generalized(base, time_limit=t_alloc_lpfw, verbose=True)
        try_seq(a_opt, "lpfw_gen")
    
    # Second round of LP-FW from the result
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 120:
            t_alloc2 = min(rem * 0.2, 240)
            a_opt2, obj2 = lp_fw_generalized(best_seq, time_limit=t_alloc2, verbose=True)
            try_seq(a_opt2, "lpfw_gen2")
    
    print(f"  Phase 3 time: {time.monotonic()-t3:.1f}s")
    
    # === Phase 4: LP-FW Tight ===
    print(f"\n=== Phase 4: LP-FW Tight ===")
    t4 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            t_alloc = min(rem * 0.25, 360)
            print(f"  Allocation: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_tight(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_tight")
    
    print(f"  Phase 4 time: {time.monotonic()-t4:.1f}s")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    t5 = time.monotonic()
    base = best_seq if best_seq is not None else best_init
    
    # Try a few different lengths
    target_lengths = []
    n_best = len(base)
    if n_best != 1024:
        target_lengths.append(1024)
    if n_best != 512:
        target_lengths.append(512)
    if n_best != 640:
        target_lengths.append(640)
    
    for target_n in target_lengths[:2]:  # Limit to 2 to save time
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")
        
        # Quick LP-FW on interpolated
        t_lpfw = min(rem * 0.3, 180)
        if t_lpfw > 30:
            a_opt, obj = lp_fw_generalized(f_interp, time_limit=t_lpfw, verbose=False)
            try_seq(a_opt, f"lpfw_n{target_n}")
    
    print(f"  Phase 5 time: {time.monotonic()-t5:.1f}s")
    
    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polishing ===")
    t6 = time.monotonic()
    if best_seq is not None:
        # Coordinate descent
        rem = deadline - time.monotonic()
        if rem > 60:
            t_alloc = min(rem * 0.3, 120)
            print(f"  Coord descent: {t_alloc:.0f}s")
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")
        
        # One more LP-FW squeeze
        if best_seq is not None:
            rem = deadline - time.monotonic()
            if rem > 40:
                t_alloc = min(rem * 0.5, 120)
                print(f"  Final LP-FW: {t_alloc:.0f}s")
                a_opt, obj = lp_fw_tight(best_seq, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, "final_lpfw_tight")
        
        # Final LP-FW general
        if best_seq is not None:
            rem = deadline - time.monotonic()
            if rem > 30:
                t_alloc = min(rem * 0.6, 90)
                print(f"  Final LP-FW gen: {t_alloc:.0f}s")
                a_opt, obj = lp_fw_generalized(best_seq, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, "final_lpfw_gen")
    
    print(f"  Phase 6 time: {time.monotonic()-t6:.1f}s")
    
    # === Fallback ===
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Something went wrong, try parent
        if parent is not None:
            best_seq = parent
            best_score = float(evaluate_sequence(best_seq.tolist()))
        else:
            best_seq = np.ones(768) / 768
            best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"TOTAL TIME: {time.monotonic()-t0:.1f}s")
    
    return [float(x) for x in best_seq]
