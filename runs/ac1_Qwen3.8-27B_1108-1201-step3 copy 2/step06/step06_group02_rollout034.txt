```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-method approach:
- Variance-minimizing gradient descent with FFT-accelerated gradients
- Improved LP-Frank-Wolfe with per-lag adaptive M constraints
- Peak-targeted coordinate descent with pair swaps
- Multi-frequency spectral initialization
- Multi-length exploration with interpolation
- Final polish with exact coordinate descent

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_corr(a, b):
    """Compute sum_k b[k] * a[k-m] for m = 0..n-1 using FFT."""
    n = len(a)
    return np.convolve(a[::-1], b, 'full')[n - 1:n - 1 + n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped initial sequence with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp, phase=0.0):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, freqs, amps, phases, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    a = np.ones(n)
    x = np.arange(n)
    for m, (f, amp, ph) in enumerate(zip(freqs, amps, phases)):
        a += amp * np.cos(2.0 * np.pi * f * x / n + ph)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_ramp(n, direction='right'):
    """Linear ramp sequence."""
    if direction == 'right':
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, separation=0.3):
    """Two Gaussian bumps."""
    x = np.arange(n) / n
    center1 = 0.5 - separation / 2
    center2 = 0.5 + separation / 2
    sigma = 0.15
    a = np.exp(-0.5 * ((x - center1) / sigma) ** 2) + np.exp(-0.5 * ((x - center2) / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_direction_per_lag(f, M_vec, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M_vec[k] for all k, g >= 0."""
    c = -np.ones(n)
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = M_vec
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_direction_uniform(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, per_lag=True):
    """Improved LP-Frank-Wolfe with per-lag M and adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        conv = fast_convolve(f)
        current_peak = conv.max()

        g = None
        if per_lag:
            # Per-lag adaptive M: allow peaks to stay, tighten non-peaks
            # Use a "target profile" that's slightly below current for peaks
            M_vec = conv.copy()
            # Slightly tighten the top 10% of lags
            threshold = np.quantile(conv, 0.9)
            M_vec[conv > threshold] *= 0.995
            g = solve_lp_direction_per_lag(f, M_vec, n)

            if g is None or (g is not None and g.sum() < 1e-12):
                # Fall back to uniform M with escalation
                for scale in [1.0, 1.02, 1.05, 1.1, 1.2, 1.5, 2.0]:
                    M_try = current_peak * scale
                    g = solve_lp_direction_uniform(f, M_try, n)
                    if g is not None and g.sum() > 1e-12:
                        break
                    g = None
        else:
            for scale in [1.0, 1.02, 1.05, 1.1, 1.2, 1.5, 2.0]:
                M_try = current_peak * scale
                g = solve_lp_direction_uniform(f, M_try, n)
                if g is not None and g.sum() > 1e-12:
                    break
                g = None

        if g is None or g.sum() < 1e-12:
            # Perturb and retry
            rng = np.random.default_rng(it * 13 + 7)
            noise_type = it % 4
            if noise_type == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            elif noise_type == 1:
                shift = rng.integers(1, max(2, n // 10))
                f = project_simplex(np.roll(best_f, shift))
            elif noise_type == 2:
                f = project_simplex(0.8 * best_f + 0.2 * best_f[::-1])
            else:
                # Random single frequency perturbation
                freq = rng.integers(1, n // 2)
                amp = rng.uniform(0.01, 0.05)
                phase = rng.uniform(0, 2 * np.pi)
                pert = amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
                f = project_simplex(best_f + pert)
            no_improve += 1
            if no_improve > 60:
                break
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            continue
        no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        alphas = [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search around best
        if best_alpha > 0 and best_alpha < 1.0:
            lo = best_alpha * 0.7
            hi = min(best_alpha * 1.3, 1.0)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                ft = ft / ft.sum()
                vm = fast_convolve(ft).max()
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid
                    lo = mid
                else:
                    if best_alpha == 0.0 or mid < best_alpha:
                        hi = mid
                    else:
                        lo = mid

        if best_alpha < 1e-12:
            no_improve += 1
            continue

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        f = f_new

        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
            if verbose and it % 20 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def variance_gradient_descent(f0, time_limit, verbose=False):
    """
    Gradient descent on sum_k (conv[k])^2 (equivalently, minimizing variance of auto-correlation).
    Uses FFT-accelerated gradient computation.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.5 / n
    momentum_val = 0.0
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)

        # Gradient of sum_k conv[k]^2:
        # d/da[m] sum_k conv[k]^2 = 2 * sum_k conv[k] * d(conv[k])/da[m]
        # d(conv[k])/da[m] = 2*a[k-m] (if valid)
        # So grad[m] = 4 * sum_k conv[k] * a[k-m] = 4 * cross_corr(a, conv)[m]
        grad = 4.0 * cross_corr(a, conv)

        # Project gradient onto tangent space of simplex (remove mean component)
        grad = grad - np.mean(grad)

        # Momentum
        momentum_val = 0.85
        v = momentum_val * v + (1.0 - momentum_val) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm

        # Adaptive step size (backtracking)
        alpha = lr
        a_new = project_simplex(a - alpha * v)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        # Check if we made progress on the actual objective
        # (variance reduction should correlate with peak reduction)
        progress = False
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            progress = True
        elif obj_new < conv.max() * 1.001:
            # Allow small increases in peak if variance is decreasing
            var_old = np.var(conv)
            var_new = np.var(conv_new)
            if var_new < var_old * 0.999:
                progress = True

        if progress:
            a = a_new
            # Decay learning rate slowly
            lr *= 0.9995
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 50:
                # Restart
                rng = np.random.default_rng(it * 41 + 3)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                lr = 0.5 / n
                no_improve = 0

        if lr < 1e-10:
            break

        if verbose and it % 50 == 0:
            print(f"    VGD iter {it}: obj={best_obj:.10f}, lr={lr:.2e}")

    return best_a, best_obj


def smooth_max_gradient_descent(f0, time_limit, verbose=False):
    """
    Gradient descent on smooth approximation of max(conv).
    Uses log-sum-exp smoothing with adaptive temperature.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = 50.0
    lr = 1.0 / n
    momentum_val = 0.8
    v = np.zeros(n)
    no_improve = 0
    it = 0

    total_time = time_limit
    while time.monotonic() - t0 < total_time:
        it += 1
        conv = fast_convolve(a)

        # Log-sum-exp smoothing
        conv_shifted = conv - conv.max()
        exp_vals = np.exp(beta * conv_shifted)
        w = exp_vals / exp_vals.sum()

        # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] = 2 * cross_corr(a, w)[m]
        grad = 2.0 * cross_corr(a, w)
        grad = grad - np.mean(grad)

        # Momentum
        v = momentum_val * v + (1.0 - momentum_val) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm

        # Step
        a_new = project_simplex(a - lr * v)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            lr *= 0.999
        else:
            no_improve += 1
            lr *= 0.93
            if no_improve > 80:
                rng = np.random.default_rng(it * 23 + 1)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr = 1.0 / n
                no_improve = 0

        # Increase beta (annealing)
        progress = (time.monotonic() - t0) / total_time
        beta = 50.0 + 200.0 * progress ** 2

        if lr < 1e-10:
            break

        if verbose and it % 30 == 0:
            print(f"    SMGD iter {it}: obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.2e}")

    return best_a, best_obj


def peak_targeted_descent(f0, time_limit, verbose=False):
    """
    Targeted mass redistribution to reduce peaks.
    Tries multiple strategies per peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            break

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find contributing indices for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        if len(contrib_idx) == 0:
            break

        products = f[contrib_idx] * f[peak_k - contrib_idx]
        total_contrib = products.sum()
        if total_contrib < 1e-15:
            break

        # Strategy 1: Reduce the most contributing coordinate
        max_prod_idx = contrib_idx[np.argmax(products)]
        partner_idx = peak_k - max_prod_idx

        if f[max_prod_idx] >= f[partner_idx]:
            source = max_prod_idx
        else:
            source = partner_idx

        # Target: redistribute to coordinates that contribute to LOW lags
        # Compute contribution of each coordinate to the peak
        # coord j contributes f[j]*f[peak_k-j] to the peak
        # We want to move mass to coords j where f[j]*f[peak_k-j] is small
        non_peak_contrib = 1.0 - products / max(total_contrib, 1e-15)

        best_trial = None
        best_trial_obj = M

        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = f[source] * frac
            if diff < 1e-15:
                continue
            trial = f.copy()
            trial[source] -= diff
            # Distribute to non-contributing coords weighted by their "room"
            weights = non_peak_contrib.copy()
            weights[source] = 0.0
            # Also weight by how much mass is already there (proportional redistribution)
            weights = weights * (trial + 0.01)
            w_sum = weights.sum()
            if w_sum < 1e-15:
                continue
            trial += diff * weights / w_sum
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_trial = trial

        if best_trial is not None:
            f = best_trial
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_f = f.copy()
                no_improve = 0
                if verbose:
                    print(f"    PTD: peak_k={peak_k}, obj={best_obj:.10f}")
        else:
            # Try different peaks
            top_peaks = np.argsort(conv)[-5:]
            found = False
            for pk in top_peaks:
                pk = int(pk)
                if pk == peak_k:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                cidx2 = np.arange(js2, je2 + 1)
                if len(cidx2) == 0:
                    continue
                prods2 = f[cidx2] * f[pk - cidx2]
                src2 = cidx2[np.argmax(prods2)]
                for frac2 in [0.1, 0.2, 0.3, 0.5]:
                    diff2 = f[src2] * frac2
                    if diff2 < 1e-15:
                        continue
                    trial = f.copy()
                    trial[src2] -= diff2
                    # Distribute uniformly to non-contributing
                    nc_mask = np.ones(n, dtype=bool)
                    nc_mask[cidx2] = False
                    if nc_mask.sum() < 1:
                        continue
                    trial[nc_mask] += diff2 / nc_mask.sum()
                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()
                    obj2 = fast_convolve(trial).max()
                    if obj2 < best_obj - 1e-14:
                        f = trial
                        best_obj = obj2
                        best_f = f.copy()
                        no_improve = 0
                        found = True
                        break
                if found:
                    break
            if not found:
                no_improve += 1

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Exact coordinate descent: try moving each coordinate to its best value."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 3:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            # Try a range of values for f[i]
            current_val = f[i]
            rest_sum = 1.0 - current_val
            if rest_sum < 1e-10:
                continue

            # Try: reduce, keep, increase
            candidates = [
                current_val * 0.5,
                current_val * 0.7,
                current_val * 1.3,
                current_val * 1.7,
                0.0,
                rest_sum * 0.1,
                rest_sum * 0.3,
            ]
            candidates = [max(0.0, min(c, 1.0)) for c in candidates]

            best_val = current_val
            best_obj_i = best_obj

            for cv in candidates:
                if abs(cv - current_val) < 1e-8:
                    continue
                trial = f.copy()
                # Redistribute the difference
                diff = cv - current_val
                trial[i] = cv
                if abs(diff) > 1e-12:
                    # Distribute to other coords proportionally
                    others = np.ones(n)
                    others[i] = 0.0
                    if others.sum() > 0:
                        trial[others] -= diff * trial[others] / max(trial[others].sum(), 1e-15)
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-10:
                    continue
                trial = trial / ts
                obj_i = fast_convolve(trial).max()
                if obj_i < best_obj_i - 1e-12:
                    best_obj_i = obj_i
                    best_val = cv

            if best_val != current_val:
                diff = best_val - current_val
                f[i] = best_val
                if abs(diff) > 1e-12:
                    others = np.ones(n)
                    others[i] = 0.0
                    osum = f[others].sum()
                    if osum > 1e-15:
                        f[others] -= diff * f[others] / osum
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                improved = True

                conv = fast_convolve(f)
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_f = f.copy()

        if improved:
            no_improve = 0
            sweep += 1
        else:
            no_improve += 1

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}")

    return best_f, best_obj


def swap_descent(f0, time_limit, verbose=False):
    """Try swapping mass between pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        if no_improve > 200:
            break

        # Pick random pair
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Try different swap amounts
        best_obj_pair = best_obj
        best_swap = 0.0

        for frac in [0.1, 0.3, 0.5, 0.7]:
            # Move frac * min(f[i], f[j]) from larger to smaller
            if f[i] >= f[j]:
                transfer = f[i] * frac
                trial = f.copy()
                trial[i] -= transfer
                trial[j] += transfer
            else:
                transfer = f[j] * frac
                trial = f.copy()
                trial[j] -= transfer
                trial[i] += transfer

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-10:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_obj_pair - 1e-13:
                best_obj_pair = obj
                best_swap = transfer

        if best_swap > 0:
            if f[i] >= f[j]:
                f[i] -= best_swap
                f[j] += best_swap
            else:
                f[j] -= best_swap
                f[i] += best_swap
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            if best_obj_pair < best_obj - 1e-14:
                best_obj = best_obj_pair
                best_f = f.copy()
                no_improve = 0
        else:
            no_improve += 1

        if verbose and no_improve % 50 == 0:
            print(f"    Swap: obj={best_obj:.10f}, no_improve={no_improve}")

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining LP-FW, gradient methods,
    peak-targeted descent, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None

        nonlocal best_seq, best_score

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
