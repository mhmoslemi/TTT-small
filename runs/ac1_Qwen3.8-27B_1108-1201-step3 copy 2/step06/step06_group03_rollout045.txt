Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced LP-Frank-Wolfe

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Build matrix A where (A @ g)[k] = (f*g)[k] for full convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = f
    return A

    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.

    best_val = vals[best_idx]
    best_alpha = alphas[best_idx]

    # Golden section on best interval
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    if best_idx > 0 and best_idx < len(alphas) - 1:
        lo = alphas[best_idx - 1]
        hi = alphas[best_idx + 1]
    elif best_idx == 0:
        lo, hi = 0.0, alphas[1]
    else:
        lo, hi = alphas[-2], 1.0

    # Golden section search
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    evals = 2

    while hi - lo > 1e-5 and evals < max_evals - 10:
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)
        evals += 1

        mid = (lo + hi) / 2.0
        fm = obj(mid)
        evals += 1
        if fm < best_val:
            best_val = fm
            best_alpha = mid

    return best_alpha, best_val


def lp_fw_optimize(f0, time_limit, verbose=False, m_relax=1.0):
    """
    Enhanced LP-Frank-Wolfe with golden section line search.
    f0: starting point (sum=1, non-negative)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    A_cache = build_conv_matrix(f)
    A_valid = True

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Solve LP
        g, _ = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.05
            if m_exp > 5.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 0.0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.05
            continue
        g = g / g_sum

        # Check convergence: gap between f and g
        gap = np.abs(g - f).sum()
        if gap < 1e-10:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.03, 3.0)
            if no_improve > 100:
                # Restart with perturbation
                rng = np.random.default_rng(it * 31 + 7)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.005, 0.0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        if elapsed > time_limit * 0.8:
            # Use simpler line search near the end
            best_alpha = 0.0
            best_val = M
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                ft = f + alpha * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
        else:
            best_alpha, best_val = line_search_golden(f, g, n, max_evals=20)

        if best_alpha > 1e-8:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}, gap={gap:.2e}, alpha={best_alpha:.4f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 3.0)

        if no_improve > 200:
            break

    return best_f, best_obj


def smoothed_mirror_descent_v2(f0, time_limit, verbose=False,
                                tau_init=10.0, tau_decay=1.005,
                                lr_init=0.01, lr_decay=0.9997):
    """
    Improved smoothed mirror descent using log-sum-exp for smooth max.
    Uses all peaks weighted by their excess above the mean.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()
        mean_c = conv.mean()
        excess = np.maximum(conv - mean_c, 0)

        # Use log-sum-exp weighting on excess
        if excess.max() > 1e-15:
            scaled = tau * excess
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws > 1e-15:
                w = w / ws
            else:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
        else:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0

        # Compute weighted subgradient (vectorized)
        sg = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            sg[js:je + 1] += w[k] * contrib

        # Multiplicative update
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        obj_new = fast_convolve(f_new).max()
        if obj_new < M:
            f = f_new
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and iter_count % 100 == 0:
                    print(f"      SMD2 iter {iter_count}: obj={best_obj:.10f}, tau={tau:.1f}")
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.8
            no_improve += 1

        tau *= tau_decay

        if no_improve > 150:
            rng = np.random.default_rng(iter_count * 13 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            tau = max(tau * 0.5, 5.0)
            no_improve = 0

        if tau > 5000 or lr < 1e-7:
            break

    return best_f, best_obj


def flatten_iteration(f0, time_limit, verbose=False):
    """
    Direct flattening: iteratively reduce the convolution peaks by
    redistributing mass from high-contributing positions to low ones.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        mean_c = conv.mean()

        # Find positions where conv exceeds mean by more than threshold
        threshold = mean_c + 0.01 * (M - mean_c)
        hot_positions = np.where(conv > threshold)[0]

        if len(hot_positions) == 0:
            break

        # For each hot position, compute which indices contribute most
        # and reduce them multiplicatively
        reduction = np.ones(n)
        eta = 0.05
        for k in hot_positions:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] * f[idx]  # pairwise products
            # Normalize contribution
            total = contrib.sum()
            if total < 1e-15:
                continue
            # Reduce indices proportional to their contribution
            frac = contrib / total
            reduction[js:je + 1] *= (1.0 - eta * frac * len(hot_positions) / max(len(hot_positions), 1))

        f_new = f * reduction
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M - 1e-15:
            f = f_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      Flatten iter {it}: obj={best_obj:.10f}")
            eta = min(eta * 1.02, 0.15)
        else:
            eta *= 0.8
            no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng(it * 23 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            eta = 0.03
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_v2(f0, time_limit, verbose=False):
    """
    Coordinate descent with finer perturbation search.
    For each coordinate, try several values and keep the best.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    full_passes = 0

    while time.monotonic() - t0 < time_limit:
        full_passes += 1
        improved = False
        rng = np.random.default_rng(full_passes * 1000 + 42)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try multiplicative perturbations
            for mult in [0.5, 0.6, 0.7, 0.8, 1.2, 1.3, 1.5, 0.3, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def make_asymmetric_init(n, variant=0, seed=42):
    """Generate diverse asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 137)
    x = np.arange(n) / n

    if variant == 0:
        a = np.exp(-x / 0.15)
    elif variant == 1:
        a = np.exp(-(1 - x) / 0.15)
    elif variant == 2:
        a = np.exp(-x / 0.3) * (1 + 0.5 * np.sin(2 * np.pi * x * 3))
        a = np.maximum(a, 0)
    elif variant == 3:
        a = x ** 2 + 0.1
    elif variant == 4:
        a = (1 - x) ** 2 + 0.1
    elif variant == 5:
        a = np.exp(-50 * (x - 0.2) ** 2) + 0.3 * np.exp(-50 * (x - 0.8) ** 2)
    elif variant == 6:
        # Very spiky: one dominant peak
        a = np.ones(n) * 0.01
        peak_pos = rng.integers(0, n)
        a[peak_pos] = 1.0
        a[max(0, peak_pos - 1):peak_pos + 2] += 0.5
    elif variant == 7:
        # Power law
        a = (np.arange(n, 0, -1)) ** 0.7
    elif variant == 8:
        # Two-scale: large slow ramp + small fast oscillation
        a = x * 0.5 + 0.5 * np.abs(np.sin(2 * np.pi * x * 7)) + 0.05
    elif variant == 9:
        # Random asymmetric
        a = rng.standard_normal(n)
        a = np.abs(a)
        a *= (1 - x)  # taper
        a = np.maximum(a, 0.01)
    elif variant == 10:
        # Step-like: high on left, low on right
        a = np.ones(n)
        a[n//3:] *= 0.3
        a[2*n//3:] *= 0.1
    elif variant == 11:
        # Gaussian centered off-center
        a = np.exp(-30 * (x - 0.3) ** 2) + 0.05
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: diverse initializations -> LP-FW (primary)
    -> SMD v2 -> flatten -> coordinate descent. Explores multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Try parent with LP-FW directly ===
    print("=== Phase 1: LP-FW on Parent ===")
    if parent is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 60:
            t_alloc = min(rem, 300)
            print(f"  LP-FW on parent (n={len(parent)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(parent, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Diverse initializations + quick SMD ===
    print("\n=== Phase 2: Diverse Initializations ===")
    test_lengths = [512, 640, 768, 896]
    init_candidates = []

    for n_try in test_lengths:
        for v in range(12):
            a_init = make_asymmetric_init(n_try, variant=v, seed=seed + v)
            init_candidates.append((a_init, f"init_v{v}_n{n_try}"))

    # Score initializations cheaply
    scored_inits = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored_inits.append((obj, a_norm, label))

    scored_inits.sort(key=lambda x: x[0])
    top_inits = scored_inits[:6]
    print(f"  Top inits: {[f'{l}:{o:.4f}' for o, _, l in top_inits]}")

    # Quick SMD on top inits
    for idx, (obj_init, a_init, label) in enumerate(top_inits[:4]):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_alloc = min(rem / max(1, 4 - idx), 80)
        if t_alloc < 20:
            t_alloc = 20
        print(f"  SMD2 on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, obj = smoothed_mirror_descent_v2(a_init, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"smd_{label}")

    # === Phase 3: LP-FW on current best ===
    print("\n=== Phase 3: LP-FW on Best ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem, 300)
            print(f"  LP-FW on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_best")

    # === Phase 4: Flatten + LP-FW combo ===
    print("\n=== Phase 4: Flatten + LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 60:
            t_flat = min(rem * 0.3, 60)
            t_lpfw = min(rem * 0.6, 120)
            print(f"  Flatten (t={t_flat:.0f}s) + LP-FW (t={t_lpfw:.0f}s)")
            a_flat, _ = flatten_iteration(best_seq, time_limit=t_flat)
            try_seq(a_flat, "flatten")
            a_opt, _ = lp_fw_optimize(a_flat, time_limit=t_lpfw, verbose=True)
            try_seq(a_opt, "lpfw_after_flat")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [640, 768, 896, 1024, 1280]:
            rem = deadline - time.monotonic() - 120
            if rem < 40:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.05:
                t_alloc = min(rem * 0.35, 80)
                if t_alloc < 20:
                    t_alloc = 20
                print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
                a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 6: Random restarts ===
    print("\n=== Phase 6: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(3):
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_alloc = min(rem / max(1, 3 - r), 100)
            if t_alloc < 20:
                t_alloc = 20

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.008
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            elif r == 1:
                asym = make_asymmetric_init(n_best, variant=rng.integers(0, 12), seed=seed + 100 + r)
                f_init = 0.6 * best_seq + 0.4 * asym
                f_init = f_init / f_init.sum()
            else:
                # Reverse + blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()

            print(f"  Restart {r} (SMD2+LP-FW): t={t_alloc:.0f}s")
            a_smd, _ = smoothed_mirror_descent_v2(f_init, time_limit=t_alloc * 0.5)
            a_opt, _ = lp_fw_optimize(a_smd, time_limit=t_alloc * 0.5)
            try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Coordinate descent polish ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_v2(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 8: Final LP-FW squeeze ===
    print("\n=== Phase 8: Final LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_lpfw")

    # === Phase 9: Final SMD with low tau ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem, 40)
            print(f"  Final SMD2 (low tau): t={t_alloc:.0f}s")
            a_opt, _ = smoothed_mirror_descent_v2(best_seq, time_limit=t_alloc,
                                                   tau_init=30.0, tau_decay=1.002,
                                                   lr_init=0.003)
            try_seq(a_opt, "final_smd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
