def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, c, top_frac=0.15, temperature=0.01):
    """Compute weighted subgradient of max(c) using top peaks with softmax."""
    n = len(f)
    len_c = len(c)
    k_top = max(5, int(len_c * top_frac))
    peaks = np.argpartition(c, -k_top)[-k_top:]
    pv = c[peaks]
    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        # dc[k]/da[m] = a[k-m] + a[m] for m in [js, je]
        # Note: if k = 2m, both terms are a[m], giving 2*a[m]
        contrib = f[k - idx] + f[idx]
        sg[js:je + 1] += w[i] * contrib
    return sg

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def peak_targeted_descent(f0, time_limit, verbose=False):
    """
    Directly target the single worst peak. For each peak position k,
    the gradient direction tells us which pairs (j, k-j) to reduce.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find the argmax position(s)
        k_best = int(np.argmax(conv))

        # Compute gradient at this peak
        js = max(0, k_best - n + 1)
        je = min(n - 1, k_best)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[js:je + 1] = f[k_best - idx] + f[idx]

        # Also consider top-3 peaks for stability
        top3 = np.argsort(conv)[-3:]
        for k2 in top3:
            k2 = int(k2)
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            grad[js2:je2 + 1] += 0.5 * (f[k2 - np.arange(js2, je2 + 1)] + f[np.arange(js2, je2 + 1)])

        # Multiplicative step
        f_new = f * np.exp(-lr * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      PTD iter {it}: obj={best_obj:.10f}")
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.7
            no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng(it * 41 + 19)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.01
            no_improve = 0

        if lr < 1e-8:
            break

    return best_f, best_obj


def coordinate_descent_fine(f0, time_limit, verbose=False):
    """Fine coordinate descent: for each coordinate, find optimal value via 1D search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int((time.monotonic() - t0) * 100) % 10000 + 1)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try perturbations
            deltas = [old_val * d for d in [-0.3, -0.2, -0.1, 0.05, 0.1, 0.2, 0.3, 0.5, -0.5]]
            for delta in deltas:
                new_val = old_val + delta
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def make_asymmetric_init(n, variant=0, seed=42):
    """Generate asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 100)

    if variant == 0:
        # Exponential decay (front-loaded)
        a = np.exp(-np.arange(n) / (n * 0.15))
    elif variant == 1:
        # Exponential decay (back-loaded)
        a = np.exp(-np.arange(n)[::-1] / (n * 0.15))
    elif variant == 2:
        # Sine-modulated (strongly asymmetric)
        a = np.exp(-np.arange(n) / (n * 0.3)) * (1 + 0.5 * np.sin(2 * np.pi * np.arange(n) / n))
        a = np.maximum(a, 0)
    elif variant == 3:
        # Ramp
        a = np.linspace(0.1, 1.0, n)
    elif variant == 4:
        # Two-bump asymmetric
        x = np.arange(n) / n
        a = np.exp(-50 * (x - 0.3) ** 2) + 0.5 * np.exp(-50 * (x - 0.7) ** 2)
    elif variant == 5:
        # Cosine window (front-heavy)
        a = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2
        a = a + 0.1 * np.ones(n)
    elif variant == 6:
        # Random with structure
        a = np.ones(n)
        a[:n//3] *= 1.5
        a[2*n//3:] *= 0.5
        a += rng.standard_normal(n) * 0.1
        a = np.maximum(a, 0)
    elif variant == 7:
        # Logarithmic
        a = np.log1p(np.arange(n) + 1)
    elif variant == 8:
        # Power law
        a = (np.arange(n) + 1) ** (-0.5)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
