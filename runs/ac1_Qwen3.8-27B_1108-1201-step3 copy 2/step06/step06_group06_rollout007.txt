import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """
    Compute gradient of log(sum_k exp((a*a)[k]/tau)) and the convolution.
    Returns (gradient, convolution, smooth_value).
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum

    # h[j] = sum_{l=0}^{n-1} w[j+l] * f[l] for j in 0..n-1
    # This is the cross-correlation of f and w, offset appropriately
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    grad = (2.0 / tau) * h
    smooth_val = tau * np.log(w_sum + 1e-300) + tau * scaled.max()
    return grad, conv, smooth_val

        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.1 * conv.max()
    else:
        tau = tau_init

    # Nesterov + Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    y = f.copy()  # Nesterov lookahead
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    mu = 0.9  # Nesterov momentum
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    tau_decay = 0.9995
    tau_increase_freq = 500
    tau_increase_factor = 1.5

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov step: evaluate gradient at lookahead point
        y = f + mu * (f - (f - lr * m / (np.sqrt(v) + eps)) if t > 1 else f)
        # Simplified: just use current f for gradient (standard Adam with momentum)
        grad, conv, _ = compute_smooth_grad(f, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature
        tau *= tau_decay
        if tau < 5e-8:
            tau = 5e-8

        # Periodic temperature increase to escape local minima
        if t % tau_increase_freq == 0 and no_improve > 50:
            tau = min(tau * tau_increase_factor, 0.15 * conv_new.max())

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.8
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(t * 13 + 7)
            noise = 0.003 + 0.005 * min(no_improve / 2000.0, 4.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.8
            tau = 0.08 * fast_convolve(f).max()
            mu = 0.9
            if verbose:
                print(f"    Restart at iter {t}: obj={2 * n * best_obj:.8f}")

        if verbose and t % 1000 == 0:
            print(f"    Iter {t}: obj={2 * n * best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """
    Direct peak-shaving: identify peak-contributing pairs and
    redistribute mass to reduce the peak.
    More aggressive than the previous version.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Find all peaks above 95% of max
        threshold = 0.95 * M
        peak_indices = np.where(conv >= threshold)[0]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]][:10]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            top_contrib = np.argsort(products)[::-1][:5]

            for ci in top_contrib:
                j1 = int(idx[ci])
                j2 = int(pk - j1)
                if j1 == j2:
                    # Diagonal: reduce f[j1]^2
                    if f[j1] < 1e-12:
                        continue
                    for frac in [0.5, 0.7, 0.85, 0.95]:
                        trial = f.copy()
                        old = trial[j1]
                        trial[j1] = old * frac
                        diff = old - trial[j1]
                        # Redistribute to positions far from all peaks
                        mask = np.ones(n, dtype=bool)
                        for pk2 in peak_indices[:5]:
                            js2 = max(0, int(pk2) - n + 1)
                            je2 = min(n - 1, int(pk2))
                            mask[js2:je2 + 1] = False
                        mask[j1] = False
                        if mask.sum() > 0 and f[mask].sum() > 1e-12:
                            trial[mask] += diff * f[mask] / f[mask].sum()
                        trial = np.maximum(trial, 0)
                        ts = trial.sum()
                        if ts > 1e-12:
                            trial /= ts
                        obj = fast_convolve(trial).max()
                        if obj < best_obj * (1 - 1e-8):
                            f = trial
                            best_obj = obj
                            best_f = f.copy()
                            improved = True
                            break
                else:
                    # Off-diagonal: reduce f[j1]*f[j2]
                    for frac1 in [0.5, 0.7, 0.85]:
                        trial = f.copy()
                        old1 = trial[j1]
                        trial[j1] = old1 * frac1
                        diff = old1 - trial[j1]
                        # Redistribute to j2's "opposite side"
                        mask = np.ones(n, dtype=bool)
                        mask[j1] = False
                        mask[j2] = False
                        # Avoid positions contributing to peak
                        for pk2 in peak_indices[:3]:
                            js2 = max(0, int(pk2) - n + 1)
                            je2 = min(n - 1, int(pk2))
                            mask[js2:je2 + 1] = False
                        if mask.sum() > 0 and f[mask].sum() > 1e-12:
                            trial[mask] += diff * f[mask] / f[mask].sum()
                        trial = np.maximum(trial, 0)
                        ts = trial.sum()
                        if ts > 1e-12:
                            trial /= ts
                        obj = fast_convolve(trial).max()
                        if obj < best_obj * (1 - 1e-8):
                            f = trial
                            best_obj = obj
                            best_f = f.copy()
                            improved = True
                            break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 7919 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
                if fast_convolve(f).max() < best_obj:
                    best_obj = fast_convolve(f).max()
                    best_f = f.copy()
        else:
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    PeakShave iter {it}: obj={2 * n * best_obj:.8f}")

    return best_f, 2 * n * best_obj


def subgradient_polish(f0, time_limit, verbose=False):
    """
    Direct subgradient method on the max-convolution objective.
    At each step, move in the negative subgradient direction (towards peak reduction).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    learning_rate = 0.01
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Subgradient: at the peak position pk, grad_j = 2*f[pk-j] (if valid)
        peak_positions = np.where(conv >= 0.99 * M)[0]
        grad = np.zeros(n)
        for pk in peak_positions[:5]:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for j in range(js, je + 1):
                grad[j] += 2.0 * f[pk - j]

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Projected gradient step
        f_new = f - learning_rate * grad
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            for _ in range(5):
                learning_rate *= 0.7
                f_new = f - learning_rate * grad
                f_new = project_simplex(f_new)
                obj_new = fast_convolve(f_new).max()
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_f = f_new.copy()
                    f = f_new
                    no_improve = 0
                    break

        if no_improve > 50:
            # Reset learning rate and perturb
            learning_rate = 0.01
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_improve = 0

        # Decay learning rate slowly
        learning_rate *= 0.9995

        if verbose and it % 200 == 0:
            print(f"    SubGrad iter {it}: obj={2 * n * best_obj:.8f}, lr={learning_rate:.5f}")

    return best_f, 2 * n * best_obj


def make_init_gaussian(n, center=0.5, width=0.25):
    x = np.arange(n) - n * center
    sigma = n * width
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_cosine(n):
    x = np.arange(n) / (n - 1) if n > 1 else np.array([0.5])
    a = np.cos(np.pi * x / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_two_lobe(n, sep=0.3):
    a1 = make_init_gaussian(n, 0.5 - sep, 0.2)
    a2 = make_init_gaussian(n, 0.5 + sep, 0.2)
    a = (a1 + a2) / 2
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_random(n, seed):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    # Smooth it a bit
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_lacunary(n, seed):
    """Initialize with a lacunary-like structure."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    # Place mass at geometrically spaced positions
    positions = [int(n * f) for f in [0.1, 0.3, 0.5, 0.7, 0.9]]
    for p in positions:
        width = max(1, n // 32)
        start = max(0, p - width)
        end = min(n, p + width)
        a[start:end] += np.exp(-0.5 * ((np.arange(start, end) - p) / (width * 0.5)) ** 2)
    a = np.maximum(a, 0)
    # Add small noise
    a += rng.random(n) * 0.01
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def optimize_single_n(n, time_budget, seed=42, verbose=False):
    """Optimize for a fixed n, return best sequence and score."""
    rng = np.random.default_rng(seed)
    best_f = None
    best_score = np.inf

    def try_it(a, label):
        nonlocal best_f, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.inf
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = a.copy()
            if verbose:
                print(f"      NEW BEST [{label}]: {score:.6f}")
        return score

    # Generate initializations
    inits = [
        ("uniform", np.ones(n) / n),
        ("gauss_c", make_init_gaussian(n, 0.5, 0.25)),
        ("gauss_l", make_init_gaussian(n, 0.35, 0.2)),
        ("cosine", make_init_cosine(n)),
        ("two_lobe", make_init_two_lobe(n, 0.3)),
        ("random1", make_init_random(n, seed)),
        ("random2", make_init_random(n, seed + 1)),
        ("lacunary", make_init_lacunary(n, seed)),
    ]

    # Score initializations
    for label, a in inits:
        try_it(a, f"init_{label}")

    if best_f is None:
        return None, np.inf

    # Phase 1: Smooth optimization (main workhorse)
    t1 = min(time_budget * 0.5, time_budget - 30)
    if t1 > 10:
        t0 = time.monotonic()
        a_opt, obj = nesterov_smooth_optimize(best_f, time_limit=min(t1, time.monotonic() + t1 - time.monotonic()), lr_init=0.005, verbose=verbose)
        try_it(a_opt, "smooth1")

    # Phase 2: Another smooth pass with different params
    if time.monotonic() < time_budget and best_f is not None:
        rem = time_budget - (time.monotonic() - time.monotonic())  # just use remaining
        t2 = min(time_budget * 0.2, 60)
        if t2 > 5:
            t0 = time.monotonic()
            a_opt, obj = nesterov_smooth_optimize(best_f, time_limit=t2, tau_init=0.03 * fast_convolve(best_f).max(), lr_init=0.002, verbose=False)
            try_it(a_opt, "smooth2")

    # Phase 3: Peak shaving
    if best_f is not None:
        t3 = min(time_budget * 0.15, 30)
        if t3 > 5:
            a_opt, obj = peak_shaving(best_f, time_limit=t3, verbose=verbose)
            try_it(a_opt, "peak_shave")

    # Phase 4: Subgradient polish
    if best_f is not None:
        t4 = min(time_budget * 0.1, 20)
        if t4 > 5:
            a_opt, obj = subgradient_polish(best_f, time_limit=t4, verbose=False)
            try_it(a_opt, "subgrad")

    # Phase 5: Final smooth polish
    if best_f is not None:
        t5 = min(time_budget * 0.05, 10)
        if t5 > 3:
            a_opt, obj = nesterov_smooth_optimize(best_f, time_limit=t5, tau_init=0.01 * fast_convolve(best_f).max(), lr_init=0.001, verbose=False)
            try_it(a_opt, "final_polish")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase, multi-n optimization for minimizing 2*n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    global_best_seq = None
    global_best_score = float('inf')

    def try_global(a, label=""):
        nonlocal global_best_seq, global_best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < global_best_score:
            global_best_score = score
            global_best_seq = a.copy()
            print(f"  GLOBAL BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Try to use parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.6f}")
            if np.isfinite(ps) and ps < global_best_score:
                global_best_score = ps
                global_best_seq = parent.copy()
    except Exception:
        pass

    # Define candidate n values (prioritize around known good range)
    n_candidates = [768, 512, 1024, 640, 896, 1280]

    # If parent available, try its length first
    if parent is not None:
        pn = len(parent)
        if pn not in n_candidates:
            n_candidates.insert(0, pn)

    # Deduplicate and sort by priority
    seen = set()
    n_ordered = []
    for n in n_candidates:
        if n not in seen and 128 <= n <= 4096:
            seen.add(n)
            n_ordered.append(n)

    print(f"Trying n values: {n_ordered}")

    # Allocate time: spend most time on top 2-3 n values
    total_time = budget_s - 40  # buffer
    # First n gets 40%, second gets 25%, third gets 15%, rest split 20%
    if len(n_ordered) >= 3:
        times = [total_time * 0.4, total_time * 0.25, total_time * 0.15]
        remaining = total_time - sum(times)
        for i in range(3, len(n_ordered)):
            times.append(max(10, remaining / (len(n_ordered) - 3)))
    elif len(n_ordered) == 2:
        times = [total_time * 0.55, total_time * 0.45]
    else:
        times = [total_time]

    for i, n in enumerate(n_ordered):
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(times[i], rem - 10)
        if t_alloc < 10:
            continue

        print(f"\n=== Optimizing n={n} (time={t_alloc:.0f}s) ===")

        # If we have a parent with similar length, use it as warm start
        if parent is not None and abs(len(parent) - n) < n * 0.3:
            warm = interpolate_sequence(parent, n)
            ws = try_global(warm, f"warmstart_n{n}")
        else:
            warm = None

        # Run optimization
        best_f, best_score = optimize_single_n(n, time_budget=t_alloc, seed=seed + i * 1000, verbose=True)

        if best_f is not None:
            try_global(best_f, f"opt_n{n}")

        # If we improved, try a quick second pass with perturbation
        if global_best_seq is not None and len(global_best_seq) == n:
            rem2 = deadline - time.monotonic()
            if rem2 > 20:
                t2 = min(rem2 - 10, t_alloc * 0.3)
                if t2 > 10:
                    perturbed = project_simplex(global_best_seq + rng.standard_normal(n) * 0.005)
                    a_opt, obj = nesterov_smooth_optimize(perturbed, time_limit=t2, lr_init=0.003, verbose=False)
                    try_global(a_opt, f"perturb_n{n}")

    # Final: try interpolation between best sequences of different n
    # (Already handled by the parent warm-start logic above)

    # Final verification
    if global_best_seq is None:
        global_best_seq = np.ones(768) / 768
        global_best_score = float(evaluate_sequence(global_best_seq.tolist()))

    final_score = float(evaluate_sequence(global_best_seq.tolist()))
    if np.isfinite(final_score):
        global_best_score = final_score

    print(f"\nFINAL SCORE: {global_best_score:.6f} (n={len(global_best_seq)})")
    return [float(x) for x in global_best_seq]
