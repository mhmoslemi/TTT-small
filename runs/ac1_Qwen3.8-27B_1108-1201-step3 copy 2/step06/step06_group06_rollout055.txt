"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse multi-start initialization
2. Adam with adaptive log-sum-exp smoothing (main workhorse)
3. Active-set subgradient descent for fine-tuning
4. Exact 1D coordinate descent for final polish
5. Multi-length exploration

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_grad(f, tau):
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse multi-start initialization with various window shapes
2. Adam with adaptive log-sum-exp smoothing (main workhorse)
3. Active-set subgradient descent for fine-tuning near the optimum
4. Exact 1D coordinate descent for final polish
5. Multi-length exploration and re-optimization
"""
import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, tau):
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv


def adam_optimize(f0, time_limit, tau_ratio=0.08, lr_init=0.004, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    f = f / s
    conv = fast_convolve(f)
    tau = tau_ratio * conv.max()
    tau_min = 1e-9
    tau_max = 0.15 * conv.max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = smooth_grad(f, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve < 20:
            tau = max(tau * 0.997, tau_min)
        elif no_improve > 80:
            tau = min(tau * 1.003, tau_max)

        if no_improve > 200:
            lr *= 0.8
        if no_improve > 700:
            rng = np.random.default_rng(1000 + t)
            noise = 0.002 * min(1.0, no_improve / 3000.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.7
            tau = 0.03 * fast_convolve(f).max()
            if verbose:
                print(f"  Adam restart t={t}: obj={2*n*best_obj:.8f}")

        if verbose and t % 500 == 0:
            print(f"  Adam t={t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def active_set_subgrad(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    step = 2.0 / (n * 2.0)
    t0 = time.monotonic()
    t = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        M = conv.max()

        threshold = M * 0.97
        active = conv >= threshold

        g = np.zeros(n)
        active_indices = np.where(active)[0]
        for k in active_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je + 1] += 2.0 * f[k - je:k - js + 1][::-1]
        g /= max(1, len(active_indices))

        f_new = f - step * g
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            step = min(step * 1.005, 6.0 / n)
            stall = 0
        else:
            step *= 0.92
            stall += 1

        if verbose and t % 500 == 0:
            print(f"  ActiveSub t={t}: obj={2*n*best_obj:.8f}, step={step:.6f}, nactive={len(active_indices)}")

        if step < 1e-13 or stall > 8000:
            break

    return best_f, 2 * n * best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    conv = fast_convolve(f)
    S = f.sum()
    best_obj = 2 * n * conv.max() / S ** 2
    best_f = f.copy()
    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        S = f.sum()
        obj = 2 * n * conv.max() / S ** 2
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()

        improved = 0
        rng = np.random.default_rng(epoch * 13 + 7)
        indices = rng.permutation(n)

        for idx in indices:
            i = int(idx)
            ai = f[i]
            if ai < 1e-15:
                continue
            S_prime = S - ai

            d = np.zeros(2 * n - 1)
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            for k in range(k_lo, k_hi + 1):
                j = k - i
                if 0 <= j < n and j != i:
                    d[k] = 2.0 * f[j]

            e_pos = 2 * i
            e_val = 1.0 if e_pos < 2 * n - 1 else 0.0

            c = conv - ai * d
            if e_val > 0:
                c[e_pos] -= ai * ai

            best_ai = ai
            best_val = 2 * n * conv.max() / S ** 2

            cands = [0.0, ai, ai * 0.5, ai * 1.5, ai * 0.7, ai * 1.3]

            top_k = np.argsort(conv)[-8:]
            for k in top_k:
                k = int(k)
                dk = d[k]
                ck = c[k]
                ek = e_val if k == e_pos else 0.0
                if dk > 1e-15 and ek == 0:
                    cand = S_prime - 2.0 * ck / dk
                    if cand > 0:
                        cands.append(cand)
                elif dk > 1e-15 and ek > 0:
                    denom = 2.0 * S_prime - dk
                    if abs(denom) > 1e-15:
                        cand = (2.0 * ck - dk * S_prime) / denom
                        if cand > 0:
                            cands.append(cand)
                elif dk == 0 and ek > 0:
                    if S_prime > 1e-15:
                        cands.append(np.sqrt(max(0.0, ck)) + 1e-6)

            for ai_new in cands:

                if abs(ai_new - ai) < 1e-12:
                    continue
                new_conv_val = c + ai_new * d
                if e_val > 0:
                    new_conv_val = new_conv_val.copy()
                    new_conv_val[e_pos] += ai_new * ai_new
                new_S = S_prime + ai_new
                if new_S < 1e-10:
                    continue

                if new_obj < best_val - 1e-15:
                    best_val = new_obj
                    best_ai = ai_new

            if best_ai != ai and best_val < obj - 1e-15:
                delta = best_ai - ai
                conv += delta * d
                if e_val > 0:
                    conv[e_pos] += (best_ai * best_ai - ai * ai)
                f[i] = best_ai
                S += delta
                improved += 1

        if verbose and epoch % 50 == 0:
            print(f"  CoordDescent epoch={epoch}: obj={2*n*conv.max()/S**2:.8f}, improved={improved}")

        if improved == 0:
            break

    best_f = best_f / best_f.sum()
    return best_f, 2 * n * fast_convolve(best_f).max()

        active = conv >= threshold
        active_indices = np.where(active)[0]

        g = np.zeros(n)
        for k in active_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je + 1] += 2.0 * f[k - je:k - js + 1][::-1]
        g /= max(1, len(active_indices))

        f_new = f * np.exp(-eta * g)
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            eta = min(eta * 1.002, 0.2 / n)
            stall = 0
        else:
            eta *= 0.95
            stall += 1

        if verbose and t % 500 == 0:
            print(f"  Mirror t={t}: obj={2*n*best_obj:.8f}, eta={eta:.2e}")

        if eta < 1e-10 or stall > 5000:
            break

    return best_f, 2 * n * best_obj


def make_gaussian(n, center=0.4, sigma=0.25):
    x = np.arange(n) - n * center
    a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width=0.5):
    x = np.arange(n) - n / 2.0
    half = n * width / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hamming(n):
    a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n, beta=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    trans = int(n * beta / 2)
    if trans > 0:
        a[:trans] = 0.5 * (1 + np.cos(np.pi * np.arange(trans) / trans - np.pi))
        a[-trans:] = 0.5 * (1 + np.cos(np.pi * np.arange(trans) / trans))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
