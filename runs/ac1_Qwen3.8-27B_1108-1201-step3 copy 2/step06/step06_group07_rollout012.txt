"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a continuation method with
log-sum-exp smoothing, FFT-based gradient computation, mirror descent,
exact subgradient polishing, and multi-start optimization.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_smooth(p, b, beta):
    """Compute gradient of log-sum-exp(beta * b) / beta w.r.t. p.
    grad_j = 2 * sum_k w_k * b_k * p[k-j] (for valid k-j)
    = 2 * conv(w*b, p) truncated to length n
    """
    n = len(p)
    # Softmax weights
    scaled = beta * b
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w /= ws
    
    c = w * b  # weighted peaks
    
    # conv(c, p) truncated to length n
    # conv(c, p)[j] = sum_k c[k] * p[j-k] for valid indices
    # = sum_k c[k] * p[j-k] where 0<=k<len(c), 0<=j-k<n
    size = len(c) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    C = np.fft.rfft(c, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_cp = np.fft.irfft(C * P, fft_size)[:n]
    
    return 2.0 * conv_cp

def compute_subgradient_exact(p, b, top_k=20, temp=0.001):
    """Compute weighted subgradient of max(b) using top-k peaks with softmax weights."""
    n = len(p)
    K = min(top_k, len(b))
    if K < len(b):
        peak_idx = np.argpartition(b, -K)[-K:]
    else:
        peak_idx = np.arange(len(b))
    
    pv = b[peak_idx]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(K)
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    c = np.zeros(len(b))
    c[peak_idx] = w * b[peak_idx]
    
    # Same as gradient computation
    size = len(c) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    C = np.fft.rfft(c, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_cp = np.fft.irfft(C * P, fft_size)[:n]
    
    return 2.0 * conv_cp

def mirror_descent_smooth(p0, time_limit, beta_sequence, verbose=False, label=""):
    """Mirror descent with log-sum-exp smoothing and continuation on beta."""
    p = np.maximum(p0.astype(np.float64), 1e-15)
    p = p / p.sum()
    n = len(p)
    
    best_obj = fast_convolve(p).max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    it = 0
    
    for beta in beta_sequence:
        # Run mirror descent at this beta level
        lr = 0.01 / (1 + 0.1 * beta)
        no_improve = 0
        
        while time.monotonic() - t0 < time_limit:
            it += 1
            b = fast_convolve(p)
            M = b.max()
            
            # Compute gradient
            grad = compute_gradient_smooth(p, b, beta)
            
            # Mirror descent update: p <- p * exp(-lr * grad), then normalize
            log_update = -lr * grad
            log_update = np.clip(log_update, -30, 30)
            p_new = p * np.exp(log_update)
            p_new = np.maximum(p_new, 1e-15)
            ss = p_new.sum()
            if ss < 1e-10:
                break
            p_new = p_new / ss
            
            b_new = fast_convolve(p_new)
            M_new = b_new.max()
            
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_p = p_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptive step size
            if no_improve > 50:
                lr *= 0.7
                no_improve = 0
            
            # Early stop for this beta level if converged
            if no_improve > 200:

def coordinate_descent_polish(p0, time_limit, verbose=False, label=""):

    p = np.maximum(p0.astype(np.float64), 0.0)
    p = p / p.sum()
    n = len(p)

    best_p = p.copy()

        it += 1
        improved = False
        
        b = fast_convolve(p)
        M = b.max()
        peak_k = int(np.argmax(b))
        
        # Find the most contributing index to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        products = p[idx] * p[peak_k - idx]
        j_max = int(idx[np.argmax(products)])
        
        # Try reducing p[j_max] and redistributing
        old_val = p[j_max]
        if old_val < 1e-12:
            continue
        
        best_trial = None
        best_trial_obj = M
        
        for frac in [0.5, 0.7, 0.85, 0.95]:
            new_val = old_val * frac
            diff = old_val - new_val
            if diff < 1e-15:
                continue
            
            trial = p.copy()
            trial[j_max] = new_val
            
            # Redistribute to non-contributing positions
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je+1] = False
            if non_contrib.sum() > 0:
                nc = trial[non_contrib].copy()
                nc_sum = nc.sum()
                if nc_sum > 1e-15:
                    trial[non_contrib] += diff * nc / nc_sum
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_trial_obj - 1e-14:
                        best_trial_obj = obj
                        best_trial = trial
        
        if best_trial is not None:
            p = best_trial
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_p = p.copy()
            improved = True
        
        if not improved:
            # Try broader reductions
            for scale in [0.9, 0.8, 0.7]:
                trial = p.copy()
                trial[js:je+1] *= scale
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je+1] = False
                if non_contrib.sum() > 0:
                    nc = trial[non_contrib]
                    nc_sum = nc.sum()
                    if nc_sum > 1e-15:
                        total = trial.sum()
                        trial[non_contrib] += (1.0 - total) * nc / nc_sum

                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()

                            best_obj = obj
                            best_p = trial.copy()
                            improved = True
                            break
            if not improved:
                break

    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)

        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
