def cross_convolve(f, g):

    F = np.fft.rfft(f, fft_size)

def build_lp_matrix(f, lags, n):

    A = np.zeros((len(lags), n))

        i_end = min(n - 1, k)
        js = np.arange(i_start, i_end + 1)
        A[idx, js] = f[k - js]
    return A

def solve_active_lp(f, M, n, active_lags, lp_time_limit=10):

                     options={'presolve': True, 'time_limit': lp_time_limit})

def smooth_gradient_polish(f0, time_limit, tau=0.01, lr=0.001, verbose=False):
    """Polish using smooth gradient of log-sum-exp approximation to max."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute smooth gradient
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # h[j] = sum_l w[j+l] * f[l] (cross-correlation with weights)
        w_rev = w[::-1]
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_result = np.fft.irfft(A * W, fft_size)
        h = conv_result[n-1:2*n-1][::-1]
        
        grad = (2.0 / tau) * h
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
        
        f = f_new
        
        # Decay tau and lr
        if t % 1000 == 0:
            tau *= 0.95
            lr *= 0.95
    
    return best_f, 2 * n * best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
