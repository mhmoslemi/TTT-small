def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    # Standard projection
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lpsubgradient(f, M, n, m_scale=1.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M*m_scale, g >= 0."""
    target_M = M * m_scale
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, target_M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Aggressive LP-FW with adaptive M tightening (m_scale < 1 for feasibility checking)
2. Flat-target gradient descent (minimizes variance of convolution, finds different basins)
3. More diverse parametric initializations
4. Bisection-based lower bound certificate
5. Better time allocation across phases
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build the LP constraint matrix efficiently using banded structure."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A_ub[j:end, j] = f[:end - j]
    return A_ub


def solve_lp_oracle(f, M, n, m_scale=1.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M*m_scale for all k, g >= 0."""
    target_M = M * m_scale
    c_obj = -np.ones(n)
    A_ub = build_lp_matrix(f, n)
    b_ub = np.full(2 * n - 1, target_M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with adaptive M scaling and aggressive tightening."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_scale = 0.95  # Start aggressive

    while time.monotonic() - t0 < time_limit:
        it += 1

        g_raw = solve_lp_oracle(f, M, n, m_scale)

        if g_raw is None:
            m_scale = min(m_scale * 1.04, 3.0)
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 7 + 3)
                noise = rng.standard_normal(n) * 0.004
                f = project_simplex(best_f + noise)
                c = fast_convolve(f)
                M = c.max()
                m_scale = 0.95
                no_improve = 0
            continue

        g_sum = g_raw.sum()
        if g_sum < 1e-15:
            m_scale = min(m_scale * 1.04, 3.0)
            continue
        g = g_raw / g_sum

        if m_scale > 1.0:
            m_scale *= 0.97

        diff = np.max(np.abs(g - f))
        if diff < 1e-11:
            no_improve += 1
            if no_improve > 30:
                m_scale = max(0.9, m_scale - 0.02)
            if no_improve > 100:
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                c = fast_convolve(f)
                M = c.max()
                m_scale = 0.93
                no_improve = 0
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            c = fast_convolve(f)
            M = c.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                m_scale = max(0.92, m_scale - 0.01)

        if no_improve > 250:
            break

    return best_f, best_obj


def smooth_gd(f0, time_limit, verbose=False):
    """Log-sum-exp smoothed gradient descent with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()

    t0 = time.monotonic()
    betas = [4, 8, 16, 32, 64, 128, 256, 512]
    mu = 0.7
    momentum = np.zeros(n)

    for beta in betas:
        if time.monotonic() - t0 > time_limit * 0.85:
            break

        stage_t0 = time.monotonic()
        stage_time = min(time_limit * 0.05, (time_limit - (time.monotonic() - t0)) * 0.25)
        lr = 0.3 / (beta * max(1, n) ** 0.2)
        no_improve = 0

        while time.monotonic() - stage_t0 < stage_time:
            c = fast_convolve(f)
            scaled = beta * c
            scaled -= scaled.max()
            exp_s = np.exp(scaled)
            w = exp_s / exp_s.sum()

            w_rev = w[::-1].copy()
            conv_len = n + len(w_rev) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]

            momentum = mu * momentum - lr * g
            f_trial = project_simplex(f + momentum)

            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()

            if obj_trial <= c.max() * 1.0005:
                f = f_trial
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.01, 1.0)
                else:
                    no_improve += 1
                    lr *= 0.96
            else:
                no_improve += 1
                lr *= 0.7
                momentum *= 0.5

            if no_improve > 40:
                break

    return best_f, best_obj


def flat_target_gd(f0, time_limit):
    """Minimize variance of convolution (flat target approach). Finds different basins."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)
    target = 1.0 / (2 * n - 1)

    best_f = f.copy()
    c = fast_convolve(f)
    best_max = c.max()
    best_var = np.sum((c - target) ** 2)

    t0 = time.monotonic()
    lr = 0.02
    mu = 0.5
    momentum = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(f)
        residual = 2.0 * (c - target)

        res_rev = residual[::-1].copy()
        conv_len = n + len(res_rev) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        F = np.fft.rfft(f, fft_size)
        R = np.fft.rfft(res_rev, fft_size)
        conv = np.fft.irfft(F * R, fft_size)[:conv_len]
        indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
        grad = 2.0 * conv[indices]

        momentum = mu * momentum - lr * grad
        f_trial = project_simplex(f + momentum)

        c_trial = fast_convolve(f_trial)
        obj_trial = np.sum((c_trial - target) ** 2)
        max_trial = c_trial.max()

        acceptance = max_trial < best_max * 1.001 and obj_trial < best_var * 1.01
        if acceptance:
            f = f_trial
            if max_trial < best_max:
                best_max = max_trial
                best_f = f.copy()
            if obj_trial < best_var:
                best_var = obj_trial
                lr = min(lr * 1.005, 0.1)
            else:
                lr *= 0.97
        else:
            lr *= 0.8
            momentum *= 0.5

    return best_f, best_max


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1.0 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell_flat(n):
    x = np.linspace(-1, 1, n)
    a = np.cos(np.pi * x / 2.0) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_parametric(n, center_width, edge_level):
    x = np.arange(n) - n / 2.0
    sigma = n * center_width
    bump = np.exp(-0.5 * (x / sigma) ** 2)
    a = (1.0 - edge_level) * bump + edge_level / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_interp(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def make_smooth_interp(f, new_n):
    """Interpolate with additional Gaussian smoothing."""
    f_interp = make_interp(f, new_n)
    kernel_size = max(3, new_n // 200)
    if kernel_size % 2 == 0:
        kernel_size += 1
    sigma_k = kernel_size / 4.0
    x_k = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x_k / sigma_k) ** 2)
    kernel /= kernel.sum()
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    s = f_smooth.sum()
    if s > 1e-15:
        f_smooth /= s
    return f_smooth

    candidates = []

    for n_val in n_values:

            candidates.append(("parent", parent))

            candidates.append((f"parent_i{n_val}", make_smooth_interp(parent, n_val)))

        candidates.append((f"gauss25_{n_val}", make_gaussian(n_val, 0.25)))
        candidates.append((f"gauss30_{n_val}", make_gaussian(n_val, 0.30)))
        candidates.append((f"gauss35_{n_val}", make_gaussian(n_val, 0.35)))
        candidates.append((f"gauss40_{n_val}", make_gaussian(n_val, 0.40)))
        candidates.append((f"raisedcos_{n_val}", make_raised_cosine(n_val)))
        candidates.append((f"bellflat_{n_val}", make_bell_flat(n_val)))
        candidates.append((f"param_a_{n_val}", make_parametric(n_val, 0.20, 0.4)))
        candidates.append((f"param_b_{n_val}", make_parametric(n_val, 0.30, 0.2)))
        candidates.append((f"param_c_{n_val}", make_parametric(n_val, 0.25, 0.5)))

    scored_candidates = []
    for name, a in candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:

        a = a / s

        try_seq(a, name)

        elif r == 2:
            noise = rng.standard_normal(n_best) * 0.008
            f_init = project_simplex(best_seq + noise)
        elif r == 3:

            f_init = 0.7 * best_seq + 0.3 * other
            f_init = f_init / f_init.sum()
        elif r == 4:
            other = make_raised_cosine(n_best)
            f_init = 0.6 * best_seq + 0.4 * other
            f_init = f_init / f_init.sum()
        else:
            other = make_bell_flat(n_best)
            f_init = 0.5 * best_seq + 0.3 * other + 0.2 * best_seq[::-1]

    return [float(x) for x in best_seq]
