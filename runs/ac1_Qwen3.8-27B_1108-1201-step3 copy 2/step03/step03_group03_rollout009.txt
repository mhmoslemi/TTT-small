import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        pmax = conv.max()
        if pmax < 1e-15:
            break

        # Identify active peaks: those within eps_rel of the max
        threshold = pmax * (1.0 - eps_rel)
        active_mask = conv >= threshold
        active_peaks = np.where(active_mask)[0]

        # Limit to top_k_max if too many
        if len(active_peaks) > top_k_max:
            active_vals = conv[active_peaks]
            idx_sort = np.argsort(active_vals)[::-1][:top_k_max]
            active_peaks = active_peaks[idx_sort]

        # Weights: proportional to how close to the max (softer focus on the top)
        peak_vals = conv[active_peaks]
        # Normalize weights so they sum to 1
        w = peak_vals / peak_vals.sum()

        sg = compute_subgrad(a, active_peaks, w)

        # Momentum update
        v = momentum * v - lr * sg
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive: reduce lr and perturb when stalled
        if stall > 80:
            lr = max(1e-5, lr * 0.6)
            v *= 0.4
            rng = np.random.default_rng(stall * 7919 + n)
            noise_scale = 0.003 * (stall / 80.0)
            noise_scale = min(noise_scale, 0.02)
            a = project_simplex(a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            stall = 0
            if lr < 2e-6:
                break

        # Slowly increase eps_rel to consider more peaks as we converge
        eps_rel = min(0.05, eps_rel + 0.00001)

    return best_a, best_obj


def soft_max_subgrad(a0, time_limit, lr_init=0.01, momentum=0.9,
                     temp_init=0.05, top_k=20):
    """Subgradient with soft-max weighting over peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv[peaks]
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]

        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        # Soft-max weights
        scaled = (peak_vals - pmax) / temp
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = compute_subgrad(a, peaks, w)
        v = momentum * v - lr * sg
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Anneal the temperature
        temp = max(0.001, temp * 0.9995)

        if stall > 150:
            lr = max(1e-5, lr * 0.5)
            v *= 0.3
            rng = np.random.default_rng(stall * 31 + n)
            a = project_simplex(a + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(a)
            stall = 0
            if lr < 2e-6:
                break

    return best_a, best_obj


def simulated_annealing(a0, time_limit, t0_temp=0.01, t_min=1e-6,
                        alpha=0.9997, moves_per_step=1):
    """Simulated annealing with targeted moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    cur_obj = best_obj

    rng = np.random.default_rng(12345)
    T = t0_temp
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        pmax = conv.max()

        # Choose move type
        move_type = rng.integers(0, 3)

        if move_type == 0:
            # Targeted move: reduce mass at a "hot" coordinate, add to "quiet"
            k_star = int(np.argmax(conv))
            contrib = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            for j in range(j_start, j_end + 1):
                contrib[j] = a[j] * a[k_star - j]

            hot_mask = contrib > 0.1 * contrib.max()
            if not hot_mask.any():
                hot_mask = a > a.max() * 0.1

            i = int(rng.choice(np.where(hot_mask)[0]))
            quiet_mask = a < np.median(a)
            if not quiet_mask.any():
                quiet_mask = np.ones(n, dtype=bool)
                quiet_mask[:n // 2] = False
            j = int(rng.choice(np.where(quiet_mask)[0]))
            if i == j:
                j = (j + 1) % n

            delta = a[i] * rng.uniform(0.01, 0.08)
            delta = min(delta, a[i])
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # Random pair swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                j = (j + 1) % n
            delta = min(a[i], a[j] + 1) * rng.uniform(0.005, 0.05)
            delta = min(delta, a[i])
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            # Gaussian perturbation
            scale = 0.002 + 0.01 * T / t0_temp
            a_new = project_simplex(a + rng.standard_normal(n) * scale)

        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn

        conv_new = fast_convolve(a_new)
        new_obj = 2.0 * n * conv_new.max()

        delta_obj = new_obj - cur_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            a = a_new
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()

        T = max(t_min, T * alpha)
        step += 1

    return best_a, best_obj


def coordinate_descent(a0, max_rounds=40, top_coords=80):
    """Coordinate descent: pairwise mass transfers with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    # Identify active coordinates
    active_thresh = 1e-6
    active = np.where(a > active_thresh)[0]
    if len(active) > top_coords:
        vals = a[active]
        idx_sort = np.argsort(vals)[::-1][:top_coords]
        active = active[idx_sort]

    for rnd in range(max_rounds):
        improved = False
        n_active = len(active)
        for idx_i in range(n_active):
            i = active[idx_i]
            if a[i] < active_thresh:
                continue
            for idx_j in range(idx_i + 1, n_active):
                j = active[idx_j]
                if a[j] < active_thresh:
                    continue
                # Try transferring mass in both directions with a few delta values
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    for direction in [1, -1]:
                        if direction == 1:
                            delta = a[i] * frac
                        else:
                            delta = a[j] * frac
                        if delta < 1e-10:
                            continue
                        a_test = a.copy()
                        if direction == 1:
                            a_test[i] -= delta
                            a_test[j] += delta
                        else:
                            a_test[j] -= delta
                            a_test[i] += delta
                        if a_test[i] < -1e-12 or a_test[j] < -1e-12:
                            continue
                        a_test = np.maximum(a_test, 0.0)
                        st = a_test.sum()
                        if st < 1e-12:
                            continue
                        a_test /= st
                        conv_test = fast_convolve(a_test)
                        obj_test = 2.0 * n * conv_test.max()
                        if obj_test < best_obj - 1e-14:
                            a = a_test
                            best_obj = obj_test
                            conv = conv_test
                            improved = True

        if not improved:
            break

    return a


def gen_qr_init(n, p=None):
    """Generate initialization based on quadratic residues."""
    if p is None:
        # Find a prime >= n
        p = n
        while True:
            if all(p % i != 0 for i in range(2, int(np.sqrt(p)) + 1)):
                break
            p += 1
    # QR indicator
    qr = np.zeros(p)
    for j in range(1, p):
        if pow(j, (p - 1) // 2, p) == 1:
            qr[j] = 1.0
    # Interpolate to length n
    x_src = np.linspace(0, 1, p)
    x_dst = np.linspace(0, 1, n)
    a = np.interp(x_dst, x_src, qr)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_chirp_init(n):
    """Generate a windowed chirp initialization."""
    x = np.arange(n, dtype=np.float64)
    # Chirp-like frequency modulation
    phase = 2.0 * np.pi * (x * x / n)
    a = 1.0 + 0.5 * np.cos(phase)
    # Apply a Hann window
    window = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    a = a * window
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_two_scale_init(n, rng):
    """Two-scale: most entries small, a few large (Bose-like)."""
    a = np.ones(n) * 0.1
    # Pick ~n/4 "active" positions
    n_active = max(4, n // 4)
    active_pos = rng.choice(n, size=n_active, replace=False)
    a[active_pos] = 1.0
    return a / a.sum()


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    inits['qr'] = gen_qr_init(n)
    inits['chirp'] = gen_chirp_init(n)
    inits['two_scale'] = gen_two_scale_init(n, rng)

    # Palindromic
    half = np.ones(n // 2) * (1.0 + 0.3 * np.sin(2 * np.pi * x[:n//2] / n))
    half = np.maximum(half, 0.1)
    pal = np.concatenate([half, half[::-1]])[:n]
    inits['palindromic'] = pal / pal.sum()

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        elif len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: active-set subgradient + soft-max subgradient +
    simulated annealing + coordinate descent. Explores multiple sequence
    lengths with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
    except Exception:
        pass

    # Phase 1: Broad scan with fast subgradient
    print("=== Phase 1: Broad Scan ===")
    lengths = [384, 512, 640, 768]
    if parent is not None:
        lengths.insert(0, len(parent))

    # Remove duplicates, keep sorted
    lengths = sorted(set(lengths))

    results = []
    phase1_end = time.monotonic() + min(400.0, (deadline - time.monotonic()) * 0.3)

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_inits(n, parent, rng)
        prio = ['parent', 'parent_ds', 'parent_up', 'qr', 'chirp', 'two_scale',
                'bimodal', 'uniform', 'palindromic', 'random', 'triangle', 'gauss', 'ramp']
        names = [nm for nm in prio if nm in inits]
        names = names[:6]

        t_per = max(10.0, 200.0 / (len(lengths) * 6))

        for name in names:
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 8)
            if t_alloc < 5:
                break

            a0 = inits[name]
            # Quick active-set subgrad
            a_opt, obj_opt = active_set_subgrad(a0, t_alloc, lr_init=0.012,
                                                 momentum=0.9, eps_rel=0.01)
            # Quick soft-max subgrad polish
            a_opt2, obj2 = soft_max_subgrad(a_opt, min(5.0, t_alloc * 0.2),
                                             lr_init=0.008, momentum=0.9,
                                             temp_init=0.03, top_k=15)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen_keys = set()
    for score, n, name, a_opt in results[:10]:
        key = (n, name)
        if key not in seen_keys:
            top_list.append((n, a_opt))
            seen_keys.add(key)
        if len(top_list) >= 5:
            break
    if best_sequence is not None:
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 250.0
    t_per = max(20.0, remaining / max(1, len(top_list) * 3))

    for n, a0 in top_list:
        if time.monotonic() > deadline - 220:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 180)
        if t_alloc < 10:
            break

        # Multiple subgradient variants
        for variant in ['active', 'softmax', 'active_fine']:
            if time.monotonic() > deadline - 170:
                break
            t_s = t_alloc / 3.0
            if t_s < 5:
                break

            if variant == 'active':
                a_opt, obj_opt = active_set_subgrad(a0, t_s, lr_init=0.008,
                                                     momentum=0.92, eps_rel=0.003)
            elif variant == 'softmax':
                a_opt, obj_opt = soft_max_subgrad(a0, t_s, lr_init=0.006,
                                                   momentum=0.92, temp_init=0.02,
                                                   top_k=25)
            else:
                # Fine active-set with very tight epsilon
                a_opt, obj_opt = active_set_subgrad(a0, t_s, lr_init=0.004,
                                                     momentum=0.94, eps_rel=0.001)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep n={n} {variant}: {score:.6f}")

    # Phase 3: Simulated annealing
    print("=== Phase 3: Simulated Annealing ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30:
            t_sa = min(80.0, remaining * 0.5)
            a_sa, obj_sa = simulated_annealing(best_sequence, t_sa,
                                                t0_temp=0.005, t_min=1e-7,
                                                alpha=0.9998)
            score_sa = float(evaluate_sequence(a_sa.tolist()))
            if np.isfinite(score_sa) and score_sa < best_score:
                best_score = score_sa
                best_sequence = a_sa.copy()
                print(f"  SA: {score_sa:.6f}")

    # Phase 4: Coordinate descent + final polish
    print("=== Phase 4: Coordinate Descent & Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 40.0

        # Coordinate descent
        t_cd = min(60.0, remaining * 0.4)
        if t_cd > 5:
            a_cd = coordinate_descent(best_sequence, max_rounds=30, top_coords=60)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")

        # Final fine subgradient cycles
        remaining = deadline - time.monotonic() - 25.0
        for cycle in range(4):
            if time.monotonic() > deadline - 20:
                break
            t_cyc = min(remaining / (4 - cycle), 25.0)
            if t_cyc < 4:
                break

            # Very fine active-set
            a_opt, obj_opt = active_set_subgrad(best_sequence, t_cyc * 0.6,
                                                lr_init=0.002, momentum=0.95,
                                                eps_rel=0.0005)
            a_opt2, obj2 = soft_max_subgrad(a_opt, t_cyc * 0.3,
                                             lr_init=0.001, momentum=0.95,
                                             temp_init=0.005, top_k=40)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Fine cycle {cycle}: {score:.6f}")

            # Quick CD
            a_cd2 = coordinate_descent(best_sequence, max_rounds=15, top_coords=50)
            score_cd2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(score_cd2) and score_cd2 < best_score:
                best_score = score_cd2
                best_sequence = a_cd2.copy()
                print(f"  CD cycle {cycle}: {score_cd2:.6f}")

        # Targeted perturbations around the best
        if time.monotonic() < deadline - 15:
            n_b = len(best_sequence)
            for ps in [0.002, 0.005, 0.01]:
                if time.monotonic() > deadline - 10:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 8:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, _ = active_set_subgrad(a_p, 3.0, lr_init=0.005,
                                                   momentum=0.9, eps_rel=0.005)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
