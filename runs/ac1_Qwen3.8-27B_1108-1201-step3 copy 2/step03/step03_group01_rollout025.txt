    best_t = 0.0
    best_f = max_obj

    # Quick candidate evaluation
    candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 1.0, 0.7, 0.4]
    for t in candidates:
        if t <= 0:
            continue
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        conv_try = fast_convolve(a_try)
        f_try = 2.0 * n * conv_try.max()
        if f_try < best_f - 1e-14:
            best_f = f_try
            best_t = t

    # Refine around best_t with golden section
    if best_t > 0.01 and best_t < 0.99:
        lo = max(0.001, best_t * 0.5)
        hi = min(1.5, best_t * 1.5)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(20):
            if hi - lo < 1e-6:
                break
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            a1 = np.maximum(a + c1 * d, 0.0)
            s1 = a1.sum()
            f1 = 2.0 * n * fast_convolve(a1 / s1).max() if s1 > 1e-15 else np.inf
            a2 = np.maximum(a + c2 * d, 0.0)
            s2 = a2.sum()
            f2 = 2.0 * n * fast_convolve(a2 / s2).max() if s2 > 1e-15 else np.inf
            if f1 < f2:
                hi = c2
            else:
                lo = c1
            mid_t = (lo + hi) / 2.0
            a_m = np.maximum(a + mid_t * d, 0.0)
            s_m = a_m.sum()
            if s_m > 1e-15:
                f_m = 2.0 * n * fast_convolve(a_m / s_m).max()
                if f_m < best_f - 1e-14:
                    best_f = f_m
                    best_t = mid_t

    return best_t, best_f


def lp_optimize(a0, time_limit, verbose=False):
    """
    Main LP-based iterative optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_norm_obj = 2.0 * n * M  # normalized objective (sum=1)
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    consecutive_fails = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed

        # Solve LP
        g, g_sum = lp_direction(a, M, margin=1e-10)

        if g is None or g_sum <= 1.0 + 1e-9:
            consecutive_fails += 1
            no_improve += 1
        else:
            consecutive_fails = 0
            # Direction: normalize g to sum=1, then direction is g_norm - a
            g_norm = g / g_sum
            d = g_norm - a
            norm_d = np.linalg.norm(d)

            if norm_d < 1e-14:
                no_improve += 1
            else:
                # Line search
                t_best, obj_best = line_search(a, d, n, best_norm_obj)

                if t_best > 0 and obj_best < best_norm_obj - 1e-13:
                    a_new = a + t_best * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                        conv = fast_convolve(a)
                        M = conv.max()
                        cur_obj = 2.0 * n * M
                        if cur_obj < best_norm_obj:
                            best_norm_obj = cur_obj
                            best_a = a.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        a = best_a.copy()
                        conv = fast_convolve(a)
                        M = conv.max()
                        no_improve += 1
                else:
                    no_improve += 1

        iteration += 1

        if verbose and iteration % 50 == 0:
            print(f"      LP iter {iteration}, norm_obj={best_norm_obj:.10f}, "
                  f"elapsed={elapsed:.0f}s, no_improve={no_improve}")

        # Handle stagnation with perturbation
        max_no_improve = 80 + 20 * min(perturb_count, 10)
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 31 + 7)
            scale = 0.003 + 0.004 * min(perturb_count, 8)
            # Structured perturbation: use autocorrelation peaks to guide
            conv = fast_convolve(best_a)
            top_peaks = np.argpartition(conv, -10)[-10:]
            noise = rng.standard_normal(n) * scale
            # Extra perturbation at peak-contributing positions
            for pk in top_peaks[:5]:
                idx = int(pk)
                for j in range(max(0, idx - n + 1), min(n, idx + 1)):
                    noise[j] += scale * 2.0 * best_a[idx - j] / (best_a.sum() + 1e-15)
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1

            if perturb_count > 40:
                break

        if consecutive_fails > 200:
            break

    return best_a, best_norm_obj


def coordinate_polish(a0, time_limit, seed=42):
    """
    Coordinate descent and pairwise swap polishing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        # Choose strategy based on iteration
        if iteration % 3 == 0:
            # Single coordinate perturbation
            j = rng.integers(n)
            deltas = [1e-3, 5e-4, 1e-4, -1e-3, -5e-4, -1e-4]
            for delta in deltas:
                a_try = a.copy()
                a_try[j] = max(0.0, a_try[j] + delta)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

        elif iteration % 3 == 1:
            # Pairwise mass transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            for delta in [2e-3, 1e-3, 5e-4, -2e-3, -1e-3, -5e-4]:
                a_try = a.copy()
                src = i if a_try[i] < a_try[j] else j
                dst = j if src == i else i
                if a_try[src] < abs(delta):
                    continue
                a_try[src] -= delta * np.sign(delta)
                a_try[dst] += delta * np.sign(delta)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

        else:
            # Targeted perturbation at peak-contributing positions
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            # Positions that contribute most to the peak
            j_start = max(0, peak_k - n + 1)
            j_end = min(n - 1, peak_k)
            contrib = a[j_start:j_end + 1] * a[peak_k - (j_start + np.arange(j_end - j_start + 1))]
            top_idx = np.argsort(contrib)[-5:] + j_start
            j = rng.choice(top_idx)
            for delta in [1e-3, 5e-4, -1e-3, -5e-4]:
                a_try = a.copy()
                a_try[j] = max(0.0, a_try[j] + delta)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

        iteration += 1

        if no_improve > 5000:
            # Perturb to escape
            scale = 0.002 + 0.002 * (no_improve % 10)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            no_improve = 0

    return best_a, best_obj


def try_length(n, time_limit, parent, rng, verbose=False):
    """Try optimizing at a given length n."""
    results = []

    # Generate initializations
    init_list = structured_init(n, rng)
    init_list.append(spectral_init(n, rng))

    # Add parent-based init if available
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                init_list.append(p / ps)
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    init_list.append(pp / pps)
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    init_list.append(pp / pps)

    # Time per initialization
    n_inits = len(init_list)
    t_per_init = max(10.0, (time_limit - 30.0) / n_inits)

    # Evaluate and sort initializations by quality
    scored_inits = []
    for init in init_list:
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 1e-15:
            init = init / s
        obj = compute_obj(init)
        scored_inits.append((obj, init))
    scored_inits.sort(key=lambda x: x[0])

    # Run LP on top initializations
    for obj_init, init in scored_inits[:min(4, len(scored_inits))]:
        t_alloc = min(t_per_init, time_limit - (time.monotonic() - t0_global) if time.monotonic() > 0 else t_per_init)
        a_opt, obj_opt = lp_optimize(init, time_limit=t_alloc, verbose=verbose)
        score = float(evaluate_sequence(a_opt.tolist()))
        results.append((score, a_opt.copy()))
        if verbose:
            print(f"    n={n}, init_obj={obj_init:.6f} -> opt={score:.6f}")

    if not results:
        a_fallback = np.ones(n) / n
        results.append((float(evaluate_sequence(a_fallback.tolist())), a_fallback))

    results.sort(key=lambda x: x[0])
    return results[0]

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP optimization with spectral init, multi-length,
    coordinate polish, and adaptive restarts.
    """
    global t0_global
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t0_global = time.monotonic()

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}")
    except Exception:
        pass

    # Phase 1: Quick scan across lengths and initializations
    print("=== Phase 1: Multi-length scan ===")
    t1_start = time.monotonic()
    phase1_budget = 400.0
    lengths_to_try = [512]
    # Also try 1024 if time permits
    if parent is not None and len(parent) > 1024:
        lengths_to_try.insert(0, 1024)

    # Allocate time: give more to 512 (proven length)
    if 1024 in lengths_to_try:
        t_allocs = {1024: phase1_budget * 0.4, 512: phase1_budget * 0.6}
    else:
        t_allocs = {512: phase1_budget}

    phase1_results = []
    for n in lengths_to_try:
        if time.monotonic() > deadline - 1000:
            break
        t_here = min(t_allocs.get(n, 100.0), deadline - time.monotonic() - 900)
        if t_here < 30:
            continue
        print(f"  Trying n={n}, budget={t_here:.0f}s")
        best_n, best_seq = try_length(n, t_here, parent, rng, verbose=True)
        phase1_results.append((best_n, best_seq))
        if best_n < best_score:
            best_score = best_n
            best_sequence = best_seq.copy()

    print(f"Phase 1 done ({time.monotonic()-t1_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: Deep LP optimization on the best result
    print("=== Phase 2: Deep LP optimization ===")
    t2_start = time.monotonic()
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 400.0
        if remaining > 60:
            # Run LP for a good chunk of time
            t_lp = min(remaining * 0.7, 1200.0)
            print(f"  Deep LP on n={n_best}, budget={t_lp:.0f}s")
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_lp, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  After deep LP: {score:.8f}")

    print(f"Phase 2 done ({time.monotonic()-t2_start:.0f}s). Best: {best_score:.6f}")

    # Phase 3: Coordinate descent polishing
    print("=== Phase 3: Coordinate polish ===")
    t3_start = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30:
            t_polish = min(remaining * 0.6, 300.0)
            print(f"  Polish budget: {t_polish:.0f}s")
            a_pol, obj_pol = coordinate_polish(best_sequence, time_limit=t_polish, seed=seed + 1)
            score = float(evaluate_sequence(a_pol.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pol.copy()
                print(f"  After polish: {score:.8f}")

    print(f"Phase 3 done ({time.monotonic()-t3_start:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Short LP restarts with perturbation
    print("=== Phase 4: Perturb & re-optimize ===")
    t4_start = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 30:
            n_best = len(best_sequence)
            n_attempts = max(1, int(remaining / 40))
            t_per = remaining / n_attempts

            for pi in range(n_attempts):
                if time.monotonic() > deadline - 15:
                    break
                if t_per < 10:
                    break

                # Decreasing perturbation scale
                scale = 0.005 / (pi + 1)
                noise = rng.standard_normal(n_best) * scale
                a_pert = best_sequence + noise
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                t_alloc = min(t_per, deadline - time.monotonic() - 10)
                if t_alloc < 5:
                    break

                a_opt, obj_opt = lp_optimize(a_pert, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi}: {score:.8f}")

    print(f"Phase 4 done ({time.monotonic()-t4_start:.0f}s). Best: {best_score:.6f}")

    # Phase 5: Final short polish
    print("=== Phase 5: Final polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10:
            a_pol, obj_pol = coordinate_polish(best_sequence, time_limit=remaining, seed=seed + 99)
            score = float(evaluate_sequence(a_pol.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pol.copy()
                print(f"  Final polish: {score:.8f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
