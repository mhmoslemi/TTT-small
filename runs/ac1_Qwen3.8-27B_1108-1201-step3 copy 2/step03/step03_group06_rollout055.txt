    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i in [0,n-1]
    # Compute via direct summation (O(n^2) but n~320 is fine)
    grad = np.zeros(n)
    for i in range(n):
        # sum_k w[k] * a[k-i] where 0 <= k-i < n means i <= k < i+n
        k_start = i
        k_end = min(len(w), i + n)
        if k_end > k_start:
            grad[i] = 2.0 * np.dot(w[k_start:k_end], a[:k_end - k_start])
    return grad


def lp_optimize_loop(a_start, time_limit, seed=0, verbose=False, use_full_lp=True):
    """Main LP optimization loop with line search and perturbations."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_start.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    perturb_scale = 0.005

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Solve LP
        if use_full_lp:
            b = solve_lp_full(a, M)
        else:
            top_k = max(30, n // 5)
            b = solve_lp_topk(a, M, conv, top_k)
            if b is None:
                b = solve_lp_full(a, M)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        # Check if direction is close to current (fixed point detection)
        dist = np.linalg.norm(b - a, 1)
        if dist < 0.005:
            no_improve += 5
        else:
            # Line search
            t_best, obj_new = line_search(a, b, n)
            if t_best > 0 and obj_new < best_obj - 1e-12:
                a = (1.0 - t_best) * a + t_best * b
                a = np.maximum(a, 0.0)
                s = a.sum()
                a = a / s
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Handle stagnation
        if no_improve >= 30:
            # Try gradient step
            grad = compute_subgradient(a, fast_convolve(a), beta=500)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                a_candidate = project_simplex(a - 0.005 * grad / gnorm)
                obj_cand = obj_normalized(a_candidate)
                if obj_cand < best_obj:
                    best_obj = obj_cand
                    best_a = a_candidate.copy()
                    a = a_candidate
                    no_improve = 0
                else:
                    no_improve += 5

        if no_improve >= 60:
            # Perturbation
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            no_improve = 0
            perturb_scale = min(0.05, perturb_scale * 1.5)
        elif no_improve >= 15:
            # Small perturbation
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            no_improve = 5
        else:
            perturb_scale = max(0.003, perturb_scale * 0.95)

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.8f}, no_improve={no_improve}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * len(best_a) * conv_final.max()
    return best_a, final_obj


def gradient_finetune(a_start, time_limit, seed=0, verbose=False):
    """Smooth gradient descent with adaptive beta and step size."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_start.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a_start, obj_normalized(a_start)
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    beta = 500.0
    lr = 0.003
    momentum = 0.92
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        grad = compute_subgradient(a, conv, beta=beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        obj_new = obj_normalized(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 100:
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = 0.003
                no_improve = 0

        beta = min(5000.0, beta * 1.002)

        if verbose and no_improve % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Grad iter: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.6f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * len(best_a) * conv_final.max()
    return best_a, final_obj


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    xm = x - (n - 1) / 2.0

    # Hann window
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    hann = np.maximum(hann, 0.01)
    hann /= hann.sum()
    inits['hann'] = hann

    # Hamming window
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    ham = np.maximum(ham, 0.01)
    ham /= ham.sum()
    inits['hamming'] = ham

    # Blackman window
    black = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    black = np.maximum(black, 0.005)
    black /= black.sum()
    inits['blackman'] = black

    # Raised cosine (smooth)
    rc = np.cos(np.pi * xm / (n / 2.0)) ** 2
    rc = np.clip(rc, 0.001, None)
    rc /= rc.sum()
    inits['raised_cos'] = rc

    # Gaussian center
    g = np.exp(-0.5 * (xm / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Bimodal edges
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    # Smooth edges
    for i in range(min(5, edge)):
        fade = (i + 1) / (min(5, edge) + 1)
        bi[i] *= fade
        bi[n-1-i] *= fade
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Triangular (peak center)
    tri = 1.0 - 2.0 * np.abs(xm) / (n - 1)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    inits['tri'] = tri

    # Random dirichlet
    inits['rand_spiky'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['rand_smooth'] = rng.dirichlet(np.ones(n) * 1.0)

    # Sum of two Gaussians
    c1, c2 = rng.uniform(0.2, 0.4) * n, rng.uniform(0.6, 0.8) * n
    mix = np.exp(-0.5 * (xm / (0.15 * n)) ** 2) + np.exp(-0.5 * (x - c2) / (0.15 * n)) ** 2
    mix = np.maximum(mix, 0.001)
    mix /= mix.sum()
    inits['mix2'] = mix

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-based direction finding with line search,
    gradient fine-tuning, and strategic perturbations.
    Focuses primarily on the parent sequence (length 320) with brief exploration
    of other lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s)) - 30.0

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p
            print(f"Parent: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    # ========== Phase 1: Quick Exploration (60s) ==========
    print("=== Phase 1: Quick Exploration ===")
    t_p1 = time.monotonic()
    p1_time = 60.0

    explore_lengths = [256, 320, 384, 448]
    if parent is not None:
        explore_lengths = [len(parent)] + [l for l in explore_lengths if l != len(parent)]

    for n_exp in explore_lengths[:4]:
        if time.monotonic() > t_p1 + p1_time - 10:
            break
        inits = generate_structured_inits(n_exp, rng)
        # If parent exists and matches this length, include it
        if parent is not None and len(parent) == n_exp:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 0.01:
                inits['parent'] = pp / ps

        init_names = list(inits.keys())
        # Prioritize
        priority = ['parent', 'hann', 'blackman', 'raised_cos', 'rand_smooth', 'bimodal']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        t_alloc_each = max(5.0, (p1_time - (time.monotonic() - t_p1) - 5) /
                           max(1, len(explore_lengths) * 3))

        for name in init_names:
            if time.monotonic() > t_p1 + p1_time - 5:
                break
            t_alloc = min(t_alloc_each, t_p1 + p1_time - time.monotonic() - 3)
            if t_alloc < 4:
                break
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize_loop(a0, time_limit=t_alloc,
                                              seed=n_exp * 10 + hash(name) % 100,
                                              use_full_lp=(n_exp <= 384))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase1: n={n_exp}, {name}, score={score:.8f}")

    print(f"Phase 1 done. Best: {best_score:.8f}")

    # ========== Phase 2: Deep Optimization on Best (1100s) ==========
    print("=== Phase 2: Deep LP Optimization ===")
    t_p2 = time.monotonic()
    p2_time = 1100.0

    if best_sequence is None:
        best_sequence = np.ones(320) / 320.0
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    n_best = len(best_sequence)
    print(f"  Optimizing length n={n_best}, current best={best_score:.8f}")

    # Split into multiple LP rounds with perturbations between
    num_rounds = 4
    time_per_round = p2_time / num_rounds

    for round_i in range(num_rounds):
        if time.monotonic() > t_p2 + p2_time - 30:
            break
        t_alloc = min(time_per_round - 10, t_p2 + p2_time - time.monotonic() - 30)
        if t_alloc < 20:
            break

        start_a = best_sequence.copy()
        if round_i > 0:
            # Perturb start for diversity
            noise_scale = 0.01 * (1 + round_i * 0.3)
            noise = rng.standard_normal(n_best) * noise_scale
            start_a = project_simplex(best_sequence + noise)

        a_opt, obj_opt = lp_optimize_loop(
            start_a, time_limit=t_alloc,
            seed=1000 + round_i * 7,
            use_full_lp=True,
            verbose=(round_i == 0)
        )
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Round {round_i}: score={score:.8f}")
        else:
            print(f"  Round {round_i}: no improvement (got {score:.8f})")

    print(f"Phase 2 done. Best: {best_score:.8f}")

    # ========== Phase 3: Gradient Fine-tuning (350s) ==========
    print("=== Phase 3: Gradient Fine-tuning ===")
    t_p3 = time.monotonic()
    p3_time = 350.0

    n_best = len(best_sequence)
    grad_configs = [
        {'beta_start': 500},
        {'beta_start': 1000},
        {'beta_start': 200},
    ]

    time_per_grad = p3_time / (len(grad_configs) + 2)

    for i, cfg in enumerate(grad_configs):
        if time.monotonic() > t_p3 + p3_time - 10:
            break
        t_alloc = min(time_per_grad - 5, t_p3 + p3_time - time.monotonic() - 10)
        if t_alloc < 10:
            break

        a_opt, obj_opt = gradient_finetune(
            best_sequence, time_limit=t_alloc, seed=500 + i * 11, verbose=False)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Grad cfg{i}: score={score:.8f}")

    # Gradient from perturbed starts
    for i in range(2):
        if time.monotonic() > t_p3 + p3_time - 10:
            break
        noise = rng.standard_normal(n_best) * 0.003
        a_pert = project_simplex(best_sequence + noise)
        t_alloc = min(time_per_grad / 2, t_p3 + p3_time - time.monotonic() - 10)
        if t_alloc < 5:
            break
        a_opt, obj_opt = gradient_finetune(a_pert, time_limit=t_alloc, seed=800 + i)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Grad pert{i}: score={score:.8f}")

    print(f"Phase 3 done. Best: {best_score:.8f}")

    # ========== Phase 4: LP + Gradient Alternation (200s) ==========
    print("=== Phase 4: Hybrid Alternation ===")
    t_p4 = time.monotonic()
    p4_time = 200.0

    n_best = len(best_sequence)
    a = best_sequence.copy()
    v = np.zeros(n_best)
    best_obj_p4 = 2.0 * n_best * fast_convolve(a).max()
    best_a_p4 = a.copy()
    no_improve_p4 = 0

    while time.monotonic() < t_p4 + p4_time:
        # LP step
        conv = fast_convolve(a)
        M = conv.max()
        b = solve_lp_full(a, M)
        if b is not None:
            b_sum = b.sum()
            if b_sum > 1e-15:
                b = b / b_sum
                # Small step for LP
                t_lp = 0.3
                a_new = (1 - t_lp) * a + t_lp * b
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                obj_new = obj_normalized(a_new)
                if obj_new < best_obj_p4:
                    best_obj_p4 = obj_new
                    best_a_p4 = a_new.copy()
                    a = a_new
                    no_improve_p4 = 0
                else:
                    no_improve_p4 += 1

        # Gradient step
        conv = fast_convolve(a)
        grad = compute_subgradient(a, conv, beta=800)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            v = 0.9 * v - 0.002 * grad / gnorm
            a_new = project_simplex(a + v)
            obj_new = obj_normalized(a_new)
            if obj_new < best_obj_p4:
                best_obj_p4 = obj_new
                best_a_p4 = a_new.copy()
                a = a_new
            else:
                no_improve_p4 += 1

        if no_improve_p4 > 40:
            noise = rng.standard_normal(n_best) * 0.004
            a = project_simplex(best_a_p4 + noise)
            v = np.zeros(n_best)
            no_improve_p4 = 0

    score = float(evaluate_sequence(best_a_p4.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_sequence = best_a_p4.copy()
        print(f"  Phase4: score={score:.8f}")

    print(f"Phase 4 done. Best: {best_score:.8f}")

    # ========== Phase 5: Final Polish (remaining time) ==========
    print("=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining time: {remaining:.1f}s")

    if remaining > 30 and best_sequence is not None:
        n_final = len(best_sequence)

        # Very small LP steps
        t_alloc = min(remaining * 0.4, 80.0)
        if t_alloc > 10:
            a_opt, obj_opt = lp_optimize_loop(
                best_sequence, time_limit=t_alloc,
                seed=9999, use_full_lp=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish LP: score={score:.8f}")

        # Tiny random perturbations
        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 10:
            num_tries = int(min(remaining2 / 3, 30))
            for i in range(num_tries):
                if time.monotonic() > deadline - 15:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_final) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Perturb[{i}] s={scale}: score={score_pert:.8f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128.0

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128.0
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
