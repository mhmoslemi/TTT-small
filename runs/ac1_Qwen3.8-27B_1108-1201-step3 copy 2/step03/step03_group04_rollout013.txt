"""
Hybrid optimization for minimizing 2*n*max(autocorrelation)/(sum)^2.
Uses: (1) fast subgradient with FFT, (2) reduced-constraint LP Frank-Wolfe,
(3) multi-length scanning, (4) coordinate refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_reduced_conv_matrix(a, conv, M, tol=0.05):
    """Build sparse matrix for only constraints near M."""
    n = len(a)
    # Include constraints where conv[k] > M*(1-tol)
    active = np.where(conv > M * (1 - tol))[0]
    
    rows = []
    cols = []
    vals = []
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)), active


def lp_oracle_reduced(a, M, tol=0.05):
    """Solve reduced LP: max sum(b) s.t. conv(a,b)[k] <= M for active k, b >= 0."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Find active constraints
    threshold = M * (1 - tol)
    active = np.where(conv > threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Build matrix for active constraints only
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 3.0})
    if res.success:
        b = np.maximum(res.x, 0.0)
        # Verify full constraints
        conv_check = fast_convolve(a, b)
        if conv_check.max() > M * (1 + 1e-6):
            # Solution violates non-active constraints; add them and re-solve
            # For now, just project
            violation = conv_check.max() - M
            if violation > 0:
                b = b * (M / conv_check.max())  # Scale down
        return b
    return None


def subgradient_descent(a0, time_limit, lr_init=0.01, momentum=0.9, verbose=False):
    """Fast subgradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient: gradient of max_k q_k(a) w.r.t. a
        # Use soft-max weighting over top peaks
        top_k = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        
        # Softmax weighting
        temp = max(peak_vals.max() * 0.1, 1e-15)
        w = np.exp(peak_vals / temp)
        w /= w.sum()
        
        # Subgradient of max is in convex hull of gradients of active q_k
        # q_k(a) = sum_i a[i]*a[k-i], gradient w.r.t. a[j] = a[k-j] + a[j-k] (with bounds)
        g = np.zeros(n)
        for idx, k in enumerate(peaks):
            if w[idx] < 1e-10:
                continue
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx_arr = np.arange(j_start, j_end + 1)
            # gradient of q_k w.r.t. a[j] for j in [j_start, j_end] is a[k-j]
            # But also need to account for the symmetric term
            # Actually q_k = sum_j a[j]*a[k-j] for valid j
            # d(q_k)/d(a[j]) = a[k-j] (for j where k-j is in range)
            # But by symmetry of autocorrelation, q_k = q_{-k}, so the full gradient
            # for the max over all lags includes both sides.
            # For linear convolution of a with a:
            # conv[k] = sum_j a[j]*a[k-j] (with a out of range = 0)
            # d(conv[k])/d(a[j]) = a[k-j]
            g[idx_arr] += w[idx] * 2.0 * a[k - idx_arr]
        
        # Update with momentum
        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        lr *= 0.999
        if no_improve > max_stall:
            # Add noise to escape
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            no_improve = 0
            max_stall = int(max_stall * 1.5)
            lr = max(lr * 0.8, 1e-6)
        
        it += 1
        if verbose and it % 500 == 0:
            print(f"    SG it={it}, obj={best_obj:.6f}, lr={lr:.2e}")
    
    return best_a, best_obj


def fw_reduced(a0, time_limit, verbose=False):
    """Frank-Wolfe with reduced LP oracle."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit:
        b = lp_oracle_reduced(a, M, tol=0.1)
        if b is None:
            # Fallback: full LP
            b = lp_oracle_full(a, M)
            if b is None:
                break
        
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            # Line search
            best_t = 0.0
            best_local = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidate = (1 - t) * a + t * b
                conv_c = fast_convolve(candidate)
                local_obj = 2.0 * n * conv_c.max()
                if local_obj < best_local:
                    best_local = local_obj
                    best_t = t
            
            if best_t > 0.01:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve >= max_stall:
            rng = np.random.default_rng(it + 42)
            noise = rng.standard_normal(n) * 0.005
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it}, obj={best_obj:.6f}, M={M:.4e}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def lp_oracle_full(a, M, time_limit=5.0):
    """Full LP oracle."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    
    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri.copy()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()
    
    # 1/sqrt edge
    edge_dist = np.minimum(x + 1, n - x)
    edge = 1.0 / np.sqrt(edge_dist)
    edge /= edge.sum()
    inits['inv_sqrt'] = edge.copy()
    
    # Cosine bell
    c = np.cos(np.pi * (x / n - 0.5)) ** 2
    c /= c.sum()
    inits['cosine'] = c.copy()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Spike at center
    spike = np.zeros(n)
    spike[n // 2] = 1.0
    inits['spike'] = spike.copy()
    
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(pp) == n:
                inits['parent'] = pp / ps
            else:
                inits['parent_interp'] = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(pp)),
                    pp) / max(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp).sum(), 1e-15)
    
    return inits

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_local_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def subgradient_descent(a0, time_limit, lr_init=0.01, momentum=0.9,
                        top_k=20, verbose=False):
    """Fast subgradient descent on the simplex using FFT."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit:
        top_k_eff = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k_eff)[-top_k_eff:]
        peak_vals = conv[peaks]

        temp = max(peak_vals.max() * 0.08, 1e-15)
        w = np.exp(peak_vals / temp)
        w /= w.sum()

        g = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx_arr = np.arange(j_start, j_end + 1)
            g[idx_arr] += w[idx] * 2.0 * a[k - idx_arr]

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9992
        if no_improve > max_stall:
            rng = np.random.default_rng(it + 7)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(a + noise, 0.0)
            a = a / max(a.sum(), 1e-15)
            conv = fast_convolve(a)
            no_improve = 0
            max_stall = int(max_stall * 1.4)
            lr = max(lr * 0.7, 1e-7)

        it += 1
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG it={it}, obj={best_obj:.6f}, lr={lr:.2e}, t={elapsed:.1f}s")

    return best_a, best_obj


def build_active_conv_matrix(a, conv, M, tol):
    """Build sparse matrix for constraints where conv[k] > M*(1-tol)."""
    n = len(a)
    threshold = M * (1.0 - tol)
    active = np.where(conv > threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    return A, active


def lp_oracle_reduced(a, M, tol=0.15, time_limit=4.0):
    """Solve reduced LP: max sum(b) s.t. conv(a,b)[k] <= M for active k."""
    n = len(a)
    conv = fast_convolve(a)
    A, active = build_active_conv_matrix(a, conv, M, tol)

    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        b = np.maximum(res.x, 0.0)
        # Verify against all constraints
        conv_check = fast_convolve(a, b)
        max_viol = conv_check.max()
        if max_viol > M * (1.0 + 1e-8):
            # Scale to satisfy constraints
            b = b * (M / max_viol)
        return b
    return None


def lp_oracle_full(a, M, time_limit=5.0):
    """Full LP oracle with all constraints."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def fw_reduced(a0, time_limit, tol=0.15, verbose=False):
    """Frank-Wolfe with reduced LP oracle and line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 40
    step_decay = 1.0

    while time.monotonic() - t0 < time_limit:
        b = lp_oracle_reduced(a, M, tol=tol, time_limit=3.0)
        if b is None:
            b = lp_oracle_full(a, M, time_limit=4.0)
            if b is None:
                break

        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb

        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            best_t = 0.0
            best_local = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidate = (1.0 - t) * a + t * b
                conv_c = fast_convolve(candidate)
                local_obj = 2.0 * n * conv_c.max()
                if local_obj < best_local:
                    best_local = local_obj
                    best_t = t

            if best_t > 0.01:
                a = (1.0 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve >= max_stall:
            rng = np.random.default_rng(it + 99)
            noise = rng.standard_normal(n) * 0.004 * step_decay
            a = np.maximum(a + noise, 0.0)
            a = a / max(a.sum(), 1e-15)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
            step_decay = max(0.3, step_decay * 0.8)

        it += 1
        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FW it={it}, obj={best_obj:.6f}, M={M:.4e}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_refine(a0, time_limit, verbose=False):
    """Coordinate-wise descent: optimize one element at a time."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    a = a / max(a.sum(), 1e-15)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(456)

    while time.monotonic() - t0 < time_limit:
        i = int(rng.integers(0, n))
        a_old = a[i]
        s_others = a.sum() - a_old

        best_val = a_old
        best_local = 2.0 * n * conv.max()

        candidates = [0.0, a_old * 0.3, a_old * 0.7, a_old * 1.3,
                      a_old * 1.7, a_old * 2.5]
        mean_val = a.sum() / n
        candidates.extend([mean_val, mean_val * 0.5, mean_val * 2.0])

        for val in candidates:
            if val < -1e-12:
                continue
            val = max(0.0, val)
            a[i] = val
            new_sum = s_others + val
            if new_sum < 1e-15:
                a[i] = a_old
                continue
            conv_new = fast_convolve(a)
            local = 2.0 * n * conv_new.max() / (new_sum ** 2)
            # Convert to comparable scale (sum=1 normalization)
            local_norm = 2.0 * n * conv_new.max() / (new_sum ** 2)
            if local_norm < best_local / (a.sum() ** 2):
                best_local = local_norm * (a.sum() ** 2)
                best_val = val

        a[i] = best_val
        a = a / max(a.sum(), 1e-15)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Coord it={it}, obj={best_obj:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations with emphasis on U-shapes."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)

    # U-shape: inverse distance from center
    dist = np.abs(x - center)
    u1 = 1.0 / (1.0 + dist * 2.0 / n)
    u1[v1 := u1] = None  # no-op, just for clarity
    inits['u_shape'] = u1 / u1.sum()

    # Sharper U
    u2 = 1.0 / (1.0 + (dist / n) ** 2)
    inits['u_sharp'] = u2 / u2.sum()

    # Inverse square root at edges
    edge_dist = np.minimum(x + 1, n - x)
    inv_sqrt = 1.0 / np.sqrt(edge_dist)
    inits['inv_sqrt'] = inv_sqrt / inv_sqrt.sum()

    # Power-law edge concentration
    edge_d = np.minimum(x + 1, n - x)
    power = edge_d ** (-0.3)
    inits['edge_power'] = power / power.sum()

    # Cosine bell (inverted U)
    cos_bell = 1.0 / (0.5 + 0.5 * np.cos(2.0 * np.pi * x / n))
    inits['cos_bell'] = cos_bell / cos_bell.sum()

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian center
    g = np.exp(-0.5 * ((x - center) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Alternating (helps flatten convolution)
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Pair symmetric U
    pair = np.minimum(x + 1, n - x) ** (-1.0)
    inits['pair_inv'] = pair / pair.sum()

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(pp) == n:
                inits['parent'] = pp / ps
            else:
                interp = np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(pp)), pp)
                inits['parent_interp'] = interp / max(interp.sum(), 1e-15)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid optimizer:
    Phase 1: Fast subgradient scan over lengths and U-shaped inits.
    Phase 2: LP-FW with reduced constraints on best candidate.
    Phase 3: Subgradient polish.
    Phase 4: More LP-FW with tighter reduction.
    Phase 5: Coordinate refinement + final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # === Phase 1: Fast subgradient scan ===
    print("=== Phase 1: Subgradient Scan ===")
    if parent is not None:
        n_parent = len(parent)
        lengths = sorted(set([n_parent, max(128, n_parent // 2),
                              min(1024, n_parent * 2), 256, 512]))
    else:
        lengths = [128, 256, 384, 512, 768]

    lengths = [l for l in lengths if 64 <= l <= 2048]

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_interp', 'u_shape', 'u_sharp', 'inv_sqrt',
                    'edge_power', 'cos_bell', 'alternating', 'pair_inv',
                    'uniform', 'gauss', 'random']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        # Allocate time per init
        n_inits = min(5, len(init_names))
        t_per_init = 25.0

        for name in init_names[:n_inits]:
            if time.monotonic() > deadline - 380:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 360)
            if t_alloc < 5:
                break

            a0 = inits[name]
            print(f"  SG: n={n}, init={name}, t={t_alloc:.0f}s")
            a_opt, obj_opt = subgradient_descent(a0, time_limit=t_alloc,
                                                 lr_init=0.008, momentum=0.91)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"    score={score:.6f}, local={obj_opt:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    print(f"After Phase 1: best={best_score:.6f}")

    # === Phase 2: LP-FW on best ===
    if best_sequence is not None:
        print("=== Phase 2: LP-FW ===")
        remaining = deadline - time.monotonic() - 200.0
        t_fw = remaining * 0.5
        if t_fw > 15:
            a_opt, obj_opt = fw_reduced(best_sequence, time_limit=t_fw,
                                        tol=0.15, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: Subgradient polish ===
    if best_sequence is not None:
        print("=== Phase 3: SG Polish ===")
        remaining = deadline - time.monotonic() - 150.0
        if remaining > 15:
            t_sg = remaining * 0.4
            a_opt, obj_opt = subgradient_descent(best_sequence, time_limit=t_sg,
                                                 lr_init=0.002, momentum=0.94,
                                                 top_k=30)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: LP-FW with tighter reduction ===
    if best_sequence is not None:
        print("=== Phase 4: LP-FW Tight ===")
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 10:
            a_opt, obj_opt = fw_reduced(best_sequence, time_limit=remaining,
                                        tol=0.08, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 5: Coordinate refinement ===
    if best_sequence is not None:
        print("=== Phase 5: Coordinate Refinement ===")
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 10:
            a_opt, obj_opt = coordinate_refine(best_sequence, time_limit=remaining)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 5 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 6: Final brief LP-FW ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_opt, obj_opt = fw_reduced(best_sequence, time_limit=remaining, tol=0.05)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 6 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # Fallback
    if best_sequence is None:
        print("WARNING: No valid sequence found, using fallback")
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
