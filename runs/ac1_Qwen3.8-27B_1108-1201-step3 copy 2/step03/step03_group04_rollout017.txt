def fast_convolve(a, b=None):

    A = np.fft.rfft(a, fft_size)

def build_conv_matrix(a):
    n = len(a)
    rows = []; cols = []; vals = []
    for k in range(2*n-1):

        j_end = min(n-1, k)

    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

**Key idea: "Convolution-aware" coordinate descent**

Instead of using a global subgradient, what if we do coordinate descent where at each step, we pick the coordinate that, if changed, would most reduce the peak?

For coordinate i, the effect on conv[k] is:
- Increasing a[i] by δ increases conv[k] by δ * a[k-i] (for each valid k-i)
- The peak is at k*, so increasing a[i] increases conv[k*] by δ * a[k*-i]
- To reduce conv[k*], we should DECREASE a[i] where a[k*-i] is large

So the "best" coordinate to decrease is the one where a[i] * a[k*-i] is largest (contributes most to the peak).

This gives a natural coordinate descent:
1. Find the peak position k*
2. Find the coordinate i that most contributes to the peak: argmax_i a[i] * a[k*-i]
3. Decrease a[i] and redistribute the mass to coordinates that don't contribute to the peak

This is a very targeted approach! Let me develop it:

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) structured initializations, (2) fast multi-peak subgradient
with adaptive temperature and Nesterov momentum, (3) LP-based Frank-Wolfe
for precision, (4) peak-targeted coordinate swaps, and (5) perturbation
restarts for escaping local minima.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        jj = np.arange(j_start, j_end + 1)
        rows.extend(k * len(jj))
        cols.extend(jj.tolist())
        vals.extend(a[k - jj].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=4.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        x = np.maximum(res.x, 0.0)
        s = x.sum()
        if s > 1e-12:
            return x / s
    return None


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def subgradient_optimize(a0, time_limit, n_peaks=12, tau_init=0.05,
                          momentum=0.92, lr_init=0.003, verbose=False):
    """Enhanced subgradient with adaptive temperature, momentum, and restarts."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    tau = tau_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit:
        # Adaptive temperature: cool down
        tau = max(0.005, tau_init * (0.9997 ** it))

        # Compute subgradient using soft-max over top peaks
        if it % 5 == 0 or conv is None:
            conv = fast_convolve(a)
            cur_peak = conv.max()
            cur_obj = 2.0 * n * cur_peak
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0

                no_improve += 1

        if s < 1e-15:

            a_new = a_new / s
        a = a_new

def coord_swap_optimize(a0, time_limit):

        k_star = int(np.argmax(conv))
        peak = conv[k_star]

        # Contribution of each coordinate to the peak
        i_vals = np.arange(n)
        j_vals = k_star - i_vals
        valid = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros(n)
        iv = i_vals[valid]
        contrib[iv] = a[iv] * a[j_vals[valid]]

        # Also consider second peak
        sorted_conv = np.sort(conv)[::-1]
        if len(sorted_conv) > 1 and sorted_conv[1] > 0.8 * sorted_conv[0]:
            # Find second peak (different position)
            mask = conv >= 0.95 * conv.max()
            peak_positions = np.where(mask)[0]
            if len(peak_positions) > 1:
                k2 = int(peak_positions[(peak_positions != k_star) & (peak_positions >= 0)][0])
                j2 = k2 - i_vals
                valid2 = (j2 >= 0) & (j2 < n)
                iv2 = i_vals[valid2]
                contrib[iv2] += a[iv2] * a[j2[valid2]]

        # Find top contributor and a good target
        top_idx = int(np.argmax(contrib))
        if contrib[top_idx] < 1e-15:
            break

        # Target: coordinate with low contribution that also has low value
        target_score = contrib + 0.1 * a  # prefer low contribution AND low current value

        target_score[top_idx] = np.inf
        bottom_idx = int(np.argmin(target_score))

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    x = np.arange(n, dtype=np.float64)

    # Edge-concentrated (various widths)
    for w_frac, name in [(0.2, 'edge_narrow'), (0.35, 'edge_med'), (0.5, 'edge_wide')]:
        edge = max(1, int(n * w_frac))
        ec = np.zeros(n)
        ec[:edge] = 1.0
        ec[-edge:] = 1.0
        ec /= ec.sum()
        inits[name] = ec

    # Two-sided exponential decay from center
    center = 0.5 * (n - 1)
    for alpha, name in [(0.02, 'exp_slow'), (0.05, 'exp_med'), (0.1, 'exp_fast')]:
        e = np.exp(-alpha * np.abs(x - center))
        e /= e.sum()
        inits[name] = e

    # Gaussian
    g = np.exp(-0.5 * ((x - center) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Triangle (linear ramp up then down)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Random sparse
    sparse = rng.random(n)
    sparse[sparse < 0.4] = 0.0
    if sparse.sum() > 1e-10:
        sparse /= sparse.sum()
    else:
        sparse = rng.dirichlet(np.ones(n))
    inits['sparse'] = sparse

    # Sine-weighted
    sine = np.sin(np.pi * (x + 0.5) / n) ** 2
    sine /= sine.sum()
    inits['sine'] = sine

    # Cosine-tapered (Hann window-like)
    hann = 0.5 * (1 + np.cos(2 * np.pi * (x + 0.5) / n))
    hann = np.maximum(hann, 0.01)
    hann /= hann.sum()
    inits['hann'] = hann

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                pp_interp = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(parent)), pp)
                ppi = pp_interp / pp_interp.sum()
                inits['parent'] = ppi

    return inits

        pp = np.maximum(parent, 0.0)
        if len(pp) != n_best:
            pp = np.interp(np.linspace(0, 1, n_best), np.linspace(0, 1, len(pp)), pp)
        pp = pp / pp.sum()
        candidates.append(pp)

    # Deduplicate
    unique_candidates = [candidates[0]]
    for c in candidates[1:]:
        if len(c) != len(unique_candidates[0]):
            continue
        if np.abs(c / c.sum() - unique_candidates[0] / unique_candidates[0].sum()).max() > 0.05:
            unique_candidates.append(c)

    phase2_end = time.monotonic() + 500.0
    t_per_candidate = max(30.0, 500.0 / max(1, len(unique_candidates)))

    for ci, cand in enumerate(unique_candidates[:2]):
        if time.monotonic() > phase2_end:
            break
        n = len(cand)
        print(f"  Candidate {ci}: n={n}")

        # Subgradient (long)
        t_sg = min(t_per_candidate * 0.5, phase2_end - time.monotonic() - 60)
        if t_sg > 10:
            a_opt, obj = subgradient_optimize(cand, time_limit=t_sg, n_peaks=15,
                                              tau_init=0.04, momentum=0.93, lr_init=0.004)
            try_update(a_opt, f"cand{ci},SG")

        # LP-FW
        t_lp = min(t_per_candidate * 0.35, phase2_end - time.monotonic() - 30)
        if t_lp > 15 and best_sequence is not None:
            a_opt, obj = lp_fw_optimize(best_sequence, time_limit=t_lp)
            try_update(a_opt, f"cand{ci},LPFW")

        # Coord swap
        t_cs = min(t_per_candidate * 0.25, phase2_end - time.monotonic() - 20)
        if t_cs > 10 and best_sequence is not None:
            a_opt, obj = coord_swap_optimize(best_sequence, time_limit=t_cs)
            try_update(a_opt, f"cand{ci},CS")

    print(f"After Phase 2: best={best_score:.6f}")

    # === Phase 3: Fine polish ===
    print("=== Phase 3: Fine Polish ===")
    if best_sequence is not None:
        phase3_end = time.monotonic() + 200.0
        n_best = len(best_sequence)

        # Very fine subgradient
        t = min(100.0, phase3_end - time.monotonic() - 60)
        if t > 10:
            a_opt, obj = subgradient_optimize(best_sequence, time_limit=t,
                                              n_peaks=20, tau_init=0.02,
                                              momentum=0.94, lr_init=0.001)
            try_update(a_opt, "polish_SG")

        # LP-FW precision
        t = min(80.0, phase3_end - time.monotonic() - 30)
        if t > 15:
            a_opt, obj = lp_fw_optimize(best_sequence, time_limit=t)
            try_update(a_opt, "polish_LPFW")

        # Coord swap final
        t = min(60.0, phase3_end - time.monotonic() - 20)
        if t > 10:
            a_opt, obj = coord_swap_optimize(best_sequence, time_limit=t)
            try_update(a_opt, "polish_CS")

    print(f"After Phase 3: best={best_score:.6f}")

    # === Phase 4: Diversification with perturbation restarts ===
    print("=== Phase 4: Perturbation Restarts ===")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        n_best = len(best_sequence)
        n_restarts = max(1, int((deadline - time.monotonic() - 15) // 20))
        n_restarts = min(n_restarts, 6)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            scale = 0.002 * (1 + 0.3 * i)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()

            t_alloc = min(18.0, deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break

            a_opt, obj = subgradient_optimize(a_pert, time_limit=t_alloc,
                                              n_peaks=15, tau_init=0.03,
                                              momentum=0.92, lr_init=0.002)
            try_update(a_opt, f"restart_{i}")

    print(f"After Phase 4: best={best_score:.6f}")

    # === Phase 5: Final LP pass if time remains ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 8
        if remaining > 15:
            print("=== Phase 5: Final LP ===")
            a_opt, obj = lp_fw_optimize(best_sequence, time_limit=remaining, verbose=True, verbose_every=100)
            try_update(a_opt, "final_LP")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
