- If all subgradients are negative, increase x

def solve_1d_convex(c, d, c_quad, has_quad, lo=0.0, hi=None):
    """
    Minimize g(x) = max(max_k (c[k] + d[k]*x), c_quad + x^2 [if has_quad]) for x >= lo.
    c, d: arrays for the linear terms.
    c_quad, has_quad: the quadratic term (if present).
    Returns optimal x.
    """
    if hi is None:
        # Find an upper bound
        hi = lo
        if hi == 0:
            hi = 1.0
        while True:
            # Check if g is still decreasing at hi
            val_quad = c_quad + hi*hi if has_quad else -np.inf
            val_lin = np.max(c + d * hi) if len(c) > 0 else -np.inf
            # Subgradient at hi
            active_lin = d[np.isclose(c + d*hi, val_lin, atol=1e-10)] if len(c) > 0 else np.array([])
            sg_min = min(np.min(active_lin) if len(active_lin) > 0 else 0, 2*hi if has_quad and val_quad >= val_lin else np.inf)
            if sg_min >= -1e-10:
                break
            hi *= 2
            if hi > 1e10:
                break
    
    # Bisection
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        val_quad = c_quad + mid*mid if has_quad else -np.inf
        val_lin = np.max(c + d * mid) if len(c) > 0 else -np.inf
        g_mid = max(val_quad, val_lin)
        
        # Subgradient at mid
        sgs = []
        if len(c) > 0:
            active = np.isclose(c + d*mid, val_lin, atol=1e-8)
            if np.max(c + d*mid) >= val_quad:
                sgs.extend(d[active].tolist())
        if has_quad and val_quad >= val_lin:
            sgs.append(2 * mid)
        
        if not sgs:
            sgs = [0]
        
        if min(sgs) >= 0:
            hi = mid
        elif max(sgs) <= 0:
            lo = mid
        else:
            # 0 is in the subgradient, we found the optimum
            return mid
    
    return 0.5 * (lo + hi)

Minimizes 2n*max(conv(a,a))/(sum(a))^2 using exact pairwise coordinate descent
on the simplex, combined with smoothed gradient descent for global exploration.
Key innovation: exact 1D optimization for each pairwise transfer, exploiting
the linearity (up to one quadratic term) of convolution changes.

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_logsumexp(obj, beta):
    """Smoothed version of max using log-sum-exp."""
    m = obj.max()
    if m < -1e300:
        return 0.0
    return m + np.log(np.sum(np.exp(obj - m))) / beta


def grad_logsumexp_conv(a, conv, beta):
    """
    Gradient of logsumexp(beta * conv(a,a)) / beta w.r.t. a.
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    
    # grad_i = 2 * sum_{k} p_k * a[k-i] (with appropriate bounds)
    # This is 2 * (p * a) cross-correlation
    # conv(p, a) at appropriate lags
    
    # Actually: d/d a[i] of conv_k = sum_j a[j]*a[k-j]
    # = a[k-i] + a[i-k] (with a[m]=0 for m out of [0,n-1])
    # So grad_i = 2 * sum_k p_k * (a[k-i] + a[i-k])
    # = 2 * (p * a)[i] (cross-correlation)
    
    # cross-correlation: (p ⊛ a)[i] = sum_k p[k] * a[k-i]
    # = sum_k p[k] * a[(k-i)] where a is zero-padded
    
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation p with a = ifft(conj(P) * A)
    cross = np.fft.irfft(np.conj(P) * A, fft_size)
    
    return 2.0 * cross[:n]

                               lr_init=0.01, momentum=0.9, seed=0):
