def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective_from_conv(conv, n):
    """Compute objective from convolution (assumes sum(a)=1)."""
    return 2.0 * n * conv.max()

def sa_update_conv(conv, a, i, j, d, n):
    """
    Compute the change in convolution when a[i] -= d, a[j] += d.
    Returns the delta vector (length 2n-1).
    Exact formula:
    delta[k] = 2d*(a[k-j] - a[k-i]) + d^2*([k=2i] - 2[k=i+j] + [k=2j])

    delta = np.zeros(2 * n - 1)
    # Linear term: 2d * (a[k-j] - a[k-i])
    # a[k-j] for k in [j, j+n-1] gives a[0], a[1], ..., a[n-1]
    # a[k-i] for k in [i, i+n-1] gives a[0], a[1], ..., a[n-1]
    end_j = min(j + n, 2 * n - 1)
    end_i = min(i + n, 2 * n - 1)
    delta[j:end_j] += 2.0 * d * a[:end_j - j]
    delta[i:end_i] -= 2.0 * d * a[:end_i - i]
    # Quadratic term

    if i + j < 2 * n - 1:

    if 2 * j < 2 * n - 1:

    return delta

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using simulated annealing with exact
O(n) incremental convolution updates, pressure-guided move proposals, LP warm
start, and local search polish. Key innovation: exact per-step convolution
updates avoid FFT overhead, enabling millions of SA steps; guided proposals
use gradient-based pressure to target peak-reducing moves.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Build constraint matrix for LP: rows=k conv positions, cols=j variables."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = 1.0
    return A


def solve_lp_direction(f, M, A_base, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    # (f*g)_k = sum_j f[k-j] * g_j
    # So A_ub[k, j] = f[k-j] if valid, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_limit, blend_init=0.4, blend_min=0.01):
    """LP-based directional optimization."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        A_ub = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                A_ub[k, j_start:j_end+1] = f_init[k - j_start:k - j_end + 1][::-1] * a[0:1]  # placeholder
        # Actually let's build it properly each time since a changes
        A_ub = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):

def simulated_annealing_guided(a_init, time_limit, T_init=3e-6, T_final=2e-9,
                                top_k_peaks=15, recompute_interval=2000):
    """
    Simulated annealing with exact O(n) convolution updates.
    Uses pressure-guided move proposals based on peak contributions.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    t0 = time.monotonic()
    elapsed_total = time_limit
    tau = elapsed_total / np.log(T_init / T_final) if T_final > 0 else elapsed_total / 5.0
    rng = np.random.default_rng(12345)

    delta_buf = np.zeros(2 * n - 1)
    step = 0
    accept_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        T = T_init * np.exp(-elapsed / tau)
        if T < T_final:
            T = T_final

        # Compute pressure vector: blame[i] = sum over peaks of weight_p * a[p-i]
        # This tells us which coordinates contribute most to the peaks
        peak_idx = np.argpartition(conv, -top_k_peaks)[-top_k_peaks:]
        peak_vals = conv[peak_idx]
        pmax_val = peak_vals.max()
        weights = np.exp(peak_vals / max(T * 0.1, 1e-10))
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.ones(len(peak_idx)) / len(peak_idx)
        else:
            weights /= w_sum

        # Compute pressure
        pressure = np.zeros(n)
        for idx in range(len(peak_idx)):
            k = int(peak_idx[idx])
            w = weights[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                pressure[j] += w * a[k - j]

        # Propose move: pick source from high pressure, target from low pressure
        # Use exponential sampling for source, uniform for target
        if rng.random() < 0.7:
            # Guided move
            src_weights = pressure + 1e-15
            src_weights /= src_weights.sum()
            # Sample source proportional to pressure (high pressure = likely source)
            i = rng.choice(n, p=src_weights)
            # Target: prefer low pressure
            tgt_weights = (1.0 / (pressure + 1e-10))
            tgt_weights[i] = 0
            tgt_sum = tgt_weights.sum()
            if tgt_sum < 1e-15:
                j = rng.integers(n)
                while j == i:
                    j = rng.integers(n)
            else:
                tgt_weights /= tgt_sum
                j = rng.choice(n, p=tgt_weights)
        else:
            # Random move
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)

        # Determine step size
        max_d = a[i] * 0.8
        if max_d < 1e-12:
            # Try swapping direction
            i, j = j, i
            max_d = a[i] * 0.8
            if max_d < 1e-12:
                T *= 0.999
                step += 1
                continue

        d = min(max_d, d_scale * max_d) if (d_scale := rng.uniform(0.1, 1.0)) <= 1.0 else max_d
        # Simpler: d = max_d * uniform(0.05, 1.0)
        d = max_d * rng.uniform(0.05, 1.0)

        # Compute exact delta in convolution
        # delta[k] = 2d*(a[k-j] - a[k-i]) + d^2*([k=2i] - 2[k=i+j] + [k=2j])
        delta_buf[:] = 0.0
        delta_buf[j:j+n] += 2.0 * d * a
        delta_buf[i:i+n] -= 2.0 * d * a
        dq = d * d
        delta_buf[2 * i] += dq
        delta_buf[i + j] -= 2.0 * dq
        delta_buf[2 * j] += dq

        new_max = (conv + delta_buf).max()
        delta_obj = new_max - cur_max

        # Accept/reject (Metropolis)
        if delta_obj <= 0 or (T > 1e-15 and rng.random() < np.exp(-delta_obj / T)):
            # Accept the move
            a[i] -= d
            a[j] += d
            conv += delta_buf
            cur_max = new_max
            accept_count += 1

            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        else:
            # Also try reverse direction (swap i and j)
            delta_buf[:] = 0.0
            delta_buf[i:i+n] += 2.0 * d * a
            delta_buf[j:j+n] -= 2.0 * d * a
            delta_buf[2 * j] += dq
            delta_buf[i + j] -= 2.0 * dq
            delta_buf[2 * i] += dq

            new_max_rev = (conv + delta_buf).max()
            delta_obj_rev = new_max_rev - cur_max

            if delta_obj_rev <= 0 or (T > 1e-15 and rng.random() < np.exp(-delta_obj_rev / T)):
                a[j] -= d
                a[i] += d
                conv += delta_buf
                cur_max = new_max_rev
                accept_count += 1
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()

        step += 1

        # Periodic recompute to prevent drift
        if step % recompute_interval == 0:
            conv = fast_convolve(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # Adaptive: if too many steps without improvement, perturb
        if step % 100000 == 0 and accept_count == 0:
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            cur_max = conv.max()
            accept_count = 0

    return best_a, 2.0 * n * best_max

def subgradient_polish(a_init, time_limit, lr_init=0.01, top_k=30):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_a = a.copy()
    best_max = conv.max()

    t0 = time.monotonic()
    lr = lr_init
    momentum = 0.9
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        # Soft weights for peaks
        scale = pmax * 0.05
        scaled = peak_vals / scale
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w /= w_sum

        # Subgradient: for each peak k, gradient of (a*a)_k w.r.t. a is 2*a_rev_k
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            wi = w[idx]
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += 2.0 * wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_max = conv.max()

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            a = project_simplex(a + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.001, lr * 0.6)

        lr *= 0.9998

    return best_a

    a_sa, obj_sa = simulated_annealing_guided(
        best_sequence, t_sa,
        T_init=5e-6, T_final=1e-9,
        top_k_peaks=15, recompute_interval=3000
    )
    score_sa = float(evaluate_sequence(a_sa.tolist()))
    if np.isfinite(score_sa) and score_sa < best_score:
        best_score = score_sa
        best_sequence = a_sa.copy()
        print(f"  SA: {score_sa:.6f}")

    # Phase 2b: Run SA from a slightly perturbed version (diversity)
    if time.monotonic() < deadline - 350:
        t_sa2 = min(300.0, deadline - time.monotonic() - 320.0)
        if t_sa2 > 30:
            noise = rng.standard_normal(len(best_sequence)) * 0.02
            a_pert = project_simplex(best_sequence + noise)
            a_sa2, obj_sa2 = simulated_annealing_guided(
                a_pert, t_sa2,
                T_init=3e-6, T_final=5e-10,
                top_k_peaks=12, recompute_interval=3000
            )
            score_sa2 = float(evaluate_sequence(a_sa2.tolist()))
            if np.isfinite(score_sa2) and score_sa2 < best_score:
                best_score = score_sa2
                best_sequence = a_sa2.copy()
                print(f"  SA2 (perturbed): {score_sa2:.6f}")

    # Phase 3: Local search polish
    print("=== Phase 3: Local Search ===")
    t_ls = min(150.0, deadline - time.monotonic() - 100.0)
    if t_ls > 10:
        a_ls = local_search_polish(best_sequence, t_ls, max_pairs=200)
        score_ls = float(evaluate_sequence(a_ls.tolist()))
        if np.isfinite(score_ls) and score_ls < best_score:
            best_score = score_ls
            best_sequence = a_ls.copy()
            print(f"  Local Search: {score_ls:.6f}")

    # Phase 4: Subgradient polish
    print("=== Phase 4: Subgradient Polish ===")
    t_sg = min(80.0, deadline - time.monotonic() - 50.0)
    if t_sg > 5:
        a_sg = subgradient_polish(best_sequence, t_sg, lr_init=0.008, top_k=40)
        score_sg = float(evaluate_sequence(a_sg.tolist()))
        if np.isfinite(score_sg) and score_sg < best_score:
            best_score = score_sg
            best_sequence = a_sg.copy()
            print(f"  Subgradient: {score_sg:.6f}")

    # Phase 5: Final SA run (short, from best)
    print("=== Phase 5: Final SA ===")
    t_final = max(10.0, deadline - time.monotonic() - 15.0)
    if t_final > 5:
        a_fin, _ = simulated_annealing_guided(
            best_sequence, t_final,
            T_init=1e-7, T_final=1e-10,
            top_k_peaks=10, recompute_interval=1000
        )
        score_fin = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(score_fin) and score_fin < best_score:
            best_score = score_fin
            best_sequence = a_fin.copy()
            print(f"  Final SA: {score_fin:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(n_primary) / n_primary
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
