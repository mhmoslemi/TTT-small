import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Matrix C where C @ g = a * g."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i+n, i] = a
    return C

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def line_search(a, d, n, max_obj):
    """Find best step size along direction d."""
    best_t = 0.0
    best_obj = max_obj
    for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        obj = 2.0 * n * fast_convolve(a_try).max() / (s*s)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj

def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(len(a))/len(a)
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        g = lp_direction(a, M * 1.001)
        if g is None:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        d = g / g.sum() - a
        if np.linalg.norm(d) < 1e-14:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        t_best, obj_best = line_search(a, d, n, best_obj)
        
        if obj_best < best_obj - 1e-14:
            a_new = a + t_best * d
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a = best_a.copy()
            else:
                a = a_new / s_new
            conv = fast_convolve(a)
            M = conv.max()
            best_obj = 2.0 * n * M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        it += 1
        if verbose and it % 200 == 0:
            print(f"    LP iter {it}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

            a = a / s if s > 1e-15 else best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()

    return best_a, best_obj
