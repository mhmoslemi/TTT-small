"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max smoothed gradient descent
with Adam optimizer, adaptive temperature annealing, FFT-based subgradients,
and multi-restart exploration. Key innovation: exploits the structure of the
subgradient (shifted-reversed copy of a) via FFT for fast per-iteration cost,
and uses a smooth approximation of the max function for stable convergence.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_subgradient(a, temp):
    """Compute smoothed subgradient of max(conv(a,a)) using softmax weighting via FFT."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # sg[i] = 2 * sum_k w[k] * a[k-i] = 2 * cross_corr(w, a)[i]
    # cross_corr(w, a)[i] = sum_j w[j+i] * a[j] = ifft(conj(fft(w)) * fft(a))[i]
    size = 1 << (2 * n).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(np.conj(W) * A, size)
    sg = 2.0 * corr[:n]
    return sg, conv


def adam_update(params, grad, state, lr, beta1, beta2, eps):
    """Adam optimizer step, modifying params in place."""
    t = state['t'] + 1
    state['t'] = t
    m = state['m']
    v = state['v']
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params -= lr * m_hat / (np.sqrt(v_hat) + eps)

    best_obj = conv.max()
    best_a = a.copy()

    temp = temp_init

    no_improve = 0

        adam_update(a, sg, state, lr, 0.9, 0.999, 1e-8)
        a = project_simplex(a)
        conv = fast_convolve(a)
        co = conv.max()
        if co < best_obj - 1e-12:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp = max(temp * 0.9999, 1e-8)
        lr *= 0.99995

        if no_improve > 200:
            # Restart from best with small perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            lr = 0.003
            no_improve = 0

    if verbose:
        print(f"    Phase 2 done: obj={best_obj:.10f}")

    # Phase 3: Very fine polishing
    temp = max(0.005 / n, 1e-8)
    lr = 0.001
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg, conv = soft_subgradient(a, temp)
        adam_update(a, sg, state, lr, 0.95, 0.999, 1e-8)
        a = project_simplex(a)
        conv = fast_convolve(a)
        co = conv.max()
        if co < best_obj - 1e-13:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        if no_improve > 300:
            break

    return best_a, best_obj


def multi_peak_gradient_optimize(a0, time_limit, verbose=False):
    """
    Gradient method using top-k peak weighting instead of full softmax.
    More focused on reducing the actual maximum.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    top_k = max(4, n // 16)
    temp = 0.02 / n
    lr = 0.008
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute top-k peak weighted gradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_peaks /= w_peaks.sum()

        # Build full weight vector (sparse)
        size = 1 << (2 * n).bit_length()
        W = np.zeros(size // 2 + 1, dtype=np.complex128)
        A = np.fft.rfft(a, size)

        # Compute weighted subgradient: sg[i] = 2 * sum_k w[k] * a[k-i]
        # Using the peaks only
        W_full = np.zeros(size)
        for i in range(len(peaks)):
            W_full[peaks[i]] = w_peaks[i]
        W_r = np.fft.rfft(W_full)
        corr = np.fft.irfft(np.conj(W_r) * A, size)
        sg = 2.0 * corr[:n]

        adam_update(a, sg, state, lr, 0.9, 0.999, 1e-8)
        a = project_simplex(a)
        conv = fast_convolve(a)
        co = conv.max()

        if co < best_obj - 1e-12:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp *= 1.001
        lr *= 0.99995

        if no_improve > 300:
            break

    return best_a, best_obj


def parametric_init(n, kind, rng):
    """Generate parametric initializations."""
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    if kind == 0:  # Uniform
        return np.ones(n) / n
    elif kind == 1:  # Gaussian center
        sigma = 0.3 * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        return g / g.sum()
    elif kind == 2:  # Edge concentrated (U-shaped)
        d = np.abs(x - mid) / n
        e = 1.0 / (d + 0.1)
        return e / e.sum()
    elif kind == 3:  # Asymmetric decay
        a = np.exp(-0.5 * x / (0.3 * n))
        return a / a.sum()
    elif kind == 4:  # Cosine squared
        c = np.sin(np.pi * (x + 0.5) / n) ** 2
        return c / c.sum()
    elif kind == 5:  # Asymmetric Gaussian
        sigma_l = 0.25 * n
        sigma_r = 0.4 * n
        g = np.where(x < mid,
                     np.exp(-0.5 * ((x - mid) / sigma_l) ** 2),
                     np.exp(-0.5 * ((x - mid) / sigma_r) ** 2))
        return g / g.sum()
    elif kind == 6:  # Random Dirichlet
        return rng.dirichlet(np.ones(n) * 0.3)
    elif kind == 7:  # Two-bump
        b1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
        b2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
        return (b1 + b2) / (b1 + b2).sum()
    elif kind == 8:  # Power law
        p = 0.7 + rng.random() * 0.6
        a = (x + 1) ** p
        return a / a.sum()
    else:  # Linear ramp
        a = x + 1
        return a / a.sum()


def try_evaluate(a, best_score_ref, label=""):
    """Evaluate and track best."""
    a_clip = np.clip(a, 0.0, 1000.0)
    score = float(evaluate_sequence(a_clip.tolist()))
    if np.isfinite(score) and score < best_score_ref[0]:
        best_score_ref[0] = score
        best_score_ref[1] = a_clip.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_clip)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search using soft-max smoothed gradient descent
    with Adam, multi-peak weighting, parametric diversification, and fine polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = [float('inf'), None]

    def is_deadline(remaining=5.0):
        return time.monotonic() > deadline - remaining

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Quick exploration with multiple inits ===
    print("=== Phase 1: Exploration ===")
    p1_start = time.monotonic()
    p1_budget = 250.0

    n_primary = len(parent) if parent is not None else 384

    # Evaluate parent
    if parent is not None:
        try_evaluate(parent, best_score, "parent_raw")

    # Quick optimization of parent
    if parent is not None and not is_deadline(50):
        t_alloc = min(100, p1_budget - (time.monotonic() - p1_start) - 30)
        if t_alloc > 15:
            print(f"  Optimizing parent (n={n_primary}), budget={t_alloc:.0f}s")
            a_opt, _ = soft_gradient_optimize_v2(parent, time_limit=t_alloc, verbose=True)
            try_evaluate(a_opt, best_score, "parent_opt")

    # Try parametric inits at same length
    inits_to_try = [1, 2, 3, 4, 5, 7]
    per_init_time = min(25, (p1_budget - (time.monotonic() - p1_start) - 20) / len(inits_to_try))
    for kind in inits_to_try:
        if is_deadline(30):
            break
        f_init = parametric_init(n_primary, kind, rng)
        if parent is not None:
            f_init = 0.5 * f_init + 0.5 * parent
            f_init = project_simplex(f_init)
        t_alloc = min(per_init_time, p1_budget - (time.monotonic() - p1_start) - 10)
        if t_alloc < 5:
            break
        a_opt, _ = soft_gradient_optimize(f_init, time_limit=t_alloc,
                                           temp_init=0.1 / n_primary,
                                           temp_min=1e-7, lr_init=0.01,
                                           beta1=0.9, beta2=0.999, eps=1e-8)
        try_evaluate(a_opt, best_score, f"init_{kind}")

    # Try a different length
    if not is_deadline(40):
        n_alt = max(128, n_primary // 2)
        f_init = parametric_init(n_alt, 4, rng)
        t_alloc = min(40, p1_budget - (time.monotonic() - p1_start) - 10)
        if t_alloc > 10:
            a_opt, _ = soft_gradient_optimize(f_init, time_limit=t_alloc,
                                               temp_init=0.1 / n_alt,
                                               temp_min=1e-7, lr_init=0.01,
                                               beta1=0.9, beta2=0.999, eps=1e-8)
            try_evaluate(a_opt, best_score, f"alt_n{n_alt}")

    print(f"  Phase 1 best: {best_score[0]:.8f}")

    # === Phase 2: Deep optimization on best ===
    print("\n=== Phase 2: Deep Optimization ===")
    p2_start = time.monotonic()
    p2_budget = 1000.0

    if best_score[1] is not None:
        a_best = best_score[1].copy()
        n_best = len(a_best)
        a_best = project_simplex(a_best)

        # Long optimization run
        rem = p2_budget - (time.monotonic() - p2_start) - 50
        if rem > 30:
            print(f"  Deep optimization (n={n_best}), budget={rem:.0f}s")
            a_opt, _ = soft_gradient_optimize_v2(a_best, time_limit=rem * 0.6, verbose=True)
            try_evaluate(a_opt, best_score, "deep_v2")

        # Multi-peak focused optimization
        rem = p2_budget - (time.monotonic() - p2_start) - 30
        if rem > 20 and best_score[1] is not None:
            a_cur = project_simplex(best_score[1])
            a_opt, _ = multi_peak_gradient_optimize(a_cur, time_limit=rem * 0.4, verbose=True)
            try_evaluate(a_opt, best_score, "multi_peak")

        # Perturbation + re-optimize
        for p in range(4):
            rem = p2_budget - (time.monotonic() - p2_start) - 20
            if rem < 20 or best_score[1] is None:
                break
            noise_scale = [0.003, 0.01, 0.03, 0.005][p]
            a_pert = project_simplex(best_score[1] + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(rem / 2, 80)
            a_opt, _ = soft_gradient_optimize(a_pert, time_limit=t_alloc,
                                               temp_init=0.05 / n_best,
                                               temp_min=1e-8, lr_init=0.005,
                                               beta1=0.9, beta2=0.999, eps=1e-8)
            try_evaluate(a_opt, best_score, f"perturb_{p}")

    print(f"  Phase 2 best: {best_score[0]:.8f}")

    # === Phase 3: Fine polishing ===
    print("\n=== Phase 3: Fine Polishing ===")
    p3_start = time.monotonic()
    p3_budget = 400.0

    if best_score[1] is not None:
        a_best = project_simplex(best_score[1])
        n_best = len(a_best)

        # Very fine soft-gradient
        rem = p3_budget - (time.monotonic() - p3_start) - 50
        if rem > 20:
            a_opt, _ = soft_gradient_optimize(a_best, time_limit=rem * 0.4,
                                               temp_init=0.01 / n_best,
                                               temp_min=1e-9, lr_init=0.002,
                                               beta1=0.95, beta2=0.999, eps=1e-9)
            try_evaluate(a_opt, best_score, "fine_sg")

        # Multi-peak with very low temp
        rem = p3_budget - (time.monotonic() - p3_start) - 30
        if rem > 15 and best_score[1] is not None:
            a_cur = project_simplex(best_score[1])
            a_opt, _ = multi_peak_gradient_optimize(a_cur, time_limit=rem * 0.3)
            try_evaluate(a_opt, best_score, "fine_mp")

        # Coordinate descent style: optimize small neighborhoods
        rem = p3_budget - (time.monotonic() - p3_start) - 20
        if rem > 15 and best_score[1] is not None:
            a_cur = best_score[1].copy()
            # Try small random perturbations and keep improvements
            for trial in range(200):
                if time.monotonic() > deadline - 60:
                    break
                idx = rng.integers(0, n_best, size=max(1, n_best // 32))
                a_trial = a_cur.copy()
                a_trial[idx] *= (1.0 + rng.standard_normal(len(idx)) * 0.02)
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    score = float(evaluate_sequence(a_trial.tolist()))
                    if np.isfinite(score) and score < best_score[0]:
                        best_score[0] = score
                        best_score[1] = a_trial.copy()
                        a_cur = a_trial.copy()
                        if trial % 50 == 0:
                            print(f"    Coord descent trial {trial}: {score:.8f}")

        # Final very fine run
        rem = deadline - time.monotonic() - 30
        if rem > 15 and best_score[1] is not None:
            a_cur = project_simplex(best_score[1])
            a_opt, _ = soft_gradient_optimize(a_cur, time_limit=rem * 0.5,
                                               temp_init=0.005 / n_best,
                                               temp_min=1e-10, lr_init=0.001,
                                               beta1=0.95, beta2=0.999, eps=1e-10)
            try_evaluate(a_opt, best_score, "final_fine")

    # Fallback
    if best_score[1] is None:
        best_score[1] = np.ones(384) / 384
        best_score[0] = float(evaluate_sequence(best_score[1].tolist()))

    # Final verification
    final_a = np.clip(best_score[1], 0.0, 1000.0)
    final_score = float(evaluate_sequence(final_a.tolist()))
    if np.isfinite(final_score):
        best_score[0] = final_score
        best_score[1] = final_a

    print(f"\nFINAL SCORE: {best_score[0]:.8f} (n={len(best_score[1])})")
    return [float(x) for x in best_score[1]]
