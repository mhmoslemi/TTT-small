def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(n):
    """Build the n x n lower-triangular constraint matrix for the LP.
    A_ub[k, j] = 1 if j <= k else 0 (the actual f values are applied via
    the matrix product with f broadcast).
    """
    # A_ub[k, j] = f[k-j] if j <= k, else 0
    # We precompute the index structure
    row_idx = []
    col_idx = []
    for k in range(n):
        for j in range(k + 1):
            row_idx.append(k)
            col_idx.append(j)
    row_idx = np.array(row_idx, dtype=np.int32)
    col_idx = np.array(col_idx, dtype=np.int32)
    return row_idx, col_idx


def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k=0,...,n-1, g >= 0.
    Uses the lower-triangular structure of the constraint matrix."""
    c = -np.ones(n)
    # Build A_ub as a dense matrix (for n up to ~1024 this is fine)

    bounds = [(0, None)] * n

                      options={'presolve': True})

    except Exception:
        pass
    return None

    f = np.maximum(f0.astype(np.float64), 0.0)

    best_obj = M

    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp(f, M * m_exp, n)

            continue

        if g_sum < 1e-15:

        g = g / g_sum

            if no_improve > 30:

            m_exp = min(m_exp * 1.08, 4.0)

        # Line search: coarse scan + bisection
        best_alpha = 0.0
        best_val = M

        coarse_alphas = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]

    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at top-k peaks with softmax weighting
        ck = len(conv)
        if top_k >= ck:
            peaks = np.arange(ck)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / max(cur_temp, 1e-12)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            # Subgradient of conv[k] w.r.t. a:
            # d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[2n-2-k-j] (if 0<=2n-2-k-j<n)
            js1 = max(0, k - n + 1)
            je1 = min(n - 1, k)
            if je1 >= js1:
                idx1 = np.arange(js1, je1 + 1)
                sg[idx1] += w[i] * f[k - idx1]

            k_sym = 2 * n - 2 - k
            js2 = max(0, k_sym - n + 1)
            je2 = min(n - 1, k_sym)
            if je2 >= js2:
                idx2 = np.arange(js2, je2 + 1)
                sg[idx2] += w[i] * f[k_sym - idx2]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.004

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0008)
            no_improve = 0

    return best_f, best_obj


def pair_swap_polish(f0, time_limit, max_pairs=2000):
    """Polish by trying pairwise mass transfers and keeping improvements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        # Pick a random pair
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if f[i] < 1e-10 and f[j] < 1e-10:
            continue

        # Try transferring mass from j to i (and vice versa)
        # a_i' = a_i + delta, a_j' = a_j - delta
        # delta range: (-a_j + eps, a_i_max) where a_i_max is bounded by sum
        # Actually: delta in (-f[j], f[i] + (1 - f[i] - f[j]))
        # Simplify: delta in (-f[j]*0.99, min(f[i], 1-f[j])*0.99) but also need f[i]+delta >= 0 and f[j]-delta >= 0
        # So delta in (-f[j], ...) and delta < ... and f[i]+delta <= 1 (always true since f[j]-delta >= 0 means delta <= f[j])

        lo = -f[j] + 1e-12
        hi = f[j] - 1e-12  # max delta is f[j] (transfer all of j to i), but also limited
        # Actually delta can be positive (j->i) up to f[j], or negative (i->j) down to -f[i]
        lo = max(lo, -f[i] + 1e-12)

        if hi - lo < 1e-12:
            continue

        # Ternary search on delta
        # The objective max_k conv_k(delta) is convex in delta (max of quadratics/linears)
        a1 = lo + (hi - lo) * 0.382
        a2 = lo + (hi - lo) * 0.618

        for _ in range(30):
            if hi - lo < 1e-14:
                break
            mid1 = lo + (hi - lo) * 0.382
            mid2 = lo + (hi - lo) * 0.618

            # Evaluate at mid1
            f_trial = f.copy()
            f_trial[i] += mid1
            f_trial[j] -= mid1
            f_trial = np.maximum(f_trial, 0.0)
            f_trial = f_trial / f_trial.sum()
            v1 = fast_convolve(f_trial).max()

            # Evaluate at mid2
            f_trial = f.copy()
            f_trial[i] += mid2
            f_trial[j] -= mid2
            f_trial = np.maximum(f_trial, 0.0)
            f_trial = f_trial / f_trial.sum()
            v2 = fast_convolve(f_trial).max()

            if v1 < v2:
                hi = mid2
            else:
                lo = mid1

        # Evaluate at the found optimum
        delta = (lo + hi) / 2.0
        f_trial = f.copy()
        f_trial[i] += delta
        f_trial[j] -= delta
        f_trial = np.maximum(f_trial, 0.0)
        st = f_trial.sum()
        if st < 1e-15:
            continue
        f_trial = f_trial / st
        val = fast_convolve(f_trial).max()

        if val < best_obj - 1e-14:
            best_obj = val
            best_f = f_trial.copy()
            f = f_trial.copy()
            stall = 0
        else:
            stall += 1

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = 0.5 * (n - 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated (heavy at edges)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Center-concentrated (heavy in middle)
    center_w = 1.0 / (np.abs(x - xm) + 1.0)
    inits.append(center_w / center_w.sum())

    # Sine window
    sine_w = np.sin(np.pi * (x + 1) / (n + 1))
    inits.append(sine_w / sine_w.sum())

    # Cosine power (p=2)
    cos_w = np.cos(np.pi * (x - xm) / n) ** 2
    cos_w = np.maximum(cos_w, 0)
    if cos_w.sum() > 1e-10:
        inits.append(cos_w / cos_w.sum())

    # Power law (heavy tail toward one end)
    pow_w = (x + 1) ** 0.5
    inits.append(pow_w / pow_w.sum())

    # Random Dirichlet (low concentration = sparse)
    inits.append(rng.dirichlet(np.ones(n) * 0.2))

    # Random Dirichlet (medium concentration)
    inits.append(rng.dirichlet(np.ones(n) * 1.0))

    # Two-peak structure
    peak1 = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (0.1 * n)) ** 2)
    peak2 = np.exp(-0.5 * ((x - 0.7 * (n - 1)) / (0.1 * n)) ** 2)
    two_peak = peak1 + peak2
    inits.append(two_peak / two_peak.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Blend parent with uniform
        parent_norm = px / ps
        blend = 0.7 * parent_norm + 0.3 * (np.ones(n) / n)
        inits.append(blend / blend.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search using LP-based Frank-Wolfe with
    halved constraints, diverse initializations, multi-length exploration,
    and final polishing via subgradient and pair-swap methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent available: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Quick multi-length exploration ===
    print("=== Phase 1: Multi-length Exploration ===")
    t_phase1_start = time.monotonic()

    # Determine which lengths to try
    lengths = [384, 256, 512, 320, 448]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(0, pn)
        else:
            lengths.remove(pn)
            lengths.insert(0, pn)

    # Limit to reasonable lengths based on available time
    phase1_budget = min(600, (deadline - time.monotonic()) * 0.35)
    per_length_budget = phase1_budget / len(lengths)

    for n in lengths:
        rem = deadline - time.monotonic() - (0.3 * (budget_s - 15))
        if rem < 30:
            break

        inits = generate_inits(n, parent, rng)
        # Limit number of initializations per length
        max_inits = min(len(inits), max(2, int(per_length_budget / 40)))
        inits = inits[:max_inits]

        t_per = min(per_length_budget / max(1, len(inits)), 80)

        for i, f_init in enumerate(inits):
            rem = deadline - time.monotonic() - (0.25 * (budget_s - 15))
            if rem < 15:
                break
            t_alloc = min(t_per, rem)
            if t_alloc < 5:
                break
            a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 1 done. Best: {best_score:.8f}")

    # === Phase 2: Deep optimization on best candidates ===
    print(f"\n=== Phase 2: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)
        phase2_budget = min(700, (deadline - time.monotonic()) * 0.45)

        # Run multiple FW optimizations with different perturbations
        perturbations = [
            ("direct", None),
            ("small_noise", 0.003),
            ("med_noise", 0.01),
            ("mirror_blend", "mirror"),
            ("uniform_blend", "uniform"),
        ]

        n_runs = len(perturbations)
        t_per_run = phase2_budget / n_runs

        for p_idx, (label, param) in enumerate(perturbations):
            rem = deadline - time.monotonic() - (0.15 * (budget_s - 15))
            if rem < 20:
                break
            t_alloc = min(t_per_run, rem)

            if param is None:
                f_init = best_seq.copy()
            elif param == "mirror":
                f_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                f_init = project_simplex(f_init)
            elif param == "uniform":
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                noise = rng.standard_normal(n_best) * param
                f_init = project_simplex(best_seq + noise)

            a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_{label}")

    # === Phase 3: Polishing ===
    print(f"\n=== Phase 3: Polishing ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Subgradient polish with various parameters
        sg_configs = [
            (80, 0.002, 0.94, max(10, n_best // 8), 0.00004),
            (60, 0.001, 0.95, max(10, n_best // 4), 0.00002),
            (40, 0.0005, 0.96, max(5, n_best // 6), 0.00001),
        ]

        for cfg_idx, (t_lim, lr, mom, tk, tmp) in enumerate(sg_configs):
            rem = deadline - time.monotonic() - 30
            if rem < 10:
                break
            a_opt, _ = sg_polish(best_seq, time_limit=min(t_lim, rem),
                                 lr=lr, mom=mom, top_k=tk, temp=tmp)
            try_seq(a_opt, f"sg{cfg_idx}")

        # Pair swap polish
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            a_opt, _ = pair_swap_polish(best_seq, time_limit=min(120, rem))
            try_seq(a_opt, "pair_swap")

        # Final short FW
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "fw_final")

        # Final SG with very fine parameters
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 40),
                                 lr=0.0005, mom=0.97, top_k=max(5, n_best // 5), temp=0.000005)
            try_seq(a_opt, "sg_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
