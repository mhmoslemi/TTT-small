Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

- Smooth L-BFGS-B with multi-beta annealing for polishing

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def grad_smooth_conv(a, w):
    """Gradient of weighted sum of conv terms via FFT cross-correlation."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def subgrad_at_lag(a, k):
    """Subgradient of conv[k] w.r.t. a: grad[i] = 2*a[k-i] if 0<=k-i<n."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        grad[i_start:i_end + 1] = 2.0 * a[k - i_start:k - i_end - 1:-1]
    return grad

    n = len(a)

    w = vals / vals.sum() if vals.sum() > 1e-15 else np.ones(top_k) / top_k
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        grad += wi * subgrad_at_lag(a, int(idx))
    return grad, float(conv[top_idx[0]])


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            beta_t = (t - 1) / (t + 2.0)
            y = a + beta_t * (a - prev_a)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = np.ones(n) / n
            else:
                y = y / sy
        else:
            y = a.copy()

        # Compute subgradient
        grad, max_conv = topk_subgradient(y, top_k)

        # Gradient step with adaptive lr
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        # Evaluate
        obj = 2.0 * n * fast_convolve(a_new).max()
        if obj < best_val:
            best_val = obj
            best_a = a_new.copy()

        # If not improving, reduce lr
        prev_a = a
        a = a_new
        lr *= decay

        if t % 200 == 0 and obj < 1.01:
            break

    return best_a, best_val


def mirror_descent(a0, n_iter=2000, lr0=0.01, top_k=3, decay=0.9995):
    """Mirror descent with entropy regularization (multiplicative updates)."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_val = 2.0 * n * fast_convolve(a).max()
    lr = lr0

    for t in range(n_iter):
        grad, max_conv = topk_subgradient(a, top_k)

        # Multiplicative update: a_i <- a_i * exp(-lr * grad_i)
        step = np.exp(-lr * grad)
        a = a * step
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_val:
            best_val = obj
            best_a = a.copy()

        lr *= decay
        if t % 200 == 0 and obj < 1.01:
            break

    return best_a, best_val


def _softmax_obj(x, n, beta):
    """Objective for scipy minimize with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _softmax_jac(x, n, beta):
    """Jacobian for scipy minimize with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_smooth_lbfgsb(a0, beta, maxiter=300):
    """L-BFGS-B optimization with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(_softmax_obj, x0, jac=_softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    for sf in [0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%d' % int(sf * 100)] = g

    # Frequency-domain init: try to make power spectrum flat
    fft_size = 1 << (n - 1).bit_length()
    for trial in range(10):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * phases[-1])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            inits['freq_%d' % trial] = a

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)

    # Symmetric inits
    half = np.ones(n // 2 + 1) / (n // 2 + 1)
    sym = np.zeros(n)
    sym[:n // 2 + 1] = half
    if n % 2 == 0:
        sym[n // 2:] = half[-2::-1]
    else:
        sym[n // 2 + 1:] = half[-1::-1]
    sym /= sym.sum()
    inits['symmetric'] = sym

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                pp = np.maximum(pp, 0)
                ps2 = pp.sum()
                if ps2 > 1e-10:
                    inits['parent_resample'] = pp / ps2

    return inits


def peak_equalize(a, n_steps=100, top_k=5):
    """Iteratively equalize peaks by boosting low-convolution regions."""
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_val = 2.0 * n * fast_convolve(a).max()

    for step in range(n_steps):
        conv = fast_convolve(a)
        top_idx = np.argsort(conv)[::-1][:top_k]
        bottom_idx = np.argsort(conv)[:top_k]

        max_val = conv[top_idx[0]]
        min_val = conv[bottom_idx[0]]

        if max_val - min_val < 1e-8:
            break

        # Direction: reduce contribution to top peaks, boost bottom
        grad_top = np.zeros(n)
        for idx in top_idx:
            grad_top += subgrad_at_lag(a, int(idx))
        grad_bot = np.zeros(n)
        for idx in bottom_idx:
            grad_bot += subgrad_at_lag(a, int(idx))

        # Move: reduce top peaks, increase bottom
        direction = grad_bot - grad_top
        direction /= (np.linalg.norm(direction) + 1e-15)

        # Try several step sizes
        improved = False
        for scale in [0.01, 0.005, 0.002, 0.001]:
            a_new = a + scale * direction
            a_new = project_simplex(a_new)
            obj = 2.0 * n * fast_convolve(a_new).max()
            if obj < best_val * (1.0 + 1e-6):
                a = a_new
                if obj < best_val:
                    best_val = obj
                    best_a = a_new.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_val


def coordinate_descent(a, n_iters=50):
    """Simple coordinate descent: for each coordinate, try scaling it."""
    n = len(a)
    a = a.copy()
    best_val = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for it in range(n_iters):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = a[i]
            for scale in [0.8, 0.9, 1.1, 1.2]:
                a[i] = old * scale
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    a[i] = old
                    break
                a = a / s
                obj = 2.0 * n * fast_convolve(a).max()
                if obj < best_val - 1e-10:
                    best_val = obj
                    best_a = a.copy()
                    improved = True
                    break
                a[i] = old
        if not improved:
            break

    return best_a, best_val


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining Nesterov subgradient,
    mirror descent, smooth L-BFGS-B, peak equalization, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: length=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d (t=%.0fs)" % (score, len(a), time.monotonic() - t_start))
        return score if np.isfinite(score) else float('inf')

    # === Phase 1: Quick multi-start scan ===
    print("=== Phase 1: Quick Scan ===")
    lengths = [128, 192, 256, 320, 384, 512]
    phase1_deadline = t_start + 250.0

    for n in lengths:
        if time.monotonic() > phase1_deadline - 20:
            break
        inits = generate_inits(n, parent, rng)
        # Pick top 4 inits
        priority = ['parent', 'parent_resample', 'freq_0', 'freq_1', 'uniform',
                    'gauss_30', 'triangle', 'random', 'symmetric']
        names = [nm for nm in priority if nm in inits] + \
                [nm for nm in inits if nm not in priority]
        names = names[:4]

        for name in names:
            if time.monotonic() > phase1_deadline - 10:
                break
            a0 = inits[name]
            # Quick Nesterov subgradient
            n_iter = int(min(500, 40.0 * 50))
            a_opt, _ = nesterov_subgrad(a0, n_iter=n_iter, lr0=0.015, top_k=3, decay=0.9997)
            score = score_and_update(a_opt)

    print("Phase 1 best: %.6f" % best_score)

    # === Phase 2: Deep optimization ===
    print("=== Phase 2: Deep Optimization ===")
    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256

    target_lengths = sorted(set([l for l in [best_n // 2, best_n, best_n * 2, 256, 384]
                                 if 64 <= l <= 1024]))
    phase2_deadline = deadline - 200.0

    # For each target length, run multiple optimization methods
    for n in target_lengths:
        if time.monotonic() > phase2_deadline - 60:
            break

        # Prepare best init for this length
        if best_sequence is not None and len(best_sequence) == n:
            a_start = best_sequence / best_sequence.sum()
        elif best_sequence is not None:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_sequence)), best_sequence)
            a_start = np.maximum(pp, 0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.ones(n) / n

        inits = generate_inits(n, parent, rng)
        inits['best'] = a_start

        # Method 1: Nesterov subgradient (long run)
        if time.monotonic() < phase2_deadline - 120:
            t_alloc = min(120.0, phase2_deadline - time.monotonic() - 120)
            n_iter = int(min(5000, t_alloc * 80))
            a_opt, _ = nesterov_subgrad(a_start, n_iter=n_iter, lr0=0.02, top_k=5, decay=0.9999)
            score = score_and_update(a_opt)
            if a_opt is not None:
                a_start = a_opt

        # Method 2: Mirror descent
        if time.monotonic() < phase2_deadline - 100:
            t_alloc = min(100.0, phase2_deadline - time.monotonic() - 100)
            n_iter = int(min(4000, t_alloc * 60))
            a_opt, _ = mirror_descent(a_start, n_iter=n_iter, lr0=0.008, top_k=5, decay=0.9999)
            score = score_and_update(a_opt)
            if score < best_score:
                a_start = a_opt

        # Method 3: Smooth L-BFGS-B with annealed beta
        if time.monotonic() < phase2_deadline - 80:
            t_alloc = min(120.0, phase2_deadline - time.monotonic() - 80)
            betas = [100.0, 50.0, 20.0, 10.0, 5.0, 3.0]
            a_cur = a_start.copy()
            for beta in betas:
                if time.monotonic() > phase2_deadline - 40:
                    break
                t_single = (phase2_deadline - time.monotonic() - 40) / len(betas)
                t_single = min(t_single, 40.0)
                if t_single < 5:
                    break
                maxiter = int(min(300, max(30, t_single * 25)))
                a_opt, _ = optimize_smooth_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
                score = score_and_update(a_opt)
                a_cur = a_opt

    print("Phase 2 best: %.6f" % best_score)

    # === Phase 3: Fine-tuning on the absolute best ===
    print("=== Phase 3: Fine-tuning ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        a_cur = best_sequence / best_sequence.sum()
        remaining = deadline - time.monotonic() - 30.0

        # 3a: Nesterov with very small lr and high top_k
        if remaining > 30:
            t_alloc = min(120.0, remaining * 0.3)
            n_iter = int(min(3000, t_alloc * 50))
            a_opt, _ = nesterov_subgrad(a_cur, n_iter=n_iter, lr0=0.005, top_k=8, decay=0.99995)
            score = score_and_update(a_opt)
            if score < best_score:
                a_cur = a_opt
            remaining = deadline - time.monotonic() - 30.0

        # 3b: Peak equalization
        if remaining > 20:
            a_opt, _ = peak_equalize(a_cur, n_steps=50, top_k=8)
            score = score_and_update(a_opt)
            if score < best_score:
                a_cur = a_opt
            remaining = deadline - time.monotonic() - 30.0

        # 3c: Smooth L-BFGS-B with very low beta
        for beta in [3.0, 2.0, 1.5, 1.0]:
            if remaining < 10:
                break
            t_single = min(40.0, remaining * 0.2)
            if t_single < 5:
                break
            maxiter = int(min(200, max(30, t_single * 20)))
            a_opt, _ = optimize_smooth_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
            score = score_and_update(a_opt)
            if score < best_score:
                a_cur = a_opt
            remaining = deadline - time.monotonic() - 30.0

        # 3d: Random perturbations + local optimization
        if remaining > 40:
            n_perturb = int(min(8, remaining / 10.0))
            for _ in range(n_perturb):
                if remaining < 10:
                    break
                scale = 0.002 + 0.003 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(a_cur + noise)
                # Quick Nesterov from perturbed point
                n_iter = int(min(500, remaining * 10))
                a_opt, _ = nesterov_subgrad(a_pert, n_iter=n_iter, lr0=0.008, top_k=5, decay=0.9999)
                score = score_and_update(a_opt)
                if score < best_score:
                    a_cur = a_opt
                remaining = deadline - time.monotonic() - 30.0

        # 3e: Coordinate descent polish
        if remaining > 15:
            a_opt, _ = coordinate_descent(a_cur, n_iters=30)
            score = score_and_update(a_opt)
            if score < best_score:
                a_cur = a_opt

        # 3f: Final Nesterov polish
        if remaining > 10:
            n_iter = int(min(1000, remaining * 30))
            a_opt, _ = nesterov_subgrad(a_cur, n_iter=n_iter, lr0=0.003, top_k=10, decay=0.99999)
            score = score_and_update(a_opt)

    # Try alternative lengths if time permits
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 80:
            alt_n = 384 if len(best_sequence) <= 256 else 256
            pp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, len(best_sequence)), best_sequence)
            a_alt = np.maximum(pp, 0)
            a_alt = a_alt / a_alt.sum()
            # Quick optimization
            a_opt, _ = nesterov_subgrad(a_alt, n_iter=1500, lr0=0.015, top_k=5, decay=0.9999)
            score = score_and_update(a_opt)
            if score < best_score:
                remaining = deadline - time.monotonic() - 20.0
                if remaining > 30:
                    a_opt2, _ = optimize_smooth_lbfgsb(a_opt, beta=5.0, maxiter=150)
                    score = score_and_update(a_opt2)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
