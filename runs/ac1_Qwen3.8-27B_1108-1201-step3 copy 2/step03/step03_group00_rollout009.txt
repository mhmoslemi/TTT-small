def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        n_p = len(parent)
        t1 = min(120, deadline - time.monotonic() - 1300)
        if t1 > 20:
            print(f"  FW-LP on parent (n={n_p}), t={t1:.0f}s")
            a_opt, _ = fw_lp_optimize(parent, time_limit=t1)
            try_seq(a_opt, "fw_parent")
    else:
        n_p = 384
        parent = np.ones(n_p) / n_p

    # Try a couple other initializations briefly
    rem = deadline - time.monotonic() - 1300
    if rem > 30:
        for fi in [1, 2]:
            rem = deadline - time.monotonic() - 1200
            if rem < 15:
                break
            inits = generate_inits(n_p, parent, rng)
            if fi < len(inits):
                a_opt, _ = fw_lp_optimize(inits[fi], time_limit=min(rem, 40))
                try_seq(a_opt, f"init_{fi}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep FW-LP (~500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 800
        if rem > 30:
            t_alloc = min(rem, 500)
            print(f"  Deep FW-LP, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

            # Try a perturbed start
            rem = deadline - time.monotonic() - 600
            if rem > 20:
                noise = rng.standard_normal(n_best) * 0.005
                f_pert = project_simplex(best_seq + noise)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=min(rem, 150))
                try_seq(a_opt2, "fw_perturbed")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Multiplicative Weight Optimization (~400s) ===
    print("\n=== Phase 3: Multiplicative Weight Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Run multiple MW passes with different hyperparameters
        mw_configs = [
            {'lr_init': 0.03, 'top_k': 10, 'temp_init': 0.0003},
            {'lr_init': 0.05, 'top_k': 20, 'temp_init': 0.0005},
            {'lr_init': 0.02, 'top_k': 15, 'temp_init': 0.0002},
            {'lr_init': 0.04, 'top_k': 30, 'temp_init': 0.0008},
        ]

        rem = deadline - time.monotonic() - 350
        t_per_mw = max(10, rem / len(mw_configs))

        for ci, cfg in enumerate(mw_configs):
            rem = deadline - time.monotonic() - 250
            if rem < 10:
                break
            t_alloc = min(t_per_mw, rem)
            print(f"  MW config {ci} (lr={cfg['lr_init']}, k={cfg['top_k']}), t={t_alloc:.0f}s")
            a_opt, _ = mult_weight_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=cfg['lr_init'], top_k=cfg['top_k'],
                temp_init=cfg['temp_init']
            )
            try_seq(a_opt, f"mw_{ci}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Peak-Chasing Subgradient (~300s) ===
    print("\n=== Phase 4: Peak-Chasing Subgradient ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple SG runs with different settings
        sg_configs = [
            {'lr': 0.003, 'mom': 0.92},
            {'lr': 0.002, 'mom': 0.94},
            {'lr': 0.004, 'mom': 0.90},
            {'lr': 0.0015, 'mom': 0.95},
        ]

        rem = deadline - time.monotonic() - 200
        t_per_sg = max(10, rem / len(sg_configs))

        for si, cfg in enumerate(sg_configs):
            rem = deadline - time.monotonic() - 120
            if rem < 10:
                break
            t_alloc = min(t_per_sg, rem)
            print(f"  PG-SG config {si} (lr={cfg['lr']}), t={t_alloc:.0f}s")
            a_opt, _ = peak_chase_sg(
                best_seq, time_limit=t_alloc,
                lr=cfg['lr'], mom=cfg['mom']
            )
            try_seq(a_opt, f"pgsg_{si}")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final Fine Polish (~150s) ===
    print("\n=== Phase 5: Final Fine Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine SG
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem, 80)
            print(f"  Fine SG, t={t_alloc:.0f}s")
            a_opt, _ = fine_sg_polish(
                best_seq, time_limit=t_alloc,
                lr=0.0005, mom=0.96, top_k=n_best // 3, temp=0.00002
            )
            try_seq(a_opt, "fine_sg")

        # Quick MW with very small lr
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            print(f"  Fine MW, t={min(rem, 40):.0f}s")
            a_opt, _ = mult_weight_optimize(
                best_seq, time_limit=min(rem, 40),
                lr_init=0.005, top_k=8, temp_init=0.0001,
                decay_lr=0.9998, decay_temp=1.001
            )
            try_seq(a_opt, "fine_mw")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            for _ in range(5):
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = fine_sg_polish(
                    f_pert, time_limit=min(rem, 15),
                    lr=0.0003, mom=0.96, top_k=n_best // 2, temp=0.00001
                )
                try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
