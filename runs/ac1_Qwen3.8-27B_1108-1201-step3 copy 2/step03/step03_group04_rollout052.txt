Now, for a given n, what's the minimum of max(conv(a,a)) over non-negative a with sum(a)=1?

This is a well-studied problem in the context of "minimum peak auto-correlation" sequences. For non-negative sequences, the optimal solution is generally not known in closed form, but numerical methods can get close.

One key insight: the problem is equivalent to finding a probability distribution on {0,...,n-1} that minimizes the maximum probability mass at any point of the sum of two independent draws from that distribution. This is related to the "addition theorem" in probability.

For continuous distributions, if X ~ Uniform[0,1], then X+X ~ Triangular[0,2] with peak density 1 (at x=1). The probability mass at any single point is 0 (continuous), but the "peak density" is 1.

For discrete: if p_j = 1/n for all j, then (p*p)[k] = sum_j p_j * p[k-j]. At k = n-1, this is n * (1/n)^2 = 1/n. So the peak is 1/n.

To reduce the peak below 1/n, we need a non-uniform distribution. The question is how much we can reduce it.

I think the key insight is that we want a distribution whose "sumset" is as spread out as possible. A distribution concentrated at the edges (like a U-shape) has a sumset that's more spread out, leading to lower peaks.

Mirror descent + coordinate descent hybrid for minimizing

Uses soft-max peak weighting, adaptive learning rates,

def fast_convolve(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def local_obj(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # U-shaped (high at edges)
    u = x.astype(np.float64)
    d = np.minimum(u - (n-1)/2, (n-1)/2 - u)
    u_shape = d / (n/2)  # 0 at center, 1 at edges
    u_shape = u_shape ** 2 + 0.01  # avoid zeros
    u_shape /= u_shape.sum()
    inits['U'] = u_shape
    
    # Inverted U (high at center) - Gaussian
    sigma = n / 4
    g = np.exp(-0.5 * ((x - (n-1)/2) / sigma) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # W-shaped (two bumps)
    c1 = (n-1) * 0.25
    c2 = (n-1) * 0.75
    w = np.exp(-0.5 * ((x - c1) / (n/8)) ** 2) + np.exp(-0.5 * ((x - c2) / (n/8)) ** 2)
    w /= w.sum()
    inits['W'] = w
    
    # Edge-concentrated (first and last n/4 positions)
    edge = max(2, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec
    
    # Cosine decay from center
    c = np.cos(np.pi * (x - (n-1)/2) / (n)) ** 2
    c = np.maximum(c, 0.001)

    inits['cos'] = c

    if r.sum() < 1e-10:
        r = rng.dirichlet(np.ones(n))
    else:
        r /= r.sum()
    inits['sparse'] = r
    
    return inits

    subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-10)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temp_scale = 0.005  # temperature for soft-max
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        # Soft-max over top peaks
        n_peaks = min(top_k, len(conv))
        peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_vals = conv[peak_idx]
        
        # Adaptive temperature: start low, increase over time
        temp = peak_vals.max() * temp_scale
        if temp < 1e-18:
            temp = 1e-18
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w /= w.sum()

        for i in range(n_peaks):
            if w[i] < 1e-14:
                continue
            k = int(peak_idx[i])

            j_max = min(n - 1, k)

        a = a / a.sum()
        
        conv = fast_convolve(a)
        cur_max = conv.max()
        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            lr *= 0.7
            no_improve = 0

    return best_a, best_max

def projected_gradient(a0, time_limit, verbose=False):

    Projecting onto the simplex (a>=0, sum=1) is O(n log n).
    """

    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()

    best_max = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

        conv = fast_convolve(a)
        current_max = conv.max()

        k = peak_idx
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        g = np.zeros(n)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)

        a = project_simplex(a)

        cur_max = conv.max()
        if cur_max < best_max - 1e-15:

            noise = rng.standard_normal(n) * 0.001
            a = project_simplex(best_a + noise)
            lr = 0.01
            no_improve = 0
        
        it += 1
    
    return best_a, best_max

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Nesterov-accelerated mirror descent with soft-max peak weighting,
multi-length scanning, diverse initializations (U-shaped, W-shaped, edge),
and perturbation restarts to minimize 2*n*max(auto-conv)/(sum)^2.
"""

import numpy as np
import time


def fast_convolve(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def local_obj(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gen_inits(n, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # U-shaped (high at edges, low at center)
    d = np.minimum(x - (n - 1) / 2.0, (n - 1) / 2.0 - x)
    u = (d / (n / 2.0)) ** 2
    u += 0.005
    u /= u.sum()
    inits['U'] = u

    # Stronger U
    u2 = (d / (n / 2.0)) ** 4
    u2 += 0.002
    u2 /= u2.sum()
    inits['U2'] = u2

    # Gaussian center
    sigma = n / 5.0
    g = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # W-shaped (two bumps at quartiles)
    c1 = (n - 1) * 0.25
    c2 = (n - 1) * 0.75
    w = np.exp(-0.5 * ((x - c1) / (n / 10.0)) ** 2) + np.exp(-0.5 * ((x - c2) / (n / 10.0)) ** 2)
    w += 0.001
    w /= w.sum()
    inits['W'] = w

    # Edge blocks
    edge = max(2, n // 5)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec

    # Cosine shape
    c = np.cos(np.pi * (x - (n - 1) / 2.0) / n) ** 2
    c = np.maximum(c, 0.001)
    c /= c.sum()
    inits['cos'] = c

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse
    r = rng.random(n)
    r[r < 0.4] = 0.0
    if r.sum() < 1e-10:
        r = rng.dirichlet(np.ones(n))
    else:
        r /= r.sum()
    inits['sparse'] = r

    return inits


def nesterov_md(a0, time_limit, top_k=30, lr_init=0.06, temp_init=0.004, verbose=False):
    """
    Nesterov-accelerated mirror descent to minimize max(auto-conv)
    subject to sum(a)=1, a>=0. Uses soft-max peak weighting.
    """
    n = len(a0)
    x = np.maximum(a0.copy(), 1e-10)
    x = x / x.sum()
    y = x.copy()

    t_mom = 1.0
    lr = lr_init
    temp_scale = temp_init
    t0 = time.monotonic()

    conv = fast_convolve(x)
    best_max = conv.max()
    best_a = x.copy()

    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov momentum step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        y = x + beta * (x - y)
        y = np.maximum(y, 1e-15)
        ys = y.sum()
        if ys < 1e-15:
            y = x.copy()
        else:
            y = y / ys
        t_mom = t_new

        # Compute soft-max weighted subgradient at y
        conv_y = fast_convolve(y)
        n_peaks = min(top_k, len(conv_y))
        peak_idx = np.argpartition(conv_y, -n_peaks)[-n_peaks:]
        peak_vals = conv_y[peak_idx]

        temp = max(peak_vals.max() * temp_scale, 1e-18)
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w /= w.sum()

        g = np.zeros(n)
        for i in range(n_peaks):
            if w[i] < 1e-14:
                continue
            k = int(peak_idx[i])
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                js = np.arange(j_min, j_max + 1)
                g[js] += 2.0 * w[i] * y[k - js]

        # Clip gradient
        gmax = g.max()
        if gmax > 5.0:
            g *= 5.0 / gmax

        # Mirror descent update
        x_new = y * np.exp(-lr * g)
        x_new = np.maximum(x_new, 1e-15)
        x_new = x_new / x_new.sum()

        conv_x = fast_convolve(x_new)
        cur_max = conv_x.max()
        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        x = x_new

        # Adaptive learning rate
        if no_improve > 150:
            lr *= 0.6
            no_improve = 0
        if no_improve > 800:
            # Perturbation restart
            rng = np.random.default_rng(it + 31337)
            scale = 0.001 + 0.002 * (it % 7)
            noise = rng.standard_normal(n) * scale
            x = np.maximum(best_a + noise, 1e-15)
            x = x / x.sum()
            y = x.copy()
            t_mom = 1.0
            lr = max(lr_init * 0.4, 0.01)
            no_improve = 0
            temp_scale = min(0.08, temp_scale * 1.3)

        # Slow decay
        lr *= 0.99997
        temp_scale = min(0.12, temp_scale * 1.00008)

        it += 1
        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD it={it}, obj={local_obj(best_a):.6f}, t={t_mom:.0f}, lr={lr:.3e}, ell={elapsed:.0f}s")

    return best_a, best_max


def plain_md(a0, time_limit, top_k=20, lr_init=0.1, verbose=False):

    n = len(a0)

    best_max = conv.max()

    t0 = time.monotonic()

        n_peaks = min(top_k, len(conv))
        peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]

def peak_balance(a0, time_limit, verbose=False):
    """
    Peak balancing: iteratively equalize the top peaks by adjusting
    elements that contribute most to the highest peak and least to others.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-10)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = 0.02

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        current_max = conv.max()
        target = conv.max() * 0.95  # try to reduce max by 5%

        # Find top peaks
        top_k = min(20, len(conv))
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_idx]

        # For the highest peak, compute which elements contribute most
        k_max = int(peak_idx[np.argmax(peak_vals)])
        j_min = max(0, k_max - n + 1)
        j_max = min(n - 1, k_max)
        contributors = np.arange(j_min, j_max + 1)
        contrib_vals = a[k_max - contributors]

        # Find elements that contribute to the max peak but NOT to other peaks
        other_contrib = np.zeros(n)
        for pk in peak_idx:
            if pk == k_max:
                continue
            pk = int(pk)
            jm = max(0, pk - n + 1)
            jM = min(n - 1, pk)
            if jm <= jM:
                other_contrib[jm:jM + 1] += a[pk - np.arange(jm, jM + 1)]

        # Elements that are "safe" to reduce (high contribution to max, low to others)
        safety = contrib_vals - 0.5 * other_contrib[contributors]
        # Only reduce elements with positive safety
        mask = safety > 0.3 * contrib_vals.max() if contrib_vals.max() > 0 else np.zeros(len(contributors), dtype=bool)

        if mask.sum() > 0:
            # Reduce those elements, redistribute to others
            total_reduction = lr * contrib_vals[mask].sum()
            if total_reduction > 0:
                for idx in np.where(mask)[0]:
                    j = contributors[idx]
                    reduction = lr * contrib_vals[idx]
                    a[j] -= reduction
                    a[j] = max(a[j], 1e-15)
                    # Add to random other positions
                    other_positions = np.random.choice([p for p in range(n) if p != j], size=3, replace=False)
                    a[other_positions] += reduction / 3.0

                a = np.maximum(a, 1e-15)
                a = a / a.sum()

        conv = fast_convolve(a)
        cur_max = conv.max()
        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.5
            no_improve = 0
        if no_improve > 1000:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(best_a + noise, 1e-15)
            a = a / a.sum()
            lr = 0.02
            no_improve = 0

        it += 1

    return best_a, best_max
