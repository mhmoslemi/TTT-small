"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using aggressive LP-based 
direction finding with bisection step selection, coordinate descent,
and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        A_ub[k_start:k_end, i] = a[k_end - k_start - 1::-1]  # Need to be careful here

def smoothed_gradient_descent(a0, time_limit, beta_init=50.0, beta_end=500.0,
                               lr_init=0.01, momentum=0.92, seed=0):
    """
    Nesterov accelerated projected gradient descent on smoothed objective.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    perturb_every = 300
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        
        # Nesterov lookahead
        y = a + momentum * v
        conv_y = fast_convolve(y)
        
        # Softmax gradient
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum
        
        w = conv_y * p
        
        # grad_i = 2 * sum_m a[m] * w[m+i] (cross-correlation)
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(A * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]
        
        # Momentum update
        v = momentum * v - lr_init * (1 - 0.5 * frac) * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)
    
    return best_a, best_obj

def generate_bimodal(n, alpha=2.0, seed=None):
    """Generate a bimodal (U-shaped) sequence."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(-alpha * (x - 2 * center * abs(x - center) / (2 * center + 1)) / center))
    # Simpler: a[i] proportional to distance from center
    a = 1.0 + alpha * (np.abs(x - center) / center) ** 2
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a

Minimizes 2n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding with
dual-guided coordinate descent, smoothed gradient descent, and SDR relaxation
for small sequences.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a, n):
    """Build the constraint matrix for the LP."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A_ub[k_start:k_end, j] = a[:k_end - k_start]
    return A_ub

    except Exception:
        pass
    
    return b / b_sum, duals
