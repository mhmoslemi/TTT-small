"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. LP-based direction finding with second-order step sizing
2. Nesterov-accelerated gradient descent with adaptive peak weighting
3. Multi-scale initialization and coordinate descent polishing
4. Hierarchical extension (optimize at smaller n, interpolate to larger n)
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_soft_gradient(a, conv, top_k=20, temp=0.005):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns (g, g_sum, success)."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step_with_sizing(a, conv, n):
    """
    One LP-based step with second-order step sizing.
    Returns (a_new, improved).
    """
    M = conv.max()
    peak_idx = int(np.argmax(conv))

    C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.0001, C)

    if not success or g_sum <= 1.0 + 1e-8:
        return a.copy(), False

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False

    # Compute second-order info for step sizing
    conv_ag = C @ g  # conv(a, g)
    conv_ag_peak = conv_ag[peak_idx]
    D = conv_ag_peak - M  # <= 0 expected

    # Try several step sizes, prefer larger
    best_a = a.copy()
    best_obj = 2.0 * n * M
    best_t = 0.0

    # Estimate max useful step from quadratic model
    if D < -1e-10:
        t_max_est = min(1.0, -D / max(abs(D) * 0.1, 1e-10))
    else:
        t_max_est = 0.3

    candidates = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    candidates = [t for t in candidates if t <= t_max_est * 1.5] or [0.05]

    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:

        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = t
            break  # First improvement in geometric sequence is good enough

    if best_t > 0:
        return best_a, True
    return a.copy(), False

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov update

        y = np.maximum(y, 0.0)

        if s_y > 1e-15:

            no_improve = 0

        temp *= 1.003
        mom = min(0.97, mom + 0.0002)

            M = conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            lp_fail_streak = 0
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, window=20):
    """Coordinate descent: try adjusting pairs of elements to reduce peak."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        batch_size = max(1, int(500000 / n))
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Pick two random positions to swap mass
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try moving small amount from i to j and vice versa
            for delta in [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
            else:
                no_improve += 1

        if no_improve > 200:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sine-based (lags at multiples)
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()

    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()

    # Bimodal (mass at edges)
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Power law
    pw = (x + 1.0) ** 0.3
    inits['power'] = pw / pw.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length via interpolation + noise."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid LP + gradient + coordinate descent search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick gradient scan across lengths (fast)
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = 200.0

    lengths = [256, 512, 768]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_initializations(n, parent, rng)

        # Prioritize parent-based inits
        priority = ['parent', 'parent_interp', 'uniform', 'sine1', 'random', 'alternating']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]

        t_per = max(10.0, (phase1_limit - 10) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 200)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj = nesterov_gradient_optimize(a0, time_limit=t_alloc,
                                                     lr_init=0.012, momentum=0.91,
                                                     top_k=40, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP-based deep optimization on best candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    # Collect top candidates (unique by length)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    # Always include best overall
    if best_sequence is not None:
        if len(best_sequence) not in seen_lengths:
            top_configs.append((len(best_sequence), best_sequence.copy()))
            seen_lengths.add(len(best_sequence))

    remaining = deadline - time.monotonic() - 300.0
    n_candidates = max(1, len(top_configs))
    t_per_lp = max(30.0, remaining * 0.6 / n_candidates)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 200)
        if t_alloc < 10:
            break

        print(f"  LP candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP polish on the single best (with more iterations)
    # =========================================================
    print("=== Phase 3: LP Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        if remaining > 50:
            # Multiple LP runs with different perturbation seeds
            for pi in range(3):
                if time.monotonic() > deadline - 100:
                    break
                t_alloc = min(remaining / (3 - pi) * 0.8, deadline - time.monotonic() - 80)
                if t_alloc < 15:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                else:
                    scale = 0.002 * pi
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()

                a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish LP {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Polish LP {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Coordinate descent + gradient fine-tuning
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)

        # Coordinate descent
        t_cd = min(remaining * 0.5, 60.0)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

        # Short gradient runs
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 10:
            for gi in range(3):
                if time.monotonic() > deadline - 10:
                    break
                t_sg = min(remaining / (3 - gi) * 0.7, 30.0)
                if t_sg < 5:
                    break

                cfgs = [
                    {'lr_init': 0.005, 'momentum': 0.93, 'top_k': 60, 'temp_init': 0.001},
                    {'lr_init': 0.003, 'momentum': 0.95, 'top_k': 100, 'temp_init': 0.0005},
                    {'lr_init': 0.01, 'momentum': 0.88, 'top_k': 30, 'temp_init': 0.003},
                ]
                cfg = cfgs[gi % len(cfgs)]
                a_opt, obj = nesterov_gradient_optimize(best_sequence, time_limit=t_sg, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG {gi}: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final LP micro-polish
    # =========================================================
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(20.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
