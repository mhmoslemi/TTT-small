import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def lp_cross_correlation(a):
    """
    Solve: min_g max_k (a*g)_k s.t. g >= 0, sum(g) = 1
    Returns g*.
    """
    n = len(a)
    C = build_conv_matrix(a)
    
    # Variables: g (n) + t (1) = n+1
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: C @ g - t <= 0  (2n-1 constraints)
    #                     -g <= 0   (n constraints, i.e., g >= 0)
    A_ub = np.zeros((2 * n - 1 + n, n + 1))
    A_ub[:2*n-1, :n] = C
    A_ub[:2*n-1, -1] = -1.0
    A_ub[2*n-1:, :n] = -np.eye(n)
    b_ub = np.zeros(2 * n - 1 + n)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n]
    return a.copy()

def obj_value(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def line_search(a, d, n, current_obj, max_evals=30):
    """Adaptive line search."""
    best_t = 0.0
    best_obj = current_obj
    
    candidates = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
    
    for t in candidates:
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try, _ = obj_value(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t
    
    # Refine
    if 0.01 < best_t < 0.99:
        step = 0.05
        for _ in range(4):

                t_try = best_t + dt
                if t_try <= 0 or t_try >= 1:

                a_try = np.maximum(a + t_try * d, 0.0)
                s = a_try.sum()
                if s < 1e-15:

                a_try = a_try / s
                obj_try, _ = obj_value(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_t = t_try
            step *= 0.4
    
    return best_t, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel cross-correlation LP
direction-finding method combined with the sum-maximizing LP, adaptive line
search, multi-phase search strategy, and diverse initializations.

Key innovation: The cross-correlation LP directly minimizes max_k (a*g)_k,
which provably gives a descent direction for the autocorrelation peak,
potentially escaping local minima that the sum-maximizing LP cannot.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    C = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C


def obj_value(a):
    """Compute 2n * max(conv(a,a)) for the given a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def lp_cross_correlation(a):
    """
    Solve: min_g max_k (a*g)_k s.t. g >= 0, sum(g) = 1.
    This gives a proven descent direction for the autocorrelation peak.
    Returns normalized g*.
    """
    n = len(a)
    C = build_conv_matrix(a)
    
    # Variables: g_0..g_{n-1}, t  (n+1 total)
    # Minimize t
    # s.t. C @ g - t <= 0  (2n-1 constraints)
    #      -g <= 0         (n constraints)
    #      sum(g) = 1     (1 equality)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1 + n, n + 1))
    A_ub[:2 * n - 1, :n] = C
    A_ub[:2 * n - 1, -1] = -1.0
    A_ub[2 * n - 1:, :n] = -np.eye(n)
    b_ub = np.zeros(2 * n - 1 + n)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 30, 'presolve': True})
        if result.success:
            g = result.x[:n]
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return a.copy()


def lp_sum_maximizing(a, M):
    """
    Solve: max sum(g) s.t. C(a) @ g <= M, g >= 0.
    Returns normalized g.
    """
    n = len(a)
    C = build_conv_matrix(a)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=np.full(2 * n - 1, M),
                         bounds=bounds, method='highs',
                         options={'time_limit': 30, 'presolve': True})
        if result.success:
            g = result.x
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return a.copy()


def line_search(a, d, current_obj, max_evals=40):
    """Adaptive line search with refinement."""
    best_t = 0.0
    best_obj = current_obj
    evals = 0
    
    # Coarse search
    candidates = [1.0, 0.85, 0.7, 0.55, 0.45, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    
    for t in candidates:
        if evals >= max_evals:
            break
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try, _ = obj_value(a_try)
        evals += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t
    
    # Fine refinement
    if 0.01 < best_t < 0.99 and evals < max_evals:
        step = 0.04
        for _ in range(5):
            improved = False
            for dt in [-step, step]:
                t_try = best_t + dt
                if t_try <= 0.001 or t_try >= 0.999 or evals >= max_evals:
                    continue
                a_try = np.maximum(a + t_try * d, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                obj_try, _ = obj_value(a_try)
                evals += 1
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_t = t_try
                    improved = True
            if not improved:
                step *= 0.4
            else:
                step *= 0.5
    
    return best_t, best_obj


def lp_optimize(a0, time_limit, verbose=False, use_cross=True, use_sum=True):
    """
    Iterative LP-based optimization using cross-correlation and/or
    sum-maximizing directions with adaptive line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj, conv = obj_value(a)
    best_a = a.copy()
    M = conv.max()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 80
    perturb_count = 0
    use_cross = True
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Choose direction
        d = None
        obj_best_new = best_obj
        
        # Cross-correlation direction (primary)
        if use_cross:
            g_cross = lp_cross_correlation(a)
            d_cross = g_cross - a
            norm_d = np.linalg.norm(d_cross)
            if norm_d > 1e-14:
                t_best, obj_val = line_search(a, d_cross, best_obj, max_evals=25)
                if obj_val < best_obj - 1e-13:
                    d = d_cross
                    obj_best_new = obj_val
        
        # Sum-maximizing direction (secondary, if cross didn't help)
        if obj_best_new >= best_obj - 1e-13 and use_sum:
            M_lp = M * 1.001
            g_sum = lp_sum_maximizing(a, M_lp)
            d_sum = g_sum - a
            norm_d = np.linalg.norm(d_sum)
            if norm_d > 1e-14:
                t_best, obj_val = line_search(a, d_sum, best_obj, max_evals=25)
                if obj_val < obj_best_new - 1e-13:
                    d = d_sum
                    obj_best_new = obj_val
        
        if d is not None and obj_best_new < best_obj - 1e-13:
            a_new = np.maximum(a + t_best * d, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
                best_obj, conv = obj_value(a)
                M = conv.max()
                best_a = a.copy()
                no_improve = 0
                if verbose and iteration % 50 == 0:
                    print(f"      iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")
            else:
                a = best_a.copy()
                no_improve += 1
        else:
            no_improve += 1
        
        iteration += 1
        
        # Stagnation: perturb and restart
        if no_improve >= max_no_improve:
            scale = 0.003 + 0.008 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            best_obj, conv = obj_value(a)
            M = conv.max()
            best_a = a.copy()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(300, 80 + 40 * perturb_count)
            
            if perturb_count > 40:
                break
    
    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.012, momentum=0.93, top_k=60, temp=0.0002):
    """
    Projected subgradient descent with momentum and softmax-weighted peaks.
    Used as a polishing step.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    _, conv = obj_value(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    batch = 40
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find top peaks
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum
            
            # Subgradient
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[i] * 2.0 * a[k - idx]
            
            v = momentum * v - lr * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()
            
            _, conv = obj_value(a)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        lr *= 0.998
        temp *= 1.003
        momentum = min(0.97, momentum + 0.0002)
        
        if no_improve >= 400:
            break
    
    return best_a, best_obj


def generate_variants(parent, n, rng, num_variants=8):
    """Generate diverse variants around the parent sequence."""
    variants = []
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent.astype(np.float64), 0)
        ps = p.sum()
        if ps > 1e-10:

            if len(p) == n:
                variants.append(p.copy())
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    variants.append(pp / pps)
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    variants.append(pp / pps)
    
    if not variants:
        variants.append(np.ones(n) / n)
    
    base = variants[0]
    
    # Small noise variants
    for scale in [0.005, 0.01, 0.02, 0.05]:
        noise = rng.standard_normal(n) * scale
        v = np.maximum(base + noise, 0)
        s = v.sum()
        if s > 1e-10:

    # Frequency modulation

        if s > 1e-10:

    # Pure random
    variants.append(rng.dirichlet(np.ones(n) * 0.5))

    variants.append(np.ones(n) / n)
    
    return variants

            n_perturb = max(2, int(remaining / 25))
            t_per_perturb = remaining / n_perturb
            
            for pi in range(n_perturb):
                remaining = deadline - time.monotonic() - 3
                if remaining < 8:
                    break
                if t_per_perturb < 5:
                    break
                
                scale = 0.002 * (pi + 1)
                noise = rng.standard_normal(n_target) * scale
                a_pert = np.maximum(best_sequence + noise, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s
                
                t_alloc = min(t_per_perturb, remaining - 3)
                if t_alloc < 3:
                    break
                a_opt, obj_opt = lp_optimize(a_pert, time_limit=t_alloc, verbose=False)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi}: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # === Fallback ===
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    print(f"Time used: {time.monotonic() - (deadline - float(budget_s) + 20):.0f}s / {budget_s}s")
    
    return [float(x) for x in best_sequence]
