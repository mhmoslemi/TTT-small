import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    tmp = u * np.arange(1, n + 1) - (cssv - 1)
    cond = tmp > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, K=50, beta=0.5):
    """Compute weighted subgradient of max_k (a*a)_k using top-K peaks with softmax weighting."""
    n = len(a)
    # Use exponential weighting over all peaks above a threshold
    threshold = conv.max() * (1.0 - beta)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Softmax weights over active peaks
    vals = conv[active]
    max_val = vals.max()
    weights = np.exp(-(max_val - vals) * n * 0.1)
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for idx in range(len(active)):
        if weights[idx] < 1e-15:
            continue
        k = int(active[idx])
        w = 2.0 * weights[idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            if j == k - j:
                sg[j] += w * a[j]
            else:
                sg[j] += w * a[k - j]
    return sg

def build_lp_matrix(n):
    """Build the constraint matrix for the LP direction problem."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = 1  # Will be scaled by f later
    return A_ub

    stall = 0
    
    while time.monotonic() - t0 < time_limit:

        if M < 1e-15:

        # Build LP: max sum(g) s.t. g>=0, (f*g)_k <= M

            break

        g = g / g_sum

        conv = fast_convolve(a)

            best_a = a.copy()
            stall = 0
        else:
            stall += 1

            blend = max(blend_min, blend * 0.5)
            stall = 0

                break
        
        if stall > blend_halve_every * 4:
            rng_local = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            a = project_simplex(a + rng_local.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            stall = 0
    
    return best_a, best_obj


def coordinate_descent(a_in, max_rounds=50, perturb_size=0.03):
    """Aggressive coordinate descent: try moving mass between coordinate pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    # Find the peak positions
    peak_pos = np.argmax(conv)
    
    for rnd in range(max_rounds):
        improved = False
        # Compute responsibility for the peak
        k = peak_pos
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        
        # Sort coordinates by responsibility (descending)
        resp = np.zeros(n)
        for j in range(j_start, j_end + 1):
            resp[j] = a[k - j]
        
        # Try moving mass from high-resp to low-resp coordinates
        high_resp = np.argsort(resp)[::-1][:n // 4]
        low_resp = np.where(resp < 1e-8)[0]
        
        for i in high_resp:
            if a[i] < 1e-8:
                continue
            for j in low_resp[:20]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = a[i] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-13:
                        a = a_test
                        best_obj = obj_test
                        best_a = a.copy()
                        conv = conv_test
                        improved = True
                        # Recompute peak
                        peak_pos = np.argmax(conv)
                        k = peak_pos
                        j_start = max(0, k - n + 1)
                        j_end = min(n - 1, k)
                        resp = np.zeros(n)
                        for jj in range(j_start, j_end + 1):
                            resp[jj] = a[k - jj]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_a


def simulated_annealing(a_in, time_limit, T_init=0.01, T_decay=0.9995, 
                        move_size=0.02, accept_ratio=0.7):
    """Simulated annealing with coordinate pair moves."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    T = T_init
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        # Random move: pick two coordinates, swap some mass
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = move_size * max(a[i], a[j])
        if delta < 1e-10:
            continue
        
        a_test = a.copy()
        a_test[i] -= delta
        a_test[j] += delta
        a_test = np.maximum(a_test, 0.0)
        st = a_test.sum()
        if st < 1e-15:
            continue
        a_test /= st
        
        conv_test = fast_convolve(a_test)
        obj_test = 2.0 * n * conv_test.max()
        cur_obj = 2.0 * n * conv.max()
        
        delta_obj = obj_test - cur_obj
        if delta_obj < 0 or (T > 1e-8 and rng.random() < np.exp(-delta_obj / (T * max(cur_obj, 1e-10)))):
            a = a_test
            conv = conv_test
            if obj_test < best_obj:
                best_obj = obj_test
                best_a = a.copy()
        
        T *= T_decay
    
    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Random (Dirichlet with low concentration for sparsity)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 8)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # W-shaped (4 peaks)
    w_shape = np.exp(-0.5 * ((x - 0.125 * n) / (0.08 * n)) ** 2) + \
              np.exp(-0.5 * ((x - 0.375 * n) / (0.08 * n)) ** 2) + \
              np.exp(-0.5 * ((x - 0.625 * n) / (0.08 * n)) ** 2) + \
              np.exp(-0.5 * ((x - 0.875 * n) / (0.08 * n)) ** 2)
    inits['w_shape'] = w_shape / w_shape.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                ps2 = p_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = p_interp / ps2
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining MWD, soft subgradient, LP, simulated annealing,
    and coordinate descent. Explores multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            best_sequence = p.copy()
            best_score = par_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    # If no parent, use a reasonable default
    if best_sequence is None:
        n_def = 512
        best_sequence = np.ones(n_def) / n_def
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"Default: n={n_def}, score={best_score:.6f}")
    
    # Phase 1: Quick exploration of different lengths and inits
    print("=== Phase 1: Quick Exploration ===")
    parent_len = len(parent) if parent is not None else 512
    lengths = sorted(set([parent_len, max(192, parent_len - 64), 
                          max(192, parent_len + 64), 384, 448, 512, 576, 640]))
    lengths = [l for l in lengths if 128 <= l <= 1024]
    
    phase1_end = time.monotonic() + min(300.0, (deadline - time.monotonic()) * 0.25)
    results = []
    
    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_inits(n, parent, rng)
        # Priority: parent first, then diverse inits
        prio = ['parent', 'parent_interp', 'random', 'random2', 'bimodal', 
                'w_shape', 'uniform', 'triangle', 'edge']
        names = [nm for nm in prio if nm in inits]
        names = names[:4]  # Limit per length
        
        t_per = max(15.0, 200.0 / (len(lengths) * 4))
        
        for name in names:
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break
            
            a0 = inits[name]
            
            # Quick MWD
            a_opt, obj_opt = optimize_mwd(a0, t_alloc * 0.5, eta_init=0.08, eta_decay=0.998)
            
            # Quick soft subgrad
            a_opt2, obj2 = optimize_soft_subgrad(a_opt, t_alloc * 0.3, lr_init=0.01, momentum=0.9, beta=0.4)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase1 n={n} {name}: {score:.6f}")
    
    # Sort results and take top candidates
    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    
    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in results[:10]:
        key = n  # One candidate per length
        if key not in seen:
            top_candidates.append((n, a_opt.copy()))
            seen.add(key)
        if len(top_candidates) >= 6:
            break
    # Always include current best
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 250.0
    t_per_cand = max(30.0, remaining / max(1, len(top_candidates)))
    
    for n, a0 in top_candidates:
        if time.monotonic() > deadline - 220:
            break
        t_alloc = min(t_per_cand, deadline - time.monotonic() - 200)
        if t_alloc < 15:
            break
        
        print(f"  Optimizing n={n}, t_alloc={t_alloc:.1f}")
        
        # Strategy: alternate between MWD and subgrad with decreasing step sizes
        a_cur = a0.copy()
        
        # Round 1: MWD with high eta
        t1 = t_alloc * 0.25
        a_opt, obj_opt = optimize_mwd(a_cur, t1, eta_init=0.05, eta_decay=0.999)
        if obj_opt < 2.0 * n * fast_convolve(a_cur).max():
            a_cur = a_opt
        
        # Round 2: Soft subgrad
        t2 = min(30.0, t_alloc * 0.25)
        a_opt2, obj2 = optimize_soft_subgrad(a_cur, t2, lr_init=0.008, momentum=0.92, beta=0.3)
        if obj2 < 2.0 * n * fast_convolve(a_cur).max():
            a_cur = a_opt2
        
        # Round 3: LP
        t3 = min(40.0, t_alloc * 0.3)
        a_opt3, obj3 = optimize_lp(a_cur, t3, blend_init=0.2, blend_min=0.003, blend_halve_every=12)
        if obj3 < 2.0 * n * fast_convolve(a_cur).max():
            a_cur = a_opt3
        
        # Round 4: MWD with low eta
        t4 = min(20.0, t_alloc * 0.15)
        a_opt4, obj4 = optimize_mwd(a_cur, t4, eta_init=0.01, eta_decay=0.9995)
        if obj4 < 2.0 * n * fast_convolve(a_cur).max():
            a_cur = a_opt4
        
        # Evaluate
        score = float(evaluate_sequence(a_cur.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_cur.copy()
            print(f"    Improved: {score:.6f}")
        else:
            print(f"    Score: {score:.6f} (no improvement)")
    
    # Phase 3: Simulated annealing + coordinate descent on best
    print("=== Phase 3: SA + CD Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        
        # Simulated annealing
        t_sa = min(80.0, remaining * 0.3)
        if t_sa > 10:
            a_sa, _ = simulated_annealing(best_sequence, t_sa, T_init=0.005, 
                                          T_decay=0.9998, move_size=0.01)
            score_sa = float(evaluate_sequence(a_sa.tolist()))
            if np.isfinite(score_sa) and score_sa < best_score:
                best_score = score_sa
                best_sequence = a_sa.copy()
                print(f"  SA: {score_sa:.6f}")
        
        # Coordinate descent (multiple rounds)
        remaining = deadline - time.monotonic() - 30.0
        for cd_round in range(3):
            if time.monotonic() > deadline - 20:
                break
            a_cd = coordinate_descent(best_sequence, max_rounds=30)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD round {cd_round}: {score_cd:.6f}")
            else:
                break
    
    # Phase 4: Final squeeze with very fine-tuning
    print("=== Phase 4: Final Squeeze ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        n_b = len(best_sequence)
        remaining = deadline - time.monotonic() - 5.0
        
        # Very small step MWD
        t_final = min(30.0, remaining * 0.4)
        if t_final > 5:
            a_f, _ = optimize_mwd(best_sequence, t_final, eta_init=0.003, eta_decay=0.9999)
            score_f = float(evaluate_sequence(a_f.tolist()))
            if np.isfinite(score_f) and score_f < best_score:
                best_score = score_f
                best_sequence = a_f.copy()
                print(f"  Final MWD: {score_f:.6f}")
        
        # Tiny perturbations + quick eval
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10:
            for ps in [0.002, 0.005, 0.001]:
                if time.monotonic() > deadline - 5:
                    break
                for _ in range(6):
                    if time.monotonic() > deadline - 3:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    # Quick MWD polish
                    a_q, _ = optimize_mwd(a_p, min(5.0, (deadline - time.monotonic() - 2) * 0.3),
                                         eta_init=0.005, eta_decay=0.999)
                    score_q = float(evaluate_sequence(a_q.tolist()))
                    if np.isfinite(score_q) and score_q < best_score:
                        best_score = score_q
                        best_sequence = a_q.copy()
                        print(f"  Perturb({ps}): {score_q:.6f}")
        
        # Final CD
        if time.monotonic() < deadline - 3:
            a_cd_final = coordinate_descent(best_sequence, max_rounds=20)
            score_final_cd = float(evaluate_sequence(a_cd_final.tolist()))
            if np.isfinite(score_final_cd) and score_final_cd < best_score:
                best_score = score_final_cd
                best_sequence = a_cd_final.copy()
                print(f"  Final CD: {score_final_cd:.6f}")
    
    # Ensure valid output
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(512) / 512
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
