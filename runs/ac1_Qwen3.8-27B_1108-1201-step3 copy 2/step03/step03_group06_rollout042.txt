"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using cyclical coordinate descent
with exact 1D subproblem solving, combined with global perturbations and
multi-length exploration.
"""

import numpy as np
import time

def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def compute_conv_and_stats(a):
    """Compute convolution, its max, and sum."""
    conv = fast_convolve(a)
    max_conv = conv.max()
    sum_a = a.sum()
    return conv, max_conv, sum_a

    if sum_a < 0.01:
