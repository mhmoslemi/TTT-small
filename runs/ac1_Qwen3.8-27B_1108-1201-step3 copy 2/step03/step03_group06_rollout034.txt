"""
Hybrid optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses improved LP with dual-guided active set management, smooth log-sum-exp
gradient with Adam optimizer, multi-scale n exploration, and structured
perturbations. Key differences from previous: Adam optimizer with adaptive
beta continuation, dual-based active set pruning, and n-value exploration
with interpolation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix: A[k, i] = a[k-i] if valid, else 0."""
    n = len(a)
    num_k = 2 * n - 1
    A = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A[k, idx] = a[k - idx]
    return A


def solve_lp_direction(a, M, conv=None, top_k=None):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for active k."""
    n = len(a)
    num_k = 2 * n - 1

    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        A_full = build_lp_matrix(a)
        A_ub = A_full[peaks]
        b_ub = np.full(len(peaks), M)
    else:
        A_ub = build_lp_matrix(a)
        b_ub = np.full(num_k, M)

    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def smooth_max_and_grad(a, beta):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # Gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    # Use FFT for efficiency
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)

    # Term 1: sum_k w[k] * a[k-i] (cross-correlation)
    # = ifft(conj(FFT(w)) * FFT(a)) ... let's use the formula directly
    # Actually: sum_k w[k]*a[k-i] for fixed i
    # Let j=k-i: sum_j w[j+i]*a[j] = cross-correlation of w and a at lag i
    # cross[i] = ifft(FFT(a) * conj(FFT(w)))[i]
    cross = np.fft.irfft(A * np.conj(W), fft_size)[:n]

    # Term 2: sum_k w[k] * a[i-k] = convolution of a and w at position i
    # = ifft(FFT(a) * FFT(w))[i]
    convaw = np.fft.irfft(A * W, fft_size)[:n]

    grad = cross + convaw
    smooth_val = conv.max() + np.log(w_sum) / beta
    return smooth_val, grad, conv, w


def adam_optimize(a0, time_limit, beta_init=200.0, beta_max=50000.0,
                  lr=0.01, beta_growth=1.005, seed=0, verbose=False):
    """Adam optimizer on the simplex using log-sum-exp smooth max."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    beta = beta_init
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        _, grad, conv, w = smooth_max_and_grad(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        t_count += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t_count)
        v_hat = v / (1 - b2 ** t_count)

        a_new = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slowly increase beta
            beta = min(beta_max, beta * beta_growth)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9
            if no_improve > 300:
                # Perturb and reset
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                lr = lr * 2.0
                beta = max(beta_init, beta * 0.5)
                no_improve = 0

        if verbose and t_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={t_count}: obj={best_obj:.6f}, beta={beta:.0f}, "
                  f"lr={lr:.5f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj

def subgradient_optimize(a0, time_limit, seed=0, verbose=False):
    """Subgradient method with momentum and adaptive step for the exact max."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    momentum = 0.9
    lr = 0.1
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        # Subgradient: pick the argmax position
        k_max = np.argmax(conv)
        # Gradient of (a*a)[k_max] wrt a
        grad = np.zeros(n)
        i_min = max(0, k_max - n + 1)
        i_max = min(n - 1, k_max)
        for i in range(i_min, i_max + 1):
            grad[i] += a[k_max - i]
            if k_max - i != i:
                grad[i] += a[i] if (i - k_max + n - 1 >= 0 and i - k_max <= n - 1 - k_max) else 0
        # Simpler: use the formula grad[i] = a[k-i] + a[i-k] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= k_max - i < n:
                grad[i] += a[k_max - i]
            if 0 <= i - k_max < n:
                grad[i] += a[i - k_max]

        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
            if no_improve > 200:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = 0.1
                momentum = 0.9
                no_improve = 0

        iter_count += 1

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def interpolate_sequence(parent, target_n):
    """Interpolate parent sequence to target_n length."""
    if len(parent) == target_n:
        return parent.copy()
    if len(parent) > target_n:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, target_n)
        interp = np.interp(x_new, x_old, parent)
    else:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, target_n)
        interp = np.interp(x_new, x_old, parent)
    interp = np.maximum(interp, 0.0)
    s = interp.sum()
    if s > 1e-15:
        interp /= s
    else:
        interp = np.ones(target_n) / target_n
    return interp


def generate_structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    # Hamming window
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1)) if n > 1 else np.ones(1)
    ham = np.maximum(ham, 1e-6)
    ham /= ham.sum()
    inits['hamming'] = ham

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Triangular (peaked at edges)
    tri = np.minimum(x + 1, n - x)
    tri /= tri.sum()
    inits['tri_edge'] = tri

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Random
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['random_unif'] = rng.dirichlet(np.ones(n) * 1.0)

    # From parent
    if parent is not None and len(parent) > 0:
        p = interpolate_sequence(parent, n)
        inits['parent'] = p

    return inits

            pert = np.cos(2 * np.pi * freq * x / n_best) * 0.005
            a_pert = project_simplex(best_sequence + pert)
            alloc = min(time_per_adam * 0.6, p3_budget - (time.monotonic() - t_p3) - 5)
            if alloc < 10:
                break
            a_opt, obj_opt = adam_optimize(a_pert, time_limit=alloc,
                                           beta_init=1000, lr=0.01,
                                           beta_growth=1.008, beta_max=50000,
                                           seed=100 + i)
            update_best(a_opt, f"Adam pert {i}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Subgradient + Fine LP (300s) ==========
    print("=== Phase 4: Subgradient + Fine LP ===")
    t_p4 = time.monotonic()
    p4_budget = 300.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Subgradient
        alloc = min(100.0, p4_budget / 3)
        a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=alloc, seed=42)
        update_best(a_opt, "subgrad")

        # Fine LP with small steps
        alloc = min(100.0, p4_budget / 3)
        a_opt, obj_opt = lp_optimize_v2(best_sequence, time_limit=alloc,
                                        step_init=0.3, top_k=None,
                                        relax=1.0, seed=777)
        update_best(a_opt, "fine LP")

        # Final Adam with very high beta
        alloc = min(100.0, p4_budget - (time.monotonic() - t_p4) - 5)
        if alloc > 20:
            a_opt, obj_opt = adam_optimize(best_sequence, time_limit=alloc,
                                           beta_init=5000, lr=0.002,
                                           beta_growth=1.002, beta_max=200000,
                                           seed=999)
            update_best(a_opt, "fine Adam")

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # ========== Phase 5: Final Polish (remaining time) ==========
    print("=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if best_sequence is not None and remaining > 30:
        n_best = len(best_sequence)

        # Tiny perturbations
        n_tries = min(30, int(remaining / 5))
        for i in range(n_tries):
            if time.monotonic() > deadline - 20:
                break
            for scale in [0.0005, 0.001, 0.002, 0.005]:
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence + noise)
                score = float(evaluate_sequence(a_pert.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_pert.copy()
                    print(f"  Perturb [{i}] s={scale}: score={best_score:.6f}")
                    break

        # Try n+delta by interpolation
        for dn in [16, -16, 32, -32]:
            if time.monotonic() > deadline - 30:
                break
            n_try = n_best + dn
            if n_try < 64 or n_try > 2048:
                continue
            a_interp = interpolate_sequence(best_sequence, n_try)
            # Quick LP to optimize
            alloc = min(30.0, deadline - time.monotonic() - 15)
            if alloc < 10:
                break
            a_opt, obj_opt = lp_optimize_v2(a_interp, time_limit=alloc,
                                            step_init=0.5, top_k=max(30, n_try // 8),
                                            seed=n_try)
            update_best(a_opt, f"n_shift {n_try}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
