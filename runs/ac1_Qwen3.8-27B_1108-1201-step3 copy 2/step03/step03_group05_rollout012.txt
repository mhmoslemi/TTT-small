def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, beta):
    """
    Log-sum-exp smoothed objective and gradient.
    f = logsumexp(beta * conv(a,a)) / beta
    grad = (1/beta) * d/d a logsumexp(beta * conv(a,a))
    """
    n = len(a)
    c = fast_convolve(a)
    scaled = beta * c
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    f = (scaled.max() + np.log(p_sum)) / beta
    
    if p_sum < 1e-300:
        # Degenerate case
        max_idx = np.argmax(c)
        grad = np.zeros(n)
        # grad of c_{max_idx} wrt a
        k = max_idx
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
        return f, grad
    
    q = p / p_sum
    
    # grad_i = (1/beta) * sum_k p_k * dc_k/da_i
    # dc_k/da_i = 2*a_{k-i} (with a_j=0 for j outside [0,n-1])
    # = (2/beta) * sum_k q_k * a_{k-i}
    # This is (2/beta) * cross_correlation(q, a)[i]
    
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    Q = np.fft.rfft(q, fft_size)
    cross = np.fft.irfft(Q * np.conj(A), fft_size)
    
    grad = (2.0 / beta) * cross[:n]
    
    return f, grad


def objective_wrapper(a, beta):
    """Unconstrained objective for optimizer (penalty for constraints)."""
    f, grad = smooth_obj_grad(a, beta)
    # Add penalty for negative values and sum != 1
    penalty = 1e6 * np.sum(np.maximum(-a, 0)**2)
    grad_penalty = -2e6 * np.minimum(a, 0)
    # For sum constraint, we'll handle via projection
    return f + penalty, grad + grad_penalty


def optimize_lbfgs(a0, time_limit, beta_start=15.0, beta_end=150.0, seed=0):
    """
    Optimize using L-BFGS-B with increasing beta schedule.
    a0: initial point on simplex
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = float('inf')
    
    # Multi-beta schedule
    n_phases = 5
    phase_budget = time_limit / n_phases
    
    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta = beta_start + (beta_end - beta_start) * phase / max(1, n_phases - 1)
        
        def obj_grad(x):
            # x is on the simplex (enforced by L-BFGS-B bounds + we project)
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-15:
                return 0, np.zeros(n)
            x = x / sx
            f, g = smooth_obj_grad(x, beta)
            # Adjust gradient for the normalization x/sx
            # Actually, since we're on the simplex, the gradient should be
            # projected: g_proj = g - mean(g) (to account for sum constraint)
            g_proj = g - g.mean()
            return f, g_proj
        
        # Use L-BFGS-B with bounds [0, 1] for each element
        # (sum constraint handled approximately)
        
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        phase_time = min(phase_budget, remaining)
        
        try:
            result = minimize(
                obj_grad, a,
                method='L-BFGS-B',
                jac=True,
                bounds=[(0, 1)] * n,
                options={
                    'maxiter': 200,
                    'maxfun': 500,
                    'ftol': 1e-12,
                    'gtol': 1e-8,
                    'maxls': 50,
                }
            )
            a_new = np.maximum(result.x, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
            score = fast_convolve(a_new).max() * 2 * n
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new
        except Exception as e:
            print(f"  L-BFGS error: {e}")
            break
        
        # Evaluate and track
        score = fast_convolve(a).max() * 2 * n
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Periodic perturbation
        if phase > 0 and phase % 2 == 0:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
    
    return best_a, best_score / (2 * n)  # Return max(conv) with sum=1


def subgradient_polish(a0, time_limit, seed=0):
    """
    Polish using subgradient method on the exact max.
    """

    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_maxc = fast_convolve(a).max()
    
    # Keep history of subgradients for bundle-like method
    history = []
    
    iters = 0
    step = 0.05
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        maxc = c.max()
        
        # Find active set (within 1% of max)
        threshold = maxc * 0.99
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient: average of gradients of active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)
        
        # Project gradient onto tangent space of simplex
        grad = grad - grad.mean()
        
        # Take step
        a_new = project_simplex(a - step * grad)
        
        # Check improvement
        c_new = fast_convolve(a_new)
        maxc_new = c_new.max()
        
        if maxc_new < best_maxc:
            best_maxc = maxc_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.1, 0.1)
        else:
            # Try smaller step
            a_new = project_simplex(a - step * 0.5 * grad)
            c_new = fast_convolve(a_new)
            maxc_new = c_new.max()
            if maxc_new < maxc:
                a = a_new
            else:
                step *= 0.8
            
            # Periodic perturbation
            iters += 1
            if iters % 100 == 0:
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                step = 0.05
    
    return best_a, best_maxc


def coordinate_descent(a0, time_limit, seed=0):
    """
    Coordinate descent: optimize one coordinate at a time.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_maxc = fast_convolve(a).max()
    
    order = list(range(n))
    
    while time.monotonic() - t0 < time_limit:
        np.random.seed(rng.integers(0, 2**31))
        np.random.shuffle(order)
        
        improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Optimize a[i] given all other coordinates
            # c_k as function of a[i]:
            # c_k = a[i]*a[k-i] + a[k-i]*a[i] + (other terms) for k != 2i
            # c_k = a[i]^2 + (other terms) for k = 2i
            
            save = a[i]
            
            # Binary search for optimal a[i]
            # The max of c_k as a function of a[i] is convex (max of quadratics/linears)
            # So we can ternary search or binary search on the derivative
            
            lo = 0.0
            hi = min(1.0, 2 * save + 0.1)  # reasonable upper bound
            
            for _ in range(30):
                mid1 = lo + (hi - lo) / 3
                mid2 = hi - (hi - lo) / 3
                
                a[i] = mid1
                # Renormalize
                s = a.sum()
                a_norm = a / s
                c1 = fast_convolve(a_norm).max()
                
                a[i] = mid2
                s = a.sum()
                a_norm = a / s
                c2 = fast_convolve(a_norm).max()
                
                if c1 < c2:
                    hi = mid2
                else:
                    lo = mid1
            
            a[i] = (lo + hi) / 2
            s = a.sum()
            a = a / s
            cur_maxc = fast_convolve(a).max()
            
            if cur_maxc < best_maxc:
                best_maxc = cur_maxc
                best_a = a.copy()
                improved = True
            else:
                a[i] = save  # revert
        
        if not improved:
            # Perturb
            idx = rng.integers(0, n)
            a[idx] = max(0, a[idx] * (1 + rng.standard_normal() * 0.1))
            s = a.sum()
            if s > 1e-10:
                a = a / s
    
    return best_a, best_maxc


def generate_fattop_window(n):
    """Generate a flat-top-like window of length n."""
    x = np.arange(n)
    # Use a sum of cosines (Harris-like)
    w = (0.35875 - 0.48829 * np.cos(2 * np.pi * x / (n - 1))
         + 0.14128 * np.cos(4 * np.pi * x / (n - 1))
         - 0.01168 * np.cos(6 * np.pi * x / (n - 1)))
    w = np.maximum(w, 0)
    w = w / w.sum()
    return w


def generate_initialization(n, parent, rng, init_type='fattop'):
    """Generate an initialization of length n."""
    if init_type == 'fattop':
        a = generate_fattop_window(n)
    elif init_type == 'raised_cos':
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif init_type == 'gauss':
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
        a = a / a.sum()
    elif init_type == 'bimodal':
        edge = max(2, n // 4)
        a = np.zeros(n)
        a[:edge] = 1.0
        a[-edge:] = 1.0
        a = a / a.sum()
    elif init_type == 'parent_interp':
        if parent is not None and len(parent) > 0:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                p = p / ps
                a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                a = np.maximum(a, 0)
                a = a / a.sum()
                return a
        a = generate_fattop_window(n)
    elif init_type == 'random':
        a = rng.dirichlet(np.ones(n) * 0.8)
    else:
        a = np.ones(n) / n
    
    # Ensure palindromic
    a = (a + a[::-1]) / 2
    a = a / a.sum()
    return a


def optimize_smoothed_nesterov(a0, time_limit, beta_init=20.0, beta_end=200.0,
                                  lr_init=0.01, momentum=0.9, seed=0):
    """
    Nesterov accelerated projected gradient on smoothed objective.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    best_a = a.copy()
    best_maxc = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        lr = lr_init * (1.0 - 0.8 * frac)
        lr = max(lr, 0.0005)
        
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        f, g = smooth_obj_grad(y, beta)
        g = g - g.mean()  # Project to simplex tangent
        
        v = momentum * v - lr * g
        a = project_simplex(a + v)
        
        # Enforce palindrome (soft)
        a = (a + a[::-1]) / 2
        a = project_simplex(a)
        
        cur_maxc = fast_convolve(a).max()
        
        if cur_maxc < best_maxc - 1e-10:
            best_maxc = cur_maxc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 200:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            a = (a + a[::-1]) / 2
            a = project_simplex(a)
            v *= 0.2
            no_improve = 0
    
    return best_a, best_maxc
