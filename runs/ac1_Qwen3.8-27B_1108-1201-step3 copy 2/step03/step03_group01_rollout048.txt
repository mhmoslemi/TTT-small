"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using minimax LP direction finding,
coordinate descent refinement, and adaptive restarts.
Key improvements over previous approach:
1. Uses minimax LP formulation (directly minimizes peak cross-convolution).
2. Adds coordinate descent refinement for fine-tuning.
3. Uses dual variables for smarter line search.
4. Employs multiple strategies and restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def solve_minimax_lp(a, A_ub=None):
    """
    Solve: minimize M s.t. (a*g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns (g, M_new, success).
    """
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize: [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints:
    # (a*g)[k] - M <= 0 for all k  =>  A_ub @ [g, M] <= 0
    # sum(g) = 1  =>  equality constraint
    # g >= 0, M unrestricted (but M >= 0 implicitly)
    
    A_min = np.hstack([A_ub, -np.ones((2*n-1, 1))])
    b_min = np.zeros(2*n-1)
    
    A_eq = np.hstack([np.ones(n), [[0]]])
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, M >= 0
    
    result = linprog(c, A_ub=A_min, b_ub=b_min, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        M_new = result.x[-1]
        return g, M_new, True
    return None, None, False

def compute_objective(a):
    """Compute the objective value."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    return 2.0 * n * max_c / (sum_a ** 2)

def line_search(a, d, n, current_obj, max_evals=20):
    """Adaptive line search."""
    best_t = 0.0
    best_obj = current_obj
    
    candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.7, 0.8, 0.9]
    
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try = compute_objective(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t
    
    # Refine
    if 0.005 < best_t < 0.995:
        step = 0.05
        for _ in range(4):
            improved = False
            for dt in [-step, step]:
                t_try = best_t + dt
                if t_try <= 0 or t_try >= 1:
                    continue
                a_try = a + t_try * d
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                obj_try = compute_objective(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_t = t_try
                    improved = True
            if not improved:
                break
            step *= 0.4
    
    return best_t, best_obj

def coordinate_descent(a, n, max_sweeps=5, tol=1e-12):
    """
    Coordinate descent: for each position i, try to optimally adjust a[i]
    to reduce the max of the self-convolution.
    """
    best_a = a.copy()
    best_obj = compute_objective(a)
    a = a.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        conv = fast_convolve(a)
        
        for i in range(n):
            # Try increasing and decreasing a[i]
            # Current contribution of a[i] to conv[k]:
            # conv[k] includes a[i]*a[k-i] (and a[k-i]*a[i] if k-i != i)
            
            # Try a[i] = 0
            a_old = a[i]
            if a_old < 1e-15:
                continue
            
            best_val = a_old
            best_local_obj = compute_objective(a)
            
            # Try several values of a[i]
            for frac in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5]:
                a_try = a.copy()
                a_try[i] = a_old * frac
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                obj_try = compute_objective(a_try)
                if obj_try < best_local_obj - 1e-13:
                    best_local_obj = obj_try
                    best_val = a_try[i]
            
            if best_val != a_old:
                a = a.copy()
                a[i] = best_val
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                improved = True
        
        if not improved:
            break
        
        obj = compute_objective(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

def lp_optimize_v2(a0, time_limit, verbose=False):
    """
    LP-based optimization using minimax formulation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 100
    
    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        
        g, M_new, success = solve_minimax_lp(a, A_ub)
        
        if not success or g is None:
            no_improve += 10
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            
            if norm_d < 1e-10:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, n, best_obj)
                
                if obj_best < best_obj - 1e-13:
                    a_new = a + t_best * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                    
                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        iteration += 1
        
        if no_improve >= max_no_improve:
            # Perturb and restart
            rng = np.random.default_rng(iteration * 7 + 13)
            noise = rng.standard_normal(n) * 0.005
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_no_improve = min(300, max_no_improve + 50)
            
            if max_no_improve > 300:
                break
    
    return best_a, best_obj

def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=50, temp=0.0002):
    """Projected subgradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    batch = 20
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find top peaks
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum
            
            # Compute subgradient
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[i] * 2.0 * a[k - idx]
            
            v = momentum * v - lr * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()
            
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        lr *= 0.998
        temp *= 1.003
        momentum = min(0.97, momentum + 0.0002)
        
        if no_improve >= 500:
            break
    
    return best_a, best_obj
