import numpy as np

from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def solve_min_max_lp(a, top_k=None, conv=None):
    """
    Solve: min t s.t. (a*b)[k] <= t for all k (or top-k), sum(b)=1, b>=0
    Returns (b, t) or (None, None)
    """
    n = len(a)
    num_k = 2 * n - 1
    
    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        k_indices = peaks
    else:
        k_indices = np.arange(num_k)
    
    num_constraints = len(k_indices)
    
    # Variables: b_0, ..., b_{n-1}, t
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: (a*b)[k] - t <= 0 for each k
    A_ub = np.zeros((num_constraints, n + 1))
    for row, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[row, idx] = a[k - idx]
        A_ub[row, -1] = -1.0  # -t
    b_ub = np.zeros(num_constraints)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # b_i >= 0, t free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def line_search_lp(a, b, n_evals=10):

"""
Multi-phase LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses min-max LP (best response) formulation, adaptive line search, diverse exploration

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_constraint_rows(a, k_indices):
    """Build rows of the constraint matrix for given k indices."""
    n = len(a)
    num_rows = len(k_indices)
    A = np.zeros((num_rows, n))
    for row, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A[row, idx] = a[k - idx]
    return A


def solve_minmax_lp(a, top_k=None, conv=None):
    """
    Solve: min t s.t. (a*b)[k] <= t for all k (or top-k), sum(b)=1, b>=0.
    Returns (b, t) or (None, None).
    """
    n = len(a)
    num_k = 2 * n - 1

    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        k_indices = peaks
    else:
        k_indices = np.arange(num_k)

    num_constraints = len(k_indices)

    # Variables: b_0,...,b_{n-1}, t
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = build_constraint_rows(a, k_indices)
    # Add -t column
    A_full = np.hstack([A_ub, -np.ones((num_constraints, 1))])
    b_ub = np.zeros(num_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    try:
        res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            b = np.maximum(res.x[:n], 0)
            return b, res.x[-1]
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(a, M, top_k=None, conv=None):
    """
    Solve: max sum(b) s.t. (a*b)[k] <= M for all k (or top-k), b>=0.
    Returns b or None.
    """

    num_k = 2 * n - 1

    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        k_indices = peaks
    else:
        k_indices = np.arange(num_k)

    c = -np.ones(n)
    A_ub = build_constraint_rows(a, k_indices)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_direction(a, method, top_k=None, relax=1.0):
    """Get LP direction. Returns (b, conv) where b is on simplex or None."""
    n = len(a)
    conv = fast_convolve(a)

        if b is None:
            return None, conv
        b_sum = b.sum()

            return None, conv
        return b / b_sum, conv

        if b is None:
            return None, conv
        b_sum = b.sum()
        if b_sum < 1e-15:
            return None, conv
        return b / b_sum, conv
    return None, conv


def line_search(a, b, n_trials=12):
    """Line search along direction (b-a). Returns best alpha and objective."""
    n = len(a)
    best_alpha = 0.0
    best_obj = 2.0 * n * fast_convolve(a).max()

    # Include 0 in the search
    alphas = np.concatenate([[0.0], np.linspace(0.05, 1.0, n_trials)])
    for alpha in alphas:
        a_trial = (1.0 - alpha) * a + alpha * b
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()

            continue
        a_trial = a_trial / s
        obj = 2.0 * n * fast_convolve(a_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    return best_alpha, best_obj

    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def coordinate_descent_refine(a, time_limit, n_passes=200):
    """Coordinate descent: try adjusting one coordinate at a time."""
    n = len(a)
    a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for _ in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        # Randomly pick a coordinate to update
        i = rng.integers(0, n)
        other_sum = 1.0 - a[i]
        a_max = max(0.0, other_sum)

        if a_max < 1e-15:
            continue

        # Try several candidate values
        current_obj = 2.0 * n * fast_convolve(a).max()
        candidates = [a[i], 0.0, a_max, a_max * 0.5, a[i] * 0.7, a[i] * 1.3,
                      a[i] + 0.01 * a_max, a[i] - 0.01 * a_max]
        candidates = [c for c in candidates if -1e-10 <= c <= a_max + 1e-10]
        candidates = [max(0.0, min(c, a_max)) for c in candidates]

        best_val = a[i]
        best_obj = current_obj

        for cv in candidates:
            a[i] = cv
            s = a.sum()
            if s < 1e-15:
                a[i] = best_val
                continue
            a_norm = a / s
            obj = 2.0 * n * fast_convolve(a_norm).max()
            if obj < best_obj:
                best_obj = obj
                best_val = cv
        a[i] = best_val

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse structured initializations (all sum to 1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Cosine window
    cos_w = np.cos(np.pi * x / (n - 1)) ** 2 if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Hann window
    if n > 2:
        hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 0.01)
        hann /= hann.sum()
        inits['hann'] = hann

    # Gaussians
    for sf in [0.08, 0.12, 0.15, 0.2, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Bimodal (edges)
    for frac in [0.15, 0.25, 0.35]:
        bi = np.zeros(n)
        edge = max(1, int(n * frac))
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        bi /= bi.sum()
        inits[f'bimodal_{frac}'] = bi

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_rev'] = ramp[::-1].copy()

    # Triangle
    tri = np.minimum(x, n - 1 - x).astype(np.float64) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Flat top (plateau in middle)
    for frac in [0.3, 0.5, 0.7]:
        ft = np.zeros(n)
        m = int(n * (1 - frac) / 2)
        ft[m:n - m] = 1.0
        if ft.sum() > 0:
            ft /= ft.sum()
            inits[f'flat_{frac}'] = ft

    # Random
    inits['rand_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['rand_uniform'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_conc'] = rng.dirichlet(np.ones(n) * 5.0)

    # Piecewise linear
    for _ in range(2):
        pts = np.sort(rng.uniform(0, n, size=4))
        vals = rng.uniform(0.1, 1.0, size=4)
        pl = np.interp(x, pts, vals)
        pl = np.maximum(pl, 0.01)
        pl /= pl.sum()
        inits[f'piecewise_{_}'] = pl

    # From parent
    if parent is not None:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
            # Also try windowed version
            w = np.ones(n)
            if n > 10:
                w[:n//10] = 0.5
                w[-n//10:] = 0.5
            pp2 = pp * w
            ps2 = pp2.sum()
            if ps2 > 1e-10:
                inits['parent_win'] = pp2 / ps2
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase LP-based optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
    Uses min-max LP best-response, adaptive line search, diverse exploration
    across lengths, coordinate descent refinement, and progressive constraint tightening.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Broad Exploration ==========
    print("=== Phase 1: Broad Exploration ===")
    t_start = time.monotonic()
    phase1_time = 180.0

    lengths = [128, 192, 256, 320, 384, 448, 512]
    scan_results = []

    for li, n in enumerate(lengths):
        if time.monotonic() + 15 > t_start + phase1_time:
            break
        inits = generate_initializations(n, rng, parent)

        # Prioritize diverse types
        priority = (['parent', 'parent_ds', 'parent_up', 'parent_win',
                     'bimodal_0.25', 'bimodal_0.35', 'uniform', 'cosine',
                     'gauss_0.15', 'gauss_0.25', 'hann', 'flat_0.5',
                     'rand_sparse', 'triangle', 'ramp'])
        init_names = [nm for nm in priority if nm in inits]
        init_names = init_names[:5]

        for name in init_names:
            if time.monotonic() + 3 > t_start + phase1_time:
                break
            t_alloc = min(12.0, max(3.0, t_start + phase1_time - time.monotonic() - 10))

            a0 = inits[name]
            a_opt, obj_opt = deep_optimize(a0, time_limit=t_alloc,
                                           method='minmax',
                                           top_k=max(30, n // 8),
                                           step_init=1.0,
                                           seed=n * 31 + hash(name) % 997,
                                           adapt_topk=True)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    if li % 2 == 0:
                        print(f"  n={n}, {name}: score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.1f}s). Best: {best_score:.6f}")

    # ========== Phase 2: Deep Optimization ==========
    print("=== Phase 2: Deep Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = min(600.0, max(60.0, deadline - time.monotonic() - 120.0))

    # Build pool of starting points
    deep_starts = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results[:12]:
        key = (n, name)
        if key not in seen_keys:
            deep_starts.append((n, a_opt.copy(), name))
            seen_keys.add(key)
        if len(deep_starts) >= 8:
            break

    if best_sequence is not None:
        deep_starts.append((len(best_sequence), best_sequence.copy(), 'best'))

    # Add fresh starts at promising lengths
    for n_fresh in [320, 384, 512]:
        if time.monotonic() < t_start + phase1_time + phase2_time - 60:
            inits_f = generate_initializations(n_fresh, rng, parent)
            for nm in ['bimodal_0.35', 'gauss_0.2', 'rand_sparse', 'flat_0.5']:
                if nm in inits_f and (n_fresh, nm) not in seen_keys:
                    deep_starts.append((n_fresh, inits_f[nm].copy(), nm))
                    seen_keys.add((n_fresh, nm))
                if len(deep_starts) >= 14:
                    break
        if len(deep_starts) >= 14:
            break

    deep_starts = deep_starts[:14]
    if not deep_starts:
        deep_starts = [(256, np.ones(256) / 256, 'init')]

    # Run deep optimization with various configs
    total_runs = len(deep_starts) * 2  # minmax + maxsum
    time_per_run = max(15.0, phase2_time / max(1, total_runs))

    run_i = 0
    for i, (n_s, a_s, name) in enumerate(deep_starts):
        for method in ['minmax', 'maxsum']:
            if time.monotonic() + 15 > phase2_start + phase2_time:
                break
            t_alloc = min(45.0, phase2_start + phase2_time - time.monotonic() - 15)
            if t_alloc < 12:
                break

            top_k = max(40, n_s // 6)
            a_opt, obj_opt = deep_optimize(a_s, time_limit=t_alloc,
                                           method=method, top_k=top_k,
                                           step_init=1.0,
                                           seed=run_i * 13 + 7,
                                           adapt_topk=True)

            score = float(evaluate_sequence(a_opt.tolist()))
            run_i += 1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep[{run_i}] n={n_s} {name} {method}: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-phase2_start:.1f}s). Best: {best_score:.6f}")

    # ========== Phase 3: Intensive Refinement on Best ==========
    print("=== Phase 3: Intensive Refinement ===")
    phase3_start = time.monotonic()
    phase3_time = min(400.0, max(30.0, deadline - time.monotonic() - 60.0))

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Multiple LP configurations
        configs = [
            {'method': 'minmax', 'top_k': None, 'step': 0.8},
            {'method': 'minmax', 'top_k': max(80, n_best // 3), 'step': 1.0},
            {'method': 'minmax', 'top_k': max(120, n_best // 2), 'step': 0.9},
            {'method': 'maxsum', 'top_k': max(60, n_best // 4), 'step': 1.0},
            {'method': 'minmax', 'top_k': max(50, n_best // 5), 'step': 0.7},
            {'method': 'maxsum', 'top_k': None, 'step': 0.8},
        ]

        time_per_cfg = phase3_time / (len(configs) + 4)

        for ci, cfg in enumerate(configs):
            if time.monotonic() + 10 > phase3_start + phase3_time:
                break
            t_alloc = min(time_per_cfg, phase3_start + phase3_time - time.monotonic() - 10)
            if t_alloc < 8:
                break

            a_opt, obj_opt = deep_optimize(best_sequence, time_limit=t_alloc,
                                           method=cfg['method'], top_k=cfg['top_k'],
                                           step_init=cfg['step'], seed=200 + ci,
                                           adapt_topk=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Refine[{ci}] {cfg['method']}: {score:.6f}")

        # Coordinate descent refinement
        if time.monotonic() < phase3_start + phase3_time - 20:
            t_cd = min(40.0, phase3_start + phase3_time - time.monotonic() - 10)
            if t_cd > 5:
                a_cd = coordinate_descent_refine(best_sequence, time_limit=t_cd)
                score = float(evaluate_sequence(a_cd.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_cd.copy()
                    print(f"  CoordDesc: {score:.6f}")
                # Follow up with LP
                if time.monotonic() < phase3_start + phase3_time - 15:
                    t_lp = min(25.0, phase3_start + phase3_time - time.monotonic() - 5)
                    if t_lp > 5:
                        a_opt, _ = deep_optimize(a_cd, time_limit=t_lp,
                                                 method='minmax',
                                                 top_k=max(60, n_best // 5),
                                                 step_init=0.7, seed=500)
                        score = float(evaluate_sequence(a_opt.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_opt.copy()
                            print(f"  CD+LP: {score:.6f}")

        # Try length changes (up/down sampling)
        if time.monotonic() < phase3_start + phase3_time - 30:
            for scale_factor in [0.75, 1.25, 1.5]:
                n_new = max(64, int(n_best * scale_factor))
                a_interp = np.interp(np.linspace(0, 1, n_new),
                                     np.linspace(0, 1, n_best), best_sequence)
                a_interp = np.maximum(a_interp, 0)
                s = a_interp.sum()
                if s > 1e-15:
                    a_interp = a_interp / s
                t_alloc = min(30.0, phase3_start + phase3_time - time.monotonic() - 10)
                if t_alloc > 5:
                    a_opt, _ = deep_optimize(a_interp, time_limit=t_alloc,
                                             method='minmax',
                                             top_k=max(30, n_new // 8),
                                             step_init=1.0, seed=600)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Resample {scale_factor} -> n={n_new}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-phase3_start:.1f}s). Best: {best_score:.6f}")

    # ========== Phase 4: Final Polish ==========
    print("=== Phase 4: Final Polish ===")
    phase4_start = time.monotonic()
    remaining = deadline - time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Full constraint LP
        if remaining > 40:
            t_alloc = min(remaining * 0.35, 80.0)
            a_opt, _ = deep_optimize(best_sequence, time_limit=t_alloc,
                                     method='minmax', top_k=None,
                                     step_init=0.5, seed=999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Full LP: {score:.6f}")

        # Tiny perturbations + quick re-optimize
        if remaining > 30:
            for _ in range(min(8, int((remaining - 15) / 4))):
                if time.monotonic() > deadline - 20:
                    break
                improved = False
                for scale in [0.0005, 0.002, 0.005, 0.01]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        improved = True
                        print(f"  Perturb s={scale}: {best_score:.6f}")
                        break
                if improved and remaining > 25:
                    t_tiny = min(10.0, deadline - time.monotonic() - 10)
                    if t_tiny > 3:
                        a_opt, _ = deep_optimize(best_sequence, time_limit=t_tiny,
                                                 method='minmax',
                                                 top_k=max(40, n_best // 8),
                                                 step_init=0.5, seed=888)
                        score = float(evaluate_sequence(a_opt.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_opt.copy()

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
