def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def lp_solve_direction(a, M, top_k=50, conv=None):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for top-k k."""
    n = len(a)
    num_k = 2 * n - 1
    
    if conv is None:
        conv = fast_convolve(a)
    
    if top_k is not None and top_k < num_k:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(num_k)
    
    num_constraints = len(peaks)
    A_ub = np.zeros((num_constraints, n))
    for ci, k in enumerate(peaks):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[ci, idx] = a[k - idx]
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(num_constraints, M)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                               'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def line_search(a, direction, n, max_iter=20):
    """Find optimal step size in direction to minimize obj = 2n*max(conv(a_t))."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return 0.0
    
    a_norm = a / s
    d_norm = direction / direction.sum() if direction.sum() > 1e-15 else direction
    
    best_t = 0.0
    best_obj = 2.0 * n * fast_convolve(a_norm).max()
    
    for i in range(max_iter):
        t = (i + 1) / max_iter
        a_t = (1 - t) * a_norm + t * d_norm
        a_t = np.maximum(a_t, 0)
        st = a_t.sum()
        if st < 1e-15:
            continue
        a_t = a_t / st
        obj = 2.0 * n * fast_convolve(a_t).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t


def generate_initializations(n, parent, rng):

    inits['bimodal'] = bi

    # Random

    # From parent

                    inits['parent_interp'] = pp / pp_s
    
    return inits


def optimize_lp(a0, time_limit, top_k=50, step_init=0.8, seed=0):
    """LP-based optimization. Returns (best_a, best_obj) with sum(a)=1."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        b = lp_solve_direction(a, M, top_k=top_k, conv=conv)
        if b is None:
            b = lp_solve_direction(a, M, top_k=None, conv=conv)
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        # Line search
        t_opt = line_search(a, b, n, max_iter=10)
        if t_opt < 0.05:
            no_improve += 1
            if no_improve > 10:
                # Perturb
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                step = step_init
                no_improve = 0
            continue
        
        a_new = (1.0 - t_opt) * a + t_opt * b
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                no_improve = 0
            elif no_improve > 5:
                step *= 0.8
    
    return best_a, best_obj


def optimize_subgrad(a0, time_limit, beta=500.0, lr=0.02, momentum=0.9, seed=0):
    """Subgradient descent with momentum on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        g, conv, w = subgradient(a, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr = max(0.001, lr * 0.85)
                v *= 0.5
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init if 'lr_init' in dir() else 0.02
                beta = max(100, beta * 0.7)
                no_improve = 0
        
        beta = min(10000.0, beta * 1.002)
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP exploration, deep LP, subgradient polishing,
    hybrid alternating, and final intensive polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass
    
    def try_sequence(a):
        nonlocal best_sequence, best_score
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")
    
    # ========== Phase 1: Quick Exploration ==========
    print("=== Phase 1: Quick Exploration ===")
    t1_start = time.monotonic()
    t1_budget = 180.0
    
    n_target = 320
    if parent is not None:
        n_target = len(parent)
    
    lengths_to_try = [n_target]
    if n_target <= 400:
        lengths_to_try.extend([n_target - 32, n_target + 32])
    else:
        lengths_to_try.extend([256, 384])
    lengths_to_try = [max(64, min(512, n)) for n in lengths_to_try]
    lengths_to_try = list(dict.fromkeys(lengths_to_try))  # dedupe
    
    scan_candidates = []
    
    for n in lengths_to_try:
        if time.monotonic() - t1_start > t1_budget - 30:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        
        # Prioritize
        priority = ['parent_crop', 'parent_interp', 'bimodal', 'uniform',
                    'random1', 'cosine', 'bump_left', 'random2']
        init_names = [nm for nm in priority if nm in init_names] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]
        
        time_per = max(5.0, (t1_budget - 30) / max(1, len(lengths_to_try)) / max(1, len(init_names)))
        
        for name in init_names:
            if time.monotonic() - t1_start > t1_budget - 10:
                break
            a0 = inits[name]
            t_alloc = min(time_per, t1_budget - (time.monotonic() - t1_start) - 5)
            if t_alloc < 3:
                break
            
            a_opt, obj_opt = optimize_lp(a0, time_limit=t_alloc, top_k=40,
                                         step_init=0.8, seed=n + hash(name) % 1000)
            try_sequence(a_opt)
            scan_candidates.append((obj_opt, n, a_opt.copy()))
    
    scan_candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")
    
    # ========== Phase 2: Deep LP Optimization ==========
    print("=== Phase 2: Deep LP Optimization ===")
    t2_start = time.monotonic()
    t2_budget = 700.0
    
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    n_best = len(best_sequence)
    
    # Starting points
    deep_starts = []
    if best_sequence is not None:
        deep_starts.append(best_sequence.copy() / best_sequence.sum())
    for obj, n, a_opt in scan_candidates[:4]:
        if n == n_best:
            deep_starts.append(a_opt / a_opt.sum())
    
    # Add some fresh starts
    inits_best = generate_initializations(n_best, parent, rng)
    for extra in ['random1', 'bimodal']:
        if extra in inits_best:
            deep_starts.append(inits_best[extra])
    
    deep_starts = deep_starts[:5]
    
    # Different LP configurations
    configs = [
        {'top_k': 40, 'step_init': 1.0},
        {'top_k': 60, 'step_init': 0.8},
        {'top_k': 30, 'step_init': 1.0},
        {'top_k': 80, 'step_init': 0.9},
        {'top_k': None, 'step_init': 0.7},
    ]
    
    total_runs = len(deep_starts) * len(configs)
    time_per_run = t2_budget / max(1, total_runs)
    time_per_run = max(10.0, time_per_run)
    
    run_count = 0
    for i, a_start in enumerate(deep_starts):
        for j, cfg in enumerate(configs):
            if time.monotonic() - t2_start > t2_budget - 15:
                break
            t_alloc = min(time_per_run, t2_budget - (time.monotonic() - t2_start) - 10)
            if t_alloc < 8:
                break
            
            a_opt, obj_opt = optimize_lp(a_start, time_limit=t_alloc,
                                         top_k=cfg['top_k'], step_init=cfg['step_init'],
                                         seed=run_count * 13 + 7)
            try_sequence(a_opt)
            run_count += 1
    
    print(f"Phase 2 done. Best: {best_score:.6f}")
    
    # ========== Phase 3: Subgradient Polishing ==========
    print("=== Phase 3: Subgradient Polishing ===")
    t3_start = time.monotonic()
    t3_budget = 300.0
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        a_norm = best_sequence / best_sequence.sum()
        
        grad_configs = [
            {'beta': 300, 'lr': 0.03, 'mom': 0.9},
            {'beta': 800, 'lr': 0.015, 'mom': 0.92},
            {'beta': 1500, 'lr': 0.008, 'mom': 0.94},
            {'beta': 2000, 'lr': 0.005, 'mom': 0.95},
        ]
        
        time_per = t3_budget / max(1, len(grad_configs) + 2)
        
        for i, gc in enumerate(grad_configs):
            if time.monotonic() - t3_start > t3_budget - 10:
                break
            t_alloc = min(time_per, t3_budget - (time.monotonic() - t3_start) - 5)
            if t_alloc < 10:
                break
            
            a_opt, obj_opt = optimize_subgrad(a_norm, time_limit=t_alloc,
                                              beta=gc['beta'], lr=gc['lr'],
                                              momentum=gc['mom'], seed=i * 31 + 3)
            try_sequence(a_opt)
        
        # Perturbed starts
        for i in range(2):
            if time.monotonic() - t3_start > t3_budget - 10:
                break
            noise = rng.standard_normal(n_best) * 0.003
            a_pert = project_simplex(a_norm + noise)
            t_alloc = min(time_per / 2, t3_budget - (time.monotonic() - t3_start) - 5)
            if t_alloc < 5:
                break
            a_opt, obj_opt = optimize_subgrad(a_pert, time_limit=t_alloc,
                                              beta=1000, lr=0.01, momentum=0.93,
                                              seed=100 + i)
            try_sequence(a_opt)
    
    print(f"Phase 3 done. Best: {best_score:.6f}")
    
    # ========== Phase 4: Hybrid Alternating ==========
    print("=== Phase 4: Hybrid Alternating ===")
    t4_start = time.monotonic()
    t4_budget = 350.0
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence / best_sequence.sum()
        best_obj_phase = 2.0 * n_best * fast_convolve(a).max()
        best_a_phase = a.copy()
        v = np.zeros(n_best)
        
        t4_t0 = time.monotonic()
        while time.monotonic() - t4_t0 < t4_budget:
            conv = fast_convolve(a)
            M = conv.max()
            
            # LP step
            b = lp_solve_direction(a, M, top_k=50, conv=conv)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b_norm = b / b_sum
                    t_opt = line_search(a, b_norm, n_best, max_iter=8)
                    if t_opt > 0.05:
                        a = (1 - t_opt) * a + t_opt * b_norm
                        a = np.maximum(a, 0)
                        s = a.sum()
                        if s > 1e-15:
                            a = a / s
                    else:
                        a = 0.9 * a + 0.1 * b_norm
                        a = np.maximum(a, 0)
                        s = a.sum()
                        if s > 1e-15:
                            a = a / s
            
            # Subgradient step
            g, conv_g, w = subgradient(a, beta=2000)
            gn = np.linalg.norm(g)
            if gn > 1e-15:
                g = g / gn
                v = 0.92 * v - 0.005 * g
                a = project_simplex(a + v)
            
            conv = fast_convolve(a)
            obj = 2.0 * n_best * conv.max()
            if obj < best_obj_phase - 1e-14:
                best_obj_phase = obj
                best_a_phase = a.copy()
                try_sequence(a)
        
        # Try the phase best
        if best_a_phase is not None:
            try_sequence(best_a_phase)
    
    print(f"Phase 4 done. Best: {best_score:.6f}")
    
    # ========== Phase 5: Final Intensive Polish ==========
    print("=== Phase 5: Final Intensive Polish ===")
    t5_start = time.monotonic()
    t5_budget = deadline - time.monotonic()
    
    if best_sequence is not None and t5_budget > 20:
        n_best = len(best_sequence)
        a = best_sequence / best_sequence.sum()
        
        # 5a: Very fine LP
        t_alloc = min(t5_budget * 0.3, 80.0)
        if t_alloc > 10:
            a_opt, obj_opt = optimize_lp(a, time_limit=t_alloc, top_k=60,
                                         step_init=0.3, seed=999)
            try_sequence(a_opt)
        
        # 5b: Very fine subgradient
        t_alloc = min(t5_budget * 0.25, 60.0)
        if t_alloc > 10:
            a_opt, obj_opt = optimize_subgrad(a, time_limit=t_alloc,
                                              beta=5000, lr=0.003, momentum=0.96,
                                              seed=777)
            try_sequence(a_opt)
        
        # 5c: Tiny perturbations
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 10:
            n_tries = int(remaining / 3)
            n_tries = min(n_tries, 50)
            a_norm = best_sequence / best_sequence.sum()
            for i in range(n_tries):
                if time.monotonic() > deadline - 25:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(a_norm + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        a_norm = a_pert / a_pert.sum()
                        print(f"  Polish perturb [{i}] s={scale}: {score_pert:.6f}")
                        break
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL SCORE: {final_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
