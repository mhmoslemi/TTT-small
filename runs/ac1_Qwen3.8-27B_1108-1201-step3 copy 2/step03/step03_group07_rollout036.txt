"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Smooth L-BFGS-B optimization with adaptive beta schedule
2. Nesterov-accelerated subgradient method for final polishing
3. Multi-length search with diverse initializations
4. Active-set KKT refinement for high-precision optimization
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = np.max(x)
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_vals / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_fft(a, w):
    """
    Compute gradient of sum_k w[k]*(a*a)[k] w.r.t. a using FFT.
    grad[i] = 2 * sum_j a[j] * w[i+j] * I(i+j < len(w))
    This equals 2 * cross_correlation(a, w, lag=i)
    Using: cross_corr(a,w,lag) = linear_conv(flip(a), w)[n-1+lag]
    """
    n = len(a)
    m = len(w)
    total_len = n + m - 1
    N = 1 << (total_len).bit_length()
    flip_a = a[::-1]
    FA = fft(flip_a, N)
    W = fft(w, N)
    conv_result = np.real(ifft(FA * W, N))
    # conv_result[n-1 + i] = cross_correlation at lag i, for i = 0, ..., n-1
    # But we need to be careful: the linear convolution has length n + m - 1
    # cross_corr at lag i = conv_result[n-1+i] for i = 0, ..., n-1
    # (since flip_a has length n, and w has length m, the conv has length n+m-1)
    grad = 2.0 * conv_result[n - 1: n - 1 + n]
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_exact(a):
    """Compute the objective 2*n*max(conv)/sum^2."""
    n = len(a)
    conv = fast_convolve(a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_conv / (sum_a ** 2)


def optimize_lbfgsb_smooth(a0, beta, maxiter=300):
    """
    Optimize using L-BFGS-B with softmax reparametrization and smooth max.
    """
    n = len(a0)
    a0_pos = np.maximum(a0, 1e-15)
    a0_pos = a0_pos / a0_pos.sum()
    x0 = np.log(a0_pos)
    
    def func(x):
        x_shifted = x - np.max(x)
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm
    
    def jac(x):
        x_shifted = x - np.max(x)
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        grad_a = compute_grad_fft(a, w)
        # Convert gradient from a-space to x-space
        # a = exp(x - max(x)) / sum(exp(x - max(x)))
        # da/dx = a * (I - 1/a.sum() * a * 1^T) ... this is the softmax Jacobian
        # Actually, since we normalized: a = softmax(x)
        # d(softmax(x))/dx = diag(a) - a*a^T
        # So grad_x = (diag(a) - a*a^T) @ grad_a = a * (grad_a - a @ grad_a)
        lagrange = np.dot(a, grad_a)
        grad_x = a * (grad_a - lagrange)
        return grad_x
    
    result = minimize(func, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50, 'gtol': 1e-10})
    
    x_opt = result.x
    a_opt = np.exp(x_opt - np.max(x_opt))
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def nesterov_subgradient(a0, n_iter, initial_lr=None):
    """
    Nesterov-accelerated subgradient method on the simplex.
    Minimizes f(a) = max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    y = a.copy()  # Nesterov lookahead
    
    # Initial step size
    conv = fast_convolve(a)
    f0 = np.max(conv)
    if initial_lr is None:
        lr = 1.0 / np.sqrt(n_iter * f0) if f0 > 1e-10 else 0.01
    else:
        lr = initial_lr
    
    best_a = a.copy()
    best_f = f0
    
    beta_nesterov = 1.0  # Standard Nesterov momentum
    
    for it in range(n_iter):
        # Nesterov lookahead
        t_k = 2.0 / (it + 2.0)
        y = a + (t_k - 1.0) * (a - y) if it > 0 else a.copy()
        # Simpler: y = a + momentum * (a - a_prev)
        # Let me use the standard form:
        # y_{k+1} = a_{k+1} + (t_k - 1)/t_{k+1} * (a_{k+1} - a_k)
        # where t_k = (1 + sqrt(1+4t_{k-1}^2))/2
        # Actually, let me use a simpler momentum version.
        
        # Recompute y using standard Nesterov
        if it == 0:
            y = a.copy()
        else:
            t_prev = 2.0 / (it) if it > 0 else 1.0
            t_curr = 2.0 / (it + 1.0)
            y = a + (t_prev - t_curr) / t_curr * (a - y)
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        max_idx = np.argmax(conv_y)
        # Subgradient of max_k (a*a)[k] at the argmax:
        # grad[i] = a[max_idx - i] + a[i - max_idx] (with valid indices)
        # More precisely: grad[i] = y[max_idx - i] if 0 <= max_idx - i < n, 
        #                    + y[i - max_idx] if 0 <= i - max_idx < n
        # But wait, for auto-convolution: (a*a)[k] = sum_j a[j]*a[k-j]
        # d/d(a[i]) = y[k-i] + y[k-i] (if valid) = 2*y[k-i] (if 0<=k-i<n)
        # Wait no: d/d(a[i]) sum_j a[j]*a[k-j] 
        # = sum_j (delta(j,i)*a[k-j] + a[j]*delta(k-j,i))
        # = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n)
        # = 2*a[k-i] (if 0<=k-i<n)
        
        # So the subgradient for the single active constraint at max_idx is:
        # grad[i] = 2 * y[max_idx - i] if 0 <= max_idx - i < n else 0
        
        # But if multiple indices are at the max, we should average.
        # For simplicity, use just the argmax.
        
        # Actually, for better convergence, let's use a "soft" argmax:
        # weight the top few peaks
        top_k = min(5, len(conv_y))
        top_indices = np.argsort(conv_y)[-top_k:]
        top_vals = conv_y[top_indices]
        # Weight by how close to max
        w = np.exp(100 * (top_vals - top_vals.max()))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for k_idx, wk in zip(top_indices, w):
            k_int = int(k_idx)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wk * y[j]
        
        # Normalize gradient for step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
            
        # Take step
        a_new = project_simplex(y - lr * grad)
        
        # Adaptive step size (reduce if function increased)
        conv_new = fast_convolve(a_new)
        f_new = np.max(conv_new)
        
        if f_new > f0 * 2:  # Too aggressive
            lr *= 0.5
            a_new = a  # Keep old
        else:
            a = a_new
        
        if f_new < best_f:
            best_f = f_new
            best_a = a.copy()
        
        # Decay learning rate
        lr *= 0.999
        
    return best_a, best_f


def subgradient_polish(a0, n_iter, lr=None):
    """
    Simple subgradient method with projection (no Nesterov, just for polishing).
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_f = np.max(fast_convolve(a))
    
    if lr is None:
        lr = 0.01 / np.sqrt(n)
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        max_val = np.max(conv)
        
        if max_val < best_f:
            best_f = max_val
            best_a = a.copy()
        
        # Weighted subgradient using top peaks
        threshold = max_val - 0.01 * max_val
        active = np.where(conv >= threshold)[0]
        
        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        
        grad /= len(active)
        
        # Step with projection
        a_new = project_simplex(a - lr * grad)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        if np.max(conv_new) < max_val:
            a = a_new
        # else: keep a (no improvement)
        
        lr *= 0.999
        
    return best_a, best_f


def active_set_refine(a0, n_newton=20, tol=1e-10):
    """
    Active-set refinement: identify near-peak positions and solve KKT system.
    Uses a simplified Newton approach.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    
    for _ in range(n_newton):
        conv = fast_convolve(a)
        max_val = np.max(conv)
        
        # Find active set: peaks within 1% of max
        threshold = max_val * 0.99
        active = np.where(conv >= threshold)[0]
        
        if len(active) < 2:
            break
        
        # Compute gradient at each active peak
        # grad_k[i] = 2 * a[k-i] if 0<=k-i<n else 0
        # KKT: sum_{k in active} mu_k * grad_k[i] = lambda (for free i)
        #     sum(a) = 1
        #     (a*a)[k] = max_val for k in active
        
        # This is a complex system. Let's use a simpler approach:
        # Perturb a in the direction that reduces the max while keeping sum=1.
        
        # Compute the "balanced" subgradient
        m = len(active)
        # Equal weights for simplicity
        mu = np.ones(m) / m
        
        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * mu[list(active).index(int(k))] * a[j]
        
        # Project gradient onto the tangent space of the simplex
        # Tangent space: {v : sum(v) = 0, v[i] >= 0 where a[i] >= 0}
        mean_grad = grad.mean()
        grad_tangent = grad - mean_grad
        
        # For indices where a[i] is very small, zero out the gradient
        free_mask = a > 1e-8
        grad_tangent[~free_mask] = 0
        grad_tangent -= grad_tangent[free_mask].mean() * free_mask
        
        # Newton step: a_new = a - alpha * grad_tangent
        # Choose alpha for sufficient decrease
        alpha = 0.01
        for _ in range(20):
            a_new = project_simplex(a - alpha * grad_tangent)
            conv_new = fast_convolve(a_new)
            if np.max(conv_new) < max_val * (1 - 1e-8):
                a = a_new
                break
            alpha *= 0.5
        else:
            # No improvement, try smaller active set
            break
        
        if np.max(fast_convolve(a)) < max_val * (1 - tol):
            continue
        else:
            break
    
    return a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian
    for sf in [0.2, 0.3, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%.1f' % sf] = g
    
    # Bimodal (edges elevated)
    edge = max(1, n // 5)
    bi = np.ones(n)
    bi[:edge] = 2.0
    bi[-edge:] = 2.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Flat spectrum (random phases)
    fft_size = 1 << (n - 1).bit_length()
    for trial in range(5):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * phases[-1] if len(phases) else 0)
        
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            inits['flat_spec_%d' % trial] = a
    
    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            elif len(parent) >= n:
                # Downsample
                idx = np.linspace(0, len(parent) - 1, n).astype(int)
                inits['parent_ds'] = pp[idx] / pp[idx].sum()
            else:
                # Upsample
                pp_up = np.interp(np.linspace(0, 1, n), 
                                  np.linspace(0, 1, len(parent)), pp)
                pp_up = np.maximum(pp_up, 0)
                inits['parent_up'] = pp_up / pp_up.sum()
    
    return inits


def multi_beta_lbfgsb(a0, betas, time_budget):
    """
    Run L-BFGS-B with a schedule of decreasing beta values.
    """
    a = a0.copy()
    t_start = time.monotonic()
    
    for beta in betas:
        t_elapsed = time.monotonic() - t_start
        t_remaining = time_budget - t_elapsed
        if t_remaining < 5:
            break
        maxiter = int(min(400, max(30, t_remaining * 20)))
        a, _ = optimize_lbfgsb_smooth(a, beta, maxiter=maxiter)
        a = project_simplex(a)
    
    return a
