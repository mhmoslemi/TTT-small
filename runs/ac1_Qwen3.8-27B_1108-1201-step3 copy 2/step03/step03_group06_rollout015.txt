def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])

def logsumexp_gradient(a, conv, beta):
    """Compute gradient of log(sum(exp(beta * conv))) w.r.t. a."""
    # This is (1/beta) * sum_k exp(beta*conv_k) * grad(conv_k)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum  # This gives softmax weights
    
    n = len(a)
    # grad = sum_k w[k] * 2*a[k-j] (for valid j)
    # For each k, the subgradient is 2*a shifted/reversed
    # Efficient computation: for each j, grad[j] = 2 * sum_k w[k] * a[k-j] (valid)
    # = 2 * (w * a reversed)[j + n - 1] ... let me think
    
    # Actually: grad[j] = 2 * sum_{k: j in S_k} w[k] * a[k-j]
    # j in S_k means max(0, k-n+1) <= j <= min(n-1, k)
    # i.e., k >= j and k <= j+n-1 and k >= n-1-j (from k-n+1 <= j means k <= j+n-1, and k >= j from j <= k)
    # Wait: max(0, k-n+1) <= j means k-n+1 <= j (if k>=n) or 0 <= j (always true). So k <= j+n-1.
    # And j <= min(n-1, k) means j <= k and j <= n-1. So k >= j.
    # Combined: j <= k <= j+n-1 (and 0 <= k <= 2n-2)
    
    # So grad[j] = 2 * sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w[k] * a[k-j]
    # Let m = k-j, then m ranges from max(0,j)-j to min(2n-2,j+n-1)-j = 0 to min(n-1, 2n-2-j)
    # grad[j] = 2 * sum_{m=0}^{min(n-1, 2n-2-j)} w[j+m] * a[m]
    # = 2 * sum_{m=0}^{n-1} w[j+m] * a[m] (with w out of range = 0)
    
    # This is a cross-correlation of w and a!
    # cross[j] = sum_m w[j+m] * a[m]
    
    # Using FFT:
    # sum_m w[j+m] * a[m] for m=0..n-1
    # = sum_m a[m] * w[m+j]
    # This is the cross-correlation of a and w at lag j
    # = ifft(fft(a) * conj(fft(w)))[j]
    
    fft_size = 1 << (2*n - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    
    grad = 2 * cross
    return grad

def solve_lp_all_constraints(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    
    # Build constraint matrix
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        A_ub[k, idx] = a[k - idx]

    b_ub = np.full(num_k, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

"""
Hybrid LP + smooth subgradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses full-constraint LP directions, log-sum-exp smoothed gradients with adaptive
beta, backtracking line search, multiplicative weights updates, and multiple
diverse initializations. Focuses computational budget adaptively on best candidates.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def logsumexp_grad(a, conv, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*conv))) w.r.t. a, scaled by beta."""
    n = len(a)
    num_k = len(conv)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(num_k)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # grad[j] = 2 * sum_{k} w[k] * a[k-j] (where 0<=k-j<n)
    # = 2 * cross_correlation(a, w)[j]
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    return 2.0 * cross


def exact_subgradient(a, conv):
    """Exact subgradient at the argmax of conv."""
    n = len(a)
    k_star = np.argmax(conv)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    j_range = np.arange(lo, hi + 1)
    g[j_range] = 2.0 * a[k_star - j_range]
    return g


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1

    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        A_ub[k, idx] = a[k - idx]

    c = -np.ones(n)
    b_ub = np.full(num_k, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_toppk(a, M, top_k, conv):
    """Solve LP with only top_k peak constraints."""
    n = len(a)
    num_k = 2 * n - 1
    if top_k >= num_k:
        return solve_lp_direction(a, M)

    peaks = np.argpartition(conv, -top_k)[-top_k:]
    A_ub = np.zeros((top_k, n))
    for i, k in enumerate(peaks):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        A_ub[i, idx] = a[k - idx]

    c = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def objective_normalized(a):
    """Return 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n (sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Edge-peaked (bimodal)
    for edge_frac in [0.15, 0.2, 0.25, 0.3]:
        edge = max(1, int(n * edge_frac))
        bi = np.zeros(n)
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        bi /= bi.sum()
        inits[f'bimodal_{edge_frac:.2f}'] = bi

    # Smooth edge-peaked
    for sf in [0.1, 0.15, 0.2, 0.25]:
        d = np.minimum(x, n - 1 - x)
        g = np.exp(-d / (sf * n))
        g = g / g.sum()
        inits[f'edge_gauss_{sf:.2f}'] = g

    # Center-peaked Gaussian
    for sf in [0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'center_gauss_{sf:.2f}'] = g

    # Cosine windows
    for power in [1, 2, 3, 4]:
        if n > 1:
            cw = np.cos(np.pi * x / (n - 1)) ** power
        else:
            cw = np.ones(1)
        cw = np.maximum(cw, 1e-6)
        cw /= cw.sum()
        inits[f'cosine_{power}'] = cw

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Random
    for alpha in [0.5, 1.0, 2.0]:
        inits[f'dirichlet_{alpha}'] = rng.dirichlet(np.ones(n) * alpha)

    # Polynomial (Kac-Rice type)
    for deg in [2, 3, 4, 5]:
        t = x / (n - 1)
        poly = np.ones(n)
        for d in range(2, deg + 1, 2):
            poly = poly + 0.1 * (2 * t - 1) ** d
        poly = np.maximum(poly, 0.01)
        poly /= poly.sum()
        inits[f'poly_{deg}'] = poly

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        if len(parent) > 0:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    # Sparse random
    for sparsity in [0.5, 0.7, 0.8]:
        mask = (rng.random(n) < sparsity).astype(float)
        vals = mask * rng.random(n)
        vs = vals.sum()
        if vs > 1e-10:
            inits[f'sparse_{sparsity:.1f}'] = vals / vs

    return inits


def hybrid_optimize(a0, time_limit, seed=0, use_full_lp=True, verbose=False):
    """
    Hybrid optimizer: alternates LP directions and smooth gradient steps.
    Uses backtracking line search and adaptive parameters.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = normalize(a)
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0.8
    beta = 200.0
    no_improve = 0
    iter_count = 0
    lp_every = 3  # Do LP every lp_every iterations
    momentum = np.zeros(n)
    lr = 0.03

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        obj = 2.0 * n * M

        # Choose direction: LP or gradient
        if iter_count % lp_every == 0 and use_full_lp:
            b = solve_lp_direction(a, M)
            if b is None:
                b = solve_lp_toppk(a, M, max(30, n // 5), conv)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    # Blend with current
                    a_try = (1.0 - step) * a + step * b
                    a_try = np.maximum(a_try, 0.0)
                    a_try = normalize(a_try)
                else:
                    a_try = a.copy()
            else:
                a_try = a.copy()
        else:
            # Smooth gradient step
            g = logsumexp_grad(a, conv, beta)
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-15:
                g = g / g_norm
            momentum = 0.9 * momentum - lr * g
            a_try = project_simplex(a + momentum)

        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
            # Slightly increase step when improving
            step = min(1.0, step * 1.05)
            lr = min(0.1, lr * 1.01)
        else:
            no_improve += 1
            # Backtracking
            if no_improve <= 5:
                step *= 0.8
                lr *= 0.9
            elif no_improve <= 15:
                step *= 0.7
                lr *= 0.8
                momentum *= 0.5
            else:
                # Perturb to escape
                noise_scale = 0.01 * min(1.0, no_improve / 30.0)
                noise = rng.standard_normal(n) * noise_scale
                a = project_simplex(a + noise)
                momentum = np.zeros(n)
                step = 0.5
                lr = 0.03
                beta = rng.uniform(100, 2000)
                no_improve = 0

        # Adapt beta (sharper approximation over time)
        beta = min(5000.0, beta * 1.002)

        # Occasionally switch between LP and gradient more aggressively
        if no_improve > 20 and iter_count % 10 == 0:
            lp_every = rng.choice([2, 3, 4, 5])

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      iter {iter_count}: obj={best_obj:.7f}, step={step:.3f}, "
                  f"beta={beta:.0f}, lr={lr:.4f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def multiplicative_weights_optimize(a0, time_limit, seed=0, verbose=False):

    n = len(a)

    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0.3
    beta = 2000.0
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Alternate LP and gradient
        if iter_count % 2 == 0:
            b = solve_lp_toppk(a, M, max(50, n // 4), conv)
            if b is not None:
                b = normalize(b)
                a = normalize((1.0 - step) * a + step * b)
            else:
                a = a.copy()
        else:
            g = logsumexp_grad(a, conv, beta)
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-15:
                g = g / g_norm
                a = project_simplex(a - 0.01 * g)

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Tiny perturbation
                noise = rng.standard_normal(n) * 0.002
                a = project_simplex(a + noise)
                step = 0.2
                no_improve = 0

        iter_count += 1
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Polish iter {iter_count}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP directions, smooth gradients,
    and multiplicative weights. Explores multiple lengths and initializations,
    then deep-optimizes the best candidates.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    def update_best(a, label=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST [{label}]: score={score:.7f}, n={len(a)}")
            return True
        return False

    # ========== Phase 1: Quick Exploration (200s) ==========
    print("=== Phase 1: Quick Exploration ===")
    t1_start = time.monotonic()
    phase1_time = 200.0

    if parent is not None:
        parent_n = len(parent)
        lengths = sorted(set([parent_n, parent_n - 32, parent_n + 32,
                              256, 320, 384, 448, 512]))
        lengths = [n for n in lengths if 64 <= n <= 640]
    else:
        lengths = [128, 192, 256, 320, 384, 512]

    scan_results = []
    time_per_n = phase1_time / max(1, len(lengths))

    for n in lengths:
        if time.monotonic() > t1_start + phase1_time - 30:
            break

        inits = generate_inits(n, parent, rng)
        # Select top initializations to try
        priority = ['parent', 'parent_interp', 'bimodal_0.20', 'edge_gauss_0.15',
                    'center_gauss_0.30', 'cosine_2', 'uniform', 'dirichlet_1.0',
                    'bimodal_0.25', 'edge_gauss_0.20']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in init_names][:3]
        init_names = init_names[:6]

        time_per_init = max(5.0, (time_per_n - 10) / max(1, len(init_names)))

        for name in init_names:
            remaining = t1_start + phase1_time - time.monotonic() - 5
            if remaining < 5:
                break
            t_alloc = min(time_per_init, remaining)
            if t_alloc < 4:
                break

            a0 = inits[name]
            a_opt, obj_opt = hybrid_optimize(a0, time_limit=t_alloc,
                                             seed=n * 7 + hash(name) % 1000,
                                             use_full_lp=(n <= 384))
            if update_best(a_opt, f"scan n={n} {name}"):
                pass
            scan_results.append((obj_opt, n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.7f}")

    # ========== Phase 2: Deep LP Optimization (700s) ==========
    print("=== Phase 2: Deep Optimization ===")
    t2_start = time.monotonic()
    phase2_time = 700.0

    # Collect starting points
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:10]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy(), name))
            seen.add(key)
        if len(deep_starts) >= 8:
            break

    if best_sequence is not None:
        deep_starts.append((len(best_sequence), best_sequence.copy(), "best_so_far"))

    # Add fresh starts at promising lengths
    if parent is not None:
        best_n = len(parent)
    else:
        best_n = 320

    inits_best = generate_inits(best_n, parent, rng)
    for extra in ['edge_gauss_0.15', 'bimodal_0.20', 'cosine_3', 'center_gauss_0.25']:
        if extra in inits_best and len(deep_starts) < 12:
            deep_starts.append((best_n, inits_best[extra].copy(), f"fresh_{extra}"))

    num_runs = len(deep_starts)
    time_per_run = phase2_time / max(1, num_runs)
    time_per_run = max(20.0, time_per_run)

    run_i = 0
    for n_start, a_start, label in deep_starts:
        remaining = t2_start + phase2_time - time.monotonic() - 15
        if remaining < 20:
            break
        t_alloc = min(time_per_run, remaining)
        if t_alloc < 15:
            break

        # Run hybrid optimizer with full LP
        a_opt, obj_opt = hybrid_optimize(a_start, time_limit=t_alloc,
                                         seed=run_i * 13 + 5,
                                         use_full_lp=(n_start <= 448),
                                         verbose=False)
        if update_best(a_opt, f"deep {label} n={n_start}"):
            pass
        run_i += 1

        # Also try multiplicative weights on the result
        remaining = t2_start + phase2_time - time.monotonic() - 15
        if remaining > 30:
            t_mw = min(40.0, remaining * 0.3)
            a_mw, obj_mw = multiplicative_weights_optimize(a_opt, time_limit=t_mw,
                                                           seed=run_i * 17 + 3)
            if update_best(a_mw, f"MW {label} n={n_start}"):
                pass

    print(f"Phase 2 done. Best: {best_score:.7f}")

    # ========== Phase 3: Focused Fine-tuning (500s) ==========
    print("=== Phase 3: Fine-tuning ===")
    t3_start = time.monotonic()
    phase3_time = 500.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a_curr = best_sequence.copy()

        # Multiple rounds of fine polishing with different settings
        polish_configs = [
            {'time_frac': 0.25, 'seed_offset': 0},
            {'time_frac': 0.25, 'seed_offset': 100},
            {'time_frac': 0.25, 'seed_offset': 200},
            {'time_frac': 0.25, 'seed_offset': 300},
        ]

        for cfg in polish_configs:
            remaining = t3_start + phase3_time - time.monotonic() - 20
            if remaining < 20:
                break
            t_alloc = min(remaining, phase3_time * cfg['time_frac'])
            if t_alloc < 15:
                break

            a_opt, obj_opt = fine_polish(a_curr, time_limit=t_alloc,
                                         seed=cfg['seed_offset'] + 42)
            if update_best(a_opt, f"polish_{cfg['seed_offset']}"):
                a_curr = a_opt.copy()

        # Try longer sequences if we haven't beaten target yet
        if best_score > 1.503 and time.monotonic() < t3_start + phase3_time - 60:
            for extra_n in [384, 448, 512]:
                if extra_n == n_best:
                    continue
                remaining = t3_start + phase3_time - time.monotonic() - 30
                if remaining < 30:
                    break
                # Interpolate best to new length
                a_interp = np.interp(np.linspace(0, 1, extra_n),
                                     np.linspace(0, 1, n_best), a_curr)
                a_interp = normalize(np.maximum(a_interp, 0))
                t_alloc = min(60.0, remaining - 20)
                a_opt, obj_opt = hybrid_optimize(a_interp, time_limit=t_alloc,
                                                 seed=extra_n + 7, use_full_lp=True)
                if update_best(a_opt, f"longer n={extra_n}"):
                    a_curr = a_opt.copy()
                    n_best = extra_n

    print(f"Phase 3 done. Best: {best_score:.7f}")

    # ========== Phase 4: Final Squeeze (remaining time) ==========
    print("=== Phase 4: Final Squeeze ===")
    remaining = deadline - time.monotonic() - 15.0
    if remaining > 20 and best_sequence is not None:
        n_best = len(best_sequence)

        # Very long fine polish
        t_alloc = min(remaining * 0.5, 150.0)
        if t_alloc > 15:
            a_opt, obj_opt = fine_polish(best_sequence, time_limit=t_alloc,
                                         seed=9999, verbose=True)
            if update_best(a_opt, "final_polish"):
                best_sequence = a_opt

        # Final tiny perturbations
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            n_best = len(best_sequence)
            for i in range(30):
                if time.monotonic() > deadline - 15:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    s = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_sequence = a_pert.copy()
                        print(f"  Perturb [{i}] s={scale}: {s:.7f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
