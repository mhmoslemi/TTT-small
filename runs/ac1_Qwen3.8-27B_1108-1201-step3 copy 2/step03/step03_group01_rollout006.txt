"""
Hybrid optimizer for min 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient
descent (logsumexp), coordinate descent, and LP-based direction finding.
"""
import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, t):
    """Compute the gradient of (1/t)*log(sum(exp(t*conv))) with respect to a."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    scaled = t * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: sg[i] = 2 * sum_k w[k] * a[k-i] (valid indices)
    sg = np.zeros(n)
    for i in range(n):
        # sum_j w[i+j] * a[j] for j in [0, n-1] and i+j in [0, 2n-2]
        j_max = min(n, 2*n-1-i)
        if j_max > 0:
            sg[i] = 2.0 * np.dot(w[i:i+j_max], a[:j_max])
    return sg, conv, w

def smooth_grad_desc(a0, time_limit, t_init=100, t_final=5000, lr_init=0.005, momentum=0.9):
    """
    Smooth gradient descent with logsumexp approximation to max.
    Temperature t increases from t_init to t_final.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t = t_init
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sg, conv, w = compute_smooth_grad(a, t)
        
        # Project gradient onto sum=1 hyperplane
        sg_proj = sg - sg.mean()
        
        # Momentum update
        v = momentum * v - lr * sg_proj
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        t *= 1.005
        if t > t_final:
            t = t_final
        lr *= 0.9995
        if no_improve % 200 == 0 and no_improve > 0:
            lr *= 0.9
        momentum = min(0.98, momentum + 0.0001)
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

def coord_descent(a0, time_limit, sweeps=50):
    """
    Coordinate descent: optimize one coordinate at a time.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    sweep = 0
    
    while sweep < sweeps and time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute affected convolution values
            # k ranges from i to i+n-1
            ai = a[i]
            S = s - ai  # sum of other elements (s is the total sum which is 1 after normalization)
            
            # Precompute for k in [i, i+n-1]:
            # conv(k) = Q_k * ai^2 + L_k * ai + C_k
            ks = np.arange(i, i + n)
            
            # C_k: convolution at k excluding the a[i] terms
            # conv(k) = sum_{i' in [max(0,k-n+1), min(k,n-1)]} a[i'] * a[k-i']
            # The terms involving a[i] are: if k-i in [0,n-1], then a[i]*a[k-i] (counted as 2 unless k=2i)
            
            C = np.zeros(n)
            L = np.zeros(n)
            Q = np.zeros(n)
            
            for idx, k in enumerate(ks):
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                # Compute conv(k) directly and subtract a[i] terms
                conv_k = 0.0
                ai_term = 0.0
                for ip in range(lo, hi + 1):
                    val = a[ip] * a[k - ip]
                    conv_k += val
                    if ip == i:
                        ai_term += val
                C[idx] = conv_k - ai_term
                if k == 2 * i:
                    Q[idx] = 1.0
                else:
                    j = k - i
                    if 0 <= j < n:
                        L[idx] = 2.0 * a[j]

            else:

                M_right = conv[i+n:].max()
            else:

            M_unchanged = max(M_left, M_right)

            # Note: S + ai = s = 1 (since we normalized)
            # So we just need to minimize max(M_unchanged, max_k(...))

            # Q for k=2i is 1, so ai^2 + C_{2i} <= M_target
            # Let's use a reasonable range

            ai_max = max(ai_max, 0.01)
            
            # Grid search
            candidates = np.linspace(0, ai_max, 40)
            best_ai_val = ai
            best_F = np.inf

                vals = Q * ai_c**2 + L * ai_c + C
                max_aff = vals.max()

"""
Hybrid optimizer for min 2*n*max(conv(a,a))/(sum(a))^2 combining smooth gradient
descent (logsumexp approximation), efficient coordinate descent, and LP-based
direction finding with multi-start exploration and time-adaptive scheduling.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_gradient(a, t):
    """Compute gradient of logsumexp approximation to max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # sg[i] = 2 * sum_j w[i+j] * a[j]
    sg = np.zeros(n)
    for i in range(n):
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            sg[i] = 2.0 * np.dot(w[i:i + j_max], a[:j_max])
    return sg, conv, w


def smooth_grad_desc(a0, time_limit, t_init=50.0, t_final=10000.0,
                     lr_init=0.003, momentum_init=0.85):
    """Smooth gradient descent with logsumexp and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    t = t_init
    lr = lr_init
    momentum = momentum_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        sg, conv, w = compute_smooth_gradient(a, t)
        sg_proj = sg - sg.mean()

        v = momentum * v - lr * sg_proj
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        iter_count += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        t *= 1.003
        if t > t_final:
            t = t_final
        lr *= 0.9997
        if no_improve % 300 == 0 and no_improve > 0:
            lr *= 0.85
            v *= 0.5
        momentum = min(0.97, momentum + 0.00005)

        if no_improve > 800:
            break

    return best_a, best_obj


def coord_descent(a0, time_limit, random_order=False, rng=None):
    """Efficient coordinate descent for the unnormalized problem."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n)
        S = float(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max() / (S * S)
    best_a = a.copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        sweep_improved = False
        order = np.random.permutation(n) if (random_order and rng is not None) else range(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            ks = np.arange(i, i + n)
            is_center = (ks == 2 * i)
            Q = is_center.astype(np.float64)
            L = np.where(is_center, 0.0, 2.0 * a[ks - i])
            C = conv[ks] - Q * ai * ai - L * ai

            # Unaffected maximum
            M_unchanged = 0.0
            if i > 0:
                M_unchanged = max(M_unchanged, float(conv[:i].max()))
            if i + n < 2 * n - 1:
                M_unchanged = max(M_unchanged, float(conv[i + n:].max()))

            S_other = S - ai

            # Find optimal ai_new
            # Upper bound heuristic
            c_center = C[is_center]
            if len(c_center) > 0:
                ai_upper = max(ai * 2.5, np.sqrt(max(M_unchanged - c_center[0], 0.0)) + 0.05)
            else:
                ai_upper = ai * 2.5
            ai_upper = max(ai_upper, 0.01, S * 0.1)

            candidates = np.unique(np.concatenate([
                np.linspace(0.0, ai_upper, 25),
                np.array([ai, ai * 0.5, ai * 0.25, 0.0])
            ]))

            best_ai_new = ai
            best_F = np.inf
            for ai_c in candidates:
                S_c = S_other + ai_c
                if S_c < 1e-12:
                    continue
                vals = Q * ai_c * ai_c + L * ai_c + C
                max_conv = max(M_unchanged, float(vals.max()))
                F_c = 2.0 * n * max_conv / (S_c * S_c)
                if F_c < best_F:
                    best_F = F_c
                    best_ai_new = ai_c

            if abs(best_ai_new - ai) > 1e-10 * max(1.0, ai):
                delta = best_ai_new - ai
                conv[ks] += Q * (best_ai_new * best_ai_new - ai * ai) + L * delta
                a[i] = best_ai_new
                S = S_other + best_ai_new
                cur_obj = 2.0 * n * conv.max() / (S * S)
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_obj


def lp_refine(a0, time_limit):
    """LP-based direction finding and line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Build constraint matrix: C @ g = a * g
        C_mat = np.zeros((2 * n - 1, n))
        for i in range(n):
            C_mat[i:i + n, i] = a

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 1.001)
        bounds = [(0, None)] * n

        try:
            result = linprog(c_obj, A_ub=C_mat, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            result = None

        if result is not None and result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-6:
                d = g / g_sum - a
                norm_d = np.linalg.norm(d)
                if norm_d > 1e-12:
                    # Line search
                    best_t = 0.0
                    best_obj_ls = best_obj
                    for t in [0.9, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                        a_try = a + t * d
                        a_try = np.maximum(a_try, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        conv_try = fast_convolve(a_try)
                        obj_try = 2.0 * n * conv_try.max()
                        if obj_try < best_obj_ls - 1e-13:
                            best_obj_ls = obj_try
                            best_t = t

                    if best_t > 0:
                        a_new = a + best_t * d
                        a_new = np.maximum(a_new, 0.0)
                        a_new = a_new / a_new.sum()
                        conv = fast_convolve(a_new)
                        M = conv.max()
                        a = a_new
                        best_obj = 2.0 * n * M
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Perturb if stuck
        if no_improve > 30:
            rng = np.random.default_rng(no_improve * 31 + 7)
            scale = 0.01 * min(no_improve / 30.0, 5.0)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0

        if no_improve > 200:
            break

    return best_a, best_obj


def generate_start(n, parent, rng):
    """Generate diverse starting sequences of length n."""
    starts = {}
    x = np.arange(n, dtype=np.float64)

    starts['uniform'] = np.ones(n) / n
    starts['dirichlet'] = rng.dirichlet(np.ones(n) * 0.5)

    ramp = x + 1.0
    starts['ramp'] = ramp / ramp.sum()

    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    starts['bimodal'] = bi / bi.sum()

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    starts['gaussian'] = g / g.sum()

    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n) + 0.05
    starts['sine'] = sine / sine.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                starts['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:

            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:

    phase1_limit = min(400.0, (deadline - time.monotonic()) * 0.25)
    phase1_deadline = t_start + phase1_limit

    lengths = [512]
    if parent is not None:
        lengths.insert(0, len(parent))
        if len(parent) != 512:
            lengths.append(512)

    candidates = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        starts = generate_start(n, parent, rng)
        # Priority order
        priority = ['parent', 'parent_ds', 'parent_up', 'bimodal', 'uniform',
                    'dirichlet', 'sine', 'gaussian', 'ramp']
        names = [nm for nm in priority if nm in starts]
        names += [nm for nm in starts if nm not in priority]
        names = names[:4]

        t_per = max(15.0, (phase1_deadline - time.monotonic() - 10) / max(1, len(names)))

        for name in names:
            if time.monotonic() > phase1_deadline - 5:
                break
            t_alloc = min(t_per, phase1_deadline - time.monotonic() - 5)
            if t_alloc < 8:
                break

            a0 = starts[name]

            # Quick smooth gradient descent
            a_opt, obj_opt = smooth_grad_desc(a0, time_limit=t_alloc * 0.5,
                                              t_init=50, t_final=2000, lr_init=0.004)
            # Quick coordinate descent
            a_opt, obj_opt2 = coord_descent(a_opt, time_limit=t_alloc * 0.4)

            # Evaluate
            a_test = a_opt / a_opt.sum()
            score = float(evaluate_sequence(a_test.tolist()))

            if np.isfinite(score):
                candidates.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase1 n={n} {name}: {score:.6f}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: Deep optimization on best candidate
    print("=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()

    if best_sequence is not None:
        # Work on the best candidate with cycling between methods
        remaining = deadline - time.monotonic() - 60.0
        current = best_sequence.copy()
        current = np.maximum(current, 0.0)
        current = current / current.sum()
        n_curr = len(current)

        # Cycling loop: alternate between methods
        cycle_cfgs = [
            ('sg', {'t_init': 200, 't_final': 5000, 'lr_init': 0.002, 'momentum_init': 0.9}),
            ('cd', {}),
            ('sg', {'t_init': 1000, 't_final': 15000, 'lr_init': 0.001, 'momentum_init': 0.93}),
            ('cd', {}),
            ('lp', {}),
            ('sg', {'t_init': 5000, 't_final': 20000, 'lr_init': 0.0005, 'momentum_init': 0.95}),
            ('cd', {}),
        ]

        cycle_len = 60  # seconds per cycle
        cycle_idx = 0
        plateau = 0

        while time.monotonic() < deadline - 50:
            method, cfg = cycle_cfgs[cycle_idx % len(cycle_cfgs)]
            t_alloc = min(cycle_len, deadline - time.monotonic() - 40)
            if t_alloc < 5:
                break

            if method == 'sg':
                a_opt, obj_opt = smooth_grad_desc(current, time_limit=t_alloc, **cfg)
            elif method == 'cd':
                a_opt, obj_opt = coord_descent(current, time_limit=t_alloc,
                                               random_order=(cycle_idx % 2 == 1), rng=rng)
            elif method == 'lp':
                a_opt, obj_opt = lp_refine(current, time_limit=t_alloc)
            else:
                a_opt = current
                obj_opt = float('inf')

            a_test = a_opt / a_opt.sum()
            score = float(evaluate_sequence(a_test.tolist()))

            if np.isfinite(score) and score < best_score - 1e-9:
                best_score = score
                best_sequence = a_opt.copy()
                current = a_test.copy()
                plateau = 0
                print(f"  Phase2 c{cycle_idx} {method}: {score:.6f}")
            else:
                plateau += 1
                # Slight perturbation to escape
                if plateau > 3:
                    scale = 0.005 * min(plateau - 2, 5)
                    current = best_sequence + rng.standard_normal(len(best_sequence)) * scale
                    current = np.maximum(current, 0.0)
                    current = current / current.sum()
                    plateau = 0

            cycle_idx += 1

            # Occasionally try a different length from parent
            if cycle_idx % 7 == 0 and parent is not None and len(parent) != n_curr:
                t_try = min(30.0, deadline - time.monotonic() - 40)
                if t_try > 10:
                    p_n = len(parent)
                    p_norm = parent / parent.sum()
                    a_try, _ = smooth_grad_desc(p_norm, time_limit=t_try * 0.5,
                                                t_init=100, t_final=3000)
                    a_try, _ = coord_descent(a_try, time_limit=t_try * 0.4)
                    a_try = a_try / a_try.sum()
                    s_try = float(evaluate_sequence(a_try.tolist()))
                    if np.isfinite(s_try) and s_try < best_score - 1e-9:
                        best_score = s_try
                        best_sequence = a_try.copy()
                        current = a_try.copy()
                        n_curr = p_n
                        print(f"  Phase2 parent-n: {best_score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: Final polish with extended coordinate descent
    print("=== Phase 3: Final Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            # Extended coordinate descent
            t_cd = min(remaining * 0.6, 200.0)
            a_opt, obj_opt = coord_descent(best_sequence, time_limit=t_cd)
            a_opt = a_opt / a_opt.sum()
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score - 1e-9:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CD polish: {score:.6f}")

            # Final smooth gradient with very high temperature
            remaining2 = deadline - time.monotonic() - 10.0
            if remaining2 > 10:
                a_opt, _ = smooth_grad_desc(best_sequence, time_limit=remaining2 * 0.5,
                                            t_init=10000, t_final=30000,
                                            lr_init=0.0003, momentum_init=0.95)
                a_opt = a_opt / a_opt.sum()
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score - 1e-9:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final SG: {score:.6f}")

            # One more CD pass
            remaining3 = deadline - time.monotonic() - 5.0
            if remaining3 > 5:
                a_opt, _ = coord_descent(best_sequence, time_limit=remaining3)
                a_opt = a_opt / a_opt.sum()
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score - 1e-9:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final CD: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        best_sequence = best_sequence / best_sequence.sum()

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
