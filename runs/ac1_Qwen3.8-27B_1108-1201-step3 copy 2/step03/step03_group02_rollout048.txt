```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed gradient descent
with adaptive temperature annealing, simplex projection, and multi-restart
exploration. Uses FFT-based convolution and np.correlate for fast gradients.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad(a, t):
    """
    Compute LSE-smoothed objective value and gradient.
    Returns (obj, grad) where obj = 2n * LSE(conv, t) and
    grad = d(obj)/da.
    """
    n = len(a)
    conv = fast_autocorr(a)
    # Numerically stable LSE
    tc = t * conv
    tc_max = tc.max()
    shifted = tc - tc_max
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp  # softmax weights
    
    obj = 2.0 * n * (tc_max / t + np.log(sum_exp) / t)
    # grad[i] = 2n * sum_j p[i+j] * a[j]
    grad = 2.0 * n * np.correlate(p, a, mode='valid')
    return obj, grad
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed gradient descent
with multi-phase temperature annealing, simplex projection, multiplicative
weight updates, and multi-restart exploration. Leverages FFT for convolution
and np.correlate for O(n^2) gradient computation (fast for n=512).
"""

import numpy as np
import time


def fast_autocorr(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
    else:
        rho = n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_obj_grad(a, t):
    """
    Compute LSE-smoothed objective and gradient.
    obj = 2n * (1/t) * log(sum exp(t*conv_k))
    grad[i] = 2n * sum_j p[i+j] * a[j]
    """
    n = len(a)
    conv = fast_autocorr(a)
    tc = t * conv
    tc_max = tc.max()
    shifted = tc - tc_max
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    obj = 2.0 * n * (tc_max / t + np.log(sum_exp) / t)
    grad = 2.0 * n * np.correlate(p, a, mode='valid')
    return obj, grad


def true_obj(a):
    """True objective: 2*n*max(conv)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max() / (s * s)


def optimize_lse_pg(a0, time_limit, t_start=5000.0, t_end=200000.0,
                    lr_start=0.008, lr_end=0.0005, momentum=0.9,
                    n_phases=6, subgrad_every=500):
    """
    LSE-smoothed projected gradient with momentum and multi-phase annealing.
    Periodically mixes in true-subgradient steps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = true_obj(a)
    best_a = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    total_iters = 0

    for phase in range(n_phases):
        phase_dur = time_limit / n_phases
        frac = phase / max(1, n_phases - 1)
        t_cur = t_start * (t_end / t_start) ** frac
        lr_cur = lr_start * (lr_end / lr_start) ** frac
        v *= 0.5

        phase_start = time.monotonic()
        iters_this_phase = 0
        while time.monotonic() - phase_start < phase_dur and time.monotonic() - t0 < time_limit:
            obj_s, grad = lse_obj_grad(a, t_cur)

            if iters_this_phase % subgrad_every == subgrad_every - 1:
                # True subgradient step: find argmax of conv, use that subgradient
                conv = fast_autocorr(a)
                k_max = int(np.argmax(conv))
                # Subgradient of max at k_max: d/d a[i] = 2n * a[k_max - i] for valid i
                j_start = max(0, k_max - n + 1)
                j_end = min(n - 1, k_max)
                sg = np.zeros(n)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] = 2.0 * n * a[k_max - idx]
                v = momentum * v - lr_cur * sg
            else:
                v = momentum * v - lr_cur * grad

            a = project_simplex(a + v)
            total_iters += 1
            iters_this_phase += 1

            if total_iters % 200 == 0:
                cur = true_obj(a)
                if cur < best_obj:
                    best_obj = cur
                    best_a = a.copy()

    cur = true_obj(a)
    if cur < best_obj:
        best_obj = cur
        best_a = a.copy()

    return best_a, best_obj

def optimize_multiplicative(a0, time_limit, t_start=5000.0, t_end=200000.0,

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n (on simplex)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        inits[f'cos_{freq}'] = a

    for width in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (width * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{width}'] = g

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    # Bell-shaped (concentrated in middle)
    bell = np.sin(np.pi * x / n) ** 2
    bell /= bell.sum()
    inits['bell'] = bell

    # Two peaks
    two_peaks = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
                np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    two_peaks /= two_peaks.sum()
    inits['two_peaks'] = two_peaks

    return inits

            {'t_start': 10000, 't_end': 500000, 'lr_start': 0.003, 'lr_end': 0.00005,
             'momentum': 0.95, 'n_phases': 10, 'subgrad_every': 200},
        ]

        for cfg in configs:
            if time.monotonic() > p2_deadline:
                break
            t_alloc = min(160, p2_deadline - time.monotonic()) - 5
            if t_alloc < 10:
                break
            print(f"  Config: t=[{cfg['t_start']},{cfg['t_end']}], lr={cfg['lr_start']}")
            a_opt, _ = optimize_lse_pg(best_sequence, time_limit=t_alloc, **cfg)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}")

    # Phase 3: Mixed LSE + subgradient (300s)
    print("=== Phase 3: Mixed LSE + Subgradient ===")
    p3_deadline = time.monotonic() + min(300, budget_s * 0.17)

    if best_sequence is not None:
        for sf in [0.2, 0.5, 0.7]:
            if time.monotonic() > p3_deadline:
                break
            t_alloc = min(90, p3_deadline - time.monotonic()) - 5
            if t_alloc < 10:
                break
            a_opt, _ = optimize_mixed(best_sequence, time_limit=t_alloc,
                                       t_start=3000, t_end=200000,
                                       lr_start=0.004, momentum=0.9,
                                       subgrad_frac=sf)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}")

    # Phase 4: Multiplicative weights (200s)
    print("=== Phase 4: Multiplicative Weights ===")
    p4_deadline = time.monotonic() + min(200, budget_s * 0.12)

    if best_sequence is not None:
        for eta_s in [0.002, 0.005, 0.001]:
            if time.monotonic() > p4_deadline:
                break
            t_alloc = min(60, p4_deadline - time.monotonic()) - 5
            if t_alloc < 10:
                break
            a_opt, _ = optimize_multiplicative(best_sequence, time_limit=t_alloc,
                                                t_start=5000, t_end=200000,
                                                eta_start=eta_s, eta_end=max(eta_s * 0.05, 0.00005))
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}")

    # Phase 5: Perturbation restarts (200s)
    print("=== Phase 5: Perturbation + Re-optimize ===")
    p5_deadline = time.monotonic() + min(200, budget_s * 0.11)

    if best_sequence is not None:
        n_best = len(best_sequence)
        for scale in [0.005, 0.01, 0.02, 0.003, 0.015]:
            if time.monotonic() > p5_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(35, p5_deadline - time.monotonic()) - 3
            if t_alloc < 5:
                break
            a_opt, _ = optimize_lse_pg(a_pert, time_limit=t_alloc,
                                        t_start=10000, t_end=500000,
                                        lr_start=0.003, lr_end=0.0001,
                                        momentum=0.94, n_phases=4,
                                        subgrad_every=400)
            try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}")

    # Phase 6: Try other lengths (150s)
    print("=== Phase 6: Other Lengths ===")
    p6_deadline = time.monotonic() + min(150, budget_s * 0.08)

    for L in [256, 1024]:
        if time.monotonic() > p6_deadline:
            break
        t_alloc = min(70, p6_deadline - time.monotonic()) - 10
        if t_alloc < 15:
            break
        # Initialize from parent or uniform
        if parent is not None and len(parent) >= L:
            a0 = np.maximum(parent[:L], 0)
            a0 = a0 / a0.sum()
        else:
            a0 = np.ones(L) / L
        a_opt, _ = optimize_lse_pg(a0, time_limit=t_alloc,
                                    t_start=3000, t_end=200000,
                                    lr_start=0.008, lr_end=0.0002,
                                    momentum=0.92, n_phases=5)
        try_update(a_opt)

    print(f"Phase 6 done: best={best_score:.6f}")

    # Phase 7: Final ultra-fine polish (remaining time)
    print("=== Phase 7: Final Polish ===")
    p7_deadline = deadline - 10

    if best_sequence is not None and time.monotonic() < p7_deadline:
        t_alloc = p7_deadline - time.monotonic() - 5
        if t_alloc > 20:
            a_opt, _ = optimize_lse_pg(best_sequence, time_limit=t_alloc,
                                        t_start=100000, t_end=1000000,
                                        lr_start=0.0005, lr_end=0.00001,
                                        momentum=0.96, n_phases=6,
                                        subgrad_every=1000)
            try_update(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(N) / N

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
