                if b is None:
                    break
                b_sum = b.sum()
                if b_sum < 1e-15:
                    break
                b = b / b_sum
        else:
            b = solve_lp(a, M * relax, top_k=top_k, conv=conv)
            if b is None:
                b = solve_lp(a, M * relax, top_k=None)
                if b is None:
                    break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        
        # Backtracking line search
        best_step = 0
        best_obj_step = best_obj
        
        for attempt in range(10):
            t = step * (0.7 ** attempt)
            a_new = (1.0 - t) * a + t * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            
            if obj_new < best_obj_step - 1e-12:
                best_step = t
                best_obj_step = obj_new
                a_new_best = a_new.copy()
            
            if obj_new < best_obj - 1e-12:
                break
        
        if best_step > 0:
            a = a_new_best
            no_improve = 0
            
            # Periodic symmetrization
            if iter_count % symmetrize_period == 0:
                a_sym = symmetrize(a)
                a_sym = np.maximum(a_sym, 0.0)
                s_sym = a_sym.sum()
                if s_sym > 1e-15:
                    a_sym = a_sym / s_sym
                conv_sym = fast_convolve(a_sym)
                obj_sym = 2.0 * n * conv_sym.max()
                if obj_sym < 2.0 * n * fast_convolve(a).max():
                    a = a_sym
            
            conv_best = fast_convolve(a)
            if 2.0 * n * conv_best.max() < best_obj:
                best_obj = 2.0 * n * conv_best.max()
                best_a = a.copy()
        else:
            no_improve += 1
            if no_improve > 20:
                # Perturb
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                step = step_init
                no_improve = 0
            elif step > 0.05:
                step *= 0.8
        
        iter_count += 1
    
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj
