```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
combined with smooth gradient optimization. The LP approach solves a linear
program to find the "flattest" sequence compatible with the current peak bound,
then blends iterations to monotonically reduce the peak. Gradient-based polish
with log-sum-exp smooth max provides final refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    result = np.real(ifft(A * A, fft_size))[:size]
    return result


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def project_simplex(v):
    """Euclidean projection onto the (scaled) simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.sum(cond)
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def compute_smooth_obj(a, beta):
    """Compute smooth objective value."""
    n = len(a)
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm / (a.sum() ** 2)


def optimize_pg_smooth(a0, beta, lr=0.05, n_iter=500, deadline=time.time() + 100):
    """Projected gradient descent on simplex with smooth max objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    best_obj = np.inf
    best_a = a.copy()

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm  # sum(a)=1 so denominator is 1

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if obj < 1.01:  # Very close to theoretical minimum
            break

        # Gradient of sum_k w[k]*conv[k] w.r.t. a
        # conv[k] = sum_i a[i]*a[k-i], d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (with bounds)
        # So grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) = 2 * (w * a)[j] (circular, but we need linear)
        # More carefully: grad[j] = sum_k w[k] * (a[(k-j) mod ...])
        # Actually for linear conv: d(conv[k])/d(a[j]) = a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n
        # This is the cross-correlation of w with a, times 2 (by symmetry)
        # But w has length 2n-1 and a has length n
        # Let's compute it via FFT
        
        N = 1 << (len(w) + n - 1 + 63).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        # grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
        # The first term sum_k w[k]*a[k-j] is cross-correlation
        # Ifft(conj(W)*A) gives sum_k w[k]*a[k-j] (circular)
        grad_part = np.real(ifft(np.conj(W) * A, N))[:n]
        # By symmetry of the problem, grad = 2 * grad_part (approximately)
        # Actually, let me be more careful.
        # conv[k] = sum_i a[i] * a[k-i] where 0<=i<n and 0<=k-i<n
        # d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # So sum_k w[k] * d(conv[k])/d(a[j]) = sum_k w[k]*a[k-j] + sum_k w[k]*a[j-k]
        # First sum: cross-correlation (w, a) at lag j
        # Second sum: cross-correlation (w, a) at lag -j = same by symmetry of w? No, w is not symmetric.
        # Let me just compute both separately.
        
        # Actually, the cross-correlation r[j] = sum_k w[k]*a[k-j] is computed by:
        # ifft(conj(FFT(w)) * FFT(a)) - but this is circular correlation
        # For linear, we need to zero-pad enough.
        
        # Let me use a simpler approach: the gradient is 2 * cross_correlation(w, a) [clipped to length n]
        # where cross_correlation(w, a)[j] = sum_k w[k+j]*a[k] (with valid bounds)
        # This equals ifft(conj(FFT(w)) * FFT(a)) for the linear case with sufficient padding.
        
        grad = 2.0 * np.real(ifft(np.conj(W) * A, N))[:n]
        
        if np.all(np.abs(grad) < 1e-15):
            break

        # Projected gradient step
        t = lr
        for _ in range(8):
            a_new = project_simplex(a - t * grad)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5
        a = a_new
        lr *= 0.999

    return best_a, best_obj


def lp_refine_step(a, target_M=None, blending=0.5):
    """
    One LP-based refinement step.
    Given current sequence a (normalized to sum=1), solve LP to find
    the "flattest" b compatible with conv(a,b) <= M, then blend.
    Returns new sequence and improvement info.
    """
    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    conv_aa = fast_convolve(a)
    M = conv_aa.max() if target_M is None else target_M
    
    # Build constraint matrix for LP
    # Variables: b[0], ..., b[n-1]
    # Constraints: conv(a, b)[k] <= M for k = 0, ..., 2n-2
    #              -conv(a, b)[k] <= M for k = 0, ..., 2n-2
    # Objective: maximize sum(b) => minimize -sum(b)
    
    # conv(a,b)[k] = sum_j a[k-j]*b[j] where 0 <= k-j < n and 0 <= j < n
    # So the (k, j) entry of the constraint matrix is a[k-j] if 0 <= k-j < n, else 0
    
    # This is a Toeplitz matrix. Let's build it efficiently.
    # Row k: [0, ..., 0, a[k], a[k-1], ..., a[0], 0, ..., 0] (shifted)
    # More precisely: A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    
    # For memory efficiency with large n, we could use sparse, but for n<=1024 it's fine as dense
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A[k_start:k_end + 1, j] = a[j - (k_end - n + 1):j + 1][::-1]
        # Wait, let me think more carefully.
        # A[k, j] = a[k-j] if 0 <= k-j < n
        # For fixed j, k ranges from j to j+n-1 (but bounded by 2n-2)
        # A[k, j] = a[k-j] for k = j, j+1, ..., min(j+n-1, 2n-2)
    
    # Let me redo this more carefully
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k such that 0 <= k-j < n, i.e., j <= k < j+n
        # Also 0 <= k <= 2n-2
        k_min = j
        k_max = min(j + n - 1, 2 * n - 2)
        if k_min <= k_max:
            A[k_min:k_max + 1, j] = a[0:k_max - k_min + 1]
    
    # The full constraint matrix includes both conv(a,b) <= M and -conv(a,b) <= M
    A_ub = np.vstack([A, -A])
    b_ub = np.full(2 * (2 * n - 1), M)
    
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not result.success:
        # Fallback: return current a
        return a.copy(), 0.0
    
    b_opt = np.maximum(result.x, 0)
    sum_b = b_opt.sum()
    if sum_b < 1e-15:
        return a.copy(), 0.0
    
    # Scale to sum = 1
    g = b_opt / sum_b
    
    # Blend
    a_new = (1 - blending) * a + blending * g
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    
    # Compute improvement
    conv_new = fast_convolve(a_new)
    M_new = conv_new.max()
    improvement = M - M_new
    
    return a_new, improvement


def cosine_init(n):
    """Cosine bump initialization."""
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n + 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - center) / (sigma_frac * n / 2)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def random_smooth_init(n, rng):
    """Random smooth initialization using random Fourier series."""
    max_freq = 5
    coeffs_real = rng.standard_normal(max_freq) * 0.1
    coeffs_img = rng.standard_normal(max_freq) * 0.1
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.5
    for k in range(1, max_freq + 1):
        a += coeffs_real[k-1] * np.cos(2 * np.pi * k * x / n)
        a += coeffs_img[k-1] * np.sin(2 * np.pi * k * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def edge_concentrated_init(n, rng, edge_frac=0.3):
    """Initialization with more mass at edges."""
    x = np.arange(n, dtype=np.float64)
    # Two Gaussians at edges
    edge_width = edge_frac * n / 3
    a = np.exp(-0.5 * ((x - edge_width) / (edge_width / 2)) ** 2) + \
        np.exp(-0.5 * ((x - (n - 1 - edge_width)) / (edge_width / 2)) ** 2)
    a += rng.uniform(0.1, 0.3) * gaussian_init(n, 0.4)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def optimize_lbfgsb_smooth(a0, beta, maxiter=300, deadline=time.time() + 60):
    """L-BFGS-B optimization with softmax reparametrization."""
    n = len(a0)
    a0_clip = np.maximum(a0, 1e-30)
    a0_clip = a0_clip / a0_clip.sum()
    x0 = np.log(a0_clip)

    def obj(x):
        if time.monotonic() > deadline:
            return np.inf
        x_s = x - x.max()
        a = np.exp(x_s)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm

    def jac(x):
        x_s = x - x.max()
        a = np.exp(x_s)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        
        N = 1 << (len(w) + n - 1 + 63).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        grad_a = 2.0 * np.real(ifft(np.conj(W) * A, N))[:n]
        
        # For softmax parametrization: d(a_i)/d(x_j) = a_i*(delta_ij - a_j)
        # d(obj)/d(x_j) = sum_i d(obj)/d(a_i) * d(a_i)/d(x_j)
        # = sum_i grad_a[i] * a[i]*(delta_ij - a_j)
        # = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
        lagrange = np.dot(a, grad_a)
        grad_x = a * (grad_a - lagrange)
        return grad_x

    result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 30})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
combined with smooth gradient optimization. The LP approach at each step solves
a linear program to find the "flattest" non-negative sequence compatible with
the current peak bound on conv(a, b), then blends to monotonically reduce the
peak. Multi-scale initialization, diverse starting points, and gradient-based
polish provide final refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    result = np.real(ifft(A * A, fft_size))[:size]
    return result


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.sum(cond)
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def optimize_pg_smooth(a0, beta, lr, n_iter, deadline):
    """Projected gradient descent on simplex with smooth max objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    best_obj = np.inf
    best_a = a.copy()

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if obj < 1.01:
            break

        N = 1 << (len(w) + n - 1 + 63).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        grad = 2.0 * np.real(ifft(np.conj(W) * A, N))[:n]

        if np.all(np.abs(grad) < 1e-15):
            break

        t = lr
        a_new = a.copy()
        for _ in range(8):
            a_new = project_simplex(a - t * grad)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5
        a = a_new
        lr *= 0.999

    return best_a, best_obj


def build_lp_matrix(a, n):
    """Build the constraint matrix for the LP. A[k,j] = a[k-j] if 0<=k-j<n."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n - 1, 2 * n - 2)
        A[j:k_end + 1, j] = a[0:k_end - j + 1]
    return A


def lp_refine_step(a, blending, target_M=None):
    """
    One LP-based refinement step. Solves LP to find max-sum b >= 0
    with conv(a,b) <= M, scales to sum=1, and blends with a.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), 0.0
    a = a / s

    conv_aa = fast_convolve(a)
    M = conv_aa.max() if target_M is None else target_M
    if M < 1e-15:
        return a.copy(), 0.0

    A = build_lp_matrix(a, n)
    A_ub = np.vstack([A, -A])
    b_ub = np.full(2 * (2 * n - 1), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
    except Exception:
        return a.copy(), 0.0

    if not result.success:
        return a.copy(), 0.0

    b_opt = np.maximum(result.x, 0)
    sum_b = b_opt.sum()
    if sum_b < 1e-15:
        return a.copy(), 0.0

    g = b_opt / sum_b
    a_new = (1 - blending) * a + blending * g
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a.copy(), 0.0
    a_new = a_new / s_new

    conv_new = fast_convolve(a_new)
    M_new = conv_new.max()
    improvement = M - M_new
    return a_new, improvement


def optimize_lbfgsb_smooth(a0, beta, maxiter, deadline):
    """L-BFGS-B optimization with softmax reparametrization."""
    n = len(a0)
    a0_clip = np.maximum(a0, 1e-30)
    a0_clip = a0_clip / a0_clip.sum()
    x0 = np.log(a0_clip)

    def obj_fn(x):
        x_s = x - x.max()
        a = np.exp(x_s)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm

    def jac_fn(x):
        x_s = x - x.max()
        a = np.exp(x_s)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        N = 1 << (len(w) + n - 1 + 63).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        grad_a = 2.0 * np.real(ifft(np.conj(W) * A, N))[:n]
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)

    try:
        result = minimize(obj_fn, x0, jac=jac_fn, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxls': 30})
        x_opt = result.x
    except Exception:
        return a0_clip

    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def cosine_init(n):
    """Cosine bump initialization."""
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n + 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def gaussian_init(n, sigma_frac):
    """Gaussian initialization."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - center) / (sigma_frac * n / 2)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def random_smooth_init(n, rng):
    """Random smooth initialization via low-frequency Fourier components."""
    max_freq = 4
    a = np.ones(n) * 0.5
    x = np.arange(n, dtype=np.float64)
    for k in range(1, max_freq + 1):
        a += rng.standard_normal() * 0.08 * np.cos(2 * np.pi * k * x / n)
        a += rng.standard_normal() * 0.08 * np.sin(2 * np.pi * k * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def edge_init(n, rng, edge_frac):
    """Edge-concentrated initialization."""
    x = np.arange(n, dtype=np.float64)
    edge_width = edge_frac * n / 3
    a = np.exp(-0.5 * ((x - edge_width) / max(edge_width / 2, 1)) ** 2) + \
        np.exp(-0.5 * ((x - (n - 1 - edge_width)) / max(edge_width / 2, 1)) ** 2)
    a += 0.2 * gaussian_init(n, 0.4)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    inits['rand_smooth'] = random_smooth_init(n, rng)
    inits['edge'] = edge_init(n, rng, 0.35)
    inits['rand_dir'] = rng.dirichlet(np.ones(n) * 0.3)

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-15:
            pp = pp / ps
            if len(pp) == n:
                inits['parent'] = pp.copy()
            else:
                pp_i = np.interp(np.linspace(0, 1, n),
                                 np.linspace(0, 1, len(pp)), pp)
                pp_i = np.maximum(pp_i, 0)
                ps2 = pp_i.sum()
                if ps2 > 1e-15:
                    inits['parent_res'] = pp_i / ps2
    return inits


def score_sequence(a):
    """Evaluate sequence using the provided function."""
    if a is None or len(a) == 0:
        return float('inf')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    try:
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')


def proxy_score(a):
    """Fast proxy for the objective (assumes sum(a) close to 1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def propose_candidate(budget_s=1800, **kwargs):
    """
    Main search: LP-based iterative refinement + smooth gradient polish.
    Multi-phase: init scan, LP refinement, gradient optimization, final check.
    """
    rng = np.random.default_rng(777)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent loaded: n=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    best_score = float('inf')
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        if a is None:
            return
        sc = score_sequence(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print("  NEW BEST: %.6f (n=%d, t=%.0fs)" % (sc, len(a), time.monotonic() - t_start))

    # === Phase 1: Quick multi-length initialization ===
    print("=== Phase 1: Initialization Scan ===")
    phase1_end = t_start + 50.0

    lengths = [128, 192, 256, 320, 384, 512]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths = sorted(set(lengths + [pn]))

    candidates = []
    for n in lengths:
        if time.monotonic() > phase1_end - 5:
            break
        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'parent_res', 'cosine', 'gauss_mid', 'rand_smooth',
                    'edge', 'uniform', 'gauss_narrow', 'rand_dir', 'gauss_wide']
        names = [nm for nm in priority if nm in inits]
        for nm in names[:3]:
            if time.monotonic() > phase1_end - 2:
                break
            a0 = inits[nm]
            dl = min(time.monotonic() + 6, phase1_end - 1)
            a_opt, _ = optimize_pg_smooth(a0, 40.0, 0.06, 150, dl)
            candidates.append((n, a_opt))
            update_best(a_opt)

    candidates.sort(key=lambda x: proxy_score(x[1]))
    print("Phase 1 done. Best proxy=%.6f, %d candidates" % (
        proxy_score(candidates[0][1]) if candidates else float('inf'), len(candidates)))

    # Select diverse top candidates
    top_cands = []
    seen_n = set()
    for n, a in candidates:
        if n not in seen_n:
            top_cands.append((n, a.copy()))
            seen_n.add(n)
        if len(top_cands) >= 4:
            break
    if not top_cands and candidates:
        top_cands.append((candidates[0][0], candidates[0][1].copy()))

    # === Phase 2: LP-based iterative refinement ===
    print("=== Phase 2: LP Refinement ===")
    phase2_end = deadline - 350.0

    for idx, (n, a_init) in enumerate(top_cands):
        if time.monotonic() > phase2_end - 20:
            break
        print("  Candidate %d: n=%d, proxy=%.6f" % (idx, n, proxy_score(a_init)))

        a_cur = a_init / a_init.sum()
        M_prev = fast_convolve(a_cur).max()
        no_improve = 0
        blend = 0.5

        while time.monotonic() < phase2_end - 10:
            a_new, imp = lp_refine_step(a_cur, blend)
            M_cur = fast_convolve(a_new).max()

            if imp < 1e-9 * M_prev:
                no_improve += 1
                # Try different blending
                blend = 0.3 if blend > 0.4 else 0.7
                if no_improve > 5:
                    # Perturb and retry
                    noise = rng.standard_normal(n) * 0.005
                    a_cur = project_simplex(a_cur + noise)
                    no_improve = 0
                    blend = 0.5
            else:
                no_improve = 0
                a_cur = a_new
                M_prev = M_cur
                update_best(a_cur)

    print("Phase 2 done. Best: %.6f" % best_score)

    # === Phase 3: Gradient polish on best ===
    print("=== Phase 3: Gradient Polish ===")
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    n_best = len(best_seq)
    a_cur = best_seq / best_seq.sum()
    remaining = deadline - time.monotonic() - 10.0

    # 3a: L-BFGS-B schedule with decreasing beta
    betas = [25.0, 12.0, 6.0, 3.0, 1.5]
    for beta in betas:
        if remaining < 8:
            break
        t_alloc = min(remaining * 0.25, 50.0)
        if t_alloc < 5:
            break
        maxit = int(min(400, max(50, t_alloc * 6)))
        a_opt = optimize_lbfgsb_smooth(a_cur, beta, maxit, time.monotonic() + t_alloc)
        update_best(a_opt)
        a_cur = a_opt
        remaining = deadline - time.monotonic() - 10.0

    # 3b: PG with low beta
    if remaining > 20:
        t_alloc = min(remaining * 0.3, 80.0)
        n_it = int(min(2000, t_alloc * 5))
        a_opt, _ = optimize_pg_smooth(a_cur, 5.0, 0.02, n_it, time.monotonic() + t_alloc)
        update_best(a_opt)
        a_cur = a_opt
        remaining = deadline - time.monotonic() - 10.0

    # 3c: Random restarts from best
    if remaining > 30:
        n_restarts = int(min(10, remaining / 35))
        for i in range(n_restarts):
            if remaining < 20:
                break
            scale = 0.001 + 0.004 * rng.uniform()
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(a_cur + noise)
            t_alloc = min(remaining * 0.25, 35.0)
            if t_alloc < 5:
                break
            a_opt = optimize_lbfgsb_smooth(a_pert, 8.0, int(min(200, t_alloc * 5)),
                                           time.monotonic() + t_alloc)
            update_best(a_opt)
            sc = score_sequence(a_opt)
            if np.isfinite(sc) and sc < best_score:
                a_cur = a_opt.copy()
            remaining = deadline - time.monotonic() - 10.0

    # 3d: Try larger n (upscale best)
    if remaining > 80:
        for alt_n in [n_best * 2, n_best * 3 // 2]:
            if remaining < 50 or alt_n > 2048:
                break
            print("  Trying upscaled n=%d" % alt_n)
            pp = np.interp(np.linspace(0, 1, alt_n),
                          np.linspace(0, 1, n_best), a_cur)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps < 1e-15:
                continue
            pp /= ps
            # Quick LP refinement
            for _ in range(40):
                if remaining < 30:
                    break
                pp, _ = lp_refine_step(pp, 0.5)
            update_best(pp)
            remaining = deadline - time.monotonic() - 10.0
            if remaining > 30:
                a_opt = optimize_lbfgsb_smooth(pp, 10.0, 200, time.monotonic() + min(remaining - 15, 50))
                update_best(a_opt)
                remaining = deadline - time.monotonic() - 10.0

    # 3e: Final LP polish on absolute best
    if remaining > 15 and best_seq is not None:
        a_final = best_seq / best_seq.sum()
        blend_seq = [0.3, 0.4, 0.5, 0.6, 0.7]
        for bl in blend_seq:
            if remaining < 8:
                break
            for _ in range(30):
                if remaining < 5:
                    break
                a_final, _ = lp_refine_step(a_final, bl)
            update_best(a_final)
            remaining = deadline - time.monotonic() - 10.0

    # === Final ===
    if best_seq is not None:
        final_sc = score_sequence(best_seq)
        if np.isfinite(final_sc):
            best_score = min(best_score, final_sc)
        print("\nFINAL: %.6f (n=%d, total_time=%.0fs)" % (
            best_score, len(best_seq), time.monotonic() - t_start))
    else:
        best_seq = np.ones(128) / 128
        best_score = score_sequence(best_seq)
        print("\nFALLBACK: %.6f" % best_score)

    return [float(x) for x in best_seq]
```
