    (B_k * a)_j = sum_i (B_k)_{ji} * a[i] = a[k-j] (if 0 <= k-j < n), else 0.
    
    So grad_j = 2 * a[k-j] (if 0 <= k-j < n), else 0.
    
    Let me verify: for k=0, grad_j = 2*a[0] if 0<=-j<n (only j=0), else 0.
    conv[0] = a[0]^2, d/da[0] = 2*a[0]. Correct!
    
    For k=1, grad_j = 2*a[1-j] if 0<=1-j<n. So grad_0 = 2*a[1], grad_1 = 2*a[0].
    conv[1] = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1].
    d/da[0] = 2*a[1], d/da[1] = 2*a[0]. Correct!
    
    Great, so grad_j = 2*a[k-j] if 0 <= k-j < n, else 0.
    This means grad = 2 * (shifted and masked version of a).
    """
    n = len(a)
    grad = np.zeros(n)
    # j ranges from 0 to n-1, need 0 <= k-j < n, i.e., k-n+1 <= j <= k
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        grad[j_start:j_end+1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
    return grad


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of weighted sum of conv values."""
    n = len(a)
    # conv[k] for k in range(2n-1), weights w[k]
    # grad_j = 2 * sum_k w[k] * a[k-j] (if 0 <= k-j < n)
    # = 2 * (w * a)[j] where the convolution is circular... no, linear.
    # Actually: grad_j = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
    # Let m = k-j, then k = j+m, so: grad_j = 2 * sum_{m: 0<=m<n, 0<=j+m<2n-1} w[j+m] * a[m]
    # = 2 * (cross-correlation of w and a)[j]
    
    # Use FFT for efficiency
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    # We want sum_k w[k] * a[k-j] for each j
    # This is the cross-correlation: (w ⋆ a)[j] = sum_k w[k] * a[k-j]
    # In FFT: ifft(conj(fft(w)) * fft(a)) gives correlation
    # Actually: ifft(fft(w) * fft(a)) gives convolution (w * a)
    # Cross-correlation (w ⋆ a)[j] = sum_k w[k+j] * a[k] = (w_rev * a)[j + len(w) - 1]
    
    # Simpler: just use the formula grad_j = 2 * sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w[k] * a[k-j]
    # This is 2 * (w * a)_reversed... let me just do it directly with FFT.
    
    # conv_weighted[j] = sum_k w[k] * a[k-j] for valid k
    # = cross_correlation(w, a)[j]
    # cross_correlation(w, a)[j] = sum_k w[k] * conj(a[k-j])
    # For real: = sum_k w[k] * a[k-j]
    
    # Using FFT: cross_corr = ifft(conj(FFT(w)) * FFT(a)) ... no
    # Let's use: ifft(FFT(w) * conj(FFT(a))) gives (w * a_flipped)[n-1-j]... 
    
    # Simplest correct approach:
    # grad_j = 2 * sum_{m=0}^{n-1} a[m] * w[j+m] (where w is zero-padded)
    # This is the convolution of a with w (flipped)... 
    
    # Let me just use: result = ifft(FFT(a) * FFT(w, N))[:n]
    # (a * w)[j] = sum_m a[m] * w[j-m] ... that's not quite right either.
    
    # OK let me think more carefully.
    # grad_j = 2 * sum_{k} w[k] * a[k-j] where 0<=k<2n-1 and 0<=k-j<n
    # Let t = k-j, so k = t+j:
    # grad_j = 2 * sum_{t=0}^{n-1} a[t] * w[t+j] (where w is defined for 0<=t+j<2n-1, else 0)
    
    # This is 2 * (a * w_padded)[j + n - 1] where w_padded is w zero-padded to length 2n-1
    # Wait: (a * w)[s] = sum_t a[t] * w[s-t]
    # We want sum_t a[t] * w[t+j] = sum_t a[t] * w[(t+j)] 
    # = (a_rev * w)[(n-1) + j] where a_rev is a reversed... 
    
    # I'm overcomplicating this. Let me just use the direct formula with FFT:
    # Define b as the zero-padded version of a (length 2n-1)
    # Then grad_j = 2 * sum_{t=0}^{n-1} a[t] * w[t+j]
    #            = 2 * (a * w)[n-1+j] ... no.
    # (a * w)[s] = sum_t a[t] * w[s-t] for t=0..n-1 (with w zero outside [0,2n-2])
    # We want sum_t a[t] * w[t+j] = sum_t a[t] * w[(-t) + (j+2t)] ... no.
    
    # Let's define c = a reversed: c[i] = a[n-1-i]
    # sum_t a[t] * w[t+j] = sum_t c[n-1-t] * w[t+j]
    # Let u = n-1-t: = sum_u c[u] * w[n-1-u+j] = (c * w)[n-1+j]
    # So grad_j = 2 * (c * w)[n-1+j] where c = a[::-1]
    
    c = a[::-1]
    # (c * w)[s] for s = n-1, n, ..., 2n-2
    # Use FFT:
    N2 = 1 << (n + len(w) - 1 + 63).bit_length()
    C = fft(c, N2)
    Wf = fft(w, N2)
    conv_cw = np.real(ifft(C * Wf, N2))
    # grad_j = 2 * conv_cw[n-1+j]
    grad = 2.0 * conv_cw[n-1:n-1+n]

def objective_and_grad(a, beta=None):

def subgradient_optimize(a0, n_iters=2000, time_budget=60.0):
    """
    Subgradient descent with Barzilai-Borwein step size on the simplex.
    Minimizes max_k conv[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    
    t = 0.1  # initial step size
    a_prev = a.copy()
    conv_val_prev = best_val
    
    t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t_start > time_budget:
            break
        
        conv = fast_convolve(a)
        val = conv.max()
        
        if val < best_val:
            best_val = val
            best_a = a.copy()
        
        if val < 1e-12:
            break
        
        k_max = int(np.argmax(conv))
        grad = grad_conv_at_k(a, k_max)
        
        # Projected gradient step
        # We want to minimize val = a^T B_k a subject to sum(a)=1, a>=0
        # grad = 2 * B_k * a
        
        # Barzilai-Borwein step size
        if it > 0:
            s = a - a_prev
            y = val - conv_val_prev
            ss = np.dot(s, s)
            sy = np.dot(s, y)
            if ss > 1e-30 and abs(sy) > 1e-15:
                t_bb = ss / abs(sy)
                t = 0.5 * (t + t_bb)  # average
            t = min(max(t, 1e-8), 1.0)
        
        a_prev = a.copy()
        conv_val_prev = val
        
        # Backtracking line search
        for _ in range(10):
            a_new = project_simplex(a - t * grad)
            conv_new = fast_convolve(a_new)
            val_new = conv_new.max()
            if val_new <= val + 1e-12:
                break
            t *= 0.5
        
        a = a_new
    
    return best_a, best_val


def symmetric_subgradient_optimize(a0, n_iters=2000, time_budget=60.0):

    s = a.sum()

        a /= s

    hs = half_vec.sum()

        # sum(a) = 2 * sum(half_vec)

    else:
        # The middle element appears once
        # sum(a) = 2 * sum(half_vec) - half_vec[half-1]
        pass

    best_a = a.copy()

def lbfgsb_smooth(a0, beta, maxiter=300):
    """L-BFGS-B with softmax parameterization and smooth max."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    
    def obj(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm
    
    def jac(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        grad_a = compute_grad_smooth_conv(a, w)
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)
    
    result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, 2.0 * n * fast_convolve(a_opt).max()


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def random_phase_init(n, rng):
    """Initialize with random phase in frequency domain."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
    for k in range(1, fft_size // 2):
        A[k] = np.exp(1j * phases[k - 1])
        A[fft_size - k] = np.exp(-1j * phases[k - 1])
    if fft_size % 2 == 0:
        A[fft_size // 2] = np.exp(1j * phases[-1])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact subgradient descent
with Barzilai-Borwein step sizes, symmetric optimization, accelerated
proximal gradient, multi-length exploration, and L-BFGS-B polishing.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_k(a, k):
    """Gradient of conv[k] w.r.t. a. Result: 2*a[k-j] for valid j."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        grad[j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
    return grad


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a using FFT."""
    n = len(a)
    c = a[::-1]
    N = 1 << (n + len(w) - 1 + 63).bit_length()
    C = fft(c, N)
    Wf = fft(w, N)
    conv_cw = np.real(ifft(C * Wf, N))
    grad = 2.0 * conv_cw[n - 1:n - 1 + n]
    return grad


def objective_exact(a):
    """Compute exact max of convolution."""
    return fast_convolve(a).max()


def subgradient_optimize(a0, n_iters=10000, time_budget=60.0, use_momentum=False):
    """
    Subgradient descent with BB step size on simplex.
    Optionally uses accelerated (FIST) momentum.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_val = fast_convolve(a).max()

    t = 0.1
    a_prev_step = a.copy()
    val_prev = best_val
    t_start = time.monotonic()

    if use_momentum:
        a_momentum = a.copy()
        a_prev_momentum = a.copy()

    total_iters = 0
    for it in range(n_iters):
        if time.monotonic() - t_start > time_budget:
            break

        conv = fast_convolve(a)
        val = conv.max()

        if val < best_val:
            best_val = val
            best_a = a.copy()

        if val < 1e-14:
            break

        k_max = int(np.argmax(conv))
        grad = grad_conv_at_k(a, k_max)

        # Barzilai-Borwein step size
        if it > 0:
            s = a - a_prev_step
            y = val - val_prev
            ss = np.dot(s, s)
            sy = np.dot(s, y)
            if ss > 1e-30 and abs(sy) > 1e-15:
                t_bb = ss / abs(sy)
                t = 0.5 * (t + t_bb)
            t = min(max(t, 1e-7), 0.5)

        a_prev_step = a.copy()
        val_prev = val

        if use_momentum:
            # FIST/ISTA acceleration
            a_old = a.copy()
            t_step = 1.0 / (np.sqrt(4.0 * ss) + 2.0) if ss > 0 else 1.0
            t_step = min(t, 2.0 * t_step)
            a_new = project_simplex(a - t * grad)
            a_momentum = a_new + (it / (it + 3.0)) * (a_new - a_old)
            a = a_momentum
        else:
            # Standard projected subgradient with backtracking
            for _ in range(8):
                a_new = project_simplex(a - t * grad)
                conv_new = fast_convolve(a_new)
                val_new = conv_new.max()
                if val_new <= val + 1e-13:
                    break
                t *= 0.5
            a = a_new
            # Slowly decrease step size
            t *= 0.9998

        total_iters += 1

    return best_a, best_val


def symmetric_subgradient_optimize(a0, n_iters=10000, time_budget=60.0):
    """Subgradient descent enforcing symmetry a[i] = a[n-1-i]."""
    n = len(a0)
    a = (a0 + a0[::-1]) / 2.0
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_val = fast_convolve(a).max()

    t = 0.05
    a_prev_step = a.copy()
    val_prev = best_val
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_budget:
            break

        conv = fast_convolve(a)
        val = conv.max()

        if val < best_val:
            best_val = val
            best_a = a.copy()

        if val < 1e-14:
            break

        k_max = int(np.argmax(conv))
        grad = grad_conv_at_k(a, k_max)
        grad = (grad + grad[::-1]) / 2.0

        if it > 0:
            s_vec = a - a_prev_step
            y = val - val_prev
            ss = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y)
            if ss > 1e-30 and abs(sy) > 1e-15:
                t_bb = ss / abs(sy)
                t = 0.5 * (t + t_bb)
            t = min(max(t, 1e-7), 0.5)

        a_prev_step = a.copy()
        val_prev = val

        for _ in range(8):
            a_new = a - t * grad
            a_new = (a_new + a_new[::-1]) / 2.0
            a_new = project_simplex(a_new)
            a_new = (a_new + a_new[::-1]) / 2.0
            s2 = a_new.sum()
            if s2 > 1e-10:
                a_new /= s2
            conv_new = fast_convolve(a_new)
            val_new = conv_new.max()
            if val_new <= val + 1e-13:
                break
            t *= 0.5

        a = a_new
        t *= 0.9998

    return best_a, best_val


def lbfgsb_smooth(a0, beta, maxiter=300):
    """L-BFGS-B with softmax parameterization and smooth max."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)

    def obj_func(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm

    def jac_func(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        grad_a = compute_grad_smooth_conv(a, w)
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)

    result = minimize(obj_func, x0, jac=jac_func, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})

    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def random_phase_init(n, rng):
    """Initialize with random phase in frequency domain (flat spectrum)."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
    for k in range(1, fft_size // 2):
        A[k] = np.exp(1j * phases[k - 1])
        A[fft_size - k] = np.exp(-1j * phases[k - 1])
    if fft_size % 2 == 0:
        A[fft_size // 2] = np.exp(1j * phases[-1])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def peak_shave(a, n_steps=100, time_budget=30.0):
    """Iteratively shave the peak of the convolution."""
    n = len(a)

        grad = grad_conv_at_k(a, k_max)

        # Reduce along gradient direction

        a_new = a - max_reduction * grad / (np.linalg.norm(grad) + 1e-15)
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new)
        if conv_new.max() < val:
            a = a_new
        else:
            # Try smaller step

            if fast_convolve(a_new).max() < val:

                if total_c > 1e-15:

        if time.monotonic() < phase2_deadline - 15:
            base = a_opt_s
            for beta in [30.0, 10.0]:
                if time.monotonic() > phase2_deadline - 10:
                    break
                remaining_here = phase2_deadline - time.monotonic() - 10
                if remaining_here < 8:
                    break
                maxiter = int(min(300, remaining_here * 4))
                a_lb = lbfgsb_smooth(base, beta, maxiter=maxiter)
                score_and_update(a_lb)
                if best_sequence is not None and len(best_sequence) == n:
                    base = best_sequence / best_sequence.sum()
                else:
                    base = a_lb

    print("After Phase 2: best=%.6f" % best_score)

    # Phase 3: Intensive polishing on the best
    print("=== Phase 3: Polishing ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        base = best_sequence / best_sequence.sum()

        # Multi-beta L-BFGS-B schedule
        for beta in [50.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.0]:
            remaining = deadline - time.monotonic() - 30
            if remaining < 10:
                break
            t_alloc = min(80.0, remaining)
            maxiter = int(min(500, t_alloc * 5))
            if maxiter < 30:
                break
            a_opt = lbfgsb_smooth(base, beta, maxiter=maxiter)
            score = score_and_update(a_opt)
            if best_sequence is not None:
                base = best_sequence / best_sequence.sum()

        # More subgradient rounds
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            a_opt, _ = subgradient_optimize(base, n_iters=20000, time_budget=remaining * 0.5)
            score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 30

        # Symmetric round
        if remaining > 20:
            a_sym = (base + base[::-1]) / 2.0
            a_sym /= a_sym.sum()
            a_opt_s, _ = symmetric_subgradient_optimize(a_sym, n_iters=20000, time_budget=remaining * 0.4)
            score_and_update(a_opt_s)
            remaining = deadline - time.monotonic() - 30

        # Random perturbations
        if remaining > 30:
            n_pert = min(8, int(remaining / 25))
            for i in range(n_pert):
                if remaining < 12:
                    break
                scale = 0.005 + 0.01 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(base + noise)
                t_alloc = min(20.0, remaining)
                a_opt_p, _ = subgradient_optimize(a_pert, n_iters=5000, time_budget=t_alloc * 0.6)
                score_and_update(a_opt_p)
                if best_sequence is not None:
                    base = best_sequence / best_sequence.sum()
                # Quick L-BFGS-B polish on perturbation result
                if time.monotonic() < deadline - 15:
                    maxiter = int(min(200, (deadline - time.monotonic() - 15) * 3))
                    if maxiter > 30:
                        a_lb = lbfgsb_smooth(a_opt_p, 10.0, maxiter=maxiter)
                        score_and_update(a_lb)
                        if best_sequence is not None:
                            base = best_sequence / best_sequence.sum()
                remaining = deadline - time.monotonic() - 30

        # Final polish
        remaining = deadline - time.monotonic() - 10
        if remaining > 10 and best_sequence is not None:
            base = best_sequence / best_sequence.sum()
            maxiter = int(min(500, remaining * 5))
            a_final = lbfgsb_smooth(base, 5.0, maxiter=maxiter)
            score_and_update(a_final)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        score_and_update(best_sequence)

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score=%.6f, n=%d, time=%.0fs" % (best_score, len(best_sequence), time.monotonic() - t_start))
    return [float(x) for x in best_sequence]
