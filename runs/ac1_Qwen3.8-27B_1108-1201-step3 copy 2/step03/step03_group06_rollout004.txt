def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix_full(a):
    """Build full constraint matrix for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp_full(a, M):
    """Solve full LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k."""
    n = len(a)

    for j, w in enumerate(peak_indices):

        # Actually peak_indices should be array of peak positions
    # Redo properly
    grad = np.zeros(n)
    for k_pos, w in zip(peak_indices, weights):
        k = int(k_pos)

    best_a = a.copy()
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead

    beta = beta_init
    lr = lr_init
    no_improve = 0
    mu = 0.5  # Nesterov parameter

    while time.monotonic() - t0 < time_limit:
        # Nesterov step
        y = a + mu * (a - (a - v / lr)) if lr > 0 else a.copy()
        # Simpler Nesterov: y = a + mu * v / lr (if lr > 0)
        if lr > 1e-10:
            y = a + mu * v / lr
            s_y = y.sum()

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n normalized to sum=1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal (edges) - often good for convolution
    for edge_frac in [0.2, 0.25, 0.3, 0.4]:
        edge = max(2, int(n * edge_frac))
        bi = np.zeros(n)
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        bi /= bi.sum()
        inits[f'bimodal_{edge_frac}'] = bi

    # Ramps
    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    inits['ramp_inc'] = ramp_inc
    inits['ramp_dec'] = ramp_inc[::-1].copy()

    # Gaussians
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Cosine
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
        inits['cosine'] = cos_w

    # Random
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['random'] = rng.dirichlet(np.ones(n) * 1.0)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

        time_per_init = max(8.0, (remaining - 10) / max(1, len(lengths) - ni) / max(1, len(init_names)))

        for name in init_names:
            remaining = phase1_time - (time.monotonic() - t1)
            if remaining < 10:
                break
            t_alloc = min(time_per_init, remaining - 5)
            if t_alloc < 5:
                break

            a0 = inits[name].copy()

            # Quick gradient optimization
            a_opt, obj_opt = gradient_optimize(a0, time_limit=t_alloc,
                                                beta_init=300, lr_init=0.1,
                                                momentum=0.9, top_k=max(20, n // 10),
                                                seed=n + hash(name) % 10000)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, score={score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}", flush=True)

    # ========== Phase 2: LP Deep Optimization on Best Candidates ==========
    print("=== Phase 2: LP Deep Optimization ===", flush=True)
    t2 = time.monotonic()
    phase2_time = 550.0

    # Collect top candidates
    deep_starts = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results[:6]:
        key = (n, name)
        if key not in seen_keys:
            deep_starts.append((n, a_opt.copy()))
            seen_keys.add(key)
    if best_sequence is not None:
        deep_starts.append((len(best_sequence), best_sequence.copy()))

    # Also add fresh starts at promising lengths
    for n_try in [256, 320, 384]:
        if any(d[0] == n_try for d in deep_starts):
            continue
        inits = generate_initializations(n_try, parent, rng)
        if 'bimodal_0.25' in inits:
            deep_starts.append((n_try, inits['bimodal_0.25'].copy()))
        elif 'uniform' in inits:
            deep_starts.append((n_try, inits['uniform'].copy()))

    deep_starts = deep_starts[:8]

    # Determine how much time per run
    n_runs = max(1, len(deep_starts))
    time_per_run = max(30.0, (phase2_time - 30) / n_runs)

    for i, (n_start, a_start) in enumerate(deep_starts):
        remaining = phase2_time - (time.monotonic() - t2)
        if remaining < 30:
            break
        t_alloc = min(time_per_run, remaining - 10)
        if t_alloc < 20:
            break

        # Full LP optimization
        a_opt, obj_opt = lp_optimize_full(a_start, time_limit=t_alloc,
                                           full_lp=True, seed=i * 7 + 3)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP [{i}]: n={n_start}, score={score:.6f}", flush=True)

    print(f"Phase 2 done. Best: {best_score:.6f}", flush=True)

    # ========== Phase 3: Gradient Polish ==========
    print("=== Phase 3: Gradient Polish ===", flush=True)
    t3 = time.monotonic()
    phase3_time = 350.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        grad_configs = [
            {'beta_init': 200, 'lr_init': 0.08, 'momentum': 0.85, 'top_k': 30},
            {'beta_init': 500, 'lr_init': 0.05, 'momentum': 0.9, 'top_k': 20},
            {'beta_init': 1000, 'lr_init': 0.03, 'momentum': 0.92, 'top_k': 15},
            {'beta_init': 800, 'lr_init': 0.04, 'momentum': 0.9, 'top_k': 40},
            {'beta_init': 300, 'lr_init': 0.06, 'momentum': 0.88, 'top_k': 50},
        ]

        time_per_grad = max(20.0, (phase3_time - 20) / (len(grad_configs) + 3))

        for i, cfg in enumerate(grad_configs):
            remaining = phase3_time - (time.monotonic() - t3)
            if remaining < 20:
                break
            t_alloc = min(time_per_grad, remaining - 10)
            if t_alloc < 15:
                break

            a_opt, obj_opt = gradient_optimize(best_sequence, time_limit=t_alloc,
                                                seed=i * 13 + 1, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad [{i}]: score={score:.6f}", flush=True)

        # Perturbed gradient runs
        for i in range(3):
            remaining = phase3_time - (time.monotonic() - t3)
            if remaining < 15:
                break
            t_alloc = min(time_per_grad / 2, remaining - 5)
            if t_alloc < 10:
                break
            noise = rng.standard_normal(n_best) * 0.008
            a_pert = project_simplex(best_sequence + noise)
            a_opt, obj_opt = gradient_optimize(a_pert, time_limit=t_alloc,
                                                beta_init=500, lr_init=0.04,
                                                momentum=0.9, top_k=25, seed=100 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad pert [{i}]: score={score:.6f}", flush=True)

    print(f"Phase 3 done. Best: {best_score:.6f}", flush=True)

    # ========== Phase 4: LP on Best + Different Lengths ==========
    print("=== Phase 4: Additional LP Exploration ===", flush=True)
    t4 = time.monotonic()
    phase4_time = 300.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Run LP on best sequence with full constraints
        remaining = phase4_time - (time.monotonic() - t4)
        t_alloc = min(120.0, remaining - 20)
        if t_alloc > 20:
            a_opt, obj_opt = lp_optimize_full(best_sequence, time_limit=t_alloc,
                                               full_lp=True, seed=500)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP best: score={score:.6f}", flush=True)

        # Try adjacent lengths
        for n_adj in [n_best - 32, n_best + 32, n_best - 64, n_best + 64]:
            remaining = phase4_time - (time.monotonic() - t4)
            if remaining < 40 or n_adj < 64:
                continue
            # Interpolate best to new length
            pp = np.interp(np.linspace(0, 1, n_adj),
                           np.linspace(0, 1, n_best), best_sequence)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                pp = pp / ps

            t_alloc = min(60.0, remaining - 20)
            if t_alloc < 20:
                continue

            a_opt, obj_opt = lp_optimize_full(pp, time_limit=t_alloc,
                                               full_lp=True, seed=n_adj)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  LP n={n_adj}: score={score:.6f}", flush=True)
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    print(f"Phase 4 done. Best: {best_score:.6f}", flush=True)

    # ========== Phase 5: Final Polish ==========
    print("=== Phase 5: Final Polish ===", flush=True)
    t5 = time.monotonic()
    phase5_time = max(10.0, deadline - time.monotonic())

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Final gradient with very sharp beta
        t_alloc = min(phase5_time * 0.3, 80.0)
        if t_alloc > 10:
            a_opt, obj_opt = gradient_optimize(best_sequence, time_limit=t_alloc,
                                                beta_init=2000, lr_init=0.02,
                                                momentum=0.93, top_k=10, seed=777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final grad: score={score:.6f}", flush=True)

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 20:
            attempts = 0
            while time.monotonic() < deadline - 25 and attempts < 500:
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    attempts += 1
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Perturb s={scale}: score={score:.6f}", flush=True)
                        break
                if time.monotonic() > deadline - 30:
                    break

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    else:
        if final_score < best_score:
            best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}", flush=True)
    return [float(x) for x in best_sequence]
