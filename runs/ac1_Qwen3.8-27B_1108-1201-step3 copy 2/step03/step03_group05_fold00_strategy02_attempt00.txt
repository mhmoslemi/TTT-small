import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    #---Security Checks---#
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    if n == 0:
        return np.inf
    sum_a = np.sum(sequence)
    # Minimum sum check
    if sum_a < 0.01:
        return np.inf
    # Compute convolution
    conv = np.convolve(sequence, sequence, mode='full')
    max_b = np.max(conv)
    return max_b / (sum_a**2)

def smoothed_convolve(a, beta):

    Includes a shift to the 'full' convolution and scaling for stability."""
    n = len(a)
    # Shift the sequence to have non-negative convolution
    conv_out = np.convolve(a, a, mode='full')
    # Scale by β to adjust the 'roughness' of the function
    conv_out = conv_out * beta
    # Calculate the shift to avoid negative values
    shift = np.max(conv_out) * (1 - 1/(n**2))
    # Only keep high part of the convolution for efficiency
    return conv_out[shift:] if n > 1 else conv_out
