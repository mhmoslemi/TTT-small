    A_ub[:, :n] = conv_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True,
                           'dual_feasibility_tolerance': 1e-12,
                           'primal_feasibility_tolerance': 1e-12})
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        M_val = res.x[n]
        s = b.sum()
        if s > 1e-15:
            b /= s
        return b, M_val
    return None, None


def lp_max_sum(a, M, conv_mat):
    """Solve: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=conv_mat, b_ub=b_ub, bounds=bounds,
                  method='highs',
                  options={'presolve': True,
                           'dual_feasibility_tolerance': 1e-12,
                           'primal_feasibility_tolerance': 1e-12})
    if res.status == 0:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-15:
            b /= s
        return b
    return None


def lp_refine_with_line_search(a0, time_limit, conv_mat, max_iters=2000,
                               step_init=0.3, step_min=0.01,
                               use_min_peak=True):
    """
    LP-based iterative refinement with backtracking line search.
    At each step, solve an LP to get a direction, then do a line search
    to find the optimal step size along that direction.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj, _ = compute_obj_normalized(a)
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    cur_method = 'min_peak' if use_min_peak else 'max_sum'

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        M = conv.max()

        if cur_method == 'min_peak':
            b, M_lp = lp_min_peak(a, conv_mat)
            if b is None:
                cur_method = 'max_sum'
                continue
            b = b / b.sum()
        else:
            b = lp_max_sum(a, M, conv_mat)
            if b is None:
                cur_method = 'min_peak'
                continue

        if np.linalg.norm(b - a) < 1e-12:
            return best_a, best_obj

        # Line search: try multiple step sizes
        best_step_obj = best_obj
        best_step_a = a.copy()
        candidates = [step * 0.25, step * 0.5, step * 0.75, step,
                      step * 1.25, step * 1.5]
        step_candidates = [t for t in candidates if 0.005 < t < 1.5]

        for t in step_candidates:
            a_try = (1.0 - t) * a + t * b
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            obj_try, _ = compute_obj_normalized(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()
                best_step_t = t

        if best_step_obj < best_obj - 1e-12:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            no_improve = 0
            step = min(1.0, best_step_t * 1.1)
        else:
            no_improve += 1
            step = max(step_min, step * 0.85)
            # Fall back to a small step if line search failed
            a = (1.0 - step) * a + step * b
            a = np.maximum(a, 0.0)
            st = a.sum()
            if st < 1e-15:
                a = best_a.copy()
                conv = fast_convolve(a)
                continue
            a = a / st
            conv = fast_convolve(a)

        if no_improve > 40:
            cur_method = 'max_sum' if cur_method == 'min_peak' else 'min_peak'
            step = step_init
            no_improve = 0

        if no_improve > 200:
            break

    return best_a, best_obj


def subgradient_step(a, conv, top_k, temperature):
    """Compute subgradient of max-k convolution using top-k peaks with soft weights."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def optimize_subgrad(a0, time_limit, conv_mat, lr_init=0.02, momentum=0.9,
                     top_k=40, temp_init=0.001, max_iters=10000):
    """Projected subgradient with momentum and adaptive parameters."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj, _ = compute_obj_normalized(a)
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        sg = subgradient_step(a, conv, top_k, temp)
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj, _ = compute_obj_normalized(a)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 1000:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            lr = lr_init * 0.5
            temp *= 0.5
        if no_improve > 5000:
            break

    return best_a, best_obj


def coordinatedescent_polish(a0, time_limit, max_rounds=20):
    """
    Coordinate descent: for each coordinate, find optimal value via 1D search
    while keeping others fixed. Uses the fact that conv(a,a) is a quadratic
    in each coordinate.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj, conv = compute_obj_normalized(a)
    best_a = a.copy()
    t0 = time.monotonic()

    rng = np.random.default_rng(777)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        # Randomize coordinate order each round
        order = rng.permutation(n)
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            # The objective as a function of a[j] (with others fixed, before re-normalization)
            # conv_k depends quadratically on a[j]:
            # conv_k = old_conv_k + a[j]*(2*sum_{j' != j} a[j']*a[k-j'] - a[j]*[j valid for conv_k])
            # After normalization, it's more complex. Let's just try a few values.
            for trial in [0.0, old_val * 0.5, old_val * 0.75, old_val * 1.25,
                          old_val * 1.5, 2.0 / n]:
                trial = max(0.0, min(10.0, trial))
                a[j] = trial
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = old_val
                    continue
                a_norm = a / s_new
                obj_try, _ = compute_obj_normalized(a_norm)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
            a[j] = old_val if not improved else a[j]
            # Reset to current best for next coordinate
            a = best_a.copy()

        if not improved:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    # Cosine-based
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        inits[f'cos_{freq}'] = a

    # Gaussian bump
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g = np.maximum(g, 0.001)
    g /= g.sum()
    inits['gauss'] = g

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sine (non-symmetric)
    s_arr = np.sin(np.pi * x / n) + 0.3 * np.sin(2 * np.pi * x / n)
    s_arr = np.maximum(s_arr, 0.01)
    s_arr /= s_arr.sum()
    inits['sine'] = s_arr

    # Exponential decay
    e_arr = np.exp(-x / (n * 0.3))
    e_arr /= e_arr.sum()
    inits['exp_decay'] = e_arr

    # Two-peak
    p1 = n // 3
    p2 = 2 * n // 3
    two = np.zeros(n)
    for p in [p1, p2]:
        two[max(0, p - 3):p + 4] = 1.0
    two = np.maximum(two, 0.01)
    two /= two.sum()
    inits['two_peak'] = two

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining LP-based refinement with
    line search, subgradient polishing, coordinate descent, and iterated
    local search with perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  ** New best: {score:.6f} (n={len(a)}, t={time.monotonic():.0f})")

    # === Phase 1: Multi-start LP exploration ===
    print("=== Phase 1: Multi-start LP ===")
    t1_start = time.monotonic()

    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(len(parent))
    lengths_to_try.extend([512, 256, 1024])
    # Deduplicate, keep in range
    seen = set()
    unique_lengths = []
    for L in lengths_to_try:
        if 64 <= L <= 2048 and L not in seen:
            unique_lengths.append(L)
            seen.add(L)
    lengths_to_try = unique_lengths

    phase1_budget = 400.0
    for n in lengths_to_try:
        if time.monotonic() + 30 > deadline - (budget_s - 20) * 0.4:
            break
        remain_p1 = min(phase1_budget, deadline - time.monotonic() - (budget_s - 20) * 0.5)
        if remain_p1 < 30:
            break

        inits = generate_inits(n, parent, rng)
        # Prioritize parent and structured inits
        priority = ['parent', 'uniform', 'sine', 'cos_1', 'random', 'gauss']
        init_names = [nm for nm in priority if nm in inits]
        init_names = init_names[:4]

        time_per_init = max(10.0, remain_p1 / (len(lengths_to_try) * len(init_names)))
        conv_mat = build_conv_matrix(np.ones(n) / n)  # placeholder, will rebuild

        for name in init_names:
            if time.monotonic() > deadline - (budget_s - 20) * 0.5:
                break
            a0 = inits[name].copy()
            # Rebuild conv matrix for this init
            conv_mat = build_conv_matrix(a0)
            t_alloc = min(time_per_init, deadline - time.monotonic() - 60)
            if t_alloc < 5:
                break

            for method_init in [True, False]:
                t_single = t_alloc / 2.0
                if t_single < 3:
                    break
                a_opt, _ = lp_refine_with_line_search(
                    a0, time_limit=t_single, conv_mat=conv_mat,
                    max_iters=1000, step_init=0.25,
                    use_min_peak=method_init)
                try_update(a_opt)
                # Update conv_mat for next iteration using the result
                if a_opt is not None:
                    conv_mat = build_conv_matrix(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # === Phase 2: Deep LP refinement on best ===
    print("=== Phase 2: Deep LP Refinement ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = deadline - 400
        n_best = len(best_sequence)
        conv_mat = build_conv_matrix(best_sequence)

        # Try multiple step sizes and methods
        configs = [
            (0.2, True), (0.3, True), (0.15, True),
            (0.2, False), (0.4, True), (0.1, True),
        ]
        for step_init, use_mp in configs:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(120.0, phase2_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break
            a_opt, _ = lp_refine_with_line_search(
                best_sequence, time_limit=t_alloc, conv_mat=conv_mat,
                max_iters=3000, step_init=step_init,
                use_min_peak=use_mp)
            try_update(a_opt)
            conv_mat = build_conv_matrix(best_sequence)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # === Phase 3: Subgradient polishing ===
    print("=== Phase 3: Subgradient Polish ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 250
        n_best = len(best_sequence)
        conv_mat = build_conv_matrix(best_sequence)

        for lr in [0.015, 0.008, 0.025]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(80.0, phase3_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break
            a_opt, _ = optimize_subgrad(
                best_sequence, time_limit=t_alloc, conv_mat=conv_mat,
                lr_init=lr, momentum=0.92, top_k=max(20, n_best // 15),
                temp_init=0.0008)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # === Phase 4: Iterated local search with perturbations ===
    print("=== Phase 4: Iterated Local Search ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 150
        n_best = len(best_sequence)
        rng_ils = np.random.default_rng(seed + 100)

        perturbations = [0.001, 0.005, 0.002, 0.01, 0.003, 0.008, 0.0015, 0.004]
        for scale in perturbations:
            if time.monotonic() > phase4_deadline:
                break
            noise = rng_ils.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            conv_mat_p = build_conv_matrix(a_pert)
            t_alloc = min(60.0, phase4_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break

            # Try both LP methods on perturbed point
            for use_mp in [True, False]:
                t_single = t_alloc / 2.0
                if t_single < 5:
                    break
                a_opt, _ = lp_refine_with_line_search(
                    a_pert, time_limit=t_single, conv_mat=conv_mat_p,
                    max_iters=1500, step_init=0.2, use_min_peak=use_mp)
                try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # === Phase 5: Coordinate descent polish ===
    print("=== Phase 5: Coordinate Descent ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        phase5_deadline = deadline - 50
        t_alloc = min(100.0, phase5_deadline - time.monotonic() - 5)
        if t_alloc > 10:
            a_opt, _ = coordinatedescent_polish(
                best_sequence, time_limit=t_alloc, max_rounds=5)
            try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # === Phase 6: Final ultra-fine LP ===
    print("=== Phase 6: Final ULF Polish ===")
    t6_start = time.monotonic()

    if best_sequence is not None:
        phase6_deadline = deadline - 5
        n_best = len(best_sequence)
        conv_mat = build_conv_matrix(best_sequence)
        t_alloc = max(5.0, phase6_deadline - time.monotonic())
        if t_alloc > 10:
            a_opt, _ = lp_refine_with_line_search(
                best_sequence, time_limit=t_alloc, conv_mat=conv_mat,
                max_iters=5000, step_init=0.05, step_min=0.005,
                use_min_peak=True)
            try_update(a_opt)

    print(f"Phase 6 done: best={best_score:.6f}, elapsed={time.monotonic()-t6_start:.0f}s")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
