"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating minimization,
frequency-domain initialization, and smooth L-BFGS-B polishing with aggressive
beta continuation.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Compute gradient of weighted convolution sum."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v = v / target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n * target_sum
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n * target_sum
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    return result * target_sum


def softmax_obj(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def softmax_jac(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=500):
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(softmax_obj, x0, jac=softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt

        # This means we want |A|^2 to be as "spread out" as possible
        # In the frequency domain, "spread out" means closer to its mean
        #
        # Actually, the circular autocorrelation being flat means |A|^2 is a 
        # delta at DC, which means a is constant. That's the wrong direction.

        # I think the issue is that "flat autocorrelation in time domain"

        # Let me abandon this approach and just use random phases.

        # 1. Compute current autocorrelation
        # 2. Find the peak
        # 3. Adjust the sequence to reduce the peak
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
            s = 1.0
        a = a / s
        
        conv = fast_convolve(a)
        peak = conv.max()
        mean = conv.mean()

        for i in range(n):

                    a[i] *= 0.99
                else:
                    a[j] *= 0.99
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s

def direct_subgradient_polish(a, n_steps=100, step_size=0.01):

        # (factor of 2 because conv_k = sum_i a_i * a_{k-i} and both terms depend on a)

def active_set_refine(a, n_iters=50):

    Active set method: identify top-K convolution lags, 
    and try to reduce them simultaneously.
    """

                break

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating iteration,
frequency-domain initialization, smooth L-BFGS-B polishing with aggressive beta
continuation, and active-set refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Compute gradient of weighted convolution sum w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v_scaled = v / target_sum
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n * target_sum
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n * target_sum
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v_scaled - theta, 0.0)
    return result * target_sum


def _softmax_obj(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _softmax_jac(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """Optimize on simplex using L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(_softmax_obj, x0, jac=_softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def solve_lp_complement(f, M, n):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)_k <= M for all k.
    Returns normalized g (sum=1) or None.
    """
    # Constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # (f*g)_k = sum_j A[k,j] * g_j
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]

    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
    except Exception:
        return None

    if res.status != 0:
        return None

    g = res.x
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def freq_domain_init(n, rng, n_trials=40):
    """Initialize using random phase designs with non-negativity projection."""
    best_a = np.ones(n) / n
    best_obj = float('inf')
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    for _ in range(n_trials):
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k])
            A[fft_size - k] = np.exp(-1j * phases[k])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * phases[fft_size // 2])

        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def lp_iteration_loop(a, max_iters, time_limit):
    """Run LP-based alternating iteration."""
    n = len(a)
    a_work = a.copy() / a.sum()
    best_a = a_work.copy()
    best_M = fast_convolve(a_work).max()
    t_step = 0.15
    t_start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break

        conv = fast_convolve(a_work)
        M = conv.max()

        if M < best_M:
            best_M = M
            best_a = a_work.copy()

        g = solve_lp_complement(a_work, M, n)
        if g is None:
            break

        # Try different step sizes
        improved = False
        for t_try in [t_step, t_step * 0.5, t_step * 0.25]:
            a_new = (1 - t_try) * a_work + t_try * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            M_new = fast_convolve(a_new).max()
            if M_new <= M + 1e-12:
                a_work = a_new
                t_step = t_try
                improved = True
                if M_new < best_M:
                    best_M = M_new
                    best_a = a_new.copy()
                break

        if not improved:
            t_step *= 0.7
            if t_step < 0.01:
                break

    return best_a


def active_set_refine(a, n_iters=80):
    """Active set method: reduce top-K convolution lags simultaneously."""
    n = len(a)
    a_work = a.copy() / a.sum()
    best_obj = 2.0 * n * fast_convolve(a_work).max()
    best_a = a_work.copy()
    top_k = 5
    lr = 0.03

    for it in range(n_iters):
        conv = fast_convolve(a_work)
        top_indices = np.argsort(conv)[::-1][:top_k]
        conv_sum = conv[top_indices].sum()
        if conv_sum < 1e-15:
            break

        grad = np.zeros(n)
        for k in top_indices:
            weight = conv[k] / conv_sum
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += weight * 2.0 * a_work[j]

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_new = project_simplex(a_work - lr * shrink * grad, target_sum=1.0)
            obj_new = 2.0 * n * fast_convolve(a_new).max()
            if obj_new < best_obj:
                a_work = a_new
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
                break

        if not improved:
            lr *= 0.8
            if lr < 1e-6:
                break

    return best_a


def subgradient_polish(a, n_steps=100, step_size=0.003):
    """Direct subgradient on the max convolution value."""
    n = len(a)
    a_work = a.copy() / a.sum()
    best_obj = 2.0 * n * fast_convolve(a_work).max()
    best_a = a_work.copy()
    lr = step_size

    for _ in range(n_steps):
        conv = fast_convolve(a_work)
        peak_k = np.argmax(conv)
        grad = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                grad[i] = 2.0 * a_work[j]

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            a_new = project_simplex(a_work - lr * shrink * grad, target_sum=1.0)
            obj_new = 2.0 * n * fast_convolve(a_new).max()
            if obj_new < best_obj:
                a_work = a_new
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
                break

        if not improved:
            lr *= 0.9

    return best_a


def blend_two(a1, a2, n_samples=20):
    """Try convex combinations of two sequences, return the best."""
    best_a = a1.copy()
    best_obj = 2.0 * len(a1) * fast_convolve(a1).max()
    n = len(a1)
    for t in np.linspace(0.0, 1.0, n_samples):
        a_blend = (1 - t) * a1 + t * a2
        a_blend = np.maximum(a_blend, 0)
        s = a_blend.sum()
        if s < 1e-10:
            continue
        a_blend = a_blend / s
        obj = 2.0 * n * fast_convolve(a_blend).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_blend.copy()
    return best_a


def interpolate_seq(src, target_n):
    """Interpolate a sequence to a new length, maintaining non-negativity and sum=1."""
    if len(src) == target_n:
        return src.copy()
    pp = np.interp(np.linspace(0, 1, target_n), np.linspace(0, 1, len(src)), src)
    pp = np.maximum(pp, 0)
    ps = pp.sum()
    if ps < 1e-10:
        return np.ones(target_n) / target_n
    return pp / ps


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search combining LP-based alternating iteration, frequency-domain
    initialization, L-BFGS-B with beta continuation, and active-set refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print("Parent: length=%d" % len(parent))
    except Exception:
        pass

    # Helper to check and update best
    def do_score(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: %.6f (n=%d, t=%.0fs)" % (score, len(a), time.monotonic() - t_start))

    # ============================================================
    # Phase 1: Generate initializations for multiple lengths
    # ============================================================
    print("=== Phase 1: Initialization ===")

    lengths_to_try = [256, 384, 512]
    if parent is not None:
        lengths_to_try.insert(0, len(parent))
    lengths_to_try = sorted(set(lengths_to_try))

    phase1_best = {}

    for n in lengths_to_try:
        if time.monotonic() > deadline - 200:
            break

        candidates = {}

        if parent is not None:
            candidates['parent'] = interpolate_seq(parent, n)

        candidates['uniform'] = np.ones(n) / n
        candidates['freq'] = freq_domain_init(n, rng, n_trials=25)
        candidates['random'] = rng.dirichlet(np.ones(n) * 0.3)

        x_arr = np.arange(n, dtype=np.float64)
        g = np.exp(-0.5 * ((x_arr - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
        candidates['gauss'] = g / g.sum()

        best_init = None
        best_obj = float('inf')
        for name, a in candidates.items():
            obj = 2.0 * n * fast_convolve(a).max()
            print("  n=%d %s: %.6f" % (n, name, obj))
            if obj < best_obj:
                best_obj = obj
                best_init = a.copy()

        phase1_best[n] = best_init
        do_score(best_init)

    print("Phase 1 done: best=%.6f" % best_score)

    # ============================================================
    # Phase 2: LP-based alternating iteration
    # ============================================================
    print("=== Phase 2: LP Iteration ===")

    if best_sequence is not None:
        n_work = len(best_sequence)
        a_work = best_sequence.copy()
    else:
        n_work = 256
        a_work = phase1_best.get(256, np.ones(256) / 256).copy()
    a_work = a_work / a_work.sum()

    # Run LP on the best sequence
    lp_time = 150.0
    a_lp = lp_iteration_loop(a_work, max_iters=300, time_limit=lp_time)
    do_score(a_lp)

    # Also try LP from other initializations
    if time.monotonic() < deadline - 150:
        for n, a_init in phase1_best.items():
            if n == n_work:
                continue
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(40.0, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break
            a_lp2 = lp_iteration_loop(a_init, max_iters=50, time_limit=t_alloc)
            do_score(a_lp2)
            if best_sequence is not None and len(best_sequence) == n_work:
                # Blend with current best
                a_blend = blend_two(best_sequence, interpolate_seq(a_lp2, n_work))
                do_score(a_blend)

    print("Phase 2 done: best=%.6f" % best_score)

    # ============================================================
    # Phase 3: L-BFGS-B polishing with beta continuation
    # ============================================================
    print("=== Phase 3: L-BFGS-B ===")

    if best_sequence is not None:
        a_pol = best_sequence.copy()
    else:
        a_pol = a_work.copy()
    a_pol = a_pol / a_pol.sum()
    n_pol = len(a_pol)

    phase3_deadline = deadline - 180.0
    betas = [20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    for beta in betas:
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 10:
            break
        t_alloc = min(60.0, remaining - 5)
        maxiter = int(min(400, max(50, t_alloc * 25)))
        a_opt = optimize_lbfgsb(a_pol, beta=beta, maxiter=maxiter)
        do_score(a_opt)
        if best_sequence is not None:
            a_pol = best_sequence.copy()

    print("Phase 3 done: best=%.6f" % best_score)

    # ============================================================
    # Phase 4: Active-set + subgradient refinement
    # ============================================================
    print("=== Phase 4: Refinement ===")

    if best_sequence is not None:
        a_ref = best_sequence.copy() / best_sequence.sum()
        n_ref = len(a_ref)
    else:
        a_ref = a_pol.copy()
        n_ref = len(a_pol)

    phase4_deadline = deadline - 80.0

    # Active set
    if time.monotonic() < phase4_deadline - 40:
        t_alloc = min(60.0, phase4_deadline - time.monotonic() - 30)
        n_act = int(min(200, max(50, t_alloc * 5)))
        a_act = active_set_refine(a_ref, n_iters=n_act)
        do_score(a_act)
        if best_sequence is not None:
            a_ref = best_sequence.copy() / best_sequence.sum()

    # Subgradient
    if time.monotonic() < phase4_deadline - 30:
        a_sub = subgradient_polish(a_ref, n_steps=150, step_size=0.002)
        do_score(a_sub)
        if best_sequence is not None:
            a_ref = best_sequence.copy() / best_sequence.sum()

    # L-BFGS-B from refined point
    if time.monotonic() < phase4_deadline - 10:
        a_opt = optimize_lbfgsb(a_ref, beta=800.0, maxiter=300)
        do_score(a_opt)

    # Another active set pass
    if best_sequence is not None and time.monotonic() < phase4_deadline - 5:
        a_act2 = active_set_refine(best_sequence / best_sequence.sum(), n_iters=50)
        do_score(a_act2)

    print("Phase 4 done: best=%.6f" % best_score)

    # ============================================================
    # Phase 5: Try alternate lengths from best
    # ============================================================
    print("=== Phase 5: Alternate Lengths ===")

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 30:
            best_n = len(best_sequence)
            alt_lengths = [l for l in [192, 256, 320, 384, 512, 640, 768] if l != best_n]

            for alt_n in alt_lengths:
                if remaining < 15:
                    break
                pp = interpolate_seq(best_sequence, alt_n)
                t_alloc = min(25.0, remaining - 10)
                if t_alloc < 8:
                    break
                maxiter = int(min(200, max(30, t_alloc * 12)))
                a_opt = optimize_lbfgsb(pp, beta=150.0, maxiter=maxiter)
                do_score(a_opt)
                # Also try active set
                if remaining > 10:
                    a_act = active_set_refine(a_opt, n_iters=40)
                    do_score(a_act)
                remaining = deadline - time.monotonic() - 30.0

    # ============================================================
    # Phase 6: Final high-beta polish
    # ============================================================
    print("=== Phase 6: Final Polish ===")

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 15:
            maxiter = int(min(500, max(50, remaining * 20)))
            a_opt = optimize_lbfgsb(best_sequence / best_sequence.sum(), beta=2000.0, maxiter=maxiter)
            do_score(a_opt)

        if best_sequence is not None and remaining > 8:
            a_act = active_set_refine(best_sequence / best_sequence.sum(), n_iters=30)
            do_score(a_act)

            # Try blending with LP result
            if a_lp is not None and len(a_lp) == len(best_sequence):
                a_blend = blend_two(best_sequence, a_lp)
                do_score(a_blend)

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
