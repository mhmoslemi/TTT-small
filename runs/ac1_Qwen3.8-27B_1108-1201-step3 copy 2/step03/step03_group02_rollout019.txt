"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
with adaptive line search, multi-resolution initialization, and diverse
structured starts. Key improvements: min-peak LP as primary direction,
line search for step size, cosine/bowl/hat initializations, multi-n search,
and perturbation-based restarts.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute the objective value for a normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(a):
    """Build matrix where (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_min_peak_lp(a, M_init=None):
    """Solve: min M' s.t. conv(a,b)[k] <= M' for all k, sum(b)=1, b>=0.
    Returns (b, M')."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    
    # Variables: b_0,...,b_{n-1}, M'
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M'
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequalities: conv(a,b)[k] - M' <= 0
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        b_sum = b.sum()
        if b_sum > 1e-15:
            b = b / b_sum
        return b, res.x[n]
    return None, None


def solve_max_sum_lp(a, M):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M, b>=0. Returns b normalized."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum > 1e-15:
            return b / b_sum
    return None

            k = int(peaks[i])
            wi = w[i]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = compute_objective(a)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9998
        if no_improve >= 2000:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    # Cosine families - these have the right "two-hump" structure
    for alpha_c in [0.3, 0.5, 0.7, 0.9]:
        a = 1.0 + alpha_c * np.cos(2.0 * np.pi * x / n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits[f'cos_{alpha_c}'] = a / s
    
    # Bowl (U-shape) - high at ends
    bowl = (x / (n - 1) - 0.5) ** 2
    bowl /= bowl.sum()
    inits['bowl'] = bowl
    
    # Hat (inverted U) - high in middle
    hat = 1.0 - (2.0 * x / (n - 1) - 1.0) ** 2
    hat = np.maximum(hat, 0)
    hat /= hat.sum()
    inits['hat'] = hat
    
    # Raised cosine window
    rc = 0.5 * (1 + np.cos(2.0 * np.pi * x / n))
    rc = np.maximum(rc, 1e-6)
    rc /= rc.sum()
    inits['raised_cos'] = rc
    
    # Two-sided exponential (bimodal)
    center = n / 2.0
    exp_two = np.exp(-np.abs(x - center) / (n * 0.15)) + np.exp(-np.abs(x - n + center) / (n * 0.15))
    exp_two += 1e-6
    exp_two /= exp_two.sum()
    inits['bimodal'] = exp_two
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Triangular (high in middle, linear falloff)
    tri = 1.0 - np.abs(2.0 * x / (n - 1) - 1.0)
    tri = np.maximum(tri, 1e-6)
    tri /= tri.sum()
    inits['triangular'] = tri
    
    # Fourth power cosine - sharper peaks at ends
    a4 = (0.5 * (1 + np.cos(2.0 * np.pi * x / n))) ** 3
    a4 = np.maximum(a4, 1e-6)
    a4 /= a4.sum()
    inits['cos_cubed'] = a4
    
    return inits


def upsample_sequence(a, new_n):
    """Upsample a sequence to new length using interpolation."""
    old_n = len(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_up = np.interp(new_x, old_x, a)
    a_up = np.maximum(a_up, 0)
    s = a_up.sum()
    if s > 1e-15:
        a_up /= s
    return a_up


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase LP-based optimization with line search,
    diverse initializations, multi-resolution, and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        a_clean = np.maximum(a, 0).tolist()
        # Check all are finite
        if not all(np.isfinite(x) for x in a_clean):
            return
        score = float(evaluate_sequence(a_clean))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = np.array(a_clean, dtype=np.float64)
            print(f"  New best: {score:.6f} (n={len(a)}, time={time.monotonic():.0f}s)")
    
    print(f"Start search at t=0, deadline in {budget_s-20}s")
    
    # Phase 1: Multi-n exploration with diverse initializations
    print("\n=== Phase 1: Multi-n Exploration ===")
    
    n_values = [256, 384, 512, 640, 768, 1024]
    
    if parent is not None:
        pn = len(parent)
        # Add parent length if reasonable
        if 128 <= pn <= 2048 and pn not in n_values:
            n_values.insert(0, pn)
        n_values = list(dict.fromkeys(n_values))  # remove duplicates, keep order
    
    # Sort by priority: parent length first, then 512, then others
    n_values.sort(key=lambda x: (0 if x == len(parent) if parent is not None else 1,
                                  0 if x == 512 else 1, 0 if x == 256 else 1, x))
    
    # Allocate time: more to mid-range n values
    time_weights = {}
    for nv in n_values:
        if nv <= 384:
            time_weights[nv] = 0.8
        elif nv <= 640:
            time_weights[nv] = 1.5
        elif nv <= 1024:
            time_weights[nv] = 1.2
        else:
            time_weights[nv] = 0.7
    
    total_weight = sum(time_weights.values())
    phase1_budget = (deadline - time.monotonic()) * 0.55  # 55% for phase 1
    
    for n in n_values:
        remain = deadline - time.monotonic() - (budget_s - 20) * 0.45
        if remain < 30:
            break
        
        t_alloc_n = phase1_budget * time_weights[n] / total_weight
        t_alloc_n = min(t_alloc_n, remain)
        if t_alloc_n < 10:
            continue
        
        print(f"\n  n={n}, time_alloc={t_alloc_n:.0f}s")
        
        inits = generate_initializations(n, parent, rng)
        
        # Priority order for initializations
        priority = ['parent', 'cos_0.7', 'cos_0.5', 'cos_0.9', 'bimodal',
                    'cos_cubed', 'raised_cos', 'bowl', 'hat', 'uniform', 'random']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits.keys() if nm not in priority]
        
        # Limit number of initializations based on time
        max_inits = max(2, int(t_alloc_n / 25))
        init_names = init_names[:max_inits]
        
        for name in init_names:
            remain_per_init = deadline - time.monotonic() - (budget_s - 20) * 0.45
            if remain_per_init < 10:
                break
            t_single = min(30.0, remain_per_init / 2)
            if t_single < 5:
                break
            
            a0 = inits[name]
            
            # Run min_peak LP (primary)
            a_opt, obj = lp_iterative_optimize(
                a0, time_limit=t_single, use_line_search=True,
                max_iters=1500, method='min_peak')
            try_update(a_opt)
            
            # If time permits, also try max_sum
            remain2 = deadline - time.monotonic() - (budget_s - 20) * 0.45
            if remain2 > 15 and a_opt is not None:
                a_opt2, obj2 = lp_iterative_optimize(
                    a_opt, time_limit=min(15, remain2 / 2), use_line_search=False,
                    max_iters=800, method='max_sum')
                try_update(a_opt2)
    
    print(f"\nPhase 1 done: best={best_score:.6f} at t={time.monotonic():.0f}s")
    
    # Phase 2: Deep refinement of best with different step sizes
    print("\n=== Phase 2: Deep Refinement ===")
    
    if best_sequence is not None:
        phase2_budget = (deadline - time.monotonic()) * 0.25
        n_best = len(best_sequence)
        
        for method in ['min_peak', 'max_sum', 'min_peak', 'max_sum']:
            if time.monotonic() > deadline - (budget_s - 20) * 0.2:
                break
            t_alloc = min(40.0, phase2_budget / 4)
            if t_alloc < 5:
                break
            
            a_opt, _ = lp_iterative_optimize(
                best_sequence, time_limit=t_alloc, use_line_search=True,
                max_iters=2000, method=method)
            try_update(a_opt)
        
        # Subgradient polish
        remain_sg = deadline - time.monotonic() - (budget_s - 20) * 0.2
        if remain_sg > 20:
            t_sg = min(30.0, remain_sg / 2)
            a_opt, _ = subgradient_optimize(
                best_sequence, time_limit=t_sg, lr_init=0.015,
                momentum=0.92, top_k=max(30, n_best // 10), temp=0.0002)
            try_update(a_opt)
    
    print(f"Phase 2 done: best={best_score:.6f} at t={time.monotonic():.0f}s")
    
    # Phase 3: Perturbation and restart
    print("\n=== Phase 3: Perturbation & Restart ===")
    
    if best_sequence is not None:
        phase3_budget = (deadline - time.monotonic()) * 0.3
        n_best = len(best_sequence)
        
        for attempt in range(8):
            remain = deadline - time.monotonic()
            if remain < 15:
                break
            t_alloc = min(30.0, phase3_budget / 8)
            if t_alloc < 5:
                break
            
            # Different perturbation strategies
            if attempt % 3 == 0:
                # Small Gaussian noise
                noise = rng.standard_normal(n_best) * 0.005
                a_pert = np.maximum(best_sequence + noise, 0)
                a_pert = a_pert / a_pert.sum()
            elif attempt % 3 == 1:
                # Sparse perturbation
                n_pert = max(5, n_best // 20)
                idx = rng.choice(n_best, size=n_pert, replace=False)
                a_pert = best_sequence.copy()
                a_pert[idx] *= (1 + rng.standard_normal(n_pert) * 0.1)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
            else:
                # Frequency-domain perturbation
                a_ft = np.fft.rfft(best_sequence)
                phase_noise = rng.standard_normal(len(a_ft)) * 0.05
                a_ft[1:] *= np.exp(1j * phase_noise[1:])
                a_pert = np.fft.irfft(a_ft, n_best)
                a_pert = np.maximum(a_pert, 0)
                s = a_pert.sum()
                if s < 1e-10:
                    a_pert = best_sequence.copy()
                else:
                    a_pert = a_pert / s
            
            a_opt, _ = lp_iterative_optimize(
                a_pert, time_limit=t_alloc, use_line_search=True,
                max_iters=1000, method='min_peak')
            try_update(a_opt)
    
    print(f"Phase 3 done: best={best_score:.6f} at t={time.monotonic():.0f}s")
    
    # Phase 4: Final polish with very small steps
    print("\n=== Phase 4: Final Polish ===")
    
    if best_sequence is not None:
        remain = deadline - time.monotonic()
        if remain > 10:
            t_polish = min(20.0, remain - 5)
            a_opt, _ = lp_iterative_optimize(
                best_sequence, time_limit=t_polish, use_line_search=True,
                max_iters=3000, method='min_peak')
            try_update(a_opt)
            
            # One more subgradient pass
            remain2 = deadline - time.monotonic()
            if remain2 > 10:
                a_opt2, _ = subgradient_optimize(
                    best_sequence, time_limit=min(15.0, remain2 - 3),
                    lr_init=0.005, momentum=0.95, top_k=20, temp=0.0001)
                try_update(a_opt2)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_sequence)} ===")
    return [float(x) for x in best_sequence]
