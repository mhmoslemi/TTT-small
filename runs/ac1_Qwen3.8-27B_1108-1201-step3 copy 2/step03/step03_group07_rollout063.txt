import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

def smooth_max_and_weights(x, beta):
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w

def compute_grad_smooth_conv(a, w):
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M):
    """Find direction b that maximizes sum(b) subject to f*b <= M and b >= 0."""
    n = len(f)
    conv_len = 2 * n - 1
    
    # Build constraint matrix A_ub such that A_ub @ b <= M * ones(conv_len)
    # (f * b)[k] = sum_i f[i] * b[k-i] for valid i
    # For k = 0, 1, ..., 2n-2:
    #   (f*b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * b[k-i]
    # Let j = k-i, so i = k-j
    #   (f*b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * b[j]
    # So A_ub[k, j] = f[k-j] if k-j is in [0, n-1], else 0
    
    A_ub = np.zeros((conv_len, n))
    for k in range(conv_len):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]
    
    # Maximize sum(b) = -minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if res.success:
        return res.x
    return None

        a = a_new

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. LP-based direction finding (from convex optimization perspective)
2. Smooth max optimization with adaptive temperature
3. Subgradient refinement with top-k peak weighting
4. Multi-length exploration and interpolation
"""

import numpy as np
import time
import sys
from scipy.optimize import linprog, minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_weighted_conv(a, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a using FFT."""
    n = len(a)
    conv_len = 2 * n - 1
    N = 1 << (conv_len + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, M, verbose=False):
    """
    Find direction b maximizing sum(b) subject to (f*b)[k] <= M for all k, b >= 0.
    Returns the optimal b vector or None if infeasible.
    """
    n = len(f)
    conv_len = 2 * n - 1
    
    # Build A_ub: A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((conv_len, n))
    for k in range(conv_len):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if res.success:
        return res.x
    return None


def lp_iterate(f, t=0.1, max_iters=200, time_limit=None, t_start=None, verbose=True):
    """
    Iterative LP-based improvement (from state of the art).
    At each step: find b via LP, rescale, blend with f.
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    f_sum = f.sum()
    f = f / f_sum
    M = fast_convolve(f).max()
    
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    
    for it in range(max_iters):
        if time_limit is not None:
            if time.monotonic() - t_start > time_limit:
                break
        
        b = lp_direction(f, M)
        if b is None:
            if verbose:
                print("  LP infeasible at iter %d" % it)
            break
        
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        # Rescale b to have same sum as f
        g = b / b_sum
        
        # Blend
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new /= s
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        score_new = 2.0 * n * M_new
        
        if score_new < best_score - 1e-10:
            best_score = score_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 10 == 0:
                print("  LP iter %d: score=%.6f, M=%.6f" % (it, score_new, M_new))
        else:
            no_improve += 1
            if no_improve > 20:
                # Try smaller t
                t *= 0.7
                if t < 0.001:
                    break
                no_improve = 0
        
        f = f_new
        M = M_new
    
    return best_f, best_score


def optimize_smooth(a, betas=None, maxiter_per_beta=300, time_limit=None, t_start=None):
    """Optimize using smooth max with L-BFGS-B and softmax parametrization."""
    n = len(a)
    if betas is None:
        betas = [100.0, 50.0, 25.0, 10.0, 5.0, 3.0, 2.0]
    
    a_clipped = np.maximum(a, 1e-15)
    a_clipped = a_clipped / a_clipped.sum()
    x = np.log(a_clipped)
    
    best_a = a_clipped.copy()
    best_score = 2.0 * n * fast_convolve(a_clipped).max()
    
    for beta in betas:
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        
        def obj(x_):
            x_s = x_ - x_.max()
            a_ = np.exp(x_s)
            a_ = a_ / a_.sum()
            conv = fast_convolve(a_)
            sm, _ = smooth_max_and_weights(conv, beta)
            return 2.0 * n * sm
        
        def jac(x_):
            x_s = x_ - x_.max()
            a_ = np.exp(x_s)
            a_ = a_ / a_.sum()
            conv = fast_convolve(a_)
            sm, w = smooth_max_and_weights(conv, beta)
            grad_a = compute_grad_weighted_conv(a_, w)
            lagrange = np.dot(a_, grad_a)
            return a_ * (grad_a - lagrange)
        
        result = minimize(obj, x, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter_per_beta, 'maxls': 50, 'maxfun': 1000})
        
        x_opt = result.x
        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()
        
        score = 2.0 * n * fast_convolve(a_opt).max()
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            x = x_opt
        else:
            x = x_opt  # Still update x for next beta
    
    return best_a, best_score


def subgradient_optimize(a, top_k=5, n_iter=1000, lr=0.005, time_limit=None, t_start=None):
    """Subgradient optimization using true max with top-k weighting."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        
        conv = fast_convolve(a)
        peak = conv.max()
        score = 2.0 * n * peak
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Top-k peaks
        top_idx = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_idx]
        
        if top_vals[0] < 1e-15:
            break
        
        # Soft weights (temperature = 5)
        temp = top_vals[0] / 5.0
        weights = np.exp(-(top_vals - top_vals[0]) / temp)
        weights /= weights.sum()
        
        # Weighted subgradient
        grad = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            j_start = max(0, idx - n + 1)
            j_end = min(n - 1, idx)
            for j in range(j_start, j_end + 1):
                grad[j] += 2.0 * w * a[idx - j]
        
        # Normalized gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Step
        a_new = project_simplex(a - lr * grad / grad_norm)
        a = a_new
        
        lr *= 0.9998
    
    return best_a, best_score


def peak_reduction(a, n_steps=50, reduction=0.97):
    """Iteratively reduce peaks in the convolution."""
    a = a.copy().astype(np.float64)
    n = len(a)
    a = a / a.sum()
    
    for _ in range(n_steps):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        
        # Find contributing elements
        j_start = max(0, max_k - n + 1)
        j_end = min(n - 1, max_k)
        
        contrib = a[j_start:j_end + 1].copy()
        # The contribution of a[j] to conv[max_k] is a[max_k - j]
        # So the "importance" of a[j] is a[j] * a[max_k - j]
        importance = np.zeros(n)
        for j in range(j_start, j_end + 1):
            importance[j] = a[j] * a[max_k - j]
        
        total = importance.sum()
        if total < 1e-15:
            break
        
        # Reduce proportionally to importance
        excess = conv.max() - conv.mean()
        if excess < 0:
            break
        
        for j in range(j_start, j_end + 1):
            a[j] -= reduction * (importance[j] / total) * excess / max(a[j], 1e-15) * 0.01
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        a /= s
    
    return a


def random_phase_init(n, rng, n_trials=20):
    """Initialize using random phases designed for flat spectrum."""
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')
    
    for _ in range(n_trials):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * phases[-1])
        
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            obj = 2.0 * n * fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_interp'] = p / ps
    
    inits['random_phase'] = random_phase_init(n, rng)
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gaussian'] = g
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)

The IFFT of conj(W)*A gives the circular cross-correlation.

The circular cross-correlation: sum_{n=0}^{N-1} W[(n+k) mod N] * A[n]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (convex relaxation of the peak constraint)
2. Smooth max L-BFGS with aggressive beta annealing
3. Subgradient refinement with top-k peak weighting (true max, no approximation)
4. Multi-length exploration and interpolation
5. Peak redistribution and final polish
"""

import numpy as np
import time
import sys
from scipy.optimize import linprog, minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_weighted_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a, where conv = a*a (linear)."""
    n = len(a)
    conv_len = 2 * n - 1
    N = 1 << (conv_len + n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build the constraint matrix for the LP: (f*b)[k] = sum_j A[k,j]*b[j]."""
    n = len(f)
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A


def lp_direction(f, M, eps=1e-9):
    """
    Solve: maximize sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.
    Returns optimal b or None.
    """
    n = len(f)
    conv_len = 2 * n - 1
    A_ub = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(conv_len, M + eps)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None),
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


def lp_iterate(f, t=0.3, max_iters=300, time_limit=60.0, t_start=None, verbose=False):
    """
    Iterative LP-based improvement.
    At each step: solve LP to find direction b, rescale, blend with f.
    """
    n = len(f)
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 0)
    fs = f.sum()
    if fs < 1e-10:
        return f, float('inf')
    f = f / fs
    M = fast_convolve(f).max()

    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0

    if t_start is None:
        t_start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break

        b = lp_direction(f, M)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-10:
            break

        # Rescale to have same sum as f
        g = b / b_sum

        # Blend
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 20 == 0:
                print("    LP iter %d: score=%.8f" % (it, score_new))
        else:
            no_improve += 1
            if no_improve > 30:
                t = max(0.01, t * 0.6)
                no_improve = 0
                if t < 0.01:
                    break

        f = f_new
        M = M_new

    return best_f, best_score


def optimize_smooth_l_bfgs(a, betas, maxiter_per_beta=300, time_limit=120.0, t_start=None):
    """
    Smooth max optimization using L-BFGS-B with log parametrization.
    Uses aggressive beta annealing schedule.
    """
    n = len(a)
    if t_start is None:
        t_start = time.monotonic()

    a_clipped = np.maximum(a, 1e-15)
    a_clipped = a_clipped / a_clipped.sum()
    x = np.log(a_clipped)

    best_a = a_clipped.copy()
    best_score = 2.0 * n * fast_convolve(a_clipped).max()

    for beta in betas:
        if time.monotonic() - t_start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 3:
            break

        def obj_func(x_):
            x_s = x_ - x_.max()
            a_ = np.exp(x_s)
            a_ = a_ / a_.sum()
            conv = fast_convolve(a_)
            sm, _ = smooth_max_and_weights(conv, beta)
            return 2.0 * n * sm

        def jac_func(x_):
            x_s = x_ - x_.max()
            a_ = np.exp(x_s)
            a_ = a_ / a_.sum()
            conv = fast_convolve(a_)
            sm, w = smooth_max_and_weights(conv, beta)
            grad_a = compute_grad_weighted_conv(a_, w)
            lagrange = np.dot(a_, grad_a)
            return a_ * (grad_a - lagrange)

        iter_budget = int(min(maxiter_per_beta, max(50, remaining * 40)))
        result = minimize(obj_func, x, jac=jac_func, method='L-BFGS-B',
                          options={'maxiter': iter_budget, 'maxls': 50,
                                   'ftol': 1e-12, 'gtol': 1e-10})

        x_opt = result.x
        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()

        score = 2.0 * n * fast_convolve(a_opt).max()
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
        x = x_opt

    return best_a, best_score


def subgradient_optimize(a, top_k=5, n_iter=2000, lr=0.003,
                         time_limit=30.0, t_start=None):
    """
    Subgradient method using the TRUE max with top-k peak weighting.
    The subgradient of max(conv) at a is the gradient of conv[k*] where k* is argmax.
    d conv[k]/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    if t_start is None:
        t_start = time.monotonic()

    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iter):
        if time.monotonic() - t_start > time_limit:
            break

        conv = fast_convolve(a)
        peak = conv.max()
        score = 2.0 * n * peak

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Get top-k peak positions
        top_idx = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_idx]

        if top_vals[0] < 1e-15:
            break

        # Soft weights: exponential tempering
        temp = max(top_vals[0] * 0.2, 1e-10)
        weights = np.exp(-(top_vals - top_vals[0]) / temp)
        weights = weights / weights.sum()

        # Weighted subgradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (valid range)
        grad = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            j_start = max(0, idx - n + 1)
            j_end = min(n - 1, idx)
            for j in range(j_start, j_end + 1):
                grad[j] += 2.0 * w * a[idx - j]

        # Normalize gradient for stable steps
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Projected gradient step
        a_new = project_simplex(a - lr * grad / gnorm)
        a = a_new

        # Slow decay
        lr *= 0.9999

    return best_a, best_score


def peak_redistribute(a, n_steps=100, alpha=0.5):
    """
    Reduce peaks by redistributing mass from elements that contribute
    most to the current peak toward elements that contribute to lower values.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    for step in range(n_steps):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        target = conv.mean()
        excess = conv[max_k] - target
        if excess < 1e-12:
            break

        # Elements contributing to conv[max_k]: a[j] where j in [max_k-n+1, max_k]
        j_start = max(0, max_k - n + 1)
        j_end = min(n - 1, max_k)

        # Importance of a[j] for peak = a[j] * a[max_k - j]
        importance = np.zeros(n)
        for j in range(j_start, j_end + 1):
            importance[j] = a[j] * a[max_k - j]

        total_imp = importance.sum()
        if total_imp < 1e-15:
            break

        # Reduce high-importance elements
        for j in range(j_start, j_end + 1):
            if importance[j] > 0 and a[j] > 1e-15:
                reduction = alpha * (importance[j] / total_imp) * excess / (a[max_k - j] + 1e-15) * 0.01
                a[j] = max(0.0, a[j] - reduction)

        # Redistribute to low-importance positions
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

    return a


def random_phase_init(n, rng, n_trials=50):
    """Initialize using random phases for approximately flat power spectrum."""
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for _ in range(n_trials):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        half = fft_size // 2
        phases = rng.uniform(0, 2 * np.pi, half)
        for k in range(1, half):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[half] = np.exp(1j * phases[-1])

        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            obj = 2.0 * n * fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def cosine_init(n, rng):
    """Initialize with cosine-modulated window."""
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    a = np.exp(-0.5 * ((x - center) / (0.4 * n)) ** 2)
    # Add small cosine modulation
    freq = rng.uniform(0.1, 0.5)
    a *= (1.0 + 0.1 * np.cos(2 * np.pi * freq * x / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random_phase'] = random_phase_init(n, rng, n_trials=30)
    inits['cosine'] = cosine_init(n, rng)

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gaussian'] = g

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.maximum(p, 0)
        if len(p) == n:
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        else:
            p_interp = np.interp(np.linspace(0, 1, n),
                                 np.linspace(0, 1, len(p)), p)
            p_interp = np.maximum(p_interp, 0)
            ps = p_interp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = p_interp / ps

    return inits

        update_best(a_opt, "L-BFGS-%s" % name)

    print("  Phase 2 best: %.8f" % best_score)

    # =====================
    # Phase 3: Multi-length exploration
    # =====================
    print("\n--- Phase 3: Multi-Length Exploration ---")
    phase3_end = deadline - 150.0

    if best_sequence is not None:
        n_curr = len(best_sequence)
    else:
        n_curr = 256

    # Try nearby lengths
    length_candidates = []
    for factor in [0.5, 0.66, 0.75, 1.0, 1.25, 1.5, 2.0]:
        cl = int(n_curr * factor)
        cl = max(64, min(1024, cl))
        if cl not in length_candidates:
            length_candidates.append(cl)

    # Standard lengths
    for sl in [128, 192, 256, 320, 384, 512, 640]:
        if sl not in length_candidates:
            length_candidates.append(sl)

    length_candidates = sorted(set(length_candidates))

    for n_try in length_candidates:
        if time.monotonic() > phase3_end - 20:
            break

        remaining = phase3_end - time.monotonic() - 10
        if remaining < 15:
            break

        # Create initial guess at this length
        if best_sequence is not None and len(best_sequence) > 0:
            bs = best_sequence
            a0 = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, len(bs)), bs)
            a0 = np.maximum(a0, 0)
            a0 = a0 / a0.sum()
        else:
            a0 = np.ones(n_try) / n_try

        t_alloc = min(30.0, remaining * 0.4)
        a_opt, _ = optimize_smooth_l_bfgs(a0, [50, 20, 10, 5],
                                           maxiter_per_beta=200,
                                           time_limit=t_alloc,
                                           t_start=time.monotonic())
        update_best(a_opt, "len=%d" % n_try)

        # Also try LP on this length if time permits
        if time.monotonic() < phase3_end - 20:
            t_alloc_lp = min(20.0, phase3_end - time.monotonic() - 15)
            if t_alloc_lp > 5:
                result_f, _ = lp_iterate(a_opt, t=0.3, max_iters=300,
                                          time_limit=t_alloc_lp,
                                          t_start=time.monotonic())
                update_best(result_f, "LP-len=%d" % n_try)

    print("  Phase 3 best: %.8f" % best_score)

    # =====================
    # Phase 4: Subgradient refinement
    # =====================
    print("\n--- Phase 4: Subgradient Refinement ---")
    phase4_end = deadline - 80.0

    if best_sequence is not None:
        a_curr = best_sequence / best_sequence.sum()
        n_curr = len(a_curr)

        for top_k in [3, 5, 8, 15]:
            if time.monotonic() > phase4_end - 10:
                break
            t_lim = min(25.0, phase4_end - time.monotonic() - 5)
            if t_lim < 5:
                break
            a_opt, _ = subgradient_optimize(a_curr, top_k=top_k, n_iter=5000,
                                             lr=0.002, time_limit=t_lim,
                                             t_start=time.monotonic())
            update_best(a_opt, "subgrad-k=%d" % top_k)

        # Re-optimize from subgradient result
        if time.monotonic() < phase4_end - 10:
            a_opt, _ = optimize_smooth_l_bfgs(a_curr, [10, 5, 3],
                                               maxiter_per_beta=300,
                                               time_limit=30,
                                               t_start=time.monotonic())
            update_best(a_opt, "post-subgrad")

    print("  Phase 4 best: %.8f" % best_score)

    # =====================
    # Phase 5: Peak redistribution + re-optimize
    # =====================
    print("\n--- Phase 5: Peak Redistribution ---")

    if best_sequence is not None:
        a_curr = best_sequence / best_sequence.sum()

        for alpha in [0.3, 0.5, 0.7]:
            if time.monotonic() > deadline - 40:
                break
            a_red = peak_redistribute(a_curr, n_steps=200, alpha=alpha)
            a_opt, _ = optimize_smooth_l_bfgs(a_red, [20, 10, 5, 3],
                                               maxiter_per_beta=200,
                                               time_limit=20,
                                               t_start=time.monotonic())
            update_best(a_opt, "peak-redist-a=%.1f" % alpha)

    print("  Phase 5 best: %.8f" % best_score)
