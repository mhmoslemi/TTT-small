"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:

import numpy as np
import time
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode='full')
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix_topk(a, top_k):
    """Build matrix for only the top-k convolution positions."""
    n = len(a)
    conv = fast_convolve(a)
    top_positions = np.argpartition(conv, -top_k)[-top_k:]
    # Sort by value descending
    top_positions = top_positions[np.argsort(conv[top_positions])[::-1]]
    
    A = np.zeros((top_k, n))
    for idx, k in enumerate(top_positions):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[idx, j] = a[k - j]
    return A, top_positions

def lp_step_fast(a, M, top_k=100):
    """One LP step using only top-k constraints."""
    n = len(a)
    A, _ = build_conv_matrix_topk(a, top_k)
    c = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def lp_step_full(a, M):
    """One LP step using all constraints."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    c = -np.ones(n)
    b_ub = np.full(num_rows, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def mirror_descent_step(a, conv, eta, top_k=50):
    """One mirror descent (exponential weights) step."""
    n = len(a)
    # Find top-k peaks
    if top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv))
    
    # Compute weights (softmax over peaks)
    peak_vals = conv[peaks]
    max_pv = peak_vals.max()
    # Use the max peak for the subgradient
    peak_idx = peaks[np.argmax(peak_vals)]
    
    # Subgradient: g_j = 2 * a[peak_idx - j] for valid j
    g = np.zeros(n)
    k = peak_idx
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    
    # Center the gradient (for simplex constraint)
    g_centered = g - g.mean()
    
    # Exponential weights update
    w = a * np.exp(-eta * g_centered)
    w_sum = w.sum()

def smoothed_objective_and_grad(a, n, t):
    """Compute log-sum-exp smoothed objective and gradient."""
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    obj = (2.0 * n / t) * np.log(sum_exp)
    p = exp_vals / sum_exp
    
    # Gradient: d/d a_j [2n/t * log(sum_k exp(t*conv_k))]
    # = (2n/t) * sum_k p_k * d(conv_k)/d(a_j)
    # = (2n/t) * sum_k p_k * 2*a_{k-j} (for valid k-j)
    # = (2n/t) * 2 * sum_k p_k * a_{k-j} (where 0 <= k-j < n)
    # Let S_j = sum_k p_k * a_{k-j} for valid k-j
    # This is like a cross-correlation of p and a
    grad = np.zeros(n)
    # S_j = sum_{k: 0<=k-j<n} p_k * a_{k-j} = sum_{i: 0<=i<n} p_{j+i} * a_i (where j+i < 2n-1)
    # = (p * a)_{reversed} ... it's a cross-correlation
    for j in range(n):
        k_vals = np.arange(j, min(j + n, 2*n - 1))
        i_vals = k_vals - j
        grad[j] = (4.0 * n / t) * np.dot(p[k_vals], a[i_vals])
    
    return obj, grad, conv

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
- Fast LP-based iterative refinement with adaptive step sizes and line search
- Mirror descent (exponential weights) for complementary search directions
- Fine perturbation + re-optimization for escaping local optima

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n <= 128:
        return np.convolve(a, a, mode='full')
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a, top_k=None):
    """Build convolution matrix, optionally only for top-k positions."""
    n = len(a)
    num_rows = 2 * n - 1
    if top_k is not None and top_k < num_rows:
        conv = fast_convolve(a)
        top_pos = np.argpartition(conv, -top_k)[-top_k:]
        top_pos = top_pos[np.argsort(conv[top_pos])[::-1]]
        A = np.zeros((top_k, n))
        for idx, k in enumerate(top_pos):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A[idx, j] = a[k - j]
        return A
    else:
        A = np.zeros((num_rows, n))
        for k in range(num_rows):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A[k, j] = a[k - j]
        return A

def mirror_descent(a0, time_limit, eta_init=0.05, top_k=50, use_weighted=True):

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient from top-k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        peak_vals = conv[peaks]

        if use_weighted:
            # Weighted sum of subgradients
            scaled = peak_vals - peak_vals.max()
            w = np.exp(scaled * 5.0)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.ones(len(peaks)) / len(peaks)
            else:
                w = w / w_sum

            g = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-10:
                    continue
                k = int(peaks[i])
                for j in range(n):
                    idx = k - j
                    if 0 <= idx < n:
                        g[j] += w[i] * 2.0 * a[idx]
        else:
            k = int(peaks[np.argmax(peak_vals)])
            g = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    g[j] = 2.0 * a[idx]

        g_centered = g - g.mean()

        # Exponential weights update

        exponent -= exponent.max()
        w_vec = a * np.exp(exponent)
        w_sum = w_vec.sum()
        if w_sum < 1e-15:
            break
        a_new = w_vec / w_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new

        eta *= 0.9995
        if no_improve > 200:
            eta *= 0.5
            no_improve = 0
        if no_improve > 1000:
            break

    return best_a, best_obj


def smoothed_obj_grad(a, t):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    obj = (2.0 * n / t) * np.log(sum_exp)
    p = exp_vals / sum_exp

    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        if k_lo <= k_hi:
            k_vals = np.arange(k_lo, k_hi + 1)
            i_vals = k_vals - j
            grad[j] = (4.0 * n / t) * np.dot(p[k_lo:k_hi + 1], a[i_vals])

    return obj, grad

def generate_initializations(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
            inits['parent_rev'] = pp[::-1] / ps

    # Various structured patterns
    for freq in [1, 2, 3, 5, 8]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        inits[f'cos_{freq}'] = a

    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sigma * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sigma}'] = g

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)

    # Quadratic chirp
    a = 1.0 + 0.6 * np.cos(np.pi * x ** 2 / n)
    a = np.maximum(a, 0.01)
    a /= a.sum()
    inits['chirp'] = a

    # Exp
    a = np.exp(-x / (n * 0.3))
    a /= a.sum()
    inits['exp'] = a

    # Symmetric bump
    mid = n // 2
    a = np.zeros(n)
    for i in range(n):
        a[i] = 1.0 / (1.0 + abs(i - mid) ** 1.5)
    a /= a.sum()
    inits['bump'] = a

    return inits
