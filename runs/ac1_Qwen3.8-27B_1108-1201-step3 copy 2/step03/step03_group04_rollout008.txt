def fast_convolve(a, b=None):

    A = np.fft.rfft(a, fft_size)

    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""

    cols = []

            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def backtracking_line_search(a, direction, alpha0=1.0, beta=0.5, c1=1e-4, max_iter=20):
    """Backtracking line search for f(x) = max(conv,1)."""
    n = len(a)
    f0 = compute_obj_normalized(a)
    grad_dir = f0  # approximate directional derivative
    
    for _ in range(max_iter):
        candidate = (1 - alpha0) * a + alpha0 * direction
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-15:
            alpha0 *= beta
            continue
        candidate = candidate / s
        f1 = compute_obj_normalized(candidate)
        if f1 <= f0 + c1 * alpha0 * grad_dir:
            return alpha0, f1
        alpha0 *= beta
    return 0.0, f0

            b = np.ones(n) / n
            no_improve += 10
        else:
            # Check improvement
            if np.allclose(a, b, atol=1e-10):
                no_improve += 1
            else:
                # Line search
                best_t = 0.0
                best_f = best_obj
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                    candidate = (1 - t) * a + t * b
                    candidate = np.maximum(candidate, 0)
                    sc = candidate.sum()
                    if sc < 1e-15:
                        continue
                    candidate = candidate / sc
                    f_cand = compute_obj_normalized(candidate)
                    if f_cand < best_f:
                        best_f = f_cand
                        best_t = t
                
                if best_t < 0.05:
                    no_improve += 1
                else:
                    a = (1 - best_t) * a + best_t * b
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = fast_convolve(a)
                    M = conv.max()
                    cur_obj = 2.0 * n * M
                    if cur_obj < best_obj - 1e-12:
                        best_obj = cur_obj
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
        
        # Stall handling: add noise
        if no_improve >= max_stall:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise = rng.standard_normal(n) * (0.005 + 0.003 * (no_improve // max_stall))
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it}, obj={best_obj:.6f}, M={M:.4e}, stall={no_improve}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def subgradient_descent(a0, time_limit, lr_init=0.01, momentum=0.9, decay=0.999):
    """Projected subgradient descent with momentum on the simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        # Compute subgradient using top peaks
        top_k = min(20, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_indices]
        
        # Temperature for softmax
        temp = max(peak_vals.max() * 0.1, 1e-15)
        weights = np.exp(peak_vals / temp - np.max(peak_vals / temp))
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for i, k in enumerate(peak_indices):
            if weights[i] < 1e-10:
                continue
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += weights[i] * 2.0 * a[k - idx]
        
        # Update with momentum
        v = momentum * v - lr * grad
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
        
        # Track best
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        lr *= decay
        it += 1
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Ramp (increasing)
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp_up'] = r.copy()
    inits['ramp_down'] = r[::-1].copy()
    
    # V-shaped (high at edges, low in middle)
    x = np.arange(n, dtype=np.float64)
    v_shape = 1.0 - np.minimum(x, n - 1 - x) / (n / 2)
    v_shape = np.maximum(v_shape, 0.01)
    v_shape /= v_shape.sum()
    inits['v_shape'] = v_shape.copy()
    
    # Inverted V (low at edges, high in middle)
    inv_v = np.minimum(x + 1, n - x)
    inv_v /= inv_v.sum()
    inits['inv_v'] = inv_v.copy()
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g.copy()
    
    # Gaussian edge
    ge = np.exp(-0.5 * ((x - (n-1)/2) / (0.1 * n)) ** 2)
    ge_edge = 1.0 / (1.0 + ge)
    ge_edge /= ge_edge.sum()
    inits['gauss_edge'] = ge_edge.copy()
    
    # Random Dirichlet (concentrated)
    inits['dirichlet_conc'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Random Dirichlet (spread)
    inits['dirichlet_spread'] = rng.dirichlet(np.ones(n) * 5.0)
    
    # Random with sparsity
    sparse = rng.random(n)
    sparse[sparse < 0.4] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse_random'] = sparse.copy()
    
    # "Plateau" - high in a random block
    block_size = max(2, n // 4)
    start = rng.integers(0, n - block_size)
    plateau = np.zeros(n)
    plateau[start:start + block_size] = 1.0
    plateau /= plateau.sum()
    inits['plateau'] = plateau.copy()
    
    # "Comb" pattern
    period = max(2, n // 8)
    comb = np.zeros(n)
    comb[::period] = 1.0
    comb /= comb.sum()
    inits['comb'] = comb.copy()
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        if len(pp) == n:
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp_interp = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(pp)), pp)
            ps = pp_interp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp_interp / ps
    
    return inits

"""
LP-based Frank-Wolfe with multi-scale optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines 'min-t' LP oracle, subgradient polishing with momentum,
FFT-based spectral initialization, and multi-length scanning.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, hstack
import math


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_min_t_lp(a, time_limit=8.0):
    """
    Solve: min t s.t. conv(a,b) <= t, sum(b) = 1, b >= 0, t >= 0.
    Returns (b_opt, t_opt) or (None, None).
    """
    n = len(a)
    A_conv = build_conv_matrix(a)
    
    t_col = csr_matrix((np.full(2 * n - 1, -1.0),
                        (np.arange(2 * n - 1), np.zeros(2 * n - 1, dtype=int))),
                       shape=(2 * n - 1, 1))
    A_ub = hstack([A_conv, t_col])
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * (n + 1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    
    if res.success:
        b_opt = np.maximum(res.x[:n], 0.0)
        s = b_opt.sum()
        if s < 1e-15:
            return None, None
        b_opt = b_opt / s
        t_opt = float(max(res.x[n], 0.0))
        return b_opt, t_opt
    return None, None


def solve_lp_fixed_m(a, M, time_limit=8.0):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A_conv = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_conv, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    
    if res.success:
        b_opt = np.maximum(res.x, 0.0)
        s = b_opt.sum()
        if s < 1e-15:
            return None
        return b_opt / s
    return None


def compute_obj_norm(a):
    """Objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def fwi_optimize(a0, time_limit, verbose=False, verbose_every=100):
    """Frank-Wolfe iterations alternating LP formulations."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 80
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        t_lp = min(6.0, remaining * 0.4)
        
        # Alternate LP formulations
        if it % 4 < 2:
            b, _ = solve_min_t_lp(a, time_limit=t_lp)
        else:
            M_try = M * (1.0 - 0.003 * min(no_improve // 10, 10))
            b = solve_lp_fixed_m(a, M_try, time_limit=t_lp)
        
        if b is None:
            no_improve += 5
            it += 1
            continue
        
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            best_t = 0.0
            best_f = best_obj
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidate = (1.0 - t) * a + t * b
                candidate = np.maximum(candidate, 0.0)
                sc = candidate.sum()
                if sc < 1e-15:
                    continue
                candidate = candidate / sc
                f_cand = compute_obj_norm(candidate)
                if f_cand < best_f - 1e-14:
                    best_f = f_cand
                    best_t = t
            
            if best_t < 0.03:
                no_improve += 1
            else:
                a = (1.0 - best_t) * a + best_t * b
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        if no_improve >= max_stall:
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            noise_scale = 0.004 * (1.0 + 0.5 * (no_improve // max_stall))
            a = np.maximum(a + rng.standard_normal(n) * noise_scale, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max(max_stall * 1.4, 60))
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it}, obj={best_obj:.6f}, M={M:.4e}, stall={no_improve}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def subgradient_descent(a0, time_limit, lr_init=0.015, momentum=0.92, decay=0.9985):
    """Projected subgradient descent with momentum on simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        top_k = min(25, len(conv))
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_idx]
        
        temp = max(peak_vals.max() * 0.08, 1e-15)
        w = np.exp(peak_vals / temp - np.max(peak_vals / temp))
        w /= w.sum()
        
        grad = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += w[i] * 2.0 * a[k - idx]
        
        v = momentum * v - lr * grad
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        lr *= decay
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse non-negative initializations (sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp_up'] = r.copy()
    inits['ramp_down'] = r[::-1].copy()
    
    # V-shape (high at edges)
    mid = (n - 1) / 2.0
    v_shape = 1.0 + 0.8 * (1.0 - np.abs(x - mid) / mid)
    v_shape = np.maximum(v_shape, 0.01)
    v_shape /= v_shape.sum()
    inits['v_shape'] = v_shape.copy()
    
    # Inverted V (high at center)
    inv_v = 1.0 - 0.8 * (1.0 - np.abs(x - mid) / mid)
    inv_v = np.maximum(inv_v, 0.01)
    inv_v /= inv_v.sum()
    inits['inv_v'] = inv_v.copy()
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - mid) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss_c'] = g.copy()
    
    # Two Gaussians at edges
    ge = np.exp(-0.5 * ((x - 0.1 * n) / (0.15 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.9 * n) / (0.15 * n)) ** 2)
    ge /= ge.sum()
    inits['gauss_edge'] = ge.copy()
    
    # FFT-based: random phase, flat amplitude
    A = np.ones(n // 2 + 1, dtype=np.complex128)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    A[1:] *= np.exp(1j * phases[1:])
    a_fft = np.fft.irfft(A * (n // 2 + 1), n)
    a_fft = np.maximum(a_fft, 0)
    if a_fft.sum() > 1e-10:
        a_fft /= a_fft.sum()
        inits['fft_phase'] = a_fft
    else:
        inits['fft_phase'] = np.ones(n) / n
    
    # Plateau
    bs = max(2, n // 5)
    st = rng.integers(0, max(1, n - bs))
    plat = np.zeros(n)
    plat[st:st + bs] = 1.0
    plat /= plat.sum()
    inits['plateau'] = plat.copy()
    
    # Random Dirichlet
    inits['dirich'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sparse random
    sp = rng.random(n)
    sp[sp < 0.4] = 0.0
    if sp.sum() > 1e-10:
        sp /= sp.sum()
        inits['sparse'] = sp
    else:
        inits['sparse'] = np.ones(n) / n
    
    if parent is not None:
        pp = np.maximum(parent, 0.0)
        if len(pp) == n:
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pi = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
            ps = pi.sum()
            if ps > 1e-10:
                inits['parent_int'] = pi / ps
    
    return inits


def propose_candidate(budget_s=1800, **kwargs):
    """
    Multi-phase search: subgradient scan -> LP-FW on diverse candidates ->
    deep LP -> subgradient polish -> perturbation restarts.
    """
    rng = np.random.default_rng(777)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_sequence = None
    best_score = float('inf')
    
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    def try_update(seq, tag):
        nonlocal best_sequence, best_score
        seq = np.maximum(seq, 0.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = seq.copy()
            print(f"  BEST [{tag}]: {score:.6f} n={len(seq)}")
        return score
    
    # Phase 1: Fast subgradient scan across lengths and inits
    print("=== Phase 1: Subgradient Scan ===")
    lengths = [256, 384, 512, 768]
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        inits = generate_inits(n, parent, rng)
        names = list(inits.keys())
        prio = ['parent', 'parent_int', 'v_shape', 'ramp_up', 'ramp_down',
                'gauss_edge', 'gauss_c', 'inv_v', 'fft_phase', 'uniform',
                'plateau', 'dirich', 'sparse']
        names.sort(key=lambda x: prio.index(x) if x in prio else 99)
        
        t_each = max(3.0, 100.0 / min(len(names), 10))
        
        for nm in names[:10]:
            if time.monotonic() > deadline - 380:
                break
            ta = min(t_each, deadline - time.monotonic() - 360)
            if ta < 2:
                break
            a_opt, _ = subgradient_descent(inits[nm], time_limit=ta,
                                           lr_init=0.025, momentum=0.93, decay=0.998)
            try_update(a_opt, f"P1 n={n} {nm}")
    
    print(f"P1 done. best={best_score:.6f}")
    
    # Phase 2: LP-FW on multiple candidates
    print("=== Phase 2: LP-FW Multi-Candidate ===")
    cands = []
    n_best = len(best_sequence) if best_sequence is not None else 384
    cands.append(('best_p1', best_sequence.copy() if best_sequence is not None else np.ones(384) / 384))
    
    # Add structured inits at best length
    inits_best = generate_inits(n_best, parent, rng)
    for nm in ['v_shape', 'ramp_up', 'ramp_down', 'gauss_edge', 'fft_phase']:
        if nm in inits_best:
            cands.append((f'P2_{nm}', inits_best[nm]))
    
    # Other lengths
    for nt in [256, 512]:
        if nt != n_best:
            it_ = generate_inits(nt, parent, rng)
            for nm in ['parent_int', 'v_shape']:
                if nm in it_:
                    cands.append((f'P2_{nm}_{nt}', it_[nm]))
    
    t_each_p2 = max(40.0, 500.0 / len(cands))
    for nm, cand in cands:
        if time.monotonic() > deadline - 250:
            break
        ta = min(t_each_p2, deadline - time.monotonic() - 200)
        if ta < 15:
            break
        print(f"  LP-FW {nm} n={len(cand)} t={ta:.0f}s")
        a_opt, _ = fwi_optimize(cand, time_limit=ta, verbose=False)
        try_update(a_opt, f"P2 {nm}")
    
    print(f"P2 done. best={best_score:.6f}")
    
    # Phase 3: Deep LP on global best
    if best_sequence is not None and time.monotonic() < deadline - 180:
        print("=== Phase 3: Deep LP ===")
        t_deep = min(350.0, deadline - time.monotonic() - 120)
        if t_deep > 20:
            print(f"  t={t_deep:.0f}s")
            a_opt, _ = fwi_optimize(best_sequence, time_limit=t_deep,
                                    verbose=True, verbose_every=80)
            try_update(a_opt, "P3 deep")
    
    # Phase 4: Subgradient polish
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("=== Phase 4: PGD Polish ===")
        t_pol = min(150.0, deadline - time.monotonic() - 60)
        if t_pol > 10:
            a_opt, _ = subgradient_descent(best_sequence, time_limit=t_pol,
                                           lr_init=0.004, momentum=0.95, decay=0.9995)
            try_update(a_opt, "P4 PGD")
    
    # Phase 5: Another LP pass
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("=== Phase 5: Final LP ===")
        t_fl = min(120.0, deadline - time.monotonic() - 30)
        if t_fl > 10:
            a_opt, _ = fwi_optimize(best_sequence, time_limit=t_fl,
                                    verbose=True, verbose_every=100)
            try_update(a_opt, "P5 LP")
    
    # Phase 6: Perturbation restarts
    if best_sequence is not None and time.monotonic() < deadline - 25:
        print("=== Phase 6: Restarts ===")
        nb = len(best_sequence)
        n_restart = min(5, max(1, int((deadline - time.monotonic() - 10) // 20)))
        for i in range(n_restart):
            if time.monotonic() > deadline - 10:
                break
            sc = 0.002 * (1.0 + 0.4 * i)
            noise = rng.standard_normal(nb) * sc
            ap = np.maximum(best_sequence + noise, 0.0)
            ap = ap / ap.sum()
            ta = min(18.0, deadline - time.monotonic() - 5)
            if ta < 5:
                break
            a_opt, _ = fwi_optimize(ap, time_limit=ta, verbose=False)
            try_update(a_opt, f"P6 r{i}")
    
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(max(0.0, x)) for x in best_sequence]
