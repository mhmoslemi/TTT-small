"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using log-space (softmax) gradient descent
with Adam optimizer, LP-based direction finding, simulated annealing perturbations,
and multi-phase diverse search. Key innovation: works in log-space to avoid simplex
projection, enabling cleaner gradient computation and potentially faster convergence.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def compute_grad_logspace(x, beta):
    """
    Compute gradient of LSE(beta * conv(softmax(x), softmax(x))) w.r.t. x.
    Returns (grad_x, conv, p) where p is the LSE weight vector.
    """
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)
    
    # LSE weights
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    
    # T1[i] = sum_k p[k] * a[k-i] (cross-correlation of p and a)
    # T2[i] = sum_k p[k] * a[i-k] (convolution of a and p)
    fft_size = 1 << (3 * n + 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    
    T1 = np.fft.irfft(P * np.conj(A), fft_size)[:n]
    T2 = np.fft.irfft(A * P, fft_size)[:n]
    
    grad_a = (T1 + T2) / beta
    mean_ga = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_ga)
    
    return grad_x, conv, p


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(a, M, n):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 20})
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum


def optimize_logspace(x0, time_limit, beta_init=50.0, beta_end=5000.0,
                      lr_init=1.0, momentum=0.9, seed=0):
    """
    Nesterov momentum optimization in log-space.
    x0: initial x (log-space)
    Returns (best_x, best_maxconv).
    """
    rng = np.random.default_rng(seed)
    x = x0.astype(np.float64).copy()
    n = len(x)
    
    a = softmax(x)
    conv = fast_convolve(a)
    best_maxconv = conv.max()
    best_x = x.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    total_iters = 0
    perturb_every = 200
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init * (beta_end / beta_init) ** frac
        
        # Nesterov lookahead
        y = x + momentum * v
        gy, conv_y, _ = compute_grad_logspace(y, beta)
        
        # Momentum update
        v = momentum * v - lr * gy
        x = x + v
        
        a = softmax(x)
        conv = fast_convolve(a)
        cur_maxconv = conv.max()
        total_iters += 1
        
        if cur_maxconv < best_maxconv - 1e-12:
            best_maxconv = cur_maxconv
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        lr = lr_init * (1.0 - 0.8 * frac)
        
        # Perturbation
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * 0.1
            x = x + noise
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.5)
            lr = max(0.01, lr * 0.5)
    
    return best_x, best_maxconv

                        lr_init=0.01, momentum=0.9, seed=0):
    """
    Projected gradient descent on the simplex (backup method).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_maxconv = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    perturb_every = 300
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init * (beta_end / beta_init) ** frac
        
        y = a + momentum * v
        # Gradient in a-space (not x-space)
        conv_y = fast_convolve(y)
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum
        
        # grad_a[i] = (1/beta) * sum_k p[k] * (a[k-i] + a[i-k])
        fft_size = 1 << (3 * n + 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        P = np.fft.rfft(p, fft_size)
        T1 = np.fft.irfft(P * np.conj(A), fft_size)[:n]
        T2 = np.fft.irfft(A * P, fft_size)[:n]
        grad = (T1 + T2) / beta
        
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_maxconv = conv.max()
        
        if cur_maxconv < best_maxconv - 1e-12:
            best_maxconv = cur_maxconv
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr = max(0.0001, lr_init * (1.0 - 0.7 * frac))
        
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.4)
    
    return best_a, best_maxconv


def simulated_annealing(a0, time_limit, seed=0):
    """
    Simulated annealing: random mass-transfer moves with temperature-based acceptance.
    Works in simplex space (a on simplex).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_maxconv = conv.max()
    best_a = a.copy()
    cur_maxconv = best_maxconv
    
    t0 = time.monotonic()
    T_init = 1e-4
    T_end = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        T = T_init * (T_end / T_init) ** frac
        
        # Random move: transfer mass from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        frac_transfer = rng.uniform(0.01, 0.2)
        amt = a[i] * frac_transfer
        if amt < 1e-10:
            continue
        
        a_new = a.copy()
        a_new[i] -= amt
        a_new[j] += amt
        if a_new[i] < 0:
            a_new[i] = 0
            a_new[j] += a[i]  # put everything
        
        conv_new = fast_convolve(a_new)
        maxconv_new = conv_new.max()
        delta = maxconv_new - cur_maxconv
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            cur_maxconv = maxconv_new
            if cur_maxconv < best_maxconv:
                best_maxconv = cur_maxconv
                best_a = a.copy()
        else:
            # Reject, but occasionally perturb more aggressively
            if rng.random() < 0.01:
                i2 = rng.integers(0, n)
                j2 = rng.integers(0, n)
                if i2 != j2:
                    amt2 = a[i2] * rng.uniform(0.05, 0.3)
                    a[i2] -= amt2
                    a[j2] += amt2
                    conv = fast_convolve(a)
                    cur_maxconv = conv.max()
    
    return best_a, best_maxconv


def lp_optimization(a0, time_limit, seed=0):
    """
    Repeated LP direction finding with line search.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_maxconv = conv.max()
    best_a = a.copy()
    current_a = a.copy()
    
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        M = conv_cur.max()
        
        # Solve LP with tight bound
        b = lp_direction(current_a, M, n)
        if b is None:
            # Try relaxed
            for factor in [1.001, 1.01, 1.05]:
                b = lp_direction(current_a, M * factor, n)
                if b is not None:
                    break
            if b is None:
                stuck += 1
                if stuck > 5:
                    break
                noise = rng.standard_normal(n) * 0.01
                current_a = project_simplex(current_a + noise)
                continue
        
        stuck = 0
        
        # Line search
        improved = False
        for t in [0.3, 0.5, 0.7, 0.9, 0.15, 0.05]:
            a_try = (1 - t) * current_a + t * b
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try /= s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()
            
            if obj_try < best_maxconv - 1e-12:
                best_maxconv = obj_try
                best_a = a_try.copy()
                current_a = a_try.copy()
                improved = True
                break
            elif obj_try < M:
                current_a = a_try.copy()
                if obj_try < best_maxconv:
                    best_maxconv = obj_try
                    best_a = a_try.copy()
                improved = True
                break
        
        if not improved:
            # Smaller step
            for t in [0.02, 0.05]:
                a_try = (1 - t) * current_a + t * b
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try /= s_try
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                if obj_try < M * 0.999:
                    current_a = a_try.copy()
                    if obj_try < best_maxconv:
                        best_maxconv = obj_try
                        best_a = a_try.copy()
                    break
            else:
                # No improvement, perturb
                noise = rng.standard_normal(n) * 0.005
                current_a = project_simplex(current_a + noise)
    
    return best_a, best_maxconv


def peak_coordinate_descent(a0, time_limit, seed=0):
    """
    Coordinate descent targeting convolution peaks.
    For each peak lag, try to reduce it by adjusting the contributing elements.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_maxconv = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()
        
        # Find top peaks
        top_k = min(3, len(conv_cur))
        peak_indices = np.argsort(conv_cur)[-top_k:][::-1]
        
        improved = False
        for k in peak_indices:
            k = int(k)
            if conv_cur[k] < max_val * 0.98:
                break
            
            # Elements contributing to conv[k]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            
            # Find the largest contributing pair
            contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            if len(contributions) == 0:
                continue
            
            max_idx = i_start + np.argmax(contributions)
            
            # Try reducing a[max_idx] and spreading to least contributing position
            for _ in range(5):
                j = rng.integers(0, n)
                if j == max_idx:
                    continue
                # Try to move mass away from peak
                for frac in [0.05, 0.1, 0.15]:
                    amt = a[max_idx] * frac
                    if amt < 1e-10:
                        break
                    a_try = a.copy()
                    a_try[max_idx] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_maxconv - 1e-12:
                        best_maxconv = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
                    elif conv_try.max() < max_val:
                        a = a_try.copy()
                        improved = True
                        break
            if improved:
                break
        
        if not improved:
            # Random pair adjustment
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = a[i] * frac
                    if amt < 1e-10:
                        break
                    a_try = a.copy()
                    a_try[i] = max(0, a_try[i] - amt)
                    a_try[j] += a_try[i] + a[i] - a[i]  # wait this is wrong
                    # Let me fix:
                    a_try = a.copy()
                    transfer = min(a[i] * frac, a[i] * 0.5)
                    a_try[i] -= transfer
                    a_try[j] += transfer
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_maxconv - 1e-12:
                        best_maxconv = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
                    elif conv_try.max() < max_val * 0.999:
                        a = a_try.copy()
                        improved = True
                        break
            if not improved:
                # Perturb
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    transfer = a[i] * 0.1
                    a[i] -= transfer
                    a[j] += transfer
                    conv = fast_convolve(a)
    
    return best_a, best_maxconv


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n, normalized to sum=1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Bell (Gaussian)
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss30'] = g
    
    # Wide Gaussian
    g2 = np.exp(-0.5 * ((x - 0.5 * n) / (0.45 * n)) ** 2)
    g2 /= g2.sum()
    inits['gauss45'] = g2
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1)))
    rc = np.maximum(rc, 1e-10)
    rc /= rc.sum()
    inits['raised_cos'] = rc
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(3, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Super-Gaussian (flatter top)
    sg = np.exp(-((x - 0.5 * n) / (0.35 * n)) ** 6)
    sg /= sg.sum()
    inits['super_gauss'] = sg
    
    # Power law decay (centered)
    pl = 1.0 / (1.0 + ((x - 0.5 * n) / (0.2 * n)) ** 4)
    pl /= pl.sum()
    inits['power_law'] = pl
    
    # Random Dirichlet
    inits['dir_05'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dir_2'] = rng.dirichlet(np.ones(n) * 2.0)
    inits['dir_5'] = rng.dirichlet(np.ones(n) * 5.0)
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses log-space gradient descent, LP direction finding, simulated annealing,
    and peak coordinate descent in a time-managed multi-phase strategy.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent available: {e}", flush=True)
    
    # === Phase 1: Quick Exploration (300s) ===
    print("=== Phase 1: Exploration ===", flush=True)
    phase1_deadline = t_start + 300.0
    
    lengths = [256, 384, 512, 640]
    scan_results = []
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 100:
            break
        
        inits = generate_inits(n, parent, rng)
        # Priority ordering
        priority = ['parent', 'parent_ds', 'parent_up', 'gauss30', 'raised_cos',
                    'super_gauss', 'uniform', 'dir_05', 'bimodal', 'power_law']
        names = [nm for nm in priority if nm in inits]
        names += [nm for nm in inits if nm not in priority]
        names = names[:6]
        
        time_for_n = max(20.0, (phase1_deadline - time.monotonic()) / max(1, len(lengths) - lengths.index(n)) / max(1, len(names)))
        per_init = min(40.0, time_for_n / max(1, len(names)))
        
        for name in names:
            if time.monotonic() > phase1_deadline - 80:
                break
            t_alloc = min(per_init, 35.0)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            # Use simplex PD for speed in exploration
            a_opt, obj_opt = optimize_simplex_pd(
                a0, t_alloc, beta_init=30, beta_end=1000,
                lr_init=0.01, momentum=0.9, seed=rng.integers(0, 10000))
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}", flush=True)
    
    # === Phase 2: Deep Optimization (1100s) ===
    print("=== Phase 2: Deep Optimization ===", flush=True)
    phase2_deadline = t_start + 1400.0
    
    # Select top configs
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and n == len(a_opt):
            top_configs.append((n, a_opt.copy()))
            seen.add(key)
        if len(top_configs) >= 4:
            break
    
    # Always include current best
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))
    
    # Add a random restart at best length
    best_n = len(best_sequence) if best_sequence is not None else 384
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.3)))
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 1.0)))
    
    n_configs = len(top_configs)
    phase2_budget = phase2_deadline - time.monotonic() - 150.0
    per_config = max(30.0, phase2_budget / max(1, n_configs))
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > phase2_deadline - 100:
            break
        t_alloc = min(per_config, 250.0)
        if t_alloc < 15:
            break
        
        print(f"  Config {i}: n={n} (alloc {t_alloc:.0f}s)", flush=True)
        
        # Sub-phase A: Log-space gradient (50%)
        t_a = t_alloc * 0.45
        if t_a > 10:
            x0 = np.log(np.maximum(a0, 1e-15))
            a_norm = a0 / a0.sum()
            x0 = np.log(np.maximum(a_norm, 1e-15))
            x_opt, obj_opt = optimize_logspace(
                x0, t_a, beta_init=50, beta_end=5000,
                lr_init=0.8, momentum=0.85, seed=rng.integers(0, 50000))
            a_log = softmax(x_opt)
            score = float(evaluate_sequence(a_log.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_log.copy()
                print(f"    LogSpace: {score:.6f}", flush=True)
            a0 = a_log  # continue from log-space result
        
        # Sub-phase B: Simplex PD refinement (25%)
        t_b = t_alloc * 0.25
        if t_b > 8:
            a_b, obj_b = optimize_simplex_pd(
                a0, t_b, beta_init=200, beta_end=10000,
                lr_init=0.005, momentum=0.92, seed=rng.integers(0, 50000))
            score = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_b.copy()
                print(f"    SimpPD: {score:.6f}", flush=True)
            a0 = a_b
        
        # Sub-phase C: LP direction (20%)
        t_c = t_alloc * 0.20
        if t_c > 5:
            a_c, obj_c = lp_optimization(a0, t_c, seed=rng.integers(0, 50000))
            score = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_c.copy()
                print(f"    LP: {score:.6f}", flush=True)
            a0 = a_c
        
        # Sub-phase D: Peak coordinate descent (10%)
        t_d = t_alloc * 0.10
        if t_d > 3:
            a_d, obj_d = peak_coordinate_descent(a0, t_d, seed=rng.integers(0, 50000))
            score = float(evaluate_sequence(a_d.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_d.copy()
                print(f"    PeakCD: {score:.6f}", flush=True)
            a0 = a_d
    
    # === Phase 3: Polishing (350s) ===
    print("=== Phase 3: Polishing ===", flush=True)
    phase3_deadline = deadline
    
    if best_sequence is not None:
        # 3a: Log-space with very high beta
        t_3a = min(120.0, (phase3_deadline - time.monotonic()) * 0.35)
        if t_3a > 10:
            a_norm = best_sequence / best_sequence.sum()
            x0 = np.log(np.maximum(a_norm, 1e-15))
            x_pol, obj_pol = optimize_logspace(
                x0, t_3a, beta_init=500, beta_end=20000,
                lr_init=0.3, momentum=0.9, seed=777)
            a_pol = softmax(x_pol)
            score = float(evaluate_sequence(a_pol.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pol.copy()
                print(f"  Polish LogSpace: {best_score:.6f}", flush=True)
        
        # 3b: LP refinement
        t_3b = min(100.0, (phase3_deadline - time.monotonic()) * 0.25)
        if t_3b > 5:
            a_lp, obj_lp = lp_optimization(best_sequence, t_3b, seed=888)
            score = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_lp.copy()
                print(f"  Polish LP: {best_score:.6f}", flush=True)
        
        # 3c: Simulated annealing
        t_3c = min(80.0, (phase3_deadline - time.monotonic()) * 0.2)
        if t_3c > 5:
            a_sa, obj_sa = simulated_annealing(best_sequence, t_3c, seed=999)
            score = float(evaluate_sequence(a_sa.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_sa.copy()
                print(f"  Polish SA: {best_score:.6f}", flush=True)
        
        # 3d: Final peak coordinate descent
        t_3d = max(0.0, (phase3_deadline - time.monotonic()) * 0.2)
        if t_3d > 3:
            a_pc, obj_pc = peak_coordinate_descent(best_sequence, t_3d, seed=1111)
            score = float(evaluate_sequence(a_pc.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pc.copy()
                print(f"  Polish PeakCD: {best_score:.6f}", flush=True)
        
        # 3e: One more LP if time
        t_3e = max(0.0, phase3_deadline - time.monotonic() - 5.0)
        if t_3e > 10:
            a_lp2, obj_lp2 = lp_optimization(best_sequence, t_3e, seed=2222)
            score = float(evaluate_sequence(a_lp2.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_lp2.copy()
                print(f"  Final LP: {best_score:.6f}", flush=True)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
