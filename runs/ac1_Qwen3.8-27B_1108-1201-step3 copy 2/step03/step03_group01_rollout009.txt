def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)

        C[i:i + n, i] = a
    return C

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Multi-M LP approach
        M_values = [M * 1.001, M * 0.995, M * 1.01]
        directions = []
        for M_trial in M_values:
            g = solve_lp(a, M_trial)
            if g is not None:
                g = g / g.sum()
                directions.append(g)
        
        if not directions:
            # Fallback: use subgradient
            no_improve += 50
            continue
        
        # Average directions
        d = np.mean(directions, axis=0) - a
        d = d - a  # Already normalized
        # Actually: d = mean(g's) - a
        d = np.mean(directions, axis=0) - a
        
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-12:
            no_improve += 1
        else:
            t_best, obj_best = line_search(a, d, best_obj, n)
            
            if obj_best < best_obj - 1e-12:
                a_new = a + t_best * d
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
                conv = fast_convolve(a)
                M = conv.max()
                best_obj = 2.0 * n * M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        iteration += 1
        
        # Stagnation handling
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 13 + 7)
            
            # Frequency-domain perturbation
            if perturb_count % 2 == 0 and n <= 2048:
                A = np.fft.rfft(a, 2*n-1)
                phases = rng.uniform(0, 2*np.pi, len(A))
                # Small phase perturbation
                new_A = A * np.exp(1j * (phases - np.pi) * 0.1)
                a_new = np.fft.irfft(new_A, 2*n-1)[:n]
                a_new = np.maximum(a_new, 0.0)
            else:
                scale = 0.005 + 0.003 * min(perturb_count, 10)
                a_new = a + rng.standard_normal(n) * scale
                a_new = np.maximum(a_new, 0.0)
            
            s = a_new.sum()
            if s > 1e-15:
                a = a_new / s
            else:
                a = best_a.copy()
            
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(300, 100 + 20 * pert_count)
            
            if perturb_count > 50:
                break
    
    return best_a, best_obj

def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Standard
    inits['uniform'] = np.ones(n) / n
    
    # Window functions
    inits['hann'] = (0.5 + 0.5 * np.cos(2 * np.pi * x / (n - 1))) 
    inits['hann'] = np.maximum(inits['hann'], 0)
    inits['hann'] = inits['hann'] / inits['hann'].sum()
    
    inits['hamming'] = 0.54 + 0.46 * np.cos(2 * np.pi * x / (n - 1))
    inits['hamming'] = np.maximum(inits['hamming'], 0)
    inits['hamming'] = inits['hamming'] / inits['hamming'].sum()
    
    inits['blackman'] = (0.42 + 0.5 * np.cos(2*np.pi*x/(n-1)) + 
                         0.08 * np.cos(4*np.pi*x/(n-1)))
    inits['blackman'] = np.maximum(inits['blackman'], 0)
    inits['blackman'] = inits['blackman'] / inits['blackman'].sum()
    
    # Raised cosine (different alpha)
    for alpha in [0.3, 0.5, 0.7]:
        wc = np.ones(n)
        edge = int(alpha * n // 2)
        if edge > 0:
            w = np.arange(edge)
            wc[:edge] = 0.5 * (1 + np.cos(np.pi * (2*w / (2*edge - 1) - 1)))
            wc[-edge:] = wc[:edge][::-1]
        inits[f'sine_{alpha}'] = wc / wc.sum()
    
    # Bimodal (mass at edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Bimodal with gradient
    bi2 = np.zeros(n)
    bi2[:edge] = np.linspace(1, 0.3, edge)
    bi2[-edge:] = np.linspace(0.3, 1, edge)
    inits['bimodal_grad'] = bi2 / bi2.sum()
    
    # Power law
    pw = (x + 1.0) ** 0.5
    inits['power'] = pw / pw.sum()
    
    pw2 = (x + 1.0) ** 1.5
    inits['power_steep'] = pw2 / pw2.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    return inits

def subgradient_polish(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp=0.0002):
    """Subgradient descent for polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # conv_k = sum_j a_j * a_{j+k}
            # d(conv_k)/d(a_i) = a_{i-k} (from j=i-k term) + a_{i+k} (from j=i term)
            # Actually: conv_k = sum_j a_j * a_{j+k}
            # d/d(a_i): if i = j, contributes a_{i+k}; if i = j+k, contributes a_{i-k}
            # So d(conv_k)/d(a_i) = a_{i+k} + a_{i-k} (where valid)
            j1 = i - k  # contribution from a_j * a_{j+k} where j+k=i => j=i-k
            j2 = i + k  # contribution from a_j * a_{j+k} where j=i => a_{i+k}
            # Wait, let me be more careful.
            # conv_k = sum_j a_j * a_{j+k} for j = max(0,-k) to min(n-1, n-1-k)
            # Actually for linear conv: conv(k) = sum_j a[j] * a[j+k] where 0<=j<n and 0<=j+k<2n-1
            # Hmm, numpy convolve: (a*b)[k] = sum_j a[j] * b[k-j]
            # For a*b = a*a: (a*a)[k] = sum_j a[j] * a[k-j]
            # d/d(a[i]): j=i gives a[k-i], j=k-i gives a[i]
            # So d(a*a)[k]/d(a[i]) = a[k-i] + a[i] (if k != i, both terms)
            # Or a[k-i] (if only j=i is valid) + a[i] (if only j=k-i is valid)
            
            # Simpler: for conv index k, the subgradient contribution to position i is:
            # a[(k-i) mod n]... no, this is linear not circular.
            # For linear convolution: (conv(a,a))[k] = sum_j a[j]*a[k-j] for valid j
            # d/d(a[i]) = a[k-i] (if 0 <= k-i < n) + a[i] (if 0 <= i < n and k-i valid... )
            # Actually: d/d(a[i]) of sum_j a[j]*a[k-j]:
            #   = a[k-i] (from term j=i, if 0<=k-i<n)
            #   + a[i] (from term k-j=i, i.e., j=k-i, if 0<=k-i<n)
            #   But wait, these are the same condition!
            #   If k != i: both terms are present if 0 <= k-i < n
            #   If k == i: it's just a[i] (from j=i, and k-j=0, so a[0]... no)
            #   
            # Let me just use the direct formula:
            # For the symmetric case (a*a), (a*a)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j]*a[k-j]
            # d/d(a[i]) = (if 0<=k-i<n) a[k-i] + (if 0<=i<n and k-i valid) ... 
            # 
            # OK let me just use a simpler approach for the subgradient:
            # If k >= 0: (a*a)[k] = sum_{j=0}^{n-1-k} a[j] * a[j+k]
            #   d/d(a[i]) = a[i+k] (from j=i) + a[i-k] (from j+k=i, i.e., j=i-k)
            #   = a[i+k] (if i+k < n) + a[i-k] (if i-k >= 0)
            # If k < 0: by symmetry, same as |k|
            
            k_abs = abs(k)
            if k >= 0:
                # (a*a)[k] = sum_{j=0}^{n-1-k} a[j] * a[j+k]
                if i + k < n:
                    sg[i] += w[i_peak] * 0  # placeholder
            # This is getting complicated. Let me use a different approach.
            pass
        
        # Actually, let me use a simpler subgradient computation
        # The subgradient of max_k (a*a)_k w.r.t. a_i is:
        # (a*a)_{k*} where k* is the argmax, and the "gradient" is the shifted a
        # Specifically: for each peak index k, the gradient contribution to position i is:
        #   a[(k-i) mod n] for circular, but for linear:
        #   a[k-i] if 0 <= k-i < n (from the j=i term)
        #   + a[i] if 0 <= k-i < n (from the j=k-i term) -- wait this is wrong
        
        # Let me just use the formula directly:
        # c_k = sum_j a_j * a_{j+k} (for the valid range)
        # dc_k/da_i: 
        #   The term where j = i: contributes a_{i+k} (valid if i+k < n and i >= 0)
        #   The term where j+k = i, i.e., j = i-k: contributes a_{i-k} (valid if i-k >= 0 and i-k < n)
        # So: dc_k/da_i = a_{i+k} * 1[0<=i+k<n] + a_{i-k} * 1[0<=i-k<n]
        
        for i_peak in range(len(peaks)):
            if w[i_peak] < 1e-12:
                continue
            k = int(peaks[i_peak])
            # Contribution from a_{i+k} (j=i term): valid when 0 <= i+k <= 2n-2
            # But a is only defined for 0 <= index < n
            # So: a[k - i] contributes (from j = k - i to a[j]*a[j+k])
            
            # Let's think of it as: c_k = sum_j a[j]*a[k-j] (numpy convention)
            # For the linear convolution (a*a)[k] = sum_{j} a[j] * a[k-j] where 0<=j<n and 0<=k-j<n
            # dc_k/da[i] = a[k-i] (if 0<=k-i<n) [from j=i term]
            #            + a[i] (if 0<=k-i<n) [from k-j=i, i.e., j=k-i term]
            # Wait no: dc_k/da[i]:
            #   In the sum sum_j a[j]*a[k-j], the index i appears in:
            #   - a[j] when j=i: contributes a[k-i] (if 0<=k-i<n)
            #   - a[k-j] when k-j=i, i.e., j=k-i: contributes a[i] (if 0<=k-i<n)
            #   These are the same condition! So:
            #   dc_k/da[i] = a[k-i] + a[i] if 0<=k-i<n, else just one term or zero.
            #   More precisely:
            #   dc_k/da[i] = a[k-i]*1[0<=k-i<n] + a[i]*1[0<=k-i<n]
            #   = (a[k-i] + a[i]) * 1[0<=k-i<n]
            #   BUT WAIT: if k-i = i (i.e., k=2i), then both terms are a[i], giving 2*a[i].
            #   This is correct because in the sum, the term a[i]*a[i] appears once but
            #   both a[j] and a[k-j] depend on a[i].
            
            # For linear conv: k ranges from 0 to 2n-2
            # 0 <= k-i < n means i-k < 0 < i... no.
            # 0 <= k-i < n means k-n < i < k, i.e., max(0, k-n+1) < i < min(n-1, k-1)
            # Hmm, actually 0 <= k-i <= n-1 means k-(n-1) <= i <= k
            # So i ranges from max(0, k-n+1) to min(n-1, k)
            
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                grad = a[k - idx] + a[idx]
                sg[idx] += w[i_peak] * grad
        
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        temp *= 1.001
        
        if no_improve >= 200:
            break
    
    return best_a, best_obj
