Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses Adam with smoothed-max subgradients, LP-FW polishing, and multi-length search.
Key innovations: Adam optimizer (faster than subgradient), smoothed max for better

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def multi_peak_subgradient(a, conv, n_peaks=20, temperature=0.1):
    """Compute a weighted subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    # Get top peaks
    top_k = min(n_peaks, len(conv))
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peak_indices]
    
    # Softmax weighting
    scaled = (peak_vals - peak_vals.max()) / max(temperature, 1e-15)
    w = np.exp(scaled)
    w /= w.sum()
    
    # Compute subgradient
    g = np.zeros(n)
    for i in range(len(peak_indices)):
        if w[i] < 1e-12:
            continue
        k = int(peak_indices[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        g[idx] += 2.0 * w[i] * a[k - idx]
    
    return g


def adam_optimize(a0, time_limit, lr=0.01, n_peaks=20, temperature_init=0.1,
                  temperature_end=0.01, verbose=False, verbose_every=500):
    """Adam optimizer with multi-peak subgradients and simplex projection."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    max_stall = 2000
    
    while time.monotonic() - t0 < time_limit:
        # Anneal temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        temperature = temperature_init * (temperature_end / temperature_init) ** progress
        
        # Compute subgradient
        g = multi_peak_subgradient(a, conv, n_peaks, temperature)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        # Evaluate

        cur_obj = 2.0 * n * conv.max()

            stall_count = 0
        else:

        a = a_new
        it += 1

            conv = fast_convolve(a)

            print(f"    Adam it={it}, best_obj={best_obj:.6f}, M={conv.max():.4e}, lr={lr:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def lp_oracle_and_update(a, M):
    """Solve LP oracle and return the Frank-Wolfe direction."""
    n = len(a)
    # Build the constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 3.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def fw_polish(a0, time_limit, max_iters=200):
    """Frank-Wolfe with LP oracle for polishing."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while it < max_iters and time.monotonic() - t0 < time_limit:
        b = lp_oracle_and_update(a, M)
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        
        if np.allclose(a, b, atol=1e-10):
            break
        
        # Line search
        best_t = 0.0
        best_local = 2.0 * n * M
        for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
            candidate = (1 - t) * a + t * b
            c = fast_convolve(candidate)
            obj = 2.0 * n * c.max()
            if obj < best_local:
                best_local = obj
                best_t = t
        
        if best_t < 0.05:
            # Try smaller steps
            for t in [0.01, 0.03, 0.05]:
                candidate = (1 - t) * a + t * b
                c = fast_convolve(candidate)
                obj = 2.0 * n * c.max()
                if obj < best_local:
                    best_local = obj
                    best_t = t

            a = (1 - best_t) * a + best_t * b
            conv = fast_convolve(a)
            M = conv.max()
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
        
        it += 1
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    # Edge-concentrated

    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    inits['edge'] = ec / ec.sum()
    
    # Gaussian
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Ramp (left-heavy)
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Sparse random
    sp = rng.random(n)
    sp[sp < 0.6] = 0.0
    if sp.sum() > 1e-10:
        inits['sparse'] = sp / sp.sum()
    
    # Palindromic random
    half = rng.dirichlet(np.ones(n // 2 + 1) * 0.3)
    pal = np.zeros(n)
    pal[:n // 2 + 1] = half
    pal[n // 2 + 1:] = half[:n // 2][::-1]
    inits['palindromic'] = pal / pal.sum()
    
    if parent is not None and parent.size > 0:
        if parent.size == n:
            p = np.maximum(parent, 0.0)
            if p.sum() > 1e-10:
                inits['parent'] = p / p.sum()
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0.0))
            if p.sum() > 1e-10:
                inits['parent_interp'] = p / p.sum()
    
    return inits
