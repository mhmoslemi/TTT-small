def fast_convolve(a):

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix A_ub for LP: (a*b)[k] <= M for all k=0..2n-2."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def build_lp_matrix_topk(a, top_k, conv):
    """Build constraint matrix using only top-k peak lags."""
    n = len(a)
    num_k = 2 * n - 1
    if top_k >= num_k:
        return build_lp_matrix(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    A_full = build_lp_matrix(a)
    return A_full[peaks]


def solve_lp_direction(a, M, A_ub=None, b_ub=None):
    """Solve LP: max sum(b) s.t. A_ub @ b <= b_ub, b >= 0."""
    n = len(a)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
        elif res.status == 9:  # time limit
            return res.x
    except Exception:
        pass
    return None


def line_search_step(a, b, best_obj, n, max_iters=12):
    """Find best step size t in (0,1] such that objective improves."""
    best_t = 0.0
    best_val = best_obj
    for t in [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_val - 1e-14:
            best_val = obj_new
            best_t = t
    return best_t, best_val


def lp_optimize_full(a0, time_limit, step_init=1.0, use_topk=None, relax=1.0,
                     verbose=False, seed=0):
    """LP-based optimization on the simplex with full or top-k constraints."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    max_no_improve = 60
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if use_topk is not None and use_topk < 2 * n - 1:
            A_ub = build_lp_matrix_topk(a, use_topk, conv)
        else:
            A_ub = build_lp_matrix(a)
        b_ub = np.full(len(A_ub), M * relax)

        b = solve_lp_direction(a, M * relax, A_ub=A_ub, b_ub=b_ub)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        # Line search
        t_found, obj_found = line_search_step(a, b, best_obj, n)

        if t_found > 0:
            a_new = (1.0 - t_found) * a + t_found * b
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            best_obj = obj_found
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if step > 0.05:
                step *= 0.8
            if no_improve > 4:
                noise_scale = 0.003 + 0.002 * (no_improve // 10)
                noise = rng.standard_normal(n) * noise_scale
                a = project_simplex(a + noise)
                step = step_init * 0.6
                no_improve = 0

        if no_improve >= max_no_improve:
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            step = step_init * 0.7
            no_improve = 0
            max_no_improve = min(250, int(max_no_improve * 1.3))

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.7f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def smooth_gradient(a, beta=500.0):
    """Compute smooth gradient of max(conv(a,a)) using softmax approximation."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (2 * n - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    grad = cross + conv_wa
    return grad, conv


def smooth_gradient_optimize(a0, time_limit, beta_init=500.0, lr_init=0.05,
                             momentum=0.9, verbose=False, seed=0):
    """Smooth gradient descent with momentum on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv = smooth_gradient(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.5
                beta = max(100.0, beta * 0.7)
                no_improve = 0

        beta = min(8000.0, beta * 1.002)

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Blackman window
    if n > 2:
        blackman = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
        blackman = np.maximum(blackman, 1e-10)
        inits['blackman'] = blackman / blackman.sum()

    # Blackman-Harris
    if n > 2:
        bh = (0.35875 - 0.48829 * np.cos(2 * np.pi * x / (n - 1)) +
              0.14128 * np.cos(4 * np.pi * x / (n - 1)) -
              0.01168 * np.cos(6 * np.pi * x / (n - 1)))
        bh = np.maximum(bh, 1e-10)
        inits['blackman_harris'] = bh / bh.sum()

    # Nuttall
    if n > 2:
        a0, a1, a2, a3 = 0.3635819, 0.4891775, 0.1365995, 0.0106411
        nuttall = (a0 - a1 * np.cos(2 * np.pi * x / (n - 1)) +
                   a2 * np.cos(4 * np.pi * x / (n - 1)) -
                   a3 * np.cos(6 * np.pi * x / (n - 1)))
        nuttall = np.maximum(nuttall, 1e-10)
        inits['nuttall'] = nuttall / nuttall.sum()

    # Kaiser window (beta=5)
    from scipy.special import i0
    if n > 2:
        beta_k = 5.0
        arg = np.sqrt(1 - (2 * x / (n - 1) - 1) ** 2)
        kaiser = i0(beta_k * arg) / i0(beta_k)
        kaiser = np.maximum(kaiser, 1e-10)
        inits['kaiser5'] = kaiser / kaiser.sum()

    # Triangular (peaked at edges)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g

    # Random
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['random_unif'] = rng.dirichlet(np.ones(n) * 1.0)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def evaluate_obj_normalized(a):
    """Evaluate the objective assuming sum(a)=1. Returns 2n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: full-constraint LP with adaptive line search,
    flat-top window initializations, multi-n exploration, gradient polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Quick Exploration ==========
    print("=== Phase 1: Quick Exploration ===")
    t_phase1 = time.monotonic()
    phase1_time = 180.0

    lengths = [192, 256, 320, 384, 448, 512, 640]
    scan_results = []

    for n in lengths:
        if time.monotonic() + 20 > t_phase1 + phase1_time:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Prioritize structured windows
        priority = ['parent', 'parent_ds', 'parent_up', 'blackman_harris',
                    'nuttall', 'kaiser5', 'blackman', 'bimodal', 'random_sparse']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        remaining = phase1_time - (time.monotonic() - t_phase1) - 10
        time_per_init = max(8.0, remaining / max(1, len(lengths) - lengths.index(n)) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() + 5 > t_phase1 + phase1_time:
                break
            t_alloc = min(time_per_init, t_phase1 + phase1_time - time.monotonic() - 5)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize_full(a0, time_limit=t_alloc,
                                              step_init=1.0, use_topk=None,
                                              relax=1.0, seed=n + 100)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}, results: {len(scan_results)}")
    for sc, nn, nm, _ in scan_results[:5]:
        print(f"    n={nn}, {nm}: {sc:.6f}")

    # ========== Phase 2: Deep LP Optimization ==========
    print("=== Phase 2: Deep LP Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = 900.0

    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 320

    # Collect diverse starting points
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:10]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy(), name))
            seen.add(key)
        if len(deep_starts) >= 6:
            break

    # Always include current best
    if best_sequence is not None:
        deep_starts.append((best_n, best_sequence.copy(), 'best'))

    # Add fresh starts at best_n
    inits_best = generate_initializations(best_n, parent, rng)
    for extra in ['blackman_harris', 'nuttall', 'kaiser5']:
        if extra in inits_best:
            deep_starts.append((best_n, inits_best[extra].copy(), extra))

    # Also try best_n+64 and best_n-64 if reasonable
    for delta in [64, -64]:
        alt_n = best_n + delta
        if 128 <= alt_n <= 1024:
            alt_inits = generate_initializations(alt_n, parent, rng)
            if 'blackman_harris' in alt_inits:
                deep_starts.append((alt_n, alt_inits['blackman_harris'].copy(), f'bh_{alt_n}'))

    num_starts = len(deep_starts)
    time_per_start = phase2_time / max(1, num_starts)
    time_per_start = max(30.0, time_per_start)

    run_count = 0
    for i, (n_start, a_start, label) in enumerate(deep_starts):
        if time.monotonic() + 15 > phase2_start + phase2_time:
            break
        t_alloc = min(time_per_start, phase2_start + phase2_time - time.monotonic() - 15)
        if t_alloc < 20:
            break

        print(f"  Deep LP [{i+1}/{num_starts}]: n={n_start}, label={label}, t_alloc={t_alloc:.0f}s")

        # Run LP with full constraints
        a_opt, obj_opt = lp_optimize_full(a_start, time_limit=t_alloc,
                                          step_init=1.0, use_topk=None,
                                          relax=1.0, seed=run_count * 13 + 7)

        score = float(evaluate_sequence(a_opt.tolist()))
        run_count += 1
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"    NEW BEST: score={score:.6f}")
        else:
            print(f"    score={score:.6f}")

        # Second run with relaxed constraints from the same start
        remaining_for_this = time_per_start - (time.monotonic() - (phase2_start + i * time_per_start))
        if remaining_for_this > 30:
            t_alloc2 = min(remaining_for_this * 0.6, 80.0)
            if t_alloc2 > 15:
                a_opt2, obj_opt2 = lp_optimize_full(a_start, time_limit=t_alloc2,
                                                    step_init=0.8, use_topk=max(50, n_start // 4),
                                                    relax=1.02, seed=run_count * 17 + 3)
                score2 = float(evaluate_sequence(a_opt2.tolist()))
                run_count += 1
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = a_opt2.copy()
                    print(f"    NEW BEST (relaxed): score={score2:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Gradient Optimization ==========
    print("=== Phase 3: Gradient Optimization ===")
    phase3_start = time.monotonic()
    phase3_time = 350.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        grad_configs = [
            {'beta_init': 400, 'lr_init': 0.04, 'momentum': 0.9},
            {'beta_init': 1000, 'lr_init': 0.02, 'momentum': 0.92},
            {'beta_init': 2000, 'lr_init': 0.01, 'momentum': 0.94},
            {'beta_init': 200, 'lr_init': 0.06, 'momentum': 0.88},
        ]

        time_per_grad = phase3_time / (len(grad_configs) + 4)

        for i, cfg in enumerate(grad_configs):
            if time.monotonic() + 10 > phase3_start + phase3_time:
                break
            t_alloc = min(time_per_grad, phase3_start + phase3_time - time.monotonic() - 10)
            if t_alloc < 15:
                break

            a_opt, obj_opt = smooth_gradient_optimize(
                best_sequence, time_limit=t_alloc, seed=i * 23 + 11, **cfg)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad [{i}]: NEW BEST score={score:.6f}")
            else:
                print(f"  Grad [{i}]: score={score:.6f}")

        # LP from gradient-improved start
        if time.monotonic() + 20 < phase3_start + phase3_time:
            t_alloc = min(60.0, phase3_start + phase3_time - time.monotonic() - 10)
            if t_alloc > 15:
                a_opt, obj_opt = lp_optimize_full(best_sequence, time_limit=t_alloc,
                                                  step_init=0.5, use_topk=None,
                                                  relax=1.0, seed=555)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP after grad: NEW BEST score={score:.6f}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Final Polish ==========
    print("=== Phase 4: Final Polish ===")
    phase4_start = time.monotonic()
    phase4_time = max(30.0, deadline - time.monotonic())

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Alternate LP and gradient with very small steps
        a = best_sequence.copy()
        v = np.zeros(n_best)
        step = 0.3
        polish_t0 = time.monotonic()
        polish_limit = phase4_time * 0.6

        while time.monotonic() - polish_t0 < polish_limit:
            conv = fast_convolve(a)
            M = conv.max()

            # LP step
            A_ub = build_lp_matrix(a)
            b_ub = np.full(len(A_ub), M)
            b = solve_lp_direction(a, M, A_ub=A_ub, b_ub=b_ub)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    t_f, obj_f = line_search_step(a, b, 2.0 * n_best * M, n_best)
                    if t_f > 0:
                        a_new = (1.0 - t_f) * a + t_f * b
                        a_new = np.maximum(a_new, 0)
                        a_new = a_new / a_new.sum()
                        a = a_new

            # Gradient step
            grad, _ = smooth_gradient(a, 2000)
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm
            v = 0.92 * v - 0.005 * grad
            a = project_simplex(a + v)

            if time.monotonic() - polish_t0 > polish_limit * 0.8:
                break

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  Polish: NEW BEST score={score:.6f}")

        # Tiny random perturbations
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 30:
            n_tries = min(int(remaining / 3), 30)
            for i in range(n_tries):
                if time.monotonic() > deadline - 20:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Perturb [{i}] s={scale}: NEW BEST {score:.6f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
