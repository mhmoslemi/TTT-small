`cross[j] = sum_m a[m] * p[m+j]` for j < 0 (which gives 0 if we extend p with zeros)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using focused LP-based iterative refinement
with adaptive step sizing, smooth gradient polishing with Nesterov acceleration,
and multi-perturbation search. Focuses on n=512 with symmetry-aware optimizations.
Key improvements: tighter time allocation to LP iteration, dual-guided step sizing,
and a warm-restart gradient phase for final polishing.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)]) if target == 1.0 else np.array([target])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    if not cond.any():
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_max_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0), A
    return None, A


def solve_lp_min_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def compute_smooth_obj_grad(a, t):
    """Compute soft-max smoothed objective and gradient for normalized a (sum=1).
    obj = (2n/t) * log(sum(exp(t * conv(a,a))))
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    max_scaled = scaled.max()
    exp_vals = np.exp(scaled - max_scaled)
    sum_exp = exp_vals.sum()
    log_sum_exp = np.log(sum_exp) + max_scaled
    p = exp_vals / sum_exp
    obj = (2.0 * n / t) * log_sum_exp

    # Gradient: grad[j] = 2n * sum_k p[k] * d(conv[k])/d(a[j])
    # d(conv[k])/d(a[j]) = (2 if 2j != k else 1) * a[k-j] for 0<=k-j<n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        factors = np.where(2 * js == k, 1.0, 2.0)
        grad[js] += 2.0 * n * p[k] * factors * a[k - js]
    return obj, grad


def lp_refine(a0, time_limit, step_init=0.35, step_decay=0.998, step_min=0.05,
              method='max_sum', m_mult=1.0, max_iters=5000):
    """LP-based iterative refinement with adaptive step size."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    current_method = method
    switch_counter = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max() * m_mult

        if current_method == 'max_sum':
            b = None
            if m_mult == 1.0:
                b, _ = solve_lp_max_sum(a, M)
            else:
                b, _ = solve_lp_max_sum(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            b, _ = solve_lp_min_peak(a)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

        if np.linalg.norm(b - a) < 1e-12:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.8, step * 1.02)
        else:
            no_improve += 1
            step = max(step_min, step * step_decay)

        a = a_new
        conv = conv_new

        # Switch method periodically
        if no_improve > 100:
            current_method = 'min_peak' if current_method == 'max_sum' else 'max_sum'
            step = max(step_init * 0.7, step_min)
            no_improve = 0
            switch_counter += 1
            if switch_counter > 6:
                break
        if no_improve > 500:
            break

    return best_a, best_obj


def nesterov_grad(a0, time_limit, t_init=20.0, lr_init=0.01, max_iters=3000):
    """Nesterov-accelerated gradient on soft-max smoothed objective."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    y = a.copy()
    v = np.zeros(n)
    mu = 0.5  # Nesterov momentum
    t_cur = t_init
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        obj_y, grad_y = compute_smooth_obj_grad(y, t_cur)

        # Nesterov update
        beta = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        y = a + mu * (a - y)
        obj_y, grad_y = compute_smooth_obj_grad(y, t_cur)
        a_new = project_simplex(a - lr * grad_y, target=1.0)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        v = a  # y update for Nesterov

        if it % 500 == 0 and t_cur < 200:
            t_cur = min(t_cur * 1.8, 200.0)
        if no_improve > 800:
            lr *= 0.5
            no_improve = 0
        if no_improve > 4000:
            break

    return best_a, best_obj


def momentum_grad(a0, time_limit, t_init=15.0, lr_init=0.02, momentum=0.9,
                  max_iters=5000):
    """Momentum-based gradient on soft-max smoothed objective."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t_cur = t_init
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        _, grad = compute_smooth_obj_grad(a, t_cur)
        v = momentum * v - lr * grad
        a = project_simplex(a + v, target=1.0)

        conv = fast_convolve(a)
        obj_new = 2.0 * n * conv.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 1000 == 0:
            lr *= 0.85
            if t_cur < 150:
                t_cur = min(t_cur * 2.0, 150.0)
        if no_improve > 1500:
            break

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using focused LP-based iterative refinement
with adaptive step sizing, smooth gradient polishing with momentum,
and multi-perturbation search. Focuses on n=512 with careful time allocation.
Key improvements over prior: tighter LP iteration budget, better step schedules,
dual-method switching, and Nesterov-accelerated gradient polishing.

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 1:
        return np.array([target]) if target > 0 else np.array([0.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    if not cond.any():
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_max_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def solve_lp_min_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0)
    return None


def compute_smooth_grad(a, t):
    """Compute gradient of soft-max smoothed objective (2n/t)*log(sum(exp(t*conv(a,a))))."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    max_scaled = scaled.max()
    exp_vals = np.exp(scaled - max_scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        factors = np.where(2 * js == k, 1.0, 2.0)
        grad[js] += 2.0 * n * p[k] * factors * a[k - js]
    return grad


def lp_refine(a0, time_limit, step_init=0.35, step_decay=0.998, step_min=0.05,
              method='max_sum', m_mult=1.0, max_iters=5000):
    """LP-based iterative refinement with adaptive step size."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    current_method = method
    switch_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max() * m_mult

        if current_method == 'max_sum':
            b = solve_lp_max_sum(a, M)
        else:
            b = solve_lp_min_peak(a)

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        if np.linalg.norm(b - a) < 1e-12:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        conv = conv_new
