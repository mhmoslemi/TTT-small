def coordinate_descent(a0, time_limit, max_sweeps=100):

    a = np.maximum(a0, 0.0)

            break
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-10:
                # Try setting it to a small value
                candidates = [0.001 / n, 0.01 / n]
            else:
                # Try several values around the current one
                candidates = [old_val * 0.5, old_val * 0.8, old_val, old_val * 1.2, old_val * 1.5]
                # Also try 0
                candidates.append(0.0)
            
            best_x = old_val
            best_j_obj = 2.0 * n * fast_convolve(a).max()
            
            for x in candidates:
                if x < -1e-10:
                    continue
                x = max(0.0, min(1.0, x))
                a[j] = x
                new_sum = a.sum()
                if new_sum < 1e-15:
                    a[j] = old_val
                    continue
                # Normalize
                a_norm = a / new_sum

        cur_obj = 2.0 * n * fast_convolve(a).max()
        if cur_obj < best_obj - 1e-10:

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}

            pp = np.maximum(parent[:n], 0)

                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    # Raised cosine
    for width in [0.3, 0.5, 0.7]:
        a = np.exp(-0.5 * ((x - 0.5 * n) / (width * n / 2)) ** 2)
        a = np.maximum(a, 0)
        a /= a.sum()
        inits[f'gauss_{width}'] = a
    
    # Cosine modulation
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        am = a.sum()
        if am > 1e-10:
            a /= am
            inits[f'cos_{freq}'] = a
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Bimodal
    peaks = [0.25 * n, 0.75 * n]
    a = np.zeros(n)
    for p in peaks:
        a += np.exp(-0.5 * ((x - p) / (0.1 * n)) ** 2)
    a /= a.sum()
    inits['bimodal'] = a
    
    # "U-shape" (high at edges)
    a = np.cos(np.pi * x / n) ** 2
    a = np.maximum(a, 0)
    a /= a.sum()
    inits['u_shape'] = a
    
    return inits
