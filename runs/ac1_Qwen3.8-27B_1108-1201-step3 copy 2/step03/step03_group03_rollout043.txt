"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on smoothed objective
with softmax parameterization, LP-based refinement, and multigrid search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def objective_lse(z, tau):
    """Smoothed objective value and gradient w.r.t. z using log-sum-exp."""
    n = len(z)
    a = softmax(z)
    C = fast_convolve(a)
    # Log-sum-exp of tau*C
    tau_C = tau * C
    m = tau_C.max()
    val = (2.0 * n / tau) * (m + np.log(np.sum(np.exp(tau_C - m))))
    # Gradient
    w = np.exp(tau_C - m)
    w /= w.sum()
    # grad_a[i] = (4n/tau) * sum_k w[k] * a[k-i]
    # Using FFT for cross-correlation
    size = 2 ** (3 * n - 1).bit_length()
    w_pad = np.zeros(size)
    w_pad[:2*n-1] = w
    a_pad = np.zeros(size)
    a_pad[:n] = a
    W = np.fft.rfft(w_pad)
    A = np.fft.rfft(a_pad)
    corr = np.fft.irfft(np.conj(W) * A, size)[:n]
    grad_a = (4.0 * n / tau) * corr
    # Map to z-space
    ga = np.dot(grad_a, a)
    grad_z = a * (grad_a - ga)
    return val, grad_z

def lbfgs_smooth(a_init, tau_init, tau_final, max_iter, time_limit):
    """L-BFGS optimization with temperature animation."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-10)
    a = a / a.sum()
    # Initialize z from a: z = log(a)
    z = np.log(a + 1e-300)
    z -= z.mean()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_val = (2.0 * n * fast_convolve(a).max())
    
    # Temperature animation
    n_anneal = 5
    for phase in range(n_anneal):
        tau = tau_init * (tau_final / tau_init) ** (phase / max(1, n_anneal - 1))
        max_iter_phase = max_iter // n_anneal
        
        def fun(z_in):
            v, g = objective_lse(z_in, tau)
            return v
        
        def grad(z_in):
            v, g = objective_lse(z_in, tau)
            return g
        
        try:
            result = minimize(fun, z, jac=grad, method='L-BFGS-B',
                            options={'maxiter': max_iter_phase, 'gtol': 1e-8})
            if result.success or result.get('nfev', 0) > 10:
                z = result.x
        except Exception:
            pass
        
        a_new = softmax(z)
        val = 2.0 * n * fast_convolve(a_new).max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_val

def lp_direction(f, M, n, max_iter=50):
    """LP-based directional optimization."""
    a = np.maximum(f, 0.0)
    a = a / a.sum()
    
    # Build constraint matrix
    # A_ub[k, j] = f[k-j] for valid k-j
    # A_ub[k, j] * g_j = sum_j f[k-j] * g_j = (f*g)[k]
    
    A_rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row[j] = f[k-j]
        A_rows.append(row)
    A_ub = np.array(A_rows)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'presolve': True, 'time_limit': 10})
        if result.success:
            g = np.maximum(result.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def lp_optimize(a_init, time_limit, blend_init=0.3, blend_min=0.005):
    """LP-based directional optimization loop."""
    a = np.maximum(a_init, 0.0)
    a = a / a.sum()
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        g = lp_direction(a, M, n)
        if g is None:
            break
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= 10:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on a log-sum-exp smoothed
objective with softmax parameterization, combined with LP-based refinement,
multigrid initialization, and basin-hopping search.

Key innovations over previous approach:
- L-BFGS with animated temperature for fast convergence to local minima
- FFT-based gradient for the smoothed objective (O(n log n))
- Multigrid coarse-to-fine initialization
- Multi-length, multi-restart exploration with time budgeting

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def objective_smooth(z, tau):
    """Compute smoothed objective value (log-sum-exp approximation of max)."""
    n = len(z)
    a = softmax(z)
    C = fast_convolve(a)
    tau_C = tau * C
    m = tau_C.max()
    val = (2.0 * n / tau) * (m + np.log(np.sum(np.exp(tau_C - m))))
    return val


def gradient_smooth(z, tau):
    """Compute gradient of smoothed objective w.r.t. z."""
    n = len(z)
    a = softmax(z)
    C = fast_convolve(a)

    # Softmax weights on convolution peaks
    tau_C = tau * C
    tau_C_shift = tau_C - tau_C.max()
    w = np.exp(tau_C_shift)
    w_sum = w.sum()
    w /= w_sum

    # h[i] = sum_{j=0}^{n-1} w[i+j] * a[j] via FFT cross-correlation
    N_fft = 2 ** (3 * n - 1).bit_length()
    w_pad = np.zeros(N_fft)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(N_fft)
    a_pad[:n] = a

    W = np.fft.fft(w_pad)
    A = np.fft.fft(a_pad)
    R = np.fft.ifft(W * np.conj(A)).real
    h = R[:n]

    # grad_z[j] = (4n/tau) * a[j] * (h[j] - dot(h, a))
    S = np.dot(h, a)
    grad_z = (4.0 * n / tau) * a * (h - S)
    return grad_z


def objective_grad(z, tau):
    """Combined objective and gradient for scipy minimize."""
    val = objective_smooth(z, tau)
    grad = gradient_smooth(z, tau)
    return val, grad


def lbfgs_optimize(a_init, tau_start=50.0, tau_end=800.0, n_anneal=6,
                   max_iter_per_phase=200, time_limit=100.0):
    """L-BFGS with temperature animation on softmax parameterization."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()

    # Initialize z from a
    z = np.log(a)
    z -= z.mean()

    t0 = time.monotonic()
    best_a = a.copy()
    best_true_obj = 2.0 * n * fast_convolve(a).max()

    for phase in range(n_anneal):
        if time.monotonic() - t0 > time_limit:
            break
        frac = phase / max(1, n_anneal - 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        def fun(z_in):
            return objective_smooth(z_in, tau)

        def grad(z_in):
            return gradient_smooth(z_in, tau)

        try:
            result = minimize(fun, z, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter_per_phase,
                                      'gtol': 1e-10, 'ftol': 1e-12})
            if result.x is not None:
                z = result.x
        except Exception:
            pass

        a_new = softmax(z)
        true_obj = 2.0 * n * fast_convolve(a_new).max()
        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a_new.copy()

    return best_a, best_true_obj


def build_constraint_matrix(n):
    """Build the LP constraint matrix A_ub where A_ub[k,j] = f[k-j]."""
    # For a given f, A_ub[k,j] = f[k-j] if 0<=k-j<n, else 0
    # But f changes, so we just store the indices
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, pairs, n, time_limit_lp=5.0):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                        method='highs',
                        options={'presolve': True, 'time_limit': time_limit_lp})
        if result.success:
            g = np.maximum(result.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def lp_refine(a_init, time_limit, blend_init=0.4, blend_min=0.003):
    """LP-based directional optimization."""
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init, 2.0 * len(a_init)
    a = a / s
    n = len(a)
    pairs = build_constraint_matrix(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        g = solve_lp_direction(a, M, pairs, n)
        if g is None:
            break

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= 8:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break

    return best_a, best_obj


def subgradient_refine(a_init, time_limit, lr_init=0.01, momentum=0.9, top_k=30):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init, 2.0 * len(a_init)
    a = a / s
    n = len(a)
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_imp = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        # Softmax weights on peaks
        scale = pmax * 0.1 + 1e-10
        w = np.exp(peak_vals / scale)
        w /= w.sum()

        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = 2.0 * w[idx]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 100:
            rng = np.random.default_rng()
            a = project_simplex(a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            v *= 0.5
            no_imp = 0
            lr = max(0.002, lr * 0.7)
            if lr < 0.002 and no_imp > 300:
                break

        lr *= 0.9995

    return best_a, best_obj


def upsample(a, new_n):
    """Upsample a sequence to new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_up = np.interp(x_new, x_old, a)
    a_up = np.maximum(a_up, 0.0)
    s = a_up.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_up / s


def downsample(a, new_n):
    """Downsample a sequence to new length via averaging blocks."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n < new_n:
        return upsample(a, new_n)
    # Average groups
    a_down = np.array([a[i * old_n // new_n:(i + 1) * old_n // new_n].mean()
                       for i in range(new_n)])
    a_down = np.maximum(a_down, 1e-10)
    return a_down / a_down.sum()


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations for a given length."""
    if rng is None:
        rng = np.random.default_rng(12345)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian (centered)
    sigma = 0.3 * n
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
    inits['gauss'] = g / g.sum()

    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Edge-concentrated
    edge = np.zeros(n)
    e = max(2, n // 8)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    # Random (Dirichlet with low concentration for sparsity)
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5 * (n - 1)) / (2 * (n - 1))))
    cos_w = np.maximum(cos_w, 0.01)
    inits['cosine'] = cos_w / cos_w.sum()

    # Ramp with slight smoothing
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp = 0.7 * ramp + 0.3 * n
    inits['ramp'] = ramp / ramp.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                inits['parent_interp'] = upsample(p / ps, n)

    return inits


def multigrid_optimize(target_n, time_limit, rng=None):
    """Multigrid: optimize at increasing resolutions."""
    if rng is None:

    # Build the sequence of sizes
    sizes = []
    n = 16
    while n < target_n:
        sizes.append(n)
        n = min(n * 2, target_n)
    if sizes[-1] < target_n:
        sizes.append(target_n)

    a = np.ones(16) / 16

def coordinate_descent(a_init, max_rounds=40, time_limit=30.0):
    """Simple coordinate descent: try adjusting pairs of elements."""
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        # Try moving mass from high-contribution to low-contribution positions
        order = np.argsort(a)[::-1]  # largest first
        for i_idx in range(min(20, n)):
            i = order[i_idx]
            if a[i] < 1e-6:
                continue
            delta = a[i] * 0.02
            for j_idx in order[i_idx + 1:i_idx + 30]:
                if time.monotonic() - t0 > time_limit:
                    break
                j = j_idx
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                if a_test[i] < 0:
                    continue
                a_test = a_test / a_test.sum()
                conv_t = fast_convolve(a_test)
                obj_t = 2.0 * n * conv_t.max()
                if obj_t < best_obj - 1e-12:
                    a = a_test
                    best_obj = obj_t
                    conv = conv_t
                    improved = True
                    break
        if not improved:
            break

    return a

                a_opt, obj_opt = lbfgs_optimize(a_p, tau_start=100, tau_end=1000,
                                                n_anneal=3, max_iter_per_phase=80,
                                                time_limit=t_alloc)
                if obj_opt < 2.0 * n_b * fast_convolve(best_sequence).max():
                    update_best(a_opt, f"bhop {perturb_scale} t{trial}")
                    best_sequence = a_opt.copy()

    # === Phase 4: Final LP + Coordinate Descent Polish ===
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic()

        # LP refinement
        t_lp = min(120.0, remaining * 0.4)
        if t_lp > 10:
            a_lp, obj_lp = lp_refine(best_sequence, t_lp, blend_init=0.15,
                                     blend_min=0.002)
            update_best(a_lp, "final LP")

        # Coordinate descent
        t_cd = min(60.0, remaining * 0.25)
        if t_cd > 5:
            a_cd = coordinate_descent(best_sequence, max_rounds=50, time_limit=t_cd)
            update_best(a_cd, "CD")

        # Another round of LP + subgrad
        if time.monotonic() < deadline - 30:
            t_final = min(60.0, deadline - time.monotonic() - 25)
            a_f, obj_f = lp_refine(best_sequence, t_final * 0.6, blend_init=0.1,
                                   blend_min=0.002)
            update_best(a_f, "final LP2")
            if time.monotonic() < deadline - 15:
                a_f2, obj_f2 = subgradient_refine(best_sequence, 10.0,
                                                  lr_init=0.003, momentum=0.93,
                                                  top_k=60)
                update_best(a_f2, "final SG")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        print("  Using fallback uniform(256)")

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
