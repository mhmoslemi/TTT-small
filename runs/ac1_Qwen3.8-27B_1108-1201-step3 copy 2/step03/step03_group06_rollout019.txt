```python
    """Initialize with flat power spectrum (random phases)."""
    # Random phases in frequency domain
    freq_size = 2 * n - 1
    fft_size = 1 << (freq_size - 1).bit_length()
    
    # Generate random phases
    phases = rng.uniform(0, 2*np.pi, size=fft_size//2 + 1)
    # Fortify: make the spectrum flat (all same magnitude)
    spectrum = np.exp(1j * phases)
    # Make it Hermitian symmetric
    spectrum_full = np.zeros(fft_size, dtype=complex)
    spectrum_full[:fft_size//2+1] = spectrum
    for i in range(1, fft_size//2):
        spectrum_full[fft_size - i] = np.conj(spectrum[i])
    
    # IFFT to get time domain
    a = np.fft.irfft(spectrum_full, fft_size)[:n]
    # Project to non-negative
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
```

```python
"""
Optimized step function search using LP direction finding with annealing,
spectral initializations, and multi-phase search strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho - (cssv - 1.0)) > 0
    if not cond.any():
        return np.ones(n) / n
    rho_val = rho[cond][-1]
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(a, constraint_indices):
    """Build sparse rows of the constraint matrix for given indices.
    
    For constraint k: sum_i a[k-i] * g[i] <= M (where valid k-i in [0,n-1])
    The row for constraint k has entries A[k, i] = a[k-i] if 0 <= k-i < n
    """
    n = len(a)
    num_constraints = len(constraint_indices)
    A = np.zeros((num_constraints, n))
    for row_idx, k in enumerate(constraint_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        idx = np.arange(i_min, i_max + 1)
        A[row_idx, idx] = a[k - idx]
    return A


def solve_lp(a, M, constraint_indices):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for k in constraint_indices."""
    n = len(a)
    A = build_constraint_matrix(a, constraint_indices)
    b = np.full(len(constraint_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_direction(a, M, top_k=None, diversity_penalty=0.0):
    """Find LP direction with optional diversity in constraint selection.
    
    Instead of just top-k, we can penalize clustering of constraints.
    """
    n = len(a)
    conv = fast_conv(a)
    num_k = 2 * n - 1
    
    if top_k is not None and top_k < num_k:
        # Select top-k constraints
        if diversity_penalty > 0:
            # Greedy selection: pick highest, then penalize neighbors
            selected = []
            scores = conv.copy()
            for _ in range(top_k):
                idx = np.argmax(scores)
                selected.append(idx)
                # Penalize neighbors
                left = max(0, idx - n // 4)
                right = min(num_k, idx + n // 4 + 1)
                scores[left:right] *= (1 - diversity_penalty)
            constraint_indices = selected
        else:
            constraint_indices = np.argpartition(conv, -top_k)[-top_k:]
    else:
        constraint_indices = list(range(num_k))
    
    return solve_lp(a, M, constraint_indices)


def lp_optimize(a0, time_limit, step_init=1.0, top_k=None, relax_init=1.0,
                relax_end=1.0, seed=0, verbose=False):

def grad_optimize(a0, time_limit, beta=500.0, lr=0.02, momentum=0.9, seed=0):
    """Subgradient/softmax optimization on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = 2.0 * n * fast_conv(a).max()
    best_a = a.copy()
    v = np.zeros(n)
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        scaled = beta * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
        # Use FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)
        
        # First part: sum_k w[k] * a[k-i] = cross-correlation
        # = ifft(conj(FFT(w)) * FFT(a))
        cross = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
        
        # Second part: sum_k w[k] * a[i-k] = convolution
        conv_wa = np.fft.irfft(W * A, fft_size)[:n]
        
        grad = cross + conv_wa
        
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm
        
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        conv_new = fast_conv(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = 0.02
                no_improve = 0
        
        beta = min(5000.0, beta * 1.002)
    
    conv_final = fast_conv(best_a)
    return best_a, 2.0 * n * conv_final.max()

def diverse_init(n, rng):
    """Generate a diverse set of initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Exponential decay
    for rate in [0.5, 1.0, 2.0, 5.0]:
        g = np.exp(-rate * x / n)
        inits[f'exp_{rate}'] = g / g.sum()
    
    # Gaussian
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()
    
    # Bimodal (edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Cosine
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        inits['cosine'] = cos_w / cos_w.sum()
    
    # Random (sparse)
    sparse = np.zeros(n)
    num_nonzero = max(1, n // 4)
    indices = rng.choice(n, size=num_nonzero, replace=False)
    sparse[indices] = rng.random(num_nonzero) + 0.1
    inits['sparse'] = sparse / sparse.sum()
    
    # Random (dense)
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_dense'] = r
    
    # Sawtooth up
    saw = (x % (n // 2 + 1)) / (n // 2)
    inits['sawtooth'] = saw / saw.sum()
    
    # Alternating (high-low pattern)
    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alternating'] = alt / alt.sum()
    
    return inits
```

```python
"""
Multi-phase step function optimization using LP direction finding with homotopy
annealing, Gerchberg-Saxton spectral initializations, subgradient polishing,
and adaptive constraint management to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho - (cssv - 1.0)) > 0
    if not cond.any():
        return np.ones(n) / n
    rho_val = rho[cond][-1]
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(a, constraint_indices):
    """Build constraint matrix rows: for constraint k, A[i] = a[k-i] if valid."""
    n = len(a)
    num_c = len(constraint_indices)
    A = np.zeros((num_c, n))
    for row_idx, k in enumerate(constraint_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        idx = np.arange(i_min, i_max + 1)
        A[row_idx, idx] = a[k - idx]
    return A


def solve_lp_direction(a, M, constraint_indices):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for k in constraint_indices."""
    n = len(a)
    A = build_constraint_matrix(a, constraint_indices)
    b = np.full(len(constraint_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def select_constraints(conv, n, strategy='topk', k=50, spread=0.0):
    """Select a subset of active constraints from the convolution peaks.
    
    strategy: 'topk' for highest values, 'diverse' for spread-out selection,
              'adaptive' for tolerance-based
    """
    num_k = 2 * n - 1
    if strategy == 'topk':
        if k < num_k:
            return np.argpartition(conv, -k)[-k:]
        return np.arange(num_k)
    elif strategy == 'diverse':
        # Greedy: pick highest, penalize neighbors
        selected = []
        scores = conv.copy()
        window = max(1, n // 8)
        for _ in range(min(k, num_k)):
            idx = np.argmax(scores)
            selected.append(idx)
            left = max(0, idx - window)
            right = min(num_k, idx + window + 1)
            scores[left:right] *= (1.0 - spread)
        return np.array(selected)
    elif strategy == 'adaptive':
        # Include all constraints within tolerance of max
        tol = 0.05
        threshold = conv.max() * (1.0 - tol)
        idx = np.where(conv >= threshold)[0]
        if len(idx) < 5:
            idx = np.argpartition(conv, -max(5, k))[-max(5, k):]
        if len(idx) > k * 2:
            # Too many, pick top k by value
            top_vals = conv[idx]
            top_k_local = np.argpartition(top_vals, -k)[-k:]
            idx = idx[top_k_local]
        return idx
    else:
        return np.argpartition(conv, -k)[-k:]


def lp_optimize(a0, time_limit, step_init=1.0, top_k=50, strategy='topk',
                relax_init=1.3, relax_end=1.0, spread=0.0, seed=0, verbose=False):
    """LP-based optimization with homotopy annealing on constraint tightness."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Smooth annealing: ease-in-out
        anneal_t = elapsed_frac * (2.0 - elapsed_frac)
        relax = relax_init + (relax_end - relax_init) * anneal_t

        conv = fast_conv(a)
        M = conv.max()

        # Select constraints
        constraint_indices = select_constraints(conv, n, strategy=strategy,
                                                k=top_k, spread=spread)
        # Solve LP
        b = solve_lp_direction(a, M * relax, constraint_indices)
        if b is None or b.sum() < 1e-15:
            # Fallback: try larger constraint set
            if top_k < 2 * n - 1:
                constraint_indices = select_constraints(conv, n, strategy='topk',
                                                        k=min(2 * top_k, 2 * n - 1))
                b = solve_lp_direction(a, M * relax, constraint_indices)
            if b is None or b.sum() < 1e-15:
                b = a.copy()

        b = b / b.sum()

        # Line search for optimal step
        best_local_obj = 2.0 * n * conv.max()
        best_local_a = a.copy()
        best_local_t = 0.0

        step_candidates = [min(1.0, step), min(1.0, step * 0.6),
                           min(1.0, step * 0.3), 0.1, 0.05]
        for t in step_candidates:
            a_try = (1.0 - t) * a + t * b
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_conv(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-10:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_local_a
            # Adapt step: increase slightly if we found good step
            step = max(0.1, min(1.0, best_local_t * 1.15))
            no_improve = 0
        else:
            no_improve += 1
            step *= 0.8
            if no_improve > 40:
                # Significant perturbation to escape local min
                noise_scale = 0.005 + 0.01 * min(3, no_improve // 50)
                noise = rng.standard_normal(n) * noise_scale
                a = project_simplex(a + noise)
                step = step_init * 0.6
                no_improve = 0
            elif no_improve > 15:
                # Mild perturbation
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                step = max(0.1, step * 0.7)

        total_iters += 1
        if verbose and total_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {total_iters}: obj={best_obj:.6f}, step={step:.3f}, "
                  f"relax={relax:.3f}, t={elapsed:.1f}s")

    conv_final = fast_conv(best_a)
    return best_a, 2.0 * n * conv_final.max()


def subgradient_optimize(a0, time_limit, beta_init=500.0, lr_init=0.02,
                         momentum=0.9, seed=0):
    """Smooth (softmax-approximated max) gradient descent on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = 2.0 * n * fast_conv(a).max()
    best_a = a.copy()
    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    no_improve = 0

    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        scaled = beta * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Gradient computation via FFT
        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)
        cross = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
        conv_wa = np.fft.irfft(W * A, fft_size)[:n]
        grad = cross + conv_wa

        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_conv(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 200:
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.7
                beta = beta_init * 0.8
                no_improve = 0

        beta = min(8000.0, beta * 1.003)

    conv_final = fast_conv(best_a)
    return best_a, 2.0 * n * conv_final.max()


def gs_init(n, num_iters=300, seed=0):
    """Gerchberg-Saxton initialization for approximately flat autocorrelation."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()

    for attempt in range(5):
        a = rng.random(n) + 0.01
        a = a / a.sum()
        success = False

        for _ in range(num_iters):
            A = np.fft.rfft(a, fft_size)
            # Target: flat magnitude
            target_mag = np.ones(fft_size // 2 + 1)
            phase = np.angle(A)
            A_new = target_mag * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a = np.maximum(a_new, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-10:
                a = rng.random(n) + 0.01
                continue
            a = a / a_sum
            success = True

        if success:
            conv = fast_conv(a)
            flatness = conv.max() / max(conv.mean(), 1e-15)
            if flatness < 2.5:  # reasonably flat
                return a

    # Fallback: just return a random positive sequence
    a = rng.random(n) + 0.01
    return a / a.sum()


def diverse_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for rate in [0.3, 1.0, 3.0]:
        g = np.exp(-rate * x / n)
        inits[f'exp_{rate}'] = g / g.sum()

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()

    # Bimodal (edges)
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    # Add slight decay within edges
    bi[:edge] *= np.linspace(1.0, 0.5, edge)
    bi[-edge:] *= np.linspace(0.5, 1.0, edge)
    inits['bimodal'] = bi / bi.sum()

    # Cosine
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.001)
        inits['cosine'] = cos_w / cos_w.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse
    sparse = np.zeros(n)
    num_nz = max(1, n // 3)
    indices = rng.choice(n, size=num_nz, replace=False)
    sparse[indices] = rng.random(num_nz) + 0.1
    inits['sparse'] = sparse / sparse.sum()

    # Alternating
    alt = np.ones(n)
    alt[1::2] = 0.4
    inits['alternating'] = alt / alt.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    # GS init
    inits['gs'] = gs_init(n, num_iters=200, seed=rng.integers(0, 100000))

    return inits

        {'step_init': 0.6, 'top_k': max(80, best_n // 3), 'strategy': 'topk',
         'relax_init': 1.1, 'relax_end': 1.0, 'spread': 0.0},
        {'step_init': 1.0, 'top_k': None, 'strategy': 'topk',
         'relax_init': 1.2, 'relax_end': 1.0, 'spread': 0.0},
    ]

    # Starting points
    starts = [best_sequence.copy()]
    for score, n, name, a_opt in scan_results[:4]:
        if n == best_n:
            starts.append(a_opt.copy())
    # Add GS init
    starts.append(gs_init(best_n, num_iters=200, seed=rng.integers(0, 99999)))
    # Add random
    inits_b = diverse_inits(best_n, rng, parent)
    if 'random' in inits_b:
        starts.append(inits_b['random'].copy())

    total_runs = len(starts) * len(configs)
    time_per_run = max(15.0, phase2_time / max(1, total_runs))

    run_count = 0
    for i, a_start in enumerate(starts):
        for j, cfg in enumerate(configs):
            if time.monotonic() - t2 > phase2_time:
                break
            t_alloc = min(time_per_run, phase2_time - (time.monotonic() - t2) - 10)
            if t_alloc < 12:
                break

            cfg_copy = dict(cfg)
            if cfg_copy.get('top_k') is not None:
                cfg_copy['top_k'] = max(15, int(cfg_copy['top_k'] * best_n / best_n))

            a_opt, obj_opt = lp_optimize(a_start, time_limit=t_alloc,
                                         seed=run_count * 17 + 3, **cfg_copy)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2 run {run_count} (start {i}, cfg {j}): {score:.6f}")
            run_count += 1

    best_n = len(best_sequence)
    print(f"Phase 2 done. Best: {best_score:.6f} (n={best_n})")

    # ========== Phase 3: Cross-Length Exploration ==========
    print(f"\n=== Phase 3: Cross-Length ===")
    t3 = time.monotonic()
    phase3_time = 250.0

    for n_try in [best_n * 2, best_n // 2, best_n + 64, best_n - 64,
                  512, 768, 1024]:
        if n_try < 64 or n_try > 2048 or n_try == best_n:
            continue
        if time.monotonic() - t3 > phase3_time - 20:
            break

        # Interpolate
        if len(best_sequence) >= n_try:
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                 np.linspace(0, 1, len(best_sequence)), best_sequence)
        else:
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                 np.linspace(0, 1, len(best_sequence)), best_sequence)
        a_interp = np.maximum(a_interp, 0.0)
        a_sum = a_interp.sum()
        if a_sum < 1e-10:
            continue
        a_interp = a_interp / a_sum

        t_alloc = min(50.0, phase3_time - (time.monotonic() - t3) - 10)
        if t_alloc < 15:
            break

        a_opt, obj_opt = lp_optimize(a_interp, time_limit=t_alloc,
                                     step_init=1.0, top_k=max(30, n_try // 8),
                                     strategy='adaptive',
                                     relax_init=1.3, relax_end=1.0,
                                     seed=hash(n_try) % 100000)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            best_n = n_try
            print(f"  P3 n={n_try}: {score:.6f}")

    best_n = len(best_sequence)
    print(f"Phase 3 done. Best: {best_score:.6f} (n={best_n})")

    # ========== Phase 4: Subgradient Polishing ==========
    print(f"\n=== Phase 4: Subgradient (n={best_n}) ===")
    t4 = time.monotonic()
    phase4_time = 350.0

    grad_configs = [
        {'beta_init': 400, 'lr_init': 0.02, 'momentum': 0.90},
        {'beta_init': 800, 'lr_init': 0.01, 'momentum': 0.92},
        {'beta_init': 1500, 'lr_init': 0.005, 'momentum': 0.94},
        {'beta_init': 2500, 'lr_init': 0.003, 'momentum': 0.95},
        {'beta_init': 200, 'lr_init': 0.03, 'momentum': 0.88},
    ]

    time_per = phase4_time / len(grad_configs)

    for i, cfg in enumerate(grad_configs):
        if time.monotonic() - t4 > phase4_time - 5:
            break
        t_alloc = min(time_per, phase4_time - (time.monotonic() - t4) - 5)
        if t_alloc < 10:
            break

        a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                              seed=i * 11 + 1, **cfg)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P4 grad[{i}]: {score:.6f}")

    # Also try from perturbed starts
    for i in range(3):
        if time.monotonic() - t4 > phase4_time - 5:
            break
        noise = rng.standard_normal(best_n) * 0.004
        a_pert = project_simplex(best_sequence + noise)
        t_alloc = min(30.0, phase4_time - (time.monotonic() - t4) - 5)
        if talloc < 10:
            break
        a_opt, obj_opt = subgradient_optimize(a_pert, time_limit=t_alloc,
                                              beta_init=1000, lr_init=0.01,
                                              momentum=0.93, seed=200 + i)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P4 pert[{i}]: {score:.6f}")

    best_n = len(best_sequence)
    print(f"Phase 4 done. Best: {best_score:.6f} (n={best_n})")

    # ========== Phase 5: Final Polish ==========
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # One more LP with very tight settings
        t_alloc = min(remaining * 0.35, 80.0)
        if t_alloc > 10:
            a_opt, _ = lp_optimize(best_sequence, time_limit=t_alloc,
                                   step_init=0.3, top_k=max(50, best_n // 5),
                                   strategy='adaptive',
                                   relax_init=1.05, relax_end=1.0,
                                   seed=77777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P5 LP: {score:.6f}")

    # Tiny perturbations
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        n_best = len(best_sequence)
        scales = [0.0005, 0.001, 0.002, 0.003, 0.005, 0.008, 0.001, 0.002]
        max_tries = int(remaining / 2)
        for i in range(min(max_tries, 20)):
            if time.monotonic() > deadline - 12:
                break
            scale = scales[i % len(scales)]
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            score = float(evaluate_sequence(a_pert.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pert.copy()
                print(f"  P5 micro[{i}] s={scale}: {score:.6f}")

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))
        best_score = final_score
    else:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
