def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        peaks = np.argpartition(conv, -top_k)[-top_k:]
        # Weight by softmax
        pv = conv[peaks]
        scaled = pv * 50.0
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[i] * f[k - idx]
    return grad

def fw_standard(f, time_limit, verbose=False):
    """Standard Frank-Wolfe with subgradient oracle."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient
        grad = compute_subgradient(f, conv, top_k=15)
        
        # FW direction: vertex with minimum gradient
        j = np.argmin(grad)
        
        # Line search
        best_alpha = 0.0
        best_val = best_obj
        
        # Try a few alpha values
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * np.zeros(n)
            ft[j] = alpha  # e_j direction
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f
            f[j] += best_alpha
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Perturb and restart
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            best_obj = conv.max()
            no_improve = 0
    
    return best_f, best_obj

def solve_lp_fast(f, M, top_constraints=200):
    """Solve LP with only the most important constraints."""
    n = len(f)
    conv_f = fast_convolve(f)  # This is f*f, not what we need
    # Actually we need to build the constraint matrix for (f*g) <= M
    # (f*g)[k] = sum_j f[j] * g[k-j]
    # For efficiency, use only the most likely binding constraints
    
    # The most binding constraints are those where f is largest
    # (since (f*g)[k] = sum_j f[j]*g[k-j] <= sum_j f[j] * max(g) ... )
    # Actually, let's just use all constraints but exploit sparsity
    
    c = -np.ones(n)
    # Build A_ub sparsely
    # A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # (f*g)[k] = sum_{j=0}^{n-1} f[j] * g[k-j] (with g[m]=0 if m outside [0,n-1])
    # So A_ub[k, j] = f[j] if 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
    
    # For efficiency, use scipy sparse
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def adam_smoothed(f0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, temp_init=10.0, temp_final=100.0):
    """Adam optimizer with smoothed max objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    log_temp = np.log(temp_init)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Temperature schedule (exponential)
        progress = min(1.0, t / max(1, int(time_limit * 50)))  # approximate
        cur_log_temp = log_temp + progress * (np.log(temp_final) - log_temp)
        cur_temp = np.exp(cur_log_temp)
        
        # Compute smoothed gradient
        conv = fast_convolve(f)
        scaled = conv * cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(2*n-1)
            w[np.argmax(conv)] = 1.0
        else:
            w /= wsum
        
        # Gradient: grad_i = 2 * sum_k w_k * f[k-i]
        # This is a cross-correlation
        grad = np.zeros(n)
        for i in range(n):
            k_start = i
            k_end = min(2*n-1, i+n)
            if k_start >= 2*n-1:
                break
            grad[i] = 2.0 * np.dot(w[k_start:k_end], f[:k_end-k_start])
        
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
    
    return best_f, best_obj

def peak_balance(f0, time_limit):
    """Try to equalize the top peaks of the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.001
    
    while time.monotonic() - t0 < time_limit:
        # Find top peaks
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        max_peak = peak_vals.max()
        min_peak = peak_vals.min()
        
        if max_peak - min_peak < 1e-10:
            break
        
        # Compute gradient that pushes down high peaks and up low peaks
        # For each peak k, the gradient contribution is 2 * f[k-i] for each i
        # We want to decrease high peaks and increase low peaks
        # So the "effective gradient" is: sum_k sign * 2 * f[k-i]
        # where sign is +1 for high peaks, -1 for low peaks
        
        grad = np.zeros(n)
        for k in peaks:
            weight = 1.0 if conv[k] > (max_peak + min_peak) / 2 else -1.0
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += weight * 2.0 * f[k - idx]
        
        # Step
        f = f - lr * grad
        f = project_simplex(f)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            lr = min(lr * 1.01, 0.01)
        else:
            lr *= 0.95
        
        if lr < 1e-7:
            break
    
    return best_f, best_obj

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent(f0, time_limit, lr=0.01, decay=0.999):
    """Mirror descent with negative entropy (multiplicative weights)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

        f = f / s

    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient (top-k peaks)
        top_k = min(20, 2*n-1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]

        w = np.exp(scaled)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_balance(f0, time_limit, top_peaks=15):
    """Iteratively balance the top peaks of the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)

        peaks = np.argpartition(conv, -top_k)[-top_k:]

        mean_peak = peak_vals.mean()

        grad = np.zeros(n)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Periodic structure
    period = max(2, n // 7)
    p = np.ones(n)
    p[::period] = 2.0
    inits.append(p / p.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def try_update(a, best_box, label=""):
    """Evaluate sequence and update best_box[0] (seq) and best_box[1] (score) if improved."""
    a = np.array(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return best_box[1]
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_box[1]:
        best_box[0] = a.copy()
        best_box[1] = score
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_box[1]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining FW-LP, standard FW,
    Adam, mirror descent, and peak balancing with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_box = [None, float('inf')]

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    # === Phase 1: Quick exploration (250s) ===
    print("=== Phase 1: Quick Exploration ===")
    phase1_end = time.monotonic() + 250.0

    if parent is not None:
        # FW-LP on parent
        t1 = min(120, phase1_end - time.monotonic() - 50)
        if t1 > 20:
            print(f"  FW-LP on parent (n={len(parent)}), t={t1:.0f}s")
            a_opt, obj = fw_lp_optimize(parent, time_limit=t1)
            try_update(a_opt, best_box, "p1_fwlp_parent")

        # Mirror descent on parent
        rem = phase1_end - time.monotonic() - 30
        if rem > 20:
            print(f"  Mirror descent on parent, t={min(rem,50):.0f}s")
            a_opt, obj = mirror_descent(parent, time_limit=min(rem, 50), lr=0.03)
            try_update(a_opt, best_box, "p1_md_parent")

    # Try other n values
    test_ns = [256, 320, 448, 512]
    for n in test_ns:
        rem = phase1_end - time.monotonic() - 20
        if rem < 25:
            break
        f_init = np.ones(n) / n
        t_alloc = min(rem // max(1, len(test_ns)), 40)
        if t_alloc < 15:
            t_alloc = 15

        a_opt, obj = fw_standard(f_init, time_limit=t_alloc)
        try_update(a_opt, best_box, f"p1_fw_n{n}")

    print(f"  Phase 1 best: {best_box[1]:.8f}")

    # === Phase 2: Deep optimization (700s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + 700.0

    if best_box[0] is not None:
        cur_best = best_box[0].copy()
        n_best = len(cur_best)

        # Extended FW-LP
        rem = phase2_end - time.monotonic() - 300
        if rem > 50:
            print(f"  Ext FW-LP (n={n_best}), t={min(rem,350):.0f}s")
            a_opt, obj = fw_lp_optimize(cur_best, time_limit=min(rem, 350))
            try_update(a_opt, best_box, "p2_fwlp")
            cur_best = best_box[0].copy() if best_box[0] is not None else cur_best

        # Adam with various settings
        rem = phase2_end - time.monotonic() - 200
        if rem > 40:
            print(f"  Adam run 1, t={min(rem,150):.0f}s")
            a_opt, obj = adam_smoothed(cur_best, time_limit=min(rem, 150),
                                       lr=0.02, temp_init=15, temp_final=150, top_k=35)
            try_update(a_opt, best_box, "p2_adam1")
            cur_best = best_box[0].copy() if best_box[0] is not None else cur_best

        rem = phase2_end - time.monotonic() - 150
        if rem > 30:
            print(f"  Mirror descent, t={min(rem,100):.0f}s")
            a_opt, obj = mirror_descent(cur_best, time_limit=min(rem, 100), lr=0.02)
            try_update(a_opt, best_box, "p2_md")
            cur_best = best_box[0].copy() if best_box[0] is not None else cur_best

        # Peak balance
        rem = phase2_end - time.monotonic() - 100
        if rem > 30:
            print(f"  Peak balance, t={min(rem,80):.0f}s")
            a_opt, obj = peak_balance(cur_best, time_limit=min(rem, 80), top_peaks=20)
            try_update(a_opt, best_box, "p2_pb")
            cur_best = best_box[0].copy() if best_box[0] is not None else cur_best

        # Another FW-LP round
        rem = phase2_end - time.monotonic() - 50
        if rem > 40:
            print(f"  FW-LP round 2, t={min(rem,150):.0f}s")
            a_opt, obj = fw_lp_optimize(cur_best, time_limit=min(rem, 150))
            try_update(a_opt, best_box, "p2_fwlp2")

    print(f"  Phase 2 best: {best_box[1]:.8f}")

    # === Phase 3: Fine polishing (400s) ===
    print(f"\n=== Phase 3: Fine Polishing ===")
    phase3_end = time.monotonic() + 400.0

    if best_box[0] is not None:
        cur_best = best_box[0].copy()

        # Fine Adam
        rem = phase3_end - time.monotonic() - 150
        if rem > 30:
            print(f"  Fine Adam, t={min(rem,150):.0f}s")
            a_opt, obj = adam_smoothed(cur_best, time_limit=min(rem, 150),
                                       lr=0.005, beta1=0.95, beta2=0.999,
                                       temp_init=80, temp_final=400, top_k=50)
            try_update(a_opt, best_box, "p3_adam_fine")
            cur_best = best_box[0].copy() if best_box[0] is not None else cur_best

        # Fine peak balance
        rem = phase3_end - time.monotonic() - 100
        if rem > 20:
            print(f"  Fine peak balance, t={min(rem,80):.0f}s")
            a_opt, obj = peak_balance(cur_best, time_limit=min(rem, 80), top_peaks=10)
            try_update(a_opt, best_box, "p3_pb_fine")
            cur_best = best_box[0].copy() if best_box[0] is not None else cur_best

        # Perturbation + Adam
        for p in range(8):
            rem = phase3_end - time.monotonic() - 20
            if rem < 10:
                break
            noise_scale = 0.002 + 0.001 * p
            noise = rng.standard_normal(len(cur_best)) * noise_scale
            f_pert = project_simplex(cur_best + noise)
            a_opt, obj = adam_smoothed(f_pert, time_limit=min(rem, 30),
                                       lr=0.01, temp_init=30, temp_final=200, top_k=40)
            try_update(a_opt, best_box, f"p3_pert{p}")
            if best_box[0] is not None:
                cur_best = best_box[0].copy()

    print(f"  Phase 3 best: {best_box[1]:.8f}")

    # === Phase 4: Final push (remaining time) ===
    print(f"\n=== Phase 4: Final Push ===")

    if best_box[0] is not None:
        cur_best = best_box[0].copy()
        n_best = len(cur_best)

        # Final FW-LP
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            print(f"  Final FW-LP, t={min(rem,120):.0f}s")
            a_opt, obj = fw_lp_optimize(cur_best, time_limit=min(rem, 120))
            try_update(a_opt, best_box, "p4_fwlp")
            cur_best = best_box[0].copy() if best_box[0] is not None else cur_best

        # Ul

tra-fine Adam
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = adam_smoothed(cur_best, time_limit=min(rem, 60),
                                       lr=0.002, beta1=0.97, beta2=0.9995,
                                       temp_init=200, temp_final=1000, top_k=60)
            try_update(a_opt, best_box, "p4_adam_ultra")

        # Tiny perturbations
        for p in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 5:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(cur_best + noise)
            a_opt, obj = mirror_descent(f_pert, time_limit=min(rem, 15), lr=0.01)
            try_update(a_opt, best_box, f"p4_tiny{p}")
            if best_box[0] is not None:
                cur_best = best_box[0].copy()

    # Fallback
    if best_box[0] is None:
        best_box[0] = np.ones(256) / 256
        best_box[1] = float(evaluate_sequence(best_box[0].tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_box[0].tolist()))
    if np.isfinite(final_score):
        best_box[1] = final_score

    print(f"\nFINAL SCORE: {best_box[1]:.8f} (n={len(best_box[0])})")
    return [float(x) for x in best_box[0]]
