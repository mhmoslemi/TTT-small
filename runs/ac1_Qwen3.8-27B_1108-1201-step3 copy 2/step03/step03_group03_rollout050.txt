"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent with
exact 1D convex line search. The key insight is that for a fixed coordinate i,
the peak of the auto-convolution is a convex function of a[i], enabling exact
1D minimization via ternary search. This is combined with multi-length search,
diverse initializations, and random restarts.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def coord_step(conv, a, i, n):
    """
    Given current convolution and coordinate index i, find the optimal value
    of a[i] that minimizes max(conv_new). Uses the fact that the peak is
    a convex function of a[i].
    Returns (x_opt, f_opt) where f_opt is the new peak value.
    """
    size = 2 * n - 1
    ai = a[i]
    
    # Compute coeff and quad arrays
    coeff = np.zeros(size)
    quad = np.zeros(size)
    
    k_start = max(0, i)
    k_end = min(size - 1, i + n - 1)
    
    for k in range(k_start, k_end + 1):
        j = k - i
        if 0 <= j < n:
            if j == i:
                quad[k] = 1.0
            else:
                coeff[k] = 2.0 * a[j]
    
    # base[k] = conv[k] - ai*coeff[k] - ai^2*quad[k]
    base = conv - ai * coeff - ai * ai * quad
    
    # f(x) = max_k (base[k] + x*coeff[k] + x^2*quad[k])
    # Check if x=0 is optimal (f is non-decreasing at 0)
    f0 = base.max()
    active = np.where(base >= f0 - 1e-14)[0]
    if len(active) > 0:
        min_coeff = coeff[active].min()
        if min_coeff >= -1e-14:
            return 0.0, f0
    
    # Find bracket for ternary search
    # Lower bound: 0
    # Upper bound: find where f(x) > f(0)
    lo = 0.0
    hi = max(ai, 0.01)
    
    # Expand hi until f(hi) > f(lo) or hi is large enough
    f_lo = f0
    for _ in range(20):
        f_hi = (base + hi * coeff + hi * hi * quad).max()
        if f_hi > f_lo:
            break
        hi *= 2.0
        if hi > 100.0:
            break
    
    # If f is still decreasing at hi, the minimum might be at 0 (but we already checked)
    # or at hi. Let's just use [0, hi] for ternary search.
    
    # Ternary search
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = (base + m1 * coeff + m1 * m1 * quad).max()
        f2 = (base + m2 * coeff + m2 * m2 * quad).max()
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    x_opt = (lo + hi) / 2.0
    f_opt = (base + x_opt * coeff + x_opt * x_opt * quad).max()
    
    return x_opt, f_opt


def coordinate_descent(a, time_limit, verbose=True):
    """
    Full coordinate descent with exact 1D line search.
    Minimizes max(conv(a,a)) (equivalently, the normalized objective).
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    n = len(a)
    conv = fast_convolve(a)
    current_peak = conv.max()
    best_a = a.copy()
    best_peak = current_peak
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            x_opt, f_opt = coord_step(conv, a, i, n)
            if f_opt < current_peak - 1e-12:
                # Update a[i] and conv
                ai = a[i]
                size = 2 * n - 1
                # Update conv incrementally
                k_start = max(0, i)
                k_end = min(size - 1, i + n - 1)
                for k in range(k_start, k_end + 1):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            conv[k] = conv[k] - ai * ai + x_opt * x_opt
                        else:
                            conv[k] = conv[k] + 2.0 * (x_opt - ai) * a[j]
                a[i] = x_opt
                current_peak = conv.max()
                improved = True
        
        if current_peak < best_peak - 1e-12:
            best_peak = current_peak
            best_a = a.copy()
        
        sweep += 1
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_peak / (best_a.sum() ** 2)
            print(f"    Sweep {sweep}: peak={best_peak:.8f}, obj={obj:.8f}, t={elapsed:.1f}s")
        
        if not improved:
            break
    
    return best_a, 2.0 * n * best_peak / (best_peak ** 2)  # placeholder

def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations of length n."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.4)
    
    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.25 * n) / (0.12 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * n) / (0.12 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()
    
    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Cosine window
    cos_w = np.cos(np.pi * x / n) ** 2 + 0.01
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
