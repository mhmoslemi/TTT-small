import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def convolve_subset(a, b, k_indices):
    """Compute (a*b)[k] for specific k values efficiently."""
    n = len(a)
    m = len(b)
    results = np.zeros(len(k_indices))
    for idx, k in enumerate(k_indices):
        i_min = max(0, k - m + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            results[idx] = 0.0
        else:
            results[idx] = np.dot(a[i_min:i_max+1], b[k-i_max:k-i_min+1][::-1])
    return results

def exact_subgradient(a, k_peak):
    """Compute exact subgradient of conv[k_peak] w.r.t. a."""
    n = len(a)
    sg = np.zeros(n)
    # sg[j] = 2*a[k_peak - j] if 0 <= k_peak - j < n
    j_min = max(0, k_peak - n + 1)
    j_max = min(n - 1, k_peak)
    if j_min <= j_max:
        sg[j_min:j_max+1] = 2.0 * a[k_peak - j_max : k_peak - j_min + 1][::-1]
    return sg

def mirror_descent_step(a, sg, eta):
    """Mirror descent (multiplicative weights) step on the simplex."""
    # a_new[j] = a[j] * exp(-eta * sg[j]) / sum(...)
    # For numerical stability, use log-domain
    log_a = np.where(a > 0, np.log(np.maximum(a, 1e-300)), -300.0)
    log_new = log_a - eta * sg
    # Subtract max for numerical stability
    log_new -= log_new.max()
    new_weights = np.exp(log_new)
    total = new_weights.sum()
    if total < 1e-300:
        return a.copy()
    return new_weights / total

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv, np.argmax(conv)

        num_k = len(k_indices)

        #              sum(b) = 1 (handled by bounds or equality)

        #      b[j] >= 0
        #      M' >= 0 (implicit since a >= 0 and b >= 0)

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()
    
    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['tri_edges'] = tri
    tri_c = (n - 1 - np.minimum(x, n - 1 - x)).astype(np.float64) + 1.0
    tri_c /= tri_c.sum()
    inits['tri_center'] = tri_c
    
    # Gaussians
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # Random
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random2'] = rng.dirichlet(np.ones(n))
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        if len(parent) < n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits

Uses exact subgradient with mirror descent (multiplicative weights) as the primary

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def exact_subgradient(a, k_peak):
    """
    Compute exact subgradient of conv[k_peak] w.r.t. a.
    d(conv[k])/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    sg = np.zeros(n)
    j_min = max(0, k_peak - n + 1)
    j_max = min(n - 1, k_peak)
    if j_min <= j_max:

        # a[k_peak - j] for j in [j_min, j_max] = a[k_peak - j_max : k_peak - j_min + 1] reversed

    """Mirror descent (multiplicative weights) step on the simplex."""
