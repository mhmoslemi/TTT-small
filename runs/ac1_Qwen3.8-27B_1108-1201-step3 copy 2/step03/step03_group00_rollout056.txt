Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with
direct minimization oracle, multi-phase search, and subgradient polishing.
Key improvements: (1) LP directly minimizes max(f*g) instead of fixing M,

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix for fixed f.
    C[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    So (C @ g)[k] = sum_j C[k,j] * g[j] = sum_j f[k-j] * g[j] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for j in range(n):
        # (f*g)[k] includes term f[k-j]*g[j]
        # k-j ranges over valid f indices: 0 <= k-j <= n-1
        # So k ranges from j to j+n-1
        C[j:j+n, j] = f
    return C

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_min_t(f):
    """Solve: min t s.t. (f*g)[k] <= t, sum(g)=1, g>=0.
    Variables: [t, g_0, ..., g_{n-1}] (n+1 variables)
    Returns (t_opt, g_opt) or (None, None)."""
    n = len(f)
    # minimize t
    c = np.zeros(n + 1)
    c[0] = 1.0  # minimize t
    
    # (f*g)[k] - t <= 0 for k = 0, ..., 2n-2
    # C @ g - t <= 0
    # In standard form: [C, -1] @ [t, g] <= 0
    C = build_conv_matrix(f)
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, 1:] = C
    A_ub[:, 0] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # sum(g) = 1 => sum(g) <= 1 and -sum(g) <= -1
    # Or use equality constraint
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    # bounds: t >= 0 (actually t >= 0 is implied), g >= 0
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[0], res.x[1:]
    return None, None

def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=48, temp=0.00005):
    """Subgradient descent with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient: d/d(a) max_k (a*a)[k] ≈ 2 * sum_k w_k * flip-shift(a, k)
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal
        cur_lr *= 0.9995
        cur_temp *= 1.005
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 99999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.0005)
            cur_temp = temp * 5
            no_improve = 0
    
    return best_f, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet (random)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Edge-concentrated (U-shaped)
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two peaks
    peak1 = np.exp(-0.5 * ((x - 0.25 * (n-1)) / (0.1 * n)) ** 2)
    peak2 = np.exp(-0.5 * ((x - 0.75 * (n-1)) / (0.1 * n)) ** 2)
    two = peak1 + peak2
    inits.append(two / two.sum())
    
    # Cosine modulation
    c = (1.0 + 0.5 * np.cos(2 * np.pi * x / n)) 
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # High-frequency oscillation
    hf = (1.0 + 0.3 * np.cos(4 * np.pi * x / n))
    hf = np.maximum(hf, 0.01)
    inits.append(hf / hf.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with
direct minimization oracle (min t s.t. (f*g)_k <= t), subgradient polishing
with adaptive softmax temperature, multi-length exploration, and
cosine-parameterized initializations. Key improvements over prior art:
(1) LP directly minimizes the next max value, (2) constraint-subset LP for
speed, (3) multi-frequency spectral inits, (4) aggressive time management.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C where (C@g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_min_t(f, active_lags=None):
    """Solve: min t s.t. (f*g)[k] <= t for k in active_lags, sum(g)=1, g>=0.
    Returns (t_opt, g_opt) or (None, None)."""
    n = len(f)
    C_full = build_conv_matrix(f)

    if active_lags is None:
        active_lags = np.arange(2 * n - 1)

    n_c = len(active_lags)
    C = C_full[active_lags, :]

    # Variables: [t, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0

    # Constraints: C @ g - t <= 0
    A_ub = np.zeros((n_c, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = C
    b_ub = np.zeros(n_c)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x[0], np.maximum(res.x[1:], 0.0)
    return None, None


def fw_direct_lp(f0, time_limit, verbose=False, top_k_constraints=None):
    """Frank-Wolfe with direct LP minimization oracle and constraint reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if top_k_constraints is None:
        top_k_constraints = max(n, 2 * n - 1)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Use top-k constraints for speed (those lags with highest f*f values)
        if top_k_constraints < 2 * n - 1:
            active = np.argpartition(conv, -top_k_constraints)[-top_k_constraints:]
        else:
            active = np.arange(2 * n - 1)

        t_opt, g = solve_lp_min_t(f, active)
        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Convergence check
        if np.max(np.abs(g - f)) < 1e-8:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in np.arange(0.05, 1.01, 0.05):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.005 + 0.008 * min(no_improve // 80, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=48, temp=0.00005):
    """Subgradient descent with adaptive softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9996
        cur_temp *= 1.004

        if no_improve > 350:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 99991)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.0005)
            cur_temp = temp * 4
            no_improve = 0

    return best_f, best_obj


def cosine_init(n, m, A, power=2):
    """Generate cosine-modulated initialization: a[j] ∝ (1 + A*cos(2π*m*j/n))^power."""
    x = np.arange(n, dtype=np.float64)
    base = 1.0 + A * np.cos(2.0 * np.pi * m * x / n)
    if power == 1:
        vals = np.maximum(base, 0.01)
    else:
        vals = base ** power
    vals = np.maximum(vals, 1e-6)
    return vals / vals.sum()


def generate_inits(n, parent, rng):
    """Generate diverse initializations including spectral ones."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine m=1
    inits.append(cosine_init(n, 1, 0.5, 2))

    # Cosine m=2
    inits.append(cosine_init(n, 2, 0.4, 2))

    # Cosine m=3
    inits.append(cosine_init(n, 3, 0.3, 2))

    # Two peaks
    p1 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.12 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append((p1 + p2) / (p1 + p2).sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits
