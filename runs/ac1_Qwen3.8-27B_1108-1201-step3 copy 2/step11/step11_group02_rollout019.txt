import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf'), 0
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max(), b

def lse_objective_and_grad(a, T):
    """
    Compute LSE-smoothed objective and gradient.
    f_T(a) = 2n * T * log(sum_k exp(b_k/T)) where b = conv(a,a)
    grad = 4n * sum_k w_k * a[k-i] (cross-correlation)
    """
    n = len(a)
    b = fast_convolve(a)
    w = np.exp((b - b.max()) / T)
    w_sum = w.sum()
    w = w / w_sum
    obj = 2.0 * n * (b.max() + T * np.log(w_sum))
    
    # Gradient: grad[i] = 4n * sum_j w[i+j] * a[j]
    # = 4n * conv(a, w[::-1])[2n-2-i]

    # Use FFT for the convolution
    N = 1 << (3 * n - 1).bit_length()
    A_ft = rfft(a, N)
    W_ft = rfft(w_rev, N)
    conv_full = irfft(A_ft * W_ft, N)
    # We need conv_full[2n-2-i] for i=0,...,n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    grad = 4.0 * n * conv_full[n-1:2*n-1][::-1]
    
    return obj, grad

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0]) if v[0] > 0 else np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_lbfgs_optimize(a0, T_stages, iters_per_stage, time_limit):
    """L-BFGS optimization on LSE-smoothed objective with decreasing T."""
    a = np.maximum(a0, 0)

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)[0]
    t0 = time.monotonic()
    n = len(a)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        
        def func(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            obj, _ = lse_objective_and_grad(a_trial, T)
            return obj
        
        def jac(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z)
            a_trial = a_trial / st
            _, grad = lse_objective_and_grad(a_trial, T)
            return grad
        
        try:

                                   'gtol': 1e-10, 'maxfun': iters_per_stage * 50,

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Compute convolve(a, b) where a has length n and b has length m."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()

def solve_lp_best_response(a, method='HiGHS'):
    """Given fixed a (normalized), find b that minimizes max conv(a,b)."""
    n = len(a)
    size = 2 * n - 1

    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # A[k, n] = -1

        je = min(n - 1, k)
        A[k, js:je+1] = a[k - np.arange(js, je+1)]
    A[:, n] = -1

    c = np.zeros(n + 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    # Bounds
    bounds = [(0, None)] * n + [(None, None)]  # b_i >= 0, t free
    
    res = linprog(c, A_ub=A, b_ub=np.zeros(size), A_eq=A_eq, b_eq=[1.0],
                  bounds=bounds, method=method)
    
    if res.status != 0:
        return None
    
    b = res.x[:n]
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:

Key differences from previous approach:
- Uses scipy HiGHS LP solver (faster than cvxpy) with efficient matrix construction
- LSE gradient uses FFT for the cross-correlation (O(n log n) per gradient)
- Dual-guided exploration: uses the LP dual to identify which constraints are active
- Adaptive Nesterov schedule with restarts based on duality gap
- Multi-scale T annealing for the LSE smoothing
- Structured random restarts using window functions at different parameters
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) for non-negative a (auto-normalizes)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def build_lp_matrices(a):
    """Build LP matrices for the best-response problem.
    Returns (c, A_ub, b_ub, A_eq, b_eq, bounds) for scipy linprog.
    """
    n = len(a)
    size = 2 * n - 1
    # A_ub[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # A_ub[k, n] = -1
    A_ub = np.zeros((size, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = a
    A_ub[:, n] = -1.0
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    b_ub = np.zeros(size)
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    return c, A_ub, b_ub, A_eq, b_eq, bounds

def solve_best_response(a):
    """Solve LP: min t s.t. conv(a,b)_k <= t, b>=0, sum(b)=1."""
    n = len(a)
    c, A_ub, b_ub, A_eq, b_eq, bounds = build_lp_matrices(a)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='HiGHS')
    if res.status != 0:
        return None
    b = np.maximum(res.x[:n], 0)
    s = b.sum()
    if s < 1e-15:
        return None
    return b / s

def lse_grad_fft(a, T):
    """Compute LSE smoothed objective and gradient via FFT.
    f_T = 2n * T * log(sum_k exp(b_k/T)), b = conv(a,a)
    grad[i] = 4n * sum_j w[i+j] * a[j] where w = softmax(b/T)
    """
    n = len(a)
    b = fast_convolve(a)
    b_shifted = (b - b.max()) / T
    exp_b = np.exp(b_shifted)
    exp_sum = exp_b.sum()
    w = exp_b / exp_sum
    obj = 2.0 * n * (b.max() + T * np.log(exp_sum))
    # grad[i] = 4n * sum_j w[i+j] * a[j]
    # = 4n * conv(a, w[::-1])[2n-2-i]
    w_rev = w[::-1]
    N = 1 << (3 * n - 1).bit_length()
    A_ft = rfft(a, N)
    W_ft = rfft(w_rev, N)
    conv_full = irfft(A_ft * W_ft, N)
    grad = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return obj, grad

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0)

def lse_optimize(a0, T_stages, iters_per_stage, time_limit):
    """LSE-smoothed optimization with decreasing T using projected gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        # Use L-BFGS via scipy for this stage
        from scipy.optimize import minimize as sp_minimize
        
        def func(z):
            a_t = np.maximum(z, 0)
            st = a_t.sum()
            if st < 1e-15:
                return 1e15
            a_t = a_t / st
            obj, _ = lse_grad_fft(a_t, T)
            return obj
        
        def jac(z):
            a_t = np.maximum(z, 0)
            st = a_t.sum()
            if st < 1e-15:
                return np.zeros(n)
            a_t = a_t / st
            _, g = lse_grad_fft(a_t, T)
            return g
        
        try:
            res = sp_minimize(func, a, jac=jac, method='L-BFGS-B',
                            bounds=[(0, None)] * n,
                            options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                     'gtol': 1e-12, 'maxfun': iters_per_stage * 100,
                                     'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception:
            pass
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        elapsed = time.monotonic() - t0
        print(f"    LSE T={T:.2f} ({elapsed:.0f}s): score={sc:.8f}")
    
    return best_a, best_score

def nesterov_lp(a0, time_limit, max_iters=500000, explore_interval=50):
    """Nesterov LP alternating with adaptive exploration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(777)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star = solve_best_response(y)
        if b_star is None:
            continue
        
        alpha = 2.0 / (np.sqrt(it + 1.0) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_star
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * (1 - alpha) * a + (it + 1) * alpha * b_star) / (it + 1)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        sc_a = compute_score(a)
        sc_y = compute_score(y)
        
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        
        if (it + 1) % explore_interval == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            
            if stall >= 5:
                # Explore: perturb toward reducing the peak
                b_cur = fast_convolve(y)
                peak_idx = int(np.argmax(b_cur))
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                idx = np.arange(js, je + 1)
                contribs = y[idx] * y[peak_idx - idx]
                
                # Reduce top contributors, add to bottom
                n_top = max(3, len(idx) // 8)
                top_pos = idx[np.argsort(contribs)[::-1][:n_top]]
                bot_pos = idx[np.argsort(contribs)[:n_top]]
                
                for p in top_pos:
                    if y[p] > 1e-10:
                        y[p] *= (1.0 - 0.03 * min(stall, 10))
                for p in bot_pos:
                    y[p] *= (1.0 + 0.02 * min(stall, 10))
                
                # Add random perturbation
                noise = rng.standard_normal(n) * 0.01 * np.sqrt(stall)
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                stall = 0
                prev_best = best_score
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, max_steps=100000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        # Find all positions near the max
        threshold = max_b * 0.999
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for peak_k in peaks[:2]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(6, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.995, 0.99]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-15:
                        continue
                    # Redistribute to others proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Try simultaneous reduction
                b_cur = fast_convolve(a)
                pk = int(np.argmax(b_cur))
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[pk - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:3]]
                a_trial = a.copy()
                for p in top_pos:
                    a_trial[p] *= 0.999
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                    elif stall > 1000:
                        break
        else:
            stall = 0
        
        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, seed):
    """Generate a structured initial sequence."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-4.0 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        beta = rng.uniform(3.0, 8.0)
        from scipy.special import i0
        a = i0(beta * np.sqrt(np.maximum(1 - (2*x-1)**2, 0))) / i0(beta)
    elif choice == 5:
        # Skewed bell
        mu = rng.uniform(0.3, 0.7)
        sigma = rng.uniform(0.15, 0.3)
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif choice == 6:
        # Rising edge
        a = x ** rng.uniform(1.0, 3.0)
    else:
        # Random smooth
        k = rng.integers(3, 10)
        points = np.sort(rng.uniform(0, 1, k))
        values = rng.uniform(0.5, 1.5, k)
        a = np.interp(x, points, values)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def try_update(best_box, a, label):
    """Try to update best solution in a mutable box."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = compute_score(a)
    if np.isfinite(sc) and sc < best_box['score']:
        best_box['score'] = sc
        best_box['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
