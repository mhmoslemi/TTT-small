"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Nesterov acceleration,
LogSumExp gradient descent, and peak shaving at multiple sequence lengths.
Uses geometric running averages and adaptive time allocation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_arr = np.clip(a_arr, 0.0, 1000.0)
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x>=0, sum(x)=target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Scale to work with sum=1 first
    u = v / target_sum if target_sum > 0 else v
    # Standard simplex projection for sum=1
    mu = np.zeros(n)
    sv = np.sort(u)[::-1]
    sv_cumsum = np.cumsum(sv)
    rho = np.where(sv * np.arange(1, n+1) > (sv_cumsum - 1))[0]
    if len(rho) > 0:
        rho = rho[-1]
        mu = (sv_cumsum[rho] - 1) / (rho + 1)
    else:
        return np.ones(n) / n
    w = np.maximum(u - mu, 0)
    return w * target_sum / w.sum() if w.sum() > 0 else np.ones(n) / n


def compute_lse_grad(z, T):
    """Compute (objective, gradient) for LogSumExp objective in z-space."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    # Gradient w.r.t. a: grad_a[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
    # Use FFT cross-correlation
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    
    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z

def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=15):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters_per_stage = max(10, min(60, int(30 + 20 * (si / n_stages))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score

                idx2 = np.arange(js2, je2 + 1)
                peak_contrib[idx2] += a[idx2] * a[k2 - idx2]

            candidates = np.where(peak_contrib < 0.3 * peak_contrib.max())[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Pick from candidates with lowest value
            cand_scores = a[candidates]
            sort_idx = np.argsort(cand_scores)
            top_candidates = candidates[sort_idx[:max(1, len(sort_idx) // 4)]]
            dst = top_candidates[rng.integers(len(top_candidates))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        # Strategy 2: Random perturbation
        if not improved:
            stall += 1
            n_pert = rng.integers(1, 5)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.04)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

            if stall > 500:
                break

    return best_a, best_score


def subgradient_polish(a0, time_limit, seed=42):
    """
    Subgradient polishing: use subgradient of max_k (a*a)_k to guide updates.
    After LP convergence, the max autocorrelation has a few dominant peaks.
    This method specifically targets reducing those peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    lr = 0.0005
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        b = fast_autocorr(a)
        max_b = b.max()

        # Find active peaks (within 0.5% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute weighted subgradient
        # For peak k: grad_k[i] = 2*a[k-i] if 0 <= k-i < n
        # Weight peaks by (b[k] - threshold) / (max_b - threshold) to focus on top peaks
        weights = (b[active] - threshold) / max(1e-15, max_b - threshold)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w_k, k in zip(weights, active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w_k * 2.0 * a[k - idx]

        # Projected subgradient step
        a_new = a - lr * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            lr *= 0.95
            stall += 1

        # Occasional random restart from nearby point
        if stall % 50 == 0 and stall > 0:
            noise = rng.normal(0, 0.001, n)
            a_noise = a + noise
            a_noise = np.maximum(a_noise, 0)
            ss = a_noise.sum()
            if ss > 1e-15:
                a_noise /= ss
                sc = 2.0 * n * fast_autocorr(a_noise).max()
                if sc < best_score:
                    a = a_noise
                    best_score = sc
                    best_a = a.copy()
                    stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        # Pseudo-random structured (m-sequence-like)
        a = 0.5 + 0.3 * np.sign(np.sin(2 * np.pi * np.arange(n) * 7.0 / n))
        a = a + 0.2 * np.sin(4 * np.pi * np.arange(n) / n)
    else:
        # Chen transform of Barker-like
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.4 * np.cos(2 * np.pi * 5 * x) + 0.1 * np.cos(2 * np.pi * 13 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization with LP Nesterov,
    LogSumExp gradient descent, subgradient polishing, and peak shaving
    at multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global_start)

    # Determine starting length
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length ===
    rem = remaining_time()
    t1 = min(rem * 0.35, 550.0)
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (time={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: LogSumExp refinement at base length ===
    rem = remaining_time()
    t2 = min(rem * 0.12, 180.0)
    print(f"\n=== Phase 2: LogSumExp at n={n_base} (time={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t2, T_start=80.0, T_end=0.05, n_stages=12)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: Subgradient polish at base length ===
    rem = remaining_time()
    t3 = min(rem * 0.10, 120.0)
    print(f"\n=== Phase 3: Subgradient polish at n={n_base} (time={t3:.0f}s) ===")
    if t3 > 10:
        a_out, _ = subgradient_polish(best_seq, t3, seed=99)
        try_update(a_out, f"subgrad_n{n_base}")

    # === Phase 4: LP Nesterov at longer length ===
    n_long = n_base + 200
    rem = remaining_time()
    t4 = min(rem * 0.25, 400.0)
    print(f"\n=== Phase 4: LP Nesterov at n={n_long} (time={t4:.0f}s) ===")
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t4, solver)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 5: LogSumExp at longer length ===
    rem = remaining_time()
    t5 = min(rem * 0.15, 200.0)
    print(f"\n=== Phase 5: LogSumExp at n={n_long} (time={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t5, T_start=200.0, T_end=0.05, n_stages=15)
        try_update(a_out, f"lse_n{n_long}")

    # === Phase 6: Peak shaving on best ===
    rem = remaining_time()
    t6 = min(rem * 0.15, 150.0)
    print(f"\n=== Phase 6: Peak shaving (time={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = peak_shaving(best_seq, t6, seed=456)
        try_update(a_out, "peak_shave_1")

    # === Phase 7: Try a different length ===
    n_alt = n_base - 100 if n_base > 1400 else n_base + 100
    rem = remaining_time()
    t7 = min(rem * 0.15, 200.0)
    print(f"\n=== Phase 7: LP Nesterov at n={n_alt} (time={t7:.0f}s) ===")
    if t7 > 30:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_out, _ = nesterov_lp_optimize(init_alt, t7, solver)
        try_update(a_out, f"lp_n{n_alt}")

        # Quick LSE polish on this length
        rem = remaining_time()
        t7b = min(rem * 0.10, 100.0)
        if t7b > 10:
            a_out, _ = logsumexp_optimize(best_seq, t7b, T_start=50.0, T_end=0.05, n_stages=10)
            try_update(a_out, f"lse_n{n_alt}")

    # === Phase 8: Final peak shaving ===
    rem = remaining_time()
    t8 = min(rem * 0.25, 120.0)
    print(f"\n=== Phase 8: Final peak shaving (time={t8:.0f}s) ===")
    if t8 > 5:
        a_out, _ = peak_shaving(best_seq, t8, seed=789)
        try_update(a_out, "peak_shave_final")

    # === Phase 9: Final subgradient polish ===
    rem = remaining_time()
    t9 = min(rem * 0.20, 80.0)
    print(f"\n=== Phase 9: Final subgradient polish (time={t9:.0f}s) ===")
    if t9 > 5:
        a_out, _ = subgradient_polish(best_seq, t9, seed=321)
        try_update(a_out, "subgrad_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
