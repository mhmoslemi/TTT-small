"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Theoretically-motivated C(2k,k)/4^k initialization (flat autocorrelation)
2. Aggressive mirror descent (multiplicative weights) with exact subgradients
3. L-BFGS with LogSumExp temperature annealing for local refinement
4. Multi-length exploration and adaptive restarts
Key insight: C(2k,k)/4^k has approximately flat self-convolution, giving
initial objective ~ pi/2, and optimization can further reduce peaks.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def init_binom(n):
    """Initialize with C(2k,k)/4^k sequence (flat autocorrelation property)."""
    a = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a / a.sum()

def compute_subgradient(a, active_lags):
    """Compute exact subgradient of max_k conv(a,a)[k] at given active lags."""
    n = len(a)
    g = np.zeros(n)
    for k in active_lags:

        je = min(n - 1, k)
        if je >= js:
            g[js:je + 1] += a[k - je: k - js + 1][::-1]

def mirror_descent(a0, time_limit, eta0=0.3, max_steps=200000, verbose=False):
    """Mirror descent (multiplicative weights) for peak shaving."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)

    total_steps = 0

        c_max = c.max()

        # Use lags within 0.5% of max for precision

            # Keep only the top 50

            top_idx = np.argsort(c_active)[-50:]
            active = active[top_idx]

        g_max = g.max()
        if g_max < 1e-15:
            break

        # Multiplicative update with backtracking
        found = False
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            a_new = a * np.exp(-eta * scale * g / g_max)

            if s < 1e-15:

            a_new = a_new / s

                a = a_new

                    best_a = a_new.copy()
                    stall_count = 0

def lse_obj_and_grad_z(z, T):
    """Compute LSE-smoothed objective and gradient in log-space.
    z are log-values, a = softmax(z).
    Objective: 2n * LSE(conv(a,a), T)
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    # Compute convolution
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    exp_c_sum = exp_c.sum()
    lse = c_max + T * np.log(exp_c_sum)
    obj = 2.0 * n * lse

    # Softmax weights over convolution
    w = exp_c / exp_c_sum

    # Gradient w.r.t. a: grad_a[j] = 2*sum_k w[k]*a[k-j] - a[j]*w[2j] (if valid)
    # First term: cross-correlation
    fft_size = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    R = irfft(W * np.conj(A), fft_size)
    b = R[:n]  # b[j] = sum_k w[k]*a[k-j]

    grad_a = 2.0 * b
    # Diagonal correction
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    grad_a[valid] -= a[valid] * w[idx[valid]]

    # Transform to z-space: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lbgfs_optimize(a0, time_limit, T_stages=None, verbose=False):
    """L-BFGS optimization with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20, 10, 5, 3, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        max_iter = min(2000, max(20, int(remaining * 50)))

        def func(z_in):
            obj, _ = lse_obj_and_grad_z(z_in, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_and_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxfun': max_iter * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if verbose:
            a = np.exp(z - z.max())
            a = a / a.sum()
            sc = evaluate_local(a)
            print(f"    LBFGS T={stage_T:.4f}: score={sc:.8f}")

    a = np.exp(z - z.max())
    a = a / a.sum()
    return a


def aggressive_peak_shave(a0, time_limit, n_rounds=5000, verbose=False):
    """Aggressive iterative peak shaving via direct constraint enforcement."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()
        
        # Target: reduce all peaks to below current max * (1 - reduction)
        target = c_max * 0.98
        
        # Identify lags that violate target
        violators = np.where(c > target)[0]
        if len(violators) == 0:
            break

        # For each violating lag k, we need to reduce conv(a,a)[k]
        # The most effective way is to reduce a_j where a[j]*a[k-j] is large
        # Use a weighted approach: reduce a proportionally to its contribution
        
        reduction = np.zeros(n)
        for k in violators:
            excess = c[k] - target
            if excess <= 0:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # Contribution of a[j] to c[k] is 2*a[j]*a[k-j] (or a[j]^2 if k=2j)
            contrib = 2.0 * a[js:je+1] * a[k-je: k-js+1][::-1]
            # Normalize contribution
            total = contrib.sum()
            if total > 1e-15:
                reduction[js:je+1] += (excess / total) * contrib
        
        # Apply reduction (with damping)
        damping = 0.3
        a_new = a - damping * reduction
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        
        if verbose and rnd % 1000 == 0:
            print(f"    PeakShave round {rnd}: score={best_score:.8f}, c_max={c_max:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, strength=0.03):
    """Perturb a sequence with various methods."""
    n = len(a)
    method = rng.integers(0, 4)
    if method == 0:
        # Gaussian
        a_new = a + strength * rng.normal(0, 1, n)
    elif method == 1:
        # Low frequency
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        a_new = a + strength * np.cos(freq * x + phase)
    elif method == 2:
        # Block perturbation
        a_new = a.copy()
        bs = rng.integers(10, max(11, n // 10))
        i = rng.integers(0, n - bs)
        a_new[i:i+bs] += strength * rng.standard_normal(bs)
    else:
        # Edge emphasis
        a_new = a.copy()
        edge = rng.integers(0, 2)
        w = rng.uniform(0.01, 0.03)
        e = max(10, n // 20)
        if edge == 0:
            a_new[:e] += w
            a_new[e:] -= w * e / (n - e)
        else:
            a_new[-e:] += w
            a_new[:-e] -= w * e / (n - e)
    return normalize(a_new)


def population_refine(a0, time_limit, pop_size=6, rng=None, verbose=False):
    """Population-based refinement with mirror descent on each member."""
    if rng is None:
        rng = np.random.default_rng(789)
    
    n = len(a0)
    population = [normalize(a0.copy())]
    for i in range(pop_size - 1):
        population.append(perturb_sequence(a0, rng, 0.02 + 0.01 * (i % 4)))
    
    scores = [evaluate_local(p) for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]
    
    t0 = time.monotonic()
    gen = 0
    
    while time.monotonic() - t0 < time_limit:
        gen += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break
        
        # Sort by score
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
        
        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()
        
        # Keep top half
        keep = max(2, pop_size // 2)
        elites = population[:keep]
        
        # Generate new population
        new_pop = list(elites)
        t_sub = max(3, (remaining - 5) / (pop_size - len(new_pop)))
        
        for i in range(pop_size - len(new_pop)):
            parent = elites[rng.integers(0, keep)]
            child = perturb_sequence(parent, rng, 0.015 + 0.01 * rng.random())
            # Brief mirror descent on child
            refined, _ = mirror_descent(child, t_sub, eta0=0.2, max_steps=2000)
            new_pop.append(refined)
        
        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]
        
        if verbose and gen % 5 == 0:
            print(f"    Pop gen {gen}: best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses C(2k,k)/4^k initialization, mirror descent peak shaving,
    L-BFGS with LSE annealing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    print(f"Budget: {budget_s:.0f}s, deadline in {budget_s-15:.0f}s")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Generate initial candidates at multiple lengths
    lengths_to_try = [1200, 800, 1500, 2000]
    if parent is not None:
        lengths_to_try.append(len(parent))
    
    # Deduplicate and sort
    lengths_to_try = sorted(set(lengths_to_try))
    
    print(f"Lengths to try: {lengths_to_try}")
    
    # Phase 0: Quick evaluation of initial candidates
    print("\n=== Phase 0: Initial Candidates ===")
    for n_try in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        init = init_binom(n_try)
        sc = evaluate_local(init)
        print(f"  Binom init n={n_try}: score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_seq = init.copy()
            print(f"  NEW BEST [binom_{n_try}]: {sc:.8f}")

    # Phase 1: Mirror descent on best initial
    print(f"\n=== Phase 1: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent(best_seq, t_alloc, eta0=0.4, max_steps=300000, verbose=True)
        local_update(a_md, "mirror_descent_1")

    # Phase 2: L-BFGS with LSE
    print(f"\n=== Phase 2: L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        T_stages = [50, 30, 20, 10, 5, 3, 2, 1, 0.7, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
        print(f"  T_stages: {len(T_stages)}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_optimize(best_seq, t_alloc, T_stages, verbose=True)
        local_update(a_lb, "lbgfs_1")

    # Phase 3: Peak shaving
    print(f"\n=== Phase 3: Peak Shave (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_ps, _ = aggressive_peak_shave(best_seq, t_alloc, n_rounds=10000, verbose=True)
        local_update(a_ps, "peak_shave_1")

    # Phase 4: Fine mirror descent
    print(f"\n=== Phase 4: Fine Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 20:
        a_md2, _ = mirror_descent(best_seq, t_alloc, eta0=0.1, max_steps=500000, verbose=True)
        local_update(a_md2, "mirror_descent_2")

    # Phase 5: L-BFGS fine polish
    print(f"\n=== Phase 5: L-BFGS Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 180.0)
    if t_alloc > 20:
        T_fine = [5, 3, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        a_lb2 = lbgfs_optimize(best_seq, t_alloc, T_fine, verbose=True)
        local_update(a_lb2, "lbgfs_fine")

    # Phase 6: Population refinement
    print(f"\n=== Phase 6: Population Refine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 180.0)
    if t_alloc > 30:
        a_pop, _ = population_refine(best_seq, t_alloc, pop_size=6, rng=rng, verbose=True)
        local_update(a_pop, "population")

    # Phase 7: Try other lengths
    print(f"\n=== Phase 7: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_best = len(best_seq)
        for n_try in [n_best + 200, n_best - 200, n_best + 500, n_best + 1000]:
            if n_try < 200 or n_try > 5000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            # Quick MD then LBFGS
            t_md = min(rem_sub * 0.4, 30.0)
            t_lb = min(rem_sub * 0.5, 30.0)
            a_try, _ = mirror_descent(init, t_md, eta0=0.3, max_steps=50000)
            local_update(a_try, f"md_len_{n_try}")
            a_try2 = lbgfs_optimize(a_try, t_lb, T_stages=[10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01], verbose=False)
            local_update(a_try2, f"lb_len_{n_try}")

    # Phase 8: Final combined polish
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        # Quick MD
        t_md = min(rem * 0.4, 40.0)
        if t_md > 5:
            a_fin1, _ = mirror_descent(best_seq, t_md, eta0=0.05, max_steps=100000)
            local_update(a_fin1, "final_md")
        
        # Quick LBFGS
        rem2 = deadline - time.monotonic()
        t_lb = min(rem2 * 0.5, 30.0)
        if t_lb > 5:
            a_fin2 = lbgfs_optimize(best_seq, t_lb, T_stages=[1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001], verbose=False)
            local_update(a_fin2, "final_lbfgs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        # Print some stats
        c = fast_convolve(best_seq)
        print(f"  Conv max: {c.max():.10f}")
        print(f"  Conv mean: {c.mean():.10f}")
        print(f"  Conv min: {c.min():.10f}")
        print(f"  Peak-to-mean ratio: {c.max()/c.mean():.4f}")
        print(f"  Theoretical lower bound: {2*len(best_seq)/(2*len(best_seq)-1):.8f}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
