def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_and_weights(a, T):
    """Compute LSE value and softmax weights over autocorrelation peaks."""
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    return lse_val, w, b


def lse_gradient(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    Uses FFT for O(n log n) computation.
    Returns (gradient, weights, b).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # grad[i] = 2 * sum_k w[k] * (if k-i is valid, a[k-i])
    # = 2 * (a_reversed * w) evaluated at position n-1
    # Using FFT: conv(a[::-1], w) gives cross-correlation
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w, b


def subgradient_descent(a0, time_limit, n_stages=30, T_start=500.0, T_end=50.0):
    """
    Subgradient descent with LSE smoothing and adaptive temperature.
    Uses simplex projection (a >= 0, sum(a) = 1).
    """

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Adaptive step size: start large, decay
        n_iters = max(50, min(500, int(remaining * 5)))
        base_lr = 1.0 / (np.sqrt(n) * np.sqrt(n_iters))
        
        for it in range(n_iters):

            # Projected gradient step onto simplex

            # Project onto simplex {x: x >= 0, sum(x) = 1}

            if score < best_score:

                best_a = a.copy()
    
    return best_a, best_score


def project_to_simplex(v):
    """Project v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

    max_Tb = Tb.max()
    if max_Tb < 1e-30:
        # All inactive, use uniform
        w = np.ones(len(b)) / len(b)
    else:
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (for valid indices)
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    
    return grad, b, w


def lse_subgradient_optimize(a0, time_limit, T_start=2000.0, T_end=100.0, n_stages=15):
    """
    LSE-subgradient optimization with:
    - Very high initial temperature for sharp peak tracking
    - Backtracking line search
    - Active-set focusing
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        n_iters = max(20, min(200, int(remaining * 3)))
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, b, w = lse_grad_active(a, T, active_frac=0.03)
            
            # Backtracking line search
            lr = 0.5 / (np.linalg.norm(grad, ord=np.inf) + 1e-10)
            improved = False
            for backtrack in range(10):
                a_trial = clip_normalize(a - lr * grad)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score or score < 2.0 * n * b.max():
                    a = a_trial
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    improved = True
                    break
                lr *= 0.5
            
            if not improved:
                # Try smaller step
                lr = lr * 0.1
                a_trial = clip_normalize(a - lr * grad)
                a = a_trial
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
    
    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build convolution matrix T where (T@b)_k = sum_j a_fixed[j] * b[k-j]."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. (a_fixed * b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = clip_normalize(y_new)
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = clip_normalize(a_new)
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def peak_shaving_aggressive(a0, time_limit, seed=42):
    """
    Aggressive peak shaving: for each top autocorrelation peak,
    find the pair (i,j) with i+j=k (or |i-j|=k) that contributes most,
    and move mass from the larger of the two to a position that
    doesn't contribute to any top peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        # Top peaks
        top_k = np.argsort(b)[::-1][:12]
        
        improved = False
        for k in top_k:
            if b[k] < b_max * 0.99:
                break
            
            # Find contributing pairs for autocorrelation at lag k
            # (a*a)_k = sum_j a[j] * a[k-j] for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Top contributor
            top_contrib_idx = np.argsort(contrib)[::-1][:5]
            
            for ci in top_contrib_idx:
                src = idx[ci]
                partner = k - src
                
                # Find a destination that reduces multiple peaks
                # Position dst that is far from all top peak contributing pairs
                peak_positions = set()
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_positions.update(range(js2, je2 + 1))
                
                candidates = [j for j in range(n) if j not in peak_positions and j != src]
                if not candidates:
                    candidates = [j for j in range(n) if j != src]
                
                # Pick candidate with lowest value (to minimize disruption)
                candidates.sort(key=lambda j: a[j])
                dst_pool = candidates[:max(1, min(10, len(candidates) // 4))]
                
                for dst in dst_pool[:3]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < 0:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Cubic-ish interpolation using sinc-windowed resampling."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    
    # Use simple linear interpolation (fast and effective)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_lp_solver():
    for solver_name in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def structured_nonuniform_init(n, rng):
    """
    Generate a non-uniform initialization that's biased toward reducing
    the central autocorrelation peak. Uses a 'U-shaped' profile
    (high at edges, low in center) which reduces the central autocorrelation.
    """
    choice = rng.integers(0, 5)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # U-shaped (reduces central peak)
        a = (2 * x - 1) ** 2 + 0.1
    elif choice == 1:
        # W-shaped
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x)
    elif choice == 2:
        # Two bumps at 1/3 and 2/3
        a = np.exp(-50 * (x - 1/3) ** 2) + np.exp(-50 * (x - 2/3) ** 2) + 0.05
    elif choice == 3:
        # Edge-concentrated
        a = (1 - 4 * (x - 0.5) ** 2) ** 2
    else:
        # Alternating
        a = 1.0 + 0.3 * np.cos(8 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
