from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_sparse(k, n):
    """Build sparse PSD Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] += 1.0
        else:
            Q[j, partner] += 0.5
            Q[partner, j] += 0.5
    return csc_matrix(Q)


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_qcqp_solver():
    """Find best available QCQP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.SCIP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.quad_form(x, cp.psd_wrap(np.eye(5)))), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def get_lp_solver():
    """Find best available LP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution matrix T: (T @ b)[k] = sum_j a_fixed[j] * b[k-j]
    # This is the matrix such that T @ b = conv(a_fixed, b)
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                return b_val / s_sum
        except Exception:
            continue
    return None


def nesterov_lp_refine(a0, time_limit, solver):
    """Nesterov accelerated LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    
    # Adaptive parameters
    base_alpha_rate = 0.0  # will use 2/(t+2)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            break
        
        alpha = 2.0 / (it + 2)
        
        # Nesterov update
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Stall detection and restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 5:
                # Restart with perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0
        
        if (it + 1) % 500 == 0:
            print(f"    LP iter={it+1} t={time.monotonic()-t0:.0f}s score={best_score:.10f}")
    
    return best_a, best_score


def solve_qcqp_active(a, active_indices, solver):
    """
    Solve min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0.
    Returns (solution, t_value) or (None, None).
    """
    n = len(a)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_indices:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    continue
                a_val = a_val / s_sum
                t_val = max(0.0, float(t_var.value))
                return a_val, t_val
        except Exception:
            continue
    
    return None, None


def active_set_qcqp_refine(a0, time_limit, solver):
    """
    Iterative active-set QCQP:
    1. Find active constraints (near-max convolution positions)
    2. Solve QCQP with active constraints
    3. Check if inactive constraints are violated
    4. If violated, add to active set and repeat
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    # Initial active set: top positions
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - 1e-6)
    active = np.where(b >= threshold)[0]
    
    if len(active) > 30:
        active = active[np.argsort(b[active])[-30:]]
    
    iteration = 0
    max_active_size = 50  # cap on active set size
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        # Solve QCQP
        a_new, t_val = solve_qcqp_active(a, active, solver)
        
        if a_new is None:
            break
        
        # Check solution quality
        b_new = fast_convolve(a_new)
        max_b_new = b_new.max()
        score_new = 2.0 * n * max_b_new
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        
        # Check if we need to add constraints
        # A constraint k is violated if (a*a)[k] > t_val + tol
        tol = 1e-10
        violated = np.where(b_new > t_val + tol)[0]
        
        # Remove constraints that are no longer active
        active_threshold = t_val * (1.0 - 1e-8)
        still_active = np.where(b_new >= active_threshold)[0]
        
        # Update active set
        new_active = np.unique(np.concatenate([still_active, violated]))
        
        if len(new_active) > max_active_size:
            # Keep the most violated ones
            new_active = new_active[np.argsort(b_new[new_active])[-max_active_size:]]
        
        # Check convergence: no violations and active set stable
        if len(violated) == 0:
            # Also check that the active set hasn't changed significantly
            if np.all(np.isin(active, new_active)) and np.all(np.isin(new_active, active)):
                break
            active = new_active
            if iteration % 5 == 0:
                print(f"    ActiveSet iter={iteration} t={time.monotonic()-t0:.0f}s "
                      f"score={best_score:.10f} |S|={len(active)}")
        else:
            active = new_active
            if iteration % 5 == 0:
                print(f"    ActiveSet iter={iteration} t={time.monotonic()-t0:.0f}s "
                      f"score={best_score:.10f} |S|={len(active)} violated={len(violated)}")
        
        # Safety: if active set is too large, use only the top ones
        if len(active) > max_active_size:
            b_check = fast_convolve(best_a)
            active = np.where(b_check >= b_check.max() * (1 - 1e-6))[0]
            active = active[np.argsort(b_check[active])[-max_active_size:]]
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_descent(a0, time_limit, n_steps=50000):
    """Mirror descent for fine-tuning."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.01
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Subgradient: average of gradients of active constraints
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-5:
                break
    
    return best_a, best_score

import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_sparse(k, n):
    """Build sparse PSD Q matrix such that a^T Q a = conv(a,a)[k]."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] += 1.0
        else:
            Q[j, partner] += 0.5
            Q[partner, j] += 0.5
    return csc_matrix(Q)


def evaluate_local(a):
    """Evaluate sequence score locally (fast, no security checks)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.SCS, cp.ECOS]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                return b_val / s_sum
        except Exception:
            continue
    return None

def solve_qcqp_with_active(a_init, active_indices, solver):
    """
    Solve min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0.
    Returns (solution_array, t_value) or (None, None).
    """
    n = len(a_init)
    if len(active_indices) == 0:
        return a_init.copy(), 0.0

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active_indices:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.SCS, cp.ECOS]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    continue
                a_val = a_val / s_sum
                t_val = max(0.0, float(t_var.value))
                return a_val, t_val
        except Exception:
            continue

    return None, None


def active_set_qcqp_refine(a0, time_limit, solver):

            active_new = np.unique(np.concatenate([active_new, violated]))

            active_new = active_new[np.argsort(b_new[active_new])[-max_active:]]

        # Check convergence
        max_violation = b_new.max() - t_val

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_descent(a0, time_limit):
    """Mirror descent (multiplicative weights) for fine-tuning active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.005
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 5e-5)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 20:
            active = active[np.argsort(b[active])[-20:]]

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Multiplicative update with step size scheduling
        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            eta *= 0.6
            no_improve = 0
            if eta < 1e-7:
                break

    return best_a, best_score


def subgradient_polish(a0, time_limit):
    """Projected subgradient method for additional refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    step = 0.01

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        k_max = int(np.argmax(b))

        # Subgradient of max_k f_k(a) at a
        threshold = max_b * (1.0 - 2e-5)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([k_max])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        # Center the gradient (for the simplex constraint)
        g = g - g.mean()

        a_new = a - step * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        step *= 0.999

    return best_a, best_score
