Now let me think about the spectral shaping approach more carefully.

**Spectral Shaping:**
The circular autocorrelation of a (periodically extended) is `IDFT(|A|^2)` where `A = DFT(a)`.
The linear autocorrelation is approximately the circular one minus boundary effects.

If we want to flatten the linear autocorrelation, we can:
1. Compute the current circular autocorrelation: `c_circ = IDFT(|A|^2)`
2. The "error" is `c_circ - target` where target is flat (or some desired shape)
3. Adjust the power spectrum: `|A_new|^2 = |A|^2 - α * FFT(c_circ - target)`
4. Clip to non-negative: `|A_new|^2 = max(|A_new|^2, 0)`
5. Keep the phase: `A_new = |A_new| * exp(i*angle(A))`
6. `a_new = IDFT(A_new)`
7. Repair non-negativity

This is a variant of the Gerchberg-Saxton / error reduction algorithm.

However, there's a subtlety: we're working with the LINEAR autocorrelation, not the circular one. The relationship between them involves the "boundary terms." For a sequence of length n, the linear autocorrelation at lag k (for 0 <= k < n) is:
`c_lin[k] = sum_{i=0}^{n-1-k} a_i * a_{i+k}`

The circular autocorrelation at lag k is:
`c_circ[k] = sum_{i=0}^{n-1} a_i * a_{(i+k) mod n} = c_lin[k] + sum_{i=0}^{k-1} a_i * a_{n-1-(k-i)}`

The second term is the "wrap-around" contribution. For a sequence that tapers to zero at the edges, this term is small.

So if we ensure the sequence tapers at the edges, the circular and linear autocorrelations are approximately the same, and we can use the spectral method.

def fft_convolve_auto(a):

    return irfft(A * A, fft_size)[:size]

    grad[m] = sum_{j=0}^{n-1} w[m+j] * a[j]
    where w = softmax(c/T), c = conv(a,a), length(w) = 2n-1.

    # grad[m] = sum_j w[m+j] * a[j]

    w_rev = w[::-1]

    grad = conv_result[n-1:2*n-1][::-1]
    return grad


def lse_objective_and_gradient(z, T):
    """LSE objective and gradient in log-parameterization.
    z: log-values, a = softmax(z)
    Returns (obj, grad_z)
    """
    n = len(z)
    # Compute a = softmax(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Compute convolution
    c = fft_convolve_auto(a)
    
    # LSE
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    
    # Weights
    w = exp_c / exp_c.sum()
    
    # Gradient w.r.t. a
    grad_a = 2.0 * fft_gradient(a, w)
    
    # Project gradient to tangent space of simplex (for log-param)
    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

            P_new = (1 - alpha) * P + alpha * P_clipped
            # Also reduce the DC component slightly to maintain sum
            P_new[0] = P[0]  # Keep DC (preserves sum of a)
            
            # New a
            A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
        
        if it % 100 == 99:
            lr *= 0.7
    
    return best_a, best_score


def lbgfs_run(a0, time_limit, T_stages):
    """L-BFGS optimization with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        def func(z_in):
            obj, _ = lse_objective_and_gradient(z_in, T)
            return obj
        
        def jac(z_in):
            _, g = lse_objective_and_gradient(z_in, T)
            return g
        
        maxiter = min(5000, max(50, int(remaining * 20)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': maxiter * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def diverse_init(n, rng):
    """Generate diverse initial sequences of length n."""
    candidates = []
    x = np.linspace(0, 1, n)
    
    # Various bump shapes
    for sigma in [0.1, 0.15, 0.2, 0.3]:
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
        candidates.append(normalize(a))
    
    # Asymmetric bumps
    for center in [0.3, 0.4, 0.6, 0.7]:
        a = np.exp(-((x - center) ** 2) / 0.05)
        candidates.append(normalize(a))
    
    # Power law (front-loaded)
    a = (1 - x) ** 2
    candidates.append(normalize(a))
    
    # Power law (back-loaded)
    a = x ** 2
    candidates.append(normalize(a))
    
    # Sine window
    a = np.sin(np.pi * x)
    candidates.append(normalize(a))
    
    # Cosine window
    a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    a = np.maximum(a, 0)
    candidates.append(normalize(a))
    
    # Two bumps (asymmetric)
    a = np.exp(-((x - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((x - 0.7) ** 2) / 0.03)
    candidates.append(normalize(a))
    
    # Random smooth
    coeffs = rng.normal(0, 1, 8)
    a = np.zeros(n)
    for i, cf in enumerate(coeffs):
        a += cf * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 0.5, 0)
    candidates.append(normalize(a))
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Triangle
    a = 1 - np.abs(2 * x - 1)
    candidates.append(normalize(a))
    
    # Exp decaying from left
    a = np.exp(-3 * x)
    candidates.append(normalize(a))
    
    # Exp decaying from right
    a = np.exp(-3 * (1 - x))
    candidates.append(normalize(a))
    
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturbation(a, rng, method='gaussian', strength=0.03):
    """Apply a perturbation to a sequence."""
    n = len(a)
    if method == 'gaussian':
        a_new = a + rng.normal(0, strength, n)
    elif method == 'low_freq':
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        a_new = a + strength * np.cos(freq * x + phase)
    elif method == 'block_swap':
        a_new = a.copy()
        bs = rng.integers(5, min(50, n // 4))
        i = rng.integers(0, n - 2 * bs)
        j = i + bs
        a_new[i:i+bs], a_new[j:j+bs] = a_new[j:j+bs].copy(), a_new[i:i+bs].copy()
    elif method == 'scaling':
        # Scale a random subset
        a_new = a.copy()
        idx = rng.integers(0, n, size=n // 3)
        factor = rng.uniform(0.5, 1.5)
        a_new[idx] *= factor
    else:
        a_new = a + rng.normal(0, strength, n)
    
    return normalize(a_new)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LSE-smoothed L-BFGS, peak multiplicative updates, spectral flattening,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize(a).copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # --- Load parent sequence if available ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = compute_score(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    # --- If no parent, generate diverse inits ---
    if parent is None:
        n_init = 1000
        candidates = diverse_init(n_init, rng)
        scores = [compute_score(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Generated init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")
    
    # === Phase 1: L-BFGS with LSE (coarse to fine) ===
    print(f"\n=== Phase 1: L-BFGS LSE ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [100, 50, 30, 20, 10, 7, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
        a_lb = lbgfs_run(best_seq, t_alloc, T_stages)
        try_update(a_lb, "lbgfs_coarse")
    
    # === Phase 2: Peak multiplicative update ===
    print(f"\n=== Phase 2: Peak Multiplicative ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mw, _ = peak_multiplicative_update(best_seq, n_steps=50000, eta0=0.1, time_limit=t_alloc)
        try_update(a_mw, "peak_mult")
    
    # === Phase 3: L-BFGS fine polish ===
    print(f"\n=== Phase 3: L-BFGS Fine ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_fine = [5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005]
        a_fine = lbgfs_run(best_seq, t_alloc, T_fine)
        try_update(a_fine, "lbgfs_fine")
    
    # === Phase 4: Spectral flattening ===
    print(f"\n=== Phase 4: Spectral Flatten ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
