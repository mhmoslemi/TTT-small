import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:

    return exp_z / s

    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()

    scaled_max = scaled.max()
    exp_terms = np.exp(scaled - scaled_max)
    log_sum = np.log(exp_terms.sum() + 1e-300)
    lse_val = max_b + T * (scaled_max + log_sum)
    obj = 2.0 * n * lse_val

    S = np.dot(grad_a, a)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    consecutive_fail = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Use threshold-based active set for efficiency
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        # Multiplicative update
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.5)
            consecutive_fail = 0
        else:
            eta *= 0.85
            consecutive_fail += 1
            if consecutive_fail > 200:
                break
        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")
    return best_a, best_score


def nesterov_mirror_descent(a0, time_limit, n_steps=200000, lr0=0.01):
    """Nesterov-accelerated mirror descent in log-domain with LSE smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    y = z.copy()  # Nesterov lookahead
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s_b = best_a.sum()
    if s_b < 1e-15:
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s_b
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    T = 1.0
    T_end = 0.01
    T_decay = 0.9995

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        # Decay temperature
        T = max(T_end, T * T_decay)
        # Nesterov step
        obj_y, grad_y = lse_obj_grad(y, n, T)
        z_new = y - (lr0 / (step + 1)) * grad_y
        # Momentum update
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * (step + 1) ** 2)) / (2.0 * (step + 1) + 2.0)
        y = (1.0 - alpha) * z + alpha * z_new
        z = y.copy()
        a = softmax_from_z(z)
        a_nn = np.maximum(a, 0)
        s_nn = a_nn.sum()
        if s_nn < 1e-15:
            continue
        a_nn = a_nn / s_nn
        score = 2.0 * n * fast_convolve(a_nn).max()
        if score < best_score:
            best_score = score
            best_a = a_nn.copy()
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov {elapsed:.0f}s: score={best_score:.8f} T={T:.4f}")
    return best_a, best_score


def adam_lse_optimize(a0, time_limit, lr=0.005, n_steps=500000):
    """Adam optimizer on LSE-smoothed objective in log-domain."""
    n = len(a0)
    z = z_from_a(a0).copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s_b = best_a.sum()
    if s_b < 1e-15:
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s_b
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    T = 2.0
    T_end = 0.005
    T_decay = 0.9997
    patience = 0
    patience_limit = 50000

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        T = max(T_end, T * T_decay)
        obj, grad = lse_obj_grad(z, n, T)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = softmax_from_z(z)
        a_nn = np.maximum(a, 0)
        s_nn = a_nn.sum()
        if s_nn < 1e-15:
            continue
        a_nn = a_nn / s_nn
        score = 2.0 * n * fast_convolve(a_nn).max()
        if score < best_score:
            best_score = score
            best_a = a_nn.copy()
            patience = 0
        else:
            patience += 1
        if patience > patience_limit:
            break
        if (step + 1) % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam {elapsed:.0f}s: score={best_score:.8f} T={T:.4f}")
    return best_a, best_score


def spectral_flatten(a, n_iters=200, strength=0.1):
    """Iterative spectral amplitude flattening to reduce auto-correlation peak."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iters):
        # Compute FFT
        N = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, N)
        # Amplitude spectrum
        mag = np.abs(A)
        angle = np.angle(A)
        # Target: flatten the amplitude spectrum (reduce peaks)
        # Use a "soft clamp" approach
        mag_mean = mag.mean()
        # Reduce bins above a threshold
        threshold = mag_mean * 1.2
        excess = np.maximum(mag - threshold, 0)
        # Redistribute to bins below threshold
        deficit = np.maximum(threshold - mag, 0)
        if deficit.sum() > 1e-15:
            # Scale excess down and redistribute
            scale_down = 1.0 - strength * (excess / (excess.sum() + 1e-15))
            mag_new = mag * scale_down
            # Add to deficit bins
            add_amount = strength * excess.sum()
            mag_new += (deficit / (deficit.sum() + 1e-15)) * add_amount
        else:
            mag_new = mag
        # Reconstruct
        A_new = mag_new * np.exp(1j * angle)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
    return best_a


def lbgfs_refine_fine(a0, time_limit, T_stages=None, maxiter_per_stage=1000):
    """L-BFGS with LogSumExp smoothing and ultra-fine temperature annealing."""
    n = len(a0)
    if T_stages is None:
        T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15,
                    0.1, 0.07, 0.05, 0.035, 0.025, 0.02, 0.015, 0.01, 0.008,
                    0.006, 0.005, 0.003, 0.002, 0.0015, 0.001]
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for i, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        func_state = LSEFuncState(n, stage_T, deadline)
        jac_state = LSEJacState(n, stage_T, deadline)

        def func_wrapper(x, fs=func_state):
            return fs.call(x)

        def jac_wrapper(x, js=jac_state):
            return js.call(x)

        try:
            res = minimize(
                fun=func_wrapper, x0=z, jac=jac_wrapper, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
        if (i + 1) % 5 == 0:
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {i+1}/{len(T_stages)} T={stage_T:.4f} ({elapsed:.0f}s): score={sc:.8f}")
    return softmax_from_z(z)


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = 0.3 + 0.2 * np.cos(2 * np.pi * 7 * x) + 0.1 * np.cos(2 * np.pi * 13 * x)
        a = np.maximum(a, 0)
    else:
        a = 0.25 + 0.15 * np.cos(2 * np.pi * 4 * x + 0.3) + 0.1 * np.cos(2 * np.pi * 11 * x + 1.7)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def fine_mw_polish(a0, time_limit, eta0=0.01, n_steps=1000000):
    """Very fine multiplicative weights for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Very tight threshold for active set
        threshold = max_b * (1.0 - 5e-5)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.001, 0.3)
            stall = 0
        else:
            eta *= 0.92
            stall += 1
            if stall > 10000:
                break
    return best_a, best_score

            a_combo = lbgfs_refine_fine(a_sp, min(rem * 0.6, 60.0), T_stages=T_c, maxiter_per_stage=1000)
            try_update(a_combo, "spectral_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
