import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:2]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_inner(a_fixed, solver):
    """Solve the inner LP: min M s.t. conv(a, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(b_val, 0)

    if s < 1e-15:

    return b_val / s

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_solver():
    """Find a working solver for both LP and QP."""
    for solver_name in [cp.CLARABEL, cp.MOSEK, cp.GUROBI, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.CLARABEL


def solve_lp_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.MOSEK, cp.GUROBI]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                return b_val / s_sum
        except Exception:
            continue
    return None


def solve_qcqp_active(a_init, active_set, solver):
    """Solve QCQP: min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0."""
    n = len(a_init)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    for k in active_set:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solvers = [cp.MOSEK, solver, cp.CLARABEL, cp.GUROBI, cp.ECOS, cp.SCS]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                a_val = np.maximum(a_val, 0)

                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)

    return float(2.0 * n * b.max() / (s * s))

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    lp_times = []
    it_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new = solve_lp_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        it_count += 1
        alpha = 2.0 / (it_count + 2)

        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()

            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it_count + 1) * a + alpha * (it_count + 2) * b_new) / (it_count + 1 + alpha * (it_count + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                # Restart with perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
            if (it + 1) % 1000 == 0:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        # Stop if LP is too slow
        if len(lp_times) > 50 and np.mean(lp_times[-50:]) > 3.0:
            print(f"    LP too slow ({np.mean(lp_times[-50:]):.2f}s/iter), stopping at iter {it+1}")
            break

        # Early stop if converged
        if stall_count >= 12 and (it + 1) > 2000:
            print(f"    NLP converged at iter {it+1}, score={best_score:.10f}")
            break

    return best_a, best_score


def iterative_qcqp_refine(a0, time_limit, solver):
    """
    Iterative active-set QCQP with KKT verification.
    Repeatedly:
    1. Identify active set (near-max convolution positions)
    2. Solve QCQP on active set
    3. Check if solution violates any inactive constraint
    4. If yes, add violated constraint to active set
    5. Repeat until KKT satisfied
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for round in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        # Identify active set
        b = fast_convolve(a)
        max_b = b.max()

        # Try multiple thresholds to find the "natural" active set
        best_round_a = None
        best_round_score = float('inf')

        for rel_tol in [1e-3, 5e-4, 1e-4, 5e-5, 1e-5, 5e-6, 1e-6]:
            if time.monotonic() - t0 > time_limit:
                break

            threshold = max_b * (1.0 - rel_tol)
            active = np.where(b >= threshold)[0]

            if len(active) == 0:
                active = np.array([np.argmax(b)])

            # Limit active set size for tractability
            if len(active) > 40:
                # Keep the largest ones
                active = active[np.argsort(b[active])[-40:]]

            a_sol, t_val = solve_qcqp_active(a, active, solver)
            if a_sol is None:
                continue

            score = 2.0 * n * fast_convolve(a_sol).max()
            if score < best_round_score:
                best_round_score = score
                best_round_a = a_sol.copy()

        if best_round_a is None:
            break

        # Check KKT: see if any inactive constraint is violated
        b_new = fast_convolve(best_round_a)
        max_new = b_new.max()
        t_active = max_new

        # Check maximum violation of inactive constraints
        violation = max_new * 0  # all are <= max by definition
        # The real check: is the solution optimal for the FULL problem?
        # At KKT optimum, for the max to be truly minimized,
        # we need the gradient to be balanced.
        # A practical check: solve with a slightly tighter active set
        # and see if the score improves.

        if best_round_score < best_score - 1e-12:
            best_score = best_round_score
            best_a = best_round_a.copy()
            a = best_round_a.copy()
            print(f"    QCQP round {round}: score={best_score:.10f}, active_set_size={len(active)}")
        else:
            # No improvement, try a different approach
            # Solve single-QP on argmax
            k_max = int(np.argmax(b_new))
            a_single = solve_qcqp_single(a, k_max, solver)
            if a_single is not None:
                sc = 2.0 * n * fast_convolve(a_single).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_single.copy()
                    a = a_single.copy()
                    print(f"    QCQP round {round} (single): score={best_score:.10f}")
            break

    return best_a, best_score


def qcqp_with_growing_active(a0, time_limit, solver):
    """
    Start with small active set, solve QCQP, then grow active set
    by adding violated constraints. Iterate until KKT satisfied.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Start with just the argmax position
    b = fast_convolve(a)
    k_max = int(np.argmax(b))
    active = [k_max]

    for it in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        a_sol, t_val = solve_qcqp_active(a, active, solver)
        if a_sol is None:
            break

        score = 2.0 * n * fast_convolve(a_sol).max()
        if score < best_score:
            best_score = score
            best_a = a_sol.copy()
            a = a_sol.copy()
            print(f"    GrowingQ it={it}: score={score:.10f}, |active|={len(active)}")

        # Check for violated constraints
        b_new = fast_convolve(a_sol)
        max_new = b_new.max()
        # Find constraints that are close to the max (within 1e-8)
        # These should be in the active set at KKT optimum
        close_to_max = np.where(b_new >= max_new * (1 - 1e-8))[0]

        # Also find any constraint that's violated relative to t_val
        if t_val is not None and t_val > 0:
            violated = np.where(b_new > t_val * (1 + 1e-6))[0]
            if len(violated) > 0:
                # Add most violated
                idx = violated[np.argmax(b_new[violated])]
                if idx not in active:
                    active.append(int(idx))

        # Update active set to include all near-max positions
        for k in close_to_max:
            if k not in active:
                active.append(int(k))

        # Limit active set size
        if len(active) > 50:
            # Keep only the most important ones
            b_vals = [b_new[k] for k in active]
            active = [active[i] for i in np.argsort(b_vals)[-50:]]

        # Check convergence: if active set hasn't changed and score is stable
        if len(close_to_max) == len(active) and set(int(k) for k in close_to_max) == set(active):
            # KKT might be satisfied
            # Do one more verification
            a_verify, _ = solve_qcqp_active(a, list(set(active + list(close_to_max))), solver)
            if a_verify is not None:
                sc_v = 2.0 * n * fast_convolve(a_verify).max()
                if sc_v < best_score:
                    best_score = sc_v
                    best_a = a_verify.copy()
            break

    return best_a, best_score


def gradient_descent_polish(a0, time_limit, n_steps=100000, lr_init=0.1):
    """Constrained gradient descent on the smoothed max objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Subgradient of max: gradient at argmax positions
        active = np.where(b >= max_b * (1 - 1e-4))[0]
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        g *= 2  # gradient of a^T Q a is 2 Q a

        # Projected gradient step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-8:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_solver()
    print(f"Using solver: {solver}")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        # Cosine initialization
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000)
        update_best(a_lp, "lp_phase1")

    # === Phase 2: Active-set QCQP refinement ===
    print(f"\n=== Phase 2: QCQP Refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.25, 400.0)
    if t_qcqp > 20:
        print(f"  Budget: {t_qcqp:.0f}s")

        # Strategy A: Iterative QCQP with growing active set
        a_qa, _ = qcqp_with_growing_active(best_seq, t_qcqp * 0.5, solver)
        update_best(a_qa, "qcqp_growing")

        # Strategy B: Multi-threshold QCQP
        rem = deadline - time.monotonic()
        if rem > 20:
            a_qb, _ = iterative_qcqp_refine(best_seq, rem * 0.5, solver)
            update_best(a_qb, "qcqp_multi")

    # === Phase 3: LP round 2 from polished solution ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=200000)
        update_best(a_lp2, "lp_phase3")

    # === Phase 4: QCQP again after second LP ===
    print(f"\n=== Phase 4: QCQP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_qc, _ = qcqp_with_growing_active(best_seq, min(rem * 0.5, 150.0), solver)
        update_best(a_qc, "qcqp_round2")

    # === Phase 5: Gradient descent polish ===
    print(f"\n=== Phase 5: Gradient Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_gd, _ = gradient_descent_polish(best_seq, min(rem * 0.3, 60.0), n_steps=50000, lr_init=0.05)
        update_best(a_gd, "grad_polish")

    # === Phase 6: Try n=1199 ===
    print(f"\n=== Phase 6: Try n=1199 (current best score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_1199 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_1199:.0f}s")
        init_1199 = interpolate_sequence(best_seq, 1199)
        a_1199, _ = nesterov_lp_refine(init_1199, t_1199 * 0.6, solver, n_max_iters=200000)
        update_best(a_1199, "lp_n1199")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1199 and rem > 30:
            a_q, _ = qcqp_with_growing_active(best_seq, min(rem * 0.5, 80.0), solver)
            update_best(a_q, "qcqp_n1199")

    # === Phase 7: Try n=1201 ===
    print(f"\n=== Phase 7: Try n=1201 (current best score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_1201 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_1201:.0f}s")
        # Interpolate from n=1200 solution (use best_seq before n=1199 change if needed)
        # Actually, just interpolate from current best
        init_1201 = interpolate_sequence(best_seq, 1201)
        a_1201, _ = nesterov_lp_refine(init_1201, t_1201 * 0.6, solver, n_max_iters=200000)
        update_best(a_1201, "lp_n1201")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1201 and rem > 30:
            a_q, _ = qcqp_with_growing_active(best_seq, min(rem * 0.5, 80.0), solver)
            update_best(a_q, "qcqp_n1201")

    # === Phase 8: Final aggressive QCQP ===
    print(f"\n=== Phase 8: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Try with very tight active set
        b = fast_convolve(best_seq)
        max_b = b.max()
        k_max = int(np.argmax(b))

        # Single constraint QCQP
        a_single = solve_qcqp_single(best_seq, k_max, solver)
        update_best(a_single, "final_single_qp")

        # Two constraint QCQP (argmax and its symmetric partner)
        k_sym = 2 * len(best_seq) - 2 - k_max
        if k_sym != k_max:
            a_two, _ = solve_qcqp_active(best_seq, [k_max, k_sym], solver)
            update_best(a_two, "final_two_qp")

        # Growing active set
        a_grow, _ = qcqp_with_growing_active(best_seq, min(rem * 0.5, 100.0), solver)
        update_best(a_grow, "final_growing")

    # === Phase 9: Final micro-polish ===
    print(f"\n=== Phase 9: Micro-polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very short LP
        a_micro, _ = nesterov_lp_refine(best_seq, min(rem * 0.3, 30.0), solver, n_max_iters=10000)
        update_best(a_micro, "micro_lp")

        # Single QP on argmax
        if best_seq is not None:
            b = fast_convolve(best_seq)
            k_max = int(np.argmax(b))
            a_s = solve_qcqp_single(best_seq, k_max, solver)
            update_best(a_s, "micro_qp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
