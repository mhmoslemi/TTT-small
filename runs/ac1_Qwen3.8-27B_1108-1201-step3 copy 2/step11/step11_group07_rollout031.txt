"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding.
Key innovation: at each step, solve an LP to find the best direction g
that maximizes sum(g) subject to (f*g)[k] <= M for all k and g >= 0.
This gives a principled descent direction guaranteed (to first order)
to not increase the auto-convolution peak. Combined with FFT-based
evaluation, adaptive line search, gradient refinement via LSE smoothing,
and spectral shaping for fine-tuning.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Evaluate score locally using FFT."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(f):
    """Build matrix T (2n-1 x n) such that T @ g = f * g (linear convolution)."""
    n = len(f)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T[k, js:je + 1] = f[k - je:k - js + 1][::-1]
    return T


def lp_direction(f, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M*relax, g >= 0.
    Returns g or None if infeasible."""
    n = len(f)
    T = build_conv_matrix(f)
    c = -np.ones(n)
    b = M * relax * np.ones(2 * n - 1)
    try:
        res = linprog(c, A_ub=T, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lse_gradient(a, T_smooth):
    """Compute LSE-smoothed gradient of the objective w.r.t. a (sum=1 parameterization).
    Returns gradient in the log-space (for softmax parameterization)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T_smooth)
    w = exp_c / exp_c.sum()

    # Gradient of 2n*LSE(c, T) w.r.t. a[i]
    # d(c[k])/da[i] = 2*a[k-i] (if valid)
    # grad[i] = 2n * sum_k w[k] * 2*a[k-i] / T * ... 
    # Actually: d(LSE)/da[i] = sum_k w[k] * d(c[k])/da[i] = 2 * sum_k w[k]*a[k-i]
    # We need the cross-correlation of w and a
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    grad = 2.0 * R[:n]

    # Transform to log-space gradient (for softmax parameterization)
    S = np.dot(grad, a)
    grad_log = a * (grad - S)
    return grad_log


def gradient_descent_logspace(z, n, T_stages, time_limit, base_lr=0.1):
    """Gradient descent in log-space with LSE smoothing and temperature annealing."""
    t0 = time.monotonic()
    z = z.copy()
    
    for T_s in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = min(2000, max(50, int(remaining * 20)))
        lr = base_lr
        
        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            
            grad = lse_gradient(a, T_s)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            z = np.log(a_new)
            
            if step % 200 == 0 and step > 0:
                lr *= 0.85
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def spectral_flatness_step(a, lr):
    """One step of spectral flatness optimization."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    r = (P - P_mean) / (P_mean + 1e-15)
    
    H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
    valid = P > 1e-12
    H[valid] = -r[valid] * P_mean * A[valid] / (P[valid] + 1e-10)
    H[0] = 0.0
    h = irfft(H, fft_size)[:n]
    h -= h.mean()
    h_norm = np.linalg.norm(h)
    if h_norm < 1e-15:
        return a
    h = h / h_norm
    a_new = a + lr * h
    return normalize(a_new)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)

            a_spec = spectral_flatness_step(a_spec, lr)
            sc_spec = eval_local(a_spec)
            if sc_spec < best_score:
                best_score = sc_spec
                best_a = a_spec.copy()
            if lr < 1e-5:
                break
            lr *= 0.95
        print(f"  After Phase 4: {best_score:.8f}")
    
    # Phase 5: Final gradient polish
    if remaining_time() > 10:
        t_alloc = min(remaining_time() * 0.15, 60.0)
        print(f"  Phase 5 (Polish): budget={t_alloc:.0f}s")
        T_final = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_polish = gradient_refine(best_a, t_alloc, T_final)
        sc_polish = eval_local(a_polish)
        if sc_polish < best_score:
            best_score = sc_polish
            best_a = a_polish.copy()
        print(f"  After Phase 5: {best_score:.8f}")
    
    return best_a, best_score


def lp_optimize_loop(a0, time_limit, relax=1.0, max_iter=2000, gradual_relax=True):
    """Main LP-based optimization loop with adaptive relaxation."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()
    consecutive_no_improve = 0
    relax_current = relax
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Compute current peak
        c = fast_convolve(a)
        M = c.max()
        
        # Solve LP for direction
        g = lp_direction(a, M, relax=relax_current)
        if g is None:
            relax_current = min(3.0, relax_current * 1.1)
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            relax_current = min(3.0, relax_current * 1.05)
            continue
        
        g = g / g_sum  # normalize
        
        # Line search
        best_t = 0.0
        best_blend_score = best_score
        for t in [0.3, 0.1, 0.2, 0.05, 0.5, 0.02, 0.15]:
            a_blend = (1 - t) * a + t * g
            a_blend = np.maximum(a_blend, 0)
            s = a_blend.sum()
            if s < 1e-10:
                continue
            a_blend = a_blend / s
            sc = eval_local(a_blend)
            if sc < best_blend_score:
                best_blend_score = sc
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = eval_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                consecutive_no_improve = 0
                # Slightly reduce relax if we're doing well
                if relax_current > 1.0:
                    relax_current = max(1.0, relax_current * 0.98)
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve >= 10:
            relax_current = min(3.0, relax_current * 1.15)
            consecutive_no_improve = 0
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: score={best_score:.8f}, relax={relax_current:.3f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Core innovation: LP-based direction finding that solves
    max sum(g) s.t. (f*g)[k] <= M, g>=0 at each step,
    giving principled descent directions for peak reduction.
    Combined with gradient refinement and spectral shaping.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    
    best_seq = None
    best_score = float('inf')
    
    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    # Determine working length
    if parent is not None:
        n = len(parent)
        a_start = parent.copy()
    else:
        n = 1200
        candidates = diverse_init(n, rng)
        scores = [eval_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        a_start = candidates[best_idx].copy()
        best_seq = a_start.copy()
        best_score = scores[best_idx]
        print(f"Generated init: n={n}, score={best_score:.8f}")
    
    print(f"Starting optimization: n={n}, score={best_score:.8f}, budget={budget_s-10:.0f}s")
    
    # === Main optimization loop with multiple strategies ===
    rem = deadline - time.monotonic()
    
    # Strategy 1: LP-based optimization from parent
    if rem > 120:
        t_alloc = min(rem * 0.35, 450.0)
        print(f"\n=== Strategy 1: LP Optimization (budget={t_alloc:.0f}s) ===")
        a_lp, sc_lp = lp_optimize_loop(a_start, t_alloc, relax=1.0, max_iter=5000)
        local_update(a_lp, "LP_main")
        a_work = a_lp if a_lp is not None else a_start
    else:
        a_work = a_start.copy()
    
    rem = deadline - time.monotonic()
    
    # Strategy 2: Gradient refinement
    if rem > 60:
        t_alloc = min(rem * 0.2, 200.0)
        print(f"\n=== Strategy 2: Gradient Refinement (budget={t_alloc:.0f}s) ===")
        T_stages = [5, 3, 2, 1, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01]
        a_grad = gradient_refine(a_work, t_alloc, T_stages)
        local_update(a_grad, "gradient_1")
        if eval_local(a_grad) < eval_local(a_work):
            a_work = a_grad
    
    rem = deadline - time.monotonic()
    
    # Strategy 3: Second LP round
    if rem > 60:
        t_alloc = min(rem * 0.3, 350.0)
        print(f"\n=== Strategy 3: LP Round 2 (budget={t_alloc:.0f}s) ===")
        a_lp2, sc_lp2 = lp_optimize_loop(a_work, t_alloc, relax=1.0, max_iter=5000)
        local_update(a_lp2, "LP_2")
        if sc_lp2 < eval_local(a_work):
            a_work = a_lp2
    
    rem = deadline - time.monotonic()
    
    # Strategy 4: Spectral shaping
    if rem > 30:
        t_alloc = min(rem * 0.15, 120.0)
        print(f"\n=== Strategy 4: Spectral Shaping (budget={t_alloc:.0f}s) ===")
        a_spec = a_work.copy()
        lr = 0.005
        t_spec_start = time.monotonic()
        it_spec = 0
        while time.monotonic() - t_spec_start < t_alloc:
            a_spec = spectral_flatness_step(a_spec, lr)
            sc_spec = eval_local(a_spec)
            if sc_spec < best_score:
                best_score = sc_spec
                best_seq = a_spec.copy()
                print(f"    NEW BEST [spectral {it_spec}]: {sc_spec:.8f}")
            it_spec += 1
            if it_spec % 50 == 0:
                lr *= 0.8
            if lr < 1e-6:
                break
        a_work = a_spec
    
    rem = deadline - time.monotonic()
    
    # Strategy 5: LP from different starting points (diversification)
    if rem > 90:
        t_alloc = min(rem * 0.25, 200.0)
        print(f"\n=== Strategy 5: Diversified LP (budget={t_alloc:.0f}s) ===")
        # Create diverse perturbations of current best
        best_a = normalize(best_seq.copy())
        perturbations = []
        
        # Gaussian perturbation
        noise = rng.normal(0, 0.03, len(best_a))
        perturbations.append(normalize(best_a + noise))
        
        # Low-frequency perturbation
        x = np.linspace(0, 2 * np.pi, len(best_a))
        noise = 0.05 * np.cos(2 * x + rng.uniform(0, 2 * np.pi))
        perturbations.append(normalize(best_a + noise))
        
        # Edge emphasis
        a_edge = best_a.copy()
        edge_w = rng.uniform(0.02, 0.05)
        a_edge[:30] += edge_w
        a_edge[30:] -= edge_w * 30 / (len(a_edge) - 30)
        perturbations.append(normalize(a_edge))
        
        a_edge2 = best_a.copy()
        a_edge2[-30:] += edge_w
        a_edge2[:-30] -= edge_w * 30 / (len(a_edge2) - 30)
        perturbations.append(normalize(a_edge2))
        
        sub_budget = t_alloc / len(perturbations)
        for pi, pert in enumerate(perturbations):
            if time.monotonic() > deadline:
                break
            t_sub = min(sub_budget, deadline - time.monotonic())
            if t_sub < 10:
                break
            print(f"    Branch {pi+1}/{len(perturbations)}: score={eval_local(pert):.8f}")
            a_branch, sc_branch = lp_optimize_loop(pert, t_sub, relax=1.0, max_iter=2000)
            local_update(a_branch, f"diverse_{pi}")
    
    rem = deadline - time.monotonic()
    
    # Strategy 6: Final LP polish
    if rem > 30:
        t_alloc = min(rem * 0.4, 120.0)
        print(f"\n=== Strategy 6: Final LP Polish (budget={t_alloc:.0f}s) ===")
        a_fin, sc_fin = lp_optimize_loop(best_seq.copy(), t_alloc, relax=1.0, max_iter=3000)
        local_update(a_fin, "final_lp")
    
    rem = deadline - time.monotonic()
    
    # Strategy 7: Final gradient polish
    if rem > 15:
        t_alloc = min(rem * 0.5, 40.0)
        print(f"\n=== Strategy 7: Final Gradient Polish (budget={t_alloc:.0f}s) ===")
        T_final = [0.3, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_polish = gradient_refine(best_seq.copy(), t_alloc, T_final)
        local_update(a_polish, "final_grad")
    
    # Final verification
    if best_seq is not None:
        fs = eval_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    if best_seq is None:
        # Fallback: return uniform
        best_seq = np.ones(100) / 100
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
