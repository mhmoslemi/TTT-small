def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

        return None, None, T
    b_val = b_val / s
    return b_val, float(M_var.value), T


def fichtner_preoptimize(a0, time_limit, n_inner=10, max_iters=2000000):
    """Fichtner multiplicative-weight pre-optimization.
    Uses fixed-a LP best-response approximation via multiplicative weights.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    T = build_conv_matrix(a)
    y = np.ones(2 * n - 1) / (2 * n - 1)
    b = a.copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Fichtner inner iterations to approximate LP best response for fixed a
        for _ in range(n_inner):
            Ty = T.T @ y
            Ty = np.maximum(Ty, 1e-15)
            b = 1.0 / Ty
            sb = b.sum()
            if sb < 1e-30:
                b = np.ones(n) / n
            else:
                b = b / sb

            Tb = T @ b
            y = np.maximum(Tb, 0)
            sy = y.sum()
            if sy < 1e-30:
                y = np.ones(2 * n - 1) / (2 * n - 1)
            else:
                y = y / sy

        # Nesterov-style update: update a towards b
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b.copy()
        else:
            a_new = a_new / sa

        y_mom = a_new + momentum * (a_new - a_old)
        y_mom = np.maximum(y_mom, 0)
        sy2 = y_mom.sum()
        if sy2 < 1e-15:
            y_mom = a_new.copy()
        else:
            y_mom = y_mom / sy2

        a = a_new

        # Rebuild T with new a (every iteration is fine for n~1200)
        T = build_conv_matrix(a)

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y_mom)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y_mom.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    Fichtner iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov accelerated LP best-response optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    T = build_conv_matrix(a)
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    lp_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp, T = solve_lp_best_response(y, T=T, solver=solver)
        lp_count += 1
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new
        T = build_conv_matrix(y)

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s, {lp_count} LPs): score={best_score:.8f} M={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=12):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 40

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 4 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f}: score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)


def frequency_init(n, rng):
    """Initialize using frequency domain flatness criterion."""
    choice = rng.integers(0, 5)
    if choice == 0:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 1:
        # Flat top (inverse FFT of flat magnitude)
        A = np.ones(n)
        phase = 2 * np.pi * rng.random(n)
        a = np.abs(np.real(np.fft.ifft(A * np.exp(1j * phase))))
    elif choice == 2:
        # Gaussian in frequency -> Gaussian in time
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        # Optimal window (Kaiser-like)
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
        a = np.abs(a)
    else:
        # Cosine bell
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    else:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def optimize_at_length(a_init, budget, solver, label=""):
    """Run the full optimization pipeline at a given length."""
    n = len(a_init)
    t0 = time.monotonic()
    a = a_init.copy()
    sc = score_normalized(a)
    print(f"  [{label}] n={n}, initial score={sc:.8f}, budget={budget:.0f}s")

    # Phase A: Fichtner pre-optimization (40% of budget)
    t_fichtner = 0.40 * budget
    if t_fichtner > 5:
        a_f, _ = fichtner_preoptimize(a, t_fichtner, n_inner=5)
        sc_f = score_normalized(a_f)
        if sc_f < sc:
            a = a_f
            sc = sc_f
            print(f"  [{label}] After Fichtner: {sc:.8f} ({time.monotonic()-t0:.0f}s)")

    # Phase B: Nesterov LP (40% of budget)
    remaining = budget - (time.monotonic() - t0)
    t_lp = 0.40 * budget
    if t_lp > 5 and remaining > 5:
        t_lp = min(t_lp, remaining * 0.7)
        a_lp, _ = nesterov_lp(a, t_lp, solver=solver)
        sc_lp = score_normalized(a_lp)
        if sc_lp < sc:
            a = a_lp
            sc = sc_lp
            print(f"  [{label}] After NesterovLP: {sc:.8f} ({time.monotonic()-t0:.0f}s)")

    # Phase C: L-BFGS LogSumExp (20% of budget)
    remaining = budget - (time.monotonic() - t0)
    t_lse = 0.20 * budget
    if t_lse > 5 and remaining > 5:
        t_lse = min(t_lse, remaining * 0.8)
        a_lse = lbgfs_logsumexp(a, t_lse, T_start=50.0, T_end=0.03, n_stages=10)
        sc_lse = score_normalized(a_lse)
        if sc_lse < sc:
            a = a_lse
            sc = sc_lse
            print(f"  [{label}] After LSE: {sc:.8f} ({time.monotonic()-t0:.0f}s)")

    # Phase D: Final Nesterov LP (if time remains)
    remaining = budget - (time.monotonic() - t0)
    if remaining > 10:
        a_final, _ = nesterov_lp(a, remaining * 0.8, solver=solver)
        sc_final = score_normalized(a_final)
        if sc_final < sc:
            a = a_final
            sc = sc_final
            print(f"  [{label}] After final LP: {sc:.8f} ({time.monotonic()-t0:.0f}s)")

    return a, sc


_global_deadline = 0.0


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, deadline in {budget_s - 20}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    total_budget = _global_deadline - time.monotonic()
    print(f"Effective budget: {total_budget:.0f}s")

    # === Strategy: Try multiple lengths with focused budgets ===
    # Primary length gets the most time
    n_primary = n_base
    n_candidates = [n_primary, n_primary + 50, n_primary - 50, n_primary + 100,
                    n_primary + 200, n_primary - 100, n_primary + 300]
    n_candidates = sorted(set(max(200, n) for n in n_candidates if n >= 200))

    # Allocate time: primary gets 40%, others share 50%, 10% reserved
    primary_budget = 0.40 * total_budget
    other_budget = 0.50 * total_budget
    n_other = len(n_candidates) - 1
    per_other = other_budget / max(1, n_other)

    print(f"\n=== Primary optimization: n={n_primary} (budget={primary_budget:.0f}s) ===")

    # Use parent as initial for primary
    a_primary = best_seq.copy()
    a_out, sc_out = optimize_at_length(a_primary, primary_budget, solver, label="primary")
    try_update(a_out, "primary")
    print(f"  Primary done: best={best_score:.8f}")

    # Try other lengths
    used_time = time.monotonic() - (total_budget + time.monotonic() - _global_deadline)
    for n_try in n_candidates:
        if n_try == n_primary:
            continue
        remaining = _global_deadline - time.monotonic()
        if remaining < 20:
            print("  Time running out, skipping remaining lengths.")
            break
        t_alloc = min(per_other, remaining * 0.4)
        if t_alloc < 10:
            continue
        print(f"\n=== Trying n={n_try} (budget={t_alloc:.0f}s) ===")
        init_try = interpolate_sequence(best_seq, n_try)
        # Also try frequency init
        freq_try = frequency_init(n_try, rng)
        # Pick better initial
        sc_interp = score_normalized(init_try)
        sc_freq = score_normalized(freq_try)
        if sc_freq < sc_interp:
            init_try = freq_try
            print(f"  Using frequency init (score={sc_freq:.8f})")

        a_try, sc_try = optimize_at_length(init_try, t_alloc, solver, label=f"n{n_try}")
        try_update(a_try, f"len_{n_try}")

    # === Final polishing on best sequence found ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Final polish (remaining={remaining:.0f}s) ===")
    if remaining > 15 and best_seq is not None:
        # L-BFGS fine tune
        t_lse = min(remaining * 0.4, 60.0)
        if t_lse > 5:
            a_ft = lbgfs_logsumexp(best_seq, t_lse, T_start=20.0, T_end=0.01, n_stages=8)
            try_update(a_ft, "final_lse")

        # Another Nesterov LP round
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            t_lp = min(remaining * 0.6, 90.0)
            if t_lp > 5:
                a_lp, _ = nesterov_lp(best_seq, t_lp, solver=solver)
                try_update(a_lp, "final_lp")

        # Subgradient micro-polish
        remaining = _global_deadline - time.monotonic()
        if remaining > 5 and best_seq is not None:
            a = best_seq.copy()
            n = len(a)
            z = z_from_a(a)
            best_sg = a.copy()
            best_sg_sc = score_normalized(a)
            t0_sg = time.monotonic()
            t_budget_sg = min(remaining * 0.5, 30.0)

            for step in range(100000):
                if time.monotonic() - t0_sg > t_budget_sg:
                    break
                b = fast_convolve(a)
                max_b = b.max()
                active = np.where(b >= max_b * 0.9)[0]
                if len(active) == 0:
                    active = np.array([int(np.argmax(b))])
                g = np.zeros(n)
                for k in active:
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    idx = np.arange(js, je + 1)
                    g[idx] += 2.0 * a[k - idx]
                g /= len(active)
                S = np.dot(g, a)
                grad_z = a * (g - S)
                eta = 0.1 * 0.98 ** (step / 1000.0)
                z = z - eta * grad_z
                a_new = softmax_from_z(z)
                sc_new = score_normalized(a_new)
                if sc_new < best_sg_sc:
                    best_sg_sc = sc_new
                    best_sg = a_new.copy()
            try_update(best_sg, "subgrad_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        else:
            # Fallback to parent
            try:
                p = np.asarray(height_sequence_1, dtype=np.float64)
                p = np.clip(p, 0.0, 1000.0)
                if p.sum() > 0.01:
                    best_seq = p / p.sum()
                    best_score = evaluate_safe(best_seq)
                    print(f"FALLBACK to parent: {best_score:.8f}")
            except Exception:
                best_seq = np.ones(n_base) / n_base
                best_score = evaluate_safe(best_seq)
                print(f"FALLBACK to uniform: {best_score:.8f}")
    else:
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"FALLBACK to uniform: {best_score:.8f}")

    return [float(x) for x in best_seq]
