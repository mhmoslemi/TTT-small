    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n + 1) - (cssv - 1)) > 0)
    if rho is None:
        rho = 0
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n + 1) - (cssv - 1)) > 0)
    if rho is None:
        rho = 0
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def evaluate_local(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 locally."""

    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(c, a):
    """Linear cross-convolution conv(c, a) via FFT. Returns length len(c)+len(a)-1."""
    lc, la = len(c), len(a)
    size = lc + la - 1
    fft_size = 1 << (size - 1).bit_length()
    C = rfft(np.asarray(c, dtype=np.float64), fft_size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(C * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n + 1) - (cssv - 1)) > 0)
    if rho is None:
        rho = 0
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def evaluate_local(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_active_matrix(a_fixed, active_indices):
    """Build submatrix T_active where T[k,j] = a_fixed[k-j] for valid ranges."""
    n = len(a_fixed)
    T = np.zeros((len(active_indices), n))
    for i, k in enumerate(active_indices):
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        if js <= je:
            j_range = np.arange(js, je + 1)
            T[i, j_range] = a_fixed[k - j_range]
    return T


def solve_lp_active(a_fixed, active_indices, solver):
    """Solve LP: min M s.t. (T_active@b)[i] <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    T = build_active_matrix(a_fixed, active_indices)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))

            continue

        # Compute averaged subgradient: g[j] = sum_k w[k] * a[k-j]
        # This is a weighted cross-correlation
        # g[j] = sum_k w[k] * a[k-j] = (w * a_flipped)[j] (approximately)
        # Use FFT for efficiency
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # cross_correlation: g[j] = sum_k w[k] * a[k-j]
        # = ifft(conj(FFT(w)) * FFT(a))[j] (circular, with proper padding)
        g = irfft(np.conj(W) * A, fft_size).real[:n]

        # Multiplicative weights update
        eta = eta_base
        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Decay eta
        eta_base *= decay
        if no_improve > 500:
            eta_base = max(0.05, eta_base * 0.5)
            no_improve = 0

        if eta_base < 0.001:
            break

        if verbose and (step + 1) % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad step={step+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def active_set_polish(a_init, deadline, solver, verbose=True):
    """Polish by solving a sequence of QPs with active constraints.
    Uses a projected gradient approach on the active quadratic form."""
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    b = fast_convolve(a)
    max_b = b.max()

    # Identify active set
    rel_tol = 1e-4
    active = np.where(b >= max_b * (1.0 - rel_tol))[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])

    # Projected gradient on objective sum_k in active conv(a,a)[k]
    # Gradient: 2 * sum_k in active (Q_k @ a) = 2 * sum_k in active a[k-j] (reversed)
    # Which is 2 * cross_correlation(one_active, a) where one_active has 1 at active positions
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    indicator = np.zeros(2 * n - 1)
    indicator[active] = 1.0 / len(active)
    IND = rfft(indicator, fft_size)

    eta = 0.1
    best_pg = best_score
    no_improve = 0

    for it in range(500000):
        if time.monotonic() > deadline:
            break

        A = rfft(a, fft_size)
        grad = 2.0 * irfft(np.conj(IND) * A, fft_size).real[:n]

        # Projected gradient step with backtracking
        alpha = eta
        for backtrack in range(10):
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            b_trial = fast_convolve(a_trial)
            # Check if all active constraints improve
            max_active_trial = np.max(b_trial[active])
            max_active_curr = np.max(b[active])
            if max_active_trial < max_active_curr:
                break
            alpha *= 0.5
        else:
            break

        a = a_trial
        b = fast_convolve(a)
        score = 2.0 * n * b.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set periodically
        if it % 500 == 0:
            max_b = b.max()
            active = np.where(b >= max_b * (1.0 - rel_tol))[0]
            if len(active) == 0:
                active = np.array([np.argmax(b)])
            indicator[:] = 0
            indicator[active] = 1.0 / len(active)
            IND = rfft(indicator, fft_size)

        if no_improve > 200:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-6:
                break

        if verbose and (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Active PG iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def default_init(n):
    """Create a reasonable initial sequence."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def load_parent_sequence():
    """Try to load the parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast active-set LP alternating, subgradient polish, and multi-n search.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = load_parent_sequence()
    if parent is not None:
        n_init = len(parent)
        a_init = parent.copy()
        best_score = evaluate_local(a_init)
        best_seq = a_init.copy()
        print(f"Parent: n={n_init}, score={best_score:.10f}")
    else:
        n_init = 1200
        a_init = default_init(n_init)
        best_score = evaluate_local(a_init)
        best_seq = a_init.copy()
        print(f"Default init: n={n_init}, score={best_score:.10f}")

    n_best = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP Alternating (main workhorse, 45% time) ===
    elapsed = time.monotonic() - (total_deadline - budget_s + 25)
    print(f"\n=== Phase 1: LP Alternating (n={n_best}, score={best_score:.10f}) ===")
    remaining = total_deadline - time.monotonic()
    t_p1 = min(remaining * 0.45, 700.0)
    if t_p1 > 60:
        print(f"  Budget: {t_p1:.0f}s")
        d_p1 = time.monotonic() + t_p1
        a_lp, sc_lp = lp_nesterov_alternating(best_seq, d_p1, solver, K_active=40)
        check_and_update(a_lp, "lp_p1")

    # === Phase 2: Subgradient Polish (10% time) ===
    print(f"\n=== Phase 2: Subgradient Polish (score={best_score:.10f}) ===")
    remaining = total_deadline - time.monotonic()
    t_p2 = min(remaining * 0.10, 150.0)
    if t_p2 > 20:
        print(f"  Budget: {t_p2:.0f}s")
        d_p2 = time.monotonic() + t_p2
        a_sg, sc_sg = subgradient_polish(best_seq, d_p2)
        check_and_update(a_sg, "subgrad")

    # === Phase 3: LP Alternating Round 2 (20% time) ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.10f}) ===")
    remaining = total_deadline - time.monotonic()
    t_p3 = min(remaining * 0.20, 350.0)
    if t_p3 > 30:
        print(f"  Budget: {t_p3:.0f}s")
        d_p3 = time.monotonic() + t_p3
        a_lp2, sc_lp2 = lp_nesterov_alternating(best_seq, d_p3, solver, K_active=60)
        check_and_update(a_lp2, "lp_p3")

    # === Phase 4: Active Set Polish (10% time) ===
    print(f"\n=== Phase 4: Active Set Polish (score={best_score:.10f}) ===")
    remaining = total_deadline - time.monotonic()
    t_p4 = min(remaining * 0.10, 120.0)
    if t_p4 > 15:
        print(f"  Budget: {t_p4:.0f}s")
        d_p4 = time.monotonic() + t_p4
        a_ap, sc_ap = active_set_polish(best_seq, d_p4, solver)
        check_and_update(a_ap, "active_polish")

    # === Phase 5: Subgradient Round 2 (5% time) ===
    print(f"\n=== Phase 5: Subgradient R2 (score={best_score:.10f}) ===")
    remaining = total_deadline - time.monotonic()
    t_p5 = min(remaining * 0.05, 80.0)
    if t_p5 > 10:
        d_p5 = time.monotonic() + t_p5
        a_sg2, sc_sg2 = subgradient_polish(best_seq, d_p5)
        check_and_update(a_sg2, "subgrad_r2")

    # === Phase 6: Try n-1 ===
    print(f"\n=== Phase 6: Try n={n_best-1} (score={best_score:.10f}) ===")
    remaining = total_deadline - time.monotonic()
    n_alt1 = n_best - 1
    if n_alt1 >= 100 and remaining > 100:
        t_alt = min(remaining * 0.30, 200.0)
        if t_alt > 60:
            print(f"  Budget: {t_alt:.0f}s for n={n_alt1}")
            init_alt = interpolate_sequence(best_seq, n_alt1)
            d_alt = time.monotonic() + t_alt * 0.7
            a_alt, sc_alt = lp_nesterov_alternating(init_alt, d_alt, solver, K_active=40)
            check_and_update(a_alt, f"lp_n{n_alt1}")
            remaining = total_deadline - time.monotonic()
            if best_seq is not None and len(best_seq) == n_alt1 and remaining > 30:
                d_sg = time.monotonic() + min(remaining * 0.3, 50.0)
                a_sg_alt, _ = subgradient_polish(best_seq, d_sg)
                check_and_update(a_sg_alt, f"sg_n{n_alt1}")

    # === Phase 7: Try n+1 ===
    print(f"\n=== Phase 7: Try n={n_best+1} (score={best_score:.10f}) ===")
    remaining = total_deadline - time.monotonic()
    n_alt2 = n_best + 1
    if n_alt2 <= 5000 and remaining > 100:
        t_alt = min(remaining * 0.30, 200.0)
        if t_alt > 60:
            print(f"  Budget: {t_alt:.0f}s for n={n_alt2}")
            init_alt = interpolate_sequence(best_seq, n_alt2)
            d_alt = time.monotonic() + t_alt * 0.7
            a_alt, sc_alt = lp_nesterov_alternating(init_alt, d_alt, solver, K_active=40)
            check_and_update(a_alt, f"lp_n{n_alt2}")
            remaining = total_deadline - time.monotonic()
            if best_seq is not None and len(best_seq) == n_alt2 and remaining > 30:
                d_sg = time.monotonic() + min(remaining * 0.3, 50.0)
                a_sg_alt, _ = subgradient_polish(best_seq, d_sg)
                check_and_update(a_sg_alt, f"sg_n{n_alt2}")

    # === Phase 8: Final LP on best (remaining time) ===
    print(f"\n=== Phase 8: Final LP (score={best_score:.10f}) ===")
    remaining = total_deadline - time.monotonic()
    if remaining > 30:
        d_final = time.monotonic() + remaining * 0.7
        a_final, sc_final = lp_nesterov_alternating(best_seq, d_final, solver, K_active=50, verbose=False)
        check_and_update(a_final, "final_lp")
        remaining = total_deadline - time.monotonic()
        if remaining > 15:
            d_sg = time.monotonic() + remaining * 0.5
            a_sg_f, _ = subgradient_polish(best_seq, d_sg, verbose=False)
            check_and_update(a_sg_f, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
