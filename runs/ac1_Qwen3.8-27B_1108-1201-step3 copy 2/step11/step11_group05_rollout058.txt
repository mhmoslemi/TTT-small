        a = a_new

def subgradient_polish(a0, time_limit, n_steps=1000000):
    """Subgradient method with decreasing step size."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    eta = 0.1  # Initial step size
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_conv = fast_convolve(a)
        max_b = b_conv.max()
        
        # Find active set (top peaks)
        threshold = max_b * (1.0 - 1e-6)
        active = np.where(b_conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b_conv)])
        
        # Subgradient: g[i] = 2 * sum_{k in active} a[k-i]
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je+1] += a[k - je : k - js + 1]
        g *= 2.0 / len(active)
        
        # Mirror descent step (multiplicative weights)
        eta_step = eta / (1 + step * 0.001)
        a_new = a * np.exp(-eta_step * g)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Don't update a, just reduce eta
            if step % 1000 == 0:
                eta *= 0.95
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP alternating with scipy HiGHS (dense matrix, no cvxpy overhead)
2. Nesterov momentum acceleration on the alternating loop
3. Reduced-constraint LP for faster iterations (top-k peaks)
4. Subgradient/mirror-descent polishing with adaptive step
5. Multi-length exploration and perturbation-based escape
6. Adaptive time allocation across phases

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        end = min(size, j + n)
        T[j:end, j] = a[:end - j]
    return T


def solve_lp_response(a_fixed):
    """Solve: min M s.t. conv(a_fixed, b)[k] <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    size = 2 * n - 1
    A = build_conv_matrix(a_fixed)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((size, 1))])
    b_ub = np.zeros(size)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            b_opt = np.maximum(res.x[:n], 0)
            s = b_opt.sum()
            if s > 1e-15:
                b_opt = b_opt / s
            return b_opt, float(res.x[-1])
    except Exception:
        pass
    return None, None


def solve_lp_reduced(a_fixed, top_k_indices):
    """Solve LP with only top-k peak constraints for speed."""
    n = len(a_fixed)
    k = len(top_k_indices)

    A_r = np.zeros((k, n))
    for i, idx in enumerate(top_k_indices):
        js = max(0, idx - n + 1)
        je = min(n - 1, idx)
        A_r[i, js:je + 1] = a_fixed[idx - je:idx - js + 1]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A_r, -np.ones((k, 1))])
    b_ub = np.zeros(k)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            s = b_opt.sum()
            if s > 1e-15:

            return b_opt, float(res.x[-1])
    except Exception:
        pass
    return None, None


def evaluate_local(a):

def lp_alternating(a0, time_limit, use_reduced=False, reduced_k=60, nesterov_beta=0.7):

        a_look = np.maximum(a_look, 0)
        sa = a_look.sum()

            a_look = a.copy()
        else:
            a_look = a_look / sa

        # Best response for b given a_look
        if use_reduced:
            b_conv = fast_convolve(a_look)
            top_k = np.argpartition(b_conv, -reduced_k)[-reduced_k:]
            b_new, _ = solve_lp_reduced(a_look, top_k)

            b_new, _ = solve_lp_response(a_look)

            if it % 500 == 0:

                b_new, _ = solve_lp_response(a_look)
                if b_new is None:
                    break
            else:

        b_new = normalize(b_new)

def subgradient_polish(a0, time_limit):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    eta = 0.05
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b_conv = fast_convolve(a)
        max_b = b_conv.max()

        # Active set: peaks within 1e-5 of max
        threshold = max_b * (1.0 - 1e-5)
        active = np.where(b_conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b_conv)])

        # Subgradient: g[i] = 2 * (1/|A|) * sum_{k in A} a[k-i]
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je + 1] += a[k - je:k - js + 1]
        g *= 2.0 / len(active)

        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 200:
                eta *= 0.8
                no_improve = 0
                if eta < 1e-5:
                    break

        if (it if 'it' in dir() else 0) % 5000 == 0:
            pass

    return best_a, best_score

    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating with Nesterov momentum, subgradient polishing,
    multi-length exploration, and perturbation escape.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}, budget={budget_s:.0f}s")

    # Track best for updates (avoiding closure)
    # We'll use a simple list to hold [best_seq, best_score]
    state = [best_seq, best_score]

    def update_state(a, label):
        if a is None:
            return
        a = normalize(a)
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < state[1]:
            state[1] = sc
            state[0] = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Full LP Alternating (accurate, main workhorse) ===
    print(f"\n=== Phase 1: Full LP Alternating n={n_best} (score={state[1]:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating(state[0], t_alloc, use_reduced=False,
                                  reduced_k=60, nesterov_beta=0.7)
        update_state(a_lp, "lp_full_1")

    # === Phase 2: Reduced LP Alternating (fast, many iterations) ===
    print(f"\n=== Phase 2: Reduced LP Alternating n={n_best} (score={state[1]:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternating(state[0], t_alloc, use_reduced=True,
                                   reduced_k=80, nesterov_beta=0.5)
        update_state(a_lp2, "lp_reduced_1")

    # === Phase 3: Subgradient Polishing ===
    print(f"\n=== Phase 3: Subgradient Polish n={n_best} (score={state[1]:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_polish(state[0], t_alloc)
        update_state(a_sg, "subgrad")

    # === Phase 4: Second LP round (full + reduced) ===
    print(f"\n=== Phase 4: LP Round 2 n={n_best} (score={state[1]:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_lp4 = min(rem * 0.20, 200.0)
        print(f"  Budget: {t_lp4:.0f}s")
        a_lp3, _ = lp_alternating(state[0], t_lp4 * 0.5, use_reduced=False,
                                   reduced_k=60, nesterov_beta=0.3)
        update_state(a_lp3, "lp_full_2")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_lp4, _ = lp_alternating(state[0], min(rem * 0.6, 100.0),
                                       use_reduced=True, reduced_k=100,
                                       nesterov_beta=0.3)
            update_state(a_lp4, "lp_reduced_2")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={state[1]:.8f}) ===")
    rem = deadline - time.monotonic()
    target_lengths = [1100, 1300, 1400, 1000, 1500]
    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_n = min(rem * 0.12, 120.0)
        print(f"  Trying n={target_n} (budget={t_n:.0f}s)")
        a_interp = interpolate_sequence(state[0], target_n)
        a_test, _ = lp_alternating(a_interp, t_n, use_reduced=True,
                                    reduced_k=50, nesterov_beta=0.5)
        update_state(a_test, f"n{target_n}")

    # === Phase 6: Perturbation escape ===
    print(f"\n=== Phase 6: Perturbation Escape (score={state[1]:.8f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(state[0])
    for trial in range(5):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        strength = 0.003 + 0.003 * trial
        x = np.linspace(0, 2 * np.pi, n_cur)
        freq = rng.integers(1, 12)
        phase = rng.uniform(0, 2 * np.pi)
        a_pert = state[0] + strength * np.cos(freq * x + phase)
        a_pert = normalize(a_pert)
        t_sub = min(rem * 0.20, 50.0)
        a_opt, _ = lp_alternating(a_pert, t_sub, use_reduced=True,
                                   reduced_k=50, nesterov_beta=0.5)
        update_state(a_opt, f"pert{trial}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (score={state[1]:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_fin = min(rem * 0.5, 30.0)
        a_fin, _ = subgradient_polish(state[0], t_fin)
        update_state(a_fin, "final_subgrad")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin2, _ = lp_alternating(state[0], min(rem * 0.6, 20.0),
                                        use_reduced=True, reduced_k=120,
                                        nesterov_beta=0.2)
            update_state(a_fin2, "final_lp")

    # Final answer
    best_seq = state[0]
    best_score = state[1]

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    fs = evaluate_local(best_seq)
    print(f"\nFINAL (local): score={fs:.8f}, n={len(best_seq)}")

    # Clip and convert for submission
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with the actual evaluator
    try:
        final_eval = evaluate_sequence(result)
        print(f"FINAL (evaluate_sequence): {final_eval:.8f}")
    except Exception as e:
        print(f"Verification error: {e}")

    return result
