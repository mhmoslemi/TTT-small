Key insight: The problem is to minimize max_k sum_i a_i * a_{k-i} subject to sum a_i = 1, a_i ≥ 0.

This is equivalent to: minimize t subject to sum_i a_i * a_{k-i} ≤ t for all k, sum a_i = 1, a_i ≥ 0.

This is a quadratic program (or rather, a non-convex problem since the constraints are quadratic). But we can think of it as: find the sequence whose "convolution profile" is as flat as possible.

For the continuous analog: minimize max_t ∫_0^1 f(x)f(t-x)dx (with appropriate domain) subject to ∫f = 1, f ≥ 0.

I recall that for the continuous case, the optimal solution is related to the "bang-bang" or "two-level" solutions in optimal control.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally using fast convolution."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gs_spectral_flatten(a0, n_iters=100000, beta=0.5, relax=1.0, time_limit=None):
    """
    Gerchberg-Saxton-like iteration to flatten the power spectrum.
    Iteratively modifies |FFT(a)| to be more flat, then projects back to non-negative.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time_limit is not None:
            if time.monotonic() - t0 > time_limit:
                break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 4  # Power spectrum of the convolution
        P_mean = P.mean() + 1e-30

        # Interpolate |A| toward flat
        # |A_new|^4 = |A|^4 * (P_mean / P)^beta  =>  |A_new| = |A| * (P_mean/P)^(beta/4)
        ratio = (P_mean / (P + 1e-30)) ** (beta / 4.0)
        A_new = np.abs(A) * ratio * np.exp(1j * np.angle(A))

        a_new_complex = irfft(A_new, fft_size)[:n]
        a_new = np.real(a_new_complex)
        a_new = np.maximum(a_new, 0)

        # Relaxation
        a_new = (1 - relax) * a + relax * a_new
        a_new = normalize(a_new)

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def gs_adaptive(a0, time_limit, n_total_iters=200000):

def lse_obj_grad_z(z, n, T):
    """Compute LSE objective and gradient in log-parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)[:size]

    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse

    w = exp_c / exp_c.sum()

    # Gradient computation
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A2 = rfft(a, N)
    R = irfft(W * np.conj(A2), N)
    b = R[:n]

    g = 2.0 * b
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]

    S = np.dot(g, a)
    grad_z = a * (g - S)

    return obj, grad_z


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(np.asarray(a), 1e-300))


def lbgfs_optimize(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, n, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, n, stage_T)
            return g

        iters = min(maxiter_per_stage, max(50, int(remaining * 50)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_check = softmax_from_z(z)
        sc = evaluate_local(a_check)
        print(f"    L-BFGS T={stage_T}: score={sc:.8f}", flush=True)

    return softmax_from_z(z)


def multiplicative_updates(a0, time_limit, n_steps=500000, eta0=0.05):
    """Multiplicative (mirror) descent with adaptive step size."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0

    for step in range(n_steps):
        if step % 1000 == 0 and time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        c = irfft(A * A, fft_size)[:2 * n - 1]
        c_max = c.max()

        # Softmax weights over convolution values
        T = max(0.01, c_max * 0.01)
        w = np.exp((c - c_max) / T)
        w /= w.sum()

        # Subgradient: g_m = 2 * sum_k w[k] * a[k-m] (cross-correlation)
        N = 1 << (len(w) + n).bit_length()
        W = rfft(w, N)
        A2 = rfft(a, N)
        R = irfft(W * np.conj(A2), N)[:n]
        g = 2.0 * R

        # Multiplicative update
        log_alpha = -eta * g
        log_alpha -= log_alpha.max()
        a_new = a * np.exp(log_alpha)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                eta *= 0.5
                no_improve = 0
                if eta < 1e-8:
                    break

    return best_a, best_score

    a = np.linspace(0.01, 1, n)
    candidates.append(normalize(a))

    # Inverted ramp
    a = np.linspace(1, 0.01, n)
    candidates.append(normalize(a))

    # Two-level: first half small, second half large
    a = np.zeros(n)
    a[:n//2] = 0.3
    a[n//2:] = 0.7
    candidates.append(normalize(a))

    # Two-level reversed
    a = np.zeros(n)
    a[:n//2] = 0.7
    a[n//2:] = 0.3
    candidates.append(normalize(a))

    # Quadratic (parabola)
    a = x * (1 - x) + 0.01
    candidates.append(normalize(a))

    # Cosine window
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    # Anti-palindromic: a[i] + a[n-1-i] = const
    c = 2.0 / n
    a = c / 2 + 0.4 * c * np.sin(np.pi * x)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    # Power law
    a = x ** 0.5 * (1 - x) ** 0.5 + 0.01
    candidates.append(normalize(a))

    # Skewed Gaussian
    a = np.exp(-30 * (x - 0.6) ** 2)
    candidates.append(normalize(a))

    # Another skew
    a = np.exp(-30 * (x - 0.4) ** 2)
    candidates.append(normalize(a))

    # Random smooth
    coeffs = rng.normal(0, 1, 8)
    a = np.zeros(n)
    for i, c_val in enumerate(coeffs):
        a += c_val * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 0.5, 0)
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def peak_shaving(a0, time_limit, n_rounds=50000):
    """Directly reduce the highest convolution peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    rng = np.random.default_rng(789)

    lr = 0.001
    for rnd in range(n_rounds):
        if rnd % 1000 == 0 and time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        c = irfft(A * A, fft_size)[:2 * n - 1]
        c_max = c.max()
        max_pos = c.argmax()

        # Find which a[i] contribute most to c[max_pos]
        # c[max_pos] = sum_i a[i] * a[max_pos - i]
        lo = max(0, max_pos - n + 1)
        hi = min(n - 1, max_pos)
        contributors = np.arange(lo, hi + 1)
        weights = np.array([a[i] * a[max_pos - i] for i in contributors])
        total = weights.sum()
        if total < 1e-15:
            continue

        # Reduce the top contributors
        top_k = min(10, len(contributors))
        top_idx = contributors[np.argsort(weights)[-top_k:]]

        # Reduce these, increase others proportionally
        reduction = lr * c_max
        a_new = a.copy()
        for i in top_idx:
            a_new[i] = max(0, a_new[i] - reduction / len(top_idx))

        # Redistribute to maintain sum
        deficit = a.sum() - a_new.sum()
        if deficit > 0:
            # Add to non-contributors
            non_contrib = set(range(n)) - set(top_idx.tolist())
            non_contrib_list = list(non_contrib)
            add_per = deficit / len(non_contrib_list)
            for i in non_contrib_list:
                a_new[i] += add_per

        a_new = normalize(a_new)
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            lr *= 0.5
            if lr < 1e-10:
                break

    return best_a, best_score


def population_search(a_init, time_limit, pop_size=12, rng=None):
    """Population-based search with adaptive mutations and GS refinement."""
    if rng is None:
        rng = np.random.default_rng(456)

    n = len(a_init)
    population = [normalize(a_init.copy())]

    # Diverse perturbations
    for i in range(pop_size - 1):
        a_pert = a_init.copy()
        method = i % 5
        if method == 0:
            a_pert += rng.normal(0, 0.02, n)
        elif method == 1:
            freq = rng.integers(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert += 0.02 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        elif method == 2:
            blk = rng.integers(20, 100)
            i1 = rng.integers(0, n - blk)
            a_pert[i1:i1+blk] *= 1.5
            a_pert[i1:i1+blk] = np.maximum(a_pert[i1:i1+blk], 0)
        elif method == 3:
            # Edge push
            side = rng.integers(0, 2)
            a_pert[:30] += 0.03
            a_pert[30:] -= 0.03 * 30 / (n - 30)
        else:
            # Random block swap
            blk = rng.integers(20, 80)
            i1 = rng.integers(0, n - blk)
            i2 = rng.integers(0, n - blk)
            if abs(i1 - i2) > blk:
                a_pert[i1:i1+blk], a_pert[i2:i2+blk] = a_pert[i2:i2+blk].copy(), a_pert[i1:i1+blk].copy()

        population.append(normalize(a_pert))

    scores = [evaluate_local(p) for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]

    t0 = time.monotonic()
    gen = 0
    adaptive_strength = 0.02

    while time.monotonic() - t0 < time_limit:
        gen += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        keep = max(3, pop_size // 3)
        elites = population[:keep]

        new_pop = list(elites)

        for i in range(pop_size - len(new_pop)):
            parent = elites[rng.integers(0, keep)]
            child = parent + rng.normal(0, adaptive_strength, n)
            child = normalize(child)

            # Occasionally apply GS refinement to a child
            if rng.random() < 0.3 and remaining > 60:
                child_gs, _ = gs_spectral_flatten(child, n_iters=500, beta=0.5, relax=0.8)
                child = child_gs

            new_pop.append(child)

        # Crossover
        if len(elites) >= 2 and len(new_pop) < pop_size:
            p1, p2 = elites[0], elites[1]
            t = rng.uniform(0.2, 0.8)
            new_pop.append(normalize(t * p1 + (1 - t) * p2))

        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]

        if gen % 20 == 0:
            if scores[0] >= best_score - 1e-10:
                adaptive_strength *= 0.85
            else:
                adaptive_strength = min(0.05, adaptive_strength * 1.1)

        if gen % 100 == 0:
            print(f"    PopSearch gen={gen}: best={best_score:.8f}", flush=True)

    # GS refine the best
    best_gs, gs_score = gs_adaptive(best_a, min(remaining, 60.0), n_total_iters=50000)
    if gs_score < best_score:
        best_score = gs_score
        best_a = best_gs

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimum peak convolution.
    Uses GS spectral flattening, LSE gradients, population search, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})", flush=True)
        return sc

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}", flush=True)
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Phase 0: If no parent, find good initialization
    if parent is None:
        print("No parent found. Generating initial candidates...", flush=True)
        best_init = None
        best_init_score = float('inf')
        for n_try in [800, 1000, 1200, 1500]:
            candidates = diverse_init(n_try, rng)
            for c in candidates:
                sc = local_update(c, f"init_n{n_try}")
                if sc < best_init_score:
                    best_init_score = sc
                    best_init = c
        best_seq = best_init
        best_score = best_init_score

    n_best = len(best_seq)
    print(f"\nStarting: n={n_best}, score={best_score:.8f}", flush=True)

    # Phase 1: GS Spectral Flattening (primary, fast)
    print(f"\n=== Phase 1: GS Spectral Flattening ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.25, 300.0)
        a_gs, gs_score = gs_adaptive(best_seq, t_alloc, n_total_iters=500000)
        local_update(a_gs, "GS_main")
        # Also try with different beta settings
        for beta_try in [0.2, 0.7, 1.0]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            a_gs2, _ = gs_spectral_flatten(best_seq, n_iters=30000, beta=beta_try, relax=0.9,
                                           time_limit=min(rem * 0.3, 30.0))
            local_update(a_gs2, f"GS_beta{beta_try}")

    # Phase 2: L-BFGS with LSE
    print(f"\n=== Phase 2: L-BFGS LSE (score={best_score:.8f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.25, 300.0)
        T_stages = [20, 10, 5, 3, 2, 1, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
        a_lb = lbgfs_optimize(best_seq, t_alloc, T_stages, maxiter_per_stage=3000)
        local_update(a_lb, "L-BFGS")

    # Phase 3: GS again (refine after gradient steps)
    print(f"\n=== Phase 3: GS Refine (score={best_score:.8f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.15, 120.0)
        a_gs3, _ = gs_adaptive(best_seq, t_alloc, n_total_iters=300000)
        local_update(a_gs3, "GS_refine")

    # Phase 4: Population search
    print(f"\n=== Phase 4: Population Search (score={best_score:.8f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.20, 240.0)
        a_pop, _ = population_search(best_seq, t_alloc, pop_size=12, rng=rng)
        local_update(a_pop, "population")

    # Phase 5: Peak shaving
    print(f"\n=== Phase 5: Peak Shaving (score={best_score:.8f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.10, 60.0)
        a_ps, _ = peak_shaving(best_seq, t_alloc, n_rounds=100000)
        local_update(a_ps, "peak_shave")

    # Phase 6: Multiplicative updates
    print(f"\n=== Phase 6: Multiplicative Updates (score={best_score:.8f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.10, 60.0)
        a_mu, _ = multiplicative_updates(best_seq, t_alloc, n_steps=200000, eta0=0.03)
        local_update(a_mu, "mult_update")

    # Phase 7: Length exploration
    print(f"\n=== Phase 7: Length Exploration (score={best_score:.8f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 90:
        for n_try in [n_best + 200, n_best - 200, n_best + 500, n_best - 500,
                      n_best + 800, n_best - 800]:
            if n_try < 200 or n_try > 5000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            # Quick GS on new length
            a_try, _ = gs_adaptive(init, min(rem_sub * 0.25, 30.0), n_total_iters=100000)
            local_update(a_try, f"len_{n_try}")

    # Phase 8: Final combined polish
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 30:
        # GS polish
        t1 = min(rem * 0.4, 30.0)
        a_polish, _ = gs_adaptive(best_seq, t1, n_total_iters=100000)
        local_update(a_polish, "polish_gs")

        # L-BFGS fine
        rem = deadline - time.monotonic()
        if rem > 20:
            t2 = min(rem * 0.5, 30.0)
            T_final = [1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_fin = lbgfs_optimize(best_seq, t2, T_final, maxiter_per_stage=2000)
            local_update(a_fin, "polish_lbfgs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})", flush=True)

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
