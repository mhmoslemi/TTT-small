Cross-correlation: (x ⋆ y)[m] = sum_n x[n+m] * conj(y[n])

In frequency: (x ⋆ y) has DFT = X[k] * conj(Y[k]).

So ifft(fft(d) * conj(fft(a))) gives (d ⋆ a)[m] = sum_n d[n+m] * a[n] (for real sequences).

And I want cross[j] = sum_{m=0}^{n-1} d[j+m] * a[m], which is (d ⋆ a)[j] for j = 0, ..., n-1.

With N >= (2n-1) + n - 1 = 3n-2, the circular cross-correlation equals the linear one for the relevant range.

So: corr = ifft(fft(d, N) * conj(fft(a, N))).real, and cross = corr[:n].

def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Find τ such that sum(max(v_i - τ, 0)) = 1
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

# Global state for L-BFGS callbacks
_g_state = {'best_a': None, 'best_score': float('inf'), 'p': 2, 'n': 0}


def _softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def _fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def _cross_corr(d, a):
    """cross[j] = sum_{m=0}^{n-1} d[j+m] * a[m] for j=0..n-1.
    d has length 2n-1, a has length n."""
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    D = np.fft.fft(d, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(D * np.conj(A)).real
    return corr[:n]


def _score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    b = _fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def _project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _projected_gradient(a0, time_limit, p, eta0=0.01, max_iters=200000):
    """Nesterov-accelerated projected gradient on p-norm objective."""
    n = len(a0)

    y = a.copy()
    prev_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Nesterov momentum
        momentum = 2.0 / (it + 3.0)
        y = a + momentum * (a - prev_a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        # Compute gradient at y
        b = _fast_conv(y)
        b = np.maximum(b, 0)
        b_p = b ** p
        S = b_p.sum()
        if S < 1e-300:
            break
        d = b ** (p - 1)
        cross = _cross_corr(d, y)
        grad = 2.0 * n * S ** (1.0 / p - 1.0) * cross

        # Adaptive step size
        eta = eta0 / (1.0 + 0.01 * it)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            eta_eff = eta * (1.0 / (grad_norm + 1e-8))
            eta_eff = min(eta_eff, 1.0)
        else:
            eta_eff = eta

        a_new = _project_simplex(y - eta_eff * grad)
        prev_a = a.copy()
        a = a_new

        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _coordinate_descent(a0, time_limit, n_steps=500000):
    """Coordinate descent focusing on peak-reducing moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = _fast_conv(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(16, len(idx))
        top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_pos:

                continue
            for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98]:

                sc = _score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial

                    improved = True
                    break

                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score
