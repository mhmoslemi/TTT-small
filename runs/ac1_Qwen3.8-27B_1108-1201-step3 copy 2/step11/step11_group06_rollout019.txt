        if g[j_max] - g[j_min] < 1e-12:
            break  # Converged
        
        # Step size: full step or line search
        gamma = 2.0 / (it + 2)  # Standard FW step size
        
        # Pairwise step: move from j_max towards j_min
        a_new = a.copy()
        a_new[j_min] += gamma * (a[j_max] - a[j_min])
        a_new[j_max] -= gamma * (a[j_max] - a[j_min])
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
        
        # Also try away step from j_max
        if a[j_max] > gamma:
            a_away = a.copy()
            a_away[j_max] -= gamma
            a_away = np.maximum(a_away, 0)
            sa = a_away.sum()
            if sa > 1e-15:
                a_away = a_away / sa
                score_away = 2.0 * n * fast_convolve(a_away).max()
                if score_away < 2.0 * n * fast_convolve(a_new).max():
                    a_new = a_away
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        
        if it > 0 and it % 50000 == 0:
            print(f"    FW: iter={it}, score={best_score:.8f}")
    
    return best_a, best_score


def multiplicative_weights_optimized(a0, time_limit, eta0=0.03, active_frac=0.02):
    """Optimized multiplicative weights with adaptive active set and step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    stall = 0
    
    for step in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_convolve(a)
        max_b = conv_a.max()
        threshold = max_b * (1.0 - active_frac)
        active = np.where(conv_a >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_a)])
        
        w = conv_a[active]
        w = w / w.sum()
        g = np.zeros(n)
        for wi, k in zip(w, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            jidx = np.arange(js, je + 1)
            g[jidx] += wi * a[k - jidx]
        
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 0.4)
            stall = 0
        else:
            stall += 1
            eta *= 0.9
            if stall > 500:
                active_frac = min(active_frac * 1.5, 0.15)
                stall = 0
            if eta < 1e-10:
                break
        
        if step % 100000 == 0 and step > 0:
            print(f"    MW: step={step}, score={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score


def nesterov_lp_refine_v2(a0, time_limit, solver, n_max_iters=1000000):
    """LP alternating with Nesterov dual averaging, dual-gap tracking, and adaptive perturbation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    rng = np.random.default_rng(9999)
    T_mat = build_conv_matrix_sparse(y)
    prev_best = best_score
    lp_times = []
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            continue
        
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 5.0:
                print(f"    LP slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 1e-300)
        y = y / y.sum()
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 1e-300)
        a = a / a.sum()
        
        T_mat = build_conv_matrix_sparse(y)
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 8:
                # Adaptive perturbation: try different frequencies
                strength = 0.005 + 0.01 * (stall_count // 10)
                strength = min(strength, 0.05)
                x = np.linspace(0, 2 * np.pi, n)
                n_freqs = np.random.default_rng(it).integers(2, 5)
                for _ in range(n_freqs):
                    freq = np.random.default_rng(it + 1).integers(1, max(2, n // 50))
                    phase = np.random.default_rng(it + 2).uniform(0, 2 * np.pi)
                    y += strength * np.cos(freq * x + phase)
                y = np.maximum(y, 1e-300)
                y = y / y.sum()
                a = y.copy()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP: iter={it+1} ({elapsed:.0f}s), score={best_score:.8f}")
    
    return best_a, best_score


def lse_obj_grad(z, n, T_temp, deadline):
    """Compute LogSumExp smoothed objective and gradient w.r.t. log-domain z."""
    if time.monotonic() > deadline:
        return 1e15, None
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    a = np.maximum(a, 1e-300)
    a = a / a.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled_shifted.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine_v2(a0, time_limit, T_stages=None, maxiter_per_stage=600):
    """L-BFGS with LogSumExp smoothing and fine temperature annealing."""
    if T_stages is None:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
    
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        stage_budget = min(remaining, (time_limit / len(T_stages)) * 2)
        
        def func(zv):
            obj, _ = lse_obj_grad(zv, n, stage_T, deadline)
            return obj
        
        def jac(zv):
            _, g = lse_obj_grad(zv, n, stage_T, deadline)
            if g is None:
                return np.zeros(len(zv))
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage,
                         'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': maxiter_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    return np.maximum(a, 0)


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        # Edge-concentrated (based on sum-distribution analysis)
        a = np.exp(-8 * x * (1 - x)) + 0.3
    else:
        # Multi-scale
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 11 * x) + 0.2 * np.cos(2 * np.pi * 23 * x + 1.0)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def optimize_at_length(a_init, length, time_budget, solver):
    """Full optimization pipeline at a specific length."""
    n = length
    a = interpolate_sequence(a_init, n) if len(a_init) != n else a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    # Phase 1: Frank-Wolfe (fast)
    t1 = min(time_budget * 0.1, 30.0)
    if t1 > 5:
        a_fw, sc_fw = frank_wolfe_pre(a, t1, max_iters=100000)
        if sc_fw < best_score:
            best_score = sc_fw
            best_a = a_fw
    
    # Phase 2: Multiplicative weights
    t2 = min(time_budget * 0.15, 60.0)
    if t2 > 5:
        a_mw, sc_mw = multiplicative_weights_optimized(best_a, t2, eta0=0.03, active_frac=0.02)
        if sc_mw < best_score:
            best_score = sc_mw
            best_a = a_mw
    
    # Phase 3: LP alternating (main workhorse)
    t3 = time_budget * 0.5
    if t3 > 20:
        a_lp, sc_lp = nesterov_lp_refine_v2(best_a, t3, solver, n_max_iters=500000)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp
    
    # Phase 4: LSE polishing
    t4 = time_budget * 0.2
    if t4 > 10:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_lse = lbgfs_refine_v2(best_a, t4, T_stages, maxiter_per_stage=400)
        sc_lse = evaluate_local(a_lse)
        if sc_lse < best_score:
            best_score = sc_lse
            best_a = a_lse
    
    # Phase 5: Final LP squeeze
    t5 = time_budget * 0.05
    if t5 > 10:
        a_fin, sc_fin = nesterov_lp_refine_v2(best_a, t5, solver, n_max_iters=100000)
        if sc_fin < best_score:
            best_score = sc_fin
            best_a = a_fin
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Frank-Wolfe pre-optimization, LP alternating with dual tracking,
    LSE polishing, and multi-length exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")
    
    # === Phase A: Optimize at parent length ===
    print(f"\n=== Phase A: Optimize at n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    tA = min(rem * 0.35, 500.0)
    if tA > 60:
        print(f"  Budget: {tA:.0f}s")
        a_A, sc_A = optimize_at_length(best_seq, len(best_seq), tA, solver)
        if sc_A < best_score:
            best_score = sc_A
            best_seq = a_A.copy()
            print(f"  NEW BEST: {best_score:.8f} (n={len(best_seq)})")
    
    # === Phase B: Palindromic optimization ===
    print(f"\n=== Phase B: Palindromic (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        tB = min(rem * 0.15, 200.0)
        a_pal = make_palindromic(best_seq)
        a_B, sc_B = optimize_at_length(a_pal, len(a_pal), tB, solver)
        if sc_B < best_score:
            best_score = sc_B
            best_seq = a_B.copy()
            print(f"  NEW BEST: {best_score:.8f} (n={len(best_seq)})")
    
    # === Phase C: Multi-length exploration ===
    print(f"\n=== Phase C: Multi-length (score={best_score:.8f}) ===")
    n_current = len(best_seq)
    # Try lengths around the current best
    n_candidates = [800, 1000, 1100, 1200, 1300, 1400, 1500, 1600, 1800, 2000]
    n_candidates = [n for n in n_candidates if n != n_current]
    # Sort by proximity to current best (most likely to help first)
    n_candidates.sort(key=lambda x: abs(x - n_current))
    
    for ln in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        # Allocate less time for further lengths
        proximity_factor = 1.0 - 0.1 * abs(ln - n_current) / max(1, n_current)
        t_C = min(rem * 0.2 * proximity_factor, 180.0)
        if t_C < 30:
            continue
        print(f"  Trying n={ln}, budget={t_C:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        a_C, sc_C = optimize_at_length(init_ln, ln, t_C, solver)
        if sc_C < best_score:
            best_score = sc_C
            best_seq = a_C.copy()
            print(f"  NEW BEST: {best_score:.8f} (n={len(best_seq)})")
        rem = deadline - time.monotonic()
        if rem < 60:
            break
    
    # === Phase D: Final polish at best length ===
    print(f"\n=== Phase D: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        tD = min(rem * 0.4, 250.0)
        # LP polish
        tD_lp = tD * 0.5
        a_dp, sc_dp = nesterov_lp_refine_v2(best_seq, tD_lp, solver, n_max_iters=200000)
        if sc_dp < best_score:
            best_score = sc_dp
            best_seq = a_dp.copy()
            print(f"  NEW BEST (LP): {best_score:.8f}")
        
        # LSE polish
        rem = deadline - time.monotonic()
        if rem > 15:
            tD_lse = min(rem * 0.6, tD * 0.4)
            T_stages_fine = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
            a_dl = lbgfs_refine_v2(best_seq, tD_lse, T_stages_fine, maxiter_per_stage=500)
            sc_dl = evaluate_local(a_dl)
            if sc_dl < best_score:
                best_score = sc_dl
                best_seq = a_dl.copy()
                print(f"  NEW BEST (LSE): {best_score:.8f}")
    
    # === Phase E: Squeeze - short aggressive LP rounds ===
    print(f"\n=== Phase E: Squeeze (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Multiple short LP rounds with perturbation between
        n_rounds = min(5, int(rem // 30))
        for r in range(n_rounds):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            if r > 0:
                # Perturb before each round
                n = len(best_seq)
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(2, 30)
                phase = rng.uniform(0, 2 * np.pi)
                best_seq = np.maximum(best_seq + 0.008 * np.cos(freq * x + phase), 0)
                best_seq = best_seq / best_seq.sum()
            t_E = min(rem * 0.6, 50.0)
            a_E, sc_E = nesterov_lp_refine_v2(best_seq, t_E, solver, n_max_iters=50000)
            if sc_E < best_score:
                best_score = sc_E
                best_seq = a_E.copy()
                print(f"  NEW BEST (squeeze {r}): {best_score:.8f}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")
    
    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
