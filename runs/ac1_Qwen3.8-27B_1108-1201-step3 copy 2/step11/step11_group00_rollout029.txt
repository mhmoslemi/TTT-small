One important optimization: instead of rebuilding the convolution matrix from scratch each iteration, I can precompute it once (since a_fixed is the extrapolation point y, which changes each iteration). Actually, y changes each iteration, so I do need to rebuild it. But the matrix construction is O(n^2) which is fast enough.

Actually wait - in the Nesterov LP method, at each iteration we:
1. Have current y (the extrapolation point)
2. Solve LP: min M s.t. conv(y, b)_k <= M, sum(b)=1, b>=0
3. The LP uses y as the fixed sequence

So the convolution matrix T(y) changes each iteration. We need to rebuild it each time. For n=1600, building a 3199x1600 matrix takes about 5ms, which is fine.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP (zero-sum game) - primary engine
2. Projected subgradient descent - local refinement
3. LogSumExp-smoothed L-BFGS - smooth optimization
4. Peak-shaving coordinate descent - final polish
5. Multi-length exploration with interpolation transfer
Key: efficient LP solving, aggressive iteration count, smart time allocation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000):
    """
    Nesterov-accelerated LP best-response iteration for the zero-sum game.
    Uses running average to track best solution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    a_prev = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max() / 1.0  # sum(a)=1 so sum^2=1
    # Actually score = 2*n*max_b/(sum_a)^2; if sum_a=1, score = 2*n*max_b
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    iter_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        iter_count += 1
        alpha_t = 2.0 / (t + 2.0)

        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / sa

        y = y_new
        a_prev = a
        a = a_new

        # Check score of a
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Check score of y
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # Gradient w.r.t. a: grad_a[i] = 2 * sum_k w[k] * (conv with reflected w)
    # conv(a[::-1], w)[n-1+i] gives sum_j a[j] * w[n-1+i-j]
    # But we need sum_k w[k] * a[k-i] (for valid indices)
    # This is the cross-correlation of a and w
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=15):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters_per_stage = max(10, min(60, int(30 + 30 * (si / max(1, n_stages - 1)))))

        def func_lse(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac_lse(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def subgradient_polish(a0, time_limit, lr_start=0.001):
    """
    Projected subgradient descent on max_k conv(a,a)_k.
    The subgradient at the active lag k* is 2*a_reflected where a_reflected[j] = a[k*-j].
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    lr = lr_start
    decay = 0.9995
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        # Subgradient: 2 * a[k_star - j] for valid j
        # a_ref[j] = a[k_star - j] if 0 <= k_star - j < n else 0
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        grad = np.zeros(n)
        # For j in [lo, hi]: grad[j] = 2 * a[k_star - j]
        js = np.arange(lo, hi + 1)
        grad[js] = 2.0 * a[k_star - js]

        # Projected subgradient step: a_new = Proj_simplex(a - lr * grad)
        a_trial = a - lr * grad
        # Project onto simplex
        a_trial = project_simplex(a_trial)

        score = 2.0 * n * fast_autocorr(a_trial).max()
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            stall = 0
        else:
            stall += 1

        lr *= decay
        if lr < 1e-10:
            lr = 1e-10
            # Reset with a smaller step
            stall += 5

        if stall > 500:
            # Try a different active lag (second worst)
            b_copy = b.copy()
            b_copy[k_star] = -np.inf
            k_alt = int(np.argmax(b_copy))
            lo2 = max(0, k_alt - (n - 1))
            hi2 = min(n - 1, k_alt)
            grad2 = np.zeros(n)
            js2 = np.arange(lo2, hi2 + 1)
            grad2[js2] = 2.0 * a[k_alt - js2]
            a_trial = project_simplex(a - lr * grad2)
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                stall = 0
            else:
                stall += 10
                lr *= 0.5

        if stall > 2000:
            break

    return best_a, best_score


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate descent: move mass from high-autocorrelation-contributing positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        improved = False

        for k in top_k:
            # For lag k, the contribution is sum_js a[j]*a[k-j]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            contrib = a[js] * a[k - js]

            if len(js) == 0:
                continue
            src_local = int(np.argmax(contrib))
            src = js[src_local]

            # Find destination: position with low value not in peak area
            peak_area = set()
            for k2 in top_k:
                l2 = max(0, k2 - (n - 1))
                h2 = min(n - 1, k2)
                for j in range(l2, h2 + 1):
                    peak_area.add(j)

            candidates = [j for j in range(n) if j not in peak_area and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Pick from low-value candidates
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = int(dst_pool[rng.integers(len(dst_pool))])

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s2 = a_trial.sum()
                if s2 < 1e-15:
                    continue
                a_trial = a_trial / s2
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s2 = a_trial.sum()
                    a_trial = a_trial / s2
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations with ramp-like shapes."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * x  # gentle ramp up
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * x  # stronger ramp
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.sin(np.pi * x)  # slight bump in middle
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * x ** 2  # quadratic ramp
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 0.8 + 0.4 * x + 0.1 * np.cos(2 * np.pi * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.ones(n)  # uniform
    elif choice == 6:
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / (x + 0.1)  # decreasing
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * x + 0.05 * np.sin(4 * np.pi * x)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s

        init1 = interpolate_sequence(best_seq, best_scan_n)
        a_out, _ = nesterov_lp_optimize(init1, t1, solver)
        try_update(a_out, f"lp_deep_n{best_scan_n}")

    # Phase 2: Try second-best length
    remaining = time_budget - (time.monotonic() - t_global_start)
    if len(scan_results) > 1:
        second_n = sorted(scan_results, key=scan_results.get)[1]
        if second_n != best_scan_n:
            print(f"\n=== Phase 2: LP at n={second_n} (remaining={remaining:.0f}s) ===")
            t2 = min(remaining * 0.20, 300.0)
            if t2 > 30:
                init2 = interpolate_sequence(best_seq, second_n)
                a_out, _ = nesterov_lp_optimize(init2, t2, solver)
                try_update(a_out, f"lp_n{second_n}")

    # Phase 3: Subgradient polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Subgradient polish (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 10:
        a_out, _ = subgradient_polish(best_seq, t3)
        try_update(a_out, f"subgrad_n{len(best_seq)}")

    # Phase 4: LogSumExp refinement
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LogSumExp (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_out, _ = logsumexp_optimize(best_seq, t4, T_start=50.0, T_end=0.1, n_stages=12)
        try_update(a_out, f"lse_n{len(best_seq)}")

    # Phase 5: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak shaving (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.30, 200.0)
    if t5 > 5:
        a_out, _ = peak_shaving(best_seq, t5, seed=456)
        try_update(a_out, f"peak_n{len(best_seq)}")

    # Phase 6: Another subgradient pass
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t6 = min(remaining * 0.5, 150.0)
        if t6 > 5:
            a_out, _ = subgradient_polish(best_seq, t6, lr_start=0.0005)
            try_update(a_out, f"subgrad2_n{len(best_seq)}")

    # Phase 7: Final peak shave with different seed
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t7 = min(remaining * 0.6, 80.0)
        if t7 > 3:
            a_out, _ = peak_shaving(best_seq, t7, seed=789)
            try_update(a_out, f"peak2_n{len(best_seq)}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
