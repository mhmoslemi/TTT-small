            a = a_new

The algorithm:

But actually, Frank-Wolfe is simpler and avoids the projection entirely. Let me go with a refined FW approach.

Key insight: In FW, the step size `gamma_t = 2/(t+2)` is known to be suboptimal in practice. A better approach is to use a "step halving" or "backtracking" line search:
- Start with `gamma = 1`
- Check if `f((1-gamma)*a + gamma*s) <= f(a) - (c)/(2*gamma) * ||a-s||^2` (strong slope condition)
- If not, halve gamma

Or simpler: just check if the objective decreased, and if not, halve the step.

Let me also think about the "away steps" variant of FW (FW-AWF). In away steps, instead of moving towards a new vertex, we can also move away from a "bad" vertex that's already in the support. This can significantly speed up convergence.

Algorithm FW-AWF:
1. Compute gradient g
2. Linear oracle: s = e_argmin(g) (move towards)
3. Away vertex: v = e_argmax(g * 1[support]) (move away from the worst in support)
4. Compute two candidate points:
   - a_towards = (1-γ)*a + γ*s
   - a_away = (a - γ*v) / (1-γ) (for γ < 1)
5. Choose the one with lower objective
6. Use backtracking line search

This is more complex but can converge much faster.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_lse_grad(a, T):
    """Compute LSE gradient and value efficiently using FFT."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    scaled = (b - b_max) / T
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    lse_val = b_max + T * np.log(exp_scaled.sum())

    # = cross_correlation(w, a)
    # w has length 2n-1, a has length n
    # cross_correlation[w,a][m] = sum_j w[j+m] * a[j] for j=0..n-1
    # We want m = 0..n-1
    
    # Using FFT: cross_corr = ifft(flip(w)_fft * a_fft) at appropriate indices
    fft_size = 1 << (3*n - 2 - 1).bit_length()  # need >= (2n-1) + n - 1 = 3n-2
    w_rev = w[::-1]
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv_res = irfft(W * A, fft_size)

    # We want: grad[m] = sum_j w[j+m] * a[j]
    # So 2n-2-p+j = j+m => p = 2n-2-m
    # grad[m] = conv_res[2n-2-m] for m=0..n-1
    # grad[0] = conv_res[2n-2], grad[n-1] = conv_res[n-1]
    grad = conv_res[2*n-2 : n-2 : -1].copy()
    # Wait: m=0 -> p=2n-2, m=n-1 -> p=n-1
    # So grad = conv_res[2n-2, 2n-3, ..., n-1]
    # In numpy: conv_res[n-1 : 2*n-1][::-1] or conv_res[2*n-2 : n-2 : -1]
    
    lse_obj = 2.0 * n * lse_val

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_obj_grad(a, T, fft_size):
    """Compute LSE objective and gradient. a must be nonneg with sum=1."""
    n = len(a)
    size = 2 * n - 1
    A = rfft(a, fft_size)
    b = irfft(A * A, fft_size)[:size]
    b_max = b.max()
    scaled = (b - b_max) / T
    exp_scaled = np.exp(scaled)
    w_sum = exp_scaled.sum()
    w = exp_scaled / w_sum
    lse_val = b_max + T * np.log(w_sum + 1e-300)
    obj = 2.0 * n * lse_val
    # Gradient: grad[i] = 2 * sum_j w[j+i] * a[j]
    # Computed as 2 * conv(w_rev, a)[2n-2-i]
    w_rev = w[::-1]
    W = rfft(w_rev, fft_size)
    conv_res = irfft(W * A, fft_size)
    grad = 2.0 * conv_res[2 * n - 2 : n - 2 : -1].copy()
    return obj, grad, b


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.arange(1, n + 1)
    cond = v_sorted - (cumsum - 1.0) / rho > 0
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.max(rho[cond])
    theta = (cumsum[rho_star - 1] - 1.0) / rho_star
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def lse_fw_backtrack(a0, T_list, iters_per_T, time_limit, verbose=True):
    """LSE Frank-Wolfe with backtracking line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    size = 2 * n - 1
    corr_size = 3 * n - 2
    fft_size = 1 << (max(size, corr_size) - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        for it in range(iters_per_T):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad, b = compute_lse_obj_grad(a, T, fft_size)
            h = np.argmin(grad)

            # Backtracking: start with gamma=1, halve until improvement
            gamma = 1.0
            a_trial = a.copy()
            a_trial -= gamma / n
            a_trial[h] += gamma
            a_trial = np.maximum(a_trial, 0)
            a_trial /= a_trial.sum()

            accepted = False
            for _bt in range(12):
                A_t = rfft(a_trial, fft_size)
                b_t = irfft(A_t * A_t, fft_size)[:size]
                b_t_max = b_t.max()
                scaled_t = (b_t - b_t_max) / T
                exp_t = np.exp(scaled_t)
                obj_t = 2.0 * n * (b_t_max + T * np.log(exp_t.sum() + 1e-300))
                if obj_t < obj:
                    accepted = True
                    break
                gamma *= 0.5
                a_trial = a.copy()
                a_trial -= gamma / n
                a_trial[h] += gamma
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()

            if accepted:
                a = a_trial
            # If not accepted even at tiny gamma, skip this iteration

            if it % 2000 == 0 and it > 0 and verbose:
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                if verbose:
                    print(f"      FW T={T:.5f} it={it} score={score:.8f}")

        # End of temperature: evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose:
                print(f"    FW end T={T:.5f}: best={best_score:.8f}")

    return best_a, best_score


def lse_fista(a0, T_list, iters_per_T, time_limit, verbose=True):
    """LSE FISTA (accelerated proximal gradient) with simplex projection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    size = 2 * n - 1
    corr_size = 3 * n - 2
    fft_size = 1 << (max(size, corr_size) - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        x = a.copy()
        y = a.copy()
        k = 1

        for it in range(iters_per_T):
            if time.monotonic() - t0 > time_limit:
                break
            # Compute gradient at y
            obj_y, grad_y, _ = compute_lse_obj_grad(y, T, fft_size)

            # Gradient step
            step_size = 0.01 * T  # conservative step
            z = y - step_size * grad_y

            # Project onto simplex
            x_new = simplex_project(z)

            # FISTA update
            k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k * k))
            y = x_new + ((k - 1) / k_new) * (x_new - x)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            else:
                y = x_new.copy()
            x = x_new
            k = k_new

            if it % 5000 == 0 and it > 0 and verbose:
                b_x = fast_convolve(x)
                score = 2.0 * n * b_x.max()
                if score < best_score:
                    best_score = score
                    best_a = x.copy()
                if verbose:
                    print(f"      FISTA T={T:.5f} it={it} score={score:.8f}")

        # End of temperature
        score = 2.0 * n * fast_convolve(x).max()
        if score < best_score:
            best_score = score
            best_a = x.copy()
            if verbose:
                print(f"    FISTA end T={T:.5f}: best={best_score:.8f}")
        a = x.copy()

    return best_a, best_score


def subgradient_fw(a0, time_limit, n_iters=500000, verbose=True):
    """Subgradient Frank-Wolfe on the true max objective (hard peaks)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    size = 2 * n - 1
    corr_size = 3 * n - 2
    fft_size = 1 << (max(size, corr_size) - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    global_it = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        global_it += 1
        b = fast_convolve(a)
        max_b = b.max()
        # Active set: positions where b is within 1% of max
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Weighted subgradient
        w_a = b[active] / b[active].sum()
        grad = np.zeros(n)
        for w_k, k_idx in zip(w_a, active):
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            grad[idx] += w_k * 2.0 * a[k_idx - idx]

        h = np.argmin(grad)
        gamma = 2.0 / (global_it + 2)
        a -= gamma / n
        a[h] += gamma
        a = np.maximum(a, 0)
        a /= a.sum()

        if it % 5000 == 0 and it > 0:
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if verbose:
                print(f"      SG-FW it={it} score={score:.8f}")

    score = 2.0 * n * fast_convolve(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = 0.3 + 0.4 * np.cos(2 * np.pi * 7 * x) * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    else:
        a = rng.uniform(0.1, 1.0, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LSE-FW with backtracking, FISTA, and subgradient refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: LSE-FW with backtracking on parent ===
    print(f"\n=== Phase 1: LSE-FW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 450.0)
    if t1 > 30:
        T_list = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        iters_per_T = [5000, 5000, 8000, 10000, 10000, 12000, 12000, 15000, 15000, 20000, 20000, 25000]
        a_fw, _ = lse_fw_backtrack(best_seq, T_list, max(iters_per_T), t1, verbose=True)
        try_update(a_fw, "fw_1")

    # === Phase 2: FISTA refinement ===
    print(f"\n=== Phase 2: FISTA n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 450.0)
    if t2 > 30:
        T_list = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        iters_per_T = [20000, 20000, 25000, 25000, 30000, 30000]
        a_fista, _ = lse_fista(best_seq, T_list, max(iters_per_T), t2, verbose=True)
        try_update(a_fista, "fista_1")

    # === Phase 3: Subgradient FW on true objective ===
    print(f"\n=== Phase 3: Subgrad-FW n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 270.0)
    if t3 > 20:
        a_sg, _ = subgradient_fw(best_seq, t3, n_iters=int(t3 * 1000), verbose=True)
        try_update(a_sg, "sg_fw")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    current_n = len(best_seq)
    for ln in [1000, 1500, 1800]:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        if current_n == ln:
            continue
        t_ln = min(rem * 0.15, 200.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick LSE-FW
        T_list_short = [0.1, 0.05, 0.01, 0.005, 0.001]
        a_ln, _ = lse_fw_backtrack(init_ln, T_list_short, 15000, t_ln, verbose=False)
        try_update(a_ln, f"fw_n{ln}")
        current_n = ln

    # === Phase 5: refined LSE-FW on best ===
    print(f"\n=== Phase 5: Fine LSE-FW n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.2, 300.0)
    if t5 > 30:
        T_list_fine = [0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
        a_fine, _ = lse_fw_backtrack(best_seq, T_list_fine, 30000, t5, verbose=True)
        try_update(a_fine, "fw_fine")

    # === Phase 6: FISTA fine polish ===
    print(f"\n=== Phase 6: FISTA fine n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 200.0)
    if t6 > 20:
        T_list_fista = [0.0001, 0.00005, 0.00002, 0.00001]
        a_fista2, _ = lse_fista(best_seq, T_list_fista, 30000, t6, verbose=True)
        try_update(a_fista2, "fista_fine")

    # === Phase 7: Final subgradient polish ===
    print(f"\n=== Phase 7: Final subgrad n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.2, 200.0)
    if t7 > 15:
        a_sg2, _ = subgradient_fw(best_seq, t7, n_iters=int(t7 * 800), verbose=True)
        try_update(a_sg2, "sg_final")

    # === Phase 8: Try different initializations for best length ===
    print(f"\n=== Phase 8: Multi-start n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monolificent() if hasattr(time, 'monolificent') else deadline - time.monotonic()
    rem = deadline - time.monotonic()
    n_final = len(best_seq)
    for trial in range(3):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        init = structured_init(n_final, rng)
        # Blend with best
        blend = 0.5 * best_seq + 0.5 * init
        blend = np.maximum(blend, 0)
        blend /= blend.sum()
        t_sub = min(rem * 0.3, 120.0)
        if t_sub < 20:
            continue
        T_list_blend = [0.01, 0.005, 0.001, 0.0005]
        a_bl, _ = lse_fw_backtrack(blend, T_list_blend, 20000, t_sub, verbose=False)
        try_update(a_bl, f"blend{trial}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
