def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_grad(a, T):
    """
    Compute gradient of 2*n*LSE(conv(a,a), T) w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    w = exp_terms / exp_terms.sum()
    
    # grad[j] = 2*n * sum_k w[k] * 2 * a[k-j] = 4n * (w cross-correlate a)[j]
    # Wait, let me re-derive.
    # LSE = T * log(sum_k exp(b[k]/T))
    # dLSE/db[k] = w[k] = exp(b[k]/T) / sum exp(b[l]/T)
    # db[k]/da[j] = 2*a[k-j] (for 0 <= k-j < n)
    # dLSE/da[j] = sum_k w[k] * 2*a[k-j]
    # d(2n*LSE)/da[j] = 2n * sum_k w[k] * 2*a[k-j] = 4n * (w ⋆ a)[j]
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)
    grad = 4.0 * n * corr[:n]
    
    return grad


def lse_value(a, T):
    """Compute 2*n*LSE(conv(a,a), T)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    return 2.0 * n * lse


def fista_lse(a0, T, n_iters, time_limit):
    """FISTA (accelerated proximal gradient) on 2n*LSE(conv(a,a), T) over simplex."""
    n = len(a0)
    a = a0.copy()
    y = a.copy()
    t = 1.0
    t0 = time.monotonic()
    L = 4.0 * n  # Lipschitz constant estimate
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Gradient at y
        grad = lse_grad(y, T)
        
        # Gradient step
        step = 1.0 / L
        z = y - step * grad
        
        # Project onto simplex
        a_new = simplex_project(z)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a_new.copy()
        else:
            y = y / sy
        
        a = a_new
        t = t_new
        
        # Track best (using actual max, not LSE)
        if (it + 1) % 100 == 0:
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    # Final check
    score = 2.0 * n * fast_convolve(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def lbgfs_lse(a0, time_limit, T_stages):

            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / T)

            obj = 2.0 * n * lse_val
            
            # Gradient w.r.t. a
            grad_a = lse_grad(a, T)  # this is d(2n*LSE)/da

            # a_i = exp(z_i) / S where S = sum exp(z_k)
            # d a_i / d z_j = a_i * (delta_ij - a_j)
            # So grad_z = a * (grad_a - dot(grad_a, a))

def build_q_sparse(k, n):
    """Build sparse Q matrix for (a*a)[k] = a^T Q_k a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_qcqp_active(a, active_set, solver=None):
    """Solve min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0."""
    n = len(a)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_set:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]
    if solver:
        solvers_to_try.insert(0, solver)
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None
                return a_val / s_sum
        except Exception:
            continue
    return None


def get_solver():
    """Find a working solver."""
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_seq(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi
