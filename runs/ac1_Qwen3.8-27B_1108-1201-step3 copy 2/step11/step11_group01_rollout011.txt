def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_subgradient(y, active):
    n = len(y)
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        g[js:je+1] += y[js:je+1] * 0 + y[k - np.arange(js, je+1)]
    if len(active) > 0:
        g /= len(active)
    return g


def compute_subgradient_fft(y, active_weights):

    total_len = nw + n - 1
    N = 1 << (total_len - 1).bit_length()

    g = np.zeros(n)

    return g


def mirror_descent(a0, time_limit, eta0=2.0, decay=0.0001, delta_init=0.05, delta_end=0.0,
                   momentum_init=0.9, momentum_end=0.999, max_iters=50000000, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    z = np.log(a)
    z_prev = z.copy()
    a_prev = a.copy()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    
    t0 = time.monotonic()
    last_print = t0
    no_improve_count = 0
    patience = 50000
    
    for t in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Momentum in log-domain
        if t == 0:
            mom = momentum_init
        else:
            progress = min(1.0, t / max_iters)
            mom = momentum_init + (momentum_end - momentum_init) * progress
        
        z = z + mom * (z - z_prev)
        
        # Convert to probability
        y = softmax_from_z(z)
        
        # Compute subgradient
        b = fast_convolve(y)
        b_max = b.max()
        
        # Adaptive delta
        progress = min(1.0, t / max_iters)
        delta = delta_init * (1.0 - progress) + delta_end * progress
        
        if delta > 1e-10:
            active = np.where(b >= b_max - delta)[0]
        else:
            active = np.array([int(np.argmax(b))])
        
        if len(active) > 300:
            # Use top-300 if too many active
            top_k = np.argsort(b)[-300:]
            active = top_k
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            g[js:je+1] += y[k - np.arange(js, je+1)]
        g /= max(1, len(active))
        
        # Step size
        eta = eta0 / (1.0 + t * decay)
        
        # Update
        z_prev = z.copy()
        z = z - eta * g
        
        a_prev = a.copy()
        a = softmax_from_z(z)
        
        # Track best
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Restart mechanism: if stuck, reduce momentum
        if no_improve_count == patience:
            z = z + np.random.default_rng(t).normal(0, 0.01, n)
            no_improve_count = 0
        
        if verbose and (t + 1) % 100000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 5:
                print(f"    MD iter={t+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.4f} |A|={len(active)}")
                last_print = elapsed
    
    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
        Tb = max_Tb
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        g[js:je+1] += 2.0 * w[k] * a[k - np.arange(js, je+1)]
    
    S = np.dot(g, a)
    grad_z = a * (g - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=500.0, T_end=0.5, n_stages=20, iters_per_stage=50):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.1f}: score={sc:.8f}")
        except Exception:
            pass
    
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-10)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        # Find worst position and try to reduce it
        worst_k = int(np.argmax(b))
        
        # Try shifting mass from peak area to low area
        js = max(0, worst_k - n + 1)
        je = min(n - 1, worst_k)
        
        # Find positions that contribute most to the peak
        contrib = np.zeros(n)
        contrib[js:je+1] = a[je:js-1:-1] if js == 0 else np.array([a[k-i] for i, k in enumerate(range(js, je+1))])
        
        # Actually let's just do a simple perturb-and-reoptimize
        # Pick a random coordinate to reduce and another to increase
        idx_reduce = np.argmax(np.maximum(a, 0))
        idx_increase = np.argmin(a)
        
        if idx_reduce == idx_increase:
            continue
        
        delta = a[idx_reduce] * 0.01
        a_trial = a.copy()
        a_trial[idx_reduce] -= delta
        a_trial[idx_increase] += delta
        a_trial = np.maximum(a_trial, 0)

        if st < 1e-15:

        a_trial = a_trial / st
        
        sc = score_normalized(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
    
    return best_a, best_score

        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")
    
    # === Phase 1: Mirror Descent (core, ~700s) ===
    print(f"\n=== Phase 1: Mirror Descent at n={n_base} (score={best_score:.8f}) ===")
    t1 = min(700.0, (_global_deadline - time.monotonic()) * 0.45)
    if t1 > 30:
        a_out, sc = mirror_descent(best_seq, t1, eta0=3.0, decay=0.00005,
                                    delta_init=0.03, delta_end=0.0,
                                    momentum_init=0.9, momentum_end=0.998,
                                    max_iters=100000000)
        try_update(a_out, f"MD1 (sc={sc:.8f})")
    print(f"  Phase 1 done: best_score={best_score:.8f}")
    
    # === Phase 2: L-BFGS LogSumExp (polish, ~250s) ===
    print(f"\n=== Phase 2: LSE L-BFGS polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(250.0, remaining * 0.2)
    if t2 > 30:
        a_lse = lbgfs_logsumexp(best_seq, t2, T_start=500.0, T_end=0.5, n_stages=15, iters_per_stage=40)
        try_update(a_lse, "LSE_polish")
    print(f"  Phase 2 done: best_score={best_score:.8f}")
    
    # === Phase 3: Another Mirror Descent round (~250s) ===
    print(f"\n=== Phase 3: MD round 2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(250.0, remaining * 0.2)
    if t3 > 30:
        a_out2, sc2 = mirror_descent(best_seq, t3, eta0=1.0, decay=0.0001,
                                      delta_init=0.01, delta_end=0.0,
                                      momentum_init=0.95, momentum_end=0.999,
                                      max_iters=100000000)
        try_update(a_out2, f"MD2 (sc={sc2:.8f})")
    print(f"  Phase 3 done: best_score={best_score:.8f}")
    
    # === Phase 4: L-BFGS with very high T (~150s) ===
    print(f"\n=== Phase 4: High-T LSE (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(150.0, remaining * 0.15)
    if t4 > 20:

    print(f"  Phase 4 done: best_score={best_score:.8f}")
    
    # === Phase 5: Multi-start with different inits (~150s) ===
    print(f"\n=== Phase 5: Multi-start (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5_total = min(150.0, remaining * 0.15)
    if t5_total > 30:
        inits = [0, 2, 4]  # Different structured inits

        for mi, init_mode in enumerate(inits):

            t5 = min(per_init, remaining * 0.4)
            if t5 < 10:
                break
            print(f"  Multi-start {mi+1}/{len(inits)}: mode={init_mode}, budget={t5:.0f}s")
            init_a = structured_init(n_base, rng, mode=init_mode)
            a_ms, sc_ms = mirror_descent(init_a, t5, eta0=3.0, decay=0.0001,
                                          delta_init=0.05, delta_end=0.0,
                                          momentum_init=0.9, momentum_end=0.995,
                                          max_iters=50000000)
            try_update(a_ms, f"multi_{init_mode} (sc={sc_ms:.8f})")
    print(f"  Phase 5 done: best_score={best_score:.8f}")
    
    # === Phase 6: Final L-BFGS fine-tune (~100s) ===
    print(f"\n=== Phase 6: Final LSE fine-tune (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6 = min(100.0, remaining * 0.3)
    if t6 > 15:
        a_final = lbgfs_logsumexp(best_seq, t6, T_start=2000.0, T_end=5.0, n_stages=8, iters_per_stage=25)

    # === Phase 7: Length exploration (~80s) ===
    print(f"\n=== Phase 7: Length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()

        t8_md = min(remaining * 0.5, 60.0)

        if remaining2 > 10 and t8_lse > 10:

                                           n_stages=5, iters_per_stage=20)

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
