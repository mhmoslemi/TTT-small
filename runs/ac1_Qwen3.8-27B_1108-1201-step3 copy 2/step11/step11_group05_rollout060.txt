def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):

    return T

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def solve_best_response(fixed_a, solver, verbose=False):
    """Given fixed a (nonnegative, sum=1), find b minimizing max conv(a,b)."""
    n = len(fixed_a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(fixed_a)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1.0]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    if s < 1e-15:

    a = a / s
    
    b = a.copy()  # Initialize b = a
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

        # Nesterov step
        alpha = 2.0 / (it + 2)

        a_ext = (1.0 - alpha) * a + alpha * b
        b_ext = (1.0 - alpha) * b + alpha * a
        a_ext = np.maximum(a_ext, 0)

def fast_convolve(a):

    n = len(a)

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n), dtype=np.float64)
    for j in range(n):
        T[j:j+n, j] = a
    return T

            threshold = max_b * (1.0 - 1e-4)
            w = np.zeros(len(b))
            active = np.where(b >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(b)])
            w[active] = 1.0 / len(active)
        else:
            exp_top = np.exp((b_top - max_top) / T)
            w_top = exp_top / exp_top.sum()
            w = np.zeros(len(b))
            w[idx] = w_top
        
        # Gradient of max (approximated by weighted sum)
        # d/da[i] sum_k w[k] conv(a,a)[k] = 2 * sum_k w[k] a[k-i] (for valid k-i)
        g_a = np.zeros(n)
        for i in range(n):
            ks = np.arange(max(0, i), min(2*n-1, i+n))
            js = ks - i
            g_a[i] = 2.0 * np.sum(w[ks] * a[js])
        
        # Convert to z-space gradient
        S = np.dot(g_a, a)
        grad_z = a * (g_a - S)
        
        # Adam update
        lr_t = lr / (1.0 + 0.5 * step / total_steps)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z ** 2
        m_hat = m / (1.0 - beta1 ** (step + 1))
        v_hat = v / (1.0 - beta2 ** (step + 1))
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()  # Centering
        
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            cur = 2.0 * n * fast_convolve(np.exp(z - z.max()) / np.exp(z - z.max()).sum()).max()
            print(f"    Adam step={step} ({elapsed:.0f}s): best={best_score:.8f} cur={cur:.8f} T={T:.4f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Load parent sequence if available
    2. Nesterov LP alternating on parent length (primary)
    3. Nesterov LP alternating on other lengths (exploration)
    4. Adam refinement as polish
    5. Final verification
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 25.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_solver()
    print(f"LP Solver: {solver}, Budget: {budget_s:.0f}s")
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = p / p.sum()
            p_score = evaluate_local(p)
            print(f"Parent: n={len(p)}, score={p_score:.8f}")
            best_seq = p.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_init = 1200
        best_seq = cosine_skew_init(n_init)
        best_score = evaluate_local(best_seq)
        print(f"Generated init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    def update_best(candidate, label):
        nonlocal best_seq, best_score
        if candidate is None:
            return
        candidate = np.maximum(np.asarray(candidate, dtype=np.float64), 0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        sc = evaluate_local(candidate)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = candidate.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (was {old_score:.8f}, n={len(candidate)}) ***")
    
    # === Phase 1: Nesterov LP on parent length (main workhorse) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Nesterov LP on n={n_parent} (current best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 30:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_opt, sc = nesterov_lp_optimize(best_seq, t_alloc, solver, verbose=True)
        update_best(a_opt, f"nlp_n{n_parent}")
    
    # === Phase 2: Nesterov LP with different initialization on same n ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Nesterov LP with ramped init on n={n_parent} (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_ramp = ramped_init(n_parent)
        a_opt, sc = nesterov_lp_optimize(a_ramp, t_alloc, solver, verbose=False)
        update_best(a_opt, f"nlp_ramped_n{n_parent}")
    
    # === Phase 3: Explore different n values ===
    n_candidates = [1100, 1300, 1400, 1500, 1600]
    # Remove the parent n if present
    n_candidates = [x for x in n_candidates if x != n_parent]
    
    for n_try in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        
        # Create initial sequence for this n
        if len(best_seq) > 0:
            init = interpolate_seq(best_seq, n_try)
        else:
            init = cosine_skew_init(n_try)
        
        t_alloc = min(rem * 0.12, 180.0)
        if t_alloc < 20:
            break
        
        print(f"\n{'='*60}")
        print(f"Phase 3: Nesterov LP on n={n_try} (best={best_score:.8f})")
        print(f"{'='*60}")
        print(f"  Time budget: {t_alloc:.0f}s")
        a_opt, sc = nesterov_lp_optimize(init, t_alloc, solver, verbose=False)
        update_best(a_opt, f"nlp_n{n_try}")
    
    # === Phase 4: Adam refinement on best ===
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_seq)
        t_alloc = min(rem * 0.20, 200.0)
        print(f"\n{'='*60}")
        print(f"Phase 4: Adam refinement on n={n_cur} (best={best_score:.8f})")
        print(f"{'='*60}")
        print(f"  Time budget: {t_alloc:.0f}s")
        a_opt, sc = adam_refine(best_seq, t_alloc, T_init=3.0, T_final=0.01, lr=0.003)
        update_best(a_opt, f"adam_n{n_cur}")
    
    # === Phase 5: Second LP polish on best ===
    rem = deadline - time.monotonic()
    if rem > 30:
        n_cur = len(best_seq)
        t_alloc = min(rem * 0.25, 150.0)
        print(f"\n{'='*60}")
        print(f"Phase 5: Final LP polish on n={n_cur} (best={best_score:.8f})")
        print(f"{'='*60}")
        print(f"  Time budget: {t_alloc:.0f}s")
        a_opt, sc = nesterov_lp_optimize(best_seq, t_alloc, solver, verbose=True)
        update_best(a_opt, f"nlp_final_n{n_cur}")
    
    # === Phase 6: Last chance improvements ===
    rem = deadline - time.monotonic()
    if rem > 20:
        n_cur = len(best_seq)
        t_alloc = min(rem * 0.4, 40.0)
        # Try a quick Adam with very low temperature
        a_opt, sc = adam_refine(best_seq, t_alloc, T_init=0.5, T_final=0.001, lr=0.001)
        update_best(a_opt, "adam_final")
        
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, sc = nesterov_lp_optimize(best_seq, min(rem * 0.6, 20.0), solver, verbose=False)
            update_best(a_opt, "nlp_ultimate")
    
    # Final verification
    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0
    
    final_local = evaluate_local(best_seq)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: local_score={final_local:.8f}, n={len(best_seq)}")
    print(f"{'='*60}")
    
    # Call the actual evaluator for verification
    try:
        eval_result = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"evaluate_sequence result: {eval_result:.8f}")
    except Exception as e:
        print(f"Evaluation error: {e}")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
