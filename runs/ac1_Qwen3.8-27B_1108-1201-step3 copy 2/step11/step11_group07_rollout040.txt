    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_cross_optimize(a, mixed=False, alpha=1.0):
    """
    Given a (normalized, sum=1), find b minimizing max(conv(a,b))
    subject to sum(b)=1, b>=0.
    Returns b.
    """
    n = len(a)

    # Use sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

        cols.append(n)
        vals.append(-1.0)

    A_eq[0, :n] = 1.0

    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Bounds
    bounds = [(0, None)] * (n+1)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based cross-convolution minimization with alternating direction
2. Subgradient peak shaving (fast O(n) per step approximation)
3. LSE-smoothed L-BFGS with adaptive temperature annealing
4. Spectral domain optimization (power spectrum flattening)
5. Multi-start with diverse initializations and length exploration
Key innovation: LP alternating direction to escape local minima, combined
with gradient-based polynomial-time refinement.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix, csc_matrix


def fast_auto_conv(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Compute objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_auto_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_lp_structure(n):
    """Precompute LP constraint matrix structure."""
    all_rows = []
    all_cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            all_rows.append(k)
            all_cols.append(j)
        all_rows.append(k)
        all_cols.append(n)
    return np.array(all_rows, dtype=np.int32), np.array(all_cols, dtype=np.int32)


def solve_lp_cross(a, rows, cols, n):
    """Solve LP: min t s.t. (a*b)[k] <= t, sum(b)=1, b>=0."""
    vals = np.empty(len(rows), dtype=np.float64)
    b_mask = cols < n
    vals[b_mask] = a[rows[b_mask].astype(np.int64) - cols[b_mask].astype(np.int64)]
    vals[~b_mask] = -1.0

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0

    bounds = [(0.0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

    if res.status != 0:
        return None
    b = res.x[:n]
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        return None
    return b / s


def lp_alternating(a0, time_limit, max_iters=5000, accept_threshold=0.999):
    """LP-based alternating minimization.
    Iterates: solve LP to get b, check if self-conv improves, accept or blend.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    rows, cols = build_lp_structure(n)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if (time.monotonic() - t0) % 50 < 0.1 and it % 500 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}")

        b = solve_lp_cross(a, rows, cols, n)
        if b is None:
            break

        sc_b = evaluate_local(b)
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()
            a = b.copy()
        else:
            # Try convex combinations
            improved = False
            for alpha in [0.7, 0.5, 0.3, 0.1]:
                a_try = normalize((1 - alpha) * a + alpha * b)
                sc_try = evaluate_local(a_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
                if sc_try < evaluate_local(a):
                    a = a_try.copy()
            if not improved:
                # Check convergence
                if np.max(np.abs(b - a)) < 1e-8:
                    break

    return best_a, best_score


def subgradient_peak_shave(a0, time_limit, n_iters=100000):
    """Fast subgradient-based peak shaving for cross-convolution.
    Uses the subgradient of max_k (a*b)[k] w.r.t. b to iteratively reduce peaks.
    """
    n = len(a0)
    a = normalize(a0.copy())
    b = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = 0.1

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 10000 == 0 and it > 0:
            sc = evaluate_local(b)
            if sc < best_score:
                best_score = sc
                best_a = b.copy()
            eta *= 0.95

        # Compute cross-conv a * b
        c = fast_cross_conv(a, b)
        k_star = int(np.argmax(c))

        # Subgradient w.r.t. b at the peak
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        for j in range(j_start, j_end + 1):
            sg[j] = a[k_star - j]

        # Update with diminishing step
        rate = eta / (1 + 0.01 * it)
        b = b - rate * sg
        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-15:
            b = a.copy()
        else:
            b = b / s

    sc = evaluate_local(b)
    if sc < best_score:
        best_score = sc
        best_a = b.copy()

    return best_a, best_score


def lse_objective_grad(a, T):
    """LSE-smoothed objective and gradient for normalized a (sum=1, a>=0)."""
    n = len(a)
    c = fast_auto_conv(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum() + 1e-300)
    obj = 2.0 * n * lse

    w = exp_c / (exp_c.sum() + 1e-300)

    # Gradient via cross-correlation
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    b_corr = R[:n]

    g = 2.0 * b_corr
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]

    return obj, g


def lbgfs_lse(a0, time_limit, T_stages, maxiter_per_stage=2000):
    """L-BFGS with LSE smoothing and temperature annealing."""
    n = len(a0)
    a = normalize(a0.copy())
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    def func_z(z_in):
        z_s = z_in - z_in.max()
        exp_z = np.exp(z_s)
        a_local = exp_z / exp_z.sum()
        obj, grad_a = lse_objective_grad(a_local, T_stages[0])
        S = np.dot(grad_a, a_local)
        grad_z = a_local * (grad_a - S)
        return obj, grad_z

    current_T = T_stages[0]

    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        current_T = T

        def func(z_in, T=current_T):
            obj, _ = func_z(z_in)
            return obj

        def jac(z_in, T=current_T):
            _, g = func_z(z_in)
            return g

        iters = min(maxiter_per_stage, max(20, int(remaining * 20)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Update best from current z
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a_new = exp_z / exp_z.sum()
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new / s

    return a


def spectral_optimize(a0, time_limit, n_iters=5000):
    """Flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    lr = 0.003

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean() + 1e-15
        r = (P - P_mean) / P_mean

        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -r[valid] * P_mean * A[valid] / (P[valid] + 1e-10)
        H[0] = 0.0
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        for scale in [lr, lr * 0.3, lr * 0.1]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            break

        if it % 500 == 0:
            lr *= 0.8

    return best_a, best_score


def lse_mirror_descent(a0, time_limit, n_steps=50000, T0=10.0):
    """Mirror descent (multiplicative weights) with LSE gradient."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    T = T0

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal temperature
        T = T0 * (0.999 ** it)
        T = max(T, 0.01)

        obj, g = lse_objective_grad(a, T)
        S = np.dot(g, a)
        grad = g - S

        # Multiplicative update
        eta = 0.5 / (np.max(np.abs(grad)) + 1e-10)
        log_update = -eta * grad
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if it % 10000 == 0 and it > 0:
            print(f"    MirrorDescent iter {it}: T={T:.4f}, score={best_score:.8f}")

    return best_a, best_score


def diverse_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    candidates = []
    x = np.linspace(0, 1, n)

    a = np.exp(-8 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))

    a = (x * (1 - x)) ** 0.7
    candidates.append(normalize(a))

    a = np.exp(-12 * (x - 0.25) ** 2) + np.exp(-12 * (x - 0.75) ** 2)
    candidates.append(normalize(a))

    a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    # Random Fourier
    coeffs = rng.normal(0, 1, 15)
    phases = rng.uniform(0, 2 * np.pi, 15)
    a = np.zeros(n)
    for i in range(15):
        a += coeffs[i] * np.cos(2 * np.pi * (i + 1) * x + phases[i])
    a = np.maximum(a + 0.8, 0)
    candidates.append(normalize(a))

    candidates.append(np.ones(n) / n)

    a = np.exp(-3 * x) + np.exp(-3 * (1 - x))
    candidates.append(normalize(a))

    # Power law
    a = x ** 2 + (1 - x) ** 2
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, strength=0.03):
    """Random perturbation."""
    n = len(a)
    method = rng.integers(0, 4)
    if method == 0:
        a_new = a + rng.normal(0, strength, n)
    elif method == 1:
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        a_new = a + strength * np.cos(freq * x + phase)
    elif method == 2:
        a_new = a.copy()
        bs = rng.integers(5, 30)
        i = rng.integers(0, max(1, n - bs))
        j = rng.integers(0, max(1, n - bs))
        if abs(i - j) > bs:
            a_new[i:i + bs], a_new[j:j + bs] = a_new[j:j + bs].copy(), a_new[i:i + bs].copy()
    else:
        a_new = a * np.exp(rng.normal(0, strength * 0.5, n))
    return normalize(a_new)


def population_refine(a_init, time_limit, pop_size=6, rng=None):
    """Lightweight population-based search."""
    if rng is None:
        rng = np.random.default_rng(789)
    n = len(a_init)
    population = [normalize(a_init.copy())]
    for i in range(pop_size - 1):
        population.append(perturb_sequence(a_init, rng, 0.02 + 0.01 * i))

    scores = [evaluate_local(p) for p in population]
    best_a = population[int(np.argmin(scores))].copy()
    best_score = scores[int(np.argmin(scores))]
    t0 = time.monotonic()
    gen = 0

    while time.monotonic() - t0 < time_limit:
        gen += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        keep = max(2, pop_size // 2)
        new_pop = list(population[:keep])

        for i in range(pop_size - len(new_pop)):
            parent = population[rng.integers(0, keep)]
            child = perturb_sequence(parent, rng, 0.015 + 0.01 * np.sin(gen * 0.05))
            new_pop.append(child)

        if len(new_pop) < pop_size and len(population) >= 2:
            t = rng.uniform(0.3, 0.7)
            child = normalize(t * population[0] + (1 - t) * population[1])
            new_pop.append(child)

        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating direction, subgradient peak shaving, LSE+L-BFGS,
    spectral shaping, and population-based search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_initializations(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = int(np.argmin(scores))
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"No parent, best init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: LP Alternating Direction (new) ===
    print(f"\n=== Phase 1: LP Alternating (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t_alloc, max_iters=10000)
        try_update(a_lp, "LP_alt")

    # === Phase 2: Subgradient Peak Shaving (new) ===
    print(f"\n=== Phase 2: Subgradient Peak Shave (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_peak_shave(best_seq, t_alloc, n_iters=200000)
        try_update(a_sg, "subgrad")

    # === Phase 3: LSE + L-BFGS (proven, with longer schedule) ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 450.0)
    if t_alloc > 30:
        T_stages = [100, 50, 30, 20, 10, 7, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003]
        print(f"  {len(T_stages)} T-stages, budget={t_alloc:.0f}s")
        a_lb = lbgfs_lse(best_seq, t_alloc, T_stages, maxiter_per_stage=2000)
        try_update(a_lb, "LSE_LBFGS")

    # === Phase 4: Spectral Shaping ===
    print(f"\n=== Phase 4: Spectral (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_spec, _ = spectral_optimize(best_seq, t_alloc, n_iters=10000)
        try_update(a_spec, "spectral")

    # === Phase 5: LSE Mirror Descent ===
    print(f"\n=== Phase 5: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_md, _ = lse_mirror_descent(best_seq, t_alloc, n_steps=100000, T0=20.0)
        try_update(a_md, "mirror_desc")

    # === Phase 6: LP again from improved point ===
    print(f"\n=== Phase 6: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_lp2, _ = lp_alternating(best_seq, t_alloc, max_iters=5000)
        try_update(a_lp2, "LP_r2")

    # === Phase 7: Fine LSE polish ===
    print(f"\n=== Phase 7: Fine Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        T_fine = [2, 1, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        a_fin = lbgfs_lse(best_seq, t_alloc, T_fine, maxiter_per_stage=1500)
        try_update(a_fin, "fine_LSE")

    # === Phase 8: Population search from best ===
    print(f"\n=== Phase 8: Population (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_pop, _ = population_refine(best_seq, t_alloc, pop_size=8, rng=rng)
        try_update(a_pop, "population")

    # === Phase 9: Length exploration ===
    print(f"\n=== Phase 9: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 200, n_best - 200, n_best + 400, n_best + 100]:
            if n_try < 200 or n_try > 4000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.25, 60.0)
            T_short = [20, 10, 5, 3, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
            a_try = lbgfs_lse(init, t_sub, T_short, maxiter_per_stage=1000)
            try_update(a_try, f"len_{n_try}")

    # === Phase 10: Final combined polish ===
    print(f"\n=== Phase 10: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Quick LP
        t_sub = min(rem * 0.3, 30.0)
        a_flp, _ = lp_alternating(best_seq, t_sub, max_iters=2000)
        try_update(a_flp, "final_LP")

        # Quick LSE
        rem = deadline - time.monotonic()
        if rem > 10:
            t_sub = min(rem * 0.5, 30.0)
            T_ult = [1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005, 0.002, 0.001]
            a_fse = lbgfs_lse(best_seq, t_sub, T_ult, maxiter_per_stage=1000)
            try_update(a_fse, "final_LSE")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
