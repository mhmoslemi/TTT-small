import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break

            continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

    obj, _ = logsumexp_obj_grad(z, n, T)  # This won't work as a closure
    return obj

    if time.monotonic() > deadline:

                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': maxiter_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def multiplicative_weights_adaptive(a0, time_limit, n_steps, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.3)
        else:
            eta *= 0.85
            if eta < 1e-9:
                break
        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    lp_times = []
    T_mat = build_conv_matrix_sparse(y)
    t_nesterov = 0.0
    a_prev = a.copy()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            continue
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 10.0:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break

        t_nesterov = (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov ** 2)) / 2.0
        alpha = (1.0 + t_nesterov - 1.0) / t_nesterov

        # Update y using Nesterov extrapolation
        y_old = y.copy()
        y = a + ((t_nesterov - 1.0) / t_nesterov) * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Update a
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        a_prev = a
        a = a_new
        T_mat = build_conv_matrix_sparse(y)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                a = y.copy()
                a_prev = y.copy()
                T_mat = build_conv_matrix_sparse(y)
                t_nesterov = 0.0
                stall_count = 0
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = 0.4 + 0.3 * np.cos(2 * np.pi * 7 * x) + 0.2 * np.cos(2 * np.pi * 13 * x + 1.0)
        a = np.maximum(a, 0)
    else:
        a = (1 + np.cos(2 * np.pi * x)) * np.exp(-2 * x * (1 - x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_palindromic(a):
    """Make a sequence palindromic (a[i] = a[n-1-i])."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def multiplicative_weights_full(a0, time_limit, n_steps, eta0=0.05):
    """Full multiplicative weights (same as adaptive but with more detail)."""
    return multiplicative_weights_adaptive(a0, time_limit, n_steps, eta0)


def lse_refine_fast(a0, time_limit, T_stages):
    """Faster LSE refinement with fewer iterations per stage."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for i, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = (time_limit - (time.monotonic() - t0)) * 0.9
        if remaining < 2:
            break
        func_args = (n, stage_T, deadline)
        maxiter = max(50, int(remaining * 20))
        maxiter = min(maxiter, 2000)
        try:
            res = minimize(
                fun=lse_objective_wrapper, x0=z, args=func_args,
                jac=lse_gradient_wrapper,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-22,
                         'gtol': 1e-15, 'maxfun': maxiter * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
        a_check = softmax_from_z(z)
        sc = evaluate_local(a_check)
        if (i + 1) % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {i+1}/{len(T_stages)} T={stage_T:.4f} ({elapsed:.0f}s): score={sc:.8f}")
    return softmax_from_z(z)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive MW, LP alternating with Nesterov, aggressive LSE L-BFGS,
    multi-length exploration, and structured perturbation escape.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Fast MW on parent (5% time) ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.05, 60.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights_adaptive(best_seq, t1, n_steps=500000, eta0=0.05)
        try_update(a_mw, "mw_1")

    # === Phase 2: LP alternating (40% time, main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.40, 600.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, n_max_iters=1000000)
        try_update(a_lp, "lp_1")

    # === Phase 3: Palindromic LP (10% time) ===
    print(f"\n=== Phase 3: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.10, 150.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = nesterov_lp_refine(a_pal, t3, solver, n_max_iters=200000)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 4: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1500, 1800]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.15, 180.0)
        if t_ln < 30:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW first
        a_ln_mw, _ = multiplicative_weights_adaptive(init_ln, t_ln * 0.2, n_steps=50000, eta0=0.05)
        try_update(a_ln_mw, f"mw_n{ln}")
        # Then LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.5, t_ln * 0.65)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw if a_ln_mw is not None else init_ln, t_lp, solver, n_max_iters=200000)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 5: Aggressive LSE L-BFGS (20% time) ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t5 = min(rem * 0.20, 300.0)
        # Very aggressive cooling schedule
        T_stages = [20.0, 10.0, 5.0, 2.5, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lse_refine_fast(best_seq, t5, T_stages)
        try_update(a_lse, "lse_fine")

    # === Phase 6: Perturbation escape (10% time) ===
    print(f"\n=== Phase 6: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            strength = 0.01 + 0.008 * trial
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem * 0.25, 50.0)
            # Quick MW then LP
            a_pert_mw, _ = multiplicative_weights_adaptive(a_pert, t_sub * 0.3, n_steps=30000, eta0=0.04)
            rem = deadline - time.monotonic()
            t_sub2 = min(rem * 0.4, t_sub * 0.6)
            if t_sub2 > 15:
                a_opt, _ = nesterov_lp_refine(a_pert_mw, t_sub2, solver, n_max_iters=100000)
                try_update(a_opt, f"pert{trial}")
            else:
                try_update(a_pert_mw, f"pert{trial}_mw")

    # === Phase 7: Final LP polish (5% time) ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.3, 60.0)
        a_fin_lp, _ = nesterov_lp_refine(best_seq, t7, solver, n_max_iters=100000)
        try_update(a_fin_lp, "final_lp")

    # === Phase 8: Final LSE micro-polish ===
    print(f"\n=== Phase 8: Final LSE polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = min(rem * 0.5, 80.0)
        T_final = [0.01, 0.005, 0.002, 0.001, 0.0005]
        a_fin_lse = lse_refine_fast(best_seq, t8, T_final)
        try_update(a_fin_lse, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
