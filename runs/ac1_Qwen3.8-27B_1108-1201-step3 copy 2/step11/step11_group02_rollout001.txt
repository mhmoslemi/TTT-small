Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated L-BFGS in log-space.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, size)
    return irfft(A * A, size)[:2 * n - 1]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def evaluate_local(a):
    """Evaluate the objective (scale-invariant, assume sum=1 for simplicity)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def logsumexp_obj_grad_x(x, T):
    """
    Compute objective and gradient of (2n/T)*LogSumExp(conv(softmax(x),softmax(x))/T) w.r.t. x.
    Uses FFT for O(n log n) gradient computation.
    """
    a = softmax(x)
    n = len(a)
    c = fast_convolve(a)

    # Compute weights w[k] = exp(c[k]/T) / sum exp(c[j]/T)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()

    # Compute g[i] = sum_{j=0}^{n-1} w[i+j] * a[j] via FFT cross-correlation
    fft_size = 1 << (3 * n).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g = irfft(W * np.conj(A), fft_size)[:n]

    # Gradient in x-space: grad_x = (4n/T) * a * (g - dot(g,a))
    dot_ga = float(np.dot(g, a))
    grad_x = (4.0 * n / T) * a * (g - dot_ga)

    # Objective value
    logsumexp_val = c_max / T + np.log(exp_c.sum())
    obj = (2.0 * n / T) * logsumexp_val

        threshold = max_c - 0.01 * max_c
        active = np.where(c >= threshold)[0]

        a = a / st

            best_score = score

            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
            return True
        return False

    # Temperature schedules (geometric)
    T_full = [200.0, 100.0, 50.0, 25.0, 12.5, 6.25, 3.125, 1.56, 0.78, 0.39, 0.2, 0.1, 0.05, 0.02, 0.01]
    T_quick = [100.0, 50.0, 25.0, 12.5, 6.25, 3.125, 1.56, 0.78, 0.39, 0.2]
    T_deep = [50.0, 25.0, 12.5, 6.25, 3.125, 1.56, 0.78, 0.39, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    # === Phase 1: L-BFGS on parent with full annealing ===
    print(f"\n=== Phase 1: L-BFGS on parent (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    rem = deadline - t0
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 30:
        n = len(best_seq)
        x0 = np.log(np.maximum(best_seq, 1e-15))
        print(f"  n={n}, budget={t_alloc:.0f}s")
        x_best, sc = lbgfs_logspace(x0, T_full, maxiter_per_stage=200,
                                     time_limit=t_alloc, t_start=t0)
        a_result = softmax(x_best)
        try_update(a_result, "lp_parent")

    # === Phase 2: Aggressive multi-restart ===
    print(f"\n=== Phase 2: Multi-restart (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    rem = deadline - t0
    t_alloc_phase2 = min(rem * 0.45, 700.0)
    if t_alloc_phase2 > 60:
        n = len(best_seq)
        x_parent = np.log(np.maximum(best_seq, 1e-15))

        # Generate diverse initializations
        inits = []
        # Parent perturbations
        for _ in range(10):
            noise = rng.standard_normal(n) * 0.1
            x_pert = x_parent + noise
            inits.append(x_pert)
        # Large perturbations
        for _ in range(5):
            noise = rng.standard_normal(n) * 0.5
            x_pert = x_parent + noise
            inits.append(x_pert)
        # Structured inits
        for c in range(8):
            inits.append(structured_init_logspace(n, rng, choice=c))
        # Random
        for _ in range(5):
            inits.append(rng.standard_normal(n) * 0.5)
        # Cosine with different frequencies
        x_lin = np.linspace(0, 1, n)
        for freq in [1, 2, 3, 5, 7, 10]:
            a_struct = 0.3 + 0.7 * np.cos(np.pi * freq * x_lin) ** 2
            a_struct = np.maximum(a_struct, 1e-10)
            inits.append(np.log(a_struct))

        print(f"  {len(inits)} initializations, total budget={t_alloc_phase2:.0f}s")
        per_init = t_alloc_phase2 / max(10, len(inits))
        per_init = max(3.0, per_init)

        candidates = []
        for i, x_init in enumerate(inits):
            if time.monotonic() > deadline - 10:
                break
            t_sub = min(per_init, deadline - time.monotonic() - 5)
            if t_sub < 3:
                break
            try:
                x_best_i, sc_i = lbgfs_logspace(x_init, T_quick, maxiter_per_stage=150,
                                                 time_limit=t_sub, t_start=time.monotonic())
                a_i = softmax(x_best_i)
                sc_true = 2.0 * n * fast_convolve(a_i).max()
                candidates.append((sc_true, a_i, x_best_i))
                try_update(a_i, f"restart_{i}")
            except Exception:
                pass

            if (i + 1) % 10 == 0:
                elapsed = time.monotonic() - t0
                print(f"    {i+1}/{len(inits)} done ({elapsed:.0f}s), best={best_score:.8f}")

        # Sort candidates and refine top ones
        candidates.sort(key=lambda x: x[0])
        top_k = min(10, len(candidates))
        if top_k > 0:
            rem2 = deadline - time.monotonic()
            t_refine_each = max(5.0, (rem2 * 0.30) / max(1, top_k))
            print(f"  Refining top {top_k} candidates")
            for j in range(top_k):
                if time.monotonic() > deadline - 10:
                    break
                sc_j, a_j, x_j = candidates[j]
                t_sub = min(t_refine_each, deadline - time.monotonic() - 5)
                if t_sub < 3:
                    break
                try:
                    x_best_j, _ = lbgfs_logspace(x_j, T_deep, maxiter_per_stage=200,
                                                  time_limit=t_sub, t_start=time.monotonic())
                    a_refined = softmax(x_best_j)
                    try_update(a_refined, f"refine_{j}")
                except Exception:
                    pass

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n exploration (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    rem = deadline - t0
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        # Try nearby n values and some very different ones
        candidate_ns = sorted(set([
            max(200, n_best - 100),
            max(200, n_best - 50),
            n_best + 50,
            n_best + 100,
            n_best + 200,
            n_best + 500,
            n_best * 2,
            max(200, n_best // 2),
        ]))
        # Filter: not too large, not equal to best
        candidate_ns = [n for n in candidate_ns if n != n_best and 100 <= n <= 5000]
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  n values: {candidate_ns}, {per_n:.0f}s each")

        for n_try in candidate_ns:
            if time.monotonic() > deadline - 10:
                break
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 5:
                continue
            try:
                x_interp = interpolate_logspace(np.log(np.maximum(best_seq, 1e-15)), n_try)
                x_best_n, _ = lbgfs_logspace(x_interp, T_quick, maxiter_per_stage=150,
                                              time_limit=t_sub, t_start=time.monotonic())
                a_n = softmax(x_best_n)
                try_update(a_n, f"n_{n_try}")
            except Exception:
                pass

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coord Descent Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 150.0)
    if t_alloc > 10:
        a_cd = coordinate_descent_polish(best_seq, t_alloc, n_steps=100000)
        try_update(a_cd, "coord_desc")

    # === Phase 5: Subgradient descent ===
    print(f"\n=== Phase 5: Subgradient Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 5:
        a_sg = subgradient_descent(best_seq, t_alloc, n_iters=10000)
        try_update(a_sg, "subgrad")

    # === Phase 6: Final deep L-BFGS ===
    print(f"\n=== Phase 6: Final Deep L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 5:
        x0 = np.log(np.maximum(best_seq, 1e-15))
        x_final, _ = lbgfs_logspace(x0, T_deep, maxiter_per_stage=300,
                                     time_limit=t_alloc, t_start=time.monotonic())
        a_final = softmax(x_final)
        try_update(a_final, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
