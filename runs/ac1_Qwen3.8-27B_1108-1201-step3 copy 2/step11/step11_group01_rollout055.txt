def fast_convolve(a):
    """Fast autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max_b for normalized a (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)

        return float('inf')


def softmax_from_z(z):
    """Convert log-space to probability simplex."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert simplex to log-space."""
    return np.log(np.maximum(a, 1e-300))

    """Build matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, restart_every=500):
    """Nesterov-accelerated LP best response with periodic restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    momentum = 0.0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov step with periodic restart
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it % restart_every) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """LogSumExp objective and gradient in log-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        Tb = Tb - max_Tb + 700
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient with respect to a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=15):
    """Multi-stage LogSumExp L-BFGS optimization."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 30

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Use a closure-free approach
        z_curr = z
        for _ in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad = lse_objective_grad(z_curr, stage_T, n)
            # Simple gradient step as fallback if L-BFGS is slow
            # Actually let's use scipy minimize properly
            break  # break inner loop, use scipy below
        
        # Use scipy L-BFGS-B
        try:
            res = minimize(
                fun=lambda z_val: lse_objective_grad(z_val, stage_T, n)[0],
                x0=z,
                jac=lambda z_val: lse_objective_grad(z_val, stage_T, n)[1],
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

    return best_a, best_score


def lse_objective(z_val, T_val, n):
    a = softmax_from_z(z_val)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lse_gradient(z_val, T_val, n):
    a = softmax_from_z(z_val)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def lse_func_wrapper(z_val, T_val, n):
    return lse_objective(z_val, T_val, n)


def lse_jac_wrapper(z_val, T_val, n):
    return lse_gradient(z_val, T_val, n)


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 50

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        try:
            res = minimize(
                fun=lse_func_wrapper,
                x0=z,
                args=(stage_T, n),
                jac=lse_jac_wrapper,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)


def peak_local_search(a0, time_limit, n_trials=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b[:n]))
        M = b[peak_k]

        n_perturb = rng.integers(1, 4)
        a_new = a.copy()
        for _ in range(n_perturb):
            i = rng.integers(0, n)
            j = min(n - 1, i + peak_k)
            delta = rng.uniform(0.001, 0.02)
            if a[i] > delta:
                a_new[i] -= delta
                a_new[j] += delta
            elif a[j] > delta:
                a_new[j] -= delta
                a_new[i] += delta

        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=200000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        eta = 0.3 * (0.005 / 0.3) ** (step / max(1, n_steps))
        z = z - eta * grad_z
        a = softmax_from_z(z)

        sc_cur = score_normalized(a)
        if sc_cur < best_score:
            best_score = sc_cur
            best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    else:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_update_state(state, a, label):
    if a is None:
        return
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    state = {'best_seq': None, 'best_score': float('inf')}
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            state['best_seq'] = p / p.sum()
            n_base = len(state['best_seq'])
            state['best_score'] = evaluate_safe(state['best_seq'])
            print(f"Parent: n={n_base}, score={state['best_score']:.8f}")
    except Exception:
        state['best_seq'] = structured_init(n_base, rng)
        state['best_score'] = evaluate_safe(state['best_seq'])
        print(f"Default init: n={n_base}, score={state['best_score']:.8f}")

    # === Phase 1: Main Nesterov LP optimization (65% of budget) ===
    total_time = float(budget_s) - 20.0
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={state['best_score']:.8f}) ===")
    t1 = 0.65 * total_time
    remaining = deadline - time.monotonic()
    t1 = min(t1, remaining)
    a_out, _ = nesterov_lp(state['best_seq'], t1, solver=solver, restart_every=400)
    try_update_state(state, a_out, "nesterov_main")
    print(f"  Phase 1 done: best_score={state['best_score']:.8f}")

    # === Phase 2: LogSumExp L-BFGS polishing (15% of budget) ===
    print(f"\n=== Phase 2: LSE L-BFGS polish (score={state['best_score']:.8f}) ===")
    remaining = deadline - time.monotonic()
    t2 = min(0.15 * total_time, remaining)
    if t2 > 15:
        a_lse = lbgfs_logsumexp(state['best_seq'], t2, T_start=50.0, T_end=0.02, n_stages=20)
        try_update_state(state, a_lse, "lse_polish")
    print(f"  Phase 2 done: best_score={state['best_score']:.8f}")

    # === Phase 3: Nesterov LP round 2 (10% of budget) ===
    print(f"\n=== Phase 3: Nesterov LP round 2 (score={state['best_score']:.8f}) ===")
    remaining = deadline - time.monotonic()
    t3 = min(0.10 * total_time, remaining)
    if t3 > 10:
        a_out2, _ = nesterov_lp(state['best_seq'], t3, solver=solver, restart_every=200)
        try_update_state(state, a_out2, "nesterov_r2")
    print(f"  Phase 3 done: best_score={state['best_score']:.8f}")

    # === Phase 4: Peak local search (5% of budget) ===
    print(f"\n=== Phase 4: Peak local search (score={state['best_score']:.8f}) ===")
    remaining = deadline - time.monotonic()
    t4 = min(0.05 * total_time, remaining)
    if t4 > 8:
        a_peak, _ = peak_local_search(state['best_seq'], t4, n_trials=100000)
        try_update_state(state, a_peak, "peak_search")
    print(f"  Phase 4 done: best_score={state['best_score']:.8f}")

    # === Phase 5: Subgradient polish (3% of budget) ===
    print(f"\n=== Phase 5: Subgradient polish (score={state['best_score']:.8f}) ===")
    remaining = deadline - time.monotonic()
    t5 = min(0.03 * total_time, remaining)
    if t5 > 5:
        a_sg, _ = subgradient_polish(state['best_seq'], t5, n_steps=100000)
        try_update_state(state, a_sg, "subgrad")
    print(f"  Phase 5 done: best_score={state['best_score']:.8f}")

    # === Phase 6: Final combined round (2% of budget) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Phase 6: Final refinement (score={state['best_score']:.8f}) ===")
        t6_lp = min(remaining * 0.5, 60.0)
        if t6_lp > 5:
            a_final, _ = nesterov_lp(state['best_seq'], t6_lp, solver=solver, restart_every=100)
            try_update_state(state, a_final, "final_nesterov")
        remaining = deadline - time.monotonic()
        if remaining > 5:
            t6_peak = min(remaining * 0.6, 30.0)
            if t6_peak > 3:
                a_fpeak, _ = peak_local_search(state['best_seq'], t6_peak, n_trials=50000)
                try_update_state(state, a_fpeak, "final_peak")
        print(f"  Phase 6 done: best_score={state['best_score']:.8f}")

    # Final verification
    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
