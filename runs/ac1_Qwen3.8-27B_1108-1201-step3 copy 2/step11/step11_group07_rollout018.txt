import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next_power_of_2 for efficiency
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr(w, a):
    """Compute cross-correlation R[j] = sum_m w[j+m]*a[m] for j=0..len(a)-1.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    R = irfft(W * np.conj(A), L)[:n]
    return R

def evaluate_local(a):
    """Evaluate the exact score for a non-negative sequence."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    c = fast_convolve(a)
    return float(2.0 * n * c.max())

def lse_score_grad(a, T):
    """Compute LSE-smoothed score and gradient for normalized a (sum=1, a>=0).
    Returns (lse_score, gradient) where lse_score = 2n * LSE_T(conv(a,a)).
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    exp_sum = exp_c.sum()
    w = exp_c / exp_sum  # softmax weights
    lse = c_max + T * np.log(exp_sum)
    score = 2.0 * n * lse
    
    # Gradient: d(score)/da[j] = 2n * (2*R[j] - w[2j]*a[j] if 2j < 2n-1)
    R = cross_corr(w, a)
    grad = 2.0 * R
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    grad[valid] -= w[idx[valid]] * a[valid]
    grad *= 2.0 * n
    
    return score, grad

def spectral_project(a, alpha=0.1):
    """Project the power spectrum of a toward flat.
    Returns a modified sequence (not necessarily normalized).
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    P = np.abs(A)
    P_mean = P.mean()
    # Blend magnitude toward flat
    P_new = (1 - alpha) * P + alpha * P_mean
    # Keep phase
    A_new = P_new * (A / np.abs(A + 1e-300))
    a_new = irfft(A_new, fft_size)[:n]
    return np.maximum(a_new, 0)

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

            sc = evaluate_local(a_new)
            if sc < best_score:

                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                no_improve = 0
                break
            elif sc < lse_val / (2.0 * n) + 0.01:  # accept if close

                improved = True
                break
        if not improved:

            eta *= 0.8

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)
