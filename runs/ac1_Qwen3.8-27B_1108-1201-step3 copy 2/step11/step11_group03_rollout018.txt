        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 + u * np.arange(1, n+1) > cssv)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated projected gradient) + Mirror Descent hybrid
2. Active-set convex QCQP (cvxpy) for precise polynomial-time polishing
3. Multi-length exploration (n=1199,1200,1201)
4. Direct full-constraint QCQP attempt

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def grad_b_k(a, k):
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1.0 + np.arange(1, n + 1) * u > cssv
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_q_sparse(k, n):
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_solver():
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def fista_optimize(a0, time_limit, eta=0.5, mode='fista'):
    """FISTA or Mirror Descent to minimize max autocorrelation."""
    n = len(a0)
    rng = np.random.default_rng(12345)
    a = project_simplex(np.maximum(a0, 0).copy())
    a_prev = a.copy()
    t_vel = 1.0
    best_a = a.copy()
    best_max_b = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    stall = 0
    n_active = 3
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        current_max = b.max()
        
        if current_max < best_max_b - 1e-12:
            best_max_b = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        it += 1
        
        if mode == 'fista':
            # FISTA extrapolation
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_vel * t_vel)) / 2.0
            if it > 1:
                y = a + ((t_vel - 1.0) / t_new) * (a - a_prev)
            else:
                y = a.copy()
            y = np.maximum(y, 0)
            
            # Subgradient at y
            threshold = current_max * (1.0 - 1e-3)
            active = np.where(b >= threshold)[0]
            if len(active) > n_active:
                active = active[np.argsort(b[active])[-n_active:]]
            
            g = np.zeros(n)
            for k in active:
                g += grad_b_k(y, k)
            if len(active) > 0:
                g /= len(active)
            
            y = y - eta * g
            a_prev = a.copy()
            a = project_simplex(y)
            t_vel = t_new
            
        elif mode == 'mirror':
            # Mirror descent (multiplicative weights)
            threshold = current_max * (1.0 - 1e-3)
            active = np.where(b >= threshold)[0]
            if len(active) > n_active:
                active = active[np.argsort(b[active])[-n_active:]]
            
            g = np.zeros(n)
            for k in active:
                g += grad_b_k(a, k)
            if len(active) > 0:
                g /= len(active)
            
            eta_md = 0.3 / np.sqrt(it + 1)
            log_upd = -eta_md * g
            log_upd -= log_upd.max()
            a_new = a * np.exp(log_upd)
            s_new = a_new.sum()
            if s_new > 1e-300:
                a_new = a_new / s_new
            else:
                a_new = a.copy()
            
            a = a_new
            
        # Restart
        if stall > 5000:
            noise = rng.uniform(-0.02, 0.02, n)
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            a_prev = a.copy()
            t_vel = 1.0
            stall = 0
        
        if it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {mode} iter={it} ({elapsed:.0f}s): max_b={best_max_b:.10f} score={2*n*best_max_b:.10f}")
    
    return best_a, best_max_b


def active_set_qcqp(a0, solver, max_rounds=30, max_active=25, rel_tol=1e-5, time_limit=120.0):
    """Active-set QCQP: solve min t s.t. a^T Q_k a <= t for active k."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).copy())
    t0 = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        threshold = max_b * (1.0 - rel_tol)
        active = np.where(b >= threshold)[0]
        if len(active) > max_active:
            active = active[np.argsort(b[active])[-max_active:]]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        
        for k in active:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        solved = False
        for s in [solver, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_new = np.asarray(a_var.value, dtype=np.float64)
                    a_new = np.maximum(a_new, 0)
                    s_sum = a_new.sum()
                    if s_sum < 1e-15:
                        break
                    a_new = a_new / s_sum
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        b_new = fast_convolve(a_new)
        max_new = b_new.max()
        
        if max_new >= max_b - 1e-10:
            # No improvement, check if truly converged
            if max_new <= max_b + 1e-10:
                a = a_new
                break
            a = a_new
        else:
            a = a_new
        
        if (rnd + 1) % 5 == 0:
            print(f"    QCQP round {rnd+1}: max_b={max_b:.10f} -> {max_new:.10f} "
                  f"score={2*n*max_new:.10f} active={len(active)} t={time.monotonic()-t0:.1f}s")
    
    final_b = fast_convolve(a)
    return a, final_b.max()

    for s in [solver, cp.MOSEK, cp.CLARABEL, cp.ECOS]:
