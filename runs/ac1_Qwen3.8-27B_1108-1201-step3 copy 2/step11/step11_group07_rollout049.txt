            break
        remaining = time_limit - (time.monotonic() - t0)
        n_steps = max(10, int(remaining * 50))
        
        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            a = lse_gradient_step(a, T, n_steps=1, lr=lr)
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Adapt lr
        lr *= 0.5
    
    return best_a, best_score

def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Gaussian
    candidates.append(normalize(np.exp(-10*(x-0.5)**2)))
    
    # Two bumps
    candidates.append(normalize(np.exp(-15*(x-0.3)**2) + np.exp(-15*(x-0.7)**2)))
    
    # Cosine modulated
    a = np.maximum(0.5 + 0.3*np.cos(2*np.pi*3*x + rng.uniform(0, 2*np.pi)) + 
                   0.2*np.cos(2*np.pi*7*x + rng.uniform(0, 2*np.pi)), 0)
    candidates.append(normalize(a))
    
    # Random smooth
    coeffs = rng.normal(0, 1, 15)
    a = np.zeros(n)
    for i, cf in enumerate(coeffs):
        a += cf * np.cos(2*np.pi*(i+1)*x + rng.uniform(0, 2*np.pi))
    candidates.append(normalize(a + abs(a).max() + 0.1))
    
    # Sine power
    candidates.append(normalize(np.sin(np.pi*x)**4))
    
    # Edge enhanced
    candidates.append(normalize(np.exp(-3*x) + np.exp(-3*(1-x))))
    
    # Asymmetric Gaussian
    candidates.append(normalize(np.exp(-8*(x-0.4)**2) * 0.7 + np.exp(-8*(x-0.6)**2) * 0.3))
    
    return candidates

def lp_peak_shave(a0, time_limit, damping=0.3):
    """
    LP-based peak shaving. Iteratively solves LP to find best b
    given constraint a*b <= M, then updates a.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_mat = build_conv_matrix(a)
        b = lp_solve(a, best_M, conv_mat)
        
        if b is None or b.sum() < 1e-10:
            break
        
        b = normalize(b)
        M_b = fast_convolve(b).max()
        
        # Try direct update
        if M_b < best_M * (1 - 1e-12):
            a = b.copy()
            best_a = b.copy()
            best_M = M_b
            if it % 10 == 0:
                print(f"    LP iter {it}: M={best_M:.8f}, score={2*n*best_M:.6f}")
        else:
            # Try damped updates
            improved = False
            for t in [damping, damping*0.5, damping*0.2, 0.05, 0.02, 0.01]:
                a_try = normalize((1-t)*a + t*b)
                M_try = fast_convolve(a_try).max()
                if M_try < best_M * (1 - 1e-12):
                    a = a_try
                    best_a = a_try.copy()
                    best_M = M_try
                    improved = True
                    break
            
            if not improved:
                # Stuck - try reducing M to force LP to find different solution
                # Or break if we've been stuck too long
                if it > 500:
                    break
                # Slightly perturb a to escape
                a = normalize(a + 0.001 * np.random.randn(n))
    
    return best_a, best_M

def subgradient_descent(a0, time_limit, n_iters=100000, eta0=0.001):
    """Subgradient descent with active-set subgradients."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Active set: positions where c is close to max
        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    if i == j:
                        g[j] += a[j]  # diagonal term
                    else:
                        g[j] += 2 * a[i]
        g /= len(active)
        
        # Update
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            a = a_new.copy()  # Still update to explore
            no_improve += 1
        
        if no_improve % 100 == 0 and no_improve > 0:
            eta *= 0.9
            if eta < 1e-8:
                break
    
    return best_a, best_score
