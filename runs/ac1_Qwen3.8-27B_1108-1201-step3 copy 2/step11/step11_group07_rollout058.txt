import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def active_set_refinement(a0, time_limit, eps=1e-4):
    """Mirror descent with active set identification."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = 0.3
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()
        
        # Active set: peaks within eps of max
        threshold = c_max * (1.0 - eps)
        active = np.where(c >= threshold)[0]
        
        # Compute gradient: g[m] = (1/|S|) sum_{k in S} 2*a[k-m]
        g = np.zeros(n)
        for k in active:

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)

        g /= len(active)

        log_update = -eta * (g - g_mean)

    # This means (A@g)[k] = (f*g)[k]

    # Instead of building the full matrix (which is 2399 x 1200), 
    # we can use the fact that it's banded.
    # But scipy linprog doesn't support sparse well for all methods.
    # Let's just build it.
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            A[k, j] = f[k-j]
    
    # Objective: maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    b = M * np.ones(2*n-1)

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (SOTA approach) for global improvement

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(f):
    """Build the convolution matrix A where (A@g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] contributes f[k-j] to (f*g)[k] for k in [j, j+n-1]
        A[j:j+n, j] = f
    return A

def lp_direction(f, M, time_budget=5.0):
    """Find LP direction: maximize sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    t0 = time.monotonic()
    
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = M * np.ones(2*n-1)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                     options={'time_limit': time_budget})
        if res.success and res.x is not None:
            return res.x
    except Exception as e:
        print(f"  LP error: {e}")
    
    # Fallback: return uniform
    return np.ones(n) / n

def lp_based_optimize(f0, time_limit):
    """Optimize using LP-based direction finding."""
    n = len(f0)
    f = normalize(f0.copy())
    best_f = f.copy()
    best_score = evaluate_local(f)
    t0 = time.monotonic()
    
    iteration = 0
    t_mix = 0.1  # Initial mixing parameter
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        c = fast_convolve(f)
        M = c.max()
        
        # Solve LP
        g = lp_direction(f, M, time_budget=max(1.0, min(5.0, remaining*0.1)))
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        # Rescale g to match sum(f)
        f_sum = f.sum()
        g = g * (f_sum / g_sum)
        
        # Line search for mixing parameter
        best_t = t_mix
        best_sc = best_score

            if s < 1e-15:

            f_new = f_new / s
            sc = evaluate_local(f_new)
            if sc < best_sc:

                best_t = t
        
        if best_sc < best_score:

            f_new = np.maximum(f_new, 0)

            best_score = best_sc

            f = f_new.copy()

            hi = min(n - 1, k)

        f_new = f * np.exp(log_update)

def coordinate_descent(f0, time_limit):
    """Coordinate descent with 1D optimization."""
    n = len(f0)
    f = normalize(f0.copy())
    best_f = f.copy()
    best_score = evaluate_local(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(789)
    
    pass_count = 0
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        # Random permutation of coordinates
        order = rng.permutation(n)
        
        for m in order:
            if time.monotonic() - t0 > time_limit:
                    break
            
            old_val = f[m]
            f_sum = f.sum()
            s_other = f_sum - old_val
            
            # Try several values of t
            candidates = [
                old_val * 0.9,
                old_val * 0.8,
                old_val * 1.1,
                old_val * 1.2,
                old_val * 0.7,
                old_val * 1.3,
                0.0,  # Try setting to zero
            ]
            
            best_t = old_val
            best_sc = best_score
            
            for t in candidates:
                t = max(0.0, t)
                f[m] = t
                sc = evaluate_local(f)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
            
            f[m] = best_t
            f = normalize(f)
            
            if best_sc < best_score:
                best_score = best_sc
                best_f = f.copy()
        
        if pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.8f}")
    
    return best_f, best_score

            f_new = f + noise
        elif method == 1:  # Low frequency
            freq = rng.integers(1, 10)
            phase = rng.uniform(0, 2*np.pi)
            noise = step_size * np.cos(freq * x + phase)
            f_new = f + noise
        elif method == 2:  # Local bump
            center = rng.integers(10, n-10)
            width = rng.integers(5, 30)

            noise = np.zeros(n)

            f_new = f + noise
        else:  # Swap blocks

            bsize = rng.integers(5, 50)

        s = f_new.sum()
        if s < 1e-15:

        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new.copy()

        else:

            if rng.random() < 0.05:

    return best_f, best_score

Wait, I have a concern about the coordinate descent. Each evaluation of `evaluate_local` takes O(n log n) time due to FFT. With n=1200 and 7 candidate values per coordinate, one full pass takes 1200 * 7 = 8400 evaluations, each taking maybe 0.1ms, so about 0.84 seconds per pass. But actually, we can optimize the coordinate descent by not calling `evaluate_local` (which does an FFT) for every candidate. Instead, we can compute the auto-correlation once per coordinate and then evaluate all candidates analytically.

Let me optimize:

For coordinate m with current value v, if we change it to t:
- The auto-correlation changes by:
  - For k != 2m (and 0 <= k-m <= n-1): delta = 2*(t-v)*f[k-m]
  - For k = 2m: delta = t^2 - v^2

So if I precompute c = f*f (the auto-correlation), then for each candidate t:
- new_c[k] = c[k] + 2*(t-v)*f[k-m] for k in [m, m+n-1] (clipped to [0, 2n-2]) and k != 2m
- new_c[2m] = c[2m] + t^2 - v^2

The new max is max(new_c), and the new score is 2n * max(new_c) / (s_other + t)^2.

This avoids the FFT for each candidate! We just need one FFT per pass (to compute c initially) and then O(n) work per coordinate.

def coordinate_descent_fast(f0, time_limit):

def build_conv_matrix_sparse(f):

    for j in range(n):

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()

        return np.ones(len(a)) / len(a)
    return a / s
