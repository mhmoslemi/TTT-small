import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

        j_end = min(n - 1, k)
        T[k, j_start:j_end + 1] = y[j_start:j_end + 1][::-1]

                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                M_val = float(M_var.value) if M_var.value is not None else 0.0
                return b_val, max(0.0, M_val)
        except Exception:
            continue
    return None, None


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP alternating with Nesterov dual averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)
    lp_times = []
    max_iters = 500000

    for it in range(max_iters):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)

        if b_new is None:
            continue

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = project_simplex(y)

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            no_improve = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0

        # Check for improvement every 100 iterations
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-9:
                prev_best = best_score
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= 50:
                # Perturb to escape local minimum
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.02 * np.cos(freq * x + phase)
                y = y + noise
                y = project_simplex(y)
                a = a + noise * 0.5
                a = project_simplex(a)
                no_improve = 0

            if verbose and (it + 1) % 1000 == 0:
                avg_lp = np.mean(lp_times[-50:]) if len(lp_times) >= 50 else lp_time
                print(f"    NLP it={it+1} ({elapsed:.0f}s): score={best_score:.8f} lp={avg_lp*1000:.1f}ms")

        # Stop if LP is too slow
        if len(lp_times) > 100:
            avg_lp = np.mean(lp_times[-50:])
            if avg_lp > 3.0:
                if verbose:
                    print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at it={it+1}")
                break

    return best_a, best_score


def fft_subgradient(p, active_indices):
    """Compute subgradient of max_k conv(p,p)[k] given active set, using FFT."""
    n = len(p)
    conv_len = 2 * n - 1
    N = 1 << (3 * n - 2).bit_length()

    w = np.zeros(conv_len)
    w[active_indices] = 1.0 / len(active_indices)

    w_pad = np.zeros(N)
    w_pad[:conv_len] = w
    p_pad = np.zeros(N)
    p_pad[:n] = p

    W = rfft(w_pad, N)
    P = rfft(p_pad, N)

    # s_i = sum_k w_k * p_{k-i} = cross-correlation
    S = W * np.conj(P)
    s = irfft(S, N)[:n]

    # t_i = sum_k w_k * p_{i-k} = convolution
    T = P * W
    t = irfft(T, N)[:n]

    return s + t


def subgradient_optimize(p0, time_limit, top_k_frac=0.1, verbose=True):
    """FFT-based subgradient method with Nesterov momentum."""
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    y = p.copy()
    best_a = p.copy()
    best_score = 2.0 * n * fast_convolve(p).max()
    t0 = time.monotonic()
    no_improve = 0
    prev_best = best_score
    eta = 0.1

    for it in range(5000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv_y = fast_convolve(y)
        max_val = conv_y.max()
        k_active = max(1, int(top_k_frac * (2 * n - 1)))
        active = np.argpartition(conv_y, -k_active)[-k_active:]

        g = fft_subgradient(y, active)

        # Nesterov step
        alpha = 2.0 / (it + 2)
        y_new = (1 - alpha) * y
        step = eta / (1 + 0.01 * it / 1000)
        y_new = y_new - step * g
        y_new = project_simplex(y_new)

        p_new = p + alpha * (y_new - p)
        p_new = project_simplex(p_new)

        score = 2.0 * n * fast_convolve(p_new).max()
        if score < best_score:
            best_score = score
            best_a = p_new.copy()
            no_improve = 0
            prev_best = best_score

        p = p_new
        y = y_new

        if (it + 1) % 5000 == 0:
            if best_score > prev_best + 1e-10:
                no_improve += 1
            if no_improve >= 20:
                eta *= 0.7
                no_improve = 0
                prev_best = best_score
            if verbose and (it + 1) % 50000 == 0:
                print(f"    SubGrad it={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.5f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    else:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 5 * x) + 0.15 * np.cos(2 * np.pi * 11 * x + 0.5)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 15)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
