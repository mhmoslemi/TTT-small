"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative weights for initialization
2. Nesterov LP alternating with constraint generation (reduced LPs)
3. Multi-stage LogSumExp L-BFGS with aggressive cooling
4. Multi-length exploration
5. Palindromic constraint exploration
6. Perturbation escape and final polish

Key improvements: constraint generation LP (faster inner solves),
warm-starting, better time allocation, and more aggressive LSE schedule.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve_auto(a):
    """Fast auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Fast cross-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_constrained(a_fixed, active_indices, solver):
    """Solve reduced LP: min M s.t. (a*b)[k] <= M for k in active_indices, sum(b)=1, b>=0."""
    n = len(a_fixed)
    K = len(active_indices)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build the constraint rows
    # (a*b)[k] = sum_j a[j] * b[k-j] for valid j
    # For each k in active_indices, we need: sum_j a[j] * b[k-j] <= M
    # This is equivalent to: (row_k) @ b <= M
    # where row_k[j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Build constraint matrix efficiently
    rows = np.zeros((K, n), dtype=np.float64)
    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            rows[i, js:je+1] = a[k - np.arange(js, je+1)][::-1]
    
    constraints = [rows @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver]
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        if s not in solvers_to_try:
            solvers_to_try.append(s)
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                return b_val / s_sum, max(0, float(M_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_cgr(a0, time_limit, solver, n_max_iters=500000, K_constraints=300):
    """Nesterov LP alternating with constraint generation and reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    b_current = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution of y with current b to identify active constraints
        conv_yb = fast_convolve_cross(y, b_current)
        
        # Select active constraints: top K values
        if len(conv_yb) > K_constraints:
            threshold_idx = np.argpartition(conv_yb, -K_constraints)[-K_constraints:]
            active_indices = threshold_idx
        else:
            active_indices = np.arange(len(conv_yb))
        
        # Solve reduced LP
        b_new, M_new = solve_lp_constrained(y, active_indices, solver)
        
        if b_new is None:
            # Fallback: use full constraint set
            active_indices = np.arange(len(conv_yb))
            b_new, M_new = solve_lp_constrained(y, active_indices, solver)
            if b_new is None:
                continue
        
        # Verify: check if solution satisfies all constraints
        conv_check = fast_convolve_cross(y, b_new)
        max_violation = conv_check.max()
        
        if max_violation > M_new * 1.001 + 1e-10:
            # Some constraints are violated, add them
            violated = np.where(conv_check > M_new * 1.001 + 1e-10)[0]
            active_indices = np.unique(np.concatenate([active_indices, violated]))
            # Cap the number of constraints
            if len(active_indices) > 500:
                # Keep top 500
                conv_vals = conv_check[active_indices]
                top_idx = np.argpartition(conv_vals, -500)[-500:]
                active_indices = active_indices[top_idx]
            b_new, M_new = solve_lp_constrained(y, active_indices, solver)
            if b_new is None:
                continue
        
        b_current = b_new.copy()
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Update best
        score_a = 2.0 * n * fast_convolve_auto(a).max()
        score_y = 2.0 * n * fast_convolve_auto(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.02 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                b_current = y.copy()
                stall_count = 0
            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP-CGR iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def multiplicative_weights(a0, time_limit, n_steps=500000, eta0=0.05):
    """Fast multiplicative weights."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    eta = eta0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve_auto(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve_auto(a_new).max()
        if score_new < best_score:
            best_score = score_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative weights for initialization
2. Nesterov LP alternating with constraint generation (reduced LPs for speed)
3. Multi-stage LogSumExp L-BFGS with aggressive geometric cooling
4. Multi-length exploration (n=1000, 1200, 1500, 2000)
5. Palindromic constraint exploration
6. Perturbation escape and final dual polish

Key improvements: constraint generation LP (faster inner solves via dense
reduced matrices), warm-starting, better time allocation, more aggressive
LSE cooling schedule, and palindromic sub-problem exploitation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve_auto(a):
    """Fast auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Fast cross-convolution of a and b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_constraint_rows(a_fixed, active_indices, n):
    """Build dense K x n matrix where row i corresponds to constraint at index active_indices[i]."""
    K = len(active_indices)
    rows = np.zeros((K, n), dtype=np.float64)
    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            # row[i, j] = a[k - j] for j in [js, je]
            for j in range(js, je + 1):
                rows[i, j] = a_fixed[k - j]
    return rows


def solve_lp_reduced(a_fixed, active_indices, n, solver):
    """Solve reduced LP with only active constraints."""
    K = len(active_indices)
    if K == 0:
        return np.ones(n) / n, 1.0 / n
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    rows = build_constraint_rows(a_fixed, active_indices, n)
    constraints = [rows @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    seen = set()
    for s in solvers_to_try:
        if s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                return b_val / s_sum, max(0.0, float(M_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_cgr(a0, time_limit, solver, n_max_iters=500000, K_constr=250):
    """Nesterov LP alternating with constraint generation and reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    b_current = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(99999)
    iter_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        it_start = time.monotonic()

        conv_yb = fast_convolve_cross(y, b_current)
        n_conv = len(conv_yb)

        if n_conv > K_constr:
            part_idx = np.argpartition(conv_yb, -K_constr)[-K_constr:]
            active_indices = part_idx
        else:
            active_indices = np.arange(n_conv)

        b_new, M_new = solve_lp_reduced(y, active_indices, n, solver)

        if b_new is None:
            active_indices = np.arange(n_conv)
            b_new, M_new = solve_lp_reduced(y, active_indices, n, solver)
            if b_new is None:
                continue

        conv_check = fast_convolve_cross(y, b_new)
        if conv_check.max() > M_new * 1.002 + 1e-9:
            violated = np.where(conv_check > M_new * 1.002 + 1e-9)[0]
            active_indices = np.unique(np.concatenate([active_indices, violated]))
            if len(active_indices) > min(n_conv, 600):
                conv_vals = conv_check[active_indices]
                top_i = np.argpartition(conv_vals, -min(n_conv, 600))[-min(n_conv, 600):]
                active_indices = active_indices[top_i]
            b_new, M_new = solve_lp_reduced(y, active_indices, n, solver)
            if b_new is None:
                continue

        b_current = b_new.copy()

        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve_auto(a).max()
        score_y = 2.0 * n * fast_convolve_auto(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        iter_times.append(time.monotonic() - it_start)
        if len(iter_times) > 50:
            iter_times.pop(0)

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 4:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.015 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                b_current = y.copy()
                stall_count = 0
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                avg_t = np.mean(iter_times) if iter_times else 0
                print(f"    LP-CGR iter={it+1} ({elapsed:.0f}s, {avg_t*1000:.1f}ms/it): score={best_score:.8f}")
            if len(iter_times) >= 30 and np.mean(iter_times) > 5.0:
                print(f"    LP too slow ({np.mean(iter_times):.2f}s/it), stopping.")
                break

    return best_a, best_score


def multiplicative_weights(a0, time_limit, n_steps=500000, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve_auto(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve_auto(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.008, 0.4)
        else:
            eta *= 0.82
            if eta < 1e-10:
                break
    return best_a, best_score

    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    else:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses constraint-generation LP with Nesterov acceleration,
    adaptive multiplicative weights, fine LSE L-BFGS, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick MW ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.03, 40.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, n_steps=500000, eta0=0.05)
        try_update(a_mw, "mw_init")
        print(f"  After MW: score={best_score:.8f}")

    # === Phase 2: Main LP-CGR ===
    print(f"\n=== Phase 2: LP-CGR n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.38, 650.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_cgr(best_seq, t2, solver, n_max_iters=1000000, K_constr=250)
        try_update(a_lp, "lp_cgr_1")
        print(f"  After LP: score={best_score:.8f}")

    # === Phase 3: Palindromic LP ===
    print(f"\n=== Phase 3: Palindromic (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t3 = min(rem * 0.10, 150.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = nesterov_lp_cgr(a_pal, t3, solver, n_max_iters=200000, K_constr=200)
        try_update(a_pal_lp, "pal_lp")
        # Also try palindromic LSE
        rem = deadline - time.monotonic()
        if rem > 30:
            T_stages_pal = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_pal_lse = lbgfs_refine(a_pal_lp, min(rem * 0.4, 60.0), T_stages_pal, maxiter_per_stage=500)
            try_update(a_pal_lse, "pal_lse")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1500, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.20, 250.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        t_mw = t_ln * 0.15
        a_ln_mw, _ = multiplicative_weights(init_ln, t_mw, n_steps=100000, eta0=0.05)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP-CGR
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_cgr(a_ln_mw, t_lp, solver, n_max_iters=500000, K_constr=200)
            try_update(a_ln_lp, f"lp_n{ln}")
        # Quick LSE on best of this length
        rem = deadline - time.monotonic()
        if rem > 20:
            T_stages_short = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_ln_lse = lbgfs_refine(best_seq, min(rem * 0.3, 50.0), T_stages_short, maxiter_per_stage=400)
            try_update(a_ln_lse, f"lse_n{ln}")

    # === Phase 5: Fine LSE L-BFGS on global best ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.30, 500.0)
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.7, 0.5, 0.3, 0.2,
                    0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005,
                    0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lbgfs_refine(best_seq, t5, T_stages, maxiter_per_stage=600)
        try_update(a_lse, "lse_fine")

    # === Phase 6: Perturbation escape ===
    print(f"\n=== Phase 6: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            strength = 0.01 + 0.008 * trial
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem * 0.3, 80.0)
            a_opt, _ = nesterov_lp_cgr(a_pert, t_sub, solver, n_max_iters=200000, K_constr=200)
            try_update(a_opt, f"pert{trial}")
            # Quick LSE polish on perturbation result
            rem = deadline - time.monotonic()
            if rem > 15 and a_opt is not None:
                T_short = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
                a_pert_lse = lbgfs_refine(a_opt, min(rem * 0.3, 30.0), T_short, maxiter_per_stage=300)
                try_update(a_pert_lse, f"pert{trial}_lse")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.35, 150.0)
        a_fin, _ = nesterov_lp_cgr(best_seq, t7, solver, n_max_iters=500000, K_constr=300)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final LSE polish (very fine) ===
    print(f"\n=== Phase 8: Final LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        T_final = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        a_fin_lse = lbgfs_refine(best_seq, min(rem * 0.6, 120.0), T_final, maxiter_per_stage=800)
        try_update(a_fin_lse, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
