import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    a = np.asarray(a, dtype=np.float64)
    T = np.zeros((size, n))
    for j in range(n):
        # T[k, j] = a[k-j] if 0 <= k-j < n
        # k ranges from j to j+n-1
        T[j:j+n, j] = a
    return T

def get_lp_solver():
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS

def solve_lp_best_response(a_fixed, solver, n_iters=1):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    continue
                b_val = b_val / s_sum
                return b_val, max(0.0, float(t_var.value))

def adam_optimize(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    
    total_stages = len(T_stages)
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break
        
        top_k = max(5, int(top_k_frac * (2 * n - 1)))
        steps_in_stage = int(remaining * 5000 / max(1, total_stages - stage_idx))
        
        for step in range(steps_in_stage):
            if time.monotonic() > deadline:
                break
            
            a = np.exp(z - z.max())
            a = a / a.sum()
            
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute gradient
            if T_stage < 0.01:
                threshold = max_b * (1.0 - 1e-4)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()
            
            # Gradient in a-space: d/da max_k LSE(b) = (1/T) * sum_k w[k] * a[k-i]
            # conv(a,a)[k] = sum_i a[i] a[k-i]
            # d/da[j] conv(a,a)[k] = a[k-j] + (a[k-j] if k-j != j else a[k-j])
            # Actually: conv(a,a)[k] = sum_i a[i] a[k-i]
            # d/da[j] = a[k-j] + (a[k-j] if we're differentiating the second factor)
            # Wait: d/da[j] sum_i a[i] a[k-i] = a[k-j] (from first factor, i=j) + sum_i a[i] * d/da[j] a[k-i]
            # d/da[j] a[k-i] = 1 if k-i = j, i.e., i = k-j. So = a[k-j].
            # Total: 2*a[k-j] (unless k-j = j, in which case it's a[k-j]... no, it's still 2*a[j] since both terms give a[j])
            # Actually: d/da[j] [a[j] * a[k-j] + ...] where the "..." includes terms where i != j.
            # The term where i=j: a[j] * a[k-j], derivative w.r.t. a[j] = a[k-j]
            # The term where k-i=j, i.e., i=k-j: a[k-j] * a[j], derivative w.r.t. a[j] = a[k-j]
            # If j != k-j (i.e., 2j != k), these are two different terms, total derivative = 2*a[k-j]
            # If j == k-j (i.e., 2j == k), it's a single term a[j]^2, derivative = 2*a[j] = 2*a[k-j]
            # So in all cases: d/da[j] conv(a,a)[k] = 2*a[k-j] (when 0 <= k-j < n, else 0)
            
            g_a = np.zeros(n)
            # g_a[j] = 2 * sum_k w[k] * a[k-j] (where k-j is valid)
            # This is 2 * correlate(w, a) at lag j
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = 2.0 * g_a_raw[n-1:2*n-1]
            
            # Gradient in z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            # Adam update
            lr = lr_init * (lr_decay ** stage_idx) / (1.0 + 0.05 * step / max(1, steps_in_stage))
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()
            
            if (step + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                cur_score = 2.0 * n * fast_convolve(a).max()
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")
    
    return best_a, best_score

def evaluate_local(a):

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new

5. Multi-length exploration (n=1200 primary, 1500 secondary)
Key improvements: top-K LP restriction for speed, better perturbation escapes,
more aggressive annealing schedules, and spectral-domain initialization.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_restricted_conv_matrix(a, k_indices):
    """Build restricted (len(k_indices)) x n matrix for selected convolution lags."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    T = np.zeros((len(k_indices), n))
    for row, k in enumerate(k_indices):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        T[row, js:je] = a[k - (je - 1):k - js + 1][::-1] if je > js else 0
    return T


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    a = np.asarray(a, dtype=np.float64)
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver, restrict_k=None, top_k=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    If restrict_k is given, only enforce constraints at those k indices.
    If top_k is given, use the top-k largest expected violations.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()

    if restrict_k is not None and len(restrict_k) < 2 * n - 1:
        T_mat = build_restricted_conv_matrix(a_fixed, restrict_k)
    else:
        T_mat = build_conv_matrix(a_fixed)

    constraints = [T_mat @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    continue
                b_val = b_val / s_sum
                return b_val, max(0.0, float(t_var.value))
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def adam_logsumexp(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.08):
    """Adam optimization in log-space with temperature annealing and top-k focusing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    global_step = 0

    total_stages = len(T_stages)

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break

        top_k = max(10, int(top_k_frac * (2 * n - 1)))
        remaining_stages = max(1, total_stages - stage_idx)
        steps_in_stage = max(100, int(remaining * 3000 / remaining_stages))

        for step in range(steps_in_stage):
            if time.monotonic() > deadline:
                break
            if (step + 1) % 5000 == 0 and time.monotonic() - t0 > time_limit * 0.95:
                break

            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            global_step += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute weights w for the log-sum-exp approximation
            if T_stage < 0.02:
                threshold = max_b * (1.0 - 1e-3)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / (exp_top.sum() + 1e-300)
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    exp_all = np.exp((b - max_b) / T_stage)
                    w = exp_all / (exp_all.sum() + 1e-300)

            # Gradient: d(LSE)/da[j] = (2/T) * sum_k w[k] * a[k-j]
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = (2.0 / T_stage) * g_a_raw[n - 1:2 * n - 1]

            # Convert to z-space gradient
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            # Adam update with decay
            progress = step / max(1, steps_in_stage)
            lr = lr_init * (lr_decay ** stage_idx) * (1.0 - 0.3 * progress)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (global_step + 1))
            v_hat = v / (1 - beta2 ** (global_step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

            if (step + 1) % 15000 == 0:
                elapsed = time.monotonic() - t0
                cur = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
                print(f"    Adam st{stage_idx}(T={T_stage:.3f}) s{step+1} ({elapsed:.0f}s): best={best_score:.8f} cur={cur:.8f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, use_restricted=True, top_k=500):
    """Nesterov dual averaging with LP alternating, optionally restricted."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Determine which k indices to restrict LP to
        k_restrict = None
        if use_restricted:
            b_current = fast_convolve(a)
            if top_k < len(b_current):
                idx = np.argpartition(b_current, -top_k)[-top_k:]
                k_restrict = list(idx)

        b_new, M_b = solve_lp_best_response(a, solver, restrict_k=k_restrict)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a_new = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sa

        score_a = 2.0 * n * fast_convolve(a_new).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        a = a_new

        if (it + 1) % 10 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freqs = rng.integers(1, 20, size=3)
                phases = rng.uniform(0, 2 * np.pi, size=3)
                noise = np.zeros(n)
                for f, p in zip(freqs, phases):
                    noise += 0.03 * np.cos(f * x + p)
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = a + noise * 0.5
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0

            if (it + 1) % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def multiplicative_weights(a0, time_limit):
    """Mirror descent (multiplicative) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.015
    no_improve = 0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g = 2.0 * g / len(active)

        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            eta *= 0.85
            no_improve = 0
            if eta < 0.001:
                break

    return best_a, best_score


def spectral_init(n, rng):
    """Create initializations designed for flat power spectrum."""
    choice = rng.integers(0, 14)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        a = 1.0 + 0.25 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x + 0.7)
        a = np.maximum(a, 0)
    elif choice == 2:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 5 * x + 0.3) + 0.1 * np.cos(2 * np.pi * 13 * x + 1.2)
        a = np.maximum(a, 0)
    elif choice == 3:
        a = np.exp(-3.0 * x * (1 - x))
    elif choice == 4:
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif choice == 5:
        # Random multi-frequency
        a = np.ones(n)
        n_freqs = rng.integers(3, 8)
        for _ in range(n_freqs):
            f = rng.uniform(1, 25)
            p = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.15)
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0)
    elif choice == 6:
        # Symmetric multi-peak
        a = 0.5 + 0.3 * np.exp(-10 * (x - 0.25) ** 2) + 0.3 * np.exp(-10 * (x - 0.75) ** 2)
        a = np.maximum(a, 0)
    elif choice == 7:
        # Inverse-cosine window
        a = np.arccos(np.cos(np.pi * x)) / (2 * np.pi) + 0.5
    elif choice == 8:
        # Kaiser-like
        a = np.exp(-2.0 * x * (1 - x) * (0.5 + x * (1 - x)))
    elif choice == 9:
        # Periodic with phase
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 4 * x) + 0.15 * np.sin(2 * np.pi * 9 * x + 0.8)
        a = np.maximum(a, 0)
    elif choice == 10:
        # Power-law decay from center
        d = np.abs(x - 0.5)
        a = (1.0 - d) ** rng.uniform(1, 4) + 0.1
    elif choice == 11:
        # Flat with structured perturbation
        a = np.ones(n)
        a[::2] += 0.1 * np.cos(2 * np.pi * np.arange(0, n, 2) / n * 6)
        a = np.maximum(a, 0)
    elif choice == 12:
        # Log-sine
        a = 1.0 / (1.0 + np.exp(-5 * (x - 0.5)))
    else:
        # Smooth random
        a = np.ones(n)
        for _ in range(2):
            f = rng.uniform(2, 15)
            p = rng.uniform(0, 2 * np.pi)
            a += rng.uniform(0.1, 0.3) * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_local(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses spectral-shaping inits, Adam LogSumExp, restricted LP alternating,
    and multiplicative weights with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = spectral_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Quick Adam warm-up ===
    print(f"\n=== Phase 1: Adam Warm-up n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
        a_adam, _ = adam_logsumexp(best_seq, t_alloc, T_stages, lr_init=0.004, lr_decay=0.85, top_k_frac=0.08)
        local_update(a_adam, "adam_warm")

    # === Phase 2: LP Alternating (main workhorse, restricted) ===
    print(f"\n=== Phase 2: LP Alt (restricted) n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, use_restricted=True, top_k=600)
        local_update(a_lp, "lp_restricted_1")

    # === Phase 3: Fine Adam ===
    print(f"\n=== Phase 3: Adam Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_fine = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002]
        a_fin, _ = adam_logsumexp(best_seq, t_alloc, T_fine, lr_init=0.003, lr_decay=0.88, top_k_frac=0.05)
        local_update(a_fin, "adam_fine")

    # === Phase 4: Full LP (no restriction, for precision) ===

                                     [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005],
                                     lr_init=0.003, lr_decay=0.87, top_k_frac=0.06)

            for f, p in zip(freqs, phases):

        if t_fin > 5:
            T_final = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
            a_fin2, _ = adam_logsumexp(best_seq, t_fin, T_final,
                                        lr_init=0.001, lr_decay=0.92, top_k_frac=0.03)
            local_update(a_fin2, "final_adam")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_mw2, _ = multiplicative_weights(best_seq, min(rem * 0.4, 25.0))
        local_update(a_mw2, "final_mw")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_lp3, _ = nesterov_lp_refine(best_seq, min(rem * 0.6, 20.0), solver, use_restricted=True, top_k=300)
        local_update(a_lp3, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
