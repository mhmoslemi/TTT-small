from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix, csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = lil_matrix((size, n), dtype=np.float64)
    for j in range(n):
        js = max(0, 0)
        je = min(n, size - j)
        if je > js:
            T[js:je, j] = a[j]
    return csc_matrix(T)


def get_best_solver():
    """Find the best available solver for LP/QCQP."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)

def solve_lp_best_response(a_fixed, solver):

    prob = cp.Problem(cp.Minimize(M_var), constraints)

                if s_sum < 1e-15:

    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n), dtype=np.float64)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    # Progressive tolerance schedule
    tolerances = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                  0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001]

    for tol in tolerances:
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 5:
            break

        for round_num in range(max_rounds):
            if time.monotonic() > deadline:
                break

            b = fast_convolve(a)
            max_b = b.max()

            # Identify active set
            active = np.where(b >= max_b * (1.0 - tol))[0]
            if len(active) == 0:
                active = np.array([np.argmax(b)])

            # Cap active set size for tractability
            max_active = 50
            if len(active) > max_active:
                active = active[np.argsort(b[active])[-max_active:]]

            # Solve QCQP
            a_new, t_val = solve_qcqp_active(a, list(active), solver)

            if a_new is None:
                break

            # Check if solution improved
            b_new = fast_convolve(a_new)
            max_b_new = b_new.max()
            score_new = 2.0 * n * max_b_new

            if score_new < best_score - 1e-15:
                best_score = score_new
                best_a = a_new.copy()
                print(f"    QCQP tol={tol:.2e} round={round_num}: score={score_new:.12f} |active|={len(active)}")

            a = a_new

            # Check for violated inactive constraints
            if max_b_new < max_b * (1.0 - tol * 0.5):
                # Significant improvement, continue to next tolerance
                break

            # Check if any previously inactive constraint is now active
            new_active = np.where(b_new >= max_b_new * (1.0 - tol * 0.1))[0]
            if len(new_active) > len(active) + 5:
                # Active set grew significantly, re-solve
                continue

            # Check KKT-like condition: if no improvement, stop this tol level
            if abs(score_new - 2.0 * n * max_b) < 1e-12:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def multiplicative_refine(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        # Efficient gradient computation for active positions
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-6:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating, iterative QCQP polishing, multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_best_solver()
    print(f"Using solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.12f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.12f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.12f} (n={len(a)})")

    # === Phase 1: LP alternating warm-up ===
    print(f"\n=== Phase 1: LP Alternating n={n_best} (score={best_score:.12f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        update_best(a_lp, "lp_phase1")

    # === Phase 2: Iterative QCQP polish (main improvement) ===
    print(f"\n=== Phase 2: Iterative QCQP Polish (score={best_score:.12f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.30, 450.0)
    if t_qcqp > 20:
        q_start = time.monotonic()
        a_qcqp, sc_qcqp = iterative_qcqp_polish(best_seq, solver,
                                                 deadline - (rem - t_qcqp) - 30,
                                                 max_rounds=50)
        update_best(a_qcqp, "qcqp_iterative")
        print(f"  QCQP polish time: {time.monotonic() - q_start:.1f}s")

    # === Phase 3: Another LP round from polished ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.12f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=200000)
        update_best(a_lp2, "lp_phase3")

    # === Phase 4: QCQP + multiplicative refinement ===
    print(f"\n=== Phase 4: Multi + QCQP (score={best_score:.12f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_mult, _ = multiplicative_refine(best_seq, min(rem * 0.2, 60.0), n_steps=50000)
        update_best(a_mult, "multiplicative")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_q2, _ = iterative_qcqp_polish(best_seq, solver,
                                             deadline - (rem - 60), max_rounds=30)
            update_best(a_q2, "qcqp_round2")

    # === Phase 5: Try n=1800 ===
    print(f"\n=== Phase 5: Try n=1800 (score={best_score:.12f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t_1800 = min(rem * 0.20, 250.0)
        print(f"  Budget: {t_1800:.0f}s")
        init_1800 = interpolate_sequence(best_seq, 1800)
        a_1800, _ = nesterov_lp_refine(init_1800, t_1800 * 0.6, solver, n_max_iters=150000)
        update_best(a_1800, "lp_n1800")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1800 and rem > 30:
            a_q1800, _ = iterative_qcqp_polish(best_seq, solver,
                                                deadline - (rem - 80), max_rounds=20)
            update_best(a_q1800, "qcqp_n1800")

    # === Phase 6: Try n=2400 ===
    print(f"\n=== Phase 6: Try n=2400 (score={best_score:.12f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_2400 = min(rem * 0.18, 200.0)
        print(f"  Budget: {t_2400:.0f}s")
        # Interpolate from the n=1200 parent (not from current best which might be n=1800)
        base_for_interp = parent if parent is not None else best_seq
        init_2400 = interpolate_sequence(base_for_interp, 2400)
        a_2400, _ = nesterov_lp_refine(init_2400, t_2400 * 0.6, solver, n_max_iters=100000)
        update_best(a_2400, "lp_n2400")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 2400 and rem > 30:
            a_q2400, _ = iterative_qcqp_polish(best_seq, solver,
                                                deadline - (rem - 80), max_rounds=15)
            update_best(a_q2400, "qcqp_n2400")

    # === Phase 7: Final QCQP polish on absolute best ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.12f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15 and best_seq is not None:
        a_final, _ = iterative_qcqp_polish(best_seq, solver,
                                           deadline - 5, max_rounds=20)
        update_best(a_final, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.12f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
