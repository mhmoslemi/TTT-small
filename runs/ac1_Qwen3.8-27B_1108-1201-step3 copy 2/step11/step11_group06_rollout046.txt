def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

            row[j] = a_fixed[k - j]
        rows.append(row)
    return np.array(rows) if rows else np.zeros((1, n))


def solve_lp_active_set(a_fixed, b_init, n_active=50, max_iters=200, time_limit=5.0):
    """Solve the LP: find b minimizing max_k conv(a_fixed, b)[k]
    using active-set constraint generation.
    """
    n = len(a_fixed)
    b = np.asarray(b_init, dtype=np.float64).copy()
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s

    t0 = time.monotonic()
    M_best = fast_cross_convolve(a_fixed, b).max()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ab = fast_cross_convolve(a_fixed, b)
        max_val = ab.max()

        # Get top n_active indices
        if n_active >= len(ab):
            top_idx = np.arange(len(ab))
        else:
            top_idx = np.argsort(ab)[-n_active:]

        # Build constraint matrix
        C = build_constraint_rows(a_fixed, top_idx)
        m = len(C)

        # LP: min M s.t. C@b <= M, sum(b)=1, b>=0
        # Variables: [b_0, ..., b_{n-1}, M]
        # c = [0, ..., 0, 1]
        c = np.zeros(n + 1)
        c[-1] = 1.0

        # Inequality constraints: C@b - M <= 0  =>  [C, -1] @ [b, M] <= 0
        A_ub = np.hstack([C, -np.ones((m, 1))])
        b_ub = np.zeros(m)

        # Equality: sum(b) = 1
        A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
        b_eq = np.ones(1)

        # Bounds: b >= 0, M free
        bounds = [(0, None)] * n + [(None, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            b_new = res.x[:n]
            b_new = np.maximum(b_new, 0)
            s_new = b_new.sum()
            if s_new < 1e-15:
                break
            b_new = b_new / s_new
        except Exception:
            break

        M_new = fast_cross_convolve(a_fixed, b_new).max()

        if M_new >= M_best - 1e-12:
            # Check if we should expand active set
            ab_new = fast_cross_convolve(a_fixed, b_new)
            if ab_new.max() > max_val + 1e-10:
                # Constraint violation, expand active set
                n_active = min(n_active * 2, len(ab))
                continue
            break

        M_best = M_new
        b = b_new

    return b, M_best


def nesterov_lp_alternating(a0, time_limit, n_active=80):
    """Nesterov-accelerated LP alternating with active-set LPs."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP: best response to y
        b_new, M_b = solve_lp_active_set(y, a, n_active=n_active, max_iters=100, time_limit=3.0)
        if b_new is None:
            continue

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Update a (Nesterov extrapolation)
        a_old = a.copy()
        a = a + (alpha / 2.0) * (b_new - a_old)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                # Perturb to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                a = np.maximum(a + 0.5 * noise, 0)
                a = a / a.sum()
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP alt iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def multiplicative_weights_nesterov(a0, time_limit, n_steps=200000, eta0=0.08):
    """Multiplicative weights with Nesterov momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        # Use top few peaks for the subgradient
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > 20:
            active = np.argsort(b)[-20:]

        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += w * y[k - j]

        # Nesterov: evaluate gradient at y, update a
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = y * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Nesterov momentum update
        alpha = 2.0 / (step + 2)
        y = (1.0 - alpha) * y + alpha * a_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.25)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def lse_objective_and_grad(z, n, T):
    """LogSumExp objective and gradient for softmax parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_max = scaled.max()
    exp_terms = np.exp(scaled - scaled_max)
    lse = max_b + T * (scaled_max + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse

    # Gradient
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad_a[j] += w[k] * a[k - j]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_vec):
            if time.monotonic() > deadline:
                return 1e15, np.zeros(len(z_vec))
            obj, grad = lse_objective_and_grad(z_vec, n, stage_T)
            return obj, grad

        try:
            res = minimize(
                fun=func, x0=z, jac=True, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': maxiter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    result = exp_z / exp_z.sum()
    return result


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = 0.3 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 7 * x + 1)
        a = np.maximum(a, 0)
    elif choice == 9:
        a = np.exp(-3 * (x - 0.5) ** 2) * (1 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 10:
        a = 0.5 * np.exp(-2 * (x - 0.25) ** 2) + 0.5 * np.exp(-2 * (x - 0.75) ** 2)
    else:
        a = np.ones(n) * (0.5 + 0.5 * np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def hybrid_lse_lp(a0, time_limit, solver_lp=True):
    """Hybrid: alternate LSE L-BFGS (smooth) and LP (exact) for better convergence."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Temperature stages for LSE
    T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
    T_idx = 0
    lp_count = 0

    while time.monotonic() - t0 < time_limit:
        # LSE L-BFGS step
        T = T_stages[min(T_idx, len(T_stages) - 1)]
        t0_sub = time.monotonic()
        z = np.log(np.maximum(a, 1e-300))

        def func(z_vec):
            if time.monotonic() - t0 > time_limit:
                return 1e15, np.zeros(len(z_vec))
            return lse_objective_and_grad(z_vec, n, T)

        try:
            res = minimize(
                fun=func, x0=z, jac=True, method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': 5000, 'disp': False}
            )
            z_new = res.x
            z_shifted = z_new - z_new.max()
            exp_z = np.exp(z_shifted)
            a_new = exp_z / exp_z.sum()
        except Exception:
            a_new = a.copy()

        # Check improvement
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new

        # LP step (every other LSE stage)
        if solver_lp and lp_count % 2 == 0:
            t0_lp = time.monotonic()
            b_new, M_b = solve_lp_active_set(a, a, n_active=60, max_iters=50, time_limit=5.0)
            if b_new is not None:
                score_lp = 2.0 * n * fast_convolve(b_new).max()
                if score_lp < best_score:
                    best_score = score_lp
                    best_a = b_new.copy()
                # Blend
                a = 0.7 * a + 0.3 * b_new
                a = np.maximum(a, 0)
                a = a / a.sum()
            lp_count += 1

        # Advance temperature
        T_idx += 1
        if T_idx >= len(T_stages):
            T_idx = len(T_stages) - 1
            # Reset to start cycle with lower T
            T_stages = [t * 0.5 for t in T_stages]

        if (lp_count + T_idx) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hybrid ({elapsed:.0f}s): score={best_score:.8f} T={T:.3f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov MW, active-set LP, hybrid LSE+LP, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    print(f"Budget: {budget_s:.0f}s, deadline in {max(0, deadline - time.monotonic()):.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1000
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick MW to warm up ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 15:
        a_mw, _ = multiplicative_weights_nesterov(best_seq, t1, n_steps=300000, eta0=0.08)
        try_update(a_mw, "mw_1")

    # === Phase 2: Nesterov LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP alternating n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_alternating(best_seq, t2, n_active=80)
        try_update(a_lp, "lp_1")

    # === Phase 3: Hybrid LSE+LP ===
    print(f"\n=== Phase 3: Hybrid LSE+LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 60:
        print(f"  Budget: {t3:.0f}s")
        a_hyb, _ = hybrid_lse_lp(best_seq, t3, solver_lp=True)
        try_update(a_hyb, "hybrid_1")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1500, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.15, 180.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights_nesterov(init_ln, t_ln * 0.25, n_steps=100000, eta0=0.06)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.55)
        if t_lp > 30:
            start_lp = a_ln_mw if a_ln_mw is not None else init_ln
            a_ln_lp, _ = nesterov_lp_alternating(start_lp, t_lp, n_active=60)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 5: Palindromic refinement ===
    print(f"\n=== Phase 5: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t5 = min(rem * 0.10, 120.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = nesterov_lp_alternating(a_pal, t5, n_active=60)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 6: Fine LSE L-BFGS on best ===
    print(f"\n=== Phase 6: LSE L-BFGS fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t6 = min(rem * 0.18, 250.0)
        T_stages = [8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.07,
                    0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        print(f"  {len(T_stages)} stages, budget={t6:.0f}s")
        a_lse = lbgfs_refine(best_seq, t6, T_stages, maxiter_per_stage=400)
        try_update(a_lse, "lse_fine")

    # === Phase 7: Perturbation + re-optimize ===
    print(f"\n=== Phase 7: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            strength = 0.01 + 0.008 * trial
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem * 0.20, 50.0)
            a_opt, _ = nesterov_lp_alternating(a_pert, t_sub, n_active=50)
            try_update(a_opt, f"pert{trial}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t8_lp = min(rem * 0.35, 80.0)
        a_fin_lp, _ = nesterov_lp_alternating(best_seq, t8_lp, n_active=100)
        try_update(a_fin_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1000) / 1000
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
