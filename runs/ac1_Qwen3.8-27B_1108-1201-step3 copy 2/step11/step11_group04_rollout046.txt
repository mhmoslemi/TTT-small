import numpy as np
import cvxpy as cp

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp, deadline):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    if time.monotonic() > deadline:
        return 1e15, np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    size = len(b)
    for k in range(size):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def power_sum_obj_grad(z, n, p, deadline):
    """Compute power-sum objective and gradient w.r.t. z.
    f = 2n * (sum_k b[k]^p)^(1/p)
    """
    if time.monotonic() > deadline:
        return 1e15, np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    b_p = b ** p
    sum_b_p = b_p.sum()
    if sum_b_p < 1e-300:
        return 0.0, np.zeros_like(z)
    f_val = (sum_b_p) ** (1.0 / p)
    obj = 2.0 * n * f_val

    # Gradient: df/da[j] = 2n * sum_b_p^(1/p - 1) * sum_k b[k]^(p-1) * a[k-j]
    prefactor = 2.0 * n * sum_b_p ** (1.0 / p - 1.0)
    weights = b ** (p - 1)
    grad_a = np.zeros(n)
    size = len(b)
    for k in range(size):
        if weights[k] < 1e-300:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += weights[k] * a[k - idx]
    grad_a *= prefactor

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_refine_logsumexp(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        iters = max(50, min(maxiter_per_stage, int(remaining * 80 / stages_left)))

        def func_lse(z_in):
            return logsumexp_obj_grad(z_in, n, stage_T, deadline)[0]

        def jac_lse(z_in):
            return logsumexp_obj_grad(z_in, n, stage_T, deadline)[1]

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 1000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def lbgfs_refine_power(a0, time_limit, p_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with power-sum smoothing and exponent annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(p_stages)

    for stage_i, stage_p in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        iters = max(50, min(maxiter_per_stage, int(remaining * 80 / stages_left)))

        def func_ps(z_in):
            return power_sum_obj_grad(z_in, n, stage_p, deadline)[0]

        def jac_ps(z_in):
            return power_sum_obj_grad(z_in, n, stage_p, deadline)[1]

        try:

                         'gtol': 1e-20, 'maxfun': iters * 1000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        if len(lp_times) >= 50:
            avg_lp_time = np.mean(lp_times[-50:])
            if avg_lp_time > 8.0:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")
                break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def peak_shave_adaptive(a, time_limit, n_steps=200000):
    """
    Adaptive peak shaving with geometric weighting of top peaks.
    Targets multiple peaks simultaneously with decreasing weights.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.015
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find peaks within adaptive threshold
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Geometric weighting based on peak height relative to max
        peak_vals = b[active]
        ratios = peak_vals / max_b  # All in [1-threshold_frac, 1]
        # Weight: higher peaks get more weight
        weights = ratios ** 50  # Sharp focusing on the top peak
        # Also add a small uniform component
        weights += 0.01

        g = np.zeros(n)
        total_w = weights.sum()
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += (weights[i] / total_w) * a[k - idx]

        # Multiplicative update
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 20:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break

        if step % 30000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.10f} scale={scale:.5f}")

    return best_a, best_score


def water_filling_peak_reduction(a, time_limit, n_steps=50000):
    """
    Directly reduce the top peaks by redistributing mass from elements
    that contribute most to those peaks.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_idx = np.argmax(b)

        # Find which (j, k-j) pairs contribute to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[peak_idx - idx]
        total_contrib = contributions.sum()

        if total_contrib < 1e-30:
            break

        # Reduce elements with high contribution, increase those with low
        contrib_frac = contributions / total_contrib
        # Target: reduce high-contribution elements
        target_reduction = contrib_frac - 1.0 / len(idx)  # Deviation from uniform
        # Only reduce positive deviations
        reduction = np.maximum(target_reduction, 0)
        reduction_sum = reduction.sum()
        if reduction_sum < 1e-30:
            break

        # Apply small reduction to high contributors
        lr = 0.005
        delta = lr * reduction / reduction_sum
        a_trial = a.copy()
        a_trial[idx] -= delta * a_trial[idx] * 5  # Proportional reduction
        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            break
        a_trial = a_trial / s_trial

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller step
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_a, best_score


def spectral_flatten(a, n_iters=100, blend=0.3, deadline=None):
    """
    Modified Gerchberg-Saxton: flatten the power spectrum to reduce time-domain peaks.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        phases = np.angle(F)
        non_dc = mags[1:L // 2]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = 1.0 - np.abs(2 * x - 1)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses power-sum/LSE smoothed L-BFGS, Nesterov LP, adaptive peak shaving,
    water-filling reduction, and systematic n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)}) (improvement: {old_score - sc:.2e})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.10f}")

    # === Phase 1: Quick spectral flattening ===
    print(f"\n=== Phase 1: Spectral Flattening (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_spec = min(rem * 0.03, 30.0)
        a_spec = spectral_flatten(best_seq, n_iters=100, blend=0.25, deadline=time.monotonic() + t_spec)
        local_update(a_spec, "spectral")

    # === Phase 2: Nesterov LP (primary optimizer) ===
    print(f"=== Phase 2: Nesterov LP n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.20, 300.0)
    if t_lp > 30:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=1000000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: L-BFGS LogSumExp (fine temperature) ===
    print(f"=== Phase 3: L-BFGS LogSumExp (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.25, 400.0)
    if t_lbfgs > 30:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.015, 0.01, 0.008,
                    0.006, 0.005, 0.004, 0.003, 0.002,
                    0.0015, 0.001, 0.0008, 0.0006, 0.0005]
        print(f"  T_stages: {len(T_stages)} stages, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine_logsumexp(best_seq, t_lbfgs, T_stages,
                                      maxiter_per_stage=10000,
                                      deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_lse_fine")

    # === Phase 4: L-BFGS Power-Sum (different smoothing) ===
    print(f"=== Phase 4: L-BFGS Power-Sum (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_ps = min(rem * 0.10, 120.0)
    if t_ps > 20:
        p_stages = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]
        print(f"  p_stages: {len(p_stages)} stages, budget={t_ps:.0f}s")
        a_psm = lbgfs_refine_power(best_seq, t_ps, p_stages,
                                   maxiter_per_stage=5000,
                                   deadline=time.monotonic() + t_ps)
        local_update(a_psm, "lbgfs_power")

    # === Phase 5: Adaptive Peak Shaving ===
    print(f"=== Phase 5: Adaptive Peak Shaving (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.08, 80.0)
    if t_peak > 10:
        a_pk, _ = peak_shave_adaptive(best_seq, t_peak, n_steps=300000)
        local_update(a_pk, "peak_shave")

    # === Phase 6: Water-filling peak reduction ===
    print(f"=== Phase 6: Water-Filling (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_wf = min(rem * 0.05, 40.0)
    if t_wf > 5:
        a_wf, _ = water_filling_peak_reduction(best_seq, t_wf, n_steps=100000)
        local_update(a_wf, "water_filling")

    # === Phase 7: Second Nesterov LP round ===
    print(f"=== Phase 7: Nesterov LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.12, 150.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=500000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 8: Perturbation escape + re-optimize ===
    print(f"=== Phase 8: Perturbation Escape (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            a_pert = perturbed_sequence(best_seq, rng, strength=0.01 + 0.008 * trial)
            # Quick spectral
            a_sp = spectral_flatten(a_pert, n_iters=15, blend=0.25)
            local_update(a_sp, f"pert_spec_{trial}")
            t_sub = min(rem_sub * 0.2, 40.0)
            if t_sub > 10:
                a_opt, _ = nesterov_lp_refine(a_sp, t_sub, solver, n_max_iters=100000)
                local_update(a_opt, f"pert_lp_{trial}")

    # === Phase 9: Try different n values ===
    print(f"=== Phase 9: n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_candidates = [1000, 1100, 1300, 1400, 1600]
        for n_try in n_candidates:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            if n_try == n_best:
                continue
            t_sub = min(rem_sub * 0.25, 80.0)
            if t_sub < 20:
                continue
            print(f"  Trying n={n_try}, budget={t_sub:.0f}s")
            init_n = interpolate_sequence(best_seq, n_try)
            a_sp_n = spectral_flatten(init_n, n_iters=20, blend=0.25)
            a_opt_n, _ = nesterov_lp_refine(a_sp_n, t_sub * 0.7, solver, n_max_iters=50000)
            local_update(a_opt_n, f"lp_n{n_try}")
            # Quick L-BFGS if improved
            rem_sub2 = deadline - time.monotonic()
            if best_seq is not None and len(best_seq) == n_try and rem_sub2 > 30:
                T_st2 = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003]
                a_lb_n = lbgfs_refine_logsumexp(best_seq, min(rem_sub2, 50.0), T_st2,
                                                maxiter_per_stage=3000,
                                                deadline=time.monotonic() + min(rem_sub2, 50.0))
                local_update(a_lb_n, f"lbgfs_n{n_try}")

    # === Phase 10: Final intensive L-BFGS ===
    print(f"=== Phase 10: Final L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.35, 80.0)
        T_stages_fin = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02,
                        0.015, 0.01, 0.008, 0.006, 0.005, 0.004,
                        0.003, 0.002, 0.0015, 0.001, 0.0008, 0.0005]
        a_fin = lbgfs_refine_logsumexp(best_seq, t_fin, T_stages_fin,
                                       maxiter_per_stage=8000,
                                       deadline=time.monotonic() + t_fin)
        local_update(a_fin, "final_lbgfs")

    # === Phase 11: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps_fin = min(rem * 0.3, 30.0)
        a_ps_fin, _ = peak_shave_adaptive(best_seq, t_ps_fin, n_steps=100000)
        local_update(a_ps_fin, "final_peak")

    # === Phase 12: Final LP polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.4, 30.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver, n_max_iters=50000)
        local_update(a_lp_fin, "final_lp")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
