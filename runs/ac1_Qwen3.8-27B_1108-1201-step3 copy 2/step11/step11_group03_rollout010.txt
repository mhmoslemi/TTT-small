    """

Wait, complementary slackness says: `lambda_k * (a^T Q_k a - t) = 0` for each active k. If lambda_k > 0, then `a^T Q_k a = t`. If lambda_k = 0, then `a^T Q_k a <= t`.

So at the optimum, all "positive multiplier" active constraints have `b_k = t`. This gives us a system of equations:
- `b_k = t` for k with lambda_k > 0
- `sum_{k in A} lambda_k * 2 * a_{k-i} = mu` for i in S
- `sum_{j in S} a_j = 1`

This is a nonlinear system (because b_k = a^T Q_k a is quadratic in a). We can solve it with Newton's method.

But this might be overkill. Let me just use a practical numerical approach.

**My final decision**: I'll use a combination of:
1. **Fast Frank-Wolfe** (using FFT-based gradients) for quick convergence
2. **scipy.optimize.minimize with SLSQP** for the QCQP (it can handle non-PSD quadratics)
3. **Active set iteration**: identify active constraints, solve QCQP, check for violations, add constraints, repeat
4. **Multi-n exploration**

Let me also consider: maybe I should just stick with the LogSumExp + L-BFGS approach but make it more efficient and with more aggressive temperature annealing. The previous program already did this, and it got to 1.504527. The issue might be that the LogSumExp approximation, even at low temperature, doesn't exactly capture the max.

The key improvement should be the **exact active-set QCQP** which should give the true optimum (or very close to it).

Let me now write the code. I'll use SLSQP for the QCQP since it's available in scipy and can handle the non-PSD quadratics.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """g[i] = sum_k w[k] * a[k-i] for i in [0, n-1]"""
    n = len(a)
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(np.asarray(w, dtype=np.float64), fft_size)
    A = rfft(np.asarray(a[::-1], dtype=np.float64), fft_size)
    return irfft(W * A, fft_size)[n-1 : 2*n-1]


def fw_iteration(a, tau):
    """One FW step. Returns (gradient, s_idx)."""
    n = len(a)
    b = fast_convolve(a)
    
    # LogSumExp weights
    scaled = b / tau
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient = 2 * cross_corr(w, a)
    g = 2.0 * fast_cross_corr(w, a)
    g = np.maximum(g, 0)
    
    s_idx = np.argmin(g)
    return g, s_idx, w


def nesterov_fw(a0, time_limit, tau, max_iters=1000000):
    """Nesterov-accelerated FW with fixed temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    
    for k in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if k > 0:
            beta = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
            gamma = (1.0 + k) / (1.0 + beta)
            y = (1.0 - gamma) * a + gamma * y
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        
        # FW step
        g, s_idx, w = fw_iteration(y, tau)
        
        t_k = 2.0 / (k + 2)
        e_s = np.zeros(n)
        e_s[s_idx] = 1.0
        a_new = (1.0 - t_k) * y + t_k * e_s
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa > 1e-15:
            a_new = a_new / sa
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Restart
        if stall >= 200:
            y = best_a.copy()
            a = best_a.copy()
            stall = 0
        
        if (k + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NFw iter={k+1} ({elapsed:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score


def build_q_dense(k, n):
    """Build dense Q_k matrix where a^T Q_k a = b_k."""
    Q = np.zeros((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return Q


def qcqp_objective(x, active_set, n):
    """Objective: t = x[-1]"""
    return x[-1]


def solve_qcqp_slsqp(a0, active_set, maxiter=500):
    """
    Solve min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0
    using SLSQP.
    """
    n = len(a0)
    a0 = np.maximum(a0, 0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / s
    
    b0 = fast_convolve(a0)
    t0 = max(b0.max(), 1e-10)
    
    x0 = np.concatenate([a0, [t0]])
    
    # Precompute Q matrices (dense for SLSQP)
    Qs = {}
    for k in active_set:
        Qs[int(k)] = build_q_dense(int(k), n)
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    constraints = []
    
    # sum(a) = 1
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - 1.0,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    })
    
    # Quadratic constraints: a^T Q_k a - t <= 0
    for k in active_set:
        Q = Qs[int(k)]
        k_int = int(k)
        
        def quad_fun(x, Q=Q, k_int=k_int):
            a = x[:n]
            return a @ Q @ a - x[-1]
        
        def quad_jac(x, Q=Q, k_int=k_int):
            a = x[:n]
            g = 2.0 * (Q @ a)
            g = np.concatenate([g, [-1.0]])
            return g
        
        constraints.append({
            'type': 'ineq',
            'fun': quad_fun,
            'jac': quad_jac
        })
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        
        if res.success or res.fun < t0:
            a_opt = np.maximum(res.x[:n], 0)
            s_opt = a_opt.sum()
            if s_opt > 1e-15:
                a_opt = a_opt / s_opt
            t_opt = max(fast_convolve(a_opt).max(), res.x[-1])
            return a_opt, float(t_opt)
    except Exception as e:
        print(f"    SLSQP failed: {e}")
    
    return None, None


def active_set_qcqp(a0, time_limit, solver='slsqp', rel_tol=1e-5, max_constraints=30):

    2. Solve QCQP
    3. Check for violated constraints
    4. Add to active set, repeat

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t0 = time.monotonic()
    active_set = set()
    best_a = a.copy()
    best_t = fast_convolve(a).max()
    
    for iteration in range(50):

        if time.monotonic() - t0 > time_limit - 30:
            break  # Leave buffer
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Identify active constraints
        threshold = max_b * (1.0 - rel_tol)
        active_candidates = np.where(b >= threshold)[0]
        
        if len(active_candidates) == 0:
            active_candidates = np.array([np.argmax(b)])
        
        # Update active set
        new_active = set(int(k) for k in active_candidates)
        
        # Limit active set size
        if len(new_active) > max_constraints:
            # Keep the top ones
            b_vals = b[active_candidates]
            top_idx = active_candidates[np.argsort(b_vals)[-max_constraints:]]
            new_active = set(int(k) for k in top_idx)

        if new_active == active_set and iteration > 3:
            # Check if we're at the optimum (relative improvement < tol)
            if abs(best_t - max_b) < rel_tol * max_b:

        if active_set:
            a_new, t_new = solve_qcqp_slsqp(a, sorted(active_set), maxiter=300)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
