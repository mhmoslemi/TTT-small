"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive spectral flattening (annealed Gerchberg-Saxton)
2. Polyak subgradient descent with peak focusing
3. Multi-temperature LogSumExp L-BFGS

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_flatten(a, n_iters=300, deadline=None, blend_start=0.5, blend_end=0.1):
    """Gerchberg-Saxton with annealed blend factor."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = normalize(a)
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Anneal blend from blend_start to blend_end
        frac = i / max(1, n_iters - 1)
        blend = blend_start + (blend_end - blend_start) * frac
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        # Vectorized magnitude update
        new_mags = (1 - blend) * mags.copy()
        new_mags[1:] = (1 - blend) * mags[1:] + blend * target
        F_new = new_mags * np.exp(1j * phases)
        # Ensure Hermitian symmetry
        # (FFT of real signal is already handled by ifft taking real part)
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a


def subgradient_descent(a0, time_limit, use_polyak=True):
    """Subgradient descent with Polyak step size and peak focusing."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = evaluate_local(a)
    lb = 2.0 * n / (2.0 * n - 1)  # theoretical lower bound
    t0 = time.monotonic()
    eta = 0.05
    no_improve = 0
    best_known = best_score
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b  # sum=1
        
        # Active set: peaks within 2% of max
        threshold = max_b * (1.0 - 0.02)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient from active peaks
        g = np.zeros(n)
        w_sum = 0.0
        for k in active:
            w = max((b[k] - threshold) / (max_b - threshold + 1e-30), 0.05)
            w_sum += w
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        g /= max(w_sum, 1e-30)
        
        # Step size
        if use_polyak and best_known > lb + 1e-6:
            gap = current_score - lb
            g_norm_sq = np.dot(g, g)
            if g_norm_sq > 1e-30:
                eta = np.clip(gap / g_norm_sq, 0.005, 0.5)
        
        # Update
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            best_known = score
            no_improve = 0
        else:
            a = a  # Don't update if no improvement
            no_improve += 1
        
        if no_improve >= 200:
            if use_polyak:
                eta = max(eta * 0.5, 0.002)
            else:
                eta *= 0.8
            no_improve = 0
            if eta < 0.001:
                break
        
        if no_improve % 10000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f} |active|={len(active)}")
    
    return best_a, best_score


def lse_obj_grad(z, n, T, deadline):
    """LogSumExp objective and gradient w.r.t. z."""
    if time.monotonic() > deadline:
        return 1e15, np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp(np.clip((b - max_b) / T, -50, 50))
    lse = max_b + T * np.log(exp_terms.sum() + 1e-300)
    obj = 2.0 * n * lse
    
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = 2.0 * n * a * (grad_a - S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    deadline = time.monotonic() + time_limit
    n_stages = len(T_stages)
    
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        stages_left = n_stages - stage_i
        iters = max(30, min(maxiter_per_stage, int(remaining * 30 / stages_left)))
        
        def func(z_in):
            return lse_obj_grad(z_in, n, stage_T, deadline)[0]
        
        def jac(z_in):
            return lse_obj_grad(z_in, n, stage_T, deadline)[1]
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-12, 'maxfun': iters * 3, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = list(dict.fromkeys([solver, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.MOSEK, cp.GUROBI]))
    for s in solvers_to_try:
        if s is None:
            continue
        try:

            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:

                b_val = b_val / sv
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None
