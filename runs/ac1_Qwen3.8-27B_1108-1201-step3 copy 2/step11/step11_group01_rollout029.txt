def lse_forward_backward(a, T):
    """Compute LSE value and gradient w.r.t. a.
    
    LSE_T = T * log(sum_k exp(b_k/T)) where b_k = (a*a)_k
    grad_i = sum_k w_k * a_{k-i} where w_k = exp(b_k/T) / sum exp(b_j/T)
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    
    # Compute softmax weights
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / exp_b.sum()
    
    lse_val = T * (b_max + np.log(exp_b.sum()))
    
    # Gradient: grad_i = sum_k w_k * a_{k-i} for valid k
    # = cross_corr(w, a) as defined above
    grad = cross_corr(w, a)
    
    return lse_val, grad, w, b


def sigmoid_param(x):
    """Stable sigmoid."""
    x = np.clip(x, -50, 50)
    return np.where(x >= 0, 1 / (1 + np.exp(-x)), np.exp(x) / (1 + np.exp(x)))


def objective_from_a(a):
    """Compute the actual objective value for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s ** 2)


def lse_adam_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_anneal_steps=2000, 
                       lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Optimize using Adam on the LSE surrogate with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))  # log parameterization
    z = z - z.sum()  # centering (doesn't change softmax)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_a = a0.copy()
    best_score = objective_from_a(a0)
    t0 = time.monotonic()
    
    for step in range(n_anneal_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current temperature
        progress = step / max(1, n_anneal_steps)
        T_cur = T_start * (T_end / T_start) ** progress
        
        # Compute a from z
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        
        # Compute LSE and gradient w.r.t. a
        lse_val, grad_a, w, b = lse_forward_backward(a, T_cur)
        
        # Gradient w.r.t. z: grad_z = a * (grad_a - a @ grad_a)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check score periodically
        if (step + 1) % 50 == 0:
            sc = objective_from_a(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
    # Final check
    z_shifted = z - z.max()
    a_final = np.exp(z_shifted) / np.exp(z_shifted).sum()
    sc_final = objective_from_a(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def lp_correction(a, solver=None, max_iters_lp=5):
    """Solve LP best response and return the direction."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build T matrix efficiently
    size = 2 * n - 1
    # T[i, j] = a_j if 0 <= i-j < n, else 0
    # (T @ b)_i = sum_j T[i,j] b_j = sum_j a_j b_{i-j} (with bounds) = (a*b)_i
    # So T is the convolution matrix of a.
    
    # Instead of building the full matrix, use the fact that T is banded
    # For LP, we need the matrix. For n=1200, T is 2399 x 1200, which is ~2.9M entries.
    # That's manageable.
    
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def game_nesterov_fast(a0, time_limit, solver=None, lp_interval=20, nesterov_iters=5000):
    """Nesterov for the game, but only solve LP every lp_interval iterations.
    Between LP calls, use gradient-based approximation of the best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = objective_from_a(a)
    t0 = time.monotonic()
    
    # Cache best response
    b_cached = None
    
    for it in range(nesterov_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP every lp_interval iterations
        if it % lp_interval == 0 or b_cached is None:
            b_new = lp_correction(y, solver)
            if b_new is not None:
                b_cached = b_new
        
        if b_cached is None:
            b_cached = a.copy()  # fallback
        
        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_cached
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_cached.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc_a = objective_from_a(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
    
    return best_a, best_score


def subgradient_mirror(a0, time_limit, n_steps=10000, eta=0.1):
    """Mirror descent (exponential weighting) for the minimax game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    z = np.log(np.maximum(a, 1e-30))
    best_a = a.copy()
    best_score = objective_from_a(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient of max(conv(a,a)) w.r.t. a
        b = fast_convolve(a)
        max_b = b.max()
        # Active set: positions where b is close to max
        active = np.where(b >= max_b * 0.9)[0]
        
        # Subgradient: average of a_{k-i} over active k
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(active)
        
        # Mirror descent update in log-space
        S = np.dot(grad, a)
        grad_z = a * (grad - S)
        
        # Decrease learning rate
        eta_cur = eta / (1 + step * 0.001)
        z = z - eta_cur * grad_z
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        
        sc = objective_from_a(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        # Always update z for exploration
        # z = z - eta_cur * grad_z  (already done above)
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * 5 * np.arange(n) / n))
    elif choice == 5:
        x2 = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x2) + 0.15 * np.cos(7 * x2)
    elif choice == 6:
        # Triangular
        a = 1 - 2 * np.abs(x - 0.5)
    else:
        # Flat with edges
        a = np.ones(n)
        a[:n//10] *= 0.5
        a[-n//10:] *= 0.5
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LogSumExp-Adam optimizer with FFT-based gradients (fast iterations)
2. Cutting-plane LP-based Nesterov for the zero-sum game (principled, powerful)
3. Multi-temperature annealing for LSE-Adam
4. Multi-length exploration
Key improvement: cutting-plane LP is much faster than full LP, enabling more Nesterov iterations.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute full auto-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolution(a, b):
    """Compute cross-convolution a*b using FFT. Result has length len(a)+len(b)-1."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def lse_grad_wrt_a(a, T):
    """Compute LSE_T value and its gradient w.r.t. a.
    
    LSE_T = T * log(sum_k exp(b_k/T)) where b_k = (a*a)_k
    grad_a_i = sum_k w_k * a_{k-i} where w_k = exp(b_k/T) / sum exp(b_j/T)
    The gradient is the cross-correlation of w and a: grad_i = sum_j w_{i+j} * a_j
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / exp_b.sum()
    
    lse_val = T * (b_max + np.log(exp_b.sum()))
    
    # grad_i = sum_j a_j * w_{i+j} for j=0..n-1
    # This equals conv(a_reversed, w)[i + n - 1]
    a_rev = a[::-1]
    conv_result = cross_convolution(a_rev, w)
    grad = conv_result[n - 1:2 * n - 1]
    
    return lse_val, grad, w, b


def objective_from_a(a):
    """Compute the actual objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s ** 2)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:

        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def lse_adam_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_steps=10000,
                      lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """Optimize using Adam on the LSE surrogate with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)).copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_a = a0.copy()
    best_score = objective_from_a(a0)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = step / max(1, n_steps - 1)
        T_cur = T_start * (T_end / T_start) ** progress
        
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        
        lse_val, grad_a, w, b = lse_grad_wrt_a(a, T_cur)
        
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if (step + 1) % 200 == 0:
            sc = objective_from_a(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    a_final = softmax_from_z(z)
    sc_final = objective_from_a(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_cutting_plane(a_fixed, solver=None, max_rounds=100, tol=1e-7):
    """Solve the best-response LP using cutting plane method.
    
    min M s.t. (a*b)_k <= M for all k, sum(b)=1, b>=0
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    base_constraints = [cp.sum(b_var) == 1]
    active_constraints = []
    
    # Start with a few initial constraints (around the center where convolution is usually max)
    for k_init in [n - 1, n, n - 2]:
        if 0 <= k_init < 2 * n - 1:
            row = np.zeros(n)
            js = max(0, k_init - n + 1)
            je = min(n - 1, k_init)
            for j in range(js, je + 1):
                row[j] = a_fixed[k_init - j]
            active_constraints.append(row @ b_var <= M_var)
    
    b_val = None
    M_val = 0.0
    
    for round_num in range(max_rounds):
        prob = cp.Problem(cp.Minimize(M_var), base_constraints + active_constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    break
                except Exception:
                    continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if b_val is not None:
                break
            return None, None
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        M_val = float(M_var.value)
        
        # Compute cross-convolution a_fixed * b_val
        cross = cross_convolution(a_fixed, b_val)
        max_cross = cross.max()
        
        if max_cross - M_val < tol:
            break
        
        # Find most violated constraint and add it
        k_violated = int(np.argmax(cross))
        row = np.zeros(n)
        js = max(0, k_violated - n + 1)
        je = min(n - 1, k_violated)
        for j in range(js, je + 1):
            row[j] = a_fixed[k_violated - j]
        active_constraints.append(row @ b_var <= M_var)
    
    if b_val is None:
        return None, None
    return b_val, M_val


def nesterov_lp_main(a0, time_limit, solver=None, max_iters=100000, lp_interval=1):
    """Nesterov acceleration for the zero-sum game with LP best responses."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = objective_from_a(a)
    t0 = time.monotonic()
    last_print = t0
    b_cached = None
    it_since_lp = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP periodically
        if it % lp_interval == 0 or b_cached is None:
            b_new, M_lp = solve_lp_cutting_plane(y, solver, max_rounds=50)
            if b_new is not None:
                b_cached = b_new
            it_since_lp = 0
        else:
            it_since_lp += 1
            if it_since_lp > 5 and b_cached is None:
                b_cached = y.copy()
        
        if b_cached is None:
            b_cached = y.copy()
        
        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_cached
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_cached.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc_a = objective_from_a(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
    
    return best_a, best_score


def subgradient_optimize(a0, time_limit, n_steps=100000, eta_start=0.3, eta_end=0.005):
    """Subgradient descent in log-space with decay."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-30)).copy()
    
    best_a = a.copy()
    best_score = objective_from_a(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        active = np.where(b >= max_b * 0.95)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(active)
        
        S = np.dot(grad, a)
        grad_z = a * (grad - S)
        
        eta = eta_start * (eta_end / eta_start) ** (step / max(1, n_steps))
        z = z - eta * grad_z
        a_new = softmax_from_z(z)
        
        sc = objective_from_a(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_score


def structured_init(n, rng):

        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * np.arange(n) / n)

        x2 = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x2) + 0.15 * np.cos(7 * x2)
    elif choice == 6:

    else:
        a = np.ones(n)
        edge = max(1, n // 15)
        a[:edge] *= 0.5

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def try_smooth_polish(a0, time_limit):
    """Try L-BFGS on LSE for final polishing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)).copy()
    t0 = time.monotonic()
    
    T = 1.0
    
    def func(z_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_val)
        lse_val, _, _, _ = lse_grad_wrt_a(a, T)
        return 2.0 * n * lse_val
    
    def jac(z_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_val)
        a = softmax_from_z(z_val)
        _, grad_a, _, _ = lse_grad_wrt_a(a, T)
        S = np.dot(grad_a, a)
        return 2.0 * n * a * (grad_a - S)
    
    try:
        res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 200, 'ftol': 1e-20, 'gtol': 1e-16, 'disp': False})
        a_res = softmax_from_z(res.x)
        return a_res
    except Exception:
        return None

    try_update(a_out, "lse_adam_1")
    print(f"  Phase 1 done: best_score={best_score:.8f}")
    
    # === Phase 2: Cutting-plane Nesterov LP (35% of budget) ===
    print(f"\n=== Phase 2: Cutting-plane Nesterov LP n={n_base} (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(0.35 * (budget_s - 30), remaining * 0.7)
    if t2 > 60:
        a_out2, sc2 = nesterov_lp_main(best_seq, t2, solver=solver, max_iters=200000, lp_interval=1)
        try_update(a_out2, "nesterov_lp_1")
    print(f"  Phase 2 done: best_score={best_score:.8f}")
    
    # === Phase 3: LSE-Adam round 2 with lower T (15% of budget) ===
    print(f"\n=== Phase 3: LSE-Adam round 2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 30), remaining * 0.6)
    if t3 > 15:
        a_out3, sc3 = lse_adam_multi(best_seq, t3,
                                       T_schedule=[(10.0, 1.0, 8000), (1.0, 0.1, 15000)],
                                       lr_schedule=[0.03, 0.01])
        try_update(a_out3, "lse_adam_2")
    
    # L-BFGS polish
    remaining = _global_deadline - time.monotonic()
    if remaining > 10:
        a_lbfgs = try_smooth_polish(best_seq, min(remaining * 0.3, 60.0))
        try_update(a_lbfgs, "lbfgs_polish")
    
    print(f"  Phase 3 done: best_score={best_score:.8f}")
    
    # === Phase 4: Nesterov LP round 2 (15% of budget) ===
    print(f"\n=== Phase 4: Nesterov LP round 2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(0.15 * (budget_s - 30), remaining * 0.6)
    if t4 > 30:
        a_out4, sc4 = nesterov_lp_main(best_seq, t4, solver=solver, max_iters=100000, lp_interval=1)
        try_update(a_out4, "nesterov_lp_2")
    print(f"  Phase 4 done: best_score={best_score:.8f}")
    
    # === Phase 5: Subgradient + LSE polish (5% of budget) ===
    print(f"\n=== Phase 5: Subgradient polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(0.05 * (budget_s - 30), remaining * 0.5)
    if t5 > 10:
        a_out5, sc5 = subgradient_optimize(best_seq, t5, n_steps=500000, eta_start=0.2, eta_end=0.003)
        try_update(a_out5, "subgrad")
        # Quick LSE after subgrad
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            a_out5b, sc5b = lse_adam_optimize(best_seq, min(remaining * 0.4, 30.0),
                                                T_start=1.0, T_end=0.05, n_steps=5000, lr=0.01)
            try_update(a_out5b, "lse_after_subgrad")
    print(f"  Phase 5 done: best_score={best_score:.8f}")
    
    # === Phase 6: Multi-length exploration (10% of budget) ===
    print(f"\n=== Phase 6: Length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6_total = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t6_total > 30:
        lengths_to_try = [n_base + 100, n_base - 100, n_base + 200, n_base - 200, n_base + 50]
        n_attempts = min(len(lengths_to_try), max(1, int(t6_total // 15)))
        for i in range(n_attempts):
            remaining = _global_deadline - time.monotonic()
            if remaining < 10:
                break
            n_try = lengths_to_try[i % len(lengths_to_try)]
            if n_try <= 0:
                continue
            t6 = min(t6_total / n_attempts, remaining * 0.4)
            if t6 < 5:
                break
            print(f"  Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick LSE-Adam then Nesterov
            a_try, _ = lse_adam_optimize(init_try, t6 * 0.4, T_start=20.0, T_end=1.0,
                                          n_steps=3000, lr=0.05)
            a_try2, _ = nesterov_lp_main(a_try, t6 * 0.5, solver=solver, max_iters=50000)
            try_update(a_try2, f"len_{n_try}")
            try_update(a_try, f"len_{n_try}_lse")
    
    # === Final: one more LSE polish ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 10:
        t_final = min(remaining * 0.5, 30.0)
        if t_final > 5:
            a_final, _ = lse_adam_optimize(best_seq, t_final, T_start=1.0, T_end=0.05,
                                            n_steps=5000, lr=0.005)
            try_update(a_final, "final_polish")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
