```python
def solve_qp(Q, n, active_set):
    """Solve min a^T Q a s.t. sum(a)=1, a[i]=0 for i not in active_set."""
    m = len(active_set)
    Q_sub = Q[np.ix_(active_set, active_set)]
    # KKT: 2*Q_sub*a_sub = mu * 1
    # sum(a_sub) = 1
    # Solve: Q_sub * a_sub = (mu/2) * 1
    # From the first eq: a_sub = (mu/2) * Q_sub^{-1} * 1
    # From the second: (mu/2) * 1^T * Q_sub^{-1} * 1 = 1
    # So mu/2 = 1 / (1^T * Q_sub^{-1} * 1)
    try:
        v = np.linalg.solve(Q_sub, np.ones(m))
        mu_half = 1.0 / np.sum(v)
        a_sub = mu_half * v
    except np.linalg.LinAlgError:
        return None
    if np.any(a_sub < -1e-10):
        return None  # Active set is wrong
    a_sub = np.maximum(a_sub, 0)
    a_sub = a_sub / a_sub.sum()
    a = np.zeros(n)
    a[active_set] = a_sub
    return a
```

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Flat-convolution-inspired initialization (generating function (1-z)^-1/2)
2. FFT-accelerated LSE-smoothed gradient with geometric temperature continuation
3. Log-parameterization for natural non-negativity
4. Direct subgradient max-reduction polish

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT. Input length n, output length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_full(a, b):
    """Full linear convolution of two arrays via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]


def evaluate_local(a):
    """Evaluate the objective locally (same as evaluate_sequence but on numpy array)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def cross_corr_via_conv(a, w):
    """Compute cross-correlation g[i] = sum_j a[j]*w[i+j] for i=0..len(a)-1.
    Uses the identity: g[i] = conv(a, w_rev)[2*len(w)-2-i]
    """
    n_a = len(a)
    n_w = len(w)
    w_rev = w[::-1]
    full = convolve_full(a, w_rev)  # length n_a + n_w - 1
    # g[i] = full[n_w - 1 - i + n_a - 1] ... let me re-derive
    # conv(a, w_rev)[k] = sum_j a[j] * w_rev[k-j] = sum_j a[j] * w[n_w-1-k+j]
    # We want g[i] = sum_j a[j] * w[i+j]
    # Setting n_w-1-k+j = i+j => k = n_w-1-i
    # So g[i] = conv(a, w_rev)[n_w-1-i]
    # For i=0: g[0] = full[n_w-1]
    # For i=n_a-1: g[n_a-1] = full[n_w-1-(n_a-1)] = full[n_w-n_a]
    # Since n_w = 2n-1 and n_a = n: indices from n_w-1=2n-2 down to n_w-n_a = n-1
    g = full[n_w - 1 - np.arange(n_a)]
    return g


def lse_obj_grad_logparam(z, T):
    """Compute LSE objective and gradient in log-parameterization.
    z: log-space parameters, a = softmax(z)
    Returns (obj, grad_z)
    """
    n = len(z)
    # Compute a from z
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    # Compute convolution
    c = fast_convolve(a)  # length 2n-1
    c_max = c.max()
    scaled = (c - c_max) / T
    exp_scaled = np.exp(scaled)
    lse = c_max + T * np.log(exp_scaled.sum())
    obj = 2.0 * n * lse

    # LSE weights
    w = exp_scaled / exp_scaled.sum()  # length 2n-1

    # Gradient w.r.t. a: g_a[i] = 2 * sum_j a[j] * w[i+j]
    g_a = 2.0 * cross_corr_via_conv(a, w)

    # Transform to z-space: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(g_a, a)
    grad_z = a * (g_a - S)

    return obj, grad_z


def direct_subgrad_score(a):
    """Compute the exact objective and the subgradient at the argmax.
    Returns (score, subgrad) where subgrad is gradient of (a*a)[k*] at a.
    """
    n = len(a)
    c = fast_convolve(a)
    k_star = int(np.argmax(c))
    score = 2.0 * n * c[k_star] / (a.sum() ** 2)
    # Gradient of (a*a)[k*] w.r.t. a[i] = 2*a[k*-i] if valid
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]
    return score, g, k_star


def flat_conv_init_left(n):
    """Initialize from flat-convolution equations: b[k] = C(2k,k)/4^k."""
    b = np.zeros(n)
    b[0] = 1.0
    for k in range(1, n):
        s = 0.0
        for j in range(1, k):
            s += b[j] * b[k - j]
        b[k] = (1.0 - s) / 2.0
    if np.any(b < -1e-10):
        b = np.maximum(b, 0)
        if b.sum() < 1e-10:
            return None
    return normalize(b)


def flat_conv_init_symmetric(n):
    """Symmetric version: b[i] = C(2*min(i,n-1-i), min(i,n-1-i)) / 4^min(i,n-1-i)."""
    m = (n - 1) // 2
    b = np.zeros(n)
    for i in range(n):
        d = min(i, n - 1 - i)
        # C(2d, d) / 4^d computed iteratively to avoid overflow
        if d == 0:
            b[i] = 1.0
        else:
            # Use log to avoid overflow: log(C(2d,d)/4^d) = log(C(2d,d)) - d*log(4)
            # C(2d,d) = (2d)!/(d!)^2
            # For large d, use Stirling or iterative
            val = 1.0
            for j in range(d):
                val *= (d + j + 1) / (j + 1) / 4.0
            b[i] = val
    return normalize(b)


def diverse_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian bump (center)
    a = np.exp(-4 * (x - 0.5) ** 2)
    inits.append(normalize(a))

    # U-shape (edges boosted)
    a = np.exp(-4 * (x - 0.5) ** 2) ** 0.5 + 0.5
    a2 = 1.0 / (1.0 + 20 * (x - 0.5) ** 2)
    inits.append(normalize(a2))

    # Inverted U (center boosted)
    a = np.exp(-2 * (x - 0.5) ** 2)
    inits.append(normalize(a))

    # Cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(normalize(a))

    # Sine squared
    a = np.sin(np.pi * x) ** 2
    inits.append(normalize(a))

    # Two bumps
    a = np.exp(-12 * (x - 0.3) ** 2) + np.exp(-12 * (x - 0.7) ** 2)
    inits.append(normalize(a))

    # Flat conv left-edge
    fc = flat_conv_init_left(n)
    if fc is not None:
        inits.append(fc)

    # Flat conv symmetric
    fc_s = flat_conv_init_symmetric(n)
    if fc_s is not None:
        inits.append(fc_s)

    # Power law edge
    d = np.minimum(x, 1 - x)
    a = d ** 0.3 + 1e-10
    inits.append(normalize(a))

    # Random smooth
    coeffs = rng.normal(0, 1, 8)
    a = np.zeros(n)
    for i, cv in enumerate(coeffs):
        a += cv * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 0.5, 0)
    inits.append(normalize(a))

    return inits

    for stage_i, T in enumerate(T_stages):

        remaining = time_budget - elapsed
        if remaining < 5:

        # Determine iterations for this stage

            obj, _ = lse_obj_grad_logparam(z_in, T)
            return obj

            _, g = lse_obj_grad_logparam(z_in, T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception as e:
            pass

        # Evaluate and update best
        a_new = _softmax(z)
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        if stage_i % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {stage_i}/{len(T_stages)}: T={T:.4f}, score={best_score:.8f}, time={elapsed:.1f}s")

    return best_a, best_score


def _softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def subgradient_polish(a0, time_budget, lr_init=0.001):
    """Direct subgradient steps targeting the convolution peak."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = lr_init

    while time.monotonic() - t0 < time_budget:
        score, g, k_star = direct_subgrad_score(a)
        if not np.isfinite(score):
            break

        # Project gradient onto tangent space of simplex (sum=1)
        g_tan = g - g.mean()
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break

        # Try multiple step sizes
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * scale
            a_new = a - step * g_tan / g_norm
            a_new = normalize(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break
        else:
            lr = min(lr * 1.05, lr_init)

    return best_a, best_score


def spectral_flatten(a0, time_budget, n_iters=5000):
    """Flatten the power spectrum to reduce autocorrelation/convolution peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    lr = 0.003

    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        # Direction to flatten P
        r = (P - P_mean) / (P_mean + 1e-15)
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -r[valid] * A[valid] / (P[valid] + 1e-10)
        H[0] = 0.0  # preserve DC
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        improved = False
        for scale in [lr, lr * 0.5, lr * 0.1]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if it % 500 == 0 and it > 0:
            lr *= 0.8

    return best_a, best_score


def multiplicative_update(a0, time_budget, n_steps=50000, eta=0.03):
    """Multiplicative weights / mirror descent on the simplex."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # LSE weights (soft)
        T_local = 0.01
        w = np.exp((c - c_max) / T_local)
        w = w / w.sum()

        # Gradient direction
        g = 2.0 * cross_corr_via_conv(a, w)
        g_tan = g - g.mean()

        # Multiplicative update in log-space
        z = np.log(np.maximum(a, 1e-30))
        z_new = z - eta * g_tan
        z_new -= z_new.max()
        a_new = np.exp(z_new)
        a_new = a_new / a_new.sum()

        sc = evaluate_local(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def coordinate_perturb(a, rng, n_perturb=5, strength=0.01):
    """Perturb a few random coordinates."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_perturb):
        idx = rng.integers(0, n)
        a_new[idx] += rng.normal(0, strength)
    return normalize(a_new)


def block_perturb(a, rng, block_size=20, strength=0.02):
    """Perturb a random block."""
    n = len(a)
    a_new = a.copy()
    start = rng.integers(0, n - block_size)
    a_new[start:start + block_size] += rng.normal(0, strength, block_size)
    return normalize(a_new)
