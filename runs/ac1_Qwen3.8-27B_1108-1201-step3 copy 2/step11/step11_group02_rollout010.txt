import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix T (2n-1 x n) where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    col_idx = np.repeat(np.arange(n), n)
    offsets = np.tile(np.arange(n), n)
    row_idx = col_idx + offsets
    data = a[offsets]
    return csr_matrix((data, (row_idx, col_idx)), shape=(size, n))

def solve_lp_response(T, n, solver, warm_b=None):
    """Solve min t s.t. T@b <= t, b>=0, sum(b)=1. Returns (b, t)."""
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    if warm_b is not None:
        try:
            b_var.value = warm_b
            t_var.value = float(np.max(T @ warm_b))
        except Exception:
            pass
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    M = float(np.max(T @ b_val))
    return b_val, M


def nesterov_lp(a0, time_limit, solver, rng=None, verbose=True):
    """Nesterov-accelerated LP alternating with adaptive exploration."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a /= s
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0
    b_prev = None
    stall_threshold = 6

    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        T = build_conv_matrix(y)
        b_new, M = solve_lp_response(T, n, solver, warm_b=b_prev)
        if b_new is None:
            continue
        b_prev = b_new

        alpha = 2.0 / (np.sqrt(it + 1.0) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy

        w1 = float(it + 1)
        w2 = alpha * float(it + 2)
        a = (w1 * a + w2 * b_new) / (w1 + w2)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa

        score_a = eval_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        score_y = eval_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 20 == 0:
            stall += 1
            if stall >= stall_threshold:
                b_cur = fast_convolve(y)
                peak_idx = int(np.argmax(b_cur))
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                idx = np.arange(js, je + 1)
                contribs = y[idx] * y[peak_idx - idx]
                n_red = max(5, len(idx) // 6)
                top_pos = idx[np.argsort(contribs)[::-1][:n_red]]
                for pos in top_pos:
                    if y[pos] > 1e-10:
                        y[pos] *= (1.0 - 0.08 * np.sqrt(stall / stall_threshold))
                y += rng.standard_normal(n) * 0.015
                y = np.maximum(y, 0)
                y /= y.sum()
                a = y.copy()
                b_prev = None
                stall = 0
                stall_threshold = min(25, stall_threshold + 1)

            if verbose and (it + 1) % 500 == 0:
                print(f"    LP it={it+1} t={time.monotonic()-t0:.0f}s sc={best_score:.8f}")

    return best_a, best_score


def lse_score_and_grad(a, T):
    """Compute (2n/T)*LSE(conv(a,a)/T) and its gradient."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    lse_val = max_b / T + np.log(exp_b.sum())
    obj = (2.0 * n / T) * lse_val

    size = 2 * n - 1
    grad = np.zeros(n)
    for i in range(n):
        d_max = min(n, size - i)
        grad[i] = np.dot(w[i:i + d_max], a[:d_max])
    grad *= (4.0 * n) / T
    return obj, grad


def lse_gradient_descent(a0, time_limit, T_stages=None, verbose=True):
    """Multi-stage LogSumExp smoothed gradient descent (L-BFGS-B)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a /= s
    best_a = a.copy()
    best_real = eval_score(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1]

    n_stages = len(T_stages)
    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit - 3:
            break
        remaining_frac = 1.0 - si / n_stages
        maxiter = int(max(20, 3000 * remaining_frac))

        def func(z):
            if time.monotonic() - t0 > time_limit - 1:
                return 1e15
            at = np.maximum(z, 0)
            st = at.sum()
            if st < 1e-15:
                return 1e15
            at = at / st
            obj, _ = lse_score_and_grad(at, T)
            return obj

        def jac(z):
            if time.monotonic() - t0 > time_limit - 1:
                return np.zeros(len(z))
            at = np.maximum(z, 0)
            st = at.sum()
            if st < 1e-15:
                return np.zeros(len(z))
            at = at / st
            _, g = lse_score_and_grad(at, T)
            return g

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxiter * 100})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a /= sa
        except Exception:
            pass

        sc = eval_score(a)
        if sc < best_real:
            best_real = sc
            best_a = a.copy()
        if verbose:
            print(f"    LSE T={T:.2f} t={time.monotonic()-t0:.0f}s sc={best_real:.8f}")

    return best_a, best_real


def coord_descent(a0, time_limit, verbose=True):

    n = len(a0)

    for step in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9995
        peaks = np.where(b >= threshold)[0][:3]

        improved = False
        for pk in peaks:
            if improved:
                break
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_m = min(10, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_pos:

                    continue
                for factor in [0.999, 0.998, 0.995, 0.99, 0.985, 0.98]:
                    a_t = a.copy()
                    a_t[pos] *= factor
                    diff = a[pos] - a_t[pos]
                    if diff < 1e-18:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_t[mask]
                    os_sum = others.sum()
                    if os_sum < 1e-15:
                        break
                    a_t[mask] = others + diff * others / os_sum
                    a_t /= a_t.sum()
                    sc = eval_score(a_t)
                    if sc < best_score - 1e-15:

                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[pk - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:5]]

                for pos in top_pos:
                    if a_t[pos] > 1e-14:
                        a_t[pos] *= 0.999
                a_t = np.maximum(a_t, 0)
                a_t /= a_t.sum()
                sc = eval_score(a_t)
                if sc < best_score - 1e-15:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    stall = 0

        if verbose and (step + 1) % 5000 == 0:

def interpolate_to_n(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        a = np.exp(-2.0 * x) * (1 + 0.5 * np.sin(6 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LP alternation, multi-n exploration, LSE gradient, and CD polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    solver = get_solver()
    print(f"Solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = eval_score(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick Nesterov LP ===
    print(f"\n=== Phase 1: Quick LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s, n={len(best_seq)}")
        a1, _ = nesterov_lp(best_seq, t1, solver, rng=rng, verbose=True)
        try_update(a1, "lp_p1")

    # === Phase 2: Multi-n Exploration ===
    print(f"\n=== Phase 2: Multi-n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if t2 > 30:
        n_best = len(best_seq)
        candidates = sorted(set([
            max(600, n_best - 200),
            max(800, n_best + 200),
            max(600, n_best + 500),
            max(500, n_best // 2 + 100),
            n_best
        ]))
        per_n = t2 / max(1, len(candidates))
        print(f"  n values: {candidates}, ~{per_n:.0f}s each")
        for n_try in candidates:
            if time.monotonic() > deadline - 10:
                break
            if n_try == n_best:
                continue
            a_interp = interpolate_to_n(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, {t_sub:.0f}s")
            a_sub, _ = nesterov_lp(a_interp, t_sub, solver, rng=rng, verbose=False)
            try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 3: Deep LP Refinement ===
    print(f"\n=== Phase 3: Deep LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.35, 500.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
        a3, _ = nesterov_lp(best_seq, t3, solver, rng=rng, verbose=True)
        try_update(a3, "deep_lp")

    # === Phase 4: LSE Gradient Descent ===
    print(f"\n=== Phase 4: LSE Gradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 300.0)
    if t4 > 20:
        T_stages = [25.0, 12.0, 6.0, 3.5, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1]
        print(f"  Budget: {t4:.0f}s")
        a4, _ = lse_gradient_descent(best_seq, t4, T_stages=T_stages, verbose=True)
        try_update(a4, "lse_grad")

    # === Phase 5: Another LP round after gradient ===
    print(f"\n=== Phase 5: LP after gradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 100.0)
    if t5 > 10:
        a5, _ = nesterov_lp(best_seq, t5, solver, rng=rng, verbose=False)
        try_update(a5, "lp_p5")

    # === Phase 6: Coordinate Descent Polish ===
    print(f"\n=== Phase 6: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.07, 80.0)
    if t6 > 10:
        a6, _ = coord_descent(best_seq, t6, verbose=True)
        try_update(a6, "coord_desc")

    # Final verification
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
