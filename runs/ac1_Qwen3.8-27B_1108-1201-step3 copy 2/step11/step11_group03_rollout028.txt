2. Periodic LP best-response with Nesterov acceleration - coarse correction

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def subgradient_at_k(a, k):
    """Compute subgradient of (a*a)[k] w.r.t. a. O(n)."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        g[j] += 2.0 * a[k - j]
    return g


def mirror_descent_optimize(a0, time_limit, max_iters, eta_init=0.5, halve_factor=0.7,
                            lp_every=50000, lp_solver=None, n_lp_steps=30):
    """
    Monotone mirror descent with subgradient and periodic LP correction.
    Uses multiplicative updates: a_new[i] = a[i] * exp(-eta * g[i]) / Z
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_b = fast_convolve(a).max()
    best_score = 2.0 * n * best_b
    t0 = time.monotonic()
    eta = eta_init
    n_steps = 0
    y = a.copy()  # Nesterov-like momentum variable for LP phase
    lp_count = 0

    while n_steps < max_iters:
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        k_max = int(np.argmax(b))
        g = subgradient_at_k(a, k_max)

        # Mirror descent step
        log_ratio = -eta * g
        log_ratio -= log_ratio.max()
        a_trial = a * np.exp(log_ratio)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            eta *= halve_factor
            n_steps += 1
            continue
        a_trial = a_trial / s_trial

        # Check improvement
        b_trial = fast_convolve(a_trial)
        max_b_trial = b_trial.max()

        if max_b_trial < max_b - 1e-15:
            a = a_trial
            if max_b_trial < best_b:
                best_b = max_b_trial
                best_score = 2.0 * n * best_b
                best_a = a.copy()
            eta = min(eta * 1.02, 2.0)  # slightly increase step size on success
        else:
            eta *= halve_factor
            if eta < 1e-8:
                # Try perturbation
                rng = np.random.default_rng(n_steps)
                a_pert = a + 0.01 * (rng.random(n) - 0.5)
                a_pert = np.maximum(a_pert, 0)
                sp = a_pert.sum()
                if sp > 1e-15:
                    a_pert = a_pert / sp
                    bp = fast_convolve(a_pert).max()
                    if bp < best_b:
                        a = a_pert
                        best_a = a.copy()
                        best_b = bp
                        best_score = 2.0 * n * bp
                eta = eta_init * 0.5

        n_steps += 1

        # Periodic LP correction
        if lp_every > 0 and lp_solver is not None and n_steps % lp_every == 0:
            for _ in range(n_lp_steps):
                if time.monotonic() - t0 > time_limit:
                    break
                b_new, M_b = solve_lp_best_response(y, lp_solver)
                if b_new is None:
                    continue
                alpha = 2.0 / (lp_count + 2)
                y = (1.0 - alpha) * y + alpha * b_new
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy < 1e-15:
                    y = a.copy()
                else:
                    y = y / sy
                a = ((lp_count + 1) * a + alpha * (lp_count + 2) * b_new) / (lp_count + 1 + alpha * (lp_count + 2))
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa < 1e-15:
                    a = y.copy()
                else:
                    a = a / sa
                lp_count += 1
                score_a = 2.0 * n * fast_convolve(a).max()
                score_y = 2.0 * n * fast_convolve(y).max()
                if score_a < best_score:
                    best_score = score_a
                    best_a = a.copy()
                    best_b = best_score / (2.0 * n)
                if score_y < best_score:
                    best_score = score_y
                    best_a = y.copy()
                    best_b = best_score / (2.0 * n)
            y = a.copy()

        if n_steps % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={n_steps} ({elapsed:.0f}s): score={best_score:.10f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 100 and np.mean(lp_times[-100:]) > 5.0:
            print(f"    LP too slow ({np.mean(lp_times[-100:]):.2f}s/iter), stopping")
            break

    return best_a, best_score


def solve_iterative_active_qcqp(a, solver, max_rounds=20, initial_tol=1e-3, final_tol=1e-7):
    """
    Iterative active-set QCQP with active set refinement.
    Repeatedly solves min t s.t. a^T Q_k a <= t for k in active_set,
    then checks for violated constraints and expands active set.
    """
    n = len(a)
    a_cur = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a_cur.sum()
    if s < 1e-15:
        return a
    a_cur = a_cur / s

    b = fast_convolve(a_cur)
    max_b = b.max()

    # Initial active set: positions within initial_tol of max
    threshold = max_b * (1.0 - initial_tol)
    active = list(np.where(b >= threshold)[0])
    if not active:
        active = [int(np.argmax(b))]

    best_a = a_cur.copy()
    best_t = max_b

    for round_num in range(max_rounds):
        # Solve QCQP with current active set
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        solved = False
        for s2 in solvers_to_try:
            if s2 is None:
                continue
            try:
                prob.solve(solver=s2)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.asarray(a_var.value, dtype=np.float64)
                    a_val = np.maximum(a_val, 0)
                    sv = a_val.sum()
                    if sv < 1e-15:
                        continue
                    a_val = a_val / sv
                    t_val = float(max(0, t_var.value))

                    # Check if better
                    b_val = fast_convolve(a_val)
                    actual_max = b_val.max()
                    if actual_max < best_t:
                        best_t = actual_max
                        best_a = a_val.copy()
                    a_cur = a_val
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        # Check for violated constraints
        b_check = fast_convolve(a_cur)
        t_cur = t_val if 't_val' in dir() else b_check.max()
        violations = np.where(b_check > t_cur + final_tol)[0]

        if len(violations) == 0:
            # No violations - try to tighten
            b_final = fast_convolve(a_cur)
            final_max = b_final.max()
            if final_max < best_t:
                best_t = final_max
                best_a = a_cur.copy()
            break

        # Add most violated constraint to active set
        worst = int(violations[np.argmax(b_check[violations])])
        if worst not in active:
            active.append(worst)

        # If active set is too large, keep only the top ones
        if len(active) > 30:
            b_now = fast_convolve(a_cur)
            active = list(sorted(active, key=lambda k: -b_now[k])[:30])

        if round_num % 5 == 0:
            print(f"    QCQP round {round_num}: |active|={len(active)}, t={t_cur:.10f}, violations={len(violations)}")

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def lse_func(z, n=n, T_temp=stage_T, deadline=deadline):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z, n, T_temp)
            return obj

        def lse_jac(z, n=n, T_temp=stage_T, deadline=deadline):
            if time.monotonic() > deadline:
                return np.zeros_like(z)
            _, g = logsumexp_obj_grad(z, n, T_temp)
            return g

        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def diverse_init(n, mode=0, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if mode == 0:
        f = np.cos(np.pi * x / 2) ** 2
    elif mode == 1:
        f = np.cos(np.pi * x) ** 2 + 0.1
    elif mode == 2:
        f = np.exp(-4 * (x - 0.5) ** 2)
    elif mode == 3:
        f = 0.5 + 0.4 * np.sin(3 * np.pi * x) * np.cos(np.pi * x / 2)
    elif mode == 4:
        f = np.ones(n) / n  # uniform
    elif mode == 5:
        f = rng.random(n)
    elif mode == 6:
        f = (1 + np.cos(2 * np.pi * x)) / 2  # bump at edges
    elif mode == 7:
        f = np.sin(np.pi * x) ** 2
    else:
        f = np.cos(np.pi * x / 2) ** 2
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    return f


def solve_single_qp(a, k_active, solver):
    """Solve min a^T Q_k a s.t. sum(a)=1, a>=0 for a single active position."""
    n = len(a)
    Q = build_q_sparse(int(k_active), n)
    a_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.quad_form(a_var, Q, assume_PSD=True)),
                      [cp.sum(a_var) == 1])
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None
                return a_val / s_sum
        except Exception:
            continue
    return None


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Key innovations:
    - Mirror descent as primary fast optimizer (O(n log n) per iter)
    - Iterative active-set QCQP with refinement loop
    - Wide n-sweep (1000-1500)
    - Multiple diverse initializations
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = diverse_init(n_init, mode=0, seed=42)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Mirror descent from multiple diverse inits ===
    print(f"\n=== Phase 1: Mirror Descent Multi-Start (current best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_per_init = min(rem * 0.30 / 4, 120.0)

    inits = [best_seq]
    # Add a few diverse inits at the same n
    for mode in [3, 6, 7]:
        inits.append(diverse_init(n_best, mode=mode, seed=seed + mode))

    for idx, init in enumerate(inits):
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        t_alloc = min(t_per_init, rem * 0.25)
        if t_alloc < 5:
            break
        print(f"  MD init {idx} (score_{init}={evaluate_local(init):.10f}), budget={t_alloc:.0f}s")
        a_md, sc = mirror_descent_optimize(init, t_alloc, max_iters=5000000,
                                           eta_init=0.3, halve_factor=0.75, lp_every=0)
        update_best(a_md, f"md_init{idx}")

    # === Phase 2: Mirror descent + LP correction (combined) ===
    print(f"\n=== Phase 2: MD + LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md_lp, sc = mirror_descent_optimize(best_seq, t_alloc, max_iters=3000000,
                                              eta_init=0.3, halve_factor=0.75,
                                              lp_every=30000, lp_solver=solver, n_lp_steps=20)
        update_best(a_md_lp, "md_lp")

    # === Phase 3: Pure LP alternating (from best) ===
    print(f"\n=== Phase 3: LP Alternating (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000)
        update_best(a_lp, "lp_phase3")

    # === Phase 4: Iterative active-set QCQP ===
    print(f"\n=== Phase 4: Iterative Active QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.12, 120.0)
    if t_qcqp > 10:
        q_start = time.monotonic()
        # Wide active set
        a_q1 = solve_iterative_active_qcqp(best_seq, solver, max_rounds=15,
                                           initial_tol=2e-3, final_tol=1e-7)
        update_best(a_q1, "qcqp_wide")
        # Narrow active set
        a_q2 = solve_iterative_active_qcqp(best_seq, solver, max_rounds=15,
                                           initial_tol=1e-4, final_tol=1e-8)
        update_best(a_q2, "qcqp_narrow")
        # Single QP on argmax
        b_cur = fast_convolve(best_seq)
        k_max = int(np.argmax(b_cur))
        a_qp = solve_single_qp(best_seq, k_max, solver)
        update_best(a_qp, "single_qp")
        print(f"  QCQP time: {time.monotonic() - q_start:.1f}s")

    # === Phase 5: LP round 2 from polished ===
    print(f"\n=== Phase 5: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 15:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        update_best(a_lp2, "lp_round2")

    # === Phase 6: Wide n-sweep ===
    print(f"\n=== Phase 6: n-Sweep (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_sweep = min(rem * 0.20, 200.0)
        print(f"  Budget: {t_sweep:.0f}s")
        # Try a range of n values
        n_candidates = list(range(1100, 1401, 4))
        # Also include values near the current best
        n_best_cur = len(best_seq)
        for dn in [-2, -1, 1, 2, 3]:
            nc = n_best_cur + dn
            if 1000 <= nc <= 1500 and nc not in n_candidates:
                n_candidates.append(nc)
        n_candidates = sorted(set(n_candidates))

        t_per_n = t_sweep / len(n_candidates)
        for nc in n_candidates:
            if time.monotonic() > deadline:
                break
            if t_per_n < 2:
                print(f"  Not enough time per n, skipping n={nc}")
                break
            init_nc = interpolate_sequence(best_seq, nc)
            a_nc, sc_nc = mirror_descent_optimize(init_nc, min(t_per_n, deadline - time.monotonic()),
                                                  max_iters=2000000, eta_init=0.3, halve_factor=0.75)
            if sc_nc < best_score:
                update_best(a_nc, f"n_sweep_{nc}")
            else:
                # Quick QCQP polish on this n
                if len(n_candidates) <= 30:
                    a_qc = solve_iterative_active_qcqp(a_nc, solver, max_rounds=5,
                                                       initial_tol=1e-3, final_tol=1e-6)
                    update_best(a_qc, f"n_sweep_{nc}_qc")

    # === Phase 7: Final L-BFGS ===
    print(f"\n=== Phase 7: Final L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbgfs = min(rem * 0.10, 100.0)
    if t_lbgfs > 15:
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_lb = lbgfs_refine(best_seq, t_lbgfs, T_stages, maxiter_per_stage=5000)
        update_best(a_lb, "lbgfs_final")

    # === Phase 8: Final QCQP with tightest tolerance ===
    print(f"\n=== Phase 8: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        a_qf = solve_iterative_active_qcqp(best_seq, solver, max_rounds=20,
                                           initial_tol=1e-5, final_tol=1e-9)
        update_best(a_qf, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
