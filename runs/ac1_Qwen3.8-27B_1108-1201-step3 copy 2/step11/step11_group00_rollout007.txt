def mirror_descent_optimize(a0, time_limit, n_iters=10000, eta_scale=1.0):
    """
    Mirror descent with negative entropy (multiplicative weights) for:

    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg_a = np.zeros(n)
    n_avg = 0
    
    # Accumulated gradient (for "smoothed" subgradient)
    grad_accum = np.zeros(n)
    
    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        c = fast_autocorr(a)
        max_c = c.max()
        
        # Compute subgradient: use the top few peaks weighted
        k_max = np.argmax(c)
        g = 2.0 * np.zeros(n)
        # dc_k/da_i = 2*a[k-i] if 0 <= k-i < n
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g[valid] = 2.0 * a[idx[valid]]
        
        # Also consider the second peak for stability
        c_copy = c.copy()
        c_copy[k_max] = -np.inf
        k_max2 = np.argmax(c_copy)
        if c_copy[k_max2] > 0.8 * max_c:
            idx2 = k_max2 - np.arange(n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            g[valid2] += 2.0 * a[idx2[valid2]]
            g /= 2.0
        
        # Step size (polyak-like)
        eta = eta_scale / np.sqrt(t + 1.0)
        eta = min(eta, 0.1)  # Cap step size
        
        # Mirror descent update: a_i *= exp(-eta * g_i)
        log_update = -eta * g
        log_update -= log_update.max()  # For numerical stability
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-300:
            break
        a = a_new / s_new
        
        # Track best
        score = 2.0 * n * c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Running average (weighted by 1/t)
        avg_a += a
        n_avg += 1
        if n_avg >= 10 and t % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg

        if t % 500 == 0:

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
