"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive multi-stage LogSumExp L-BFGS with ultra-fine cooling
2. Multiplicative weights for fast initial progress
3. Multi-length exploration (n=800, 1000, 1200, 1500)
4. Frequency-domain perturbation for escaping local minima
5. Adaptive step-size mirror descent
Key: much finer LSE cooling schedule and longer low-T optimization phases.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_weighted(z, n, T, peak_weights):
    """Compute LSE objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    exp_sum = exp_terms.sum()
    lse = max_b + T * (scaled.max() + np.log(exp_sum + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_sum
    # If peak_weights provided, use them
    if peak_weights is not None:
        w = w * peak_weights
        sw = w.sum()
        if sw > 1e-300:
            w = w / sw
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class Timer:
    def __init__(self, total_budget):
        self.start = time.monotonic()
        self.total = total_budget
    
    def remaining(self):
        return self.total - (time.monotonic() - self.start)
    
    def elapsed(self):
        return time.monotonic() - self.start
    
    def done(self):
        return self.remaining() < 1.0


def multiplicative_weights(a0, time_budget, n_steps=200000, eta0=0.05):
    """Adaptive multiplicative weights."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Use top peaks with exponential weighting
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Weight by how close to max
        b_active = b[active]
        w = (b_active - b_active.min() + 1e-10)
        w = w / w.sum()
        g = np.zeros(n)
        for wi, k in zip(w, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += wi * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.2)
        else:
            eta *= 0.85
            if eta < 1e-8:
                break
    return best_a, best_score


def lse_lbfgs_refine(a0, time_budget, T_stages=None):
    """L-BFGS with LSE smoothing and temperature annealing.
    Uses many stages with fine cooling."""
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.5, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
    
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    # Allocate time proportionally to stages, with more time at low T
    total_stages = len(T_stages)
    # Give more time to later stages (lower T)
    weights = np.arange(1, total_stages + 1, dtype=float)
    weights = weights / weights.sum()
    time_per_stage = [time_budget * w for w in weights]
    
    for i, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 3:
            break
        T = stage_T
        deadline = t0 + time_budget
        
        def func(x, _T=T):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_obj_weighted(x, n, _T, None)
            return obj
        
        def jac(x, _T=T):
            if time.monotonic() > deadline:
                return np.zeros(len(x))
            _, g = lse_obj_weighted(x, n, _T, None)
            return g
        
        maxiter = max(100, int(time_per_stage[i] * 100))  # generous maxiter
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': maxiter * 100,
                         'disp': False}
            )
            z = res.x
            score = evaluate_local(softmax_from_z(z))
            elapsed = time.monotonic() - t0
            if i % 3 == 0 or i == total_stages - 1:
                print(f"    LSE T={T:.4f}: score={score:.8f} (t={elapsed:.0f}s)")
        except Exception as e:
            print(f"    LSE stage T={T} failed: {e}")
    
    return softmax_from_z(z)

def perturb_frequency(a, rng, strength=0.02):
    """Perturb in frequency domain for more structured changes."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a_norm = a / s
    # FFT
    fft_size = 1 << (2*n - 1).bit_length()
    A = rfft(a_norm, fft_size)
    # Add small random phase perturbation
    comp = len(A)
    max_k = max(1, int(0.3 * comp))
    dA = np.zeros_like(A)
    for k in range(1, max_k):
        dA[k] = strength * (rng.standard_normal() + 1j * rng.standard_normal())
        dA[fft_size - k] = np.conj(dA[k]) if k < fft_size - k else dA[k]
    # Actually for rfft we only have k=0..fft_size//2
    # Let's just perturb the real part
    A_new = A + dA[:comp]
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a_norm
    return a_new / s_new


def perturb_additive(a, rng, strength=0.02):
    """Add structured frequency perturbation."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    num_freqs = rng.integers(1, 6)
    pert = np.zeros(n)
    for _ in range(num_freqs):
        freq = rng.integers(1, 15)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        pert += amp * np.cos(freq * x + phase)
    pert = strength * pert / max(1.0, np.abs(pert).max())
    a_new = np.maximum(a + pert, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def mirror_descent_subgrad(a0, time_budget, n_steps=100000):
    """Mirror descent with analytical subgradient of max autocorrelation.
    Uses the peak position for the subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    # Initial step size
    step = 1.0 / n
    patience = 0
    
    for step_i in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Find peak(s)
        peak = np.argmax(b)
        # Subgradient: grad of (a*a)[peak] w.r.t. a
        g = np.zeros(n)
        js = max(0, peak - n + 1)
        je = min(n - 1, peak)
        idx = np.arange(js, je + 1)
        g[idx] = 2 * a[peak - idx]  # factor of 2 for symmetric
        # Actually for linear conv, (a*a)[k] = sum_j a_j * a_{k-j}
        # d/d(a_i) = sum_j [delta(j=i)*a_{k-i} + a_j*delta(k-j=i)]
        # = a_{k-i} + a_{k-i} = 2*a_{k-i} if i and k-i both in [0,n-1]
        # But only if i is in [js, je]
        
        # Mirror descent update: a_new_i ∝ a_i * exp(-step * g_i)
        log_mult = -step * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            step = min(step * 1.01, 5.0 / n)
        else:
            patience += 1
            step *= 0.9
            if patience > 200 or step < 1e-8 / n:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def direct_subgrad_max(a0, time_budget, n_steps=50000):
    """Direct subgradient method on the max autocorrelation.
    Uses top-k peaks for a smoothed subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    t = 1.0  # temperature for peak selection
    step = 0.1
    
    for i in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Soft max over peaks
        scaled = (b - max_b) / t
        weights = np.exp(scaled - scaled.max())
        weights = weights / weights.sum()
        # Weighted subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if weights[k] < 1e-12:
                continue

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Ultra-fine multi-stage LogSumExp L-BFGS (T from 20 down to 0.0005)
2. Adaptive multiplicative weights for fast initial progress
3. Frequency-domain and additive perturbations for local minima escape

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T):
    """Compute LSE objective and gradient w.r.t. z for softmax parameterization."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    exp_sum = exp_terms.sum()
    lse = max_b + T * (scaled.max() + np.log(exp_sum + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_sum
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class LSEProblem:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def obj(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_obj_grad(z, self.n, self.T)
        return obj

    def grad(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_obj_grad(z, self.n, self.T)
        return g


class LSEProblemWeighted:
    def __init__(self, n, T, deadline, peak_weights):
        self.n = n
        self.T = T
        self.deadline = deadline
        self.peak_weights = peak_weights

    def obj(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        return self._compute(z)[0]

    def grad(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        return self._compute(z)[1]

    def _compute(self, z):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        scaled = (b - max_b) / self.T
        scaled_shifted = scaled - scaled.max()
        exp_terms = np.exp(scaled_shifted)
        w = exp_terms / exp_terms.sum()
        pw = self.peak_weights
        w = w * pw
        sw = w.sum()
        if sw > 1e-300:
            w = w / sw
        exp_sum = (w * exp_terms).sum()  # recompute for obj
        # Actually let's just use the weighted LSE
        weighted_scaled = scaled_shifted + np.log(pw + 1e-300)
        max_ws = weighted_scaled.max()
        exp_ws = np.exp(weighted_scaled - max_ws)
        lse = max_b + self.T * (max_ws + np.log(exp_ws.sum() + 1e-300))
        obj = 2.0 * self.n * lse
        grad_a = np.zeros(self.n)
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - self.n + 1)
            je = min(self.n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return obj, 2.0 * self.n * grad_z


def multiplicative_weights(a0, time_budget, n_steps=200000, eta0=0.05):
    """Adaptive multiplicative weights with peak-weighted subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    patience = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        b_active = b[active]
        w = (b_active - b_active.min() + 1e-10)
        w = w / w.sum()
        g = np.zeros(n)
        for wi, k in zip(w, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += wi * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            eta = min(eta * 1.003, 0.15)
        else:
            patience += 1
            eta *= 0.88
            if patience > 300 or eta < 1e-8:
                break
    return best_a, best_score


def lse_lbfgs_refine(a0, time_budget, T_schedule=None):
    """L-BFGS with LSE smoothing and temperature annealing.
    Uses geometric cooling for ultra-fine optimization."""
    if T_schedule is None:
        # Generate geometric schedule from 20 down to 0.0005
        T_schedule = []
        T = 20.0
        while T > 0.0004:
            T_schedule.append(T)
            T *= 0.82

    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    total_stages = len(T_schedule)

    for i, stage_T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)

def perturb_frequency(a, rng, strength=0.03):
    """Perturb in frequency domain."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a_norm = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a_norm, fft_size)
    comp = len(A)
    max_k = max(1, int(0.4 * comp))
    dA = np.zeros(comp, dtype=np.complex128)
    for k in range(1, max_k):
        dA[k] = strength * (rng.standard_normal() + 1j * rng.standard_normal())
    A_new = A + dA
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a_norm
    return a_new / s_new


def perturb_additive(a, rng, strength=0.02):
    """Add structured frequency perturbation."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    num_freqs = rng.integers(1, 6)
    pert = np.zeros(n)
    for _ in range(num_freqs):
        freq = rng.integers(1, 20)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        pert += amp * np.cos(freq * x + phase)
    pmax = np.abs(pert).max()
    if pmax < 1e-15:
        return a.copy()
    pert = strength * pert / pmax
    a_new = np.maximum(a + pert, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def perturb_peak_targeted(a, rng, strength=0.02):
    """Perturb specifically to reduce autocorrelation peaks."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    # Find top peaks
    num_peaks = min(5, len(b))
    peak_indices = np.argsort(b)[-num_peaks:][::-1]
    # Create perturbation that reduces contribution to these peaks
    pert = np.zeros(n)
    for k in peak_indices:
        if b[k] < max_b * 0.9:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        pert[idx] -= a[k - idx]
    pmax = np.abs(pert).max()
    if pmax < 1e-15:
        return a.copy()
    pert = strength * pert / pmax
    a_new = np.maximum(a + pert, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def direct_subgrad(a0, time_budget, n_steps=50000):
    """Direct subgradient with adaptive temperature on peak weights."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    T = 1.0
    step = 0.5
    patience = 0

    for i in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        b = fast_convolve(a)
        max_b = b.max()
        scaled = (b - max_b) / T
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights = weights / weights.sum()
        g = np.zeros(n)
        for k in range(len(b)):
            if weights[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * weights[k] * a[k - idx]
        a_new = a - step * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            T = max(0.005, T * 0.998)
            step = min(step * 1.002, 2.0)
        else:
            patience += 1
            step *= 0.97
            T = min(T * 1.005, 5.0)
            if patience > 500 or step < 1e-8:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if t_lse > 20:
            a_ln_lse = lse_lbfgs_refine(a_ln_mw if a_ln_mw is not None else init_ln, t_lse)
            try_update(a_ln_lse, f"lse_n{ln}")

    # === Phase 5: Very fine LSE on best ===
    print(f"\n=== Phase 5: Very fine LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t5 = min(rem * 0.28, 300.0)
        print(f"  Budget: {t5:.0f}s")
        a_vf = lse_lbfgs_refine_very_fine(best_seq, t5)
        try_update(a_vf, "very_fine_lse")

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: Subgrad polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t6 = min(rem * 0.15, 50.0)
        a_sg, _ = direct_subgrad(best_seq, t6, n_steps=30000)
        try_update(a_sg, "subgrad")

    # === Phase 7: Final LSE ===
    print(f"\n=== Phase 7: Final LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t7 = min(rem * 0.5, 60.0)
        a_fin = lse_lbfgs_refine_very_fine(best_seq, t7)
        try_update(a_fin, "final_lse")

    # === Phase 8: One more perturbation attempt ===
    print(f"\n=== Phase 8: Final escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_pert = perturb_peak_targeted(best_seq, rng, strength=0.01)
        t8 = min(rem * 0.6, 55.0)
        if t8 > 15:
            a_last = lse_lbfgs_refine_very_fine(a_pert, t8)
            try_update(a_last, "final_escape")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
