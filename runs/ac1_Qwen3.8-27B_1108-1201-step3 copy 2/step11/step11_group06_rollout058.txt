```python
import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_grad_a(a, T):
    """Compute gradient of 2n*LSE(conv(a,a)/T) w.r.t. a, using FFT."""
    n = len(a)
    b = fast_convolve(a)
    # Softmax weights
    shifted = (b - b.max()) / T
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    # grad_a[j] = 2n * sum_k w[k] * a[k-j] (for valid k-j)

    # Using FFT: ifft(conj(FFT(a)) * FFT(w))[j]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(A) * W, fft_size)
    return 2.0 * n * corr[:n]
```

```python
def lse_objective_and_grad_z(z, n, T):
    """Compute LSE objective and gradient w.r.t. z (softmax parameterization)."""
    # Softmax
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Objective
    b = fast_convolve(a)
    shifted_b = (b - b.max()) / T
    exp_shifted = np.exp(shifted_b)
    lse = b.max() + T * np.log(exp_shifted.sum())
    obj = 2.0 * n * lse
    
    # Gradient w.r.t. a (using FFT)
    w = exp_shifted / exp_shifted.sum()
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(A) * W, fft_size)[:n]
    grad_a = 2.0 * n * corr
    
    # Gradient w.r.t. z
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- O(n log n) FFT-based gradient computation for LSE-smoothed objective
- Multi-stage geometric temperature cooling with L-BFGS-B
- Fast multiplicative weights for coarse initialization
- Multi-length exploration and perturbation-based escape
- Softmax parameterization ensuring non-negativity and sum=1
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad_z(z, n, T):
    """
    Compute LSE-smoothed objective and gradient w.r.t. z (softmax param).
    Uses O(n log n) FFT-based gradient.
    """
    # Softmax to get a
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Convolution
    b = fast_convolve(a)
    
    # LSE objective
    b_max = b.max()
    b_shifted = (b - b_max) / T
    exp_b = np.exp(b_shifted)
    exp_sum = exp_b.sum()
    lse = b_max + T * np.log(exp_sum)
    obj = 2.0 * n * lse
    
    # Weights for gradient
    w = exp_b / exp_sum
    
    # Gradient w.r.t. a: grad_a[j] = 2n * sum_k w[k] * a[k-j]
    # = 2n * cross-correlation of a and w, evaluated at lags 0..n-1
    # Using FFT: ifft(conj(FFT(a_padded)) * FFT(w_padded))[:n]
    fft_size = 1 << (3 * n - 3).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    grad_a = 2.0 * n * irfft(np.conj(A) * W, fft_size)[:n]
    
    # Gradient w.r.t. z: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lse_obj(z, n, T):
    """Objective only."""
    obj, _ = lse_obj_grad_z(z, n, T)
    return obj


def lse_jac(z, n, T):
    """Gradient only."""
    _, g = lse_obj_grad_z(z, n, T)
    return g

    def __init__(self, n, T):

        self.T = T
    def __call__(self, z):
        return lse_jac(z, self.n, self.T)

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)

    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def multiplicative_weights_fast(a0, time_limit, n_steps, eta0=0.05):
    """Fast MW with O(n log n) gradient per step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = eta0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Subgradient: at the argmax lag k*, grad[j] = a[k*-j] (valid indices)
        # Use top few peaks for stability
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > 20:
            # Just use top 20
            top_idx = np.argsort(b[active])[-20:]
            active = active[top_idx]
        
        w_active = b[active] / b[active].sum()
        grad_a = np.zeros(n)
        for w_k, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w_k * a[k - idx]
        
        # MW update
        log_mult = -eta * grad_a
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.2)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- O(n log n) FFT-based gradient for LSE-smoothed objective (key speedup)
- Multi-stage geometric temperature cooling with L-BFGS-B in softmax space
- Fast multiplicative weights for coarse initialization
- Multi-length exploration and perturbation-based local escape
- Softmax parameterization ensuring a>=0, sum(a)=1 automatically
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad_z(z, n, T):
    """LSE-smoothed objective and gradient w.r.t. z (softmax param). O(n log n)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    b = fast_convolve(a)
    b_max = b.max()
    b_shifted = (b - b_max) / T
    exp_b = np.exp(b_shifted)
    exp_sum = exp_b.sum()
    lse = b_max + T * np.log(exp_sum)
    obj = 2.0 * n * lse

    w = exp_b / exp_sum

    fft_size = 1 << (3 * n - 3).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    grad_a = 2.0 * n * irfft(np.conj(A) * W, fft_size)[:n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lse_obj(z, n, T):
    obj, _ = lse_obj_grad_z(z, n, T)
    return obj


def lse_jac(z, n, T):
    _, g = lse_obj_grad_z(z, n, T)
    return g


class LSEFunc:
    def __init__(self, n, T):
        self.n = n
        self.T = T

    def __call__(self, z):
        return lse_obj(z, self.n, self.T)


class LSEJac:
    def __init__(self, n, T):
        self.n = n
        self.T = T

    def __call__(self, z):
        return lse_jac(z, self.n, self.T)


def softmax_vec(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def log_of(a):

    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def lse_cooling_optimize(a0, T_start, T_end, decay, maxiter_per_stage, time_limit, verbose=False):
    """Multi-stage LSE cooling with L-BFGS-B in softmax space."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

    else:
        a = a / s

    z = log_of(a)
    t0 = time.monotonic()
    T = T_start
    stage = 0
    best_a = a.copy()
    best_score = compute_score(a)

    while T > T_end:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        func = LSEFunc(n, T)
        jac = LSEJac(n, T)
        iters = min(maxiter_per_stage, max(200, int(remaining * 800)))

        try:
            res = minimize(
                func, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_curr = softmax_vec(z)
        score_curr = compute_score(a_curr)
        if score_curr < best_score:
            best_score = score_curr
            best_a = a_curr.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Stage {stage}: T={T:.5f}, score={score_curr:.8f}, best={best_score:.8f}, t={elapsed:.1f}s")

        stage += 1

        func = LSEFunc(n, T_end)
        jac = LSEJac(n, T_end)
        try:
            res = minimize(
                func, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-14, 'disp': False}

            z = res.x
        except Exception:
            pass
        a_curr = softmax_vec(z)
        score_curr = compute_score(a_curr)
        if score_curr < best_score:

            best_a = a_curr.copy()
        if verbose:

    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-5)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        w_active = b[active] / b[active].sum()
        grad_a = np.zeros(n)
        for w_k, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w_k * a[k - idx]

        log_mult = -eta * grad_a
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.003, 0.25)
        else:
            eta = eta * 0.82
            if eta < 1e-12:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses O(n log n) FFT-based LSE gradient with multi-stage cooling.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structural_init(n_init, rng)
        best_score = compute_score(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")
    print(f"Budget: {budget_s:.0f}s")

    # === Phase 1: Quick MW on parent ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 5:
        print(f"\n=== Phase 1: MW (t={t1:.0f}s) ===")
        a_mw, _ = multiplicative_weights_fast(best_seq, t1, n_steps=1000000, eta0=0.05)
        best_seq, best_score = update_best(best_seq, best_score, a_mw, "mw")

    # === Phase 2: Main LSE cooling (broad) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LSE broad cooling (t={t2:.0f}s) ===")
        a_lse = lse_cooling_optimize(
            best_seq, T_start=3.0, T_end=0.005, decay=0.55,
            maxiter_per_stage=4000, time_limit=t2, verbose=True
        )
        best_seq, best_score = update_best(best_seq, best_score, a_lse, "lse_broad")

    # === Phase 3: Fine LSE cooling ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        print(f"\n=== Phase 3: LSE fine cooling (t={t3:.0f}s) ===")
        a_fine = lse_cooling_optimize(
            best_seq, T_start=0.3, T_end=0.0005, decay=0.65,
            maxiter_per_stage=5000, time_limit=t3, verbose=True
        )
        best_seq, best_score = update_best(best_seq, best_score, a_fine, "lse_fine")

    # === Phase 4: Ultra-fine LSE ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 180.0)
    if t4 > 20:
        print(f"\n=== Phase 4: LSE ultra-fine (t={t4:.0f}s) ===")
        a_ultra = lse_cooling_optimize(
            best_seq, T_start=0.05, T_end=0.0001, decay=0.7,
            maxiter_per_stage=6000, time_limit=t4, verbose=True
        )
        best_seq, best_score = update_best(best_seq, best_score, a_ultra, "lse_ultra")

    # === Phase 5: Multi-length exploration ===
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1500, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.12, 150.0)
        if t_ln < 15:
            continue
        print(f"\n=== Phase 5: n={ln} (t={t_ln:.0f}s) ===")
        a_interp = interpolate_to_length(best_seq, ln)
        a_opt = lse_cooling_optimize(
            a_interp, T_start=2.0, T_end=0.002, decay=0.6,
            maxiter_per_stage=3000, time_limit=t_ln, verbose=False
        )
        best_seq, best_score = update_best(best_seq, best_score, a_opt, f"n{ln}")

    # === Phase 6: Perturbation + re-optimize ===
    rem = deadline - time.monotonic()
    n_pert = max(3, int(rem / 50))
    for trial in range(n_pert):
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        strength = 0.003 + 0.004 * trial
        if trial % 2 == 0:
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
        else:
            a_pert = perturb_random(best_seq, rng, strength=strength * 0.5)
        t_sub = min(rem * 0.25, 80.0)
        a_opt = lse_cooling_optimize(
            a_pert, T_start=0.5, T_end=0.001, decay=0.65,
            maxiter_per_stage=3000, time_limit=t_sub, verbose=False
        )
        best_seq, best_score = update_best(best_seq, best_score, a_opt, f"pert{trial}")

    # === Phase 7: Final fine polish ===
    rem = deadline - time.monotonic()
    if rem > 25:
        t7 = min(rem * 0.4, 100.0)
        print(f"\n=== Phase 7: Final polish (t={t7:.0f}s) ===")
        a_fin = lse_cooling_optimize(
            best_seq, T_start=0.1, T_end=0.0001, decay=0.7,
            maxiter_per_stage=5000, time_limit=t7, verbose=True
        )
        best_seq, best_score = update_best(best_seq, best_score, a_fin, "final")

    # === Phase 8: Try fresh random inits at best length ===
    rem = deadline - time.monotonic()
    n_inits = max(2, int(rem / 60))
    for i in range(n_inits):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        a_init = structural_init(n_best, rng)
        t_sub = min(rem * 0.3, 50.0)
        a_opt = lse_cooling_optimize(
            a_init, T_start=5.0, T_end=0.001, decay=0.55,
            maxiter_per_stage=2000, time_limit=t_sub, verbose=False
        )
        best_seq, best_score = update_best(best_seq, best_score, a_opt, f"init{i}")

    # Final verification
    if best_seq is not None:
        final_score = compute_score(best_seq)
        print(f"\nFINAL: score={final_score:.8f}, n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
