def fast_convolve(a):

    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)

            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:

            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")

def lse_grad_direct(a, T_val, n):
    """Compute LogSumExp objective and gradient directly."""
    b = fast_convolve(a)
    Tb = b / T_val
    max_Tb = Tb.max()
    if max_Tb > 500:
        max_Tb = 500
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n * T_val) * log_sum_exp

    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad, a)
    grad_z = a * (grad - S)
    return obj, grad_z, w

        if remaining < 2:
            break

        steps_this_stage = max(100, int(remaining / (n_stages - si) * 2))
        steps_this_stage = min(steps_this_stage, 5000)

        for step in range(steps_this_stage):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_from_z(z)
            obj, grad_z, w = lse_grad_direct(a, stage_T, n)

            step_count += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** step_count)
            v_hat = v / (1 - beta2 ** step_count)

            lr = lr_start * (lr_end / lr_start) ** (step / max(1, steps_this_stage))
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if (si + 1) % 5 == 0:
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            print(f"    Adam-LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={sc:.8f}")

    return best_a, best_score


def adam_logsumexp_v2(a0, time_limit, T_start=30.0, T_end=0.02, n_stages=25,
                      lr=0.05, beta1=0.9, beta2=0.999):
    """Improved Adam with adaptive LR and gradient clipping."""
    n = len(a0)
    z = z_from_a(a0).copy()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0
    best_stage_score = best_score

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time.monotonic_time_limit - time.monotonic() if hasattr(time, 'monotonic_time_limit') else time_limit - (time.monotonic() - t0)
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        steps = max(50, int(remaining / max(1, n_stages - si)))
        steps = min(steps, 3000)

        stage_best = best_score
        for step in range(steps):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = b / stage_T
            max_Tb = min(Tb.max(), 400)
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            grad = np.zeros(n)
            n_active = min(len(b), max(50, int(0.1 * len(b))))
            top_idx = np.argsort(w)[::-1][:n_active]
            for k in top_idx:
                if w[k] < 1e-15:
                    break
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad[idx] += 2.0 * w[k] * a[k - idx]

            S = np.dot(grad, a)
            grad_z = a * (grad - S)

            grad_norm = np.linalg.norm(grad_z)
            if grad_norm > 1.0:
                grad_z = grad_z / grad_norm

            step_count += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** step_count)
            v_hat = v / (1 - beta2 ** step_count)

            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            if sc < stage_best:
                stage_best = sc
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if (si + 1) % 5 == 0:
            print(f"    AdamV2 stage={si+1}/{n_stages} T={stage_T:.3f}: score={best_score:.8f}")

    return best_a, best_score


def subgradient_with_momentum(a0, time_limit, n_steps=100000, eta=0.1, momentum=0.9):
    """Subgradient descent with momentum on the max-conv objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    velocity = np.zeros(n)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()

        active_frac = max(0.05, 0.5 * (1 - step / max(1, n_steps)))
        threshold = max_b * (1 - active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        eta_curr = eta * (0.5 ** (step / max(1, n_steps // 3)))
        velocity = momentum * velocity - eta_curr * grad_z
        z = z + velocity
        a_new = softmax_from_z(z)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        if (step + 1) % 10000 == 0:
            print(f"    SubGrad step={step+1}: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 5 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = x * (1 - x) * (4 - 3 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.3 * np.exp(-5 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, strength=0.05):
    """Add random perturbation and renormalize."""
    n = len(a)
    noise = rng.normal(0, strength, n)
    a_pert = a * (1 + noise)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP main optimization (45% of budget) ===
    remaining = global_deadline - time.monotonic()
    t1 = 0.45 * (budget_s - 30)
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score:.8f}, budget={t1:.0f}s) ===")
    a_out, sc1 = nesterov_lp(best_seq, t1, solver=solver)
    try_update(a_out, "nesterov_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Adam LogSumExp with annealing (25% of budget) ===
    remaining = global_deadline - time.monotonic()
    t2 = min(0.25 * (budget_s - 30), remaining * 0.6)
    print(f"\n=== Phase 2: Adam LSE (score={best_score:.8f}, budget={t2:.0f}s) ===")
    if t2 > 20:
        a_lse, sc2 = adam_logsumexp(best_seq, t2, T_start=50.0, T_end=0.05,
                                     n_stages=20, lr_start=0.1, lr_end=0.001)
        try_update(a_lse, "adam_lse")
        # Second Adam run with different params
        remaining = global_deadline - time.monotonic()
        if remaining > 30:
            a_lse2, sc2b = adam_logsumexp_v2(best_seq, remaining * 0.5,
                                              T_start=20.0, T_end=0.02,
                                              n_stages=20, lr=0.03)
            try_update(a_lse2, "adam_lse_v2")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Nesterov LP round 2 (15% of budget) ===
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 30), remaining * 0.6)
    print(f"\n=== Phase 3: Nesterov LP round 2 (score={best_score:.8f}, budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out2, sc3 = nesterov_lp(best_seq, t3, solver=solver, use_active_set=True)
        try_update(a_out2, "nesterov_r2")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Perturbation restarts (10% of budget) ===
    remaining = global_deadline - time.monotonic()
    t4_total = min(0.10 * (budget_s - 30), remaining * 0.6)
    print(f"\n=== Phase 4: Perturbation restarts (score={best_score:.8f}) ===")
    if t4_total > 30:
        for idx in range(3):
            remaining = global_deadline - time.monotonic()
            if remaining < 20:
                break
            t4 = min(t4_total / 3.0, remaining * 0.5)
            if t4 < 8:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.02 + 0.02 * idx)
            print(f"  Perturb {idx+1}: strength={0.02+0.02*idx:.3f}, budget={t4:.0f}s")
            a_res, _ = nesterov_lp(a_pert, t4, solver=solver)
            try_update(a_res, f"perturb_{idx+1}")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Subgradient polish (5% of budget) ===
    remaining = global_deadline - time.monotonic()
    t5 = min(0.05 * (budget_s - 30), remaining * 0.6)
    print(f"\n=== Phase 5: Subgradient polish (score={best_score:.8f}, budget={t5:.0f}s) ===")
    if t5 > 10:
        a_sg, sc5 = subgradient_with_momentum(best_seq, t5, eta=0.05, momentum=0.95)
        try_update(a_sg, "subgrad")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Length exploration (remaining time) ===
    remaining = global_deadline - time.monotonic()
    print(f"\n=== Phase 6: Length exploration (score={best_score:.8f}, remaining={remaining:.0f}s) ===")
    if remaining > 30:
        n_tries = [n_base + 100, n_base - 100, n_base + 200, n_base + 50]
        per_try = max(10, remaining * 0.6 / len(n_tries))
        for n_try in n_tries:
            remaining = global_deadline - time.monotonic()
            if remaining < 15:
                break
            t_try = min(per_try, remaining * 0.4)
            if t_try < 5:
                break
            print(f"  Trying n={n_try}, budget={t_try:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t_try, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Final Adam polish on best ===
    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 7: Final Adam polish (remaining={remaining:.0f}s) ===")
        t7 = min(remaining * 0.7, 60.0)
        if t7 > 5:
            a_final, _ = adam_logsumexp(best_seq, t7, T_start=10.0, T_end=0.01,
                                         n_stages=10, lr_start=0.05, lr_end=0.0005)
            try_update(a_final, "final_adam")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
