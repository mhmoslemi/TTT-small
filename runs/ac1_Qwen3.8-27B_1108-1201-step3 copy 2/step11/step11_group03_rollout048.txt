"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive cutting-plane QCQP (main workhorse)
2. LP alternating with Nesterov (for diversification)
3. Low-temperature LogSumExp L-BFGS for final precision
4. Multi-n exploration

Key innovation: The cutting-plane QCQP directly solves the convex problem
min t s.t. a^T Q_k a <= t, sum(a)=1, a>=0 iteratively, adding constraints
only when they're violated. This converges in a small number of iterations
and provides a globally optimal solution (up to solver tolerance).
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q_k a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_available_solvers():
    """Find available solvers, sorted by preference."""
    solvers = []
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.SCS]:
        if s is not None:
            solvers.append(s)
    # Remove duplicates
    seen = set()
    unique = []
    for s in solvers:
        if s not in seen:
            seen.add(s)
            unique.append(s)
    return unique


def solve_qcqp_cutting_plane(a0, time_limit, solvers, max_active=100, rel_tol=1e-6):
    """
    Solve min max_k c_k(a) s.t. sum(a)=1, a>=0 using cutting planes.
    
    Iteratively:
    1. Identify active constraints (near-max c_k values)
    2. Solve QCQP with active constraints
    3. Check for violations
    4. Add violated constraints if any
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Start with the most active constraint
    c = fast_convolve(a)
    max_c = c.max()
    active_set = [int(np.argmax(c))]
    
    iteration = 0
    for iteration in range(50):  # Max 50 cutting-plane iterations
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build QCQP with current active set
        constraints = [cp.sum(a_var) == 1]
        Q_matrices = []
        for k in active_set:
            Q = build_q_sparse(k, n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
            Q_matrices.append(Q)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        solved = False
        for solver in solvers:
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Compute new convolution
        c_new = fast_convolve(a_new)
        max_new = c_new.max()
        score_new = 2.0 * n * max_new
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        # Check for violated constraints
        threshold = max_new * (1.0 - rel_tol)
        violated = np.where((c_new >= threshold) & (np.array(active_set) != np.arange(2*n-1)))[0]
        
        # More precisely: find constraints not in active_set that are violated
        non_active_mask = np.ones(2 * n - 1, dtype=bool)
        for k in active_set:
            non_active_mask[k] = False
        violated_new = np.where((c_new >= max_new * (1.0 - 1e-4)) & non_active_mask)[0]
        
        if len(violated_new) == 0:
            # No new violations - we've converged
            # But let's also add near-active constraints for robustness
            near_active = np.where(c_new >= max_new * (1.0 - rel_tol))[0]
            new_actives = [int(k) for k in near_active if int(k) not in active_set]
            if len(new_actives) == 0:
                break
            active_set.extend(new_actives)
        else:
            # Add violated constraints
            for k in violated_new:
                if int(k) not in active_set:
                    active_set.append(int(k))
        
        # Cap active set size
        if len(active_set) > max_active:
            # Keep only the most active ones
            c_vals = c_new[np.array(active_set)]
            order = np.argsort(c_vals)[::-1]
            active_set = [int(active_set[i]) for i in order[:max_active]]
        
        a = a_new
        print(f"    CP iter {iteration+1}: score={score_new:.10f}, active={len(active_set)}, t_val={t_var.value:.10f}")
    
    return best_a, best_score


def solve_lp_best_response(a_fixed, solvers):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build the constraint matrix: conv(a, b)[k] = sum_j a[j] * b[k-j]
    # This is linear in b, so it's an LP
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                return b_val / s, float(max(0, M_var.value))
        except Exception:
            continue
    
    return None, None


def nesterov_lp(a0, time_limit, solvers, max_iters=50000):
    """Nesterov-accelerated LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, _ = solve_lp_best_response(y, solvers)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if (it + 1) % 500 == 0:
            print(f"    NesterovLP iter={it+1}: score={best_score:.10f}")
    
    return best_a, best_score


def lse_refine(a0, time_limit, T_stages=None):
    """LogSumExp L-BFGS refinement."""
    n = len(a0)
    if T_stages is None:
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    # Work in log domain for non-negativity
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for T in T_stages:
        if time.monotonic() > deadline:
            break
        
        def obj_grad(z_val):
            z_shifted = z_val - z_val.max()
            exp_z = np.exp(z_shifted)
            a_val = exp_z / exp_z.sum()
            a_val = np.maximum(a_val, 1e-300)
            
            b = fast_convolve(a_val)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / T)
            lse = max_b + T * np.log(exp_terms.sum())
            obj = 2.0 * n * lse
            
            w = exp_terms / exp_terms.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_val[k - idx]
            
            S_val = np.dot(grad_a, a_val)
            grad_z = a_val * (grad_a - S_val)
            return obj, 2.0 * n * grad_z
        
        def fun(z_val):
            obj, _ = obj_grad(z_val)
            return obj
        
        def jac(z_val):
            _, g = obj_grad(z_val)
            return g
        
        try:
            res = minimize(fun, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-15, 'disp': False})
            z = res.x
            a = np.exp(z - z.max())
            a = a / a.sum()
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"    LSE T={T}: score={score:.10f}")
        except Exception:
            pass
    
    return best_a, best_score


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
