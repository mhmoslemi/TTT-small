import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_fastest_lp_solver():

    constraints = [T_test @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    results = {}

        active = {int(np.argmax(b))}

    a_cur = a_init.copy()
    t_cur = max_b

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a_new, t_new = solve_qp_active(a_cur, sorted(active), solver)
        if a_new is None:
            break

        a_cur = a_new
        t_cur = t_new

        # Check if all constraints are satisfied
        b_new = fast_convolve(a_cur)

        # Find violated constraints
        violations = b_new - t_cur
        max_viol = violations.max()

        if max_viol < 1e-8:
            print(f"    Active-set QP converged in {it+1} iterations, t={t_cur:.15f}")
            break

        # Add the most violated constraint
        k_viol = int(np.argmax(violations))
        active.add(k_viol)

        # Also add any other significantly violated constraints
        many_viol = np.where(violations > 0.5 * max_viol)[0]
        for k in many_viol:
            active.add(int(k))

        # Limit active set size
        if len(active) > 50:
            b_check = fast_convolve(a_cur)
            top_k = np.argsort(b_check)[-30:]
            active = set(int(k) for k in top_k)

        if (it + 1) % 10 == 0:
            print(f"    Active-set QP iter {it+1}: |A|={len(active)}, t={t_cur:.10f}, max_viol={max_viol:.2e}")

    return a_cur, t_cur


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters, fallback_solvers=None):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, fallback_solvers)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                # Restart with perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.12f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 50 and np.mean(lp_times[-50:]) > 8.0:
            print(f"    LP too slow, stopping")
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def multiplicative_subgradient(a0, time_limit, n_steps):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.05
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def kkt_newton_polish(a0, solver, time_limit):
    """
    Newton-based polish using KKT conditions.
    At optimum: for active set A, b^T Q_k b = t for all k in A,
    and sum_{k in A} w_k * 2 Q_k b = lambda * 1.
    We solve this nonlinear system with Newton's method.
    """
    n = len(a0)
    t0 = time.monotonic()

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()

    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - 1e-5)
    active = np.where(b >= threshold)[0]

    if len(active) > 20:
        active = active[np.argsort(b[active])[-20:]]

    if len(active) == 0:
        return a, 2.0 * n * max_b

    print(f"    KKT Newton: |active|={len(active)}")

    # Build Q matrices for active set
    Qs = {}
    for k in active:
        Qs[int(k)] = build_q_sparse(int(k), n)

    # Unknowns: b (n), t (1), lambda (1)
    # Equations: b^T Q_k b - t = 0 for k in active (|A| eqs)
    #            sum_b b = 1 (1 eq)
    #            We need |A| more equations from KKT:
    #            For the |A| active constraints, the KKT condition gives:
    #            sum_k w_k * grad_k = lambda * 1
    #            This is n equations but with unknowns w (|A|) and lambda (1).
    #            So we have n + |A| + 1 equations in n + |A| + 2 unknowns.
    #            The system is underdetermined by 1 (scale invariance of w).
    #            We fix sum(w) = 1 to make it square.

    # Alternative simpler approach: just solve the QP directly with all active constraints
    # and verify optimality.

    # Let's just use the QP with all detected active constraints as a final check
    a_qp, t_qp = solve_qp_active(a, list(active), solver)
    if a_qp is not None:
        score_qp = 2.0 * n * fast_convolve(a_qp).max()
        score_cur = 2.0 * n * max_b
        if score_qp < score_cur:
            return a_qp, score_qp

    return a, 2.0 * n * max_b


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating with Nesterov acceleration, active-set QP with
    add-violated-constraint loop, KKT polish, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Benchmark LP solver
    print("=== Benchmarking LP solvers ===")
    solver = get_fastest_lp_solver()
    fallback = [cp.ECOS, cp.SCS, cp.CLARABEL]
    print(f"Using solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.12f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:

    if parent is None:
        n_init = 1200

        best_seq = np.cos(np.pi * x / 2) ** 2

        best_score = evaluate_local(best_seq)

    n_best = len(best_seq)

    def update_best(a, label):

        if a is None:
            return

        s = a.sum()
        if s < 0.01:

    rem = deadline - time.monotonic()
