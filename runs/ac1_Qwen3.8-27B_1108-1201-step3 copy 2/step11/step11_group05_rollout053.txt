def fast_conv(a):

    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)

        idx1 = j
        idx2 = k - j
        g[idx1] += a[idx2]
        if idx2 != idx1:
            g[idx2] += a[idx1]

        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()
        
        # Periodic LP
        if solver is not None and step % n_lp_every == 0:
            b_lp = solve_lp(a, solver)
            if b_lp is not None:
                # Nesterov dual averaging update
                alpha = 2.0 / (n_lp_count + 2)
                y_nesterov = (1 - alpha) * y_nesterov + alpha * b_lp
                y_nesterov = np.maximum(y_nesterov, 0)
                sy = y_nesterov.sum()
                if sy > 1e-15:
                    y_nesterov = y_nesterov / sy
                    y_score = 2.0 * n * fast_conv(y_nesterov).max()
                    if y_score < best_score:
                        best_score = y_score
                        best_a = y_nesterov.copy()
                n_lp_count += 1
    
    return best_a, best_score
