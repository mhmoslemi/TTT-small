        a_new, t_val = solve_qcqp(a, S, solver)

            violated = np.where(b_new > t_val + tol)[0]
            S = np.union1d(S, violated)
            a = a_new
    
    return a, fast_convolve(a).max()

Alternative: **Direct Newton on KKT conditions**

Once we have a good active set S (say |S| <= 50), we can solve the KKT system directly:

Unknowns: a (n), t (1), μ (1), λ (|S|) with the constraint sum(λ) = 1
Equations:
- a^T Q_k a = t for k in S (|S| equations)
- 2 * sum_{k in S} λ_k Q_k a + μ * 1 = 0 (n equations)
- sum(a) = 1 (1 equation)
- sum(λ) = 1 (1 equation)

Total: |S| + n + 2 equations, n + 1 + 1 + |S| = n + |S| + 2 unknowns. Balanced!

The system is nonlinear (due to the a^T Q_k a terms), so we use Newton's method:
- Linearize around the current iterate
- Solve the linear system
- Update

The Jacobian of the system with respect to (a, t, μ, λ) is:
- ∂(a^T Q_k a)/∂a = 2 Q_k a (for each k in S)
- ∂(a^T Q_k a)/∂t = -1
- ∂(a^T Q_k a)/∂μ = 0
- ∂(a^T Q_k a)/∂λ = 0

- ∂(2 sum λ_k Q_k a + μ 1)/∂a = 2 sum λ_k Q_k
- ∂(2 sum λ_k Q_k a + μ 1)/∂t = 0
- ∂(2 sum λ_k Q_k a + μ 1)/∂μ = 1
- ∂(2 sum λ_k Q_k a + μ 1)/∂λ_j = 2 Q_j a

- ∂(sum a)/∂a = 1
- Others = 0

- ∂(sum λ)/∂λ = 1
- Others = 0

So the Jacobian is a block matrix:
[2 Q_k a (rows for k in S)  -1_S  0        0       ]
[2 sum λ_k Q_k             0      1        2 Q_j a  ]
[1^T                        0      0        0       ]
[0                           0      0        1_S^T  ]

This is a (n + |S| + 2) x (n + |S| + 2) system. For n=1200 and |S|=20, it's about 1222 x 1222. Solving this with a direct method (LU factorization) takes about O(n^3) = 1.8 * 10^9 operations, which is too slow per Newton iteration.

However, we can use the Sherman-Morrison-Woodbury formula or exploit the structure. The main block is 2 * sum λ_k Q_k, which is a sum of sparse matrices (each Q_k is sparse with at most n non-zeros per row). So the main block is dense but has a specific structure.

Actually, for n=1200, solving a 1222 x 1222 dense system with numpy's lstsq or solve takes about 0.1-0.5 seconds. If we need 10-20 Newton iterations, that's 1-10 seconds total. This is feasible!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating with Nesterov acceleration for initial convergence
2. Cutting-plane QCQP (cvxpy) for exact solution on active constraints
3. Multi-n exploration (n=1198..1202)
Key improvement over previous: direct QCQP solving gives exact solutions
on the active set, avoiding the slow O(1/t) convergence of Nesterov.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working fast LP solver."""
    for solver_name in [cp.CLARABEL, cp.GUROBI, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                b_val = b_val / s_sum
                return b_val
        except Exception:
            continue
    return None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    from scipy.sparse import lil_matrix, csc_matrix
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_qcqp_active_set(a, active_set, solver):
    """
    Solve: min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0
    """
    n = len(a)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_set:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.GUROBI, cp.CLARABEL, solver, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception as e:
            continue
    return None, None


def cutting_plane_qcqp(a0, solver, tol=1e-12, max_iters=30):
    """
    Cutting-plane method for the QCQP.
    Iteratively solves the problem with an active set, checks for violations,
    and adds violated constraints.
    """
    n = len(a0)
    a = a0.copy()
    b = fast_convolve(a)
    
    # Initial active set: top few peaks
    max_b = b.max()
    threshold = max_b * (1 - 1e-6)
    S = set(np.where(b >= threshold)[0].astype(int).tolist())
    
    # Limit active set size for solver efficiency
    if len(S) > 50:
        # Keep only the top 50
        idx = np.argsort(b)[::-1][:50]
        S = set(idx.astype(int).tolist())
    
    for it in range(max_iters):
        a_new, t_val = solve_qcqp_active_set(a, S, solver)
        if a_new is None:
            break
        
        # Check for violated constraints
        b_new = fast_convolve(a_new)
        max_new = b_new.max()
        
        if max_new <= t_val + tol:
            # Converged
            return a_new
        
        # Add violated constraints
        violated = np.where(b_new > t_val + tol)[0]
        if len(violated) > 0:
            S.update(violated.astype(int).tolist())
        
        # Limit active set size
        if len(S) > 100:
            b_check = fast_convolve(a_new)
            idx = np.argsort(b_check)[::-1][:100]
            S = set(idx.astype(int).tolist())
        
        a = a_new
    
    return a


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov accelerated LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                b_val = b_val / s_sum
                return b_val
        except Exception:
            continue
    return None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_qcqp_active_set(a_init, active_set, solver):
    """
    Solve: min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0
    Returns (a_opt, t_opt) or (None, None) on failure.
    """
    n = len(a_init)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active_set:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, cp.CLARABEL, solver, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def cutting_plane_qcqp(a0, solver, time_limit=300.0, tol=1e-12):
    """
    Cutting-plane method for the QCQP.
    Iteratively solves with growing active set until no violations.
    """
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()

    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - 1e-5)
    S = list(np.where(b >= threshold)[0].astype(int))

    if len(S) > 30:
        idx = np.argsort(b)[::-1][:30]
        S = sorted(idx.astype(int).tolist())

    best_a = a.copy()
    best_score = 2.0 * n * max_b

    for it in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        sol_t0 = time.monotonic()
        a_new, t_val = solve_qcqp_active_set(a, S, solver)
        sol_time = time.monotonic() - sol_t0

        if a_new is None:
            print(f"    QCQP solve failed at iter {it}")
            break

        b_new = fast_convolve(a_new)
        max_new = b_new.max()
        score_new = 2.0 * n * max_new

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        print(f"    CutPlane iter={it} |S|={len(S)} t_val={t_val:.10f} "
              f"max_new={max_new:.10f} viol={max_new - t_val:.2e} time={sol_time:.1f}s")

        if max_new <= t_val + tol:
            print(f"    Converged: no violations")
            return best_a

        violated = np.where(b_new > t_val + 1e-10)[0]
        if len(violated) > 0:
            S = sorted(set(S) | set(violated.astype(int).tolist()))
            if len(S) > 60:
                idx = np.argsort(b_new)[::-1][:60]
                S = sorted(idx.astype(int).tolist())

        a = a_new

    return best_a


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov accelerated LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-100:]) * 1000 if len(lp_times) >= 100 else 0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): "
                  f"score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 200 and np.mean(lp_times[-200:]) > 10.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def evaluate_local(a):
    """Evaluate sequence score locally (fast, no external call)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def optimize_for_n(a_init, n_target, solver, time_limit):
    """Optimize a sequence of given length using LP alternating + QCQP polish."""
    if n_target != len(a_init):
        a_init = interpolate_sequence(a_init, n_target)

    n = len(a_init)
    a = np.maximum(a_init, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    initial_score = evaluate_local(a)
    print(f"    n={n} initial score={initial_score:.10f}")

    # Phase A: LP alternating
    t_lp = time_limit * 0.5
    a_lp, score_lp = nesterov_lp_refine(a, t_lp, solver)
    if score_lp < initial_score:
        a = a_lp
        initial_score = score_lp

    # Phase B: Cutting-plane QCQP
    t_qcqp = time_limit - (time.monotonic() - (time_limit + time.monotonic()) + time_limit)
    elapsed_a = 0
    remaining = time_limit - (elapsed_a + 0)
    a_cp = cutting_plane_qcqp(a, solver, time_limit=min(remaining * 0.6, 200.0))
    score_cp = evaluate_local(a_cp)
    if score_cp < initial_score:
        a = a_cp
        initial_score = score_cp

    return a, initial_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent or create initial
    2. LP alternating + QCQP polish for n=1200
    3. Try nearby n values
    4. Final QCQP polish
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP Alternating for n=1200 ===
    print(f"\n=== Phase 1: LP Alternating n={len(best_seq)} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000)
        update_best(a_lp, "lp_phase1")

    # === Phase 2: Cutting-plane QCQP polish ===
    print(f"\n=== Phase 2: Cutting-Plane QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.25, 250.0)
    if t_qcqp > 30:
        print(f"  Budget: {t_qcqp:.0f}s")
        a_cp = cutting_plane_qcqp(best_seq, solver, time_limit=t_qcqp)
        update_best(a_cp, "cutting_plane_n1200")

    # === Phase 3: Second LP round from polished ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        update_best(a_lp2, "lp_phase3")

    # === Phase 4: QCQP polish again ===
    print(f"\n=== Phase 4: QCQP Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp2 = min(rem * 0.10, 100.0)
    if t_qcqp2 > 15:
        a_cp2 = cutting_plane_qcqp(best_seq, solver, time_limit=t_qcqp2)
        update_best(a_cp2, "qcqp_round2")

    # === Phase 5: Try n=1199 ===
    print(f"\n=== Phase 5: n=1199 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_1199 = min(rem * 0.12, 150.0)
        print(f"  Budget: {t_1199:.0f}s")
        init_1199 = interpolate_sequence(best_seq, 1199)
        a_1199, sc_1199 = nesterov_lp_refine(init_1199, t_1199 * 0.6, solver, n_max_iters=200000)
        update_best(a_1199, "lp_n1199")
        rem = deadline - time.monotonic()
        if rem > 30 and (best_seq is None or len(best_seq) != 1199 or evaluate_local(a_1199) < best_score + 0.001):
            a_cp1199 = cutting_plane_qcqp(a_1199, solver, time_limit=min(rem * 0.5, 80.0))
            update_best(a_cp1199, "qcqp_n1199")

    # === Phase 6: Try n=1201 ===
    print(f"\n=== Phase 6: n=1201 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_1201 = min(rem * 0.12, 150.0)
        print(f"  Budget: {t_1201:.0f}s")
        init_1201 = interpolate_sequence(best_seq, 1201)
        a_1201, sc_1201 = nesterov_lp_refine(init_1201, t_1201 * 0.6, solver, n_max_iters=200000)
        update_best(a_1201, "lp_n1201")
        rem = deadline - time.monotonic()
        if rem > 30:
            a_cp1201 = cutting_plane_qcqp(a_1201, solver, time_limit=min(rem * 0.5, 80.0))
            update_best(a_cp1201, "qcqp_n1201")

    # === Phase 7: Try n=1198 and n=1202 ===
    print(f"\n=== Phase 7: n=1198,1202 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        for n_try in [1198, 1202]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_try = min(rem * 0.10, 80.0)
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_refine(init_try, t_try * 0.6, solver, n_max_iters=100000)
            update_best(a_try, f"lp_n{n_try}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_cpt = cutting_plane_qcqp(a_try, solver, time_limit=min(rem * 0.4, 40.0))
                update_best(a_cpt, f"qcqp_n{n_try}")

    # === Phase 8: Final QCQP on best ===
    print(f"\n=== Phase 8: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_final = cutting_plane_qcqp(best_seq, solver, time_limit=min(rem * 0.5, 60.0))
        update_best(a_final, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
