def iterative_active_set_qcqp(a_init, solver, max_iters=50, time_limit=300):
    n = len(a_init)
    a = a_init.copy()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find all active constraints (within tolerance)
        threshold = max_b * (1.0 - 1e-6)
        active = np.where(b >= threshold)[0]
        
        # Also add near-active constraints
        threshold2 = max_b * (1.0 - 0.01)
        near_active = np.where(b >= threshold2)[0]
        active = np.union1d(active, near_active)
        
        if len(active) > 20:
            # Keep only top 20
            active = active[np.argsort(b[active])[-20:]]
        
        # Solve QCQP with these active constraints
        a_new = solve_qcqp(a, active, solver)
        
        if a_new is None:
            break
        
        # Check if any constraint is violated
        b_new = fast_convolve(a_new)
        new_max = b_new.max()
        
        if new_max <= max_b * (1.0 + 1e-8):
            # Converged
            a = a_new
            break
        
        a = a_new
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP alternating with over-relaxation (main workhorse)
2. Iterative active-set convex QCQP polish (exact via CVXPY)
3. Peak-shaving projected gradient for fine-tuning
4. Multi-n exploration (n=1199, 1200, 1201)
5. Dual gap tracking for convergence certification

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b_value, M_value) or (None, None)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    solved = False
    for s in solvers_to_try:
        if s is None or (s != solver and not solved and solvers_to_try.index(s) < solvers_to_try.index(solver)):
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_sum = b_val.sum()
    if s_sum < 1e-15:
        return None, None
    b_val = b_val / s_sum
    M_val = float(max(0, M_var.value))
    return b_val, M_val


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters, over_relax=1.0):
    """Nesterov dual averaging LP alternating with optional over-relaxation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    lp_times = []
    print_every = 200

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        alpha_eff = alpha * over_relax

        # Nesterov momentum update for y
        y = (1.0 - alpha_eff) * y + alpha_eff * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Primal update
        a = ((it + 1) * a + alpha_eff * (it + 2) * b_new) / (it + 1 + alpha_eff * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection and restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                # Add structured noise
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % print_every == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-min(100, len(lp_times)):]) * 1000
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        # Stop if LPs are too slow
        if len(lp_times) > 50 and np.mean(lp_times[-50:]) > 3.0:
            print(f"    LP too slow ({np.mean(lp_times[-50:]):.2f}s/iter), stopping at iter {it+1}")
            break

    return best_a, best_score


def solve_active_qcqp(a_init, solver, rel_tol=1e-5, max_active=15, time_limit=120):
    """
    Iterative active-set QCQP:
    min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0
    Iteratively adds violated constraints.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_convolve(a).max()

    for outer_it in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        if max_b < best_score * (1 - 1e-10):
            best_score = max_b
            best_a = a.copy()

        # Find active set (constraints near the maximum)
        threshold = max_b * (1.0 - rel_tol)
        active = np.where(b >= threshold)[0]

        if len(active) > max_active:
            active = active[np.argsort(b[active])[-max_active:]]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Solve QCQP
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        a_new = None
        for s_solve in solvers_to_try:
            if s_solve is None:
                continue
            try:
                prob.solve(solver=s_solve, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_new = np.asarray(a_var.value, dtype=np.float64)
                    a_new = np.maximum(a_new, 0)
                    sn = a_new.sum()
                    if sn < 1e-15:
                        a_new = None
                    else:
                        a_new = a_new / sn
                    break
            except Exception:
                continue

        if a_new is None:
            break

        # Check for new violations
        b_new = fast_convolve(a_new)
        new_max = b_new.max()
        t_opt = float(max(0, t_var.value)) if t_var.value is not None else new_max

        # Find violated constraints
        violated = np.where(b_new > t_opt * (1.0 + 1e-6))[0]

        if len(violated) == 0:
            # No violations - converged
            a = a_new
            final_max = fast_convolve(a).max()
            if final_max < best_score:
                best_score = final_max
                best_a = a.copy()
            break

        # Add violated to active set and continue
        a = a_new
        if new_max < best_score:
            best_score = new_max
            best_a = a.copy()

        # Check convergence
        if abs(new_max - max_b) < 1e-12:
            break

    return best_a, best_score * 2.0 * n


def peak_shave_gradient(a0, time_limit, n_steps=50000):
    """
    Projected multiplicative gradient descent targeting active peaks.
    Uses the subgradient of max_k a^T Q_k a which is in conv{Q_k a : k active}.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.02
    no_improve = 0
    cooldown = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if cooldown > 0:
            cooldown -= 1
            continue

        b = fast_convolve(a)
        max_b = b.max()

        # Find active peaks (top few)
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient: average of Q_k a over active k
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Projected gradient step (Euclidean projection onto simplex)
        # Use multiplicative (mirror) descent for non-negativity
        a_new = a * np.exp(-eta * g)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        # Ensure non-negativity
        a_new = np.maximum(a_new, 0)
        sn2 = a_new.sum()
        if sn2 > 1e-15:
            a_new = a_new / sn2

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            eta *= 0.7
            no_improve = 0
            cooldown = 10
            if eta < 1e-5:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def solve_full_qcqp_iterative(a_init, solver, time_limit=180):
    """
    More aggressive iterative active-set QCQP.
    Starts with a small active set and grows it.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    active_set = set()

    # Initialize active set with top peaks
    b = fast_convolve(a)
    top_k = np.argsort(b)[-5:]
    active_set = set(int(k) for k in top_k)

    for it in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve QCQP with current active set
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in sorted(active_set):
            Q = build_q_sparse(k, n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        a_new = None
        t_val = None
        for s_solve in [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS]:
            if s_solve is None:
                continue
            try:
                prob.solve(solver=s_solve, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_new = np.asarray(a_var.value, dtype=np.float64)
                    a_new = np.maximum(a_new, 0)
                    sn = a_new.sum()
                    if sn < 1e-15:
                        a_new = None
                    else:
                        a_new = a_new / sn
                    t_val = float(max(0, t_var.value))
                    break
            except Exception:
                continue

        if a_new is None:
            break

        b_new = fast_convolve(a_new)
        new_max = b_new.max()

        # Update best
        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()

        # Find violated constraints
        if t_val is not None:
            violated = np.where(b_new > t_val + 1e-10)[0]
        else:
            violated = np.where(b_new > new_max * (1.0 - 1e-8))[0]

        if len(violated) == 0:
            # Converged - no violations
            break

        # Add violated constraints (limit to prevent explosion)
        if len(active_set) < 25:
            for k in violated:
                active_set.add(int(k))
        else:
            # Keep only the most violated
            worst = violated[np.argmax(b_new[violated])]
            active_set.add(int(worst))
            # Remove least violated from active set if too many
            if len(active_set) > 25:
                active_list = list(active_set)
                active_list.sort(key=lambda k: b_new[k], reverse=True)
                active_set = set(active_list[:25])

        a = a_new

    return best_a, 2.0 * n * best_max

        init_1199 = interpolate_sequence(best_seq, 1199)
        a_1199, _ = nesterov_lp_refine(init_1199, t_1199 * 0.6, solver, n_max_iters=300000, over_relax=1.0)
        update_best(a_1199, "lp_n1199")
        rem = deadline - time.monotonic()
        if len(best_seq) == 1199 and best_score < 1.504 and rem > 30:
            a_q, _ = solve_full_qcqp_iterative(best_seq, solver, time_limit=min(rem * 0.3, 60))
            update_best(a_q, "qcqp_n1199")

    # === Phase 7: Try n=1201 ===
    print(f"\n=== Phase 7: Try n=1201 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_1201 = min(rem * 0.12, 180.0)
        print(f"  Budget: {t_1201:.0f}s")
        # Interpolate from the best n=1200 solution
        if best_seq is not None and len(best_seq) != 1201:
            init_1201 = interpolate_sequence(best_seq, 1201)
        else:
            init_1201 = best_seq.copy()
        a_1201, _ = nesterov_lp_refine(init_1201, t_1201 * 0.6, solver, n_max_iters=300000, over_relax=1.0)
        update_best(a_1201, "lp_n1201")
        rem = deadline - time.monotonic()
        if len(best_seq) == 1201 and best_score < 1.504 and rem > 30:
            a_q, _ = solve_full_qcqp_iterative(best_seq, solver, time_limit=min(rem * 0.3, 60))
            update_best(a_q, "qcqp_n1201")

    # === Phase 8: Final combined polish on best ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Quick peak shave
        t_ps2 = min(rem * 0.2, 30.0)
        a_ps2, _ = peak_shave_gradient(best_seq, t_ps2, n_steps=50000)
        update_best(a_ps2, "final_ps")

        # Final QCQP
        rem = deadline - time.monotonic()
        if rem > 20:
            a_qf, _ = solve_active_qcqp(best_seq, solver, rel_tol=1e-7, max_active=5, time_limit=min(rem * 0.5, 60))
            update_best(a_qf, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
