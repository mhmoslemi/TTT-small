  conv[4] = a^2
  
  To minimize max(a^2, 2ab, b^2+2a^2) subject to 2a+b=1:
  If a^2 = b^2+2a^2, then b^2 = -a^2, impossible.
  If 2ab = b^2+2a^2, then 2ab = b^2+2a^2 => b^2-2ab+2a^2=0 => (b-a)^2+a^2=0, impossible.
  If a^2 = 2ab, then a=2b (or a=0). With 2a+b=1: 4b+b=1, b=1/5, a=2/5.
  Then: a^2=4/25, 2ab=4/25, b^2+2a^2=1/25+8/25=9/25. Max=9/25.
  If 2ab = a^2, then b=a/2. With 2a+b=1: 2a+a/2=1, 5a/2=1, a=2/5, b=1/5. Same as above.
  
  So the best symmetric solution for n=3 is [2/5, 1/5, 2/5] with obj=54/25=2.16 > 2.
  
  What about non-symmetric? Try a=[a,b,c]:
  conv[0]=a^2, conv[1]=2ab, conv[2]=b^2+2ac, conv[3]=2bc, conv[4]=c^2
  This is getting complex. The uniform gives 2, which seems hard to beat for n=3.

```python
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    s = a.sum()
    a = a / s  # normalize to sum=1
    
    b = fast_convolve(a)
    max_b = b.max()
    lse = max_b + T * np.log(np.sum(np.exp((b - max_b) / T)))
    obj = 2.0 * n * lse
    
    # Gradient of lse w.r.t. a: w = softmax(b/T)
    w = np.exp((b - max_b) / T)
    w = w / w.sum()
    
    # Gradient w.r.t. a: 2 * correlate(w, a) (centered)
    g_a = 2.0 * np.correlate(w, a, mode='full')[n-1:2*n-1]
    
    # Gradient w.r.t. z: a * (g_a - sum(g_a * a))
    S = np.dot(g_a, a)
    grad_z = a * (g_a - S)
    
    return 2.0 * n * grad_z, obj
```

Let me also think about the **structure of the optimal solution**. 

For the auto-convolution to be flat, we need:
sum_{j} a[j] * a[k-j] ≈ constant for all k.

This is a system of n(2n-1)/2 equations (actually 2n-1 equations by symmetry) in n unknowns. For n > 1, this is underdetermined, so there should be many solutions. The question is whether a non-negative solution exists that's close to flat.

For the uniform sequence, the convolution is triangular (linearly decreasing from center). To make it flatter, we need to "lift" the edges and "lower" the center. This means putting more mass at the edges.

A natural candidate: a[i] proportional to 1/sqrt(1 + |i - (n-1)/2|) or similar.

Actually, let me think about it differently. If a[i] = c * f(i/n) for some function f, then:
conv[k] ≈ c^2 * n * ∫ f(x) f(x + (k/n - 1/2)) dx (for large n, approximately)

We want this to be constant in k. By Fourier analysis, this means the Fourier transform of f should have constant magnitude (which is impossible for a real non-negative function with compact support, but we can approximate).

The best approximation in the non-negative, compactly-supported case would be something like a raised cosine or a Fejér-type kernel... but actually, we want the OPPOSITE of flat spectrum. We want flat autocorrelation, which means flat spectrum. But a non-negative function cannot have exactly flat spectrum (the DC component is always the largest).

The best we can do is minimize the variance of the autocorrelation. This is related to the "minimum variance" design in experimental design theory.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def lse_objective_grad(z, n, T):
    """
    LogSumExp smoothed objective and gradient in log-space.
    z is the log-parameter, a = softmax(z).
    Returns (objective, gradient_wrt_z).
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    
    # LogSumExp
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    # Weights
    w = exp_terms / exp_terms.sum()
    
    # Gradient w.r.t. a
    # d(lse)/d(a_i) = (2/T) * sum_k w[k] * a[k-i] (for valid k-i)
    # This is (2/T) * correlate(w, a) centered
    g_a_raw = np.correlate(w, a, mode='full')
    g_a = 2.0 * g_a_raw[n-1:2*n-1] / T
    
    # Convert to z-space
    S = np.dot(g_a, a)
    grad_z = a * (g_a - S)
    
    return obj, grad_z

def lbfgs_logsumexp(a0, time_limit, T_values=None, max_iter=500):
    """
    L-BFGS-B optimization in log-space with LogSumExp smoothing.
    T_values: list of temperatures to anneal through.
    """
    n = len(a0)
    if T_values is None:
        T_values = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    
    z = np.log(np.maximum(a0, 1e-300))
    z -= z.mean()  # centering
    
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a0.copy()
    best_score = evaluate_local(a0)
    
    for T in T_values:
        if time.monotonic() > deadline:
            break
        
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        # Each T stage gets a portion of remaining time
        stages_left = len(T_values) - T_values.index(T)
        t_per_stage = min(remaining / stages_left, 60.0)
        
        def fun_g(z_vec):
            return lse_objective_grad(z_vec, n, T)
        
        result = minimize(fun_g, z, method='L-BFGS-B',
                         options={'maxiter': 200, 'maxfun': 500,
                                 'gtol': 1e-8, 'iprint': -1})
        
        if result.x is not None:
            z = result.x
            z -= z.mean()
            a_new = normalize(np.exp(z - z.max()))
            score = evaluate_local(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-300))
                z -= z.mean()
        
        elapsed = time.monotonic() - t0
        if (T_values.index(T) + 1) % 3 == 0:
            print(f"    L-BFGS T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def nesterov_lp(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov LP alternating."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize(y)
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = normalize(a)
        
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
        
        # Stall detection and restart
        if (it + 1) % 500 == 0:
            # Periodic perturbation to escape local optima
            if it > 1000:
                x = np.linspace(0, 2*np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2*np.pi))
                y = y + noise
                y = normalize(y)
    
    return best_a, best_score

def subgradient_nesterov(a0, time_limit, n_steps=500000):
    """Nesterov-accelerated subgradient method on max(conv(a,a))."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # velocity
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    gamma0 = 0.1  # initial step size
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Evaluate at lookahead point
        b_y = fast_convolve(y)
        k_star = np.argmax(b_y)
        
        # Subgradient at y: 2*y[k_star - i] for valid i
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            g[i] = 2.0 * y[k_star - i]
        
        # Nesterov update
        gamma_t = gamma0 / np.sqrt(step + 1)
        v = 0.9 * v - gamma_t * g
        a_new = a + v
        a_new = normalize(a_new)
        
        # Update y (lookahead)
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * (step+1)**2)) / 2.0
        y = (beta_t * a_new - (step+1) * a) / (beta_t - 1.0) if step > 0 else a_new.copy()
        y = normalize(y)
        
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradNest step={step+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating with PDLP/CLARABEL (tight tolerances, many iterations)
2. L-BFGS-B on LogSumExp smoothed objective (new, quasi-Newton steps)
3. Nesterov-accelerated subgradient on max function
4. Multi-length exploration (n=1200, 1500, 2000)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast, for internal use)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring accurate ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            kwargs = {}
            if s == cp.PDLP:
                kwargs = {'max_iters': 500, 'eps_abs': 1e-8, 'eps_rel': 1e-8}
            elif s == cp.CLARABEL:
                kwargs = {'tol_gap': 1e-8, 'tol_feas': 1e-8, 'tol_infeas': 1e-8}
            prob.solve(solver=s, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                    break
            except Exception:
                continue
    else:
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def lse_objective_and_grad(z, n, T):
    """
    LogSumExp smoothed objective and gradient in log-space.
    z is log-parameter, a = softmax(z), sum(a)=1.
    Returns (objective_value, gradient_wrt_z).
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    b = fast_convolve(a)
    max_b = b.max()

    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()

    g_a_raw = np.correlate(w, a, mode='full')
    g_a = 2.0 * g_a_raw[n - 1:2 * n - 1] / T

    S = np.dot(g_a, a)
    grad_z = a * (g_a - S)

    return obj, grad_z


def lse_wrapper(z, n, T):
    """Top-level wrapper for scipy minimize (no closure)."""
    return lse_objective_and_grad(z, n, T)

def nesterov_lp(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov LP alternating with periodic restarts."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    prev_best = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize(y)

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = normalize(a)

        score_a = evaluate_local(a)
        score_y = evaluate_local(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            elapsed = time.monotonic() - t0
            if (it + 1) % 1000 == 0:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

        if stall_count >= 30 and (it + 1) % 200 == 0:
            x = np.linspace(0, 2 * np.pi, n)
            freq = rng.integers(1, 8)
            noise_scale = 0.01 + 0.01 * min(stall_count // 30, 5)
            noise = noise_scale * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
            y = y + noise
            y = normalize(y)
            stall_count = 0

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, n_steps=500000):
    """Nesterov-accelerated subgradient on max(conv(a,a))."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    gamma0 = 0.5

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_convolve(y)
        k_star = int(np.argmax(b_y))

        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            g[lo:hi + 1] = 2.0 * y[k_star - hi:k_star - lo + 1][::-1]

        gamma_t = gamma0 / math.sqrt(step + 1)
        v = 0.9 * v - gamma_t * g
        a_new = a + v
        a_new = normalize(a_new)

        if step == 0:
            y = a_new.copy()
        else:
            beta_t = (1.0 + math.sqrt(1.0 + 4.0 * (step + 1) ** 2)) / 2.0
            y = (beta_t * a_new - (step + 1) * a) / (beta_t - 1.0)
            y = normalize(y)

        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if (step + 1) % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def geometric_combine(a1, a2, alpha=0.5):
    """Geometric mean combination of two normalized sequences."""
    a1 = normalize(a1)
    a2 = normalize(a2)
    n = len(a1)
    combined = a1 ** alpha * a2 ** (1.0 - alpha)
    return normalize(combined)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)
```
