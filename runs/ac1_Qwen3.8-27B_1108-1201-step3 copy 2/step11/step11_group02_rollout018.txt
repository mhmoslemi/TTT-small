import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_sparse_conv_matrix(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
        data.extend(a)
    T = csr_matrix((data, (np.array(row_ind), np.array(col_ind))), shape=(size, n))
    return T

def solve_lp_best_response(a_fixed, T_sparse, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_sparse @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:

        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=100000, top_k_peaks=3):

    Nesterov LP alternating with convolution-aware stall recovery.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0
    stall_threshold = 10

    rng = np.random.default_rng(99999)

        b_new = solve_lp_best_response(y, T_sparse, solver)
        if b_new is None:
            continue

        # Nesterov step
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # FTRL-style average
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:

            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 15 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_threshold:
                # Convolution-aware perturbation
                b_cur = fast_convolve(y)
                # Find top-k peaks
                peak_threshold = b_cur.max() * 0.999
                peaks = np.where(b_cur >= peak_threshold)[0]

                for pk in peaks[:top_k_peaks]:
                    js = max(0, pk - n + 1)
                    je = min(n - 1, pk)
                    idx = np.arange(js, je + 1)
                    influence[idx] += y[idx] * y[pk - idx]

                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:

def subgradient_momentum(a0, time_limit, n_max_iters=500000, top_k=5):
    """
    Subgradient method with Nesterov momentum and Polyak step size.
    Tracks top-k peaks for richer subgradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Lower bound: 2n/(2n-1)
    lb = 2.0 * n / (2 * n - 1)
    
    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum coefficient
    
    # Polyak parameters
    eta_min = 1e-8
    eta_max = 1.0
    eta = 0.1
    no_improve = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 0.5 and it > 100:
            break

        # Compute convolution and find top-k peaks
        b = fast_convolve(y)
        max_b = b.max()
        score = 2.0 * n * max_b  # since sum(y)=1
        
        # Find top-k peak positions
        if top_k <= len(b):
            peak_positions = np.argsort(b)[::-1][:top_k]
        else:
            peak_positions = np.array([int(np.argmax(b))])
        
        # Only use peaks that are close to max (within 1%)
        peak_positions = peak_positions[b[peak_positions] >= max_b * 0.999]
        if len(peak_positions) == 0:
            peak_positions = np.array([int(np.argmax(b))])
        
        # Compute subgradient: average of shifted a over active peaks
        g = np.zeros(n)
        for pk in peak_positions:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            g[idx] += y[pk - idx]
        g /= len(peak_positions)
        
        # Nesterov momentum update
        v = beta * v + (1 - beta) * g
        # Polyak step size
        gap = score - lb
        g_norm_sq = np.dot(g, g)
        if g_norm_sq > 1e-20:
            eta_polyak = gap / (4.0 * g_norm_sq)
        else:
            eta_polyak = eta
        eta = max(eta_min, min(eta_max, 0.5 * eta + 0.5 * eta_polyak))
        
        y = y - eta * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
            sy = 1.0
        y = y / sy
        
        # Update a (running average)
        t_eff = it + 1
        a = (t_eff * a + y) / (t_eff + 1)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        # Check improvement
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic shake
        if no_improve >= 2000:
            rng_shake = np.random.default_rng(it * 7 + 3)
            noise = rng_shake.standard_normal(n) * 0.02
            y = y + noise
            y = np.maximum(y, 0)
            y = y / y.sum()
            a = y.copy()
            v = np.zeros(n)
            no_improve = 0
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_max_sweeps=500):
    """
    Coordinate descent: for each coordinate, try to find a better value
    by redistributing mass from other coordinates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for sweep in range(n_max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved_in_sweep = 0
        b = fast_convolve(a)
        max_b = b.max()
        
        # Process coordinates in order of influence on the peak
        peak_pos = int(np.argmax(b))
        js = max(0, peak_pos - n + 1)
        je = min(n - 1, peak_pos)
        
        # Compute influence of each coordinate on the peak
        influence = np.zeros(n)
        idx = np.arange(js, je + 1)
        influence[idx] = a[idx] * a[peak_pos - idx]
        
        # Process in order of decreasing influence (most impactful first)
        order = np.argsort(influence)[::-1]
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            
            # Try reducing a[i] by various fractions
            for frac in [0.7, 0.8, 0.9]:
                delta = a[i] * (1.0 - frac)
                if delta < 1e-15:
                    continue
                
                # Redistribute delta proportionally to other elements
                a_trial = a.copy()
                a_trial[i] = a[i] * frac
                others = np.ones(n)
                others[i] = 0
                others_sum = a_trial @ others  # sum of all other elements
                if others_sum < 1e-15:
                    break
                a_trial[others.astype(bool)] += delta * a_trial[others.astype(bool)] / others_sum
                a_trial /= a_trial.sum()
                
                sc = evaluate_local(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved_in_sweep += 1
                    break
            
            # Early exit if too slow
            if time.monotonic() - t0 > time_limit - 5:
                break
        
        # If no improvement in this sweep, break
        if improved_in_sweep == 0:
            # Try a random perturbation
            rng_cd = np.random.default_rng(sweep * 13 + 7)
            noise = rng_cd.standard_normal(n) * 0.005
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if (sweep + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc sweep={sweep+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
