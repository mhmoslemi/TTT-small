"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid LP + LogSumExp gradient approach.
Key innovations over previous:
- Precomputed sparse matrix structure (O(n^2) build once, O(nnz) update per iteration)
- scipy HiGHS solver for fast LP best-response computation
- FFT-accelerated LogSumExp gradient via np.correlate for O(n log n) gradient eval
- Adaptive hybrid: LP every K iterations, LSE gradient steps in between
- Multi-n sweep with interpolation-based transfer
- Multi-stage LSE annealing for precise convergence
- Coordinate descent polish targeting peak-contributing elements
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, hstack


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Evaluate the objective (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


class ConvMatrix:
    """Precomputed structure for convolution constraint matrix T(a)."""

    def __init__(self, n):
        self.n = n
        self.size = 2 * n - 1
        row_ind = []
        col_ind = []
        for j in range(n):
            row_ind.extend(range(j, j + n))
            col_ind.extend([j] * n)
        self.row_ind = np.array(row_ind, dtype=np.int32)
        self.col_ind = np.array(col_ind, dtype=np.int32)
        self._indices = self.row_ind - self.col_ind
        self._data = None

    def update(self, a):
        self._data = a[self._indices].astype(np.float64, copy=False)

    def get_csr(self):
        return csr_matrix((self._data, (self.row_ind, self.col_ind)),
                          shape=(self.size, self.n))


def solve_lp_best_response(a, conv_mat, solver=None):
    """Solve: min t s.t. T(a)@b <= t, b >= 0, sum(b) = 1. Returns (b*, t*)."""
    n = len(a)
    size = 2 * n - 1
    conv_mat.update(a)
    T_csr = conv_mat.get_csr()

    # Use scipy linprog with HiGHS
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        negative_col = -csr_matrix(np.ones((size, 1)))
        A_ub = hstack([T_csr, negative_col], format='csr')
        b_ub = np.zeros(size)
        A_eq_row = np.ones(n + 1)
        A_eq_row[-1] = 0.0
        A_eq = csr_matrix(A_eq_row.reshape(1, -1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 5.0, 'presolve': True})
        if result.status != 0:
            return None, None
        b_val = np.maximum(result.x[:n], 0)
        t_val = result.x[n]
    except Exception:
        # Fallback to cvxpy
        b_val, t_val = _solve_lp_cvxpy(a, conv_mat, solver)
        return b_val, t_val

    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(t_val)


def _solve_lp_cvxpy(a, conv_mat, solver=None):
    """Fallback LP solve using cvxpy."""
    if solver is None:
        solver = _find_solver()
    n = len(a)
    conv_mat.update(a)
    T = conv_mat.get_csr()
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
    except Exception:
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    t_val = float(t_var.value) if t_var.value is not None else fast_convolve(a).max()
    return b_val, t_val


_solver_found = None


def _find_solver():
    global _solver_found
    if _solver_found is not None:
        return _solver_found
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _solver_found = s
                return s
        except Exception:
            continue
    _solver_found = cp.ECOS
    return cp.ECOS


def lse_score_and_grad(a, T):
    """Compute LogSumExp smoothed objective and gradient. a must have sum=1, a>=0."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    # grad[i] = (4n/T) * sum_j w[i+j] * a[j]
    g = np.correlate(w, a, mode='valid')
    grad = (4.0 * n / T) * g
    # Objective: 2n * LSE_T(b) = 2n * T * log(sum(exp(b/T)))
    obj = 2.0 * n * (max_b + T * np.log(exp_b.sum()))
    return obj, grad


def nesterov_lp_hybrid(a0, time_limit, conv_mat, lp_every=8,
                       T_init=5.0, T_min=0.3, max_iters=1000000):
    """
    Hybrid Nesterov LP + LSE gradient optimization.
    Every `lp_every` iterations, do an LP best response.
    Between LP calls, do LSE gradient steps with adaptive learning rate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()

    y = a.copy()  # Nesterov momentum variable
    eta = T_init
    stall = 0
    prev_best = best_score
    lr = 0.1

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if (it + 1) % lp_every == 0:
            # LP best response
            b_new, M_lp = solve_lp_best_response(y, conv_mat)
            if b_new is not None:
                # Nesterov update
                alpha = 2.0 / (np.sqrt(it % 5000 + 1) + 1.0)
                y = (1.0 - alpha) * y + alpha * b_new
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                # Weighted average
                a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                else:
                    a = y.copy()

                sc = eval_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                    eta = max(T_min, eta * 0.95)
                else:
                    sc_y = eval_score(y)
                    if sc_y < best_score:
                        best_score = sc_y
                        best_a = y.copy()
                        a = y.copy()
                        stall = 0
                    else:
                        stall += 1

                # Restart Nesterov on stall
                if stall >= 15:
                    y = best_a.copy()
                    a = best_a.copy()
                    stall = 0
                    eta = max(T_min, eta * 1.3)
                    lr = 0.1
        else:
            # LSE gradient step
            T_cur = max(T_min, eta)
            obj, grad = lse_score_and_grad(a, T_cur)
            # Project gradient onto tangent space of simplex
            grad_tangent = grad - np.mean(grad)
            # Adaptive learning rate
            gnorm = np.linalg.norm(grad_tangent)
            if gnorm > 1e-15:
                step = lr * grad_tangent / gnorm
                # Line search (backtracking)
                for _ in range(20):
                    a_trial = a - step
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-15:
                        a_trial = a_trial / st
                        sc = eval_score(a_trial)
                        if sc < best_score:
                            a = a_trial
                            y = a.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                            break
                    step *= 0.5
                else:
                    # No improvement; decay lr
                    lr *= 0.9
                    stall += 1
                    if stall >= 30:
                        y = best_a.copy()
                        a = best_a.copy()
                        stall = 0
                        lr = 0.1
                        eta = max(T_min, eta * 1.5)

        # Decay eta slowly
        if (it + 1) % 100 == 0:
            eta = max(T_min, eta * 0.99)
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Hybrid iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}, eta={eta:.3f}")

    return best_a, best_score


def lse_lbgfs_refine(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp smoothed objective with multi-stage annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1]

    best_a = a.copy()
    best_score = eval_score(a)
    n_stages = len(T_stages)
    time_per_stage = time_limit / n_stages

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break
        stage_limit = time_per_stage

        def func_proj(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            obj, _ = lse_score_and_grad(a_trial, T)
            sc = eval_score(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
            return obj + 1e6 * (1.0 - st) ** 2  # penalty for sum != 1

        def jac_proj(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z)
            a_trial = a_trial / st
            _, grad = lse_score_and_grad(a_trial, T)
            # Penalty gradient
            grad_pen = -2e6 * (1.0 - st)
            return grad + grad_pen

        try:
            res = minimize(fun=func_proj, x0=a, jac=jac_proj, method='L-BFGS-B',
                           options={'maxiter': int(200 * time_per_stage / 5),
                                    'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 100000,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception:
            pass

        sc = eval_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        elapsed = time.monotonic() - t0
        print(f"    LSE-LBFGS T={T:.2f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coord_descent_refine(a0, time_limit, max_steps=500000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(6, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-14:
                continue
            for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                old_val = a_trial[pos]
                new_val = old_val * factor
                a_trial[pos] = new_val
                diff = old_val - new_val
                if diff < 1e-16:
                    continue
                # Redistribute to maintain sum
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = eval_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Try simultaneous small reduction
                a_trial = a.copy()
                for pos in top_positions[:3]:
                    if a_trial[pos] > 1e-14:
                        a_trial[pos] *= 0.999
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = eval_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 1000:
                    break
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length (preserves sum=1 normalization)."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new


def structured_init(n, rng):
    """Create a structured initialization."""
    r = rng.integers(0, 5)
    x = np.linspace(0, 1, n)
    if r == 0:
        a = np.cos(np.pi * x) ** 2
    elif r == 1:
        a = np.exp(-4 * x * (1 - x))
    elif r == 2:
        a = np.sin(np.pi * x) ** 0.8
    elif r == 3:
        a = np.exp(-3 * (x - 0.5) ** 2 / 0.15)
    else:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: n-sweep ===
    print(f"\n=== Phase 1: n-sweep (score={best_score:.8f}) ===")
    n_current = len(best_seq)
    n_candidates = sorted(set([
        max(500, n_current - 200),
        max(500, n_current - 100),
        n_current,
        n_current + 100,
        n_current + 200,
        n_current + 400
    ]))
    t_sweep_total = min(180.0, (deadline - time.monotonic()) * 0.1)
    per_n = t_sweep_total / max(1, len(n_candidates))
    print(f"  n candidates: {n_candidates}, ~{per_n:.0f}s each")

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        rem = min(per_n, deadline - time.monotonic() - 30)
        if rem < 10:
            continue
        a_interp = interpolate_seq(best_seq, n_try)
        conv_mat = ConvMatrix(n_try)
        a_res, _ = nesterov_lp_hybrid(a_interp, rem, conv_mat, lp_every=10,
                                       T_init=3.0, T_min=0.5)
        try_update(a_res, f"sweep_n{n_try}")

    # === Phase 2: Main hybrid LP optimization ===
    print(f"\n=== Phase 2: Main Hybrid LP (score={best_score:.8f}, n={len(best_seq)}) ===")
    n_main = len(best_seq)
    conv_mat_main = ConvMatrix(n_main)
    rem = deadline - time.monotonic()
    t_main = min(rem * 0.55, 1000.0)
    if t_main > 30:
        print(f"  Budget: {t_main:.0f}s")
        a_res, _ = nesterov_lp_hybrid(best_seq, t_main, conv_mat_main,
                                       lp_every=5, T_init=3.0, T_min=0.2)
        try_update(a_res, "main_lp")

    # === Phase 3: LSE L-BFGS refinement ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbgfs = min(rem * 0.25, 400.0)
    if t_lbgfs > 20:
        print(f"  Budget: {t_lbgfs:.0f}s")
        T_stages = [8.0, 5.0, 3.0, 2.0, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12, 0.08]
        a_res = lse_lbgfs_refine(best_seq, t_lbgfs, T_stages=T_stages)
        try_update(a_res, "lse_lbgfs")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_cd = min(rem * 0.15, 250.0)
    if t_cd > 15:
        print(f"  Budget: {t_cd:.0f}s")
        a_res, _ = coord_descent_refine(best_seq, t_cd)
        try_update(a_res, "coord_desc")

    # === Phase 5: Fine n adjustment ===
    print(f"\n=== Phase 5: Fine n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_fine = min(rem * 0.25, 300.0)
    if t_fine > 20:
        n_base = len(best_seq)
        fine_ns = [n_base - 50, n_base + 50, n_base - 100, n_base + 100]
        fine_ns = [n for n in fine_ns if n >= 200]
        per_fine = t_fine / max(1, len(fine_ns))
        print(f"  Fine n: {fine_ns}, ~{per_fine:.0f}s each")
        for n_f in fine_ns:
            if time.monotonic() > deadline - 15:
                break
            a_interp = interpolate_seq(best_seq, n_f)
            conv_mat_f = ConvMatrix(n_f)
            a_res, _ = nesterov_lp_hybrid(a_interp, min(per_fine, deadline - time.monotonic() - 10),
                                           conv_mat_f, lp_every=5, T_init=2.0, T_min=0.2)
            try_update(a_res, f"fine_n{n_f}")

    # === Phase 6: Final LSE polish ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_final = min(rem * 0.3, 150.0)
    if t_final > 10:
        T_final = [1.0, 0.5, 0.3, 0.2, 0.1, 0.06]
        a_res = lse_lbgfs_refine(best_seq, t_final, T_stages=T_final)
        try_update(a_res, "final_polish")

        # Final coord descent
        rem2 = deadline - time.monotonic()
        t_cd2 = min(rem2 * 0.5, 60.0)
        if t_cd2 > 5:
            a_res, _ = coord_descent_refine(best_seq, t_cd2, max_steps=200000)
            try_update(a_res, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
