    n = len(a)
    fft_size = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    # Add random phase perturbation

    A_perturbed = A * np.exp(1j * phase_perturb)

    s = a_new.sum()
    if s < 1e-11:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a refined multi-phase approach:
1. Exact subgradient with backtracking line search
2. LSE-smoothed gradient descent with temperature annealing  
3. Spectral-domain perturbation and refinement
4. Block coordinate descent with exact sub-solves
5. Multi-restart with diverse perturbations for escaping local minima

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_obj_grad(a, T):
    """Compute LogSumExp-smoothed objective and gradient."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse

    w = exp_c / exp_c.sum()

    # Gradient computation using FFT for cross-correlation
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    b = R[:n]

    g = 2.0 * b
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]

    return obj, g


def subgradient_step(a, eta, active_tol=1e-6):
    """One subgradient step with backtracking line search."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    
    # Active set: peaks within tolerance of max
    active = np.where(b >= b_max * (1.0 - active_tol))[0]
    
    # Compute subgradient (average of gradients over active set)
    g = np.zeros(n)
    for k in active:
        # ∂b[k]/∂a[m] = 2*a[k-m] if 0 <= k-m < n
        shift = k - np.arange(n)
        valid = (shift >= 0) & (shift < n)
        g[valid] += 2.0 * a[shift[valid]]
    g /= len(active)
    
    # Project gradient onto the constraint sum(a) = 1
    # The feasible direction should have zero sum (to preserve normalization)
    g = g - g.mean()
    
    # Backtracking line search
    best_a = a.copy()
    best_score = b_max  # We're tracking max_b (since sum=1)
    
    for scale in [eta, eta*0.5, eta*0.25, eta*0.1, eta*0.05]:
        a_new = a - scale * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        b_new = fast_convolve(a_new)
        max_new = b_new.max()
        if max_new < best_score:
            best_score = max_new
            best_a = a_new.copy()
    
    return best_a, best_score


def lse_descent(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """LSE-smoothed gradient descent with temperature annealing."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Gradient descent with backtracking
        lr = 1.0 / (n * stage_T + 1)
        for it in range(maxiter_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            obj, g = lse_obj_grad(a, stage_T)
            
            # Project gradient to preserve normalization
            g = g - g.mean()
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            g = g / g_norm
            
            # Backtracking line search
            improved = False
            for scale in [lr, lr*0.5, lr*0.25, lr*0.1]:
                a_new = a - scale * g
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                sc = evaluate_local(a_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    break
            
            if not improved:
                lr *= 0.7
                if lr < 1e-10:
                    break

    return best_a, best_score


def spectral_refine(a0, time_limit, n_iters=5000):
    """Refine by making the power spectrum flatter."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    lr = 0.01

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean() + 1e-15

        # Compute direction to flatten P
        r = (P - P_mean) / P_mean
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -r[valid] * P_mean * A[valid] / (P[valid] + 1e-10)
        H[0] = 0.0
        h = irfft(H * 2, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        improved = False
        for scale in [lr, lr*0.5, lr*0.1]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if not improved:
            lr *= 0.7

        if it % 500 == 0:
            lr *= 0.8

    return best_a, best_score

        end = start + block_size

        # Compute gradient for the block
        T = 0.1
        _, g_full = lse_obj_grad(a, T)
        g_block = g_full[start:end].copy()

        lr = 0.01

        for it in range(30):
            a_new_block = a_block_orig - lr * g_block
            a_new_block = np.maximum(a_new_block, 0)

        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, method='gaussian', strength=0.02):
    """Perturb a sequence."""
    n = len(a)
    x = np.linspace(0, 2*np.pi, n)
    if method == 'gaussian':
        a_new = a + rng.normal(0, strength, n)
    elif method == 'low_freq':
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2*np.pi)
        a_new = a + strength * np.cos(freq * x + phase)
    elif method == 'freq1':
        phase = rng.uniform(0, 2*np.pi)
        a_new = a + strength * np.cos(x + phase)
    elif method == 'freq2':
        phase = rng.uniform(0, 2*np.pi)
        a_new = a + strength * np.cos(2*x + phase)
    elif method == 'spectral_phase':
        fft_size = 1 << (2*n - 1).bit_length()
        A = rfft(a, fft_size)
        phase_pert = rng.uniform(-strength, strength, len(A))
        A_p = A * np.exp(1j * phase_pert)
        A_p[0] = A[0]
        a_new = irfft(A_p, fft_size)[:n]
    else:
        a_new = a + rng.normal(0, strength, n)
    
    return normalize(a_new)


def population_search(a_init, time_limit, pop_size=12, rng=None):
    """Population-based search."""
    if rng is None:
        rng = np.random.default_rng(456)

    n = len(a_init)
    methods = ['gaussian', 'low_freq', 'freq1', 'freq2', 'spectral_phase']
    
    population = [normalize(a_init.copy())]
    for i in range(pop_size - 1):
        m = methods[i % len(methods)]
        strength = 0.01 + 0.01 * (i % 3)
        population.append(perturb_sequence(a_init, rng, m, strength))

    scores = [evaluate_local(p) for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]

    t0 = time.monotonic()
    gen = 0
    adaptive_strength = 0.015

    while time.monotonic() - t0 < time_limit:
        gen += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        keep = max(3, pop_size // 3)
        elites = population[:keep]
        new_pop = list(elites)

        for i in range(pop_size - len(new_pop)):
            parent_idx = rng.integers(0, keep)
            parent = elites[parent_idx]
            method = methods[rng.integers(0, len(methods))]
            strength = adaptive_strength * (0.8 + 0.4 * np.random.random())
            child = perturb_sequence(parent, rng, method, strength)
            new_pop.append(child)

        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]

        if gen % 50 == 0:
            if scores[0] >= best_score - 1e-10:
                adaptive_strength *= 0.85
            else:
                adaptive_strength = min(0.04, adaptive_strength * 1.1)

    return best_a, best_score


def lse_descent_fast(a0, time_limit, T_stages, maxiter_per_stage=10000):
    """Faster LSE descent using vectorized operations."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        lr = 1.0 / (n * 0.1 + 1)
        iters = max(10, int(remaining * 20))
        iters = min(iters, maxiter_per_stage)
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            obj, g = lse_obj_grad(a, stage_T)
            g = g - g.mean()
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-14:
                break
            g = g / g_norm
            
            improved = False
            for scale in [lr, lr*0.5, lr*0.2]:
                a_new = a - scale * g
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                sc = evaluate_local(a_new)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    break
            
            if not improved:
                lr *= 0.6
                if lr < 1e-12:
                    break

    return best_a, best_score
