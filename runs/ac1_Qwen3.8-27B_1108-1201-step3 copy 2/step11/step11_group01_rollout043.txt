"""
Search for a low-peak non-negative linear-convolution sequence.

Strategy:
- Use the provided parent sequence when available.
- Interpolate to longer lengths (often improves the finite-n triangle effects).
- Use a fast custom active-set / smoothed-gradient line search in log-softmax
  coordinates.
- Use a Nesterov accelerated LP best-response method when CVXPY is available.
- Polish with a couple of short L-BFGS-B LogSumExp stages.
- Try several nearby lengths and keep the best sequence.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

try:
    import cvxpy as cp
except Exception:
    cp = None

_lp_solver = None


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1e6)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones_like(a)
        s = float(a.sum())
    return a / s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    A = rfft(a, N)
    b = irfft(A * A, N)[:L]
    # Remove tiny negative roundoff; it is harmless and keeps weights sane.
    if b.size:
        b = np.maximum(b, 0.0)
    return b


def score_norm(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s < 1e-15:
        return float("inf")
    b = fast_conv(a)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    return float(2.0 * n * M / (s * s))


def evaluate_safe(a):
    try:
        out = []
        for x in a:
            x = float(x)
            if x < 0.0:
                x = 0.0
            elif x > 1000.0:
                x = 1000.0
            out.append(x)
        if not out:
            return float("inf")
        val = evaluate_sequence(out)
        return float(val)
    except Exception:
        return float("inf")


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    n = len(z)
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    mz = np.max(z)
    if not np.isfinite(mz):
        return np.ones(n, dtype=np.float64) / n
    e = np.exp(z - mz)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return e / s


def z_from_a(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    z = np.log(np.maximum(a, 1e-300))
    if np.isfinite(z).all():
        z = z - float(np.mean(z))
    else:
        z = np.zeros(n, dtype=np.float64)
    return z


def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    old_n = len(a)
    if old_n == new_n:
        return normalize_seq(a)
    if old_n == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    if new_n <= 0:
        return np.array([1.0], dtype=np.float64)
    a = normalize_seq(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, int(new_n))
    out = np.interp(new_x, old_x, a)
    return normalize_seq(out)


def structured_init(n, seed=42):
    rng = np.random.default_rng(seed)
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    choice = int(rng.integers(0, 8))

    if choice == 0:
        p = float(rng.uniform(0.9, 1.6))
        a = x ** p
    elif choice == 1:
        p = float(rng.uniform(0.9, 1.6))
        a = (1.0 - x) ** p
    elif choice == 2:
        p = float(rng.uniform(0.7, 1.4))
        q = float(rng.uniform(0.0, 0.4))
        a = (x + 1e-3) ** p * (1.0 - x) ** q
    elif choice == 3:
        p = float(rng.uniform(0.0, 0.4))
        q = float(rng.uniform(0.7, 1.4))
        a = x ** p * (1.0 - x + 1e-3) ** q
    elif choice == 4:
        a = 1.0 + 0.25 * np.sin(2.0 * np.pi * x)
    elif choice == 5:
        a = 1.0 + 0.20 * np.cos(2.0 * np.pi * x - float(rng.uniform(0, 2 * np.pi)))
    elif choice == 6:
        a = np.ones(n)
        for k in range(1, 5):
            amp = float(rng.uniform(0.03, 0.12)) / k
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp * np.cos(2.0 * np.pi * k * x + ph)
    else:
        a = 1.0 + 0.15 * np.sin(2.0 * np.pi * x) + 0.08 * np.cos(4.0 * np.pi * x)

    a = np.maximum(a, 1e-6)
    return normalize_seq(a)


def make_weights(b, M, temp):
    b = np.asarray(b, dtype=np.float64)
    M = float(M)
    if M <= 1e-300:
        w = np.ones_like(b)
        w /= max(1, w.size)
        return w
    scale = max(1e-300, M * max(1e-12, float(temp)))
    x = (b - M) / scale
    x = np.clip(x, -40.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        s = 1.0
    return e / s


def grad_from_weights(a, w):
    """
    For weighted lags w, compute grad_a_j = sum_k w_k * d(b_k)/d(a_j).
    Here d(b_k)/d(a_j) = 2 * a_{k-j} when valid.
    The valid correlation is computed by FFT.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()

    # Numerically enforce zero sum? Not necessary, but helps.
    ws = float(w.sum())
    if ws < 1e-300 or not np.isfinite(ws):
        w = np.zeros(L)
        w[0] = 1.0
    else:
        w = w / ws

    w_rev = w[::-1]
    A = rfft(a, N)
    W = rfft(w_rev, N)
    c = irfft(W * A, N)[:L]

    idx = np.arange(L - 1, L - n - 1, -1)
    g = c[idx]
    g = np.maximum(g, 0.0)
    return 2.0 * g


def active_subgradient_polish(
    a0,
    deadline,
    temp_start=0.08,
    temp_end=0.003,
    max_steps=500000,
    prefix="ACT"
):
    a = normalize_seq(np.asarray(a0, dtype=np.float64))
    if a.size == 0:
        return a, float("inf")

    n = len(a)
    best = a.copy()
    best_sc = score_norm(a)
    z = z_from_a(a)
    z = z - float(np.mean(z))

    eta = 0.5
    adaptive_temp = 1.0
    no_accept = 0
    f_accepted = None

    for step in range(max_steps):
        now = time.monotonic()
        if now >= deadline:
            break

        frac = min(1.0, float(step) / float(max(1, max_steps)))
        base_temp = temp_start * (temp_end / max(1e-15, temp_start)) ** frac
        temp = min(0.30, max(1e-5, base_temp * adaptive_temp))

        b = fast_conv(a)
        M = float(np.max(b))
        if not np.isfinite(M) or M <= 0.0:
            break
        f0 = float(2.0 * n * M)  # a is normalized

        w = make_weights(b, M, temp)
        ga = grad_from_weights(a, w)
        mean_g = float(np.dot(a, ga))
        grad_z = n * (a * (ga - mean_g))
        grad_z = grad_z - float(np.mean(grad_z))
        nrm = float(np.linalg.norm(grad_z))

        if not np.isfinite(nrm) or nrm < 1e-30:
            break

        d = -grad_z / nrm
        accepted = False
        f_accepted = None

        for tr in range(8):
            alpha = eta * (0.5 ** tr)
            if alpha <= 1e-9:
                break
            zt = z + alpha * d
            at = softmax_from_z(zt)
            bt = fast_conv(at)
            Mt = float(np.max(bt))
            ft = float(2.0 * n * Mt)

            if np.isfinite(ft) and ft < f0 - 1e-15:
                z = zt
                a = at
                f_accepted = ft
                accepted = True

                if tr == 0:
                    eta = min(2.0, eta * 1.25)
                elif tr <= 2:
                    eta = min(2.0, eta * 1.05)
                else:
                    eta = max(0.05, eta * 0.7)

                adaptive_temp = max(1.0, adaptive_temp * 0.97)
                no_accept = 0
                break

        if not accepted:
            no_accept += 1
            if no_accept >= 10:
                adaptive_temp = min(5.0, adaptive_temp * 1.4)
                no_accept = 0
            if no_accept >= 20:
                eta = max(0.02, eta * 0.5)
                no_accept = 0
            if eta < 0.02:
                break

        z = z - float(np.mean(z))
        if not np.all(np.isfinite(z)):
            z = z_from_a(a)

        cur_sc = f_accepted if accepted else f0
        if np.isfinite(cur_sc) and cur_sc < best_sc:
            best_sc = float(cur_sc)
            best = a.copy()

        if (step + 1) % 5000 == 0 and time.monotonic() - now > 0:
            print(f"  {prefix} step={step+1} best={best_sc:.10f}")

    final_sc = score_norm(a)
    if np.isfinite(final_sc) and final_sc < best_sc:
        best = a.copy()
        best_sc = float(final_sc)

    return best, float(best_sc)


def lse_obj(z, T, n):
    a = softmax_from_z(z)
    b = fast_conv(a)
    M = float(np.max(b))
    x = T * (b - M)
    x = np.clip(x, -700.0, 700.0)
    logsum = float(np.log(float(np.sum(np.exp(x)))))
    val = M + logsum / T
    return float(n * val)


def lse_jac(z, T, n):
    a = softmax_from_z(z)
    b = fast_conv(a)
    M = float(np.max(b))
    x = T * (b - M)
    x = np.clip(x, -700.0, 700.0)
    e = np.exp(x)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.zeros_like(z)
    w = e / s
    ga = grad_from_weights(a, w)
    mean_g = float(np.dot(a, ga))
    return n * a * (ga - mean_g)


def lse_stage(a0, T, maxiter=50):
    a = normalize_seq(np.asarray(a0, dtype=np.float64))
    z = z_from_a(a)
    n = len(a)
    try:
        bounds = [(-50.0, 50.0)] * n
        res = minimize(
            fun=lse_obj,
            x0=z,
            jac=lse_jac,
            args=(float(T), n),
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-16,
                "gtol": 1e-10,
                "maxls": 12,
                "disp": False,
            },
        )
        if hasattr(res, "x") and res.x is not None:
            z = np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    out = softmax_from_z(z)
    return normalize_seq(out)


def build_conv_matrix(y):
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    L = 2 * n - 1
    T = np.empty((L, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = y
    return T


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    if cp is None:
        _lp_solver = None
        return None

    candidates = []
    for name in ["PDLP", "ECOS", "SCS", "CLARABEL", "OSQP", "GLOP", "GLPK", "CBC", "GUROBI", "MOSEK", "XPRESS"]:
        s = getattr(cp, name, None)
        if s is not None and s not in candidates:
            candidates.append(s)

    for s in candidates:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _lp_solver = s
                return s
        except Exception:
            continue

    _lp_solver = None
    return None


def solve_lp_best_response(y, solver=None):
    if cp is None:
        return None, None
    if solver is None:
        solver = get_lp_solver()
    if solver is None:
        return None, None

    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(y)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [solver]
    for s in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL, cp.GLOP, cp.GLPK, cp.CBC]:
        if s is not None and s not in solvers:
            solvers.append(s)

    ok = False
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
        except Exception:
            continue
        if prob.status in ("optimal", "optimal_inaccurate"):
            ok = True
            break

    if not ok:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = float(b_val.sum())
    if s < 1e-15 or not np.isfinite(s):
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    if not np.isfinite(M_val):
        M_val = float(fast_conv(b_val).max())
    return b_val, M_val


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, prefix="NLP"):
    a0 = normalize_seq(np.asarray(a0, dtype=np.float64))
    if cp is None:
        return a0, score_norm(a0)
    if solver is None:
        solver = get_lp_solver()
    if solver is None:
        return a0, score_norm(a0)

    a = a0.copy()
    y = a.copy()
    best = a.copy()
    best_sc = score_norm(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        tk = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * float(it) * float(it)))
        alpha = 2.0 / (tk + 1.0)
        momentum = (tk - 1.0) / tk

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = normalize_seq(a_new)

        y_new = a_new + momentum * (a_new - a_old)
        y_new = normalize_seq(y_new)

        a = a_new
        y = y_new

        sc_a = score_norm(a)
        if np.isfinite(sc_a) and sc_a < best_sc:
            best_sc = float(sc_a)
            best = a.copy()

        sc_y = score_norm(y)
        if np.isfinite(sc_y) and sc_y < best_sc:
            best_sc = float(sc_y)
            best = y.copy()

        if (it + 1) % 100 == 0:
            el = time.monotonic() - t0
            if el - last_print > 8.0:
                print(f"  {prefix} iter={it+1} t={el:.0f}s best={best_sc:.10f} M_lp={M_lp:.8f}")
                last_print = el

    return best, float(best_sc)


def has_budget(deadline, seconds):
    return time.monotonic() + seconds < deadline

    if has_budget(deadline, 90):
        a_cur = interpolate_sequence(best_a, n_main)
        end = min(deadline, time.monotonic() + 200.0)
        a_act, sc_act = active_subgradient_polish(
            a_cur, end,
            temp_start=0.03, temp_end=0.002,
            max_steps=400000, prefix="MAIN-ACT2"
        )
        if np.isfinite(sc_act) and sc_act < best_sc:
            best_a = a_act
            best_sc = float(sc_act)
            print(f"  main active 2 improved: {best_sc:.10f}")

    # A small LogSumExp L-BFGS polish on the main length.
    if has_budget(deadline, 20) and len(best_a) <= 2500:
        a_lse = lse_stage(best_a, T=5000.0, maxiter=25)
        sc = score_norm(a_lse)
        if np.isfinite(sc) and sc < best_sc:
            best_a = a_lse
            best_sc = float(sc)
            print(f"  LSE5k improved: {best_sc:.10f}")

    if has_budget(deadline, 20) and len(best_a) <= 2500:
        a_lse = lse_stage(best_a, T=20000.0, maxiter=25)
        sc = score_norm(a_lse)
        if np.isfinite(sc) and sc < best_sc:
            best_a = a_lse
            best_sc = float(sc)
            print(f"  LSE20k improved: {best_sc:.10f}")

    # Try larger lengths from the best shape.
    large_set = {len(best_a) + 200, len(best_a) + 400, len(best_a) + 600,
                 2400, 2700, 3000, 3200}
    large_lengths = sorted([x for x in large_set if 300 <= x <= 3200 and x > len(best_a) - 1])

    for nt in large_lengths:
        if not has_budget(deadline, 80):
            break
        a_t = interpolate_sequence(best_a, int(nt))
        sc = score_norm(a_t)
        if np.isfinite(sc) and sc < best_sc:
            best_a = a_t
            best_sc = float(sc)
            print(f"  large interp n={nt}: {best_sc:.10f}")

        if has_budget(deadline, 60):
            end = min(deadline, time.monotonic() + 70.0)
            a_t, sc_t = active_subgradient_polish(
                a_t, end,
                temp_start=0.04, temp_end=0.002,
                max_steps=200000, prefix=f"LEN{nt}"
            )
            if np.isfinite(sc_t) and sc_t < best_sc:
                best_a = a_t
                best_sc = float(sc_t)
                print(f"  large active n={nt}: {best_sc:.10f}")

        if has_budget(deadline, 25) and nt <= 3000:
            a_lse = lse_stage(a_t, T=10000.0, maxiter=15)
            sc = score_norm(a_lse)
            if np.isfinite(sc) and sc < best_sc:
                best_a = a_lse
                best_sc = float(sc)
                print(f"  large LSE n={nt}: {best_sc:.10f}")

    # Final very short active polish on the current best.
    if has_budget(deadline, 35):
        end = min(deadline, time.monotonic() + 45.0)
        a_fin, sc_fin = active_subgradient_polish(
            best_a, end,
            temp_start=0.02, temp_end=0.001,
            max_steps=100000, prefix="FINAL"
        )
        if np.isfinite(sc_fin) and sc_fin < best_sc:
            best_a = a_fin
            best_sc = float(sc_fin)
            print(f"  final active improved: {best_sc:.10f}")

    best_a = normalize_seq(best_a)

    # Final verification against the actual evaluator when possible.
    try:
        ver = evaluate_safe(best_a)
        if np.isfinite(ver) and ver < best_sc:
            best_sc = float(ver)
    except Exception:
        pass

    print(f"FINAL: n={len(best_a)}, score={best_sc:.10f}")
    return [float(x) for x in best_a]
