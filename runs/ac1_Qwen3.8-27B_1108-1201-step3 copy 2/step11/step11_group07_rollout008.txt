    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_objective_and_grad(a, T):
    """LSE-smoothed objective and gradient for normalized a (sum=1, a>=0).
    Returns (obj, grad_wrt_p) where grad is on the simplectangent space."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    
    # gradient of lse wrt a: grad_a[m] = sum_k w[k] * d(c[k])/d(a[m])
    # c[k] = sum_i a[i]*a[k-i], so d(c[k])/d(a[m]) = a[k-m] + (a[m-k] if m!=k-m else a[k-m])
    # Actually: d(c[k])/d(a[m]) = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n and m-k != k-m)
    # For auto-conv: d(c[k])/d(a[m]) = a[k-m] + a[m-k] if k-m != m-k, else 2*a[k-m]... 
    # Wait: c[k] = sum_i a[i]*a[k-i]
    # d(c[k])/d(a[m]) = a[k-m] (from the i=m term: d(a[m]*a[k-m])/d(a[m]) = a[k-m])
    #                  + a[m-k] (from the i=k-m term if k-m != m: d(a[k-m]*a[m])/d(a[m]) = a[k-m])
    # Wait, I'm confusing myself. Let me be careful.
    # c[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
    # d(c[k])/d(a[m]) = sum over terms where a[m] appears:
    #   If m is the first index: a[k-m] (valid if 0<=k-m<n)
    #   If m is the second index (i.e., k-i=m, so i=k-m): a[k-m] (valid if 0<=k-m<n and k-m != m)
    #   If m is both (i=m and k-i=m, i.e., k=2m): only one term a[m]*a[m], derivative = 2*a[m]
    # So: d(c[k])/d(a[m]) = a[k-m] * 1_{0<=k-m<n} + a[m-k] * 1_{0<=m-k<n, m != k-m}
    # Since a is the same in both positions: = a[|k-m|] * (1_{0<=k-m<n} + 1_{0<=m-k<n} - 1_{k=2m})
    # Hmm, this is getting complicated. Let me use a different approach.
    
    # Alternative: compute gradient via correlation
    # grad_a[m] = 2 * sum_k w[k] * a[k-m] - correction for the diagonal
    # Actually, since c[k] = <a, shift_k(a)>, we have:
    # d(c[k])/d(a) = shift_{-k}(a) + shift_k(a) (contributions from both a's)
    # But for auto-convolution specifically:
    # c[k] = sum_i a[i]*a[k-i]
    # d(c[k])/d(a[m]) = a[k-m] + a[m-k] (when both are valid and distinct)
    #                   = 2*a[k-m] (when k-m = m-k, i.e., k=2m)
    #                   = a[k-m] (when only one is valid)
    
    # Efficient computation: grad_a = 2 * (w * a) (cross-correlation) minus diagonal correction
    # Let's compute R = cross_correlation(w, a) where R[m] = sum_k w[k]*a[k-m]
    # Then grad_a[m] = R[m] + R2[m] where R2[m] = sum_k w[k]*a[m-k]
    # But R2[m] = sum_k w[k]*a[m-k] = sum_j w[m+j]*a[j] (substituting j=m-k)
    # = (w reversed) correlated with a... 
    
    # Actually, since w is the softmax of the convolution which IS symmetric (w[k] = w[2(n-1)-k] if a is such that conv is symmetric), 
    # but in general w is NOT symmetric.
    
    # Let me just compute it directly:
    # grad_a[m] = sum_k w[k] * (a[k-m] + a[m-k]) with appropriate bounds
    # = sum_k w[k] * a[k-m] + sum_k w[k] * a[m-k]
    # First term: (w * a) at lag m = cross-correlation
    # Second term: sum_k w[k]*a[m-k] = sum_j w[m+j]*a[j] = (w flipped) cross-correlated with a
    
    # For efficiency, use FFT:
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    
    # First term: sum_k w[k]*a[k-m] for valid k-m
    # This is the cross-correlation <w, shift_m(a)>
    # = inverse FFT of W * conj(A) ... hmm, let me think
    # corr[m] = sum_k w[k]*a[k-m]
    # In FFT: this is the circular cross-correlation, which is IFFT(conj(W) * A) or IFFT(W * conj(A)) depending on convention
    
    # Let me use: R1 = irfft(W * np.conj(A), N) => R1[m] = sum_k w[k+m]*a[k] = sum_j w[j]*a[j-m]
    # That's sum_j w[j]*a[j-m] which is what we want for the first term (with k->j)
    
    R1 = irfft(W * np.conj(A), N)[:n]
    # R1[m] = sum_j w[j]*a[j-m] (with a[j-m]=0 if j-m out of bounds, since we zero-padded)
    
    # Second term: sum_k w[k]*a[m-k]
    # Let j = m-k, so k = m-j: sum_j w[m-j]*a[j]
    # This is the cross-correlation of w and a with a different shift direction
    # = IFFT(conj(W) * A)[m] ... let me verify
    # IFFT(conj(W)*A)[m] = (1/N) sum_k conj(W[k])*A[k]*exp(2pi i k m/N)
    # = sum_n w[n]*a[n+m] (circular correlation)
    # Hmm, this gives sum_n w[n]*a[n+m] which is sum_k w[k]*a[k+m], not what we want.
    
    # Let me try: IFFT(W * conj(A))[m] = sum_n w[n-m]*a[n] (circular)
    # = sum_k w[k]*a[k+m] (substituting k = n-m)
    # That's not quite right either.
    
    # OK let me just use the direct formula:
    # grad_a[m] = sum_k w[k] * (a[k-m] + a[m-k])
    # First sum: sum_k w[k]*a[k-m] = (w cross-correlate a)[m]
    # Second sum: sum_k w[k]*a[m-k] = (a cross-correlate w)[m] = (w cross-correlate a)[m] by symmetry of cross-correlation!
    # Wait: (a cross-correlate w)[m] = sum_k a[k]*w[k+m] = sum_j w[j]*a[j-m] = (w cross-correlate a)[m]
    # YES! They're the same! Because cross-correlation satisfies (f g~)[m] = (g f~)[m]
    
    # So grad_a[m] = 2 * sum_k w[k]*a[k-m] = 2 * R1[m]
    # EXCEPT for the diagonal correction: when k = 2m, the term a[k-m] = a[m] is counted twice
    # in the sum "a[k-m] + a[m-k]" but in the actual derivative, c[k] has only one a[m]*a[m] term.
    # So we need to subtract w[2m]*a[m] for valid 2m.
    
    grad_a = 2.0 * R1
    # Diagonal correction
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    grad_a[valid] -= a[valid] * w[idx[valid]]
    
    return obj, grad_a

def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-parameterization (softmax)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    obj, grad_a = lse_objective_and_grad(a, T)
    
    # Transform to z-space: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def subgradient_descent(a0, time_limit, n_steps=50000, eta0=0.05):
    """Subgradient method using the true (not smoothed) objective.
    Uses the active set (peaks within epsilon of max) for the subgradient."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Active set: lags within 1% of max
        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        
        # Subgradient: average of gradients at active peaks
        # grad[m] = (1/|S|) sum_{k in S} (a[k-m] + a[m-k]) with bounds
        grad = np.zeros(n)
        for k in active:
            # a[k-m] for m = 0..n-1, valid when 0 <= k-m < n
            # a[m-k] for m = 0..n-1, valid when 0 <= m-k < n
            m_vals = np.arange(n)
            km = k - m_vals  # k - m
            mk = m_vals - k  # m - k
            valid_km = (km >= 0) & (km < n)
            valid_mk = (mk >= 0) & (mk < n)
            grad[valid_km] += a[km[valid_km]]
            valid_mk2 = valid_mk & ~np.eye(n, dtype=bool)  # avoid double counting
            # Actually, let's just do it simply:
            # The gradient of c[k] w.r.t. a[m] is:
            # a[k-m] if 0<=k-m<n, plus a[m-k] if 0<=m-k<n and m-k != k-m
            # = a[|k-m|] * (indicator(0<=k-m<n) + indicator(0<=m-k<n)) - a[k-m]*indicator(k==2m)
            # For simplicity, use the R1 formula: grad_at_k[m] = 2*a[k-m] (if valid) - a[m]*delta(k==2m)
            pass
        
        # Use FFT-based computation for speed
        N = 1 << (2*n).bit_length()
        A = rfft(a, N)
        # For each active k, the gradient contribution is 2*shift_k(a) - diagonal
        # Sum over active k: sum_k active * 2 * a[k-m]
        # = 2 * sum_k (active_indicator[k] * a[k-m])
        # = 2 * (active_indicator * a) cross-correlation
        W_act = rfft(np.zeros(2*n-1), N)
        # Actually, let's just do it with the active set as weights
        w_act = np.zeros(2*n-1)
        w_act[active] = 1.0 / len(active)
        W_act = rfft(w_act, N)
        R = irfft(W_act * np.conj(A), N)[:n]
        grad = 2.0 * R
        # Diagonal correction
        for k in active:
            m = k // 2
            if k % 2 == 0 and m < n:
                grad[m] -= a[m] / len(active)
        
        # Multiplicative update
        for scale in [1.0, 0.5, 0.25, 0.1]:
            log_alpha = -eta * scale * grad
            log_alpha -= log_alpha.max()
            a_new = a * np.exp(log_alpha)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                no_improve = 0
                break
        else:
            no_improve += 1
        
        if no_improve >= 30:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-8:
                break
    
    return best_a, best_score

def lbgfs_multi_stage(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with multi-stage temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        stage_time = min(remaining, time_limit / len(T_stages) * 1.5)
        
        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, stage_T)
            return obj
        
        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g
        
        iters = min(maxiter_per_stage, max(10, int(stage_time * 50)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"  L-BFGS error at T={stage_T}: {e}")
    
    return softmax_from_z(z)

def active_set_optimization(a0, time_limit, n_rounds=10000):
    """Optimize by directly minimizing the max of the convolution.
    At each step, identify the active peaks and solve a small QP or use
    a projected gradient step that specifically reduces those peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    lr = 0.01
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Active set: top peaks
        k_active = max(1, min(n, int(n * 0.01)))  # top 1% of peaks
        top_k = np.argsort(c)[-k_active:]
        
        # Compute subgradient from active set
        # Use FFT for efficiency
        w_act = np.zeros(2*n-1)
        w_act[top_k] = 1.0 / len(top_k)
        N = 1 << (2*n).bit_length()
        W = rfft(w_act, N)
        A = rfft(a, N)
        R = irfft(W * np.conj(A), N)[:n]
        grad = 2.0 * R
        for k in top_k:
            if k % 2 == 0:
                m = k // 2
                if m < n:
                    grad[m] -= a[m] / len(top_k)
        
        # Projected gradient step on the simplex
        # a_new = normalize(max(0, a - lr * grad))
        for scale in [1.0, 0.5, 0.25, 0.1]:
            a_new = a - lr * scale * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                break
        
        # Adaptive learning rate
        if rnd % 100 == 0 and rnd > 0:
            # Check if we're making progress
            c_check = fast_convolve(a)
            if c_check.max() > c_max * 0.9999:  # barely improving
                lr *= 0.8
    
    return best_a, best_score

def multiplicative_weights_refined(a0, time_limit, n_steps=30000, eta0=0.1):
    """Refined multiplicative weights with better step size."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Use all peaks with weight proportional to (c[k] - mean)^+ 
        # Or use top-K peaks
        k_top = min(n, max(10, n // 10))
        top_k = np.argpartition(c, -k_top)[-k_top:]
        
        # Weighted subgradient
        w_act = np.zeros(2*n-1)
        w_act[top_k] = c[top_k]  # weight by value
        w_act /= w_act.sum()
        
        N = 1 << (2*n).bit_length()
        W = rfft(w_act, N)
        A = rfft(a, N)
        R = irfft(W * np.conj(A), N)[:n]
        grad = 2.0 * R
        for k in top_k:
            if k % 2 == 0:
                m = k // 2
                if m < n:
                    grad[m] -= a[m] * w_act[k]
        
        # Multiplicative update
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            log_alpha = -eta * scale * grad
            log_alpha -= log_alpha.max()
            a_new = a * np.exp(log_alpha)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                no_improve = 0
                break
        else:
            no_improve += 1
        
        if no_improve >= 20:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-8:
                break
    
    return best_a, best_score
