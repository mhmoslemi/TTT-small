import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Scale-invariant score computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix_rows(a, K):
    """Build sparse matrix rows for specific convolution indices K.
    Returns (row_ind, col_ind, data, shape) for rows in K."""
    n = len(a)
    size = 2 * n - 1
    row_ind = []
    col_ind = []
    data = []
    for k in K:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row_ind.append(k)
            col_ind.append(j)
            data.append(a[j])
    row_ind = np.array(row_ind)
    col_ind = np.array(col_ind)
    data = np.array(data)
    T = csr_matrix((data, (row_ind, col_ind)), shape=(size, n))
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_cg(a_fixed, K_init, solver, max_new_constraints=50, tol=1e-8):
    """Solve LP with constraint generation.
    Minimize t s.t. rows of T(b) <= t for k in active_set, sum(b)=1, b>=0
    where T(k) is row k of the convolution matrix.
    
    Returns (b_opt, t_opt) or (None, None).
    """
    n = len(a_fixed)
    K = list(K_init)
    b_val = np.ones(n) / n
    
    for cg_iter in range(max_new_constraints + 1):
        # Build LP with current constraint set
        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        
        constraints = [cp.sum(b_var) == 1]
        for k in K:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            row_vec = a_fixed[js:je+1]
            # (row_vec) @ b_var[js:je+1] <= t_var
            constraints.append(cp.sum(cp.multiply(row_vec, b_var[js:je+1])) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            return None, None
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        t_val = float(t_var.value)
        
        # Check for violated constraints
        conv_val = fast_convolve(a_fixed) @ b_val
        violated = np.where(conv_val > t_val + tol)[0]
        
        if len(violated) == 0:
            return b_val, t_val
        
        # Add the most violated constraints (up to 10 per round)
        violated_sorted = violated[np.argsort(conv_val[violated])[::-1]]
        new_K = list(violated_sorted[:10])
        K = list(set(K) | set(new_K))
        # Cap the number of constraints
        if len(K) > 500:
            # Keep the most important ones (highest conv values)
            conv_at_K = conv_val[np.array(K)]
            K = list(np.array(K)[np.argsort(conv_at_K)[::-1][:500]])
    
    return b_val, t_val

    beta = 1.0  # Nesterov momentum

            M_b = 1.0 / n
        
        # Nesterov update for y
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        alpha = (1.0 - beta) / beta_new

        sy = y.sum()
        if sy < 1e-15:

def logsumexp_grad(a, T):
    """Compute gradient of (2n/T)*LogSumExp(conv(a,a)/T) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    w = exp_b / exp_b.sum()
    
    # grad_a[i] = (2n/T) * 2 * sum_j w[i+j] * a[j] for valid j
    # Using cross-correlation
    grad_a = np.zeros(n)
    for i in range(n):
        valid_len = min(n, 2 * n - 1 - i)
        if valid_len > 0:
            grad_a[i] = 2.0 * np.dot(w[i:i + valid_len], a[:valid_len])
    grad_a *= (2.0 * n) / T
    return grad_a

    iters_per_stage = max(30, maxiter // n_stages)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        def func(z):
            if time.monotonic() - t0 > time_limit - 1:
                return 1e15
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            b = fast_convolve(a_trial)
            max_b = b.max()
            exp_b = np.exp(b / T - max_b / T)
            lse = max_b / T + np.log(exp_b.sum())
            obj = (2.0 * n / T) * lse
            sc = 2.0 * n * max_b
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
            return obj
        
        def jac(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z)
            a_trial = a_trial / st
            return logsumexp_grad(a_trial, T)
        
        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'maxfun': iters_per_stage * 100,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception:
            pass
    
    return best_a, best_score


def coord_descent_line_search(a0, time_limit, n_steps=100000):
    """Coordinate descent with line search on top-contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find top peaks
        threshold = max_b * 0.9995
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])
        
        improved = False
        for peak_k in peaks[:5]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(10, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                # Try multiple reduction factors via line search
                for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    # Redistribute diff proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    sc = eval_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Multi-coordinate reduction
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:6]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.997
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = eval_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
        else:
            stall = 0
        
        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def mirror_descent(a0, time_limit, eta0=0.03):
    """Mirror descent with KL divergence on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    no_improve = 0
    eta = eta0
    m = 0  # running average count
    
    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1.0 / (it + 10))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 50:
            active = active[np.argsort(b[active])[::-1][:50]]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Mirror descent with step decay
        eta_t = eta0 / (1.0 + 0.01 * it) ** 0.5
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta_t * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        a = a_new
        
        # Running average
        m += 1
        a_avg = ((m - 1) * a_avg + a) / m if m > 1 else a.copy()
        
        sc = eval_score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 300:
            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.standard_normal(n) * 0.005
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        a = np.exp(-1.5 * (x - 0.5) ** 2 / 0.15) * (1 + 0.1 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses constraint-generation LP, mirror descent, L-BFGS, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: Constraint-Generation LP Alternating ===
    print(f"\n=== Phase 1: CG-LP Nesterov (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_cg(best_seq, t_alloc, solver=solver,
                                  n_max_iters=500000, K_init_size=100)
        try_update(a_lp, "CG_LP_p1")
    
    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent(best_seq, t_alloc, eta0=0.03)
        try_update(a_md, "MD")
    
    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        candidate_ns = sorted(set([
            max(600, n_best - 200),
            max(800, n_best + 100),
            max(600, n_best + 300),
            max(1000, int(n_best * 1.5)),
        ]))
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n: {candidate_ns}, ~{per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 15:
                break
            if n_try == n_best:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_sub, _ = nesterov_lp_cg(a_interp, t_sub, solver=solver,
                                       n_max_iters=200000, K_init_size=80)
            try_update(a_sub, f"multi_n_{n_try}")
    
    # === Phase 4: Deep CG-LP Refinement ===
    print(f"\n=== Phase 4: Deep CG-LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_deep, _ = nesterov_lp_cg(best_seq, t_alloc, solver=solver,
                                    n_max_iters=500000, K_init_size=150)
        try_update(a_deep, "deep_cg")
    
    # === Phase 5: LogSumExp L-BFGS ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        T_stages = [12.0, 6.0, 3.5, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages, maxiter=3000)
        try_update(a_lb, "LSE_LBFGS")
    
    # === Phase 6: Coordinate Descent Polish ===
    print(f"\n=== Phase 6: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_cd, _ = coord_descent_line_search(best_seq, t_alloc, n_steps=200000)
        try_update(a_cd, "coord_desc")
    
    # === Phase 7: Final Quick CG-LP ===
    print(f"\n=== Phase 7: Final CG-LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 80.0)
    if t_alloc > 5:
        a_fin, _ = nesterov_lp_cg(best_seq, t_alloc, solver=solver,
                                   n_max_iters=200000, K_init_size=120)
        try_update(a_fin, "final")
    
    # Final verification
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
