import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_restarts(a0, time_limit, solver=None, max_iters=10000000,
                         restart_interval=800, n_restarts_max=8):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    nesterov_iter = 0
    restart_count = 0

    while True:
        if time.monotonic() - t0 > time_limit:
            break

        # Periodic restart: reset Nesterov state but keep best
        if restart_count < n_restarts_max and nesterov_iter % restart_interval == 0 and nesterov_iter > 0:
            a = best_a.copy()
            y = best_a.copy()
            nesterov_iter = 0
            restart_count += 1
            continue

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            nesterov_iter += 1
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * nesterov_iter * nesterov_iter)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new
        nesterov_iter += 1

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (nesterov_iter + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"      NesterovLP iter={nesterov_iter} restarting={restart_count} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.8f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 50

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0:
                print(f"      LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)


def peak_directed_subgrad(a0, time_limit, n_steps=500000, T_start=5.0, T_end=0.1):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        T_cur = T_start * (T_end / T_start) ** (step / max(1, n_steps))
        Tb = T_cur * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[k] * a[k - idx]

        S = np.dot(g, a)
        grad_z = a * (g - S)

        eta = 0.3 * (0.01 / 0.3) ** (step / max(1, n_steps))
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            z = z_new
        a = softmax_from_z(z)

        if (step + 1) % 50000 == 0:
            print(f"      PeakSG step={step+1}: best_score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def multiplicative_weights_step(a, y, eta):
    b = fast_convolve(y)
    max_b = b.max()
    if max_b < 1e-30:
        return a.copy()
    cost = np.zeros(len(a))
    for k in range(len(b)):
        if b[k] < max_b * 0.5:
            continue
        js = max(0, k - len(a) + 1)
        je = min(len(a) - 1, k)
        idx = np.arange(js, je + 1)
        cost[idx] += b[k] / max_b * y[k - idx]
    new_a = a * np.exp(-eta * cost)
    s = new_a.sum()
    if s < 1e-30:
        return a.copy()
    return new_a / s

        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 5000 == 0:
            print(f"      MWU iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')
    n_base = 1200

    # Get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.10f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.10f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Main Nesterov LP with restarts (50% of budget) ===
    print(f"\n=== Phase 1: Nesterov LP restarts at n={n_base} (score={best_score:.10f}) ===")
    t1 = 0.50 * (budget_s - 20)
    a_out, _ = nesterov_lp_restarts(best_seq, t1, solver=solver,
                                     restart_interval=600, n_restarts_max=6)
    try_update(a_out, "nesterov_lp_p1")
    print(f"  Phase 1 done: best_score={best_score:.10f}")

    # === Phase 2: LSE L-BFGS polishing (15% of budget) ===
    print(f"\n=== Phase 2: LSE L-BFGS polish (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(0.15 * (budget_s - 20), remaining * 0.4)
    if t2 > 10:
        a_lse = lbgfs_logsumexp(best_seq, t2, T_start=50.0, T_end=0.02, n_stages=25)
        try_update(a_lse, "lse_polish_p2")
    print(f"  Phase 2 done: best_score={best_score:.10f}")

    # === Phase 3: Second Nesterov LP round (20% of budget) ===
    print(f"\n=== Phase 3: Nesterov LP round 2 (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(0.20 * (budget_s - 20), remaining * 0.4)
    if t3 > 10:
        a_out2, _ = nesterov_lp_restarts(best_seq, t3, solver=solver,
                                          restart_interval=500, n_restarts_max=5)
        try_update(a_out2, "nesterov_lp_p3")
    print(f"  Phase 3 done: best_score={best_score:.10f}")

    # === Phase 4: Peak-directed subgradient (10% of budget) ===
    print(f"\n=== Phase 4: Peak-directed subgradient (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 20), remaining * 0.4)
    if t4 > 5:
        a_sg, _ = peak_directed_subgrad(best_seq, t4, n_steps=300000, T_start=3.0, T_end=0.05)
        try_update(a_sg, "peak_sg_p4")
    print(f"  Phase 4 done: best_score={best_score:.10f}")

    # === Phase 5: MWU accelerated (5% of budget) ===
    print(f"\n=== Phase 5: MWU accelerated (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(0.05 * (budget_s - 20), remaining * 0.3)
    if t5 > 5:
        a_mwu, _ = mwu_accelerated(best_seq, t5, eta_start=0.5, eta_end=0.02, max_iters=2000000)
        try_update(a_mwu, "mwu_p5")
    print(f"  Phase 5 done: best_score={best_score:.10f}")

    # === Phase 6: Length exploration (5% of budget) ===
    print(f"\n=== Phase 6: Length exploration (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6_total = min(0.05 * (budget_s - 20), remaining * 0.5)
    if t6_total > 15:
        n_candidates = [n_base + 50, n_base - 50, n_base + 100, n_base + 200]
        t6_each = t6_total / len(n_candidates)
        for n_try in n_candidates:
            remaining = _global_deadline - time.monotonic()
            if remaining < 10 or t6_each < 5:
                break
            t6 = min(t6_each, remaining * 0.4)
            if t6 < 5 or n_try < 100:
                continue
            print(f"    Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_restarts(init_try, t6, solver=solver,
                                             restart_interval=400, n_restarts_max=3)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 6 done: best_score={best_score:.10f}")

    # === Phase 7: Final LSE fine-tune ===
    print(f"\n=== Phase 7: Final LSE (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 8:
        t7 = min(remaining * 0.6, 40.0)
        if t7 > 5:
            a_final = lbgfs_logsumexp(best_seq, t7, T_start=10.0, T_end=0.01, n_stages=15)
            try_update(a_final, "final_lse_p7")
    print(f"  Phase 7 done: best_score={best_score:.10f}")

    # === Phase 8: Very short Nesterov LP final push ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 10:
        t8 = min(remaining * 0.7, 30.0)
        if t8 > 5:
            a_push, _ = nesterov_lp_restarts(best_seq, t8, solver=solver,
                                              restart_interval=300, n_restarts_max=2)
            try_update(a_push, "final_push_p8")
            print(f"  Phase 8 done: best_score={best_score:.10f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
