import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS, cp.GLPK, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def nesterov_subgradient(a0, time_limit, verbose=True):
    """Nesterov accelerated subgradient on the simplex for min max_k (a*a)[k]."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Nesterov parameters
    x_k = a.copy()
    x_prev = a.copy()
    y_k = a.copy()

    # Adaptive step size
    eta = 1.0 / np.sqrt(n)
    eta_min = 1e-6
    eta_decay = 0.9999

    # Track for Polyak-like step
    running_best = best_score
    improvement_count = 0

    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        if (it + 1) % 100 == 0:
            # Nesterov momentum
            momentum = (it) / (it + 3.0)
            y_k = x_k + momentum * (x_k - x_prev)
            y_k = np.maximum(y_k, 0)
            sy = y_k.sum()
            if sy > 1e-15:
                y_k = y_k / sy
            else:
                y_k = x_k.copy()

        # Compute convolution and find active set
        conv = fast_convolve(y_k)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = y_k.copy()
            improvement_count += 1

        if (it + 1) % 5000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSubGrad iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f} eta={eta:.6f}")

        # Adaptive active set: include lags close to max
        if improvement_count > 200 or cur_score < 1.55:
            threshold_frac = 1.0 - 0.001
        elif cur_score < 1.6:
            threshold_frac = 1.0 - 0.005
        else:
            threshold_frac = 1.0 - 0.01

        threshold = max_val * threshold_frac
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient: g[i] = 2 * sum_{k in active} w_k * a[k-i]
        # where w_k = 1/|active|
        g = np.zeros(n)
        w_each = 1.0 / len(active)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            # g[i] += w_each * a[k-i]
            # k-i ranges from k-i_hi to k-i_lo
            g[i_lo:i_hi + 1] += w_each * a[k - i_lo:k - i_hi + 1][::-1]
        g *= 2.0

        # Step size (Nesterov O(1/t) for non-smooth)
        eta = eta_min + (1.0 / np.sqrt(n + it * 0.01)) * 0.5
        # Actually use a more practical adaptive step
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            # Polyak-like: step toward best
            eta_adaptive = max(eta_min, 1.0 / (np.sqrt(n) * (1.0 + it * 1e-5)))
            eta = eta_adaptive
        else:
            eta = eta_decay * eta

        # Update
        x_new = y_k - eta * g
        x_new = np.maximum(x_new, 0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y_k.copy()
        else:
            x_new = x_new / s_new

        x_prev = x_k.copy()
        x_k = x_new

    return best_a, best_score


def multiplicative_weights(a0, time_limit, verbose=True):
    """Mirror descent / multiplicative weights on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 1.0 / np.sqrt(n)
    no_improve = 0

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 200:
            eta *= 0.9
            no_improve = 0
            if eta < 1e-8:
                break

        if (it + 1) % 10000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")

        # Active set
        threshold = max_val * (1.0 - 0.005)
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Subgradient
        g = np.zeros(n)
        w_each = 1.0 / len(active)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            g[i_lo:i_hi + 1] += w_each * a[k - i_lo:k - i_hi + 1][::-1]
        g *= 2.0

        # Multiplicative update
        update = np.exp(-eta * g)
        a_new = a * update
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP alternating (forbidden directions) with Nesterov dual averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    y = a.copy()  # Nesterov dual variable
    lp_times = []

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP: min M s.t. T(y) @ b <= M, b >= 0, sum(b) = 1
        lp_t0 = time.monotonic()
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS]
        for s_try in solvers_to_try:
            if s_try is None:
                continue
            try:
                prob.solve(solver=s_try)
                if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue

        if not solved or b_var.value is None:
            break

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
        b_val[np.isnan(b_val)] = 0
        sb = b_val.sum()
        if sb < 1e-15:
            break
        b_val = b_val / sb
        lp_times.append(time.monotonic() - lp_t0)

        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_val) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 20 == 0 and verbose:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-10:]) * 1000 if lp_times else 0
            print(f"    LPAlt iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={avg_lp:.0f}ms")

        # Stop if LP too slow
        if len(lp_times) > 30:
            avg = np.mean(lp_times[-10:])
            if avg > 5.0:
                print(f"    LP too slow ({avg:.2f}s/iter), stopping")
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x)**2
    elif choice == 1:
        a = np.sin(np.pi * x)**2
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.1 * np.cos(2 * np.pi * 11 * x + 0.7)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-4*(x-0.3)**2) + 0.6*np.exp(-4*(x-0.7)**2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            best_seq = init_2000.copy()
            best_score = sc_2000

        # Quick Nesterov at n=2000
        t_ns2 = min(t_n2000 * 0.4, 200.0)
        if t_ns2 > 20:
            a_ns2, _ = nesterov_subgradient(init_2000, t_ns2, verbose=True)
            try_update(a_ns2, "nesterov_n2000")

        # LP at n=2000
        rem = deadline - time.monotonic()
        if rem > 60:
            t_lp2 = min(rem * 0.6, 300.0)
            a_lp2, _ = lp_alternating(best_seq, t_lp2, solver, verbose=True)
            try_update(a_lp2, "lp_n2000")

    # === Phase 5: Final polish at best n ===
    print(f"\n=== Phase 5: Final Polish (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 100.0)
        a_fin, _ = nesterov_subgradient(best_seq, t_fin, verbose=False)
        try_update(a_fin, "final_nesterov")

    rem = deadline - time.monotonic()
    if rem > 20:
        a_mw2, _ = multiplicative_weights(best_seq, min(rem * 0.5, 50.0), verbose=False)
        try_update(a_mw2, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
