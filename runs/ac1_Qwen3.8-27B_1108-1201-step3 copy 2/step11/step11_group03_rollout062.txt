import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Project vector onto {x>=0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_T(a):
    """Build (2n-1) x n matrix T with T[k,j] = a[k-j] (0 if out of range)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        ks = max(0, j)
        ke = min(size, j + n)
        T[ks:ke, j] = a[:ke - ks]
    return T


def lp_best_response(a_fixed, time_limit_lp=10.0):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_T(a_fixed)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'time_limit': time_limit_lp, 'dual_feasibility_tolerance': 1e-9,
                              'primal_feasibility_tolerance': 1e-9})
    if result.status != 0:
        return None, None
    b_val = np.maximum(result.x[:n], 0)
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val = b_val / s_val
    M_val = max(0.0, float(result.x[n]))
    return b_val, M_val

            sy = 1.0

        score_y = 2.0 * n * fast_convolve(y).max()

def fft_subgradient(a, active_set):
    """Compute subgradient of max_k (a*a)[k] at a using FFT cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    # Build weight vector w (length 2n-1)
    w = np.zeros(size)
    for k in active_set:
        w[k] = 1.0
    w /= len(active_set)
    
    # Cross-correlation via FFT: (w *flip* a)[j] = sum_k w[k] * a[k-j]
    # = ifft(FFT(w) .* conj(FFT(a_pad)))[:n]
    Z = 1 << (size + n - 1 - 1).bit_length()
    W = fft(w, Z)
    A = fft(a, Z)
    corr = ifft(W * np.conj(A), Z).real[:n]
    return 2.0 * corr


def fft_smooth_gradient(a, T_temp):
    """Compute gradient of smooth max T*log(sum exp(b[k]/T)) via FFT."""
    n = len(a)
    size = 2 * n - 1
    b = fast_convolve(a)
    # Compute weights
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / T_temp)
    Z_val = exp_b.sum()
    if Z_val < 1e-300:
        w = np.zeros(size)
        w[np.argmax(b)] = 1.0
    else:
        w = exp_b / Z_val
    
    # Cross-correlation
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    W = fft(w, fft_size)
    A = fft(a, fft_size)
    corr = ifft(W * np.conj(A), fft_size).real[:n]
    grad = 2.0 * T_temp * corr
    return grad, b, float(T_temp * np.log(Z_val) + b.max())


def subgradient_refine(a0, time_limit, max_iters=200000):
    """FFT-based subgradient descent with adaptive step size and projection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.5 / np.sqrt(n)
    best_eta = eta
    no_improve = 0
    momentum = a.copy()
    mu = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: within 0.1% of max
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient via FFT
        g = fft_subgradient(a, active)
        
        # Momentum update
        momentum = mu * momentum + (1 - mu) * g
        a_new = a - eta * momentum
        a_new = project_simplex(a_new)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slowly decay eta
            eta *= 0.9995
        else:
            no_improve += 1
            if no_improve >= 100:
                eta *= 0.5
                no_improve = 0
                if eta < 1e-8:
                    break
    
    return best_a, best_score


def lbgfs_smooth_max(a0, time_limit, T_stages=None, maxiter_total=5000):
    """L-BFGS on smooth max with softmax parameterization and temperature annealing."""
    n = len(a0)
    if T_stages is None:
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Initialize z from a
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a_init = a_init / a_init.sum()
    z = np.log(a_init)
    z -= z.mean()  # center for numerical stability
    
    t0 = time.monotonic()
    deadline = t0 + time_limit
    total_iters = maxiter_total
    iters_per_stage = max(20, total_iters // len(T_stages))
    
    for T_cur in T_stages:
        if time.monotonic() > deadline - 1.0:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            break
        iters_this = min(iters_per_stage, max(10, int(remaining * 20)))
        
        res = _lbgfs_single_stage(z, n, T_cur, remaining, iters_this)
        if res is not None:
            z = res
    
    # Convert back
    a = np.exp(z - z.max())
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a_sum
    return a


def _lbgfs_single_stage(z0, n, T_temp, time_limit, maxiter):
    """Single stage of L-BFGS for smooth max."""
    t0 = time.monotonic()
    deadline = t0 + time_limit
    fft_size = 1
    size = 2 * n - 1
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    
    def smooth_obj(z_val):
        if time.monotonic() > deadline:
            return 1e15, np.zeros(n)
        a = np.exp(z_val - z_val.max())
        a_sum = a.sum()
        if a_sum < 1e-300:
            return 1e15, np.zeros(n)
        a = a / a_sum
        b = fast_convolve(a)
        b_shifted = b - b.max()
        exp_b = np.exp(b_shifted / T_temp)
        Z_val = exp_b.sum()
        if Z_val < 1e-300:
            return 1e15, np.zeros(n)
        obj = T_temp * np.log(Z_val) + b.max()
        
        # Gradient w.r.t. a
        w = exp_b / Z_val
        W = fft(w, fft_size)
        A = fft(a, fft_size)
        corr = ifft(W * np.conj(A), fft_size).real[:n]
        grad_a = 2.0 * T_temp * corr
        
        # Gradient w.r.t. z (softmax Jacobian)
        S_val = np.dot(grad_a, a)
        grad_z = a * (grad_a - S_val)
        return 2.0 * n * obj, 2.0 * n * grad_z
    
    def fun(z_val):
        val, _ = smooth_obj(z_val)
        return val
    
    def jac(z_val):
        _, g = smooth_obj(z_val)
        return g
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'eps': 1e-7, 'disp': False})
        return result.x
    except Exception:
        return z0


def active_set_refine(a0, time_limit, max_active=8, max_iters=5000):
    """Refine using only the top active constraints with weighted subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.1 / np.sqrt(n)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Tight active set
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:max_active]
        
        # Weighted subgradient (weight by proximity to max)
        weights = np.exp((b[active] - max_b) / (1e-6 * max_b + 1e-30))
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w_k, k in zip(weights, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += w_k * 2.0 * a[k - j]
        
        a_new = a - eta * g
        a_new = project_simplex(a_new)
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta *= 0.999
        else:
            no_improve += 1
            if no_improve >= 500:
                eta *= 0.5
                no_improve = 0
                if eta < 1e-10:
                    break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases: LP alternating -> subgradient -> L-BFGS smooth max -> active set -> multi-n.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2.0) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: LP Alternating (main workhorse) ===
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.40, 600.0)
    if t_lp > 60:
        print(f"\n=== Phase 1: LP Alternating n={n_best} (score={best_score:.10f}) ===")
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_alternate(best_seq, t_lp, max_iters=20000)
        update_best(a_lp, "lp_phase1")
    
    # === Phase 2: Subgradient refinement ===
    rem = deadline - time.monotonic()
    t_sg = min(rem * 0.15, 200.0)
    if t_sg > 20:
        print(f"\n=== Phase 2: Subgradient (score={best_score:.10f}) ===")
        print(f"  Budget: {t_sg:.0f}s")
        a_sg, _ = subgradient_refine(best_seq, t_sg, max_iters=500000)
        update_best(a_sg, "subgradient")
    
    # === Phase 3: L-BFGS smooth max ===
    rem = deadline - time.monotonic()
    t_lb = min(rem * 0.18, 280.0)
    if t_lb > 30:
        print(f"\n=== Phase 3: L-BFGS Smooth Max (score={best_score:.10f}) ===")
        print(f"  Budget: {t_lb:.0f}s")
        T_stages = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        a_lb = lbgfs_smooth_max(best_seq, t_lb, T_stages=T_stages, maxiter_total=3000)
        update_best(a_lb, "lbgfs_smooth")
    
    # === Phase 4: Active-set refinement ===
    rem = deadline - time.monotonic()
    t_as = min(rem * 0.10, 100.0)
    if t_as > 10:
        print(f"\n=== Phase 4: Active Set (score={best_score:.10f}) ===")
        a_as, _ = active_set_refine(best_seq, t_as, max_active=5, max_iters=50000)
        update_best(a_as, "active_set")
    
    # === Phase 5: Second LP pass (from improved point) ===
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.20, 300.0)
    if t_lp2 > 40:
        print(f"\n=== Phase 5: LP Round 2 (score={best_score:.10f}) ===")
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_alternate(best_seq, t_lp2, max_iters=10000)
        update_best(a_lp2, "lp_phase2")
    
    # === Phase 6: Multi-n exploration ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 6: Multi-n (score={best_score:.10f}) ===")
        for n_try in [n_best + 1, n_best - 1, n_best + 2, n_best - 2]:
            if n_try < 100 or n_try > 5000:
                continue
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_n = min(rem * 0.25, 100.0)
            if t_n < 15:
                continue
            print(f"  Trying n={n_try}, budget={t_n:.0f}s")
            init_n = interpolate_sequence(best_seq, n_try)
            # Quick LP
            a_n, _ = nesterov_lp_alternate(init_n, t_n * 0.6, max_iters=3000)
            update_best(a_n, f"lp_n{n_try}")
            # Quick L-BFGS
            rem = deadline - time.monotonic()
            if rem > 10:
                a_lb_n = lbgfs_smooth_max(best_seq, min(rem * 0.3, 30.0),
                                          T_stages=[0.1, 0.05, 0.01, 0.005],
                                          maxiter_total=500)
                # Re-interpolate result if needed
                if len(a_lb_n) == n_try:
                    update_best(a_lb_n, f"lbgfs_n{n_try}")
    
    # === Phase 7: Final polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
        # L-BFGS with very low temperature
        a_final = lbgfs_smooth_max(best_seq, min(rem * 0.5, 30.0),
                                   T_stages=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                   maxiter_total=1000)
        update_best(a_final, "final_lbgfs")
        rem = deadline - time.monotonic()
        if rem > 5:
            a_as2, _ = active_set_refine(best_seq, min(rem, 15.0), max_active=3, max_iters=20000)
            update_best(a_as2, "final_active")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Also try evaluate_sequence for ground truth
        try:
            test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
            gt_score = evaluate_sequence(test_seq)
            if np.isfinite(gt_score):
                print(f"\nGround truth score: {gt_score:.10f}")
            else:
                print(f"\nGround truth score: invalid!")
        except Exception as e:
            print(f"evaluate_sequence failed: {e}")
        print(f"FINAL LOCAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
