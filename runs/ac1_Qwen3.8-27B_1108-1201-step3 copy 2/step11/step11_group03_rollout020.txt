def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def get_lp_solver():
    """Find a working LP solver."""

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

def lse_objective(z, n, T_temp):
    """LogSumExp smoothed objective and gradient."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)

    exp_terms = np.exp((b - max_b) / T_temp)
    w = exp_terms / exp_terms.sum()
    lse = max_b + T_temp * np.log(w.sum())
    obj = 2.0 * n * lse
    
    # Gradient computation
    grad_a = np.zeros(n)
    for k in range(len(b)):

            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating with Nesterov acceleration (coarse optimization)
2. High-precision L-BFGS with progressive LogSumExp temperature annealing (fine polishing)
3. Multi-restart L-BFGS with perturbations to escape local minima
4. Multi-n exploration (1199, 1200, 1201) with cross-interpolation
5. Final coordinate descent polish
Key insight: The problem is non-convex (max of non-PSD quadratics), so we rely on
multiple local optimization methods and multi-restart to find the global optimum.
"""

import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast, mimics evaluate_sequence)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_limit, solver, max_iters=100000):
    """LP alternating with Nesterov acceleration and adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        y = y / sy if sy > 1e-15 else a.copy()

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        a = a / sa if sa > 1e-15 else y.copy()

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 4)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

        if (it + 1) % 5000 == 0:
            print(f"    LP iter={it+1}: score={best_score:.10f}")

    return best_a, best_score


def lse_objective_grad(z, n, T_temp):
    """Compute LogSumExp smoothed objective and gradient w.r.t. z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    z_sum = exp_z.sum()
    if z_sum < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / z_sum

    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    w = exp_terms / exp_terms.sum()
    lse = max_b + T_temp * np.log(w.sum())
    obj = 2.0 * n * lse

    # grad_a[i] = (2/T) * sum_k w[k] * a[k-i] for valid k-i
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += (2.0 / T_temp) * wk * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lbgfs_polish(a0, time_limit, T_stages, deadline_check=None):
    """L-BFGS polishing with progressive LogSumExp temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        def fun_lse(z_vec):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_objective_grad(z_vec, n, T)
            return obj

        def jac_lse(z_vec):
            if time.monotonic() > deadline:
                return np.zeros_like(z_vec)
            _, g = lse_objective_grad(z_vec, n, T)
            return g

        maxiter = max(50, int(remaining * 30))
        try:
            res = minimize(
                fun=fun_lse, x0=z, jac=jac_lse,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': maxiter * 1000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    return a
