def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_rows(a, indices):
    """Build constraint rows for given convolution indices."""
    n = len(a)
    K = len(indices)
    rows = np.zeros((K, n))
    for i, k in enumerate(indices):
        if k < n:
            rows[i, 0:k+1] = a[k:-1:-1]
        else:
            start = k - n + 1
            rows[i, start:] = a[n-1:start-1:-1]
    return rows

def solve_best_response(a, active_k, max_cp_iters=30):
    """Solve best-response LP for b given fixed a, using cutting plane."""
    n = len(a)
    # Start with given active constraints
    active_k = list(set(active_k))
    # Add the argmax of conv(a, uniform) as initial
    conv_a = fast_convolve(a)
    top_k = np.argsort(conv_a)[-min(10, len(conv_a)):]
    active_k = sorted(set(active_k + list(top_k)))
    
    M_val = np.inf
    b_opt = None
    
    for cp_iter in range(max_cp_iters):
        K = len(active_k)
        T_A = build_rows(a, active_k)
        
        # LP: min M s.t. T_A @ b <= M, sum(b) = 1, b >= 0
        # Variables: [b_0, ..., b_{n-1}, M]

        b_eq = np.array([1.0])

                      bounds=bounds, method='highs')
        
        if not res.success:

            b_opt = np.ones(n) / n
            M_val = fast_convolve(a).max()  # not quite right
            break

        if s < 1e-15:

            break
    
    return b_opt, M_val

def cross_convolve_fft(a, b):

    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]
