# --- Helper functions (all top-level) ---

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_best_response_lp(a_fixed, solver):
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def nesterov_lp(a0, time_limit, solver, max_iters=100000):
    n = len(a0)
    a = normalize(a0.copy())
    x_prev = a.copy()
    x_cur = a.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: find best response to y
        b_star = solve_best_response_lp(y, solver)
        if b_star is None:
            continue
        
        # Nesterov update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        gamma = t_k / t_next
        x_new = gamma * x_cur + (1.0 - gamma) * b_star
        x_new = normalize(x_new)
        y = x_new + gamma * (x_new - x_prev)
        y = normalize(y)
        x_prev = x_cur
        x_cur = x_new
        t_k = t_next
        
        # Check score
        sc = eval_score(x_cur)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = x_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.standard_normal(n) * 0.005
            x_cur = x_cur + noise
            x_cur = normalize(x_cur)
            x_prev = x_cur.copy()
            y = x_cur.copy()
            t_k = 1.0
            no_improve = 0
            best_score = eval_score(x_cur)
        
        if (it + 1) % 500 == 0:
            print(f"      NL iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lse_grad_x(x, T):
    """Gradient of 2n*logsumexp(conv(softmax(x), softmax(x))/T) wrt x."""
    x_s = x - x.max()
    exp_x = np.exp(x_s)
    a = exp_x / exp_x.sum()
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    # grad_a[i] = (4n/T) * sum_j w[j+i] * a[j]
    grad_a = np.zeros(n)
    for i in range(n):
        valid = min(n, 2 * n - 1 - i)
        if valid > 0:
            grad_a[i] = np.dot(w[i:i+valid], a[:valid])
    grad_a *= 4.0 * n / T
    # Chain rule
    grad_x = grad_a - np.dot(grad_a, a) * a
    return grad_x
