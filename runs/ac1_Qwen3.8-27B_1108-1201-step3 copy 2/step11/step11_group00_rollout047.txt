**Alternative: Direct optimization of the non-smooth objective using subgradient methods**

The objective f(a) = 2*n*max_m c[m] (with sum(a)=1) is convex and piecewise linear. Its subgradient at a is:
∂f(a) = 2*n * {sum_{m in argmax} λ_m * ∇c[m] : λ in simplex, λ_m > 0 only if m in argmax}

A natural subgradient is 2*n * ∇c[m*] where m* is any argmax.

1. Compute a subgradient g = 2*n * ∇c[m*]

3. a_new = proj_simplex(a_trial)

The projection onto the simplex is O(n log n) using sorting.

This is much cheaper than LP! Each iteration is O(n log n) for the projection + O(n log n) for the FFT-based gradient.

def softmax_fn(z):

def compute_lse_gradient(a, T):
    """Gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w.copy(), fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad

def lse_value(a, T):
    """Value of (1/T)*log(sum exp(T*(a*a)_k))."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    return (max_Tb + np.log(np.exp(Tb - max_Tb).sum())) / T

def approximate_best_response(y, n_inner=30, T_br=20.0):
    """
    Approximate the LP best response: find b minimizing max_k (y*b)_k
    using mirror descent on LSE approximation.
    """
    n = len(y)
    b = np.ones(n) / n
    
    for _ in range(n_inner):
        # Compute conv(y, b)
        c = np.convolve(y, b)  # length 2n-1
        Tc = T_br * c
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        w = exp_Tc / exp_Tc.sum()
        
        # Gradient of LSE w.r.t. b: 2 * conv(w, y[::-1]) reversed... 
        # Actually, c[m] = sum_j y[j]*b[m-j]
        # dc[m]/db[i] = y[m-i] if valid → same structure
        # grad_b[i] = 2 * sum_k w[i+k] * y[k] (by symmetry of the formula)
        
        # Use FFT
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        Y_rev = rfft(y[::-1].copy(), fft_size)
        W = rfft(w.copy(), fft_size)
        conv = irfft(Y_rev * W, fft_size)
        grad_b = 2.0 * conv[n - 1:n - 1 + n]
        
        # Mirror descent step
        eta_br = 1.0 / (2.0 * T_br**0.5)
        log_b = np.log(np.maximum(b, 1e-30))
        b = np.exp(log_b - eta_br * grad_b)
        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-30:
            b = np.ones(n) / n
        else:
            b = b / s
    
    return b

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using PCHIP-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
