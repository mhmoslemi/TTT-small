def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def power_sum_obj_grad(z, n, p, T):
    """
    Power sum approximation: f(a) = T * log((1/(2n-1)) * sum((b/T)^p))
    For large p, this approximates max(b).
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    num_b = len(b)
    # Power sum: (1/m * sum((b_i/T)^p))^(1/p) * T ≈ max(b_i) for large p
    # We use: T * log((1/m) * sum(exp(p * (b_i - max_b)/T)))
    # Actually let's use: (sum(b_i^p) / m)^(1/p) which for large p ≈ max(b_i)
    # But for numerical stability, use log-sum-exp style
    
    max_b = b.max()
    if T < 1e-15:
        return 2.0 * n * max_b, None
    
    # f = T * log(sum(exp(b/T)) / m)  [LogSumExp style]
    # Or for power sum: (sum(b^p)/m)^(1/p)
    # Let's stick with LogSumExp but parameterize differently
    
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum() / num_b)
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * n * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

                best_eta = eta_trial

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init_asymmetric(n, rng):
    """Create asymmetric structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Exponential decay (strongly asymmetric)
        a = np.exp(-3 * x)
    elif choice == 1:
        # Bump off-center
        a = np.exp(-10 * (x - 0.3) ** 2)
    elif choice == 2:
        # Two bumps
        a = np.exp(-8 * (x - 0.2) ** 2) + 0.5 * np.exp(-8 * (x - 0.8) ** 2)
    elif choice == 3:
        # Power law
        a = (1 - x) ** 2 + 0.01
    elif choice == 4:
        # Sinc-like, shifted
        center = int(0.4 * n)
        t = (np.arange(n) - center) / (n / 2)
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 5:
        # Cosine window, asymmetric
        a = np.cos(np.pi * x * 0.7) ** 2
        a = np.maximum(a, 0)
    elif choice == 6:
        # Gaussian mixture
        a = np.exp(-5 * (x - 0.25) ** 2) + 0.7 * np.exp(-5 * (x - 0.6) ** 2) + 0.3 * np.exp(-5 * (x - 0.9) ** 2)
    elif choice == 7:
        # Triangle, asymmetric
        a = np.where(x < 0.4, x / 0.4, (1 - x) / 0.6)
    elif choice == 8:
        # Raised cosine, shifted
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.2)))
        a = np.maximum(a, 0)
    else:
        # Random smooth
        nyq = np.fft.rfftfreq(n)
        spectrum = np.random.default_rng(rng.integers(0, 1000000)).exponential(1, len(nyq))
        spectrum[0] = 0
        a = np.fft.irfft(spectrum * np.exp(1j * np.angle(np.fft.rfft(np.ones(n)))), n)
        a = np.maximum(a, 0)
    
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def perturbed_sequence(a, rng, strength=0.03):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    n_freqs = rng.integers(1, 6)
    perturbation = np.zeros(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        perturbation += strength / n_freqs * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Ultra-fine LogSumExp L-BFGS with 35+ temperature stages down to T=0.0001
2. Peak shaving with Polyak step size and cooling schedule
3. Nesterov-accelerated LP alternating method
4. Multi-scale diversification with asymmetric initializations
5. Adaptive time allocation with final combined polish
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    num_b = len(b)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum() / num_b)
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * n * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def _lbgfs_obj(z, n, T_stage, deadline):
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_stage)
    return obj


def _lbgfs_jac(z, n, T_stage, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_stage)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    if deadline is None:
        deadline = time.monotonic() + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        stages_left = max(1, n_stages - stage_i)
        iters = max(30, min(maxiter_per_stage, int(remaining * 80 / stages_left)))

        def func(z_in):
            return _lbgfs_obj(z_in, n, stage_T, deadline)

        def jac(z_in):
            return _lbgfs_jac(z_in, n, stage_T, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 300,
                         'maxls': 30, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_shave_polyak(a, time_limit, n_steps=200000):
    """Peak shaving with Polyak step size and cooling."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Lower bound: 2n/(2n-1) for sum(a)=1
    lower_bound = 2.0 * n / (2 * n - 1)
    f_star = lower_bound

    eta = 0.05
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b

        # Find active peaks (within 0.5% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient (average of shifted a at active peaks)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Polyak step size with cap
        g_norm_sq = np.dot(g, g)
        if g_norm_sq > 1e-30:
            eta_polyak = (current_score - f_star - 1e-10) / g_norm_sq
            eta_polyak = max(1e-8, min(eta_polyak, 1.0))
        else:
            eta_polyak = eta

        # Blend with cooling schedule
        eta_eff = 0.3 * eta_polyak + 0.7 * eta * scale

        # Multiplicative update
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        new_score = 2.0 * n * fast_convolve(a_new).max()
        if new_score < best_score - 1e-15:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            eta *= 0.8
            if scale < 0.001:
                break

    return best_a, best_score


def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a

def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init_asymmetric(n, rng):

    elif choice == 3:
        a = (1 - x) ** 2 + 0.01
    elif choice == 4:
        center = int(0.4 * n)
        t = (np.arange(n) - center) / (n / 2)
        a = np.sinc(t)
        a = np.maximum(a, 0)

        a = np.cos(np.pi * x * 0.7) ** 2
        a = np.maximum(a, 0)

    else:

def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    n_freqs = rng.integers(1, 5)
    perturbation = np.zeros(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, 12)
        phase = rng.uniform(0, 2 * np.pi)
        perturbation += strength / n_freqs * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

            if n_try == n_best:
                continue
            t_sub = min(rem_sub * 0.25, 50.0)
            if t_sub > 12:
                print(f"  Trying n={n_try}, budget={t_sub:.0f}s")
                init_try = interpolate_sequence(best_seq, n_try)
                a_sp = spectral_flatten(init_try, n_iters=15, blend=0.4)
                T_sub = [2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008]
                a_opt = lbgfs_refine(a_sp, t_sub * 0.85, T_sub, maxiter_per_stage=5000,
                                     deadline=time.monotonic() + t_sub * 0.85)
                local_update(a_opt, f"n={n_try}")

    # === Phase 10: Final Polish ===
    print(f"=== Phase 10: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 50.0)
        if t_fin > 10:
            T_fin = [0.5, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0007]
            a_fin = lbgfs_refine(best_seq, t_fin, T_fin, maxiter_per_stage=5000,
                                deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.35, 30.0)
        if t_ps > 5:
            a_ps, _ = peak_shave_polyak(best_seq, t_ps, n_steps=50000)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_flp = min(rem * 0.4, 30.0)
        if t_flp > 8:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=20000)
            local_update(a_flp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
