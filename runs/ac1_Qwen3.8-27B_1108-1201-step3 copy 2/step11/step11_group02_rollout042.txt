import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def build_conv_matrix(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # i.e., for column j, T[j+j', j] = a[j'] for j'=0,...,n-1
    rows = []
    cols = []
    data = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(a)
    T = csr_matrix((data, (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))), 
                   shape=(size, n))
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated LP alternating method.
    Minimizes max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    Uses the minimax duality: min_a max_b (a*b) = max_b min_a (a*b)
    where a,b are probability vectors.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Pre-build LP variables (reuse across iterations for speed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    stall = 0
    rng = np.random.default_rng(42)
    
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: min t s.t. T(y) @ b <= t, sum(b)=1, b>=0
        T = build_conv_matrix(y)
        constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0)
        sb = b_new.sum()
        if sb < 1e-15:
            continue
        b_new = b_new / sb
        
        # Nesterov update
        alpha = 2.0 / (it + 3)
        y_new = (1 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a.copy()
            sy = 1.0
        y_new = y_new / sy
        
        # Weighted average for a (FTRL / multiplicative weights style)
        a_new = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
            sa = 1.0
        a_new = a_new / sa
        
        y = y_new
        a = a_new
        
        # Check improvement
        val_a = fast_convolve(a).max()
        val_y = fast_convolve(y).max()
        
        if val_a < best_val:
            best_val = val_a
            best_a = a.copy()
            stall = 0
        elif val_y < best_val:
            best_val = val_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic exploration on stall
        if stall >= 50:
            # Perturb y
            noise = rng.standard_normal(n) * (0.02 * np.sqrt(stall / 50))
            y = y + noise
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            a = y.copy()
            stall = 0
        
        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_val
            print(f"    iter={it+1} ({elapsed:.0f}s): raw_val={best_val:.8f}, score={score:.8f}")
    
    return best_a


def subgradient_mirror_descent(a0, time_limit, verbose=True):
    """
    Mirror descent (KL divergence) with subgradient for the convex problem:
    min max_k (a*a)[k] s.t. a >= 0, sum(a) = 1.
    Uses the "active set" subgradient and Polyak step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Estimate upper bound (for Polyak step)
    U = best_val  # upper bound on optimal value (crude)
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution and find active peaks
        b = fast_convolve(a)
        max_b = b.max()
        
        # Use top peaks for subgradient (smoothing)
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        
        # Subgradient: average of gradients from active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Mirror descent step (log barrier / multiplicative weights)
        # Step size: use diminishing schedule
        eta = 1.0 / (np.sqrt(it + 1) + 1.0)
        eta = np.clip(eta, 0.001, 0.5)
        
        # Update in log space
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        a = a_new
        
        # Check
        val = fast_convolve(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
        
        if verbose and (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_val
            print(f"    MD iter={it+1} ({elapsed:.0f}s): score={score:.8f}")
    
    return best_a


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: reduce the top-contributing elements to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        top_m = min(5, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                # Redistribute to others proportionally
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_val - 1e-14:
                    a = a_trial
                    best_val = val
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 100:
                # Try simultaneous reduction
                a_trial = a.copy()
                for pos in top_positions[:3]:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.998
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_val - 1e-14:
                    a = a_trial
                    best_val = val
                    best_a = a.copy()
                    stall = 0
                elif stall > 500:
                    break
        else:
            stall = 0
        
        if verbose and (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_val
            print(f"    CD step={step+1} ({elapsed:.0f}s): score={score:.8f}")
    
    return best_a


def structured_init(n, rng):

    choice = rng.integers(0, 5)

    else:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    return a / s
