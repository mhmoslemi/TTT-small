import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute h_j = sum_k w_k * a_{k-j} for j=0..n-1 using FFT."""
    n = len(a)
    size = 4 * n
    W = fft(np.asarray(w, dtype=np.float64), size)
    A_rev = fft(a[::-1].copy(), size)
    conv_result = ifft(W * A_rev).real
    return conv_result[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_and_grad_z(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. z (in softmax param)."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    w = exp_terms / exp_terms.sum()
    lse_val = max_b + T * np.log(w.sum())
    obj = 2.0 * n * lse_val

    W = w.dot(b)
    h = fast_cross_corr(w, a)
    grad_z = 2.0 * n * a * (h - W)
    return obj, grad_z


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_best_solver():
    """Find the fastest working LP solver."""
    for solver_name in [cp.PDLP, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve inner LP: min M s.t. T@a b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.MOSEK, cp.ECOS, cp.SCS, cp.CLARABEL]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                return b_val / s_sum, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None

def adam_logsumexp(a0, time_limit, T_start=5.0, T_end=0.005, lr=0.05,
                   beta1=0.9, beta2=0.999, eps=1e-8, warmup_steps=100):
    """Adam optimizer on LogSumExp with continuous temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    max_steps = 50000

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        t = step + 1

        progress = min(1.0, t / max_steps)
        T = T_start * (T_end / T_start) ** progress

        obj, grad = lse_obj_and_grad_z(z, n, T)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr_t = lr * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
        lr_t = min(lr_t, 1.0 / (np.sqrt(v_hat + eps).max() + eps))
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate periodically
        if step % 10 == 0:
            a = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()

        if step % 500 == 0 and step > 0:
            a_check = softmax_from_z(z)
            score_check = 2.0 * n * fast_convolve(a_check).max()
            print(f"    AdamLogS {step}: T={T:.4f} score={score_check:.8f} best={best_score:.8f}")

    a_final = softmax_from_z(z)
    score_final = 2.0 * n * fast_convolve(a_final).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()
    return best_a, best_score


def spectral_flatten(a, n_iters=100, blend=0.3, deadline=None):
    """Gerchberg-Saxton spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:L // 2 + 1]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    return a

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LP alternating, Adam LogSumExp, spectral methods,
    and targeted diversification with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = find_best_solver()
    print(f"LP solver: {solver}")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 50 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_local(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(1200, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            n_best = len(a)
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={n_best})")

    rem = deadline - time.monotonic()
    print(f"Budget: {rem:.0f}s, starting: {best_score:.8f}")

    # === Phase 0: Quick spectral pre-processing (10s) ===
    print(f"\n=== Phase 0: Spectral Pre-processing ===")
    rem = deadline - time.monotonic()
    t_spec0 = min(rem * 0.02, 10.0)
    if t_spec0 > 3:
        a_sf = spectral_flatten(best_seq, n_iters=50, blend=0.3,
                                deadline=time.monotonic() + t_spec0)
        local_update(a_sf, "spec0")

    # === Phase 1: Nesterov LP (40% of remaining) ===
    print(f"=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp1 = min(rem * 0.40, 500.0)
    if t_lp1 > 30:
        print(f"  Budget: {t_lp1:.0f}s")
        a_lp1, _ = nesterov_lp_refine(best_seq, t_lp1, solver, n_max_iters=500000)
        local_update(a_lp1, "nlp1")

    # === Phase 2: Adam LogSumExp (25% of remaining) ===
    print(f"=== Phase 2: Adam LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_adam = min(rem * 0.25, 350.0)
    if t_adam > 30:
        print(f"  Budget: {t_adam:.0f}s")
        a_adam, _ = adam_logsumexp(best_seq, t_adam, T_start=5.0, T_end=0.005,
                                   lr=0.05, max_steps_check=50000)
        local_update(a_adam, "adam1")

    # === Phase 3: Second LP round on best (15%) ===
    print(f"=== Phase 3: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.15, 200.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=300000)
        local_update(a_lp2, "nlp2")

    # === Phase 4: Adam LogSumExp round 2 with lower temp (10%) ===
    print(f"=== Phase 4: Adam LogSumExp Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_adam2 = min(rem * 0.10, 120.0)
    if t_adam2 > 15:
        print(f"  Budget: {t_adam2:.0f}s")
        a_adam2, _ = adam_logsumexp(best_seq, t_adam2, T_start=1.0, T_end=0.003,
                                    lr=0.03)
        local_update(a_adam2, "adam2")

    # === Phase 5: Diversification (10%) ===
    print(f"=== Phase 5: Diversification (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_div = min(rem * 0.10, 120.0)
    if t_div > 30:
        n_trials = 4
        t_per_trial = t_div / n_trials
        for trial in range(n_trials):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 15:
                break
            x = np.linspace(0, 2 * np.pi, n_best)
            freqs = rng.integers(1, 12, size=2)
            phases = rng.uniform(0, 2 * np.pi, size=2)
            noise = 0.04 * np.cos(freqs[0] * x + phases[0]) + 0.02 * np.cos(freqs[1] * x + phases[1])
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
            # Quick optimize: brief LP + peak shave
            t_sub = min(rem_sub * 0.4, t_per_trial)
            if t_sub > 10:
                a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=50000)
                local_update(a_opt, f"div{trial}_lp")
            rem_sub = deadline - time.monotonic()
            if rem_sub > 10:
                a_ps, _ = peak_shave_fast(a_pert, min(rem_sub * 0.3, 20.0), n_steps=50000)
                local_update(a_ps, f"div{trial}_ps")

    # === Phase 6: Try different n (8%) ===
    print(f"=== Phase 6: Different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [int(n_best * 0.85), int(n_best * 1.15)]:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            if n_try < 200 or n_try > 3000:
                continue
            print(f"  Trying n={n_try}, budget={rem_sub:.0f}s")
            a_interp = interpolate_sequence(best_seq, n_try)
            a_sf = spectral_flatten(a_interp, n_iters=20, blend=0.3)
            t_sub = min(rem_sub * 0.4, 80.0)
            if t_sub > 10:
                a_opt, _ = nesterov_lp_refine(a_sf, t_sub, solver, n_max_iters=30000)
                local_update(a_opt, f"n{n_try}_lp")
            rem_sub = deadline - time.monotonic()
            if rem_sub > 15:
                a_ad, _ = adam_logsumexp(a_opt if a_opt is not None else a_sf,
                                         min(rem_sub * 0.5, 30.0), T_start=2.0, T_end=0.01, lr=0.04)
                local_update(a_ad, f"n{n_try}_adam")

    # === Phase 7: Final polish (remaining) ===
    print(f"=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin_lp = min(rem * 0.5, 60.0)
        if t_fin_lp > 10:
            a_fin, _ = nesterov_lp_refine(best_seq, t_fin_lp, solver, n_max_iters=100000)
            local_update(a_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin_adam = min(rem * 0.6, 40.0)
        if t_fin_adam > 10:
            a_fa, _ = adam_logsumexp(best_seq, t_fin_adam, T_start=0.5, T_end=0.002, lr=0.02)
            local_update(a_fa, "final_adam")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_ps, _ = peak_shave_fast(best_seq, min(rem * 0.7, 20.0), n_steps=30000)
        local_update(a_ps, "final_ps")

    # Final output
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            ev = evaluate_sequence(test_seq)
            print(f"Verified: {ev:.8f}")
        except Exception as e:
            print(f"Verify error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
