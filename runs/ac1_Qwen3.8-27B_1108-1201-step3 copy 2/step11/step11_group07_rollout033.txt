    """Solve the LP to find the best response to f."""
    n = len(f)
    # Build constraint matrix A such that A @ g <= M * ones(2n-1)
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    k_vals = np.arange(2*n - 1)
    i_vals = np.arange(n)
    # A[k, i] = f[k-i] if 0 <= k-i < n
    diff = k_vals[:, None] - i_vals[None, :]  # (2n-1) x n

    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    return res.x if res.success else None

import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def score(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def lp_response(f, M, verbose=False):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    # Build constraint matrix
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0

    k_vals = np.arange(2*n - 1)

    A = np.zeros((2*n-1, n))
    A[mask] = f[diff[mask]]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': 10})

    return None, res

    a = normalize(rng.dirichlet(np.ones(n)))
    fft_size = 1 << (2*n - 1).bit_length()
    
    for _ in range(200):
        A = rfft(a, fft_size)
        P = np.abs(A)**2
        P_mean = P.mean()
        
        # Compute gradient direction to flatten spectrum
        r = (P - P_mean) / (P_mean + 1e-15)
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-10
        H[valid] = -r[valid] * P_mean * A[valid] / (P[valid] + 1e-10)
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)

            if s < 1e-15:

    a = normalize(a0.copy())
    best_a = a.copy()

    lb = 2.0 * n / (2*n - 1)  # Lower bound

        c_max = c.max()
        S = np.where(c >= c_max * (1 - 1e-10))[0]

            je = min(n - 1, k)
            g[js:je+1] += a[k - js : k - je + n - js]  # a[k-m] for m in [js, je]
        g *= 2.0 / len(S_use)

        gap = current_obj - lb
        g_norm_sq = np.sum(g**2)

            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_score

"""
LP-based alternating optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key strategies:
1. LP-based alternating optimization (solve exact best-response LP at each step)
2. Spectral initialization for diverse starting points

4. Subgradient polishing with Polyak step
5. Dual certificates for convergence detection

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def build_conv_matrix(n, f):
    """Build the (2n-1) x n convolution matrix A where A[k,i] = f[k-i] if valid."""
    k_vals = np.arange(2 * n - 1)
    i_vals = np.arange(n)
    diff = k_vals[:, None] - i_vals[None, :]
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = f[diff[mask]]
    return A


def solve_lp_response(f, M, timeout=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_y, success)."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    
    if res.success:
        g = res.x
        dual_y = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
        return g, dual_y, True
