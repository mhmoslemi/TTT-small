    T_mat = build_conv_matrix(a_fixed)

    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]

        except Exception:
            continue

2. Compute `b = conv(a,a)`, find `k_max = argmax(b)`
3. Solve: `min a^T Q_{k_max} a` subject to `a ≥ 0, sum(a) = 1` (a single QP)
4. This gives a new `a`
5. Go to step 2

This is a cyclic best-response method. It should converge (each step doesn't increase the objective at the previously active point). But it might converge very slowly.

A better version: at each step, solve the QP for the CURRENTLY most active constraint, but also check if the solution violates other constraints, and add them.

This is essentially the active-set method, which is what the previous program does.

1. Binary search on `t`
2. For a given `t`, check feasibility: is there `a ≥ 0, sum(a)=1` with `a^T Q_k a ≤ t` for all k?
3. The feasibility check is: `min max_k a^T Q_k a ≤ t`, which is the same as our original problem... circular.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using high-precision LP alternating
with Nesterov acceleration, active-set QCQP polishing, KKT-based verification,
spectral restarts for diversity, and broad multi-n exploration (1195-1205).
Key improvements over prior: tighter LP tolerances, duality-gap monitoring,
adaptive step sizing, frequency-domain perturbations, and longer convergence.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_best_lp_solver():
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x[0] <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, tol=1e-11):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.MOSEK, cp.GUROBI, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            kw = {}
            if s == cp.CLARABEL:
                kw = {'tol_gap_abs': tol, 'tol_gap_rel': tol, 'tol_feas': tol, 'tol_infeas': tol}
            elif s == cp.MOSEK:
                kw = {'MSDK_OPTIMIZER_MAX_TIME': 60}
            elif s == cp.GUROBI:
                kw = {'TimeLimit': 60}
            prob.solve(solver=s, **kw)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    continue
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def build_q_sparse(k, n):
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp(a, solver, rel_tol=1e-5, max_active=25):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]

    if len(active) > max_active:
        active = active[np.argsort(b[active])[-max_active:]]

    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    continue
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        # Adaptive Nesterov: use standard 2/(t+2) but switch to smaller
        # steps when close to convergence
        if it < 5000:
            alpha = 2.0 / (it + 2)
        else:
            # Switch to geometric decay for final convergence
            alpha = max(0.01, 2.0 / (it + 2))

        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate every 10 iterations to save time
        if it % 10 == 0:
            score_a = 2.0 * n * fast_convolve(a).max()
            score_y = 2.0 * n * fast_convolve(y).max()

            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        # Stall detection and spectral restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                # Frequency-domain perturbation
                A = rfft(y, 2 * n)
                phase_noise = 0.02 * rng.standard_normal(len(A))
                A_pert = A * np.exp(1j * phase_noise)
                y_new = irfft(A_pert, 2 * n)[:n]
                y_new = np.maximum(y_new, 0)
                sy2 = y_new.sum()
                if sy2 > 1e-15:
                    y = y_new / sy2
                else:
                    y = y + 0.01 * rng.standard_normal(n)
                    y = np.maximum(y, 0)
                    y = y / y.sum()
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.12f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 200 and np.mean(lp_times[-200:]) > 8.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_subgradient(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.02
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.7
            no_improve = 0
            if scale < 0.002:
                break

    return best_a, best_score


def kkt_check(a, n_active_tol=1e-6):
    """
    Check KKT conditions. Returns (is_optimal, active_set, dual_weights).
    At optimum: there exist dual weights w_k >= 0, sum(w_k)=1 such that
    sum_k w_k * 2*Q_k*a = lambda * 1 (for positive a_i)
    and sum_k w_k * 2*Q_k*a >= lambda * 1 (for zero a_i)
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - n_active_tol)
    active = np.where(b >= threshold)[0]
    return active, max_b


def solve_full_active_qcqp(a, solver, rel_tol=1e-4, max_active=50):
    """Solve QCQP with a broader active set for better polishing."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]

    if len(active) > max_active:
        # Keep the top max_active
        active = active[np.argsort(b[active])[-max_active:]]

        active = np.array([int(np.argmax(b))])

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
