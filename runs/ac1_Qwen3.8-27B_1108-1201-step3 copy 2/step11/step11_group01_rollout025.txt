    n = len(a)
    fft_size = 2*n - 1  # or next power of 2
    a = a.copy()
    for i in range(n_iters):
        # Go to frequency domain
        A = fft(a, fft_size)
        mag = abs(A)
        # Flatten the magnitude (keep DC)
        mag[1:] = np.ones(fft_size - 1) * mag[1:].mean()  # or some target
        # Actually, we want to reduce the dynamic range
        # New target: interpolate between current mag and flat
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]  # keep DC
        target_mag[1:] = np.median(mag[1:])  # or mean
        # Blend
        mag_new = 0.5 * mag + 0.5 * target_mag
        # Reconstruct with current phase, new magnitude
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = ifft(A_new).real[:n]
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        # Normalize
        a_new = a_new / a_new.sum()
        a = a_new
    return a

F(z) = (2n/T) * log(sum_k exp(T * conv(softmax(z), softmax(z))[k]))

The gradient is:
dF/dz_i = (dF/da_i) * (da/da)_i * (da_i/dz_i) ... 

Actually, if a = softmax(z), then da_i/dz_j = a_i * (delta_{ij} - a_j).

def fast_convolve(a):
    """Compute full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def lse_objective_grad(z_in, T_val, n):
    """Compute LSE objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
        Tb = Tb - Tb.max() + 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient: grad_a[i] = 2 * sum_j a[j] * w[i+j]
    # Use cross-correlation via FFT
    n_w = len(w)  # 2n-1
    fft_size = 1 << (n + n_w - 1 - 1).bit_length()
    a_rev = a[::-1]
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(A_rev * W, fft_size)[:n]
    # cross[i] = sum_j a_rev[j] * w[i-j] = sum_j a[n-1-j] * w[i-j]
    # We want: grad_a[i] = 2 * sum_j a[j] * w[i+j]
    # conv(a_rev, w)[k] = sum_j a_rev[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
    # For k = n-1+i: sum_j a[n-1-j] * w[n-1+i-j]
    #   Let m = n-1-j: sum_m a[m] * w[i+m] ✓
    
    # So we need conv(a_rev, w)[n-1 : n-1+n]
    # The full conv has length n + (2n-1) - 1 = 3n-2
    # cross computed above has length n (first n elements)
    # But we need elements [n-1, n-1+n) of the full conv
    
    # Let me recompute properly
    full_size = n + n_w - 1  # 3n-2
    fft_size2 = 1 << (full_size - 1).bit_length()
    A_rev2 = rfft(a_rev, fft_size2)
    W2 = rfft(w, fft_size2)
    full_cross = irfft(A_rev2 * W2, fft_size2)[:full_size]
    grad_a = 2.0 * full_cross[n-1:n-1+n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def subgradient_polish(a0, time_limit, n_steps=200000):
    """Subgradient descent on the exact max function."""
    n = len(a0)

    s = a.sum()
    if s < 1e-15:

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Use all "active" positions (within 5% of max)
        threshold = max_b * 0.9
        active = np.where(b >= threshold)[0]
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        # Adaptive step size
        eta = 0.1 / (1.0 + 0.01 * step)
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
        else:
            # Occasionally accept worse solutions (simulated annealing)
            if rng.random() < 0.1:
                z = z_new
                a = a_new

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
