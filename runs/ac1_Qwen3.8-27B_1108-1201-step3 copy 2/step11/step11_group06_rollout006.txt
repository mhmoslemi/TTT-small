**Another novel idea: Use the "water-filling" metaphor.**

A concrete algorithm:
1. Compute `b = a*a`.
2. Find the peak positions.
3. For each peak at position `k`, compute the "excess" `b[k] - mean(b)`.
4. Adjust `a` to reduce the excess at peak positions.

The adjustment: to reduce `(a*a)_k`, we can reduce `a_j` for the `j` that contribute most to the peak. Specifically, the contribution of `a_j` to `(a*a)_k` is `2 * a_j * a_{k-j}` (or `a_j^2` if `j = k/2`). So we should reduce `a_j` proportional to `a_{k-j}` (the "partner" coefficient).

This is exactly what the subgradient does! So the subgradient method is the natural "water-filling" algorithm.

def compute_subgradient(a, b, n_active=10, epsilon=1e-6):
    """Compute weighted subgradient of max_k b[k] at point a."""
    n = len(a)
    M = b.max()
    # Active set: peaks within epsilon of max
    threshold = M - max(epsilon * M, 1e-10)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    # Use top n_active if too many
    if len(active) > n_active:
        idx = np.argsort(b[active])[::-1][:n_active]
        active = active[idx]
    # Weights proportional to values
    w = b[active]
    w = w / w.sum()
    # Compute subgradient
    g = np.zeros(n)
    for k, wk in zip(active, w):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_idx = np.arange(js, je + 1)
        g[j_idx] += 2.0 * wk * a[k - j_idx]
    return g

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast momentum subgradient with FFT cross-correlation
2. Multi-stage LogSumExp L-BFGS (log-domain) with fine temperature annealing
3. Peak-directed refinement with adaptive step sizes
4. Multi-length exploration with structured initializations
Innovations: FFT-based cross-correlation for O(n log n) gradients,
log-domain parameterization (no projection needed), momentum acceleration,
and peak-aware final polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def crosscorr_fft(p, a):
    """Compute c_i = sum_k p[k] * a[k-i] for i in [0, len(a)-1]."""
    n_p = len(p)
    n_a = len(a)
    N = 1 << (n_p + n_a - 1).bit_length()
    P = rfft(p, N)
    A = rfft(a, N)
    return irfft(P * A.conj(), N)[:n_a]


def lse_obj_grad_a(a, T):
    """LSE objective and gradient w.r.t. a (a on simplex, sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    exp_shifted = np.exp(shifted)
    lse = max_b + T * np.log(exp_shifted.sum() + 1e-300)
    p = exp_shifted / (exp_shifted.sum() + 1e-300)
    corr = crosscorr_fft(p.astype(np.float64), a)
    grad = 2.0 * corr
    return lse, grad


def softmax_fn(z):
    z_shift = z - z.max()
    e = np.exp(z_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def log_fn(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-domain (z space)."""
    a = softmax_fn(z)
    n = len(a)
    lse, grad_a = lse_obj_grad_a(a, T)
    obj = 2.0 * n * lse
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = v / target_sum
    u_sorted = np.sort(u)[::-1]
    cumulative = np.cumsum(u_sorted)
    cond = u_sorted * np.arange(1, n + 1) > (cumulative - 1)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cumulative[rho] - 1) / (rho + 1.0)
    proj = np.maximum(u - theta, 0)
    return proj * target_sum


def subgradient_momentum(a0, time_limit, eta0=0.3, n_active=15):
    """Momentum subgradient method with simplex projection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a_prev = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    eta = eta0
    momentum = 0.9
    it = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()
        # Active set
        threshold = M * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > n_active:
            idx = np.argsort(b[active])[::-1][:n_active]
            active = active[idx]
        w = b[active]
        w = w / w.sum()
        # Subgradient
        g = np.zeros(n)
        for k, wk in zip(active, w):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_idx = np.arange(js, je + 1)
            g[j_idx] += 2.0 * wk * a[k - j_idx]
        
        # Momentum
        v = momentum * (a - a_prev) - eta * g
        a_prev = a.copy()
        a_new = simplex_project(a + v)
        
        # Evaluate
        score_new = 2.0 * n * fast_convolve(a_new).max()
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            stagnant = 0
            eta = min(eta * 1.01, 1.0)
        else:
            stagnant += 1
            eta *= 0.9
            if stagnant > 200:
                # Restart momentum
                momentum *= 0.5
                eta = eta0 * 0.5
                stagnant = 0
                if momentum < 0.01:
                    break
        
        a = a_new
        it += 1
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG-M: t={elapsed:.0f}s iter={it} score={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def lse_lbfgs_refine(a0, time_limit, T_stages, maxiter_per_stage=1000):
    """L-BFGS with LogSumExp smoothing in log-domain."""
    n = len(a0)
    z = log_fn(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    for stage_T in T_stages:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_obj_grad_z(z_in, stage_T)
            return obj
        
        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros(len(z_in))
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-16, 'maxfun': maxiter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_fn(z)
            sc = evaluate_local(a_check)
            if time.monotonic() - t0 < time_limit:
                print(f"    LSE T={stage_T}: score={sc:.8f} (t={time.monotonic()-t0:.0f}s)")
        except Exception as e:
            print(f"    LSE T={stage_T}: error {e}")
    
    return softmax_fn(z)
