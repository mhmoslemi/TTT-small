def fast_autocorr(a):

    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_corr(a, b):
    """Cross-correlation: (a*b)_k = sum_j a[j]*b[j+k] for k in range."""
    n = len(a)

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=25):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    # Power-law temperature schedule (spends more time at high T)
    temps = T_start * (T_end / T_start) ** (np.arange(n_stages) / max(n_stages - 1, 1))
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # More iterations at high temperature
        frac = si / max(n_stages - 1, 1)
        iters_per_stage = max(10, int(60 * (1 + frac) + 20 * (1 - frac)))
        
        def func(z_in):
            val, _ = lse_obj_and_grad_z(z_in, stage_T)
            return val
        
        def jac(z_in):
            _, g = lse_obj_and_grad_z(z_in, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:

def peak_shaving(a0, time_limit, seed=42):
    """
    Adaptive peak shaving using gradient-guided coordinate descent.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

        # Sort peaks by height

        improved = False
        for k in top_k:
            if b[k] <= b.mean() * 1.01:
                break

            je = min(n - 1, k)
            if js > je:
                continue

            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Compute how much each position contributes to the top peaks

                    peak_contrib[idx2] += a[idx2] * a[k2 - idx2]

            scores = peak_contrib * 2 + a
            target_candidates = np.argsort(scores)[:max(5, n // 20)]

                target_candidates = [j for j in range(n) if j != src]
            
            dst = target_candidates[rng.integers(min(len(target_candidates), 10))]

                if amt < 1e-15:
                    continue

                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:

                best_a = a.copy()

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def spectral_init(n, rng):
    """
    Generate initialization with relatively flat power spectrum.
    Uses random phase + inverse FFT approach.
    """
    # Create a power spectrum with some structure
    k = np.arange(n // 2 + 1)
    # Start with flat, add some randomness
    P = np.ones(n // 2 + 1) * (0.5 + 0.5 * rng.random(n // 2 + 1))
    # Ensure symmetry for real output
    # Construct full spectrum
    spec = np.zeros(n, dtype=np.complex128)
    phases = 2 * np.pi * rng.random(n)
    for kk in range(n // 2 + 1):
        mag = np.sqrt(P[kk])
        spec[kk] = mag * np.exp(1j * phases[kk])
        if kk > 0 and kk < n - kk:
            spec[n - kk] = np.conj(spec[kk])
    
    a = np.real(np.fft.ifft(spec)).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        # Fallback
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    a = a / a.sum()
    return a

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based subgradient Nesterov (no LP, O(n log n) per step)
2. Multi-stage LogSumExp gradient optimization with L-BFGS (aggressive annealing)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_and_grad_z(z, T):
    """
    LSE-smoothed max autocorrelation and gradient w.r.t. z (where a=softmax(z)).
    Uses FFT for O(n log n) gradient.
    """
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)

    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def subgradient_nesterov(a0, time_limit, max_iters=1000000):
    """
    Fast subgradient Nesterov using FFT. Minimizes max(a*a)_k over simplex.
    O(n log n) per iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        momentum = (t - 1.0) / (t + 2.0)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        b_y = fast_autocorr(y)
        k_star = int(np.argmax(b_y))

        grad = np.zeros(n)
        if k_star == 0:
            grad = 2.0 * y
        else:
            end1 = n - k_star
            if end1 > 0:
                grad[:end1] += y[k_star:]
            if k_star < n:
                grad[k_star:] += y[:n - k_star]

        step = 1.5 / np.sqrt(t)
        a_new = y - step * grad
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn

        a_prev = a
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:

            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=30):
    """
    LogSumExp optimization with L-BFGS and aggressive multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = T_start * (T_end / T_start) ** (np.arange(n_stages) / max(n_stages - 1, 1))

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        frac = si / max(n_stages - 1, 1)
        iters = max(20, int(80 + 40 * frac))

        def func(z_in):
            val, _ = lse_obj_and_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_and_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters * 400,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Gradient-guided adaptive peak shaving with multi-direction search.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(20, 2 * n - 1)]

        improved = False
        for k in top_k:
            if b[k] <= b.mean() * 1.005:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            # Compute peak contribution map
            peak_contrib = np.zeros(n)
            for k2 in top_k[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 > je2:
                    continue
                idx2 = np.arange(js2, je2 + 1)
                peak_contrib[idx2] += a[idx2] * a[k2 - idx2]

            # Score targets: low peak contribution
            target_scores = peak_contrib + 0.1 * a
            n_targets = max(10, n // 15)
            target_pool = np.argsort(target_scores)[:n_targets]
            target_pool = [j for j in target_pool if j != src]

            if not target_pool:
                target_pool = [j for j in range(n) if j != src]

            best_trial_score = best_score
            best_move = None

            for dst in target_pool[:min(15, len(target_pool))]:
                for frac in [0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_trial_score:
                        best_trial_score = score
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_m, dst_m, amt_m = best_move
                a[src_m] -= amt_m
                a[dst_m] += amt_m
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_trial_score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Try random pair moves
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-15:
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break

            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_init(n, rng):
    """Generate initialization with relatively flat power spectrum."""
    k = np.arange(n // 2 + 1)
    P = np.ones(n // 2 + 1) * (0.3 + 0.7 * rng.random(n // 2 + 1))
    spec = np.zeros(n, dtype=np.complex128)
    phases = 2 * np.pi * rng.random(n // 2 + 1)
    for kk in range(n // 2 + 1):
        mag = np.sqrt(P[kk])
        spec[kk] = mag * np.exp(1j * phases[kk])
        if kk > 0 and kk < n - kk:
            spec[n - kk] = np.conj(spec[kk])
    a = np.real(np.fft.ifft(spec)).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        s = a.sum()
    return a / s


def generate_inits(n, rng, count=12):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** 2
    inits.append(a / a.sum())

    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    inits.append(a / a.sum())

    x = np.linspace(-1, 1, n)
    a = np.exp(-x ** 2 * 3)
    inits.append(a / a.sum())

    x = np.linspace(0, 2 * np.pi, n)
    a = 0.5 + 0.3 * np.cos(2 * x) + 0.15 * np.cos(5 * x) + 0.05 * np.cos(13 * x)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())

    x = np.linspace(0, 1, n)
    a = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    inits.append(a / a.sum())

    for _ in range(max(0, count - 5)):
        inits.append(spectral_init(n, rng))

    inits.append(np.ones(n) / n)
    return inits


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization with iterative refinement.
    Phase structure: init -> subgrad -> LSE -> peak -> re-optimize -> peak
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_start = 1200
        print("No parent available, generating initializations...")
        inits = generate_inits(n_start, rng, count=12)
        best_init = None
        best_init_score = float('inf')
        for init in inits:
            sc = 2.0 * n_start * fast_autocorr(init).max()
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        best_seq = best_init
        best_score = evaluate_safe(best_seq)
        print(f"Best init: n={n_start}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: Subgradient Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: SubGrad Nesterov n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.18, 300.0)
    if t1 > 20:
        a_out, _ = subgradient_nesterov(best_seq, t1)
        try_update(a_out, f"subgrad_n{n_base}")

    # Phase 2: LogSumExp at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LogSumExp n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 10:
        a_out, _ = logsumexp_optimize(best_seq, t2, T_start=80.0, T_end=0.01, n_stages=30)
        try_update(a_out, f"lse_n{n_base}")

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Peak shaving ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 5:
        a_out, _ = peak_shaving(best_seq, t3, seed=111)
        try_update(a_out, "peak1")

    # Phase 4: Re-optimize with LogSumExp (second pass)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LogSumExp pass 2 ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 10:
        a_out, _ = logsumexp_optimize(best_seq, t4, T_start=30.0, T_end=0.005, n_stages=25)
        try_update(a_out, "lse2")

    # Phase 5: Try longer length
    n_long = int(round(n_base * 1.4)) if n_base < 2500 else int(round(n_base * 1.2))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: SubGrad at n={n_long} ({remaining:.0f}s left) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = subgradient_nesterov(init_long, t5)
        try_update(a_out, f"subgrad_n{n_long}")

    # Phase 6: LogSumExp at longer length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: LogSumExp n={n_long} ({remaining:.0f}s left) ===")
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 10:
        a_out, _ = logsumexp_optimize(best_seq, t6, T_start=60.0, T_end=0.01, n_stages=25)
        try_update(a_out, f"lse_n{n_long}")

    # Phase 7: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Peak shaving final ({remaining:.0f}s left) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 5:
        a_out, _ = peak_shaving(best_seq, t7, seed=999)
        try_update(a_out, "peak_final")

    # Phase 8: Final LogSumExp polish (very low temperature)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final LSE polish ({remaining:.0f}s left) ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 10:
        a_out, _ = logsumexp_optimize(best_seq, t8, T_start=10.0, T_end=0.003, n_stages=20)
        try_update(a_out, "lse_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
