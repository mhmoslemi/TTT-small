def fast_autocorr(a):

def solve_lp_best_response(a_fixed, solver):

    and take a projected gradient step.
    """

    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    lr = lr_init
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_val = b.max()
        score = 2.0 * n * max_val
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            # Reduce learning rate
            lr *= 0.5
            stall = 0
            if lr < 1e-8:
                break

        top_idx = np.argsort(b)[::-1][:top_k]

        peak_vals = b[top_idx]
        w = np.exp(peak_vals - peak_vals.max())
        w = w / w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            valid_i = np.arange(js, je + 1)
            ki_vals = k - valid_i
            valid_mask = (ki_vals >= 0) & (ki_vals < n)
            valid_i = valid_i[valid_mask]
            ki_valid = ki_vals[valid_mask]
            grad[valid_i] += wi * 2.0 * a[ki_valid]
        
        # Project gradient step onto feasible set (simplex: a>=0, sum(a)=1)
        # Simple approach: a_new = a - lr * grad, then project
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check if this improves things (with some tolerance)
        new_score = 2.0 * n * fast_autocorr(a_new).max()
        
        if new_score < score:
            a = a_new
            # Slightly increase lr if improving
            lr = min(lr * 1.01, 0.1)
        else:

            a_new2 = a - lr * 0.1 * grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive LP Nesterov-accelerated best-response (primary driver)
2. FFT-based LogSumExp gradient with aggressive multi-stage annealing
3. Peak-directed subgradient with adaptive step size
4. Multi-length exploration with smooth interpolation
5. Symmetry-aware coordinate descent for final polish
Key improvements: more LP iterations, better annealing, symmetry exploitation,
adaptive time allocation based on convergence rate.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_autocorr(a, b):
    """Compute score given a and its autocorrelation b."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=200000, verbose_every=100):
    """
    Nesterov-accelerated LP best-response iteration with running average.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    b = fast_autocorr(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    last_improve_t = 0
    patience = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        # Check score of a
        b_a = fast_autocorr(a)
        score_a = 2.0 * n * b_a.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve_t = t
            patience = 0

        # Check score of y
        b_y = fast_autocorr(y)
        score_y = 2.0 * n * b_y.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve_t = t
            patience = 0

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg >= 5 and n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            b_avg = fast_autocorr(avg_check)
            score_avg = 2.0 * n * b_avg.max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                last_improve_t = t
                patience = 0

        patience += 1
        if t % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}, patience={patience}")

    return best_a, best_score


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        a = np.ones(len(z)) / len(z)
    else:
        a = e / s

    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # FFT-based gradient computation
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def logsumexp_optimize(a0, time_limit, T_start=200.0, T_end=0.01, n_stages=25, verbose_every=5):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage geometric annealing.
    """
    from scipy.optimize import minimize

    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters_per_stage = max(20, min(100, int(30 + 30 * (si / n_stages))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            b = fast_autocorr(a)
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123, top_k_peaks=10):
    """
    Coordinate-descent peak shaving: reduce top autocorrelation peaks
    by moving mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    best_a = a.copy()
    b = fast_autocorr(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            break

        top_k = np.argsort(b)[::-1][:top_k_peaks]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find target positions
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                b_trial = fast_autocorr(a_trial)
                new_score = 2.0 * n * b_trial.max()
                if new_score < best_score - 1e-14:
                    a = a_trial
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            # Random perturbation to escape
            n_perturb = rng.integers(2, 6)
            for _ in range(n_perturb):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        b_trial = fast_autocorr(a_trial)
                        new_score = 2.0 * n * b_trial.max()
                        if new_score < best_score:
                            a = a_trial
                            stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(11 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        # Triangular (Bartlett)
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)
    else:
        # Hamming-like window
        x = np.linspace(0, 1, n)
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def subgradient_optimize(a0, time_limit, top_k=8, lr_init=0.005):
    """
    Peak-directed subgradient with adaptive step size.
    Uses the top-k autocorrelation peaks to compute a weighted subgradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    best_a = a.copy()
    b = fast_autocorr(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()

    lr = lr_init
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        score = 2.0 * n * b.max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 200:
            lr *= 0.5
            stall = 0
            if lr < 1e-10:
                break

        # Compute weighted subgradient from top-k peaks
        top_idx = np.argsort(b)[::-1][:top_k]
        peak_vals = b[top_idx]
        # Softmax weights over top peaks (sharper = more focused)
        w = np.exp(5.0 * (peak_vals - peak_vals.max()))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(top_k) / top_k
        else:
            w = w / w_sum

        # Compute gradient: for peak k, grad[i] += 2*a[k-i] for valid i
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += wi * 2.0 * a[ki]

        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        b_new = fast_autocorr(a_new)
        new_score = 2.0 * n * b_new.max()

        if new_score < score:
            a = a_new
            lr = min(lr * 1.02, 0.05)
        else:
            # Try half step
            a_half = a - 0.5 * lr * grad
            a_half = np.maximum(a_half, 0)
            s_half = a_half.sum()
            if s_half > 1e-15:
                a_half = a_half / s_half
                b_half = fast_autocorr(a_half)
                if 2.0 * n * b_half.max() < score:
                    a = a_half
                    lr = min(lr * 1.01, 0.05)
                else:
                    lr *= 0.8
            else:
                lr *= 0.8

        it += 1
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD t={it} ({elapsed:.0f}s): best={best_score:.6f}, cur={score:.6f}, lr={lr:.6f}")

    return best_a, best_score

    n_long = max(n_long, len(best_seq) + 100)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP Nesterov at n={n_long} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t3, solver, max_iters=200000, verbose_every=100)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 4: LSE at longer length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LSE at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_out, _ = logsumexp_optimize(best_seq, t4, T_start=50.0, T_end=0.005, n_stages=20)
        try_update(a_out, f"lse_n{len(best_seq)}_2")

    # === Phase 5: Subgradient polishing ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Subgradient at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 120.0)
    if t5 > 5:
        a_out, _ = subgradient_optimize(best_seq, t5, top_k=8, lr_init=0.003)
        try_update(a_out, f"sgd_n{len(best_seq)}")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 100.0)
    if t6 > 5:
        a_out, _ = peak_shaving(best_seq, t6, seed=111, top_k_peaks=12)
        try_update(a_out, "peak_shave_1")

    # === Phase 7: Final LP at best length (short burst) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: LP refinement at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 15:
        a_out, _ = nesterov_lp_optimize(best_seq, t7, solver, max_iters=100000, verbose_every=50)
        try_update(a_out, f"lp_final_n{len(best_seq)}")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 3:
            print(f"\n=== Phase 8: Final peak shave (remaining={remaining:.0f}s) ===")
            a_out, _ = peak_shaving(best_seq, t8, seed=999, top_k_peaks=15)
            try_update(a_out, "peak_shave_final")

    # === Phase 9: Very short LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 8:
        t9 = min(remaining * 0.6, 40.0)
        if t9 > 3:
            print(f"\n=== Phase 9: Final LSE polish (remaining={remaining:.0f}s) ===")
            a_out, _ = logsumexp_optimize(best_seq, t9, T_start=5.0, T_end=0.005, n_stages=10)
            try_update(a_out, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
