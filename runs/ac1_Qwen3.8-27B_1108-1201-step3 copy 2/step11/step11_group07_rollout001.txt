def lp_iteration(a, M):
    # Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0
    # Using scipy.optimize.linprog
    n = len(a)
    # Variables: b_0, ..., b_{n-1}
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: conv(a,b)[k] <= M for k = 0, ..., 2n-2
    # conv(a,b)[k] = sum_j a[j] * b[k-j] (valid j)
    # For a given k, the constraint is: sum_{j: 0<=k-j<n} a[j] * b[k-j] <= M
    # Let i = k-j, so j = k-i, and 0 <= k-i < n means k-n < i <= k, so max(0,k-n+1) <= i <= min(n-1, k)
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            A_ub[k, i] = a[j]  # a[k-i] * b[i]
    
    b_ub = np.full(2*n-1, M)
    
    # Solve LP
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_solve(a, M):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    # Build constraint matrix
    # conv(a,b)[k] = sum_i a[k-i] * b[i] for valid i
    # We need: for each k, sum_i a[k-i] * b[i] <= M
    
    # The matrix is banded. Let's build it as a sparse matrix.
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None

def lp_iteration(a, M, t):
    """One step of the LP-based iteration."""
    b = lp_solve(a, M)
    if b is None:
        return a
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        return a
    b = b / s
    # Blend: a_new = (1-t)*a + t*b
    a_new = (1 - t) * a + t * b
    return normalize(a_new)

    i_vals = col_idx
    vals = a[k_vals - i_vals].astype(np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_based_optimize(a0, time_limit, initial_t=0.5, verbose=True):
    """LP-based fixed-point iteration."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    row_idx, col_idx = build_lp_indices(n)
    
    t = initial_t
    no_improve = 0
    iteration = 0
    stalled = False
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = c.max()
        
        b = lp_solve(a, M, row_idx, col_idx)
        if b is None or b.sum() < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        b_norm = b / b.sum()
        
        # Try multiple step sizes
        improved = False
        for trial_t in [t, t * 0.5, t * 0.25]:
            a_new = normalize((1 - trial_t) * a + trial_t * b_norm)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                t = trial_t  # Keep the step size that worked
                no_improve = 0
                improved = True
                break
        
        if not improved:
            no_improve += 1
            t *= 0.9
            if t < 0.05:
                # Restart with perturbation
                rng = np.random.default_rng(iteration)
                pert = rng.normal(0, 0.01, n)
                a = normalize(a + pert)
                t = initial_t
                no_improve = 0
        
        iteration += 1
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iteration}, score={best_score:.8f}, t={t:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def lse_gradient(a, T):
    """Compute gradient of LSE-smoothed objective for normalized a (sum=1)."""
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    
    # Gradient: grad[m] = 4n * sum_k w[k] * a[k-m]
    # Compute via cross-correlation FFT
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    grad = 4.0 * n * R[:n]
    
    return grad


def lse_objective(a, T):
    """Compute LSE-smoothed objective for normalized a (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    return 2.0 * n * lse


def lbgfs_optimize(a0, time_limit, T_stages, verbose=True):
    """L-BFGS optimization with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for stage_i, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Work in unconstrained space (log parameters)
        z = np.log(np.maximum(a, 1e-300))
        
        def func(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            return lse_objective(a_in, T)
        
        def jac(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            g = lse_gradient(a_in, T)
            # Transform to z-space: grad_z = a * (grad_a - sum(grad_a * a))
            S = np.dot(g, a_in)
            return a_in * (g - S)
        
        maxiter = max(20, int(remaining * 50))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': maxiter * 100, 'disp': False}
            )
            z_new = res.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
        except Exception:
            pass
        
        if verbose and stage_i % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS stage {stage_i} (T={T}), score={best_score:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score
