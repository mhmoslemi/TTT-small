def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalized_eval(a):
    """Evaluate assuming sum(a)=1, returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def perplexity_grad(a, top_k=None):
    """
    Compute gradient of max_k (a*a)_k (or top-k smoothed version).
    Returns (objective_value, gradient).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    if top_k is None or top_k >= len(b):
        # Hard max: subgradient is 2 * P_k * a for k = argmax
        k = int(np.argmax(b))
        # Gradient of (a*a)_k w.r.t. a is 2 * a[k::-1] padded
        g = np.zeros(n)
        # (a*a)_k = sum_j a_j * a[k-j], d/da_i = 2 * a[k-i] (with bounds)
        for i in range(n):
            ji = k - i
            if 0 <= ji < n:
                g[i] = 2.0 * a[ji]
        return max_b, g
    else:
        # Top-k smoothing
        idx = np.argsort(b)[::-1][:top_k]
        weights = b[idx] - b[idx[-1]]  # relative weights
        weights = np.maximum(weights, 0)
        if weights.sum() < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / weights.sum()
        
        g = np.zeros(n)
        for wi, ki in zip(weights, idx):
            for i in range(n):
                ji = ki - i
                if 0 <= ji < n:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Evaluate 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def reverse_shift(a, k):
    """Compute (R_k a)_i = a_{k-i} with zero padding for out-of-bounds."""
    n = len(a)
    result = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        result[lo:hi + 1] = a[k - lo:k - hi - 1:-1]
    return result


def topk_subgrad(a, top_k=1):
    """
    Compute top-k weighted subgradient of max_k (a*a)_k.
    Returns (objective, gradient_vector).
    Assumes sum(a)=1.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()

    if top_k <= 1 or top_k >= len(b):
        k = int(np.argmax(b))
        g = 2.0 * reverse_shift(a, k)
        return max_b, g
    else:
        # Top-k with softmax weights
        idx = np.argsort(b)[::-1][:top_k]
        vals = b[idx]
        w = vals - vals.min()
        w = np.maximum(w, 0)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / ws
        g = np.zeros(n)
        for wi, ki in zip(w, idx):
            g += wi * reverse_shift(a, int(ki))
        g *= 2.0
        obj = float(np.dot(w, vals))
        return obj, g


def lse_obj_grad(a, T=0.1):
    """
    LogSumExp smoothed objective and gradient.
    f_T(a) = 2n * T * log(sum_k exp((a*a)_k / T))
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    # Threshold for sparsity
    w[w < 1e-10] = 0
    w = w / w.sum()

    g = np.zeros(n)
    # Vectorized: for each active k, add w[k] * 2 * R_k(a)
    active = np.where(w > 0)[0]
    for ki in active:
        g += w[ki] * reverse_shift(a, int(ki))
    g *= 2.0
    # Scale by 2n/T for the full gradient of 2n*T*log(sum exp(b/T))
    # Actually: d/d a [2n * T * log(sum exp(b_k/T))] = 2n * sum_k (exp(b_k/T)/sum) * (1/T) * db_k/da
    # = 2n * sum_k w_k * (1/T) * 2*R_k(a)
    # Wait, I already have g = 2 * sum w_k R_k(a), so multiply by 2n/T... 
    # No wait. Let me redo.
    # f = 2n * T * log(sum_k exp(b_k/T))
    # df/da_i = 2n * T * (1/sum exp) * sum_k exp(b_k/T) * (1/T) * db_k/da_i
    # = 2n * sum_k w_k * db_k/da_i
    # db_k/da_i = 2 * a[k-i] = 2 * (R_k a)_i
    # So df/da_i = 2n * 2 * sum_k w_k * (R_k a)_i = 4n * (sum_k w_k R_k a)_i
    # But I computed g = 2 * sum w_k R_k a, so I need to multiply by 2n more.
    g *= (2.0 * n)
    return obj, g


def lse_obj_grad_topk(a, T=0.1, top_k=20):
    """LogSumExp but only over top-k peaks (more targeted)."""
    n = len(a)
    b = fast_convolve(a)
    # Only consider top-k
    if top_k < len(b):
        threshold = np.partition(b, len(b) - top_k)[len(b) - top_k]
        mask = b >= threshold
    else:
        mask = np.ones(len(b), dtype=bool)

    b_masked = b[mask]
    max_b = b_masked.max()
    exp_terms = np.exp((b_masked - max_b) / T)
    lse_val = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse_val

    w = exp_terms / exp_terms.sum()
    w[w < 1e-12] = 0
    if w.sum() < 1e-30:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w.sum()

    g = np.zeros(n)
    active_indices = np.where(mask)[0]
    for ki, wi in zip(active_indices, w):
        if wi > 0:
            g += wi * reverse_shift(a, int(ki))
    g *= (4.0 * n)  # 2n * 2 for the gradient
    return obj, g


def topk_subgrad_step(a, eta, top_k=1, momentum_v=None, beta=0.9):
    """
    One step of multiplicative (mirror descent) subgradient method.
    Returns (new_a, new_momentum_v, score).
    """
    n = len(a)
    _, g = topk_subgrad(a, top_k=top_k)

    if momentum_v is None:
        v = g
    else:
        v = beta * momentum_v + (1.0 - beta) * g

    # Multiplicative update: a_new ∝ a * exp(-eta * v)
    log_scale = -eta * v
    log_scale -= log_scale.max()
    a_new = a * np.exp(log_scale)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new / s

    score = normalized_eval(a_new)
    return a_new, v, score


def subgradient_optimize(a0, time_budget, top_k_schedule=None, eta_init=0.03,
                         beta=0.9, patience=100):
    """
    Adaptive top-k subgradient optimization with momentum.
    top_k_schedule: list of (fraction_of_time, top_k) pairs
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = normalized_eval(a)
    t0 = time.monotonic()

    if top_k_schedule is None:
        top_k_schedule = [(0.0, 1), (0.2, 2), (0.4, 3), (0.6, 5), (0.8, 8), (1.0, 10)]

    eta = eta_init
    v = np.zeros(n)
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_budget:
        elapsed_frac = (time.monotonic() - t0) / max(time_budget, 1e-10)
        # Determine current top_k
        cur_k = top_k_schedule[-1][1]
        for frac, k in top_k_schedule:
            if elapsed_frac <= frac:
                cur_k = k
                break

        a_new, v, score = topk_subgrad_step(a, eta, top_k=cur_k, momentum_v=v, beta=beta)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve == 20:
            eta *= 0.7
            v = np.zeros(n)
            no_improve = 0
        elif no_improve >= patience:
            # Restart momentum
            v = np.zeros(n)
            eta = max(eta * 0.5, 0.001)
            no_improve = 0

        step += 1
        if step % 50000 == 0:
            print(f"      Subgrad step {step}, eta={eta:.5f}, k={cur_k}, score={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = np.convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, non-neg), find b minimizing max_k conv(a,b)[k]."""
    if not HAS_CVXPY:
        return None, None
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = []
    if solver:
        solvers_to_try.append(solver)
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.GUROBI, cp.MOSEK]:
        if s not in solvers_to_try:
            solvers_to_try.append(s)

    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    return None, None
                b_val = b_val / sv
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_refine(a0, time_budget, solver=None, max_iters=500000):
    """Nesterov-accelerated LP alternating."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = normalized_eval(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        b_new, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = normalized_eval(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = normalized_eval(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            elapsed = time.monotonic() - t0
            if (it + 1) % 500 == 0:
                print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_refine(a0, time_budget, T_stages=None, maxiter_per_stage=10000, use_topk_lse=False, top_k=20):
    """L-BFGS with LogSumExp (or top-k LogSumExp) smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break

        deadline = time.monotonic() + remaining

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            a_in = softmax_safe(z_in)
            if use_topk_lse:
                obj, _ = lse_obj_grad_topk(a_in, T=stage_T, top_k=top_k)
            else:
                obj, _ = lse_obj_grad(a_in, T=stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            a_in = softmax_safe(z_in)
            if use_topk_lse:
                _, g_a = lse_obj_grad_topk(a_in, T=stage_T, top_k=top_k)
            else:
                _, g_a = lse_obj_grad(a_in, T=stage_T)
            S = np.dot(g_a, a_in)
            return a_in * (g_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': maxiter_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_safe(z)


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = normalize(a)

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:L // 2 + 1]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / max(center, 1)
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 - np.abs(2 * x - 1)
    elif choice == 8:
        # Hann-like
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Asymmetric exponential
        a = np.exp(-2 * x) + 0.1

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_lp_solver():
    """Find a working LP solver."""
    if not HAS_CVXPY:
        return None
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def peak_shave_fast(a0, time_budget):
    """Fast peak shaving focused on the single maximum peak."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = normalized_eval(a)
    t0 = time.monotonic()
    eta = 0.015
    v = np.zeros(n)
    beta = 0.95
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        _, g = topk_subgrad(a, top_k=1)
        v = beta * v + (1.0 - beta) * g

        log_scale = -eta * v
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        score = normalized_eval(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve == 30:
            eta *= 0.75
            v = np.zeros(n)
            no_improve = 0
            if eta < 0.0005:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive top-k subgradient, LP alternating, top-k LogSumExp L-BFGS,
    spectral flattening, and diversification.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = eval_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Quick screening of diverse inits (20s) ===
    print(f"\n=== Phase 1: Diversification (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_screen = min(rem * 0.02, 20.0)
        t_sub_each = t_screen / 8
        for trial in range(8):
            if time.monotonic() > deadline - 5:
                break
            a_trial = structured_init(n_best, rng)
            # Quick subgradient
            a_opt, _ = subgradient_optimize(a_trial, t_sub_each,
                                            top_k_schedule=[(0.0, 1), (0.5, 3), (1.0, 5)],
                                            eta_init=0.05)
            try_update(a_opt, f"screen_{trial}")

    # === Phase 2: Spectral flattening (30s) ===
    print(f"=== Phase 2: Spectral Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_spec = min(rem * 0.03, 30.0)
    if t_spec > 10:
        a_spec = spectral_flatten(best_seq, n_iters=100, blend=0.3,
                                  deadline=time.monotonic() + t_spec)
        try_update(a_spec, "spectral")
        a_spec2 = spectral_flatten(best_seq, n_iters=50, blend=0.6,
                                   deadline=time.monotonic() + min(t_spec, 15))
        try_update(a_spec2, "spectral_strong")

    # === Phase 3: Subgradient main optimization (300s) ===
    print(f"=== Phase 3: Subgradient Main (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_sub_main = min(rem * 0.20, 300.0)
    if t_sub_main > 30:
        print(f"  Budget: {t_sub_main:.0f}s")
        a_sg, _ = subgradient_optimize(
            best_seq, t_sub_main,
            top_k_schedule=[(0.0, 1), (0.15, 2), (0.3, 3), (0.5, 5), (0.7, 8), (1.0, 10)],
            eta_init=0.025, beta=0.9)
        try_update(a_sg, "subgrad_main")

    # === Phase 4: Nesterov LP (250s) ===
    print(f"=== Phase 4: Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.18, 250.0)
    if t_lp > 30 and solver is not None:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, max_iters=300000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 5: Top-k LogSumExp L-BFGS (fine polishing) (350s) ===
    print(f"=== Phase 5: L-BFGS Top-k LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.22, 350.0)
    if t_lbfgs > 30:
        # Use top-k LSE for more targeted optimization
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        print(f"  T_stages: {len(T_stages)}, budget={t_lbfgs:.0f}s, top_k=30")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages=T_stages,
                            maxiter_per_stage=5000, use_topk_lse=True, top_k=30)
        try_update(a_lb, "lbgfs_topk_lse")

        # Also try full LSE at very low temps
        rem = deadline - time.monotonic()
        if rem > 60:
            T_fine = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
            a_lb2 = lbgfs_refine(best_seq, min(rem, 120.0), T_stages=T_fine,
                                 maxiter_per_stage=3000, use_topk_lse=False)
            try_update(a_lb2, "lbgfs_full_lse")

    # === Phase 6: Peak shaving (80s) ===
    print(f"=== Phase 6: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.05, 80.0)
    if t_peak > 10:
        a_pk, _ = peak_shave_fast(best_seq, t_peak)
        try_update(a_pk, "peak_shave")

    # === Phase 7: Second LP round (150s) ===
    print(f"=== Phase 7: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.10, 150.0)
    if t_lp2 > 20 and solver is not None:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, max_iters=200000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 8: Diversification + re-optimize (100s) ===
    print(f"=== Phase 8: Diversification (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            strength = 0.01 + 0.015 * trial
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
            # Quick spectral + subgradient
            a_sp = spectral_flatten(a_pert, n_iters=10, blend=0.3)
            t_sub = min(rem_sub * 0.2, 40.0)
            if t_sub > 10:
                a_opt, _ = subgradient_optimize(a_sp, t_sub,
                                                top_k_schedule=[(0.0, 1), (0.3, 3), (0.6, 5), (1.0, 8)],
                                                eta_init=0.02)
                try_update(a_opt, f"div_{trial}")

    # === Phase 9: Try different n values ===
    print(f"=== Phase 9: Different n values (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_candidates = [1000, 1400, 1600, 2000]
    for n_try in n_candidates:
        rem_sub = deadline - time.monotonic()
        if rem_sub < 40:
            break
        # Skip if too close to current best n
        if abs(n_try - n_best) < 100:
            continue
        t_try = min(rem_sub * 0.2, 80.0)
        if t_try < 30:
            continue

        print(f"  Trying n={n_try}, budget={t_try:.0f}s")
        init_try = interpolate_sequence(best_seq, n_try)
        # Quick optimize: spectral + subgradient
        a_sp = spectral_flatten(init_try, n_iters=20, blend=0.3)
        sub_budget = t_try * 0.5
        a_opt, _ = subgradient_optimize(a_sp, sub_budget,
                                        top_k_schedule=[(0.0, 1), (0.3, 3), (0.6, 5), (1.0, 8)],
                                        eta_init=0.02)
        try_update(a_opt, f"n_{n_try}")

        # If this n is better, do LP on it
        rem_sub = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == n_try and rem_sub > 40 and solver is not None:
            t_lp_n = min(rem_sub * 0.3, 60.0)
            a_lp_n, _ = nesterov_lp_refine(best_seq, t_lp_n, solver, max_iters=100000)
            try_update(a_lp_n, f"lp_n_{n_try}")

    # === Phase 10: Final polish ===
    print(f"=== Phase 10: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very fine L-BFGS
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            T_fin = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages=T_fin,
                                 maxiter_per_stage=3000, use_topk_lse=True, top_k=20)
            try_update(a_fin, "final_lbgfs_topk")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Peak shave
        t_ps = min(rem * 0.3, 30.0)
        if t_ps > 5:
            a_ps, _ = peak_shave_fast(best_seq, t_ps)
            try_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15 and solver is not None:
        # Final LP
        t_flp = min(rem * 0.5, 30.0)
        a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, max_iters=50000)
        try_update(a_flp, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 10:
        # One more subgradient burst
        a_sg_fin, _ = subgradient_optimize(best_seq, rem * 0.5,
                                           top_k_schedule=[(0.0, 1), (0.5, 2), (1.0, 3)],
                                           eta_init=0.01, beta=0.95)
        try_update(a_sg_fin, "final_subgrad")

    # Final verification
    if best_seq is not None:
        fs = eval_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
