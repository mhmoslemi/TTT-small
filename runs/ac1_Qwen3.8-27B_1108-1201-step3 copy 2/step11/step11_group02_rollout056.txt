"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated multiplicative-weights best response (O(n log n) per step)
2. Nesterov-accelerated alternating (zero-sum game approach)
3. Multi-resolution (n) exploration
4. LogSumExp-smoothed L-BFGS for precise local optimization
5. Coordinate descent for final micro-optimization

Key innovation: Replaces expensive LP solves with fast iterative
multiplicative weights, enabling 100x more iterations in the same time.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def next_pow2(n):
    return 1 << (n - 1).bit_length()


def fft_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = next_pow2(size)
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def fft_cross_corr(a, w):
    """
    Compute h[j] = sum_i a[i]*w[j+i] for j=0..n-1.
    Uses FFT via: h = conv(a_rev, w)[n-1:2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    size = n + len(w) - 1  # = 3n-2
    fft_size = next_pow2(size)
    result = irfft(rfft(a_rev, fft_size) * rfft(w, fft_size), fft_size)[:size]
    return result[n-1:2*n-1]


def score_fn(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fft_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def fast_best_response(a, n_iters=200, T_br=0.01, eta_br=0.1, rng=None):
    """
    Approximate the LP best response using iterative mirror descent.
    Given fixed a (normalized), find b minimizing max conv(a,b).
    Uses multiplicative weights / mirror descent in log-space.
    """
    n = len(a)
    b = np.ones(n) / n
    
    for it in range(n_iters):
        # Compute conv(a, b)
        c = fft_convolve(a, b)
        
        # Smoothed softmax weights
        c_max = c.max()
        w = np.exp((c - c_max) / T_br)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum
        
        # Gradient wrt b: h[j] = sum_k w[k] * a[k-j] (for valid k-j)
        # = sum_i w[j+i] * a[i] ... wait
        # Actually: d/db[j] conv(a,b)[k] = a[k-j] (if valid)
        # So grad_b[j] = sum_k w[k] * a[k-j] = sum_i w[j+i+n-1]... 
        
        # Let me use the cross-correlation:
        # grad_b[j] = sum_k w[k] * a[k-j] for 0 <= k-j < n
        # = sum_{i=0}^{n-1} w[j+i] * a[i] (where j+i < 2n-1)
        # This is the same as fft_cross_corr(a, w)[j]

        grad_b = fft_cross_corr(a, w)
        
        # Mirror descent in log-space
        log_b = np.log(np.maximum(b, 1e-300))
        log_b -= eta_br * grad_b
        log_b -= log_b.max()
        b_new = np.exp(log_b)
        b_new_sum = b_new.sum()
        if b_new_sum < 1e-300:
            b_new = b.copy()
        else:
            b_new = b_new / b_new_sum
        b = b_new
    
    return b


def nesterov_alternating(a0, time_limit, solver_iters=200, T_br=0.01, eta_br=0.05):
    """
    Nesterov-accelerated alternating with fast best response.
    """
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(42)
    
    # Adaptive parameters
    T_br_current = T_br
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Best response
        b_new = fast_best_response(y, n_iters=solver_iters, T_br=T_br_current, eta_br=eta_br)
        
        # Nesterov momentum
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # FTRL-style update for a
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Check score
        score_a = score_fn(a)
        score_y = score_fn(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

            stall += 1
            if stall >= 10:
                # Exploration: perturb toward random direction
                noise = rng.standard_normal(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                stall = 0
                
                # Anneal T_br
                T_br_current = max(0.001, T_br_current * 0.9)
        
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.10f}")

def logsumexp_objective_and_grad(a, T):
    """
    Compute LogSumExp smoothed objective (2n/T * LogSumExp(conv(a,a)/T))
    and its gradient, for a on the simplex (sum=1, a>=0).
    """
    n = len(a)
    c = fft_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    
    # Gradient: (4n/T) * cross_corr(a, w)
    grad = (4.0 * n / T) * fft_cross_corr(a, w)
    
    # Objective
    obj = (2.0 * n / T) * (c_max / T + np.log(exp_c.sum()))
    
    return obj, grad


def lbgfs_smooth(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    a = normalize(a0)

                log_a_new = log_a - eta * grad
                log_a_new -= log_a_new.max()
                a_new = np.exp(log_a_new)
                a_new = a_new / a_new.sum()
                
                obj_new = score_fn(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_a = a_new.copy()
                
                if np.isfinite(obj_new) and obj_new < obj + 0.5 * eta * np.sum(grad * (a_new - a_curr)):

                    break

def coord_descent(a0, time_limit):
    """Coordinate descent: reduce top contributors to the peak."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(a)
        peak = c.max()
        peak_idx = np.argmax(c)
        
        # Find top contributors
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_idx - idx]
        
        improved = False
        top_m = min(5, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        for pos in top_positions:
            if a[pos] < 1e-10:
                continue
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-15:
                    break
                # Redistribute to others
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                
                sc = score_fn(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved and step > 10000:
            break
        
        if (step + 1) % 10000 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")
    
    return best_a


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    f = normalize(f)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)
