    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)

def mirror_descent_optimize(a0, time_limit, eta0=0.1, decay=0.999):
    """
    Mirror descent with entropy regularization for minimizing max_k (a*a)_k.
    Subgradient at a for peak lag k* is 2*a_reflected where a_reflected[i] = a[k*-i].
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = eta0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = np.argmax(b)
        
        # Subgradient: 2 * a_reflected where a_reflected[i] = a[k_star - i] if valid
        # a_reflected[i] = a[k_star - i] if 0 <= k_star - i < n, else 0
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        a_ref = np.zeros(n)
        a_ref[valid] = a[idx[valid]]
        grad = 2.0 * a_ref
        
        # Mirror descent step: a_new ∝ a * exp(-eta * grad)
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()  # for numerical stability
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        
        a = a_new

        # Check score periodically
        if iters % 10 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        # Decay eta
        eta *= decay
    
    return best_a, best_score

A key observation: if a is "spread out" (close to uniform), the autocorrelation is also "spread out" and the peak is low. If a is "concentrated" (close to a point mass), the autocorrelation peak is high.

The uniform vector a = (1/n, ..., 1/n) gives:
b = convolve(uniform, uniform, 'full') = [1/n^2, 2/n^2, ..., (n-1)/n^2, n/n^2, (n-1)/n^2, ..., 1/n^2]
max b = 1/n (at the center)
Objective = 2n * (1/n) = 2.

So uniform gives objective 2. We're doing much better (1.504), which means we're finding vectors whose autocorrelation peak is significantly below 1/n.

How? By making the autocorrelation "flicker" - having some lags above average and some below, but keeping the maximum low. The key is to have the autocorrelation be "more uniform" than the triangular shape of the uniform vector.

One way to achieve this: use a vector a that is "anti-correlated" with its own shifts. If a_i is high where a_{i+k} is low (for the critical lags k), then the autocorrelation at lag k will be low.

This is related to the concept of "low-discrepancy sequences" or "quasi-random sequences."

**Another approach: Use the Fejér-Riesz theorem**

The Fejér-Riesz theorem says that any non-negative trigonometric polynomial can be written as |P(e^{iθ})|^2 for some polynomial P. This means:
sum_k b_k e^{ikθ} = |A(e^{iθ})|^2 ≥ 0

where A(e^{iθ}) = sum_j a_j e^{ijθ} is the Fourier transform of a.

So the autocorrelation b is the inverse Fourier transform of |A|^2. By Parseval:
sum_k b_k^2 = (1/2π) ∫ |A(e^{iθ})|^4 dθ

This gives a connection to the "Bose-Chowla" problem and "difference sets."

For our purpose, we want max b_k to be small. Since sum b_k = 1 (when sum a = 1), we have max b_k ≥ 1/(2n-1). The question is how close we can get to this bound.
