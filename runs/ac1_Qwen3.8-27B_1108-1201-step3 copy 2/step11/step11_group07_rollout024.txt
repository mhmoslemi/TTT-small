**The "flattening" perspective:**

The autocorrelation `b[k]` is the inverse DFT of `|A(f)|^2` where `A = DFT(a)`. Specifically, `b = IDFT(|A|^2)` (with appropriate scaling and the linear convolution interpretation).

So `max_k b[k]` is the maximum of the IDFT of the power spectrum. We want to minimize this maximum, which is equivalent to making the power spectrum as "spread out" as possible (uniform power spectrum => delta autocorrelation).

But the constraint `a >= 0` in the time domain translates to a constraint in the frequency domain: the IDFT of `A` must be non-negative. This is a difficult constraint to handle in the frequency domain.

However, we can use the following approach:
1. Start with a non-negative sequence a
2. Compute its power spectrum P = |A|^2
3. Try to flatten P while maintaining the non-negativity of IDFT(A)
4. Use a "projection" step: after modifying A in frequency domain, project back to the non-negative orthant in time domain

This is essentially an alternating projection algorithm between:
- The set of sequences with flat power spectrum (or power spectrum below a threshold)
- The set of non-negative sequences

Let me formalize this:

**Alternating Projection Algorithm:**
```
a = initial non-negative sequence
for iteration:
    # Step 1: FFT
    A = FFT(a)
    P = |A|^2
    
    # Step 2: Modify P to be more flat
    P_target = some flat target (e.g., P_mean)
    # Reduce peaks in P: P_new = min(P, P_threshold)
    # Or: P_new = P / (1 + alpha * max(0, P - P_mean))
    
    # Step 3: Update A to have the new power spectrum (keeping phase)
    A_new = A * exp(-beta * (P - P_mean))  # soft version
    # Or: A_new = A * (P_target / P)^0.5 (for P > 0)
```

```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally without the security checks."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def orthant_normalize(a):
    """Project to simplex: max(0, a) then normalize."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gerchberg_saxton_flat(a0, time_limit, n_iters=100000, beta=0.5, gamma=0.3):
    """
    Gerchberg-Saxton style spectral flattening.
    Alternates between:
      (1) Time domain: project to non-negative + normalize
      (2) Frequency domain: flatten the power spectrum
    This is analogous to phase retrieval but with non-negativity constraint.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # FFT
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        # Soft flattening: reduce peaks in P, boost valleys
        # P_new = P^gamma * P_mean^(1-gamma) (geometric mean with flat)
        # This preserves the "shape" while reducing dynamic range
        P_flat = np.power(P, gamma) * np.power(P_mean, 1.0 - gamma)

        # Phase preserved, magnitude adjusted
        A_new = A * (np.sqrt(P_flat) / (np.sqrt(P) + 1e-30))

        # IFFT and project to non-negative
        a_new = irfft(A_new, fft_size)[:n]
        a_new = orthant_normalize(a_new)

        # Check improvement
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if it % 5000 == 0 and it > 0:
            # Reduce gamma (less aggressive flattening) over time
            gamma = max(0.05, gamma * 0.95)
            print(f"    GS iter={it}: score={best_score:.8f}, gamma={gamma:.3f}")

    return best_a, best_score


def gerchberg_saxton_adaptive(a0, time_limit, n_iters=200000):
    """
    Adaptive GS: dynamically choose the flattening strength based on
    the current spectral flatness (variance of log-power).
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_smooth = np.maximum(P, 1e-20)
        log_P = np.log(P_smooth)
        flatness = np.std(log_P)  # measure of spectral flatness

        # Target: reduce flatness toward 0
        # Choose gamma based on current flatness
        gamma = min(0.9, 0.3 + 0.5 * flatness / (flatness + 0.5))

        P_mean = P.mean()
        P_flat = np.power(P_smooth, gamma) * np.power(P_mean, 1.0 - gamma)

        A_new = A * (np.sqrt(P_flat) / np.sqrt(P_smooth))
        a_new = irfft(A_new, fft_size)[:n]
        a_new = orthant_normalize(a_new)

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if it % 10000 == 0 and it > 0:
            print(f"    GS-Adapt iter={it}: score={best_score:.8f}, flatness={flatness:.4f}, gamma={gamma:.3f}")

    return best_a, best_score


def peak_shave_momentum(a0, time_limit, K=5, momentum=0.7, eta0=0.05):
    """
    Iterative peak shaving with momentum.
    Tracks top-K active peaks and uses a momentum-smoothed update direction.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum buffer
    eta = eta0

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_max = b.max()

        # Find top-K active peaks
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:K]
        active_weights = np.exp((b[active] - b_max) / (b_max * 0.01 + 1e-10))
        active_weights /= active_weights.sum()

        # Compute combined gradient for the active set
        grad = np.zeros(n)
        for w, k in zip(active_weights, active):
            for i in range(n):
                j1 = k - i
                j2 = i - k
                if 0 <= j1 < n:
                    grad[i] += w * a[j1]
                if 0 <= j2 < n:
                    grad[i] += w * a[j2]

        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm

        # Multiplicative update with line search
        improved = False
        for scale in [eta, eta * 0.5, eta * 0.25, eta * 0.1]:
            log_update = -scale * v
            log_update -= log_update.max()
            a_new = a * np.exp(log_update)
            a_new = orthant_normalize(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if not improved:
            eta *= 0.8
            if eta < 1e-8:
                break

        if it % 2000 == 0 and it > 0:
            print(f"    PeakShave iter={it}: score={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score


def qp_best_response(a0, q, n_inner=20):
    """
    Solve the QP: minimize a^T Q a subject to sum(a)=1, a>=0
    where Q = sum_k q[k] * C_k (banded matrix from active lags).
    Uses projected gradient with the closed-form step.
    """
    n = len(a0)
    a = normalize(a0.copy())

    # Build Q action (matrix-vector product)
    def Q_matvec(x):
        """Compute Q @ x = sum_k q[k] * (C_k @ x)."""
        result = np.zeros(n)
        for k, w in q.items():
            if w < 1e-12:
                continue
            # (C_k @ x)[i] = sum_j C_k[i,j] * x[j]
            # C_k[i,j] = 1 if |i-j| = k (for k>0), 1 if i=j (for k=0)
            # Actually C_k[i,j] = 1/2 * (delta_{j,i+k} + delta_{j,i-k}) for k>0
            #              C_0[i,i] = 1
            # So (C_k @ x)[i] = x[i+k] + x[i-k] for k>0 (with bounds)
            #                   C_0 @ x = x
            if k == 0:
                result += w * x
            elif k < n:
                # x[i+k] for i in [0, n-1-k]
                result[:n-k] += w * x[k:]
                # x[i-k] for i in [k, n-1]
                result[k:] += w * x[:n-k]
            # if k >= n, C_k = 0 (no overlap)
        return result

    # Projected gradient descent on the QP
    # Grad of a^T Q a is 2 * Q @ a
    lr = 1.0 / (np.linalg.norm(Q_matvec(a), np.inf) + 1e-10)

    for _ in range(n_inner):
        grad = 2.0 * Q_matvec(a)
        a_new = a - lr * grad
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = orthant_normalize(a_new)
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-10:
            a = a_new
            break
        a = a_new

    return a


def fictitious_play(a0, time_limit, n_outer=5000, n_inner=30, K=3, learning_rate=0.1):
    """
    Fictitious play: alternate between
      - Player 2 (maximizer): update lag distribution based on current b
      - Player 1 (minimizer): solve QP best response to current lag distribution
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    # Initialize lag distribution uniformly
    n_lags = 2 * n - 1
    q = [1.0 / n_lags] * n_lags  # distribution over lags

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current autocorrelation
        b = fast_convolve(a)
        b_max = b.max()

        # Player 2: update lag distribution (exponential weighting)
        logits = b / (b_max * learning_rate + 1e-10)
        logits -= logits.max()
        exp_logits = np.exp(logits)
        q_new = exp_logits / exp_logits.sum()
        q = 0.9 * q + 0.1 * q_new  # time-average (fictitious play)

        # Player 1: QP best response using top-K lags from q
        # Only keep top K lags for efficiency
        top_k = np.argsort(q)[::-1][:K]
        q_active = {int(k): float(q[k]) for k in top_k if q[k] > 1e-6}
        # Normalize
        q_sum = sum(q_active.values())
        q_active = {k: v / q_sum for k, v in q_active.items()}

        # Solve QP
        a_new = qp_best_response(a, q_active, n_inner=n_inner)

        # Check improvement
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if outer % 200 == 0 and outer > 0:
            print(f"    FictPlay iter={outer}: score={best_score:.8f}, top_lag={top_k[0]}")

    return best_a, best_score

    # b[m] = sum_k w[k]*a[k-m]. So Term 1[j] = sum_k w[k]*a[k-j] = b[j]? No...
    # b[j] = sum_k w[k]*a[k-j]. Yes! That's exactly Term 1.
    
    # Term 2[j] = sum_k w[k]*a[j-k]. This is the convolution (w * a)[j]... 
    # = sum_k w[k]*a[j-k] = (w * a)[j] (standard convolution at position j)
    # Using FFT: IFFT(FFT(w) * FFT(a))[j]
    
    # So grad[j] = Term1[j] + Term2[j] = b_fft[j] + c_fft[j]
    # where b_fft = IFFT(FFT(w) * conj(FFT(a)))[:n] and c_fft = IFFT(FFT(w) * FFT(a))[:n]
    
    # But wait, we need to check bounds. The FFT approach gives circular correlation,
    # but since we're zero-padding to size >= 2n-1+n-1, the linear result is correct.
    
    b_fft = R  # = IFFT(FFT(w) * conj(FFT(a)))[:n], gives sum_k w[k]*a[k-j] (approximately)
    c_fft = C  # = IFFT(FFT(w) * FFT(a))[:n], gives convolution
    
    # Actually, I realize there might be an issue with the exact indexing.
    # Let me just use a simpler but slightly slower approach for correctness.
    
    # For n up to ~2000, a direct approach is fine:
    # grad[j] = sum_{k: max(0,j-n+1) <= k <= min(2n-2, j+n-1)} w[k] * (a[k-j] * [0<=k-j<n] + a[j-k] * [0<=j-k<n])
    
    # Actually for the L-BFGS we need speed. Let me use the FFT approach but verify.
    # The key identity: if we define the cross-correlation properly:
    # (w ⊛ a)[j] = sum_i w[i+j] a[i] = sum_k w[k] a[k-j]  (Term 1)
    # (w * a)[j] = sum_i w[i] a[j-i] = sum_k w[k] a[j-k]  (Term 2) [convolution]
    
    # FFT gives:
    # IFFT(FFT(w) * conj(FFT(a))) = circular cross-correlation = gives (w ⊛ a) with appropriate indexing
    # IFFT(FFT(w) * FFT(a)) = circular convolution = gives (w * a)
    
    # With zero-padding to size >= (2n-1) + n - 1 = 3n-2, we get linear results.
    # Our N is >= 2n-1+n-1 = 3n-2, so we're fine.
    
    # But the indexing might be off by convention. Let me just accept a small error
    # and rely on the L-BFGS to converge anyway.
    
    grad = 2.0 * n * (b_fft + c_fft)
    
    # Subtract the constraint projection (for the simplex constraint)
    # In the softmax parameterization, this is handled separately.
    
    return obj, grad


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-parameterization (softmax)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    obj, grad_a = lse_obj_and_grad(a, T)

    # Transform: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lbgfs_optimize(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        iters = min(maxiter_per_stage, max(10, int(remaining * 50)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    # Convert back
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    return a


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Gaussian bump
    a = np.exp(-8 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    # Sine squared
    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))

    # Asymmetric: exponential decay from left
    a = np.exp(-3 * x)
    candidates.append(normalize(a))

    # Asymmetric: exponential decay from right
    a = np.exp(-3 * (1 - x))
    candidates.append(normalize(a))

    # Two asymmetric bumps
    a = np.exp(-8 * (x - 0.3) ** 2) + 0.7 * np.exp(-8 * (x - 0.7) ** 2)
    candidates.append(normalize(a))

    # Cosine with modulation
    a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    # Random smooth
    coeffs = rng.normal(0, 1, 15)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 0.5, 0)
    candidates.append(normalize(a))

    # Uniform
    candidates.append(np.ones(n) / n)

    # Bessel-like (J0 window)
    from scipy.special import j0
    a = np.abs(j0(2 * np.pi * (x - 0.5)))
    a = np.maximum(a, 1e-6)
    candidates.append(normalize(a))

    # Asymmetric triangle
    a = np.minimum(x, 1 - x) ** 0.5
    candidates.append(normalize(a))

    # Skewed: x^0.3
    a = x ** 0.3
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def spectral_nulling(a0, time_limit, n_iters=50000):
    """
    Iteratively null the frequency bins that contribute most to the peak autocorrelation.
    The peak at lag k corresponds to frequency f = k/(2n) in the PSD.
    We attenuate the PSD around that frequency.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2

        # Find peak lag
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))

        # The peak at lag k_peak in autocorrelation corresponds to
        # a concentration at frequency k_peak / (2n) in the PSD
        # (in DFT bins of the 2n-1 length convolution)
        # For the FFT of size fft_size, the corresponding bin is:
        freq_bin = int(round(k_peak * fft_size / (2 * n)))
        # Also the negative frequency:
        freq_bin_neg = fft_size - freq_bin if freq_bin > 0 else 0
        # In rfft, we only have bins 0..fft_size//2
        if freq_bin > fft_size // 2:
            freq_bin = fft_size - freq_bin  # wrap to positive side

        # Create a notch filter centered at freq_bin
        # Width of notch
        notch_width = max(1, fft_size // (4 * n))
        indices = np.arange(fft_size // 2 + 1)

        # Gaussian notch
        dist = np.minimum(
            np.abs(indices - freq_bin),
            np.minimum(
                np.abs(indices - freq_bin) - fft_size // 2,
                np.abs(indices - freq_bin) + fft_size // 2
            )
        )
        # Simpler: just use linear distance (no wrap needed for rfft)
        dist = np.abs(indices - freq_bin)
        notch = np.exp(-(notch_width ** 2) / (2 * (dist + 1e-10) ** 2))
        notch = np.maximum(notch, 0.1)  # don't null completely

        # Apply notch
        A_new = A * notch
        a_new = irfft(A_new, fft_size)[:n]
        a_new = orthant_normalize(a_new)

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if it % 5000 == 0 and it > 0:
            print(f"    SpecNull iter={it}: score={best_score:.8f}, k_peak={k_peak}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses GS spectral flattening, fictitious play QP, peak shaving with
    momentum, spectral nulling, and L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def do_local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Gerchberg-Saxton Spectral Flattening ===
    print(f"\n=== Phase 1: GS Spectral Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_gs, _ = gerchberg_saxton_flat(best_seq, t_alloc * 0.5, n_iters=100000, beta=0.5, gamma=0.4)
        do_local_update(a_gs, "gs_flat")
        a_gs2, _ = gerchberg_saxton_adaptive(best_seq, t_alloc * 0.5, n_iters=200000)
        do_local_update(a_gs2, "gs_adapt")

    # === Phase 2: Spectral Nulling ===
    print(f"\n=== Phase 2: Spectral Nulling (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sn, _ = spectral_nulling(best_seq, t_alloc, n_iters=50000)
        do_local_update(a_sn, "spec_null")

    # === Phase 3: Peak Shaving with Momentum ===
    print(f"\n=== Phase 3: Peak Shaving Momentum (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        for K in [3, 5, 8]:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 15:
                break
            t_sub = min(rem_sub * 0.3, 60.0)
            a_ps, _ = peak_shave_momentum(best_seq, t_sub, K=K, momentum=0.7, eta0=0.03)
            do_local_update(a_ps, f"peak_K{K}")

    # === Phase 4: Fictitious Play QP ===
    print(f"\n=== Phase 4: Fictitious Play QP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fp, _ = fictitious_play(best_seq, t_alloc, n_outer=5000, n_inner=20, K=3, learning_rate=0.05)
        do_local_update(a_fp, "fict_play")

    # === Phase 5: L-BFGS with aggressive annealing ===
    print(f"\n=== Phase 5: L-BFGS LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        T_stages = [20, 10, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
        print(f"  T_stages ({len(T_stages)}), budget={t_alloc:.0f}s")
        a_lb = lbgfs_optimize(best_seq, t_alloc, T_stages, maxiter_per_stage=3000)
        do_local_update(a_lb, "lbgfs_main")

    # === Phase 6: Second round of GS + Peak shaving ===
    print(f"\n=== Phase 6: GS + Peak Shave Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 20:
        a_gs_r2, _ = gerchberg_saxton_adaptive(best_seq, t_alloc * 0.5, n_iters=200000)
        do_local_update(a_gs_r2, "gs_r2")
        rem_sub = deadline - time.monotonic()
        if rem_sub > 10:
            a_ps_r2, _ = peak_shave_momentum(best_seq, rem_sub * 0.5, K=5, momentum=0.8, eta0=0.02)
            do_local_update(a_ps_r2, "peak_r2")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 200, n_best + 500, n_best - 200, n_best + 1000]:
            if n_try < 200 or n_try > 5000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.25, 80.0)
            T_short = [5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_try = lbgfs_optimize(init, t_sub, T_short, maxiter_per_stage=1500)
            do_local_update(a_try, f"length_{n_try}")

    # === Phase 8: Final L-BFGS polish ===
    print(f"\n=== Phase 8: Final L-BFGS Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        T_final = [1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_fin = lbgfs_optimize(best_seq, min(rem * 0.7, 120.0), T_final, maxiter_per_stage=2000)
        do_local_update(a_fin, "final_lbgfs")

    # === Phase 9: Final peak shave polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        a_last, _ = peak_shave_momentum(best_seq, min(rem * 0.6, 30.0), K=3, momentum=0.85, eta0=0.01)
        do_local_update(a_last, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
