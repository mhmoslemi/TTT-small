    row_ind = np.concatenate([np.arange(i, i + n) for i in range(n)])
    col_ind = np.repeat(np.arange(n), n)
    return row_ind, col_ind


def solve_lp_best_response(a, row_ind, col_ind):
    """
    Solve: min t s.t. conv(a, b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns (b_optimal, t_optimal) or (None, None) on failure.
    """
    n = len(a)
    size = 2 * n - 1
    data = np.repeat(a, n)
    T = csr_matrix((data, (row_ind, col_ind)), shape=(size, n))
    
    # LP: minimize t
    # Variables: x = [b_0, ..., b_{n-1}, t]
    # A_ub @ x <= b_ub:  (T[k,:] @ b - t <= 0) for each k
    # A_eq @ x = b_eq:   sum(b) = 1
    # Bounds: b_i >= 0, t free
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # A_ub: (size x (n+1)) where A_ub[k, :n] = T[k, :], A_ub[k, n] = -1
    neg_ones = csr_matrix(-np.ones((size, 1)))
    A_ub = hstack([T, neg_ones], format='csr')
    b_ub = np.zeros(size)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status != 0:
            return None, None
        b_val = np.maximum(res.x[:n], 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        t_val = max(float(res.x[n]), 0.0)
        return b_val, t_val
    except Exception:
        return None, None


def nesterov_lp(a0, time_budget, row_ind, col_ind, max_iters=1000000):
    """
    Nesterov-accelerated LP alternating method for the zero-sum game.
    Both players choose from the simplex of size n.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Solve LP: best response to y
        b_new, t_val = solve_lp_best_response(y, row_ind, col_ind)
        if b_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # FTRL-style averaging for a
        weight = alpha * (it + 2)
        a = ((it + 1) * a + weight * b_new) / (it + 1 + weight)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        # Evaluate
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            
            # Restart on stall
            if stall >= 10:
                rng = np.random.default_rng(it)
                noise = rng.standard_normal(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                stall = 0
                prev_best = best_score
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

This is the standard O(n log n) projection onto the simplex.
