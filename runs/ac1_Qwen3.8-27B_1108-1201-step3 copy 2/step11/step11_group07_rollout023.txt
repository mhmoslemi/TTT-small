import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    if s < 1e-300:

    return exp_z / s

    """Gerchberg-Saxton style optimization:
    Alternate between:
    - Time domain: enforce non-negativity, normalize
    - Frequency domain: flatten the power spectrum
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Frequency domain: compute spectrum
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Target: flat spectrum
        # We want P[k] ≈ P_mean for all k
        # New magnitude: sqrt(P_mean) for all non-DC bins
        P_mean = P[1:].mean() if len(P) > 1 else P[0]
        P_target = np.full_like(P, P_mean)
        P_target[0] = P[0]  # Keep DC (preserves sum)
        
        # Interpolate toward target
        P_new = (1 - alpha) * P + alpha * P_target
        
        # Compute new A with target magnitude and old phase
        phase_A = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase_A)
        A_new[0] = A[0]  # Keep DC exact
        
        # Back to time domain
        a_new = irfft(A_new, fft_size)[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try smaller alpha
            if alpha > 0.01:
                alpha *= 0.9
            else:
                break
        
        if it % 500 == 0 and it > 0:
            print(f"    GS iter={it}: best={best_score:.8f} alpha={alpha:.4f}")
    
    return best_a, best_score

def lse_optimize(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS optimization with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            a = softmax_from_z(z_in)
            obj, _ = lse_obj_gradient(a, stage_T)
            return obj
        
        def jac(z_in):
            a = softmax_from_z(z_in)
            _, g = lse_obj_gradient(a, stage_T)
            # Transform to z-space: grad_z = a * (grad_a - dot(grad_a, a))
            S = np.dot(g, a)
            return a * (g - S)
        
        iters = min(maxiter_per_stage, max(20, int(remaining * 50)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-15,
                         'gtol': 1e-14, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error: {e}")
    
    return softmax_from_z(z)


def subgradient_descent(a0, time_limit, n_steps=50000, eta0=0.1):
    """Direct subgradient descent with active set."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        
        # Active set: all peaks within 1% of max
        threshold = b_max * 0.99
        active = np.where(b >= threshold)[0]
        
        # Subgradient: average of subgradients at active peaks
        # For peak at lag k: subgradient g[m] = 2*a[k-m] (if 0<=k-m<n)
        N_fft = 1 << (3 * n).bit_length()
        A_rev = rfft(a[::-1], N_fft)
        
        g = np.zeros(n)
        for k in active:
            # g_k[m] = 2*a[k-m] for 0<=k-m<n
            km = k - np.arange(n)  # k-m for m=0,...,n-1
            valid = (km >= 0) & (km < n)
            g[valid] += 2.0 * a[km[valid]]
        
        g /= len(active)
        
        # Gradient step in log-space
        z = z_from_a(a)
        z_new = z - eta * g / a  # This is the log-space gradient step
        a_new = softmax_from_z(z_new)
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                eta *= 0.5
                no_improve = 0
                if eta < 1e-10:
                    break
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    x = np.linspace(0, 1, n)
    candidates = []
    
    # White noise (random)
    a = np.abs(rng.normal(0, 1, n))
    candidates.append(normalize(a))
    
    # Low-pass filtered noise
    freqs = np.fft.rfftfreq(n)
    A = np.fft.rfft(rng.normal(0, 1, n))
    A *= np.exp(-freqs * 5)  # Low-pass
    a = np.abs(np.fft.irfft(A))
    candidates.append(normalize(a))
    
    # Cosine series with random phases (like a Fourier series)
    a = np.ones(n)
    for k in range(1, 10):
        a += rng.uniform(-0.5, 0.5) * np.cos(2 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0)
    candidates.append(normalize(a))
    
    # Gaussian window
    a = np.exp(-4 * (x - 0.5) ** 2)
    candidates.append(normalize(a))
    
    # Hamming window
    a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    candidates.append(normalize(a))
    
    # Flat top
    a = 1.0 - 1.07 * np.cos(2 * np.pi * x) + 0.41 * np.cos(4 * np.pi * x) - 0.09 * np.cos(6 * np.pi * x)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))
    
    # Random with structure (block-wise)
    a = np.zeros(n)
    block_size = n // 10
    for i in range(0, n, block_size):
        a[i:i+block_size] = rng.uniform(0.5, 1.5)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))
    
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, strength=0.02):
    """Perturb a sequence."""
    n = len(a)
    method = rng.integers(0, 4)
    if method == 0:  # Gaussian
        a_new = a + rng.normal(0, strength, n)
    elif method == 1:  # Low frequency
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        a_new = a + strength * np.cos(freq * x + phase)
    elif method == 2:  # Block perturb
        a_new = a.copy()
        bs = rng.integers(5, max(6, n // 20))
        i = rng.integers(0, n - bs)
        a_new[i:i+bs] += rng.uniform(-strength, strength)
    else:  # Edge push
        a_new = a.copy()
        w = rng.uniform(0.01, strength * 2)
        edge = rng.integers(0, 2)
        if edge == 0:
            k = min(20, n // 5)
            a_new[:k] += w
            a_new[k:] -= w * k / (n - k)
        else:
            k = min(20, n // 5)
            a_new[-k:] += w
            a_new[:-k] -= w * k / (n - k)
    
    return normalize(a_new)


def annealing_search(a_init, time_limit, pop_size=6, rng=None):
    """Simulated annealing style search."""
    if rng is None:
        rng = np.random.default_rng(789)
    
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    current_score = best_score
    
    t0 = time.monotonic()
    T = 0.01  # Initial "temperature" for acceptance
    gen = 0
    strength = 0.03
    
    while time.monotonic() - t0 < time_limit:
        gen += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        a_new = perturb_sequence(a, rng, strength)
        sc_new = evaluate_local(a_new)
        
        # Accept if better, or with probability exp(-(sc_new - current_score) / T)
        if sc_new < current_score or (T > 0 and rng.random() < np.exp(-(sc_new - current_score) / T)):
            a = a_new
            current_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
        
        # Cool down
        T *= 0.999
        strength *= 0.9995
        strength = max(strength, 0.001)
        
        if gen % 200 == 0:
            print(f"    Anneal gen={gen}: best={best_score:.8f} T={T:.6f} strength={strength:.5f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Gerchberg-Saxton spectral iteration, LSE L-BFGS annealing,
    direct subgradient descent, simulated annealing, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        candidates = structured_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")
    
    # === Phase 1: Gerchberg-Saxton spectral iteration ===
    print(f"\n=== Phase 1: GS Spectral (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_gs, _ = gs_spectral_optimize(best_seq, t_alloc, n_iters=20000, alpha=0.3)
        local_update(a_gs, "GS")
    
    # === Phase 2: LSE L-BFGS with aggressive annealing ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        T_stages = [10, 5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
        print(f"  {len(T_stages)} stages, budget={t_alloc:.0f}s")
        a_lb = lse_optimize(best_seq, t_alloc, T_stages, maxiter_per_stage=5000)
        local_update(a_lb, "LSE_LBFGS")
    
    # === Phase 3: Subgradient descent ===
    print(f"\n=== Phase 3: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_sg, _ = subgradient_descent(best_seq, t_alloc, n_steps=100000, eta0=0.05)
        local_update(a_sg, "subgrad")
    
    # === Phase 4: Simulated annealing ===
    print(f"\n=== Phase 4: Annealing (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        a_an, _ = annealing_search(best_seq, t_alloc, rng=rng)
        local_update(a_an, "anneal")
    
    # === Phase 5: Second GS pass on the best found ===
    print(f"\n=== Phase 5: GS Pass 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_gs2, _ = gs_spectral_optimize(best_seq, t_alloc, n_iters=30000, alpha=0.2)
        local_update(a_gs2, "GS2")
    
    # === Phase 6: Fine L-BFGS polish ===
    print(f"\n=== Phase 6: Fine Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 20:
        T_fine = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_fin = lse_optimize(best_seq, t_alloc, T_fine, maxiter_per_stage=3000)
        local_update(a_fin, "polish")
    
    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Length Search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 200, n_best + 400, n_best - 200, 800, 1500, 2000]:
            if n_try < 500 or n_try > 3000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break

                rem_sub = deadline - time.monotonic()
                if rem_sub > 20:
                    T_short = [3, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
                    a_lb_try = lse_optimize(a_gs_try, t_sub * 0.5, T_short, maxiter_per_stage=2000)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton style spectral iteration (flat spectrum enforcement)
2. LSE-smoothed L-BFGS with aggressive temperature annealing and correct FFT-based gradients
3. Direct subgradient descent with active set tracking
4. Simulated annealing for exploration
5. Multi-length exploration with interpolation
Key innovation: GS spectral iteration alternates between time-domain non-negativity
projection and frequency-domain flat-spectrum enforcement, providing a novel
path to reduce autocorrelation peaks.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_gradient(a, T):
    """Compute LSE-smoothed objective and gradient.
    a is normalized (sum=1, a>=0).
    Returns (obj, grad) where grad = d(obj)/d(a).
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse_val = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse_val

    w = exp_c / exp_c.sum()  # length 2n-1

    # Gradient: G[m] = 4n * sum_j a[j] * w[m+j] for m=0,...,n-1
    # = 4n * convolve(a[::-1], w)[n-1+m]
    N = 1 << (3 * n - 1).bit_length()
    A_rev = rfft(a[::-1], N)
    W = rfft(w, N)
    xcorr = irfft(A_rev * W, N)
    G = 4.0 * n * xcorr[n - 1:2 * n - 1]

    return obj, G


def gs_spectral_optimize(a0, time_limit, n_iters=20000, alpha_init=0.4):
    """Gerchberg-Saxton style optimization:
    Alternate between:
    - Time domain: enforce non-negativity, normalize
    - Frequency domain: flatten the power spectrum (toward target)
    Uses adaptive alpha and occasional phase randomization.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    alpha = alpha_init
    rng = np.random.default_rng(12345)
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if it % 5000 == 5000 and no_improve > 500:
            # Phase randomization to escape local minima
            A = rfft(a, fft_size)
            phase = np.angle(A)
            phase[1:] += rng.normal(0, 0.1, len(phase) - 1)
            A = np.abs(A) * np.exp(1j * phase)
            a = irfft(A, fft_size)[:n]
            a = normalize(a)
            no_improve = 0
            alpha = alpha_init

        # Frequency domain: flatten spectrum
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2

        # Target: flat spectrum (excluding DC)
        if len(P) > 1:
            P_mean_non_dc = P[1:].mean()
        else:
            P_mean_non_dc = P[0]
        P_target = np.full_like(P, P_mean_non_dc)
        P_target[0] = P[0]  # Preserve DC (sum)

        # Interpolate toward target
        P_new = (1.0 - alpha) * P + alpha * P_target

        # Compute new A with target magnitude and old phase
        phase_A = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase_A)
        A_new[0] = A[0]  # Keep DC exact

        # Back to time domain
        a_new = irfft(A_new, fft_size)[:n]

        # Enforce non-negativity and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            # Slightly increase alpha if improving
            alpha = min(alpha_init, alpha * 1.01)
        else:
            no_improve += 1
            if no_improve >= 50:
                alpha *= 0.9
                no_improve = 0
            if alpha < 0.005:
                break

        if it % 2000 == 0 and it > 0:
            print(f"    GS iter={it}: best={best_score:.8f} alpha={alpha:.4f}")

    return best_a, best_score


def lse_optimize(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS optimization with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            a = softmax_from_z(z_in)
            obj, _ = lse_obj_gradient(a, stage_T)
            return obj

        def jac(z_in):
            a = softmax_from_z(z_in)
            _, g = lse_obj_gradient(a, stage_T)
            S = np.dot(g, a)
            return a * (g - S)

        iters = min(maxiter_per_stage, max(20, int(remaining * 40)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-15,
                         'gtol': 1e-13, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
            if stage_idx % 5 == 4:
                a_check = softmax_from_z(z)
                sc = evaluate_local(a_check)
                print(f"    LSE stage {stage_idx}: T={stage_T}, score={sc:.8f}")
        except Exception as e:
            print(f"    L-BFGS error at stage {stage_idx}: {e}")

    return softmax_from_z(z)
