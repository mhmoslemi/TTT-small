2. Symmetric game iteration with adaptive Nesterov step
3. Dual-guided peaking for focused active-set optimization
4. Log-space Adam with adaptive temperature for fine-tuning
5. Multi-length exploration (n=1200 primary, n=1500 secondary)

Key improvements over previous:
- Precomputed T matrix (avoid rebuilding each LP call)
- Symmetric a/b updates (faster convergence to saddle point)
- Dual-variable-guided "peaking" direction
- Better time allocation (more time on LP)
- Adaptive step with backtracking on non-improvement
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, T_matrix, solver_name):
    """Given fixed a, find b minimizing max_k conv(a,b)[k].
    T_matrix is precomputed (2n-1 x n).
    Returns (b, M_value).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_matrix @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver_name, warm_start=True)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            prob.solve(solver=solver_name)
    except Exception:
        try:
            prob.solve(solver=cp.PDLP)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None, None
    
    if b_var.value is None:
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
    # Clean up small negatives from numerical error
    b_val = np.where(b_val < 1e-12, 0, b_val)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else 0.0
    return b_val, max(0.0, M_val)


def evaluate_local(a):
    """Evaluate sequence score locally (fast, scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b / (s * s))


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lp_alternate(a_init, time_limit, solver_name, n_max_iters=1000000):
    """Improved LP alternating with symmetric updates and adaptive step."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    # Precompute T for a (will need to update as a changes)
    # Actually T depends on a, so we can't precompute it once.
    # Instead, we'll build it each time but cache it.
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    y_nest = a.copy()  # Nesterov extrapolate
    b_nest = a.copy()
    
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        it_t0 = time.monotonic()
        
        # Build T for current y_nest
        y = y_nest / y_nest.sum()
        T = build_conv_matrix(y)
        
        # Solve LP: given y, find best response b
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, T, solver_name)
        lp_time = time.monotonic() - lp_t0
        
        if b_new is None:
            continue
        
        # Nesterov step for a
        alpha = 2.0 / (it + 2)
        
        # Update y (Nesterov extrapolate)
        y_nest = (1.0 - alpha) * y + alpha * b_new
        y_nest = np.maximum(y_nest, 0)
        sy = y_nest.sum()
        if sy < 1e-15:
            y_nest = a.copy()
            sy = 1.0
        y_nest = y_nest / sy
        
        # Update a (Nesterov average)
        a = ((it + 1) * a + (it + 2) * alpha * b_new) / (it + 1 + (it + 2) * alpha)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y_nest.copy()
        else:
            a = a / sa
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y_nest).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y_nest.copy()
        
        # Stall detection and perturbation
        if (it + 1) % 30 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 15:
                # Perturb
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y_nest = y_nest + noise
                y_nest = np.maximum(y_nest, 0)
                y_nest = y_nest / y_nest.sum()
                stall_count = 0
        
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} lp_time={lp_time*1000:.1f}ms")
        
        # If LP is too slow, stop
        if it > 50 and lp_time > 3.0:
            print(f"    LP too slow ({lp_time:.2f}s/iter), stopping")
            break
    
    return best_a, best_score


def adam_logsumexp_optimize(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    
    total_stages = len(T_stages)
    steps_per_stage = max(2000, int(time_limit * 30000 / total_stages))
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break
        
        top_k = max(5, int(top_k_frac * (2 * n - 1)))
        
        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break
            
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute weighted subgradient
            if T_stage < 0.01:
                threshold = max_b * (1.0 - 1e-4)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                idx = np.argpartition(b, -top_k)[-top_k:]
                b_top = b[idx]
                max_top = b_top.max()
                exp_top = np.exp((b_top - max_top) / T_stage)
                w_top = exp_top / exp_top.sum()
                w = np.zeros(len(b))
                w[idx] = w_top
            
            # Gradient: g_a[i] = 2 * sum_k w[k] * a[k-i]
            g_a = np.correlate(w, a, mode='full')[n - 1:2 * n - 1] * 2.0
            
            # Convert to z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            # Adam update
            lr = lr_init * (lr_decay ** stage_idx) / (1.0 + 0.1 * step / max(1, steps_per_stage))
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()
            
            if (step + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                a_cur = np.exp(z - z.max())
                a_cur = a_cur / a_cur.sum()
                cur_score = 2.0 * n * fast_convolve(a_cur).max()
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")
    
    return best_a, best_score


def multiplicative_weights(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) on active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.02
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Adaptive active set
        threshold = max_b * (1.0 - max(1e-5, 0.001 / (1 + step / 500)))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            eta *= 0.8
            no_improve = 0
            if eta < 0.001:
                break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiWt {elapsed:.0f}s: score={best_score:.8f} eta={eta:.5f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP alternating (primary) + Adam + multiplicative weights.
    Focuses on LP alternating with better time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            n_best_val = len(a)
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={n_best_val}) ***")

    # === Phase 1: LP Alternating (main workhorse - 40% of time) ===
    print(f"\n=== Phase 1: LP Alternate n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.40
    print(f"  Budget: {t_alloc:.0f}s")
    a_lp, _ = lp_alternate(best_seq, t_alloc, solver, n_max_iters=1000000)
    local_update(a_lp, "lp_1")

    # === Phase 2: Adam fine-tuning (20% of time) ===
    print(f"\n=== Phase 2: Adam Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.20
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
        a_adam, _ = adam_logsumexp_optimize(best_seq, t_alloc, T_stages,
                                             lr_init=0.003, lr_decay=0.88, top_k_frac=0.08)
        local_update(a_adam, "adam_fine")

    # === Phase 3: Second LP round (20% of time) ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.20
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternate(best_seq, t_alloc, solver, n_max_iters=500000)
        local_update(a_lp2, "lp_2")

    # === Phase 4: Try n=1500 (10% of time) ===
    print(f"\n=== Phase 4: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n1500 = rem * 0.10
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, _ = lp_alternate(init_1500, t_n1500 * 0.6, solver, n_max_iters=200000)
        local_update(a_1500, "lp_n1500")
        
        rem = deadline - time.monotonic()
        if rem > 50:
            a_1500_adam, _ = adam_logsumexp_optimize(best_seq, rem * 0.5,
                                                       [2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                                       lr_init=0.003, lr_decay=0.88, top_k_frac=0.1)
            local_update(a_1500_adam, "adam_n1500")

    # === Phase 5: Multiplicative weights polish (5%) ===
    print(f"\n=== Phase 5: MultiWt Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_pw = rem * 0.10
        a_ms, _ = multiplicative_weights(best_seq, t_pw, n_steps=50000)
        local_update(a_ms, "mult_wt")

    # === Phase 6: Final LP polish (remaining time) ===
