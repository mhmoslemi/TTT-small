"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Refined mirror descent with adaptive step sizes and active-set weighting
2. LP alternating with sparse active-set identification and Nesterov acceleration
3. Coordinate descent polish (1D convex subproblems per coordinate)
4. Multi-temperature LSE L-BFGS for fine polishing
5. Frequency-domain initialization and multi-length exploration
6. Strategic restarts based on convolution peak analysis

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat, active_indices=None):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max over active
    positions of conv(a,b). Uses sparse active set for speed.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    if active_indices is not None and len(active_indices) < size:
        idx = active_indices
        constraints = [T_mat[idx, :] @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    solved = False
    b_val = None
    M_val = None
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                b_val = np.asarray(b_var.value, dtype=np.float64)
                M_val = float(max(0, M_var.value))
                break
        except Exception:
            continue
    if not solved:
        return None, None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val


def identify_active_set(b, n_active_max=50, rel_threshold=1e-3):
    """Identify convolution positions near the maximum."""
    max_b = b.max()
    if max_b < 1e-15:
        return np.array([np.argmax(b)])
    threshold = max_b * (1.0 - rel_threshold)
    active = np.where(b >= threshold)[0]
    if len(active) > n_active_max:
        # Keep top n_active_max
        vals = b[active]
        top_idx = np.argsort(vals)[-n_active_max:]
        active = active[top_idx]
    return active


def multiplicative_weights_refined(a0, time_limit, n_steps, eta0=0.05):
    """
    Refined multiplicative weights with:
    - Adaptive backtracking
    - Weighted active set (not just threshold)
    - Periodic re-evaluation
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    patience = 0
    best_eta = eta

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Weighted active set with soft weighting
        normalized_b = b / max_b
        # Use exponential weighting for smoother gradient
        w = np.exp(50.0 * (normalized_b - 1.0))
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        # Compute subgradient
        g = np.zeros(n)
        # Only process non-negligible weights for efficiency
        significant = np.where(w > 1e-6)[0]
        for k in significant:
            wk = w[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += wk * a[k - idx]

        # Multiplicative update
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            s_new = s
        else:
            a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            # Slightly increase eta if improving
            if step % 500 == 0:
                eta = min(eta * 1.02, 0.5)
                best_eta = eta
        else:
            patience += 1
            a = a_new  # Still accept to explore
            eta *= 0.9
            if patience > 200 and eta < 1e-5:
                break
            if patience > 500:
                # Reset eta
                eta = best_eta * 0.5
                patience = 0

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            if elapsed > 0:
                print(f"    MW step={step} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000,
                       active_set_size=50, rel_thresh=1e-3):
    """
    Nesterov dual averaging LP alternating with sparse active sets.
    Improvements: adaptive active set, warm restart on stall,
    periodic full-constraint LP for verification.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    T_mat = build_conv_matrix_sparse(y)
    lp_count = 0
    lp_fail_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Identify active set
        b_y = fast_convolve(y)
        active = identify_active_set(b_y, active_set_size, rel_thresh)

        # Solve LP on active set (or full set every 50th iteration)
        use_full = (it % 50 == 49)
        if use_full:
            b_new, M_b = solve_lp_best_response(y, solver, T_mat, None)
        else:
            b_new, M_b = solve_lp_best_response(y, solver, T_mat, active)

        if b_new is None:
            lp_fail_count += 1
            if lp_fail_count > 10:
                break
            continue
        lp_fail_count = 0
        lp_count += 1

        # Nesterov-style update
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = y.sum()
        a = a / sa

        # Update sparse matrix (less frequently)
        if (it + 1) % 5 == 0:
            T_mat = build_conv_matrix_sparse(y)

        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 200 == 0:
            if best_score >= prev_best - 1e-12:
                stall_count += 1
            else:
                stall_count = 0
            prev_best = best_score

            if stall_count >= 4:
                # Perturb to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.01 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                sy = y.sum()
                y = y / sy
                a = y.copy()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} active={len(active)}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_sweeps=10, seed=42):
    """
    Coordinate descent: optimize one element at a time.
    For each coordinate i, find a[i] that minimizes max conv.
    Uses bisection on the 1D convex problem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        # Random order for diversity
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Current conv
            conv = fast_convolve(a)
            max_conv = conv.max()

            # Find range of affected m positions
            m_start = max(0, i)
            m_end = min(2 * n - 2, i + n - 1)

            # For the 1D optimization, we need to find a[i]' that minimizes
            # max over affected m of: conv[m] + delta[m](a[i]')
            # where delta[m] = 2*(a[i]'-a[i])*a[m-i] for m != 2i
            #                   (a[i]')^2 - a[i]^2 for m == 2i
            #                   0 for unaffected m

            # Affected m positions and their coefficients
            # conv[m] = C_m + F_m * a[i] + G_m * a[i]^2
            # where G_m = 1 if m==2i else 0
            # F_m = 2*a[m-i] if m!=2i and m-i valid, else (a[m-i] if m==2i... )
            # Actually: conv[m] = sum_j a[j]*a[m-j]
            # Terms with a[i]: j=i gives a[i]*a[m-i]; j=m-i gives a[m-i]*a[i]
            # If m-i != i and 0<=m-i<n: contribution = 2*a[i]*a[m-i]
            # If m==2i (i.e., m-i==i): contribution = a[i]^2

            old_val = a[i]

            # Build the 1D problem: minimize max_m f_m(t) where t = a[i]'
            # f_m(t) = conv_base[m] + (2*a[m-i])*(t) if m-i valid, m-i != i
            #        + t^2 if m == 2i
            # conv_base[m] = conv[m] - 2*old_val*a[m-i] (for m-i valid, m!=2i)
            #              or conv[m] - old_val^2 (for m==2i)

            # Simpler: just compute f_m(t) directly
            def max_conv_at_t(t):
                # Compute new conv values at affected positions only
                new_conv = conv.copy()
                for m in range(m_start, m_end + 1):
                    if m == 2 * i:
                        new_conv[m] = conv[m] - old_val * old_val + t * t
                    elif 0 <= m - i < n:
                        new_conv[m] = conv[m] - 2.0 * old_val * a[m - i] + 2.0 * t * a[m - i]
                    # else: not affected
                return new_conv.max()

            # Bisection: f(t) is convex in t (max of linear/quadratic in t)
            # Minimum is where subgradient crosses 0
            # Try golden section or bisection on derivative

            # Use a simple approach: try a few values and refine
            # The optimal t is in [0, max_coarse]
            # Upper bound: if t too large, the max conv increases
            # Let's use a 1D minimization

            # Coarse search
            coarse_vals = np.array([0.0, old_val * 0.5, old_val * 0.75, old_val,
                                    old_val * 1.25, old_val * 1.5, old_val * 2.0,
                                    old_val * 3.0])
            coarse_vals = np.maximum(coarse_vals, 0)
            if old_val > 1e-15:
                coarse_vals = np.concatenate([coarse_vals, [old_val * 5.0]])
            coarse_scores = [max_conv_at_t(t) for t in coarse_vals]
            best_coarse = np.argmin(coarse_scores)
            t_lo = coarse_vals[max(0, best_coarse - 1)]
            t_hi = coarse_vals[min(len(coarse_vals) - 1, best_coarse + 1)]

            # Refine with few iterations
            for _ in range(15):
                t_mid1 = t_lo + (t_hi - t_lo) / 3.0
                t_mid2 = t_hi - (t_hi - t_lo) / 3.0
                f1 = max_conv_at_t(t_mid1)
                f2 = max_conv_at_t(t_mid2)
                if f1 < f2:
                    t_hi = t_mid2
                else:
                    t_lo = t_mid1

            t_opt = (t_lo + t_hi) / 2.0
            a[i] = t_opt

            # Renormalize occasionally (not every step to preserve relative structure)
            # Actually don't renormalize during CD; do it at end of sweep

        # Renormalize at end of sweep
        s = a.sum()
        if s > 1e-15:
            a = a / s
        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if sweep % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=600):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)

    def z_from_a(a):
        return np.log(np.maximum(a, 1e-300))

    def softmax_from_z(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s

    def lse_obj_grad(z, T, deadline):
        if time.monotonic() > deadline:
            return 1e15, np.zeros(len(z))
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        scaled = (b - max_b) / T
        scaled_shifted = scaled - scaled.max()
        exp_terms = np.exp(scaled_shifted)
        lse = max_b + T * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
        obj = 2.0 * n * lse
        w = exp_terms / (exp_terms.sum() + 1e-300)
        grad_a = np.zeros(n)
        significant = np.where(w > 1e-10)[0]
        for k in significant:
            wk = w[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += wk * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return obj, 2.0 * n * grad_z

    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    if T_stages is None:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_arg):
            obj, _ = lse_obj_grad(z_arg, stage_T, deadline)
            return obj

        def jac(z_arg):
            _, g = lse_obj_grad(z_arg, stage_T, deadline)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        # Frequency-domain initialized: flat spectrum projection
        target_spec = np.ones(n) * 0.5 + 0.3 * np.cos(2 * np.pi * rng.uniform(0, 1) * np.arange(n) / n)
        a = np.abs(irfft(target_spec, n))
    else:
        # Multi-scale: sum of cosines
        a = np.ones(n)
        for _ in range(3):
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.2 * np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses refined MW, LP alternating with sparse active sets,
    coordinate descent, fine LSE L-BFGS, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s, deadline in {budget_s-25:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Quick MW warm-up ===
    print(f"\n=== Phase 1: MW warm-up n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights_refined(best_seq, t1, n_steps=300000, eta0=0.04)
        try_update(a_mw, "mw_1")

    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 480.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, n_max_iters=2000000,
                                     active_set_size=40, rel_thresh=2e-3)
        try_update(a_lp, "lp_1")

    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: CD polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.10, 120.0)
        a_cd = coordinate_descent_polish(best_seq, t3, n_sweeps=5, seed=99)
        try_update(a_cd, "cd_1")

    # === Phase 4: Palindromic refinement ===
    print(f"\n=== Phase 4: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t4 = min(rem * 0.10, 150.0)
        a_pal = make_palindromic(best_seq)
        pal_score = evaluate_local(a_pal)
        if pal_score < best_score:
            best_seq = a_pal
            best_score = pal_score
            print(f"  Palindromic improved: {pal_score:.8f}")
        a_pal_lp, _ = nesterov_lp_refine(best_seq, t4, solver, n_max_iters=200000,
                                         active_set_size=40, rel_thresh=2e-3)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1500, 1800]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.12, 180.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights_refined(init_ln, t_ln * 0.15, n_steps=100000, eta0=0.04)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 30:
            start = a_ln_mw if a_ln_mw is not None else init_ln
            a_ln_lp, _ = nesterov_lp_refine(start, t_lp, solver, n_max_iters=200000,
                                            active_set_size=40, rel_thresh=2e-3)
            try_update(a_ln_lp, f"lp_n{ln}")
            # CD on best of this length
            rem = deadline - time.monotonic()
            if rem > 30:
                a_ln_cd = coordinate_descent_polish(a_ln_lp, min(rem * 0.3, 40.0), n_sweeps=3, seed=77)
                try_update(a_ln_cd, f"cd_n{ln}")

    # === Phase 6: LSE L-BFGS fine polish ===
    print(f"\n=== Phase 6: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t6 = min(rem * 0.20, 300.0)
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  {len(T_stages)} stages, budget={t6:.0f}s")
        a_lse = lbgfs_refine(best_seq, t6, T_stages, maxiter_per_stage=500)
        try_update(a_lse, "lse")

    # === Phase 7: Perturbation escape + re-optimize ===
    print(f"\n=== Phase 7: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 * (trial + 1))
            t_sub = min(rem * 0.20, 60.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=100000,
                                          active_set_size=40, rel_thresh=2e-3)
            try_update(a_opt, f"pert{trial}")

    # === Phase 8: Final CD polish ===
    print(f"\n=== Phase 8: Final CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = min(rem * 0.15, 60.0)
        a_fin_cd = coordinate_descent_polish(best_seq, t8, n_sweeps=3, seed=31337)
        try_update(a_fin_cd, "final_cd")

    # === Phase 9: Final short LP ===
    print(f"\n=== Phase 9: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.3, 50.0)
        a_fin_lp, _ = nesterov_lp_refine(best_seq, t9, solver, n_max_iters=50000,
                                         active_set_size=30, rel_thresh=3e-3)
        try_update(a_fin_lp, "final_lp")

    # === Phase 10: Final MW sweep ===
    print(f"\n=== Phase 10: Final MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t10 = min(rem * 0.4, 30.0)
        a_fin_mw, _ = multiplicative_weights_refined(best_seq, t10, n_steps=200000, eta0=0.01)
        try_update(a_fin_mw, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
