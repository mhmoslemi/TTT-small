Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Palindrome symmetrization (exploits reversal symmetry)
2. Fast alternating LP with adaptive Nesterov step size
3. Log-sum-exp smoothing + L-BFGS (multi-temperature)

5. Peak shaving for final refinement

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def symmetrize(a):
    """Make sequence palindromic."""
    a = np.asarray(a, dtype=np.float64)
    return (a + a[::-1]) / 2.0


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = []
    if solver:
        solvers_to_try.append(solver)
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        if s not in solvers_to_try:
            solvers_to_try.append(s)
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                b_val = b_val / sv
                return b_val
        except Exception:
            continue
    return None


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.GUROBI, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return None


def alternating_lp(a0, time_limit, solver, n_max_iters=1000000):

    Alternating LP with Nesterov step size and symmetrization.
    """
    n = len(a0)
    a = normalize(a0.copy())
    a = symmetrize(a)
    a = normalize(a)
    
    y = a.copy()  # Nesterov momentum
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        lp_t0 = time.monotonic()
        b_new = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_t0
        
        if b_new is None:
            break
        
        # Symmetrize
        b_new = symmetrize(b_new)
        b_new = normalize(b_new)
        
        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = symmetrize(y)
        y = normalize(y)
        
        a = (( (it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2)))
        a = np.maximum(a, 0)
        a = symmetrize(a)
        a = normalize(a)
        
        # Track best
        score_a = eval_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = eval_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Stall detection
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 10:
                break
            
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    AltLP iter={it+1} ({elapsed:.0f}s, {lp_time*1000:.0f}ms/LP): score={best_score:.8f}")

def lbgfs_refine(a0, time_limit, T_stages, deadline=None):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    # Parameterize in log-domain: a_j = exp(z_j) / sum exp(z)
    a = normalize(a0.copy())
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    
    for stage_T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        iters = max(20, int(remaining * 10))
        
        def func(z_in):
            a_tmp = softmax_from_z(z_in)
            a_tmp = symmetrize(a_tmp)
            a_tmp = normalize(a_tmp)
            obj, _ = lse_grad(a_tmp, stage_T)
            return obj
        
        def jac(z_in):
            a_tmp = softmax_from_z(z_in)
            a_tmp = symmetrize(a_tmp)
            a_tmp = normalize(a_tmp)
            obj, g = lse_grad(a_tmp, stage_T)
            # Chain rule: z -> a via softmax+sym+normalize
            # This is complex. Let me skip the chain rule and just use the gradient w.r.t. a
            # as a gradient w.r.t. z (approximate).
            # Actually, for the softmax parameterization, da_j/dz_l = a_j(δ_jl - a_l)
            # So grad_z_l = sum_j grad_a_j * a_j * (δ_jl - a_l) = grad_a_l * a_l - a_l * sum_j grad_a_j * a_j
            S = np.dot(g, a_tmp)
            grad_z = a_tmp * (g - S)
            # Apply symmetrization: grad_z = (grad_z + grad_z[::-1]) / 2
            grad_z = symmetrize(grad_z)
            return grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-15, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = softmax_from_z(z)
    a_out = symmetrize(a_out)
    a_out = normalize(a_out)
    return a_out


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def subgradient_descent(a0, time_limit, eta_init=0.1, eta_min=0.0001, decay=0.999):
    """
    Multiplicative subgradient (mirror descent on simplex).
    Very fast, no LP needed.
    """
    n = len(a0)
    a = normalize(a0.copy())
    a = symmetrize(a)
    a = normalize(a)
    
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    eta = eta_init
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        if eta < eta_min:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        
        # Active set
        threshold = b_max * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient: average of rows
        T_a = build_conv_matrix(a)
        g = T_a[active].mean(axis=0)  # Average of active rows
        
        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        a_new = symmetrize(a_new)
        a_new = normalize(a_new)
        
        score = eval_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Don't update a, just decrease eta
            pass
        
        eta *= decay
        step += 1
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step} ({elapsed:.0f}s): score={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score


def spectral_flatten(a, n_iters=50, blend=0.3, deadline=None):
    """Gerchberg-Saxton spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = normalize(a.copy())
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        a_new = symmetrize(a_new)
        a_new = normalize(a_new)
        a = a_new
    
    return a
