import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_flatten(a, n_iters=100, blend=0.3, deadline=None):
    """Gerchberg-Saxton spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = normalize(a)

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        a_new = np.fft.ifft(F_new).real[:n]
        a = normalize(a_new)

    return a


def solve_lp_best_response(a_fixed, solver, prev_b=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] subject to sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                b_val = b_val / sv
                return b_val
        except Exception:
            continue
    return None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):

        y = normalize(y)

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = normalize(a)

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:

def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return 1e15, np.zeros(n)
    a = exp_z / s
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return o

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros(n)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        iters = max(30, min(maxiter_per_stage, int(remaining * 20)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    return a

        improved = False
        # Random order for diversity
        perm = np.random.permutation(n)

        for j in perm:
            if time.monotonic() - t0 > time_limit:
                break

            a_old_j = a[j]
            if a_old_j < 1e-15:
                # Try setting to small values
                for val in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    a[j] = val
                    s = a.sum()
                    if s < 1e-15:
                        continue
                    a_norm = a / s
                    score = 2.0 * n * fast_convolve(a_norm).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a_norm.copy()
                        a = a_norm.copy()
                        improved = True
                    a[j] = a_old_j
                a[j] = a_old_j
                continue

            for mult in multipliers:
                a[j] = a_old_j * mult
                s = a.sum()
                if s < 1e-15:
                    a[j] = a_old_j
                    continue
                a_norm = a / s
                score = 2.0 * n * fast_convolve(a_norm).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    a_old_j = a[j]
                    improved = True
                a[j] = a_old_j

            a[j] = a_old_j

        if not improved and sweep > 10:
            break

        if sweep % 100 == 0 and sweep > 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc sweep={sweep} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=100000):
    """Peak shaving with multiplicative updates targeting active peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 - np.abs(2 * x - 1)
    elif choice == 8:
        # Kaiser-like
        alpha = 5.0
        t = 2 * x - 1
        a = np.ones(n)
        # Simple approximation
        a = np.exp(-alpha * t ** 2)
    else:
        # Random positive
        a = np.abs(rng.standard_normal(n)) + 0.1

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
