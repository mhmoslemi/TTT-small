"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Convolution-based subgradient OMD/FTRL (fast, many iterations)
2. Active-set LP alternating with warm starts
3. LSE L-BFGS refinement
4. Multi-restart perturbation
Key innovation: O(n log n) subgradient via single FFT convolution,
enabling 100x more iterations than previous subgradient approaches.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(u, v):
    """Linear cross-convolution of u and v via FFT (same length)."""
    n = len(u)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    U = rfft(np.asarray(u, dtype=np.float64), fft_size)
    V = rfft(np.asarray(v, dtype=np.float64), fft_size)
    return irfft(U * V, fft_size)[:size]


def compute_subgradient(a, b, n_top_frac=0.1):
    """
    Compute subgradient of max_k (a*a)[k] at point a.
    Uses top fraction of active k's for efficiency.
    Returns (gradient, max_b, num_active).
    """
    n = len(a)
    max_b = b.max()
    # Select active set: top n_top_frac of peaks
    sorted_idx = np.argsort(b)[::-1][:max(1, int(n_top_frac * (2*n-1)))]
    # Weight by value
    w = b[sorted_idx].copy()
    w[w <= 0] = 0
    sw = w.sum()
    if sw < 1e-30:
        w = np.ones(len(sorted_idx)) / len(sorted_idx)
    else:
        w = w / sw
    
    # Subgradient: g = 2 * conv(w_extended, a) truncated to n
    # w is defined on indices sorted_idx (subset of {0,...,2n-2})
    # g[i] = 2 * sum_k w_k * a[k-i] where 0 <= k-i < n
    # = 2 * (w_full * a)[i] where w_full is the full (2n-1)-vector
    
    # Build full w vector
    w_full = np.zeros(2*n-1)
    w_full[sorted_idx] = w
    
    # conv(w_full, a)[i] = sum_j w_full[i+j] * a[j] for j=0..n-1 (where i+j < 2n-1)
    # This is the cross-convolution of w_full and a, evaluated at indices 0..n-1
    g_full = fast_cross_convolve(w_full, a)
    g = 2.0 * g_full[:n]
    
    return g, max_b, len(sorted_idx)


def omd_step(a, g, eta):
    """Online Mirror Descent (multiplicative weights) step."""
    log_mult = -eta * g
    log_mult -= log_mult.max()
    a_new = a * np.exp(log_mult)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def ftrl_simplex_update(z, h):
    """FTRL update: a = proj_simplex(-z/h)."""
    v = -z / h
    n = len(v)
    # Project onto simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1.0) / np.arange(n, 0, -1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def omd_phase(a0, time_limit, n_top_frac=0.05, eta0=None, use_ftrl=False):
    """
    OMD/FTRL phase with convolution-based subgradient.
    Fast: O(n log n) per iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    # Adaptive eta
    if eta0 is None:
        eta = 0.5
    else:
        eta = eta0
    
    # FTRL state
    z = np.zeros(n)  # cumulative gradient
    h = 1.0  # regularization parameter
    n_steps = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        g, max_b, _ = compute_subgradient(a, b, n_top_frac)
        
        if use_ftrl:
            z += g
            a_new = ftrl_simplex_update(z, h)
            h *= 1.001  # slowly increase regularization
        else:
            a_new = omd_step(a, g, eta)
            # Decay eta
            n_steps += 1
            eta = eta0 / (1 + 0.01 * n_steps) if eta0 else eta * 0.999
        
        a = a_new
        n_steps += 1
        
        if n_steps % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    OMD iter={n_steps} ({elapsed:.0f}s): best={best_score:.8f} cur={2*n*fast_convolve(a).max():.8f} eta={eta:.6f}")
    
    return best_a, best_score, n_steps

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Convolution-based subgradient OMD (O(n log n) per iter, 100x faster than per-k loop)
2. Active-set LP alternating with cutting plane and warm starts
3. LSE L-BFGS with adaptive temperature
4. Multi-scale perturbation restarts
Key innovation: single FFT for subgradient (vs O(n*|active|) previously),
enabling far more iterations in the same time budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(u, v):
    """Linear cross-convolution via FFT. Both inputs same length."""
    n = len(u)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    U = rfft(np.asarray(u, dtype=np.float64), fft_size)
    V = rfft(np.asarray(v, dtype=np.float64), fft_size)
    return irfft(U * V, fft_size)[:size]


def compute_subgradient_fast(a, b, top_frac=0.05):
    """
    Compute subgradient of max_k (a*a)[k] using top-fraction active set.
    g[i] = 2 * sum_k w_k * a[k-i] for valid (k,i) pairs.
    This equals 2 * (w_full * a)[i] (cross-convolution, truncated to n).
    Returns (gradient array of length n, max_b).
    """
    n = len(a)
    size = 2 * n - 1
    max_b = b.max()
    n_active = max(10, int(top_frac * size))
    top_idx = np.argpartition(b, -n_active)[-n_active:]
    w_vals = b[top_idx].copy()
    w_vals = np.maximum(w_vals, 0)
    sw = w_vals.sum()
    if sw < 1e-30:
        w_vals = np.ones(n_active) / n_active
    else:
        w_vals = w_vals / sw
    w_full = np.zeros(size)
    w_full[top_idx] = w_vals
    g_full = fast_cross_convolve(w_full, a)
    g = 2.0 * g_full[:n]
    return g, max_b


def omd_phase(a0, time_limit, top_frac=0.05, eta_init=1.0, label="omd"):
    """
    Online Mirror Descent (multiplicative weights) with conv-based subgradient.
    O(n log n) per iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    eta = eta_init
    it = 0
    while time.monotonic() - t0 < time_limit:
        g, max_b = compute_subgradient_fast(a, b, top_frac)
        # OMD update (multiplicative weights)
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        b = fast_convolve(a_new)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        eta *= 0.9995
        it += 1
        if it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): best={best_score:.8f} cur={2*n*b.max():.8f} eta={eta:.6f}")
    return best_a, best_score


def ftrl_phase(a0, time_limit, top_frac=0.05, label="ftrl"):
    """
    FTRL with L2 regularizer and simplex projection.
    O(n log n) per iteration (FFT for subgradient + O(n log n) for projection).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    z = np.zeros(n)
    h = 0.1
    it = 0
    while time.monotonic() - t0 < time_limit:
        g, max_b = compute_subgradient_fast(a, b, top_frac)
        z += g
        # Project -z/h onto simplex
        v = -z / h
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho_candidates = np.where(u > (cssv - 1.0) / np.arange(n, 1, -1))[0]
        if len(rho_candidates) == 0:
            a_new = np.ones(n) / n
        else:
            rho = rho_candidates[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(v - theta, 0.0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / sn
        b = fast_convolve(a_new)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        h *= 1.005
        it += 1
        if it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): best={best_score:.8f} cur={2*n*b.max():.8f} h={h:.4f}")
    return best_a, best_score


def build_sparse_conv_matrix(a, active_k):
    """Build sparse matrix T_active where T_active @ b = (a*b) for k in active_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for sn in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sn)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return sn
        except Exception:
            continue
    return cp.ECOS

def lse_obj_grad(z, n, T):
    """LogSumExp objective and gradient w.r.t. z (log-domain parameter)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    # gradient of lse w.r.t. a: sum_k w_k * grad_k(a) / (2n)
    # Then chain rule for z
    g_full = fast_cross_convolve(w, a)
    grad_a = 2.0 * g_full[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_lbgfs_phase(a0, time_limit, label="lse"):
    """LSE L-BFGS with adaptive temperature cooling."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    
    for T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        max_iter = max(20, int(remaining * 50))
        
        def func(x):
            if time.monotonic() > deadline:
                return 1e15
            val, _ = lse_obj_grad(x, n, T)
            return val
        
        def jac(x):
            if time.monotonic() > deadline:
                return np.zeros(len(x))
            _, g = lse_obj_grad(x, n, T)
            return g
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-30,
                                   'gtol': 1e-20, 'disp': False})
            z = res.x
        except Exception:
            pass
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

def make_palindromic(a):

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.12, 200.0)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_omd, _ = omd_phase(best_seq, t1, top_frac=0.05, eta_init=2.0, label="omd1")
        try_update(a_omd, "omd1")
    
    # === Phase 2: LP Alternating (main workhorse) ===
    print(f"\n=== P2: LP n={n_best} score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 600.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = lp_alternating_phase(best_seq, t2, solver, label="lp1")
        try_update(a_lp, "lp1")
    
    # === Phase 3: FTRL refinement ===
    print(f"\n=== P3: FTRL n={n_best} score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 200.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        a_ftrl, _ = ftrl_phase(best_seq, t3, top_frac=0.03, label="ftrl1")
        try_update(a_ftrl, "ftrl1")
    
    # === Phase 4: OMD + LP cycle ===
    print(f"\n=== P4: OMD+LP cycle score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.2, 350.0)
    if t4 > 60:
        print(f"  Budget: {t4:.0f}s")
        # OMD first
        t4a = t4 * 0.4
        a_omd2, _ = omd_phase(best_seq, t4a, top_frac=0.03, eta_init=1.0, label="omd2")
        try_update(a_omd2, "omd2")
        # Then LP
        rem = deadline - time.monotonic()
        t4b = min(rem * 0.5, t4 * 0.55)
        if t4b > 30:
            a_lp2, _ = lp_alternating_phase(best_seq, t4b, solver, label="lp2")
            try_update(a_lp2, "lp2")
    
    # === Phase 5: LSE L-BFGS ===
    print(f"\n=== P5: LSE L-BFGS score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 250.0)
    if t5 > 30:
        print(f"  Budget: {t5:.0f}s")
        a_lse = lse_lbgfs_phase(best_seq, t5)
        try_update(a_lse, "lse")
    
    # === Phase 6: Multi-restart perturbation ===
    print(f"\n=== P6: Perturbation restarts score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    n_restarts = 3
    for r in range(n_restarts):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        strength = 0.01 + 0.015 * r
        a_pert = perturb_sequence(best_seq, rng, strength)
        # Also try palindromic
        if r == 1:
            a_pert = make_palindromic(a_pert)
        t_sub = min(rem * 0.3, 120.0)
        # Quick OMD
        t_omd = t_sub * 0.4
        a_r_omd, _ = omd_phase(a_pert, t_omd, top_frac=0.05, eta_init=1.5, label=f"pert{r}")
        try_update(a_r_omd, f"pert{r}_omd")
        # Quick LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_sub * 0.55)
        if t_lp > 20:
            a_r_lp, _ = lp_alternating_phase(a_r_omd, t_lp, solver, label=f"pert{r}_lp")
            try_update(a_r_lp, f"pert{r}_lp")
    
    # === Phase 7: Final OMD polish ===
    print(f"\n=== P7: Final polish score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.5, 100.0)
        a_fin, _ = omd_phase(best_seq, t7, top_frac=0.02, eta_init=0.5, label="final_omd")
        try_update(a_fin, "final_omd")
        rem = deadline - time.monotonic()
        if rem > 30:
            t7b = min(rem * 0.6, 80.0)
            a_fin2, _ = lp_alternating_phase(best_seq, t7b, solver, label="final_lp")
            try_update(a_fin2, "final_lp")
    
    # === Phase 8: Try different length ===
    rem = deadline - time.monotonic()
    if rem > 100:
        print(f"\n=== P8: Different length score={best_score:.8f} ===")
        for alt_n in [1000, 1500]:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            if len(best_seq) == alt_n:
                continue
            # Interpolate
            old_x = np.linspace(0, 1, len(best_seq))
            new_x = np.linspace(0, 1, alt_n)
            a_interp = np.interp(new_x, old_x, best_seq)
            a_interp = np.maximum(a_interp, 0)
            s = a_interp.sum()
            if s > 1e-15:
                a_interp = a_interp / s
            t_alt = min(rem * 0.35, 150.0)
            a_alt, _ = omd_phase(a_interp, t_alt * 0.4, top_frac=0.05, eta_init=1.5, label=f"alt_n{alt_n}")
            try_update(a_alt, f"alt_n{alt_n}_omd")
            rem = deadline - time.monotonic()
            if rem > 40:
                a_alt2, _ = lp_alternating_phase(a_alt, min(rem * 0.5, t_alt * 0.5), solver, label=f"alt_n{alt_n}_lp")
                try_update(a_alt2, f"alt_n{alt_n}_lp")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")
    
    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
