def fast_convolve(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:

        if len(a_list) == 0:

        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def lse_objective_grad(z_in, T_val, n):

    a = softmax_from_z(z_in)
    b = fast_convolve(a)

    Tb = T_val * b
    max_Tb = Tb.max()

        max_Tb = 700

    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    # Gradient w.r.t. a: d/da_j [2n * sum_k w_k * c_k]
    # c_k = sum_{i+j=k} a_i * a_j
    # dc_k/da_j = 2 * a_{k-j} (by symmetry)
    # So grad_a_j = 2n * sum_k w_k * 2 * a_{k-j} = 4n * sum_k w_k * a_{k-j}
    # This is 4n * cross_correlation(w, a)
    # Actually: grad_a = 4n * conv(w, a_reversed) [appropriate slice]
    # Let me compute it as: grad_a_j = 4n * sum_k w_k * a_{k-j} (with a_x=0 for x outside [0,n-1])
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 4.0 * w[k] * a[k - idx]
    grad_a *= n  # Wait, I already have the 4n factor... let me recheck.
    
    # Actually: obj = (2n/T) * log(sum exp(T*b))
    # d obj / d b_k = (2n/T) * T * exp(T*b_k) / sum(exp(T*b)) = 2n * w_k
    # d obj / d a_j = sum_k (2n * w_k) * dc_k/da_j = 2n * sum_k w_k * 2*a_{k-j} = 4n * sum_k w_k * a_{k-j}
    
    # So grad_a_j = 4n * sum_k w_k * a_{k-j}

    S = np.dot(grad_a, a)

def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=25):
    """Multi-stage LSE L-BFGS optimization."""

    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        try:
            max_iter = max(20, int(remaining * 5))  # adaptive
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': max_iter * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0 or si == n_stages - 1:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={sc:.10f}")
        except Exception as e:
            print(f"    LSE stage {si+1} failed: {e}")
    
    return softmax_from_z(z)


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
                last_print = elapsed
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n

    """
    Fast LSE objective and gradient using FFT for the cross-correlation.
    f_T(a) = (2n/T) * log(sum_k exp(T * c_k)) where c = conv(a,a)
    grad_z = a * (grad_a - dot(grad_a, a))
    grad_a_j = 4n * sum_k w_k * a_{k-j}  (cross-correlation, computed via FFT)
    """
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Compute grad_a_j = 4n * sum_k w_k * a_{k-j} via FFT cross-correlation
    # cross[j] = sum_k w_k * a[k-j] = conv(w, a_rev)[j + n - 1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(np.zeros(3 * n - 2))
    # Place a[::-1] at the start
    a_rev_padded = np.zeros(3 * n - 2)
    a_rev_padded[:n] = a[::-1]
    A_rev = rfft(a_rev_padded, fft_size)
    cross = irfft(W * A_rev, fft_size)[:3 * n - 2]
    grad_a = 4.0 * n * cross[n - 1:2 * n - 1]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_obj_grad_loop(z_in, T_val, n):
    """LSE objective and gradient using direct loop (simpler, for verification)."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 4.0 * wk * a[k - idx]
    grad_a *= n

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=30):
    """Multi-stage LSE L-BFGS with FFT-accelerated gradients."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_obj_grad_fast(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_obj_grad_fast(z_val, stage_T, n)
            return grad

        try:
            # Adaptive max iterations based on remaining time
            max_iter = max(30, int(remaining * 10))
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': max_iter * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0 or si == n_stages - 1:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f} ({elapsed:.0f}s): score={sc:.10f}")
        except Exception as e:
            pass

    return softmax_from_z(z)


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} M_lp={M_lp:.8f}")
                last_print = elapsed

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_and_polish(a_best, rng, time_per_trial, n_trials):
    """
    Perturbation-based local search: add random noise, polish with short LSE,
    keep the best result.
    """
    global _global_deadline
    n = len(a_best)
    best_a = a_best.copy()
    best_score = score_normalized(a_best)
    epsilons = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]

    total_budget = time_per_trial * n_trials
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t0 > total_budget:
            break
        if time.monotonic() > _global_deadline:
            break

        eps = epsilons[trial % len(epsilons)]
        # Random perturbation in log-space
        z = z_from_a(a_best)
        noise = rng.normal(0, eps * np.sqrt(n), n)
        z_pert = z + noise
        a_pert = softmax_from_z(z_pert)

        # Short LSE polish from perturbed point
        remaining_trial = min(time_per_trial, _global_deadline - time.monotonic())
        if remaining_trial < 3:
            break
        a_polished = lbgfs_logsumexp(a_pert, remaining_trial, T_start=20.0, T_end=0.1, n_stages=8)
        sc = score_normalized(a_polished)
        if sc < best_score:
            best_score = sc
            best_a = a_polished.copy()
            print(f"    Perturb trial {trial+1} eps={eps}: NEW BEST {sc:.10f}")

    return best_a, best_score


def fista_l2(a0, time_limit, max_iters=20000):
    """
    FISTA on convex relaxation: min sum(conv(a,a)^2) s.t. a>=0, sum(a)=1.
    This gives a good starting point and a lower bound on the true objective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_val = np.sum(fast_convolve(a) ** 2)
    t0 = time.monotonic()

    # Estimate Lipschitz constant: ||Hessian|| <= 4 * n * max(conv(a,a))
    L = 4.0 * n * fast_convolve(a).max()
    L = max(L, 1.0)

    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    a_rev_padded = np.zeros(3 * n - 2)
    a_rev_padded[:n] = a[::-1]

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c_y = fast_convolve(y)
        # Gradient of sum(c^2) w.r.t. a: 4 * cross_corr(c_y, y)
        W = rfft(c_y, fft_size)
        a_rev_padded[:n] = y[::-1]
        A_rev = rfft(a_rev_padded, fft_size)
        cross = irfft(W * A_rev, fft_size)[:3 * n - 2]
        grad = 4.0 * cross[n - 1:2 * n - 1]

        # FISTA step
        a_new = project_simplex(y - grad / L)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + ((t_k - 1.0) / t_new) * (a_new - a)

        a = a_new
        t_k = t_new

        val = np.sum(fast_convolve(a) ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # Update Lipschitz estimate
        L = max(L * 0.99, 4.0 * n * val + 1.0)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA-L2 iter={it+1} ({elapsed:.0f}s): l2val={val:.10f} score={score_normalized(a):.10f}")

    return best_a, best_val


_global_deadline = 0.0


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, deadline in {budget_s - 30}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.10f}")
    except Exception:
        # Default initialization
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.10f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 0: FISTA on L2 relaxation (quick, 5% budget) ===
    print(f"\n=== Phase 0: FISTA L2 relaxation (score={best_score:.10f}) ===")
    t0_phase = 0.05 * (budget_s - 30)
    a_fista, _ = fista_l2(best_seq, t0_phase, max_iters=5000)
    try_update(a_fista, "fista_l2")
    print(f"  Phase 0 done: best_score={best_score:.10f}")

    # === Phase 1: Main Nesterov LP (40% budget) ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t1 = min(0.40 * (budget_s - 30), remaining * 0.6)
    a_out, _ = nesterov_lp(best_seq, t1, solver=solver)
    try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 1 done: best_score={best_score:.10f}")

    # === Phase 2: Aggressive LSE L-BFGS polishing (25% budget) ===
    print(f"\n=== Phase 2: LSE L-BFGS aggressive (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(0.25 * (budget_s - 30), remaining * 0.5)
    if t2 > 10:
        a_lse = lbgfs_logsumexp(best_seq, t2, T_start=80.0, T_end=0.02, n_stages=35)
        try_update(a_lse, "lse_aggressive")
    print(f"  Phase 2 done: best_score={best_score:.10f}")

    # === Phase 3: Perturbation-based local search (20% budget) ===
    print(f"\n=== Phase 3: Perturbation search (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(0.20 * (budget_s - 30), remaining * 0.5)
    if t3 > 30:
        n_trials = 5
        time_per_trial = t3 / n_trials
        a_pert, _ = perturb_and_polish(best_seq, rng, time_per_trial, n_trials)
        try_update(a_pert, "perturbation")
    print(f"  Phase 3 done: best_score={best_score:.10f}")

    # === Phase 4: Second Nesterov LP round (10% budget) ===
    print(f"\n=== Phase 4: Nesterov LP round 2 (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t4 > 10:
        a_out2, _ = nesterov_lp(best_seq, t4, solver=solver)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 4 done: best_score={best_score:.10f}")

    # === Phase 5: Final LSE polish at very low temperature ===
    print(f"\n=== Phase 5: Final LSE polish (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 10:
        t5 = min(remaining * 0.5, 120.0)
        if t5 > 5:
            a_final = lbgfs_logsumexp(best_seq, t5, T_start=10.0, T_end=0.01, n_stages=20)
            try_update(a_final, "final_lse")
    print(f"  Phase 5 done: best_score={best_score:.10f}")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length exploration (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        lengths_to_try = [n_base + 100, n_base - 100, n_base + 200, n_base + 50]
        for n_try in lengths_to_try:
            remaining = _global_deadline - time.monotonic()
            if remaining < 15:
                break
            if n_try < 100:
                continue
            t_try = min(remaining * 0.3, 90.0)
            if t_try < 5:
                break
            print(f"  Trying n={n_try}, budget={t_try:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick LSE polish at the new length
            a_try = lbgfs_logsumexp(init_try, t_try, T_start=30.0, T_end=0.05, n_stages=15)
            try_update(a_try, f"len_{n_try}")

    # === Phase 7: One more perturbation round if time remains ===
    print(f"\n=== Phase 7: Final perturbation (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 20:
        a_final2, _ = perturb_and_polish(best_seq, rng, min(remaining * 0.3, 30.0), 3)
        try_update(a_final2, "final_perturb")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
