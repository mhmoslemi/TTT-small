def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max_b / sum_a^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        beta1 = 0.8
        beta2 = 0.999
        eps = 1e-8
        t_adam = 0
        
        max_iters = int(min(2000, remaining * 50))  # aim for ~50 iters/sec
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            val, grad_a = lse_grad_computation(a, stage_T, top_K=top_K)
            
            # Project gradient to simplex (sum(a)=1 constraint)
            # The effective gradient on the simplex is grad_a - mean(grad_a * a) / ... 
            # Actually, for the constraint sum(a)=1, the projected gradient is:
            # grad_proj = grad_a - np.mean(grad_a)  (for the affine constraint)
            # But we also have a >= 0, so we use multiplicative updates.
            
            # Use multiplicative (mirror) descent with Adam:
            # a_new[i] = a[i] * exp(-lr * grad_i) normalized
            # With Adam adaptation on the log-space gradient
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_a
            v = beta2 * v + (1 - beta2) * grad_a ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            # Adaptive learning rate
            lr_eff = lr * np.sqrt(t_adam)  # warmup
            step = m_hat / (np.sqrt(v_hat) + eps)
            
            # Multiplicative update (stays non-negative)
            a_new = a * np.exp(-lr_eff * step)
            
            # Clip and normalize
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            a = a_new
            
            # Check score
            if it % 10 == 0:
                score = compute_objective(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        # Update a to best
        score = compute_objective(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        print(f"    LSE-Adam stage {si+1}/{n_stages} (T={stage_T:.2f}): best={best_score:.6f}")
    
    return best_a, best_score


def lp_best_response_approx(a_fixed, n_constraints_sample=None):
    """
    Approximate LP best response: find b >= 0, sum(b)=1 minimizing max_k (a_fixed * b)_k.
    Uses subgradient method as approximation when n is large.
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    T = 500  # iterations
    lr = 0.1
    
    for t in range(1, T + 1):
        # Compute conv = a_fixed * b
        conv = fast_autocorr_combined(a_fixed, b)
        # Subgradient: the position of max
        k_max = np.argmax(conv)
        # Gradient of max w.r.t. b: d conv[k_max] / d b[j] = a_fixed[k_max - j] (if valid)
        # conv[k] = sum_j a_fixed[j] * b[k-j]
        # d conv[k] / d b[j] = a_fixed[k-j] (if 0 <= k-j < n)
        # So grad[j] = a_fixed[k_max - j] for j = max(0, k_max-n+1) to min(n-1, k_max)
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        if js <= je:
            grad[js:je+1] = a_fixed[k_max - np.arange(js, je + 1)]
        
        # Mirror descent step
        b_new = b * np.exp(-lr / np.sqrt(t) * grad)
        b_new = b_new / b_new.sum()
        b = b_new
    
    return b


def fast_autocorr_combined(a, b):
    """Compute convolution of a and b using FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def nesterov_lp_optimize(a0, time_limit, use_approx_lp=True, max_iters=50000):
    """
    Nesterov-accelerated LP best-response iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Get best response
        if use_approx_lp:
            b_star = lp_best_response_approx(y)
        else:
            b_star = _solve_lp_exact(y)
            if b_star is None:
                b_star = lp_best_response_approx(y)
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        if t % 5 == 0:
            score_a = compute_objective(a)
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = compute_objective(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def _solve_lp_exact(a_fixed):
    """Try to solve the LP exactly using cvxpy."""
    n = len(a_fixed)
    if n > 3000:  # Too large for exact LP
        return None
    try:
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        # Build constraint matrix
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a_fixed
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=cp.ECOS, max_iterations=500)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s
    except Exception:
        pass
    return None


def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Coordinate descent: move mass between pairs of positions to reduce peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        
        improved = False
        for k in top_k:
            # For peak at position k, the contributing pairs are (j, k-j)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue
                
            src = idx[np.argmax(contrib)]
            
            # Find good target: low a[j], far from peak
            # Score each position by how much it would help
            candidates = np.arange(n)
            # Prefer positions with low a[j] and not in peak neighborhoods
            peak_zone = set()
            for k2 in top_k[:3]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_zone.update(range(js2, je2 + 1))
            
            weights = np.array([1.0 if j not in peak_zone else 0.1 for j in range(n)])
            # Among candidates, prefer low a[j]
            scores = a * weights
            scores[src] = np.inf
            # Pick from the lowest
            bottom_idx = np.argsort(scores)
            n_candidates = max(5, n // 20)
            dst_pool = bottom_idx[:n_candidates]
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                a_trial = a_trial / s_t
                score = compute_objective(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = compute_objective(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            stall += 1
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    # This is a cross-correlation
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    
    return val, grad


def lse_adam_optimize(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=12, lr=0.01, top_K=None):
    """Adam optimization of log-sum-exp smoothed objective."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t_start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 3:
            break
        
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        t_adam = 0
        
        max_iters = min(5000, int(remaining * 30))
        for it in range(max_iters):
            if time.monotonic() - t_start > time_limit:
                break
            
            val, grad = lse_value_grad(a, stage_T, top_K=top_K)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_eff = lr * min(1.0, np.sqrt(t_adam) / 10.0)
            step = m_hat / (np.sqrt(v_hat) + eps)
            
            # Multiplicative update (preserves non-negativity)
            a_new = a * np.exp(-lr_eff * step)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            a = a_new
            
            if it % 20 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        print(f"    LSE-Adam stage {si+1}/{n_stages} (T={stage_T:.2f}): best={best_score:.6f}")
    
    return best_a, best_score


def coord_descent_polish(a0, time_limit, seed=42):
    """Targeted coordinate descent: move mass from peak-contributing positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    stall = 0
    
    while time.monotonic() - t_start < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        b_max = b.max()
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-18:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find target: prefer positions with low a[j] and far from peaks
            peak_zone = set()
            for k2 in top_k[:4]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_zone.update(range(js2, je2 + 1))
            
            # Score: prefer low a[j], penalize being in peak zone
            pen = np.array([2.0 if j in peak_zone else 1.0 for j in range(n)])
            scores = a * pen
            scores[src] = np.inf
            bottom = np.argsort(scores)[:max(10, n // 10)]
            
            # Try multiple targets
            targets = bottom[rng.integers(len(bottom), size=min(5, len(bottom)))]
            for dst in targets:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            stall += 1
            if stall > 300:
                break
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new


def fine_coord_descent(a0, time_limit, seed=123):
    """Very fine coordinate descent with small steps for final polish."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    
    while time.monotonic() - t_start < time_limit:
        b = fast_autocorr(a)
        k_max = np.argmax(b)
        
        # Find the top contributor to the peak
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]
        top_srcs = idx[np.argsort(contrib)[::-1][:5]]
        
        improved = False
        for src in top_srcs:
            # Try moving tiny amounts to random positions
            for _ in range(10):
                dst = rng.integers(n)
                if dst == src:
                    continue
                for frac in [0.0005, 0.001, 0.002, 0.003, 0.005]:
                    amt = a[src] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try a small random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                amt = a[i] * rng.uniform(0.0005, 0.002)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
    
    return best_a, best_score
