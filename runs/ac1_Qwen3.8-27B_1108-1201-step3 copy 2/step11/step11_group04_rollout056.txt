import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross(a, b):
    """Cross-correlation a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():

            prob.solve(solver=solver_name)

        except Exception:

def solve_lp_best_response(a_fixed, solver, T_mat=None):

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z.
    Objective: 2*n*logsumexp(b/T)/T where b = conv(a,a), a = softmax(z)
    Simplified: 2*n*(max_b + T*log(sum(exp((b-max_b)/T))))
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_objective(z, n, T_stage, deadline):
    """Objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_stage)
    return obj


def lbgfs_jacobian(z, n, T_stage, deadline):
    """Gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_stage)
    return g


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton style) with peak weighting."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        phases = np.angle(F)

        # Compute autocorrelation to identify peaks
        R = irfft(F * np.conj(F), L)  # This is the circular autocorrelation
        # Use linear autocorrelation
        R_lin = fast_convolve(a)
        max_R = R_lin.max()
        threshold = max_R * 0.95
        peak_mask = R_lin >= threshold

        # Weighted target: reduce peaks more
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            # Apply stronger flattening at frequencies that contribute to peaks
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def spectral_flatten_weighted(a, n_iters=50, blend=0.3, deadline=None):
    """Spectral flattening with explicit peak suppression in frequency domain."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        phases = np.angle(F)

        # Compute current autocorrelation peaks
        R_lin = fast_convolve(a)
        max_R = R_lin.max()

        # Find top peaks and compute their frequency contribution
        top_k = max(1, len(R_lin) // 50)
        peak_indices = np.argsort(R_lin)[-top_k:]

        # For each peak at position k, the frequency contribution is ~exp(i*2*pi*f*k/L)
        # We want to reduce |F(f)| where these contributions are large
        # Approximate: reduce F(f) proportionally to the "peak energy" at each f
        peak_weight = np.ones(L)
        for k in peak_indices:
            weight_k = R_lin[k] / max_R
            freq_response = np.cos(2 * np.pi * np.arange(L) * k / L)
            peak_weight[1:] += weight_k * np.abs(freq_response[1:])

        peak_weight = peak_weight / peak_weight.max()

        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            effective_blend = blend * (0.5 + 0.5 * peak_weight[k])
            new_mag = (1 - effective_blend) * mags[k] + effective_blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def peak_shave(a, time_limit, n_steps=100000):

        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Weight by peak height
        peak_weights = b[active] / max_b
        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += peak_weights[i] * a[k - idx]
        g /= peak_weights.sum()

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive step and duality tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    rng = np.random.default_rng(54321)
    lp_times = []
    T_mat = build_conv_matrix(a)  # Will rebuild when a changes significantly
    T_valid = False

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        if len(lp_times) >= 30:
            avg_lp_time = np.mean(lp_times[-50:])
            if avg_lp_time > 10.0 and len(lp_times) > 100:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")
                break

        # Nesterov dual averaging with adaptive step
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Nesterov extrapolation
        x_old = a.copy()
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Also try the "dual" solution: use lambda from dual as guidance
        if it % 100 == 0 and it > 0:
            # Check if we can get the dual
            pass

        # Stall detection with perturbation
        if (it + 1) % 200 == 0:
            prev_best_local = best_score
            if best_score < 2.0 * n * fast_convolve(x_old).max() - 1e-12:
                stall_count = 0
            else:
                stall_count += 1

            if stall_count >= 10:
                # Perturb y to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} lp_time={np.mean(lp_times[-10:]):.3f}s")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        iters = max(30, min(maxiter_per_stage, int(remaining * 30 / max(1, stages_left))))

        def func(z_in):
            return lbgfs_objective(z_in, n, stage_T, deadline)

        def jac(z_in):
            return lbgfs_jacobian(z_in, n, stage_T, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_pair_subgradient(a0, time_limit, n_steps=50000):
    """Target the top 2 peaks simultaneously with adaptive subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.02
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Get top 2 peaks
        sorted_idx = np.argsort(b)[-2:]
        k1, k2 = sorted_idx

        g = np.zeros(n)
        for k in [k1, k2]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= 2.0

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 - np.abs(2 * x - 1)
    elif choice == 8:
        # Chirp-like magnitude
        a = np.abs(np.sin(np.pi * x * (1 + 2 * x))) + 0.1
    else:
        # Gaussian sum
        a = np.exp(-8*(x-0.2)**2) + 0.7*np.exp(-8*(x-0.5)**2) + 0.5*np.exp(-8*(x-0.8)**2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gaussian_sum_param_init(n, rng, n_gaussians=4):
    """Initialize using a sum of Gaussians (guarantees non-negativity)."""
    centers = np.sort(rng.uniform(0.1, 0.9, n_gaussians))
    widths = rng.uniform(0.05, 0.3, n_gaussians)
    amplitudes = rng.uniform(0.5, 2.0, n_gaussians)
    x = np.arange(n) / (n - 1)
    a = np.zeros(n)
    for c, w, amp in zip(centers, widths, amplitudes):
        a += amp * np.exp(-((x - c) / w) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def chirp_perturbation(a, rng, strength=0.02):
    """Perturb with a chirp signal."""
    n = len(a)
    x = np.arange(n) / n
    beta = rng.uniform(2, 10)
    chirp = np.sin(2 * np.pi * (x + beta * x**2))
    a_pert = a + strength * chirp
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def diversify_and_select(parent, rng, budget_s=15):
    """Try several perturbations and spectral variants, return the best."""
    n = len(parent)
    candidates = [parent.copy()]

    for strength in [0.02, 0.05, 0.08, 0.12]:
        candidates.append(perturbed_sequence(parent, rng, strength=strength))
        candidates.append(chirp_perturbation(parent, rng, strength=strength * 0.5))

    # Gaussian sum perturbation
    a_gauss = gaussian_sum_param_init(n, rng, n_gaussians=3)
    a_blend = 0.7 * parent + 0.3 * a_gauss
    a_blend = np.maximum(a_blend, 0)
    a_blend /= a_blend.sum()
    candidates.append(a_blend)

    best_c = parent.copy()
    best_sc = evaluate_local(parent)

    t_start = time.monotonic()
    for c in candidates:
        if time.monotonic() - t_start > budget_s:
            break
        sc = evaluate_local(c)
        if sc < best_sc:
            best_sc = sc
            best_c = c.copy()

    # Also try spectral flattening on the best
    if time.monotonic() - t_start < budget_s * 0.7:
        sf = spectral_flatten_weighted(best_c, n_iters=15, blend=0.3,
                                       deadline=time.monotonic() + (budget_s - (time.monotonic() - t_start) * 0.3))
        sc_sf = evaluate_local(sf)
        if sc_sf < best_sc:
            best_sc = sc_sf
            best_c = sf

    return best_c, best_sc


def low_dim_gaussian_opt(n, budget_s=30):
    """Optimize a Gaussian-sum parameterization in low dimension."""
    rng = np.random.default_rng(777)
    best_a = None
    best_score = float('inf')
    t_start = time.monotonic()

    for n_gauss in [3, 4, 5]:
        if time.monotonic() - t_start > budget_s * 0.6:
            break
        for trial in range(5):
            if time.monotonic() - t_start > budget_s:
                break
            centers = np.sort(rng.uniform(0.05, 0.95, n_gauss))
            widths = rng.uniform(0.03, 0.25, n_gauss)
            amplitudes = rng.uniform(0.3, 2.0, n_gauss)

            def gauss_params(params):
                c = np.sort(np.abs(params[:n_gauss]) * 0.9 + 0.05)
                w = np.abs(params[n_gauss:2*n_gauss]) * 0.2 + 0.03
                amp = np.abs(params[2*n_gauss:3*n_gauss]) + 0.1
                x = np.arange(n) / (n - 1)
                a = np.zeros(n)
                for ci, wi, ai in zip(c, w, amp):
                    a += ai * np.exp(-((x - ci) / wi) ** 2)
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    return 1e10
                a = a / s
                return evaluate_local(a)

            x0 = np.concatenate([centers, widths, amplitudes])
            try:
                res = minimize(gauss_params, x0, method='Nelder-Mead',
                              options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-8})
                if res.fun < best_score:
                    best_score = res.fun
                    c = np.sort(np.abs(res.x[:n_gauss]) * 0.9 + 0.05)
                    w = np.abs(res.x[n_gauss:2*n_gauss]) * 0.2 + 0.03
                    amp = np.abs(res.x[2*n_gauss:3*n_gauss]) + 0.1
                    x = np.arange(n) / (n - 1)
                    a = np.zeros(n)
                    for ci, wi, ai in zip(c, w, amp):
                        a += ai * np.exp(-((x - ci) / wi) ** 2)
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    best_a = a
            except Exception:
                pass

    if best_a is None:
        best_a = gaussian_sum_param_init(n, rng, 4)
        best_score = evaluate_local(best_a)

    return best_a, best_score

            print(f"  Trying n={n_try}, budget={t_nt:.0f}s")
            init_nt = interpolate_sequence(best_seq, n_try)
            a_sp_nt = spectral_flatten_weighted(init_nt, n_iters=20, blend=0.3,
                                                deadline=time.monotonic() + min(t_nt * 0.3, 20))
            a_nt, _ = nesterov_lp_refine(a_sp_nt, min(t_nt * 0.5, 50), solver, n_max_iters=50000)
            local_update(a_nt, f"lp_n{n_try}")

            # Quick L-BFGS
            rem = deadline - time.monotonic()
            if best_seq is not None and len(best_seq) == n_try and rem > 30:
                T_st2 = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
                a_ntb = lbgfs_refine(best_seq, min(rem, 40), T_st2,
                                     maxiter_per_stage=5000,
                                     deadline=time.monotonic() + min(rem, 40))
                local_update(a_ntb, f"lbgfs_n{n_try}")

    # === Phase 9: Final polish ===
    rem = deadline - time.monotonic()
    print(f"=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    if rem > 15:
        T_stages_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        t_fin = min(rem * 0.4, 50.0)
        if t_fin > 10:
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin, maxiter_per_stage=5000,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_ps, _ = peak_shave(best_seq, min(rem * 0.3, 25), n_steps=30000)
        local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_flp, _ = nesterov_lp_refine(best_seq, min(rem * 0.5, 20), solver, n_max_iters=10000)
        local_update(a_flp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
            if eval_score < best_score:
                best_score = eval_score
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
