    cols = np.zeros(n * n, dtype=np.int32)
    offsets = np.zeros(n * n, dtype=np.int32)
    count = 0
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows[count] = k
            cols[count] = j
            offsets[count] = k - j
            count += 1
    return rows[:count], cols[:count], offsets[:count], size


def make_sparse_conv(n, a, indices):
    rows, cols, offsets, size = indices
    vals = a[offsets]
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def get_solver():
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GLOP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, indices, solver):
    n = len(a_fixed)
    T = make_sparse_conv(n, a_fixed, indices)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver]
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        if s != solver:
            solvers.append(s)
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                b_val = b_val / sv
                return b_val, float(max(0.0, M_var.value))
        except Exception:
            continue
    return None, None


def multiplicative_weights(a0, time_limit, n_steps=1000000, eta0=0.05):
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        w = b[active] / b[active].sum()
        g = np.zeros(n)
        for wi, k in zip(w, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += wi * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.3)
        else:
            eta *= 0.85
            if eta < 1e-9:
                break
    return best_a, best_score


def lp_alternating(a0, time_limit, solver, n_c=5.0, max_restarts=20):
    """LP alternating with Nesterov acceleration and Gaussian restarts."""
    n = len(a0)
    indices = precompute_conv_indices(n)
    a = normalize_a(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    restarts = 0
    rng = np.random.default_rng(12345)
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_val = solve_lp_best_response(y, indices, solver)
        if b_new is None:
            continue
        alpha = n_c / (it + n_c)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
            sy = 1.0
        y = y / sy
        a = ((it + 1) * a + alpha * (it + n_c) * b_new) / (it + 1 + alpha * (it + n_c))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8 and restarts < max_restarts:
                # Gaussian blur restart
                sigma = max(2.0, n / 80.0)
                a_smooth = gaussian_filter1d(a, sigma=sigma, mode='nearest')
                a_smooth = normalize_a(a_smooth)
                y = a_smooth.copy()
                stall = 0
                restarts += 1
                if restarts % 5 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP restart {restarts} ({elapsed:.0f}s): score={best_score:.8f}")
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_and_grad(z, n, T_temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + math.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=500):
    n = len(a0)
    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.mean()  # center for numerical stability
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for T in T_stages:
        if time.monotonic() > deadline:
            break
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        # Define objective and gradient (without closures)
        # We'll use a wrapper class
        obj_func = _LSEObj(n, T, deadline)
        jac_func = _LSEJac(n, T, deadline)
        maxiter = min(maxiter_per_stage, int(remaining * 5))
        try:
            res = minimize(
                fun=obj_func, x0=z, jac=jac_func, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-25,
                         'gtol': 1e-18, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


class _LSEObj:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline
    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_objective_and_grad(z, self.n, self.T)
        return obj


class _LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline
    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_objective_and_grad(z, self.n, self.T)
        return g


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    return normalize_a(a_pal)


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    return normalize_a(a)
