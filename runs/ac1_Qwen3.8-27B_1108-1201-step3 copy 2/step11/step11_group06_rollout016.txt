import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def lse_obj_and_grad(z, n, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_max = scaled.max()
    exp_terms = np.exp(scaled - scaled_max)
    log_sum = scaled_max + np.log(exp_terms.sum() + 1e-300)
    lse_val = max_b + T * log_sum
    obj = 2.0 * n * lse_val
    # Gradient w.r.t. a
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    # Chain rule: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE L-BFGS with 60+ temperature stages (core improvement)
2. LP alternating with Nesterov acceleration for fast convergence
3. Multiplicative weights for rapid warmup
4. Coarse-to-fine initialization across multiple lengths
5. Multi-length exploration (n=1000, 1200, 1500)
Key innovation: O(N log N) gradient via FFT cross-correlation, very fine
temperature schedule, and aggressive L-BFGS iterations per stage.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_fft(z, n, T, fft_size):
    """
    Compute LSE objective and gradient w.r.t. z using FFT-based cross-correlation.
    Key optimization: gradient computed in O(N log N) via FFT instead of O(n^2).
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_max = scaled.max()
    exp_terms = np.exp(scaled - scaled_max)
    log_sum = scaled_max + np.log(exp_terms.sum() + 1e-300)
    lse_val = max_b + T * log_sum
    obj = 2.0 * n * lse_val

    # Weights for the LSE
    w = exp_terms / exp_terms.sum()

    # Gradient via FFT cross-correlation: grad_a[j] = 2 * sum_k w[k] * a[k-j]
    # This is 2 * cross_correlation(w, a)
    # cross_correlation(w, a)[j] = sum_k w[k] * a[k-j]
    # Compute via: ifft(conj(fft(w)) * fft(a))[:n]
    W_fft = rfft(w, fft_size)
    A_fft = rfft(a, fft_size)
    cross = irfft(W_fft * A_fft, fft_size)[:n]
    grad_a = 2.0 * cross

    # Chain rule for softmax parameterization
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in ['ECOS', 'CLARABEL', 'PDLP', 'SCS', 'MOSEK']:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return 'ECOS'


def build_conv_matrix_sparse(y):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(y, b)."""
    n = len(y)
    size = 2 * n - 1
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, size)
        length = k_end - k_start
        rows_list.append(np.arange(k_start, k_end))
        cols_list.append(np.full(length, j, dtype=np.int32))
        vals_list.append(y[:length])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def solve_lp_best_response(y_fixed, solver, T_mat):
    """Given fixed y (sum=1), find b minimizing max_k conv(y,b)[k]."""
    n = len(y_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, 'ECOS', 'SCS', 'PDLP']:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None
                return b_val / s_val
        except Exception:
            continue
    return None


def multiplicative_weights(a0, time_limit, n_steps=500000, eta0=0.05):
    """Fast multiplicative weights with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Use top few peaks
        top_k = min(5, len(b))
        active = np.argsort(b)[-top_k:]
        w_active = b[active]
        w_active = w_active / w_active.sum()

        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]

        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.3)
        else:
            eta *= 0.85
            if eta < 1e-9:
                break
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov-accelerated LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    T_mat = build_conv_matrix_sparse(y)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, solver, T_mat)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        T_mat = build_conv_matrix_sparse(y)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.015 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            if (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def lse_lbfgs_refine(a0, time_limit, n_stages=60, maxiter_per_stage=300):
    """
    L-BFGS optimization of the LSE-smoothed objective with very fine
    temperature schedule. Uses FFT-based gradient (O(N log N)).
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    # Temperature schedule: logarithmic from high to very low
    T_high = 50.0
    T_low = 0.0005
    log_ratios = np.geomspace(np.log(T_high / T_low), np.log(T_high / T_low / 60.0), n_stages)
    T_stages_list = T_high * np.exp(-log_ratios)
    T_stages_list = np.sort(T_stages_list)[::-1]

    t0 = time.monotonic()
    deadline = t0 + time_limit
    best_z = z.copy()
    best_obj = float('inf')

    for stage_idx, T in enumerate(T_stages_list):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        def func(z_eval):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_obj_grad_fft(z_eval, n, T, fft_size)
            return obj

        def jac(z_eval):
            if time.monotonic() > deadline:
                return np.zeros(n)
            _, g = lse_obj_grad_fft(z_eval, n, T, fft_size)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-15, 'maxfun': maxiter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Evaluate current point
        try:
            cur_obj, _ = lse_obj_grad_fft(z, n, T, fft_size)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_z = z.copy()
        except Exception:
            pass

        if (stage_idx + 1) % 15 == 0:
            a_cur = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_cur).max()
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {stage_idx+1}/{n_stages} ({elapsed:.0f}s): "
                  f"T={T:.6f}, LSE={cur_obj:.8f}, true_max_score={sc:.8f}")

    return softmax_from_z(best_z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length (preserves shape, normalizes to sum 1)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    else:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):

            best_score = bl_score
            print(f"  Coarse-to-fine improved: {best_score:.8f}")

    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, n_steps=500000, eta0=0.05)
        try_update(a_mw, "mw")

    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")

            try_update(a_ln_lse, f"lse_n{ln}")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t6 = min(rem * 0.4, 120.0)
        a_fin_lp, _ = nesterov_lp_refine(best_seq, t6, solver, n_max_iters=200000)
        try_update(a_fin_lp, "final_lp")

    # === Phase 7: Final LSE with very fine temps ===
    print(f"\n=== Phase 7: Final fine LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.5, 200.0)
        a_fin_lse = lse_lbfgs_refine(best_seq, t7, n_stages=40, maxiter_per_stage=400)
        try_update(a_fin_lse, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
