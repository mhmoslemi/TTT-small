def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):

    n = len(a0)

    if s < 1e-15:
        a = np.ones(n) / n

    a_prev = a.copy()
    best_a = a.copy()

            # Accept
            a_prev = a.copy()
            a = a_candidate
            t_nest = t_new
            consecutive_increases = 0
            
            if sc_candidate < best_score:
                best_score = sc_candidate

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue

                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

def structured_init_asymmetric(n, rng):
    """Generate an asymmetric structured initialization."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Tilted uniform
        a = 1.0 + 0.8 * (2.0 * x - 1.0)
    elif choice == 1:
        # Ramp
        a = x + 0.1
    elif choice == 2:
        # Step-like
        a = 2.0 * np.exp(-3.0 * x) + 0.3
    elif choice == 3:
        # Asymmetric bump
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.2
    elif choice == 4:
        # Cosine with phase shift (asymmetric)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (x + 0.1)) + 0.15 * np.cos(4 * np.pi * x)
    elif choice == 5:
        # Two bumps asymmetric
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.7) ** 2) + 0.1
    elif choice == 6:
        # Power law
        a = (x + 0.05) ** 0.5
    else:
        # Sinusoidal asymmetric
        a = 0.8 + 0.4 * np.sin(2 * np.pi * x + 0.5)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov LP with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        # Adaptive restart
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0005 and t_nest > 5.0:
            # Function increased - reduce momentum
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 40 if si < n_stages // 2 else 20
        try:

def peak_targeted_refine(a0, time_limit, K=30, max_steps=50000):
    """Multiplicative weights update targeting top K peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.05
    last_improve_step = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.92

        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Compute gradient toward peaks
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        # Multiplicative update: a'_i = a_i * exp(-eta * grad_i) / Z
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.002, 0.15)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, max_steps=100000):

    n = len(a0)

    eta = 0.3
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()

        grad_a = np.zeros(n)

            je = min(n - 1, k)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

                z = z_new

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init_asymmetric(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            elapsed = time.monotonic()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # === Phase 1: Fast subgradient Nesterov pre-optimization (90s) ===
    print(f"\n=== Phase 1: Subgradient Nesterov at n={n_base} (score={best_score:.8f}) ===")
    t1 = min(90.0, (_global_deadline - time.monotonic()) * 0.1)
    if t1 > 5:
        a_out, _ = subgradient_nesterov(best_seq, t1, max_steps=200000)
        try_update(a_out, "subgrad_nesterov_p1")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: LP-based Nesterov (main workhorse, 800s) ===
    print(f"\n=== Phase 2: Nesterov LP at n={best_n} (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(800.0, remaining * 0.55)
    if t2 > 30:
        a_out, _ = nesterov_lp_adaptive(best_seq, t2, solver=solver)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Peak-targeted multiplicative refinement (300s) ===
    print(f"\n=== Phase 3: Peak-targeted refinement (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(300.0, remaining * 0.2)
    if t3 > 10:
        a_pt, _ = peak_targeted_refine(best_seq, t3, K=30, max_steps=50000)
        try_update(a_pt, "peak_targeted")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: LSE L-BFGS polishing (350s) ===
    print(f"\n=== Phase 4: LSE L-BFGS polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(350.0, remaining * 0.25)
    if t4 > 20:
        a_lse = lbgfs_logsumexp(best_seq, t4, T_start=80.0, T_end=0.03, n_stages=25)
        try_update(a_lse, "lse_polish")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Second LP round (200s) ===
    print(f"\n=== Phase 5: Nesterov LP round 2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(200.0, remaining * 0.2)
    if t5 > 15:
        a_out2, _ = nesterov_lp_adaptive(best_seq, t5, solver=solver)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final peak-targeted + LSE (100s) ===
    print(f"\n=== Phase 6: Final refinement (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 20:
        t6a = min(50.0, remaining * 0.4)
        if t6a > 5:
            a_pt2, _ = peak_targeted_refine(best_seq, t6a, K=15, max_steps=30000)
            try_update(a_pt2, "peak_targeted_r2")
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            t6b = min(50.0, remaining * 0.5)
            if t6b > 5:
                a_lse2 = lbgfs_logsumexp(best_seq, t6b, T_start=20.0, T_end=0.01, n_stages=15)
                try_update(a_lse2, "lse_final")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 60:
        for n_try in [best_n - 100, best_n + 100, best_n + 200]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 20:
                break
            if n_try < 100:
                continue
            t7 = min(40.0, remaining * 0.3)
            if t7 < 8:
                break
            print(f"  Trying n={n_try}, budget={t7:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_adaptive(init_try, t7, solver=solver, max_iters=100000)
            try_update(a_try, f"len_{n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
