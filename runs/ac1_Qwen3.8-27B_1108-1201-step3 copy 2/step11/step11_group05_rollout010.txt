**Alternative: Frank-Wolfe (Conditional Gradient) for the outer problem**

The outer problem is: min_a max_k (a*a)[k] s.t. a >= 0, sum(a) = 1.

This is a min-max problem. At each iteration:
1. Evaluate the current point: find k* = argmax_k (a*a)[k]
2. Find the "vertex" of the simplex that minimizes the linearized objective: min_b (2 * (a*a)[k* gradient at b direction])

The gradient of (a*a)[k] at a in the direction b is:
`∇_a (a*a)[k] · b = sum_i (2*a_{k-i}) * b_i = 2 * sum_i a_{k-i} * b_i = 2 * (a*b)[k]`

So the Frank-Wolfe step is: find b ∈ simplex that minimizes `2 * (a*b)[k*]`, which is equivalent to finding b that minimizes `(a*b)[k*]`.

`min over simplex of sum_i c_i * b_i = min_i c_i` where `c_i = a_{k*-i}` (for valid i).

`a_new = (1-t) * a + t * e_j` where `j = argmin_i a_{k*-i}` and `t = 2/(step+2)` (FW step size).

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):

    B = rfft(b, fft_size)

def subgradient_polish(a0, time_limit, n, verbose=True):

    lb = 1.0 / (2 * n - 1)
    
    t0 = time.monotonic()

    for step in range(5000000):

            break
        
        b_conv = fast_convolve(a)
        max_b = b_conv.max()
        current_val = max_b
        
        # Find active set
        threshold = max_b * (1.0 - 1e-6)
        active = np.where(b_conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b_conv)])
        
        # Subgradient: average over active set
        g = np.zeros(n)
        for k in active:
            # (a*a)[k] = sum_i a_i * a[k-i]
            # d/d a_j = a[k-j] + a[k-j] = 2*a[k-j] (simplified)
            # More precisely: for j in [0,n-1], if k-j in [0,n-1], add a[k-j]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Polyak step size
        gap = current_val - lb
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        t_polyak = gap / g_norm_sq
        # Cap the step size
        t_polyak = min(t_polyak, 0.1)
        
        # Nesterov momentum (light)
        beta = 0.9
        m = beta * m + (1 - beta) * g
        step_dir = m

        a_new = a - t_polyak * step_dir
        a_new = np.maximum(a_new, 0)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
