def lp_br_scipy(y, time_limit=20.0):
    y = np.asarray(y, dtype=np.float64)
    n = y.size
    if n < 2 or n > 1500: return None
    s=y.sum()
    if s <=0: return None
    y=y/s
    N=2*n-1
    data = np.empty((n,n), dtype=np.float64)
    data[:] = y[:, None]  # each row constant
    try:
       T = diags(data, offsets=np.arange(n), shape=(N,n), format='csc')
    except Exception:
       return None
    mcol = csc_matrix((np.ones(N), (np.arange(N), [0]*N)), shape=(N,1))
    A_ub = hstack([T, -mcol], format='csc')
    b_ub = np.zeros(N)
    A_eq = np.zeros((1,n+1)); A_eq[0,:n]=1.0
    b_eq = np.array([1.0])
    c = np.zeros(n+1); c[-1]=1.0
    bounds = [(0.0, None)]*(n+1)
    options = {'time_limit': max(1.0, float(time_limit))}
    try:
       res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options=options)
    except Exception:
       res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:
       b = res.x[:n].copy(); b=np.clip(b,0,None); s=b.sum()
       if s>1e-15: b/=s
       return b, float(res.x[-1])
    return None

  * FFT-based fast exact / smoothed-overlap optimization in softmax space.
  * Very fine LogSumExp temperature cooling, which is crucial for recent gains.
  * Entropy-multiplicative-weight polishing with adaptive steps and EMA.
  * Diverse low-dimensional starts computed over several lengths.
  * Optional true alternating LP best-response refinement for moderate n.
"""

import time
import operator

import numpy as np

try:
    from scipy.fft import rfft, irfft
except Exception:
    from numpy.fft import rfft, irfft

from scipy.optimize import minimize, linprog
from scipy.sparse import csc_matrix, diags, hstack


# ----------------------------------------------------------------------
# Basic numerical helpers
# ----------------------------------------------------------------------

def fast_conv_self(a):
    a = np.ascontiguousarray(a, dtype=np.float64)
    n = int(a.size)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    N = 1 << max(0, (size - 1).bit_length())
    A = rfft(a, N)
    c = irfft(A * A, N)
    return c[:size]


def fast_cross_grad(a, p):
    """
    Return H_i = sum_j p_{i+j} a_j for i=0..n-1.
    This is the core needed for gradients of max_k conv(a,a).
    """
    a = np.ascontiguousarray(a, dtype=np.float64)
    p = np.ascontiguousarray(p, dtype=np.float64)
    n = int(a.size)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = n + L - 1
    N = 1 << max(0, (size - 1).bit_length())
    A = rfft(a, N)
    P = rfft(p[::-1], N)
    c = irfft(A * P, N)
    return c[n - 1:2 * n - 1][::-1]


def softmax_log(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return np.zeros(0, dtype=np.float64)
    v = v - np.max(v)
    e = np.exp(np.clip(v, -745.0, 0.0))
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    return e / s


def norm_nonneg(a):
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        return None
    s = float(np.sum(a))
    if s <= 1e-15:
        return None
    return a / s


def score_normalized(a):
    a = norm_nonneg(a)
    if a is None:
        return float("inf")
    n = int(a.size)
    c = fast_conv_self(a)
    return float(2.0 * n * float(np.max(c)))


def score_fast(a):
    a = norm_nonneg(a)
    if a is None:
        return float("inf")
    n = int(a.size)
    c = fast_conv_self(a)
    return float(2.0 * n * float(np.max(c)))


def update_best(best_a, best_score, cand, label=None):
    cand = norm_nonneg(cand)
    if cand is None:
        return best_a, best_score
    sc = score_normalized(cand)
    if np.isfinite(sc) and sc < best_score:
        if label is not None:
            print(f"  NEW BEST ({label}): {sc:.10f}, n={len(cand)}")
        return cand, sc
    return best_a, best_score


def formal_base(n):
    y = np.empty(n, dtype=np.float64)
    if n >= 1:
        y[0] = 1.0
    for k in range(1, n):
        y[k] = y[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return y


def interpolate_to(parent, n):
    if parent is None:
        return None
    parent = norm_nonneg(parent)
    if parent is None:
        return None
    if parent.size == n:
        return parent.copy()
    xs = np.linspace(0.0, 1.0, parent.size)
    xn = np.linspace(0.0, 1.0, n)
    a = np.interp(xn, xs, parent)
    return norm_nonneg(a)


def random_fourier_base(n, rng, strength=0.8):
    t = np.linspace(0.0, 1.0, n)
    z = float(rng.uniform(-0.5, 0.5))
    for j in range(1, 7):
        amp = float(rng.uniform(0.0, strength))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z = z + amp * np.cos(2.0 * np.pi * j * t + ph)
    a = softmax_log(z)
    return norm_nonneg(a)


def smooth_noise(n, rng, strength=0.03):
    t = np.linspace(0.0, 2.0 * np.pi, n)
    out = np.zeros(n, dtype=np.float64)
    for _ in range(3):
        f = int(rng.integers(1, 5))
        amp = strength * float(rng.uniform(0.5, 1.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        out += amp * np.sin(f * t + ph)
    return out


def perturb_log(a, rng, strength=0.05):
    a = norm_nonneg(a)
    if a is None:
        return None
    loga = np.log(np.maximum(a, 1e-300))
    loga -= np.max(loga)
    loga = loga + smooth_noise(a.size, rng, strength)
    return norm_nonneg(softmax_log(loga))


# ----------------------------------------------------------------------
# LogSumExp L-BFGS in softmax(z) space
# ----------------------------------------------------------------------

def a_from_z(z, n):
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64) / float(max(1, n))
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    m = max(0.0, float(np.max(z)))
    e = np.empty(n, dtype=np.float64)
    e[:-1] = np.exp(z - m)
    e[-1] = np.exp(-m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def z_from_a(a, n):
    if n <= 1:
        return np.zeros(max(0, n - 1), dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-300)
    z = np.log(a[:-1] / a[-1])
    z = np.where(np.isfinite(z), z, 0.0)
    return np.clip(z, -50.0, 50.0)


class LSEOracle:
    def __init__(self, n, T, deadline):
        self.n = int(n)
        self.T = float(T)
        self.deadline = float(deadline)
        self.last = None
        self.fval = None
        self.gval = None

    def compute(self, z):
        if time.monotonic() > self.deadline:
            return 1e15, np.zeros(int(z.size), dtype=np.float64)

        n = self.n
        T = self.T
        a = a_from_z(z, n)
        c = fast_conv_self(a)
        maxc = float(np.max(c))

        if not np.isfinite(maxc) or maxc <= 0.0:
            obj = 2.0 * n * max(0.0, maxc)
            return float(obj), np.zeros(int(z.size), dtype=np.float64)

        if T <= 1e-300:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
            lse = maxc
        else:
            zc = (c - maxc) / T
            zc -= float(np.max(zc))
            zc = np.clip(zc, -50.0, 0.0)
            expz = np.exp(zc)
            Z = float(np.sum(expz))
            if not np.isfinite(Z) or Z <= 0.0:
                p = np.zeros_like(c)
                p[int(np.argmax(c))] = 1.0
                lse = maxc
            else:
                p = expz / Z
                lse = maxc + T * float(np.log(Z))

        obj = 2.0 * n * lse
        H = fast_cross_grad(a, p)
        G = 4.0 * n * H
        S = float(np.dot(G, a))
        grad = a * (G - S)
        if n > 1:
            g = grad[:-1]
        else:
            g = np.zeros(0, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        return float(obj), g

    def f(self, z):
        if self.last is not None and np.array_equal(z, self.last):
            return self.fval
        val, g = self.compute(z)
        self.last = np.asarray(z, dtype=np.float64).copy()
        self.fval = val
        self.gval = g
        return val

    def g(self, z):
        if self.last is not None and np.array_equal(z, self.last):
            return self.gval
        val, g = self.compute(z)
        self.last = np.asarray(z, dtype=np.float64).copy()
        self.fval = val
        self.gval = g
        return g


def lse_optimize_z(a_init, end, min_T_factor=1e-6, rng=None):
    a = norm_nonneg(a_init)
    if a is None:
        return None, float("inf")
    best_a = a.copy()
    best_score = score_normalized(a)
    n = int(a.size)
    if n < 50:
        return best_a, best_score

    z = z_from_a(a, n)
    t0 = time.monotonic()
    total = max(0.0, float(end) - t0)

    if total <= 5.0:
        return best_a, best_score

    all_factors = [
        0.2, 0.05, 0.01, 0.002, 0.0004,
        0.00008, 0.000015, 0.000003, 0.0000006
    ]

    if total > 180.0:
        factors = all_factors[:8]
    elif total > 90.0:
        factors = all_factors[1:8]
    elif total > 45.0:
        factors = all_factors[2:8]
    else:
        factors = all_factors[4:8]

    for fac in factors:
        if time.monotonic() >= float(end):
            break
        if fac < min_T_factor:
            break
        cmax = float(np.max(fast_conv_self(a)))
        if not np.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(cmax * fac, 1e-15)
        rem = float(end) - time.monotonic()
        if rem < 6.0:
            break

        maxiter = int(min(5000, max(80, rem * 18.0)))
        oracle = LSEOracle(n, T, float(end))
        try:
            res = minimize(
                fun=oracle.f,
                jac=oracle.g,
                x0=z,
                method="L-BFGS-B",
                options={
                    "maxiter": maxiter,
                    "maxls": 50,
                    "ftol": 1e-20,
                    "gtol": 1e-12,
                    "maxfun": max(maxiter * 50, 1000),
                    "disp": False,
                }
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a2 = a_from_z(res.x, n)
                a2 = norm_nonneg(a2)
                if a2 is not None:
                    sc2 = score_normalized(a2)
                    if sc2 < best_score:
                        best_score = sc2
                        best_a = a2.copy()
                        a = a2.copy()
                        z = z_from_a(a, n)
        except Exception:
            pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Multiplicative-weight polishing
# ----------------------------------------------------------------------

def mpw_optimize(a_init, end, mode="exact", rng=None):
    if rng is None:
        rng = np.random.default_rng(20240501)

    a = norm_nonneg(a_init)
    if a is None:
        return None, float("inf")

    best_a = a.copy()
    best_score = score_normalized(a)
    n = int(a.size)
    if n < 20:
        return best_a, best_score

    loga = np.log(np.maximum(a, 1e-300))
    loga -= np.max(loga)
    ema = a.copy()

    t0 = time.monotonic()
    total = max(0.0, float(end) - t0)
    if total <= 5.0:
        return best_a, best_score

    if mode == "smooth":
        factors = [0.2, 0.05, 0.01, 0.002]
        eta_init = 0.55
    else:
        factors = [
            0.02, 0.004, 0.0008, 0.00015,
            0.00003, 0.000006, 0.000001
        ]
        eta_init = 0.35

    cmax = float(np.max(fast_conv_self(a)))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return best_a, best_score

    nf = len(factors)
    for idx, fac in enumerate(factors):
        rem_total = float(end) - time.monotonic()
        if rem_total < 10.0:
            break
        stage_budget = rem_total / float(nf - idx)
        if stage_budget < 6.0:
            stage_budget = rem_total * 0.5
        stage_end = time.monotonic() + stage_budget
        if stage_end > float(end):
            stage_end = float(end)

        T = max(cmax * fac, 1e-15)
        eta = eta_init
        stall = 0
        step = 0

        while time.monotonic() < stage_end:
            c = fast_conv_self(a)
            maxc = float(np.max(c))
            cur = 2.0 * n * maxc

            zc = (c - maxc) / T
            zc -= float(np.max(zc))
            zc = np.clip(zc, -50.0, 0.0)
            expz = np.exp(zc)
            Z = float(np.sum(expz))
            if not np.isfinite(Z) or Z <= 0.0:
                p = np.zeros_like(c)
                p[int(np.argmax(c))] = 1.0
            else:
                p = expz / Z

            H = fast_cross_grad(a, p)
            G = 4.0 * n * H
            G -= float(np.mean(G))

            gnorm = float(np.linalg.norm(G))
            if gnorm < 1e-14:
                break

            delta = -eta * G
            delta -= float(np.mean(delta))
            dnorm = float(np.linalg.norm(delta))
            if dnorm > 0.5:
                delta *= 0.5 / dnorm

            new_log = loga + delta
            a_new = softmax_log(new_log)
           =sc_new = score_fast(a_new)

            if np.isfinite(sc_new) and sc_new < cur:
                a = a_new
                loga = new_log
                cur = sc_new
                ema = ema + 0.10 * (a - ema)

                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a.copy()

                eta = min(eta * 1.035, 1.2)
                stall = 0
            else:
                eta *= 0.78
                stall += 1
                if eta < 1e-4:
                    noise = smooth_noise(n, rng, 0.02)
                    pert_log = loga + noise
                    a_pert = softmax_log(pert_log)
                    sc_pert = score_fast(a_pert)
                    if np.isfinite(sc_pert) and sc_pert < cur - 1e-14:
                        a = a_pert
                        loga = pert_log
                        ema = a.copy()
                        cur = sc_pert
                        if sc_pert < best_score:
                            best_score = sc_pert
                            best_a = a.copy()
                        eta = eta_init * 0.55
                        stall = 0
                    else:
                        break

            step += 1
            if step % 60 == 0:
                sc_ema = score_fast(ema)
                if np.isfinite(sc_ema):
                    if sc_ema < best_score:
                        best_score = sc_ema
                        best_a = ema.copy()
                    if sc_ema < cur - 1e-14:
                        a = ema.copy()
                        loga = np.log(np.maximum(a, 1e-300))
                        loga -= np.max(loga)
                        cur = sc_ema

        cmax = float(np.max(fast_conv_self(a)))
        if not np.isfinite(cmax) or cmax <= 0.0:
            break

    return best_a, best_score

def quick_scan(rng, deadline, parent):
    cands = []

    lengths = [1000, 1200, 1400, 1600, 1800, 2000, 2400]
    if parent is not None:
        lp = int(len(parent))
        if not any(abs(lp - l) <= 120 for l in lengths):
            lengths.append(lp)

    for n in lengths:
        if time.monotonic() > float(deadline) - 25.0:
            break

        base = formal_base(n)
        t = np.linspace(0.0, 1.0, n)
        idx = np.arange(n, dtype=np.float64) + 1.0

        a = norm_nonneg(base)
        if a is not None:
            sc = score_normalized(a)
            if np.isfinite(sc):
                cands.append((sc, a))

        for p in [0.35, 0.45, 0.5, 0.55, 0.65, 0.8]:
            a = norm_nonneg(np.power(idx, -p))
            if a is not None:
                sc = score_normalized(a)
                if np.isfinite(sc):
                    cands.append((sc, a))

        if parent is not None:
            a = interpolate_to(parent, n)
            if a is not None:
                sc = score_normalized(a)
                if np.isfinite(sc):
                    cands.append((sc, a))
                for st in [0.02, 0.05, 0.09]:
                    ap = perturb_log(a, rng, st)
                    if ap is not None:
                        sc = score_normalized(ap)
                        if np.isfinite(sc):
                            cands.append((sc, ap))

        trials = 150 if n <= 1600 else 80
        for _ in range(trials):
            if time.monotonic() > float(deadline) - 25.0:
                break
            kind = int(rng.integers(0, 6))

            if kind == 0:
                c0 = float(rng.uniform(-2.0, 2.0))
                c1 = float(rng.uniform(-2.0, 2.0))
                c2 = float(rng.uniform(-2.0, 2.0))
                a = base * np.exp(c0 * t + c1 * t * t + c2 * t * t * t)

            elif kind == 1:
                amp = float(rng.uniform(0.0, 0.75))
                f = float(int(rng.integers(1, 6)))
                ph = float(rng.uniform(0.0, 6.283185307179586))
                factor = 1.0 + amp * np.sin(2.0 * np.pi * f * t + ph)
                a = base * np.maximum(factor, 0.05)

            elif kind == 2:
                pv = float(rng.uniform(0.2, 0.9))
                qv = float(rng.uniform(-0.5, 0.5))
                eps = float(rng.uniform(0.005, 0.10))
                a = np.power(t + eps, -pv) * np.power(1.0 - t + eps, -qv)

            elif kind == 3:
                pv = float(rng.uniform(0.3, 0.75))
                beta = float(rng.uniform(-1.5, 1.5))
                a = base * np.power(np.maximum(t + 0.01, 0.01), beta) * np.power(idx, -pv)

            elif kind == 4:
                a = random_fourier_base(n, rng, 0.85)

            else:
                if parent is not None:
                    a = interpolate_to(parent, n)
                    if a is not None:
                        a = perturb_log(a, rng, 0.06)
                else:
                    a = base * np.exp(float(rng.uniform(-1.0, 1.0)) * t)

            a = norm_nonneg(a)
            if a is not None:
                sc = score_normalized(a)
                if np.isfinite(sc):
                    cands.append((sc, a))

        if len(cands) > 9000:
            cands.sort(key=operator.itemgetter(0))
            cands = cands[:4500]

    if len(cands) > 0:
        cands.sort(key=operator.itemgetter(0))
        return cands[:40]
    return []

def refine_candidate(a_init, end, rng):
    a = norm_nonneg(a_init)
    if a is None:
        return None, float("inf")

    best_a = a.copy()
    best_score = score_normalized(a)

    now = time.monotonic()
    total = max(0.0, float(end) - now)
    if total <= 10.0:
        return best_a, best_score

    if total > 25.0:
        t1 = min(total * 0.45, 220.0)
        end1 = now + t1
        a1, sc1 = lse_optimize_z(a, end1, min_T_factor=1e-5, rng=rng)
        best_a, best_score = update_best(best_a, best_score, a1, label="lse1")

    rem = float(end) - time.monotonic()
    if rem > 25.0:
        t2 = min(total * 0.35, rem * 0.75, 300.0)
        if t2 < 12.0:
            t2 = rem * 0.45
        end2 = time.monotonic() + t2
        a2, sc2 = mpw_optimize(best_a, end2, mode="exact", rng=rng)
        best_a, best_score = update_best(best_a, best_score, a2, label="mpw")

    rem = float(end) - time.monotonic()
    if rem > 25.0:
        t3 = min(total * 0.25, rem * 0.8, 220.0)
        if t3 < 12.0:
            t3 = rem * 0.5
        end3 = time.monotonic() + t3
        a3, sc3 = lse_optimize_z(best_a, end3, min_T_factor=1e-7, rng=rng)
        best_a, best_score = update_best(best_a, best_score, a3, label="lse2")

    return best_a, best_score

# Optional LP alternating refinement

def lp_br_scipy(y, time_limit=20.0):
    y = np.asarray(y, dtype=np.float64)
    n = int(y.size)
    if n < 2 or n > 1500:
        return None
    s = float(np.sum(y))
    if s <= 1e-15 or not np.isfinite(s):
        return None
    y = y / s

    N = 2 * n - 1
    try:
        data = np.empty((n, n), dtype=np.float64)
        data[:] = y[:, None]
        T = diags(data, offsets=np.arange(n), shape=(N, n), format="csc")
        inds = np.zeros(N, dtype=int)
        mcol = csc_matrix((np.ones(N, dtype=np.float64), (np.arange(N), inds)), shape=(N, 1))
        A_ub = hstack([T, -mcol], format="csc")
        b_ub = np.zeros(N, dtype=np.float64)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)
        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0
        bounds = [(0.0, None)] * (n + 1)

        options = {"time_limit": max(1.0, float(time_limit))}
        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options=options,
            )
        except Exception:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )

        if res is not None and int(res.status) == 0:
            b = np.asarray(res.x[:n], dtype=np.float64).copy()
            b = np.clip(b, 0.0, None)
            sb = float(np.sum(b))
            if sb > 1e-15:
                b /= sb
                return b, float(res.x[-1])
    except Exception:
        return None
    return None


def lp_alternate(a_init, end):
    a = norm_nonneg(a_init)
    if a is None or a.size > 1500:
        return a, score_normalized(a) if a is not None else float("inf")

    x = a.copy()
    y = a.copy()
    ema_x = x.copy()
    ema_y = y.copy()

    best_a = x.copy()
    best_score = score_normalized(x)
    best_a, best_score = update_best(best_a, best_score, y, label="lp_y_init")

    it = 0
    while time.monotonic() < float(end):
        rem = float(end) - time.monotonic()
        if rem < 18.0:
            break

        tl = min(22.0, rem * 0.35)

        if it % 2 == 0:
            rhs = lp_br_scipy(x, tl)
            if rhs is None:
                break
            b, _ = rhs
            alpha = 2.0 / float(it + 2)
            y = (1.0 - alpha) * y + alpha * b
            y = norm_nonneg(y)
            ema_y = 0.90 * ema_y + 0.10 * b
            best_a, best_score = update_best(best_a, best_score, y, label=f"lp_y{it}")
            best_a, best_score = update_best(best_a, best_score, b, label=f"lp_y_br{it}")
            best_a, best_score = update_best(best_a, best_score, ema_y, label=f"lp_ema_y{it}")
        else:
            rhs = lp_br_scipy(y, tl)
            if rhs is None:
                break
            b, _ = rhs
            alpha = 2.0 / float(it + 2)
            x = (1.0 - alpha) * x + alpha * b
            x = norm_nonneg(x)
            ema_x = 0.90 * ema_x + 0.10 * b
            best_a, best_score = update_best(best_a, best_score, x, label=f"lp_x{it}")
            best_a, best_score = update_best(best_a, best_score, b, label=f"lp_x_br{it}")
            best_a, best_score = update_best(best_a, best_score, ema_x, label=f"lp_ema_x{it}")

        it += 1
        if it % 8 == 0:
            print(f"    LP iter {it}: score={best_score:.10f}")

    return best_a, best_score

# Main entry point

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(1.0, float(budget_s) - 35.0)

    best_a = None
    best_score = float("inf")

    # Load previously found sequence if available.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if (
            p.ndim == 1
            and p.size > 0
            and np.all(np.isfinite(p))
            and float(np.sum(p)) > 1e-10
        ):
            parent = p / float(np.sum(p))
            best_a = parent.copy()
            best_score = score_normalized(best_a)
            print(f"Parent: n={len(best_a)}, score={best_score:.10f}")
    except Exception:
        parent = None

    if best_a is None:
        a0 = formal_base(1200)
        best_a = norm_nonneg(a0)
        best_score = score_normalized(best_a)
        print(f"Initial formal n=1200: score={best_score:.10f}")

    # Quick diversity scan.
    print("Phase: quick diversity scan")
    scan_deadline = min(deadline, start + 95.0)
    cands = quick_scan(rng, scan_deadline, parent)
    for sc, a in cands:
        best_a, best_score = update_best(best_a, best_score, a, label="scan")
    print(f"After scan: best={best_score:.10f}, n={len(best_a) if best_a is not None else 0}")

    # Build processing list.
    process = [(best_score, best_a)]
    for sc, a in cands:
        if len(process) >= 12:
            break
        skip = False
        for _, b in process:
            if b is not None and a is not None and len(a) == len(b):
                if np.allclose(a, b, rtol=1e-5, atol=1e-12):
                    skip = True
                    break
        if not skip:
            process.append((sc, a))

    # Allocate time with decaying weights.
    base_weights = [
        0.34, 0.19, 0.12, 0.08,
        0.06, 0.05, 0.04, 0.03,
        0.02, 0.01, 0.01, 0.01
    ]
    if len(process) < len(base_weights):
        weights = np.array(base_weights[:len(process)], dtype=np.float64)
    else:
        tail = np.full(len(process) - len(base_weights), 0.01, dtype=np.float64)
        weights = np.concatenate([np.array(base_weights, dtype=np.float64), tail])
    sw = float(np.sum(weights))
    if sw > 0.0:
        weights = weights / sw
    else:
        weights = np.ones(len(process), dtype=np.float64) / float(max(1, len(process)))

    print("Phase: candidate refinement")
    for i, (sc, a) in enumerate(process):
        rem = deadline - time.monotonic()
        if rem < 90.0:
            break
        w = float(weights[i])
        suffix = float(np.sum(weights[i:]))
        if suffix <= 0.0:
            suffix = 1.0
        alloc = rem * (w / suffix)
        alloc = min(alloc, 380.0)
        if alloc < 25.0:
            alloc = min(rem * 0.3, 60.0)
        if alloc < 18.0:
            break

        end = time.monotonic() + alloc
        print(
            f"  Refine candidate {i + 1}/{len(process)}: "
            f"score={sc:.10f}, n={len(a) if a is not None else 0}, "
            f"alloc={alloc:.0f}s"
        )
        try:
            a_new, sc_new = refine_candidate(a, end, rng)
            best_a, best_score = update_best(best_a, best_score, a_new, label=f"refine{i + 1}")
        except Exception as e:
            print(f"  refine candidate {i + 1} failed: {e}")

    # Optional LP alternating on the best moderate-length sequence.
    rem = deadline - time.monotonic()
    if best_a is not None and len(best_a) <= 1500 and rem > 260.0:
        t_lp = min(rem * 0.28, 430.0)
        end_lp = time.monotonic() + t_lp
        print(f"Phase: LP alternating on best, n={len(best_a)}, budget={t_lp:.0f}s")
        try:
            a_lp, sc_lp = lp_alternate(best_a, end_lp)
            best_a, best_score = update_best(best_a, best_score, a_lp, label="lp")
        except Exception as e:
            print(f"LP phase failed: {e}")
        print(f"After LP: best={best_score:.10f}, n={len(best_a) if best_a is not None else 0}")

    # Final polish.
    rem = deadline - time.monotonic()
    if best_a is not None and rem > 50.0:
        t_mpw = min(rem * 0.45, 280.0)
        end_mpw = time.monotonic() + t_mpw
        print(f"Phase: final exact MPW, budget={t_mpw:.0f}s")
        a_fin, sc_fin = mpw_optimize(best_a, end_mpw, mode="exact", rng=rng)
        best_a, best_score = update_best(best_a, best_score, a_fin, label="final_mpw")

    rem = deadline - time.monotonic()
    if best_a is not None and rem > 35.0:
        t_lse = min(rem * 0.55, 200.0)
        end_lse = time.monotonic() + t_lse
        print(f"Phase: final fine LSE, budget={t_lse:.0f}s")
        a_fin, sc_fin = lse_optimize_z(best_a, end_lse, min_T_factor=1e-7, rng=rng)
        best_a, best_score = update_best(best_a, best_score, a_fin, label="final_lse")

    final = norm_nonneg(best_a)
    if final is None:
        final = np.ones(1200, dtype=np.float64) / 1200.0

    final = np.clip(final, 0.0, 1000.0)
    fs = score_normalized(final)
    print(f"FINAL local score={fs:.10f}, n={len(final)}")

    out = []
    for x in final:
        out.append(float(max(0.0, min(1000.0, x))))
    return out
