def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def subgradient_at_peak(a, top_frac=0.05):
    """Compute subgradient of max(conv(a,a)) w.r.t. a, focusing on top peaks."""
    n = len(a)
    b = fast_convolve(a)
    
    # Select top peaks

    idx = np.argpartition(b, -k)[-k:]

    w_full[idx] = w

    # Using correlation
    g = np.correlate(w_full, a, mode='full')

def mirror_descent_step(a, peak_pos, eta):
    """
    Mirror descent step targeting a specific peak position.
    At peak position k, the cost of a[j] is proportional to a[k-j].
    We reduce high-cost elements multiplicatively.
    """
    n = len(a)
    k = peak_pos
    
    # Cost vector: cost[j] = a[k-j] for valid j
    cost = np.zeros(n)
    for j in range(n):
        ki = k - j
        if 0 <= ki < n:
            cost[j] = a[ki]
    
    # Multiplicative update: a_new[j] = a[j] * exp(-eta * cost[j])
    log_factor = -eta * cost
    log_factor -= log_factor.max()  # For numerical stability
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

    t0 = time.monotonic()
    
    step = 0
    stall = 0
    prev_best = best_score
    eta = lr_init
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        a_nest = y + momentum * (a - y)
        a_nest = np.maximum(a_nest, 0)
        s_nest = a_nest.sum()
        if s_nest > 1e-15:
            a_nest = a_nest / s_nest
        else:
            a_nest = a.copy()
        
        # Compute subgradient at lookahead point
        g, max_b = subgradient_at_peak(a_nest, top_frac)
        
        # Update velocity
        v = momentum * v - eta * g
        
        # Update position
        y = y + v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()
        
        a = y.copy()
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive learning rate
        if stall > 50:
            eta *= 0.9
            stall = 0
        elif stall == 0:
            eta = min(eta * 1.01, lr_init * 2)
        
        step += 1
        if step % 5000 == 0:
            print(f"    NSG step={step} eta={eta:.6f} score={best_score:.8f}")
    
    return best_a, best_score


def adaptive_mirror_descent(a0, time_limit):

        a_new = mirror_descent_step(a, peak_pos, eta)

def lp_refine(a0, time_limit, solver=None):
    """LP-based best response refinement."""
    if not HAS_CVXPY:
        return a0, evaluate_local(a0)
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    # Build convolution matrix (sparse for efficiency)
    # For n=1200, the matrix is 2399 x 1200, which is manageable
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS] if solver else [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    
    t0 = time.monotonic()
    b_val = None
    for sol in solvers_to_try:
        if sol is None:
            continue
        if time.monotonic() - t0 > time_limit * 0.8:
            break
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                break
        except Exception:
            continue
    
    if b_val is None:
        return a0, evaluate_local(a0)
    
    b_val = b_val / b_val.sum()
    
    # Interpolate between a and b_val
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for t in [0.5, 0.7, 0.8, 0.9, 1.0]:
        a_try = (1-t) * a + t * b_val
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        sc = 2.0 * n * fast_convolve(a_try).max()
        if sc < best_score:
            best_score = sc
            best_a = a_try.copy()
    
    return best_a, best_score


def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix."""
    import scipy.sparse as sp
    n = len(a)
    size = 2 * n - 1
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    T = sp.csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def lp_refine_sparse(a0, time_limit, solver=None):

            continue

        try:

                b_val = np.maximum(b_val, 0)
                bv_sum = b_val.sum()
                if bv_sum > 1e-15:

        if sc < best_score:

    return best_a, best_score


def pair_cost_direction(a, peak_pos):
    """
    Compute the pairwise cost direction at a given peak position.
    Returns a direction that reduces the peak most efficiently.
    """
    n = len(a)
    k = peak_pos
    
    # For each element j, its "cost" at peak k is a[k-j] (if valid)
    # We want to move mass from high-cost to low-cost elements
    cost = np.zeros(n)
    for j in range(n):
        ki = k - j
        if 0 <= ki < n:
            cost[j] = a[ki]
    
    # Direction: move from high cost to low cost
    # d[j] = cost[j] - mean(cost) (positive means reduce, negative means increase)
    d = cost - cost.mean()
    
    # Make it a zero-sum direction (preserves sum)
    # Project onto {x : sum(x) = 0, and we can apply a+eta*d >= 0}
    # Simple: just return d, the caller will handle projection
    return d


def directional_refinement(a0, time_limit, n_steps=100000):
    """
    IceShard-style directional refinement:
    At each step, identify the peak, compute the pairwise cost,
    and move in the direction that most reduces the peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.005
    momentum = 0.7
    v = np.zeros(n)
    step = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and step < n_steps:
        b = fast_convolve(a)
        max_b = b.max()
        
        # Get top few peaks
        k = max(1, min(5, int(0.01 * len(b))))
        top_idx = np.argpartition(b, -k)[-k:]
        
        # Compute combined direction
        d = np.zeros(n)
        for pk in top_idx:
            d += pair_cost_direction(a, pk)
        
        # Nesterov momentum
        v = momentum * v - eta * d
        a_new = a + v
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.005, 0.02)
        else:
            stall += 1
            a = a_new  # Still move (exploration)
            if stall > 20:
                eta *= 0.9
                stall = 0
                if eta < 0.0005:
                    break
        
        step += 1
        if step % 10000 == 0:
            print(f"    DR step={step} eta={eta:.6f} score={best_score:.8f}")
    
    return best_a, best_score


def multi_peak_subgradient(a0, time_limit, top_k=5, lr_schedule='decay'):
    """
    Subgradient method targeting multiple peaks simultaneously.
    Uses softmax weighting over top-k peaks for a smooth subgradient approximation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.002
    beta = 0.9  # Momentum
    v = np.zeros(n)
    step = 0
    stall = 0
    temperature = 0.1  # For softmax over peaks
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        
        # Select top-k peaks with softmax weighting
        k = max(1, min(top_k, len(b)))
        idx = np.argpartition(b, -k)[-k:]
        b_top = b[idx]
        
        # Softmax weights (inverse temperature)
        inv_T = 1.0 / temperature
        w_top = np.exp(inv_T * (b_top - b_top.max()))
        w_top = w_top / w_top.sum()
        
        # Full weight vector
        w_full = np.zeros(len(b))
        w_full[idx] = w_top
        
        # Subgradient: g[j] = 2 * sum_k w[k] * a[k-j]
        g = np.correlate(w_full, a, mode='full')
        g = g[n-1:2*n-1] * 2.0
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Momentum update
        v = beta * v - eta * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Don't update a if it got worse (stay put)
            if stall > 30:
                eta *= 0.9
                temperature *= 0.95  # Focus more on the top peak
                stall = 0
                if eta < 0.0001:
                    break
        
        # Anneal temperature
        temperature = max(0.01, temperature * 0.9999)
        
        step += 1
        if step % 10000 == 0:
            print(f"    MPSG step={step} eta={eta:.6f} T={temperature:.4f} score={best_score:.8f}")
    
    return best_a, best_score


def logsumexp_adam(a0, time_limit, T_init=5.0, T_min=0.01, lr_init=0.003):
    """Adam optimization on log-sum-exp approximation in log-space."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    a = a / s
    z = np.log(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    T = T_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        b = fast_convolve(a)
        
        # Log-sum-exp approximation of max
        max_b = b.max()
        exp_b = np.exp((b - max_b) / T)
        lse = max_b + T * np.log(exp_b.sum())
        
        # Score
        score = 2.0 * n * lse
        if lse < best_score / (2.0 * n):
            actual_score = 2.0 * n * b.max()
            if actual_score < best_score:
                best_score = actual_score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Gradient of LSE w.r.t. a
        w = exp_b / exp_b.sum()
        g_a = np.correlate(w, a, mode='full')
        g_a = g_a[n-1:2*n-1] * 2.0
        
        # Convert to z-space
        S = np.dot(g_a, a)
        grad_z = a * (g_a - S)
        
        # Adam update
        lr = lr_init
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()  # Centering
        
        # Anneal temperature
        T = max(T_min, T * 0.9995)
        
        # Adaptive learning rate
        if stall > 100:
            lr *= 0.9
            stall = 0
        elif stall == 0:
            lr = min(lr * 1.001, lr_init * 2)
        
        step += 1
        if step % 10000 == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            cur_score = 2.0 * n * fast_convolve(a_cur).max()
            print(f"    LSE-Adam step={step} T={T:.4f} lr={lr:.6f} best={best_score:.8f} cur={cur_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def find_best_solver():
    """Find a working LP solver."""
    if not HAS_CVXPY:
        return None
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    n = len(a)

    return float(2.0 * n * b.max() / (s * s))


def find_best_lp_solver():

def build_sparse_conv_matrix(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = j + n
        rows.extend(range(k_start, k_end))
        cols.extend([j] * n)
        vals.extend(a[:n].tolist())
    T = sp.csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def solve_lp_best_response(a_fixed, solver, time_limit=30.0):

def multi_peak_subgradient(a0, time_limit, top_k=5, lr_init=0.003):
    """Subgradient targeting top-k peaks with adaptive softmax temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    t0 = time.monotonic()

    beta = 0.85
    v = np.zeros(n)
    temperature = 0.05
    step = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k = max(1, min(top_k, len(b)))
        idx = np.argpartition(b, -k)[-k:]
        b_top = b[idx]
        max_top = b_top.max()

        w_top = np.exp((b_top - max_top) / temperature)
        w_top = w_top / w_top.sum()

        w_full = np.zeros(len(b))
        w_full[idx] = w_top

        g = np.correlate(w_full, a, mode='full')
        g = g[n - 1:2 * n - 1] * 2.0

        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm

        v = beta * v - eta * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()

            best_score = score

    return best_a, best_score


def pairwise_directional_refine(a0, time_limit, n_max_steps=100000):
    """
    IceShard-style: at each step compute pairwise cost at peak and move
    mass from high-cost to low-cost positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.003
    momentum = 0.7
    v = np.zeros(n)
    step = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and step < n_max_steps:
        b = fast_convolve(a)
        k = max(1, min(3, int(0.005 * len(b)) + 1))
        top_idx = np.argpartition(b, -k)[-k:]

        # Compute combined pairwise cost direction
        d = np.zeros(n)
        for pk in top_idx:
            cost = np.zeros(n)
            for j in range(n):
                ki = pk - j
                if 0 <= ki < n:
                    cost[j] = a[ki]
            d += cost - cost.mean()

        # Normalize
        d_norm = np.linalg.norm(d)
        if d_norm > 1e-15:
            d = d / d_norm

        v = momentum * v - eta * d
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()

def mirror_descent_refine(a0, time_limit):

    n = len(a0)

        max_b = b.max()
        peak_pos = int(np.argmax(b))

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < cur_score:
            cur_score = sc
            cur_best = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}) ***")

    # === Phase 1: Nesterov LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} score={cur_score:.10f} ===")
    rem = deadline - time.monotonic()
    if solver is not None and rem > 100:
        t_alloc = min(rem * 0.25, 400.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_r, sc_r = nesterov_lp_alternate(cur_best, t_alloc, solver)
        try_update(a_r, "nlp_1")
    else:
        if rem > 60:
            t_alloc = min(rem * 0.15, 200.0)
            a_r, sc_r = multi_peak_subgradient(cur_best, t_alloc, top_k=5, lr_init=0.003)
            try_update(a_r, "mps_p1")

    # === Phase 2: Pairwise Directional Refinement ===
    print(f"\n=== Phase 2: PDR score={cur_score:.10f} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_r, sc_r = pairwise_directional_refine(cur_best, t_alloc)
        try_update(a_r, "pdr")

    # === Phase 3: Second LP round ===
    print(f"\n=== Phase 3: LP Round 2 score={cur_score:.10f} ===")
    rem = deadline - time.monotonic()
    if solver is not None and rem > 60:
        t_alloc = min(rem * 0.20, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_r, sc_r = nesterov_lp_alternate(cur_best, t_alloc, solver)
        try_update(a_r, "nlp_2")

    # === Phase 4: Fine subgradient ===
    print(f"\n=== Phase 4: Fine MPSG score={cur_score:.10f} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_r, sc_r = multi_peak_subgradient(cur_best, t_alloc, top_k=3, lr_init=0.001)
        try_update(a_r, "fine_mpsg")

    # === Phase 5: Mirror Descent ===
    print(f"\n=== Phase 5: Mirror Descent score={cur_score:.10f} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_r, sc_r = mirror_descent_refine(cur_best, t_alloc)
        try_update(a_r, "mirror")

    # === Phase 6: Try n=1500 ===
    print(f"\n=== Phase 6: Try n=1500 score={cur_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_1500 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_1500:.0f}s")
        init_1500 = interpolate_sequence(cur_best, 1500)
        t_sub = t_1500 * 0.4
        a_1500, _ = multi_peak_subgradient(init_1500, t_sub, top_k=5, lr_init=0.002)
        try_update(a_1500, "mps_n1500")

        # LP on n=1500
        rem = deadline - time.monotonic()
        if solver is not None and rem > 40:
            t_lp1500 = min(rem * 0.3, 120.0)
            a_1500b, _ = nesterov_lp_alternate(cur_best, t_lp1500, solver)
            try_update(a_1500b, "nlp_n1500")

    # === Phase 7: Iterative refinement loop ===
    print(f"\n=== Phase 7: Iterative Refinement score={cur_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        print(f"  Budget: {rem:.0f}s")
        for round_idx in range(30):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 8:
                break
            t_sub = min(rem_sub * 0.08, 25.0)
            if t_sub < 5:
                if rem_sub > 20:
                    a_r, _ = fine_local_search(cur_best, rem_sub * 0.3, rng)
                    try_update(a_r, f"final_fs_{round_idx}")
                else:
                    break
                continue

            if round_idx % 4 == 0:
                a_r, _ = multi_peak_subgradient(cur_best, t_sub, top_k=3, lr_init=0.001)
                try_update(a_r, f"iter_mpsg_{round_idx}")
            elif round_idx % 4 == 1:
                a_r, _ = pairwise_directional_refine(cur_best, t_sub, n_max_steps=50000)
                try_update(a_r, f"iter_pdr_{round_idx}")
            elif round_idx % 4 == 2:
                a_r, _ = mirror_descent_refine(cur_best, t_sub)
                try_update(a_r, f"iter_md_{round_idx}")
            else:
                a_r, _ = fine_local_search(cur_best, t_sub, rng)
                try_update(a_r, f"iter_fs_{round_idx}")

        # Final LP if time
        rem = deadline - time.monotonic()
        if solver is not None and rem > 15:
            a_r, _ = nesterov_lp_alternate(cur_best, min(rem * 0.5, 30.0), solver)
            try_update(a_r, "final_lp")

    # === Phase 8: Final fine local search ===
    print(f"\n=== Phase 8: Final Local Search score={cur_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 5:
        a_r, _ = fine_local_search(cur_best, min(rem * 0.8, 30.0), rng)
        try_update(a_r, "final_local")

    # Final verification
    if cur_best is not None:
        fs = evaluate_local(cur_best)
        if np.isfinite(fs) and fs < cur_score:
            cur_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in cur_best])
        print(f"\nFINAL SCORE (local): {cur_score:.10f} (n={len(cur_best)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.10f}")

    if cur_best is not None:
        best_seq = cur_best
    else:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
