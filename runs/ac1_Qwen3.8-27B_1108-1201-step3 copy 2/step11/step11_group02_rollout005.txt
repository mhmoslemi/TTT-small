def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def score_fn(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()

        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    data = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    return csr_matrix((data, (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))), shape=(size, n))

def smooth_obj_grad(a, T):
    """Compute LogSumExp smoothed objective and gradient using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Compute convolution via FFT
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]

    # Softmax weights (temperature T)
    max_c = conv.max()
    exp_c = np.exp(conv / T - max_c / T)
    w = exp_c / exp_c.sum()

    # Gradient: (4n/T) * sum_{j=0}^{n-1} w[m+j] * a[j] for m=0..n-1
    # This is the cross-correlation of w and a
    fft2_size = 1 << (size + n - 2).bit_length()
    W_padded = np.zeros(fft2_size)
    W_padded[:size] = w
    A_padded = np.zeros(fft2_size)
    A_padded[:n] = a
    Wf = rfft(W_padded)
    Af = rfft(A_padded)
    cc = irfft(np.conj(Wf) * Af, fft2_size)
    grad = cc[:n] * (4.0 * n) / T

    # Objective value
    lse_val = max_c / T + np.log(exp_c.sum())
    obj = (2.0 * n / T) * lse_val

    return obj, grad


def logsumexp_lbgfs(a0, time_limit, use_sym=True):
    """L-BFGS on LogSumExp smoothed objective with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    if use_sym:
        a = symmetrize(a)
        a = np.maximum(a, 0)
        a = a / a.sum()

    t0 = time.monotonic()
    best_a = a.copy()
    best_true_score = score_fn(a)

    # Temperature schedule
    T_stages = [30.0, 15.0, 8.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 3:
            break

        elapsed_slot = time_limit - (time.monotonic() - t0)
        remaining_stages = len(T_stages) - T_stages.index(T)
        iters = max(30, int(2000 * elapsed_slot / (time_limit / max(1, len(T_stages)))))
        iters = min(iters, 5000)

        def func(z_inner):
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            if use_sym:
                a_trial = symmetrize(a_trial)
                a_trial = np.maximum(a_trial, 0)
                st2 = a_trial.sum()
                if st2 > 1e-15:
                    a_trial = a_trial / st2
            obj, _ = smooth_obj_grad(a_trial, T)
            # Track true score
            nonlocal best_true_score, best_a
            ts = score_fn(a_trial)
            if ts < best_true_score - 1e-14:
                best_true_score = ts
                best_a = a_trial.copy()
            return obj

        def jac(z_inner):
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z_inner)
            a_trial = a_trial / st
            if use_sym:
                a_trial = symmetrize(a_trial)
                a_trial = np.maximum(a_trial, 0)
                st2 = a_trial.sum()
                if st2 > 1e-15:
                    a_trial = a_trial / st2
            _, grad = smooth_obj_grad(a_trial, T)
            return grad

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters * 500,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
            if use_sym:
                a = symmetrize(a)
                a = np.maximum(a, 0)
                a = a / a.sum()
        except Exception as e:
            print(f"    LSE-LBFGS error at T={T}: {e}")

    return best_a, best_true_score


def peak_deflation(a0, time_limit, use_sym=True):
    """Coordinate descent that specifically targets reducing the max peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    if use_sym:
        a = symmetrize(a)
        a = np.maximum(a, 0)
        a = a / a.sum()

    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0

    it = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        # Find all near-max peaks
        threshold = max_b * 0.9995
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:5]:
            if improved or time.monotonic() - t0 > time_limit:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]

            # Try reducing top contributors
            n_try = min(12, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:n_try]]

            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                # Try different reduction factors
                for factor in [0.999, 0.995, 0.99, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    diff = a_trial[pos] * (1.0 - factor)
                    a_trial[pos] *= factor
                    if diff < 1e-16:
                        continue
                    # Redistribute to non-contributing positions
                    mask = np.ones(n, dtype=bool)
                    if use_sym and pos < n:
                        mask[n - 1 - pos] = False
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    if use_sym:
                        a_trial = symmetrize(a_trial)
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()

                    sc = score_fn(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Try simultaneous reduction of multiple top contributors
                peak_k = int(np.argmax(fast_convolve(a)))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:6]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.997
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                if use_sym:
                    a_trial = symmetrize(a_trial)
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                sc = score_fn(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 1000:
                    # Random restart perturbation
                    noise = rng.standard_normal(n) * 0.01
                    a = a + noise
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    if use_sym:
                        a = symmetrize(a)
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                    stall = 0
            elif stall > 50:
                # Small random perturbation
                noise = rng.standard_normal(n) * 0.003
                a_trial = a + noise
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                if use_sym:
                    a_trial = symmetrize(a_trial)
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                sc = score_fn(a_trial)
                if sc < best_score + 0.01:  # Allow slight increase for exploration
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
        else:
            stall = 0

        it += 1
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDef iter={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def mirror_descent_subgrad(a0, time_limit, use_sym=True, eta0=0.05):
    """Mirror descent using subgradient of the max function."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    if use_sym:
        a = symmetrize(a)
        a = np.maximum(a, 0)
        a = a / a.sum()

    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    eta = eta0
    no_improve = 0

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find active set (peaks close to max)
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient (average of gradients at active peaks)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += 2.0 * a[k - j]
        g /= len(active)

        # Normalize gradient
        g_norm = g / max(np.sum(np.abs(g)), 1e-15)

        # Mirror descent step (exponential family)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum
        if use_sym:
            a_new = symmetrize(a_new)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        a = a_new

        sc = score_fn(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.05, 1.0)  # Gentle increase
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                eta = max(eta * 0.9, 0.001)  # Decrease step size

        if no_improve >= 500:
            # Perturb and reset
            noise = rng.standard_normal(n) * 0.015
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            if use_sym:
                a = symmetrize(a)
                a = np.maximum(a, 0)
                a = a / a.sum()
            no_improve = 0
            eta = eta0

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternation, LogSumExp L-BFGS, peak deflation, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = score_fn(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        a = np.cos(np.pi * x / 2) ** 2
        a = a / a.sum()
        best_seq = a
        best_score = score_fn(a)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: LP Alternation (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternation (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternation(best_seq, t_alloc, solver=solver, use_sym=True)
        try_update(a_lp, "LP_sym")

        # Also try without symmetrization
        rem2 = deadline - time.monotonic()
        t_alloc2 = min(rem2 * 0.10, 150.0)
        if t_alloc2 > 20:
            a_lp2, _ = lp_alternation(best_seq, t_alloc2, solver=solver, use_sym=False)
            try_update(a_lp2, "LP_nosym")

    # === Phase 2: LogSumExp L-BFGS ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb, _ = logsumexp_lbgfs(best_seq, t_alloc, use_sym=True)
        try_update(a_lb, "LSE_sym")

        # Try without symmetry
        rem2 = deadline - time.monotonic()
        t_alloc2 = min(rem2 * 0.08, 100.0)
        if t_alloc2 > 20:
            a_lb2, _ = logsumexp_lbgfs(best_seq, t_alloc2, use_sym=False)
            try_update(a_lb2, "LSE_nosym")

    # === Phase 3: Mirror Descent with Subgradient ===
    print(f"\n=== Phase 3: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_subgrad(best_seq, t_alloc, use_sym=True, eta0=0.04)
        try_update(a_md, "MD_subgrad")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        candidate_ns = sorted(set([
            max(400, n_best - 300),
            max(500, n_best + 200),
            max(400, int(n_best * 1.3)),
            max(400, int(n_best * 0.7))
        ]))
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, ~{per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 15:
                break
            if abs(n_try - n_best) < 50:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 10)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            # Quick LP alternation at this n
            a_sub, _ = lp_alternation(a_interp, min(t_sub * 0.6, 100), solver=solver, use_sym=True)
            try_update(a_sub, f"multi_n_{n_try}_lp")
            # Then L-BFGS polish
            rem_sub = deadline - time.monotonic() - 5
            t_sub2 = min(t_sub * 0.4, rem_sub)
            if t_sub2 > 10:
                a_sub2, _ = logsumexp_lbgfs(a_sub, t_sub2, use_sym=True)
                try_update(a_sub2, f"multi_n_{n_try}_lb")

    # === Phase 5: Peak Deflation (targeted coordinate descent) ===
    print(f"\n=== Phase 5: Peak Deflation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pd, _ = peak_deflation(best_seq, t_alloc, use_sym=True)
        try_update(a_pd, "peak_def")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_fin, _ = lp_alternation(best_seq, t_alloc, solver=solver, use_sym=True)
        try_update(a_fin, "final_lp")

    # === Phase 7: Final coordinate descent ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        a_cd, _ = peak_deflation(best_seq, t_alloc, use_sym=False)
        try_update(a_cd, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = score_fn(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
