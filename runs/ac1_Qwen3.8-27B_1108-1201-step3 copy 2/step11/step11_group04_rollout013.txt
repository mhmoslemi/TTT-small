def spectral_flatten(a, n_iters=100, blend=0.3, deadline=None):
    """Spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        mags = np.abs(F)
        non_dc = mags[1:L//2+1]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a

def build_conv_matrix(a):
    """Build matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_best_response_lp(a_fixed):
    """Given fixed a (sum=1, nonneg), find b (sum=1, nonneg) minimizing max_k (a*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T_mat = build_conv_matrix(a_fixed)
    
    # LP: min t s.t. T @ b <= t, sum(b) = 1, b >= 0
    # Use scipy's linprog
    from scipy.optimize import linprog
    
    # Variables: b[0], ..., b[n-1], t
    # Total: n+1 variables
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality: T @ b - t <= 0
    # [T_mat, -1] @ [b; t] <= 0
    A_ub = np.hstack([T_mat, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.ones(1)
    
    # Bounds: b >= 0, t unbounded (but will be >= 0 from constraints)
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.status == 0:
            b_val = res.x[:n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
            return b_val, res.x[n]
    except Exception:
        pass
    
    return None, None

def lp_alternate(a0, time_limit, n_max_iters=100000):
    """LP alternating method with Nesterov acceleration."""

    y = a.copy()
    lp_times = []
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

                break
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)

            y = y / sy
        else:
            y = a.copy()

        a = np.maximum(a, 0)
        sa = a.sum()

        score_a = 2.0 * n * fast_convolve(a).max()

def structured_init(n, seed):
    """Create structured initial sequence."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 - np.abs(2 * x - 1)
    elif choice == 8:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Super-Gaussian
        a = np.exp(-4 * np.abs(x - 0.5) ** 4)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    max_scaled = scaled.max()
    exp_terms = np.exp(scaled - max_scaled)
    s = exp_terms.sum()
    lse = max_b + T * (max_scaled + np.log(s))
    obj = 2.0 * n * lse
    w = exp_terms / s
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

    if time.monotonic() > deadline_ref[0]:
