def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """Gerchberg-Saxton spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = normalize(a)

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):

            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    prev_best = best_score
    stall_count = 0
    iter_count = 0
    best_iter = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val = solve_lp_best_response(y)
        if b_new is None:
            continue

        iter_count += 1

        # Nesterov dual averaging with alpha = 2/(t+2)
        alpha = 2.0 / (it + 2)

        # Update dual variable y
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Update primal variable a (running average)
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            best_iter = it + 1

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            best_iter = it + 1

        # Stall detection and restart
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                # Adaptive perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                strength = 0.01 + 0.01 * min(stall_count // 15, 3)
                noise = strength * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = a + noise
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                stall_count = 0

            if verbose and (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                rate = iter_count / elapsed if elapsed > 0 else 0
                print(f"    NLP: iter={it+1}, t={elapsed:.0f}s, rate={rate:.0f}it/s, "
                      f"score={best_score:.8f}, best_at_iter={best_iter}")

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    NLP done: {iter_count} iters in {elapsed:.1f}s, score={best_score:.8f}")
    return best_a, best_score


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. log-domain z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s

    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    w_sum = exp_terms.sum()
    if w_sum < 1e-300:
        w = np.ones(len(b)) / len(b)
    else:
        w = exp_terms / w_sum

    lse = max_b + T_temp * np.log(max(w_sum, 1e-300))
    obj = 2.0 * n * lse

    # Gradient: grad_a[j] = 2 * sum_k w_k * (a_{k-j} + a_{j-k})
    # = 2 * (w * a) + 2 * (w_rev * a) via convolution
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # conv(w, a) gives sum_k w_k * a_{mk - k} ... need to be careful
    # Actually: grad_a[j] = 2 * sum_k w_k * a_{k-j} + 2 * sum_k w_k * a_{j-k}
    # First term: (w convolved with reversed a) or (reversed w) convolved with a
    # Let's use: sum_k w_k * a_{k-j} = (w * a_rev)[len(w)-1-j] where a_rev = a[::-1]
    # Simpler: just compute via FFT cross-correlation

    # sum_k w_k * a_{k-j}: this is the cross-correlation of w and a at lag j
    # = ifft(conj(fft(w)) * fft(a))[j] ... but need full correlation
    # Actually: cross_corr[j] = sum_k w[k] * a[k+j] (for k+j < n)
    # Using FFT: ifft(fft(w_padded) * conj(fft(a_padded))) gives sum_k w[k] * conj(a[k-lag])
    # Since a is real: sum_k w[k] * a[k-lag]

    # Let me just do it directly for clarity (n is at most ~2000, so O(n^2) is fine for the gradient)
    # Actually no, let's use FFT for efficiency.

    # sum_k w_k * a_{k-j}: pad both to size, use cross-correlation via FFT
    # cross_corr[m] = sum_k w[k+m] * a[k] = sum_k a[k] * w[k+m]
    # Using: ifft(fft(w_full) * conj(fft(a_full))) gives sum_k w[k] * a[k-m]
    # We want sum_k w[k] * a[k-j] for j = 0..n-1

    w_full = np.zeros(fft_size)
    w_full[:len(w)] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    cross1 = irfft(fft(w_full) * np.conj(fft(a_full), fft_size=fft_size))
    # cross1[m] = sum_k w[k] * a[k-m] (circular)
    # We want sum_k w[k] * a[k-j] = cross1[j] for j >= 0
    # But we need linear, not circular. Since w has length 2n-1 and a has length n,
    # and fft_size >= 3n-2, the linear result is in the right positions.

    # Actually let me just use a simpler approach:
    # grad_a[j] = 2 * sum_k w[k] * (a[k-j] if 0<=k-j<n else 0) + 2 * sum_k w[k] * (a[j-k] if 0<=j-k<n else 0)

    # First sum: for fixed j, sum over k where 0 <= k-j < n, i.e., j <= k <= j+n-1
    #   = sum_{m=0}^{n-1} w[j+m] * a[m]
    # Second sum: for fixed j, sum over k where 0 <= j-k < n, i.e., j-n+1 <= k <= j
    #   = sum_{m=0}^{n-1} w[j-m] * a[m] (with w[negative] = 0)

    # First sum is the cross-correlation of w and a at lag j
    # Let's compute both using FFT

    # method: use scipy's convolve or just FFT
    # cross1[j] = sum_{m} w[j+m] * a[m] = (w cross-correlate a)[j]
    # Using FFT: ifft(conj(fft(w)) * fft(a)) gives sum_k w[k] * a[k+n-1-j] ... this is getting confusing.

    # Let me just do it the simple O(n^2) way since n <= 2000
    # Actually for n=1200, O(n^2) = 1.44M ops per gradient eval. L-BFGS might call
    # gradient 50-100 times per iteration, and we might have 1000 iterations.
    # That's 1000 * 100 * 1.44M = 144B ops. Too slow!

    # Let me use FFT properly.
    # We want: g1[j] = sum_{m=0}^{n-1} w[j+m] * a[m] for j = 0, ..., n-1
    # and:        g2[j] = sum_{m=0}^{n-1} w[j-m] * a[m] for j = 0, ..., n-1 (w[neg]=0)

    # g1[j] = sum_m w[j+m] * a[m]: This is the cross-correlation, can be done by
    #   reversing w: w_rev = w[::-1], then g1 = conv(w_rev_padded, a)[offset]
    #   Specifically: g1[j] = sum_m w_rev[len(w)-1-j-m] * a[m] ... still confusing.

    # Let me use a different formulation:
    # g1[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    # Define w_ext of length 2n-1 (same as w). Then g1[j] is the "offset dot product".
    # This is equivalent to: (w_ext * a_reversed)[(2n-2) - j] where * is linear convolution.
    # Let a_rev = a[::-1] (length n).
    # conv(w_ext, a_rev)[k] = sum_i w_ext[i] * a_rev[k-i]
    # For k = (2n-2) - j: sum_i w_ext[i] * a_rev[2n-2-j-i]
    #   = sum_i w_ext[i] * a[n-1-(2n-2-j-i)]
    #   = sum_i w_ext[i] * a[i+j-n+1]
    # This is sum over i where 0 <= i+j-n+1 < n, i.e., n-1-j <= i <= n-1
    #   = sum_{i=n-1-j}^{n-1} w_ext[i] * a[i+j-n+1]
    # Let m = i+j-n+1: = sum_{m=0}^{j} w_ext[n-1-m+j] * a[m] ... not quite what we want.

    # OK let me just use the direct formula with numba-like vectorization:
    # For the gradient, I'll compute it as:
    # grad_a = 2 * (elementwise correlation terms)
    # Using the fact that the autocorrelation gradient is:
    # d(conv(a,a)_k)/d(a_j) = a_{k-j} + a_{k-j} (when k!=2j) or 2*a_{k/2} (when k=2j)
    # So: grad_a[j] = 2 * sum_k w_k * a_{|k-j|} ... NO that's wrong.
    # Correct: grad_a[j] = sum_k w_k * (a_{k-j} + a_{j-k}) where a_i=0 outside [0,n-1]
    #          = sum_k w_k * a_{k-j} (for k>=j) + sum_k w_k * a_{j-k} (for k<j)
    #          = sum_{m=0}^{n-1-j} w_{j+m} * a_m + sum_{m=1}^{j} w_{j-m} * a_m
    #          (the m=0 term in the second sum is w_j * a_0 which is already in the first)
    #          Wait: second sum is sum_{k=0}^{j-1} w_k * a_{j-k} = sum_{m=1}^{j} w_{j-m} * a_m
    #          First sum: sum_{k=j}^{2n-2} w_k * a_{k-j} = sum_{m=0}^{min(n-1, 2n-2-j)} w_{j+m} * a_m
    #          Since j < n and 2n-2-j >= n-1 when j <= n-1, the first sum is sum_{m=0}^{n-1} w_{j+m} * a_m

    # So: grad_a[j] = sum_{m=0}^{n-1} w_{j+m} * a_m + sum_{m=1}^{j} w_{j-m} * a_m
    #              = a_0 * w_j + sum_{m=1}^{min(j,n-1)} (w_{j+m} + w_{j-m}) * a_m + (j < n-1: w_{j+n-1} * a_{n-1} if j+n-1 < 2n-1)

    # This is hard to vectorize simply. Let me just use the FFT approach:
    # Define w_ext (length 2n-1) and compute:
    # h[j] = sum_{m=0}^{n-1} w_ext[j+m] * a[m]  (this is the "sliding window" sum)
    # This is the cross-correlation of w_ext and a, which can be computed as:
    # h = ifft(conj(fft(w_ext_padded)) * fft(a_padded)) ... let me verify.
    # ifft(conj(W) * A)[j] = sum_k w_ext[k] * a[k-j] (circular)
    # For linear: we need j such that k-j is in [0,n-1] and k is in [0,2n-2].
    # h[j] = sum_k w_ext[k] * a[k-j] = sum_{k=j}^{j+n-1} w_ext[k] * a[k-j] = sum_{m=0}^{n-1} w_ext[j+m] * a[m]
    # Yes! So h = ifft(conj(fft(w_ext_padded)) * fft(a_padded)) gives us what we want
    # (in the positions j = 0..n-1, as long as there's no circular wraparound).

    w_ext = np.zeros(fft_size)
    w_ext[:2*n-1] = w  # w has length 2n-1
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a

    W_ext = rfft(w_ext, fft_size)
    A_pad = rfft(a_pad, fft_size)

    # h[j] = sum_k w_ext[k] * a_pad[k-j] = ifft(conj(W_ext) * A_pad)[j]
    h = irfft(np.conj(W_ext) * A_pad, fft_size)[:n]

    # Now grad_a[j] = h[j] + (second sum)
    # Second sum: sum_{m=1}^{j} w_{j-m} * a[m] = sum_{m=0}^{j-1} w_ext[j-m] * a[m] (w_ext[0]=w[0])
    # This is: sum_k w_ext[k] * a[j-k] for k=0..j-1
    # = ifft(W_ext * conj(A_pad))[j] ... let me verify.
    # ifft(W * conj(A))[j] = sum_k w_ext[k+j] * a[k] (circular)
    # For linear: sum_k w_ext[k+j] * a[k] for k=0..n-1 = sum_{m=j}^{j+n-1} w_ext[m] * a[m-j]
    # That's not quite right either.

    # Let me try: ifft(W_ext * conj(A_pad))[j] = sum_k a[k] * w_ext[(k-j) mod fft_size]
    # For no wraparound: = sum_k a[k] * w_ext[k-j] = sum_{k=j}^{j+n-1} a[k] * w_ext[k-j]
    # Let m = k-j: = sum_{m=0}^{n-1} a[m+j] * w_ext[m]
    # That's the same as h but with a and w swapped... not what we want.

    # OK, the second sum is: sum_{m=1}^{j} w_{j-m} * a[m]
    # = sum_{k=0}^{j-1} w_ext[k] * a[j-k]
    # Let m = j-k: = sum_{m=1}^{j} w_ext[j-m] * a[m]
    # This is the cross-correlation of w_ext and a at lag -j (or a and w_ext at lag j).
    # = ifft(W_ext * conj(A_pad))[?] ...

    # Actually: ifft(W_ext * conj(A_pad))[j] = sum_k w_ext[k] * conj(a[k-j])
    # For real a: = sum_k w_ext[k] * a[k-j]
    # For k-j in [0,n-1]: = sum_{k=j}^{j+n-1} w_ext[k] * a[k-j] = sum_{m=0}^{n-1} w_ext[j+m] * a[m]
    # That's h again! So ifft(W * conj(A)) and ifft(conj(W) * A) give the same for real signals?
    # No: ifft(conj(W)*A)[j] = sum_k a[k] * w_ext[k-j] (by the correlation theorem)
    #    ifft(W*conj(A))[j] = sum_k w_ext[k] * a[k-j]
    # These are the same! (just relabeling the sum index)
    # So I only need one FFT convolution.

    # Therefore: h[j] = sum_{m=0}^{n-1} w_ext[j+m] * a[m] (first sum, what we want for term 1)
    # And the second sum sum_{m=1}^{j} w_ext[j-m] * a[m] is:
    #   Let's call it g[j]. g[j] = sum_{k=0}^{j-1} w_ext[k] * a[j-k]
    #   = (w_ext cross-correlate a)[j] evaluated differently...
    #   Actually g[j] = sum_k w_ext[k] * a[j-k] for k < j
    #   The full cross-correlation (w_ext ⋆ a)[j] = sum_k w_ext[k] * a[j+k]
    #   Hmm, that's not it either.

    # Let me just note that for the autocorrelation gradient:
    # b_k = sum_i a_i * a_{k-i}
    # db_k/d(a_j) = a_{k-j} + a_{k-j} (counting both positions) if k != 2j
    #            = 2*a_j if k = 2j
    # In general: db_k/d(a_j) = a_{k-j} + a_{j-k} (with a_i = 0 for i < 0 or i >= n)
    #   where a_{k-j} means: if 0 <= k-j < n then a[k-j] else 0
    #   and a_{j-k} means: if 0 <= j-k < n then a[j-k] else 0
    #
    # So: grad_a[j] = sum_k w_k * (a_{k-j} + a_{j-k})
    #   = sum_k w_k * a_{k-j} (for 0<=k-j<n) + sum_k w_k * a_{j-k} (for 0<=j-k<n)
    #   = sum_{k=j}^{j+n-1} w_k * a_{k-j} + sum_{k=0}^{j} w_k * a_{j-k}
    #   (second sum: k from max(0,j-n+1) to j, but since j < n, it's k from 0 to j)
    #
    # First term: sum_{m=0}^{n-1} w_{j+m} * a_m = h[j] (computed above)
    # Second term: sum_{m=0}^{j} w_{j-m} * a_m
    #   Note: when m=0: w_j * a_0. This is included in h[j] as the m=0 term? No!
    #   h[j] = sum_{m=0}^{n-1} w_{j+m} * a_m, the m=0 term is w_j * a_0.
    #   Second term m=0: w_j * a_0. So they share the m=0 term!
    #   But wait: in the original formula, when k=j: a_{k-j}=a_0 and a_{j-k}=a_0.
    #   So db_j/d(a_j) includes 2*a_0? No:
    #   b_j = sum_i a_i * a_{j-i}. The term a_j appears when i=j (giving a_j * a_0)
    #   and when j-i=j, i.e., i=0 (giving a_0 * a_j). So yes, db_j/d(a_j) = 2*a_0 + ...
    #   Wait: b_j = sum_i a_i * a_{j-i}. d(b_j)/d(a_j) = a_0 (from i=j term) + a_0 (from i=0 term, since j-0=j)
    #   Only if j != 0. If j=0: b_0 = a_0^2, db_0/d(a_0) = 2*a_0. And a_{j-k} with j=k=0 gives a_0.
    #   So for j=0: first term = w_0 * a_0, second term = w_0 * a_0. Total = 2*w_0*a_0. Correct!
    #
    # So the formula grad_a[j] = h[j] + g[j] where:
    #   h[j] = sum_{m=0}^{n-1} w_{j+m} * a_m (first term, includes m=0: w_j*a_0)
    #   g[j] = sum_{m=0}^{j} w_{j-m} * a_m (second term, includes m=0: w_j*a_0)
    #   Total at m=0: 2*w_j*a_0. Correct (matches the 2*a_0 contribution).
    #
    # Now I need to compute g[j] = sum_{m=0}^{j} w_{j-m} * a_m efficiently.
    # g[j] = (w_ext convolved with a)[j] (linear convolution, but only first n terms)
    # Because (w * a)[j] = sum_m w[m] * a[j-m] = sum_{m=0}^{j} w[m] * a[j-m]
    # Let k = j-m: = sum_{k=0}^{j} w[j-k] * a[k]. Yes! That's exactly g[j].
    #
    # So g = linear_convolution(w_ext, a)[:n]

    # We already have W_ext and A_pad from above.
    # g = irfft(W_ext * A_pad, fft_size)[:n]

    g = irfft(W_ext * A_pad, fft_size)[:n]

    grad_a = 2.0 * (h + g)

    # Now convert to z-gradient (log domain)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, 2.0 * n * grad_z


def lbgfs_obj_wrapper(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_temp)
    return obj


def lbgfs_jac_wrapper(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_temp)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        iters = max(20, min(maxiter_per_stage, int(remaining * 30 / max(1, stages_left))))

        def func(z_in):
            return lbgfs_obj_wrapper(z_in, n, stage_T, deadline)

        def jac(z_in):
            return lbgfs_jac_wrapper(z_in, n, stage_T, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Update a periodically for tracking
        a = np.exp(z - z.max())
        s = a.sum()
        if s > 1e-15:
            a = a / s

    a_final = np.exp(z - z.max())
    s = a_final.sum()
    if s > 1e-15:
        a_final = a_final / s
    else:
        a_final = normalize(a0)

    return a_final


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp = f_interp / s
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

        best_score = evaluate_local(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}, n={n_best}")

    # === Phase 0: Quick spectral flattening (10s) ===
    print(f"\n=== Phase 0: Spectral Flattening ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_spec = min(rem * 0.02, 10.0)
        a_spec = spectral_flatten(best_seq, n_iters=200, blend=0.3,
                                  deadline=time.monotonic() + t_spec)
        local_update(a_spec, "spectral_init")

    # === Phase 1: Main Nesterov LP (60% of time) ===
    print(f"=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp1 = rem * 0.55
    print(f"  Budget: {t_lp1:.0f}s")
    a_lp, _ = nesterov_lp_refine(best_seq, t_lp1, max_iters=500000, verbose=True)
    local_update(a_lp, "nesterov_lp_main")

    # === Phase 2: L-BFGS LogSumExp polishing (20% of time) ===
    print(f"=== Phase 2: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = rem * 0.25
    if t_lbfgs > 20:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
        print(f"  Budget: {t_lbfgs:.0f}s, {len(T_stages)} T-stages")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages, maxiter_per_stage=3000)
        local_update(a_lb, "lbgfs_polish")

    # === Phase 3: Second LP round from improved starting point (10%) ===
    print(f"=== Phase 3: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = rem * 0.15
    if t_lp2 > 15:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, max_iters=200000, verbose=False)
        local_update(a_lp2, "nesterov_lp_r2")

    # === Phase 4: Try different n values (10%) ===
    print(f"=== Phase 4: Multi-n exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_candidates = [1000, 1500, 2000]
    n_candidates = [x for x in n_candidates if x != n_best]

    for n_try in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_sub = min(rem * 0.3, 80.0)
        if t_sub < 15:
            continue
        print(f"  Trying n={n_try}, budget={t_sub:.0f}s")
        init_try = interpolate_sequence(best_seq, n_try)
        init_try = spectral_flatten(init_try, n_iters=30, blend=0.3)
        a_try, _ = nesterov_lp_refine(init_try, t_sub * 0.8, max_iters=100000, verbose=False)
        local_update(a_try, f"lp_n{n_try}")

        # Quick L-BFGS if this n is promising
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == n_try and rem > 20:
            T_st2 = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_try2 = lbgfs_refine(best_seq, min(rem * 0.3, 30.0), T_st2, maxiter_per_stage=2000)
            local_update(a_try2, f"lbgfs_n{n_try}")

    # === Phase 5: Perturbation escape + quick re-optimize (5%) ===
    print(f"=== Phase 5: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 15:
                break
            strength = 0.01 + 0.01 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            a_sp = spectral_flatten(a_pert, n_iters=15, blend=0.3)
            t_sub = min(rem_sub * 0.25, 30.0)
            if t_sub > 8:
                a_opt, _ = nesterov_lp_refine(a_sp, t_sub, max_iters=50000, verbose=False)
                local_update(a_opt, f"pert_r{trial}")

    # === Phase 6: Final L-BFGS polish ===
    print(f"=== Phase 6: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_fin = min(rem * 0.5, 30.0)
        T_stages_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003]
        a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin, maxiter_per_stage=3000)
        local_update(a_fin, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")

        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
