def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_sparse_conv_matrix(a):
    n = len(a)
    T = lil_matrix((2*n-1, n))

    return T.tocsr()

def build_q_sparse(k, n):
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):

        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

def iterative_active_qcqp(a, solver, time_limit, max_iter=30):

def lse_obj_grad(z, n, T_temp):
    """LogSumExp objective and gradient."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_sparse_conv_matrix(a):
    """Build (2n-1) x n sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    T = lil_matrix((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T.tocsr()


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_lp_solver():
    """Find a working LP/QP solver, preferring fast ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def lp_best_response(a, M, solver):
    """Solve max sum(b) s.t. T_a @ b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_sparse_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    solvers = [solver, cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP, cp.MOSEK, cp.GUROBI]
    seen = set()
    for s in solvers:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                return b_val, float(s_val)
        except Exception:
            continue
    return None, None

    n = len(a0)

            continue

        b_norm = b_new / S_val

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_norm
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_norm) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_y = 2.0 * n * fast_convolve(y).max()

            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

                stall_count = 0
            else:
                stall_count += 1

                if sy2 > 1e-15:
                    y = y / sy2

def iterative_active_qcqp(a, solver, time_limit, max_iter=50, initial_active=None):
    """
    Iterative active-set expansion QCQP.
    Start with a small active set, solve QP, check violations, expand.
    """
    n = len(a)
    a_cur = np.maximum(np.asarray(a, dtype=np.float64), 0)
    sc = a_cur.sum()
    if sc < 1e-15:
        return None, None
    a_cur = a_cur / sc

    b = fast_convolve(a_cur)
    max_b = b.max()

    if initial_active is not None:
        active_set = set(int(k) for k in initial_active)
    else:
        top_k = min(3, 2 * n - 1)
        top_indices = np.argsort(b)[-top_k:]
        active_set = set(int(k) for k in top_indices)

    t0 = time.monotonic()
    a_sol = None
    t_sol = None

    for iteration in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in sorted(active_set):
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        a_sol_new = None
        t_sol_new = None
        solvers = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        seen = set()
        for s in solvers:
            if s is None or s in seen:
                continue
            seen.add(s)
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_sol_new = np.asarray(a_var.value, dtype=np.float64)
                    t_sol_new = float(max(0, t_var.value))
                    break
            except Exception:
                continue

        if a_sol_new is None:
            break

        a_sol_new = np.maximum(a_sol_new, 0)
        ss = a_sol_new.sum()
        if ss < 1e-15:
            break
        a_sol_new = a_sol_new / ss

        a_sol = a_sol_new
        t_sol = t_sol_new

        b_sol = fast_convolve(a_sol)
        max_sol = b_sol.max()

        if max_sol <= t_sol * (1.0 + 1e-7):
            return a_sol, 2.0 * n * max_sol

        violations = b_sol - t_sol
        violated = np.where(violations > 1e-9)[0]
        if len(violated) == 0:
            return a_sol, 2.0 * n * max_sol

        k_violated = int(violated[np.argmax(violations[violated])])
        active_set.add(k_violated)

        if (iteration + 1) % 10 == 0:
            print(f"    ActiveQCQP iter={iteration+1}: active={len(active_set)}, t={t_sol:.10f}, max={max_sol:.10f}")

    if a_sol is not None:
        b_sol = fast_convolve(a_sol)
        return a_sol, 2.0 * n * b_sol.max()
    return None, None


def _lse_objective_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    exp_sum = exp_z.sum()
    if exp_sum < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / exp_sum

    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    exp_sum_t = exp_terms.sum()
    if exp_sum_t < 1e-300:
        lse = max_b
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        lse = max_b + T_temp * np.log(exp_sum_t)
        w = exp_terms / exp_sum_t

    obj = 2.0 * n * lse

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def _lse_func(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = _lse_objective_grad(z, n, T_temp)
    return obj


def _lse_jac(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = _lse_objective_grad(z, n, T_temp)
    return g


def lse_lbfgs_refine(a0, time_limit, T_stages=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    if T_stages is None:
        T_stages = [3.0, 1.5, 0.8, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    sc = a.sum()
    if sc < 1e-15:
        a = np.ones(n) / n
    a = a / sc
    z = np.log(np.maximum(a, 1e-300))

    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_idx, stage_T in enumerate(T_stages):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 5:
            break

        maxiter = max(20, int(300 * remaining / max(1, n_stages - stage_idx)))

        try:
            res = minimize(
                fun=_lse_func, x0=z, jac=_lse_jac,
                args=(n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    return a


def multiplicative_weights_refine(a0, time_limit, n_steps=50000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.02
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def cosine_init(n, frequency=1.0):

    s = a.sum()
    if s < 1e-15:

        best_seq = cosine_init(n_init)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_cur = len(best_seq)

    def do_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_local(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(arr)})")

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alt n={n_cur} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 750.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=200000, rng=rng)
        do_update(a_lp, "lp_phase1")

    # === Phase 2: Iterative Active-Set QCQP (wide) ===
    print(f"\n=== Phase 2: Active QCQP Wide (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_qcqp = min(rem * 0.15, 250.0)
        a_q, _ = iterative_active_qcqp(best_seq, solver, t_qcqp, max_iter=60)
        do_update(a_q, "qcqp_wide")

    # === Phase 3: LP Alt Round 2 (from polished) ===
    print(f"\n=== Phase 3: LP Alt Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=100000, rng=rng)
        do_update(a_lp2, "lp_phase3")

    # === Phase 4: Active-Set QCQP (tight, single constraint) ===
    print(f"\n=== Phase 4: QCQP Tight (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        b_cur = fast_convolve(best_seq)
        k_max = int(np.argmax(b_cur))
        a_qt, _ = iterative_active_qcqp(best_seq, solver, min(rem * 0.1, 80.0),
                                        max_iter=30, initial_active=[k_max])
        do_update(a_qt, "qcqp_tight")

    # === Phase 5: L-BFGS Low-Temp ===
    print(f"\n=== Phase 5: L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_lb = min(rem * 0.10, 150.0)
        print(f"  Budget: {t_lb:.0f}s")
        a_lb = lse_lbfgs_refine(best_seq, t_lb)
        do_update(a_lb, "lbgfs")

    # === Phase 6: Multiplicative weights polish ===
    print(f"\n=== Phase 6: Mult Weights (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_mw, _ = multiplicative_weights_refine(best_seq, min(rem * 0.1, 30.0), n_steps=30000)
        do_update(a_mw, "mult_weights")

    # === Phase 7: Final QCQP ===
    print(f"\n=== Phase 7: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_qf, _ = iterative_active_qcqp(best_seq, solver, min(rem * 0.2, 80.0), max_iter=40)
        do_update(a_qf, "qcqp_final")

    # === Phase 8: Multi-n exploration ===
    print(f"\n=== Phase 8: Multi-n (score={best_score:.10f}) ===")
    n_vals_to_try = []
    for dn in [-5, -3, -2, -1, 1, 2, 3, 5, 10, 20, 50, 100, 200]:
        nt = n_cur + dn
        if 100 <= nt <= 2000 and nt != n_cur:
            n_vals_to_try.append(nt)

    for n_try in n_vals_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break

        # Interpolate from best sequence
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0
, 1, n_try)
        init = np.interp(x_new, x_old, best_seq)
        init = np.maximum(init, 0)
        s = init.sum()
        if s < 1e-15:
            continue
        init = init / s

        t_alloc = min(rem * 0.18, 100.0)
        if t_alloc < 20:
            continue
        print(f"  Trying n={n_try}, budget={t_alloc:.0f}s")
        a_try, _ = nesterov_lp_refine(init, t_alloc * 0.7, solver, n_max_iters=50000, rng=rng)
        do_update(a_try, f"lp_n{n_try}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == n_try:
            a_qn, _ = iterative_active_qcqp(best_seq, solver, min(rem * 0.2, 40.0), max_iter=20)
            do_update(a_qn, f"qcqp_n{n_try}")

    # === Phase 9: Final squeeze with LSE + QCQP ===
    print(f"\n=== Phase 9: Final Squeeze (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_squeeze = lse_lbfgs_refine(best_seq, min(rem * 0.3, 30.0),
                                     T_stages=[0.05, 0.02, 0.01, 0.005])
        do_update(a_squeeze, "squeeze_lbfgs")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_qs, _ = iterative_active_qcqp(best_seq, solver, min(rem * 0.5, 20.0), max_iter=20,
                                            initial_active=[int(np.argmax(fast_convolve(best_seq)))])
            do_update(a_qs, "squeeze_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
