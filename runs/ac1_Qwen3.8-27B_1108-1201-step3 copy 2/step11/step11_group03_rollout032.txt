- Given current point a, find the vertex of the simplex that minimizes the linear approximation of the objective

- The subgradient at a in direction e_i is related to the gradient of the active constraint

Actually, for the Frank-Wolfe method applied to min f(a) where f is convex and the domain is the simplex:
1. Compute a subgradient g of f at current point a
2. Solve min g^T v s.t. v in simplex → v = e_{argmin g_i}
3. Update: a_new = (1-t)*a + t*v

For min f(a) over the simplex, where f is convex and differentiable (or subdifferentiable):
1. Compute a subgradient g at current point a

- No projection needed (which is O(n))
- The update is a simple convex combination with a coordinate
- Each iteration is O(n log n) for the FFT + O(n) for the line search

And with a proper line search, Frank-Wolfe has the same O(1/T) rate as projected subgradient but often converges faster in practice.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-method approach:

2. Nesterov-accelerated LP alternating (main workhorse for the saddle-point problem)

import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    return proj


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if a[j] == 0:
            continue
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix_sparse(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def frank_wolfe_refine(a0, time_limit, n_max_iters=5000000):
    """Frank-Wolfe (conditional gradient) with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    scale = 1.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_max = int(np.argmax(b))
        max_b = b[k_max]

        # Subgradient of max at a: gradient of the active constraint
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g = np.zeros(n)
        g[valid] = 2.0 * a[idx[valid]]

        # Also check nearby active constraints for a better subgradient
        threshold = max_b * (1.0 - 1e-6)
        active = np.where(b >= threshold)[0]
        if len(active) > 1:
            g = np.zeros(n)
            for k_act in active:
                idx_k = k_act - np.arange(n)
                valid_k = (idx_k >= 0) & (idx_k < n)
                g[valid_k] += a[idx_k[valid_k]]
            g /= len(active)
            g *= 2.0

        # Frank-Wolfe step: find vertex e_i minimizing g^T e_i = g_i
        i_min = int(np.argmin(g))

        # Step size with adaptive scaling
        t_step = (2.0 / (it + 2)) * scale

        # Update
        a_new = (1.0 - t_step) * a
        a_new[i_min] += t_step

        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1

        if no_improve >= 200:
            scale *= 0.7
            no_improve = 0
            if scale < 0.01:
                break

        if (it + 1) % 50000 == 0:
            print(f"    FW iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-30:]) * 1000 if len(lp_times) >= 30 else 0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 200 and np.mean(lp_times[-100:]) > 5.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def solve_qcqp_active_set(a_start, solver, rel_tol, max_active):
    """Solve active-set QCQP with given tolerance and max active constraints."""
    n = len(a_start)
    b = fast_convolve(a_start)
    max_b = b.max()

    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]

    if len(active) > max_active:
        active = active[np.argsort(b[active])[-max_active:]]

    if len(active) == 0:
        active = np.array([np.argmax(b)])

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def cutting_plane_qcqp(a0, solver, time_limit, max_rounds=50):
    """Iterative cutting-plane: solve QCQP, add violated constraints."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    t0 = time.monotonic()
    active_set = [int(np.argmax(fast_convolve(a)))]

    for round_idx in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve QCQP with current active set
        a_val, t_val = solve_qcqp_active_set(a, solver, rel_tol=1e-10, max_active=max(50, len(active_set)))
        if a_val is None:
            break
        a = a_val

        # Check for violated constraints
        b = fast_convolve(a)
        max_b = b.max()

        # Find constraints that are violated (exceed current t by more than tolerance)
        violated = np.where(b > t_val * (1 + 1e-8) + 1e-12)[0]
        if len(violated) == 0:
            # Also check if any non-active constraint is close to the max
            for k in range(len(b)):
                if k not in active_set and b[k] > max_b * 0.999:
                    violated = np.array([k])
                    break

        if len(violated) == 0:
            break

        # Add new constraints to active set
        for k in violated:
            if int(k) not in active_set:
                active_set.append(int(k))

        if len(active_set) > 80:
            # Keep only the most binding constraints
            b_check = fast_convolve(a)
            active_set = list(np.argsort(b_check)[-80:])

        if (round_idx + 1) % 5 == 0:
            print(f"    CP-QCQP round {round_idx+1}: |active|={len(active_set)}, t={t_val:.10f}")

    return a, 2.0 * n * fast_convolve(a).max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lse_func(z, n, T_temp, deadline):
    """Top-level LSE objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_temp)
    return obj


def lse_jac(z, n, T_temp, deadline):
    """Top-level LSE gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_temp)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(10, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                args=(n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_subgradient(a0, time_limit, n_steps=200000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.05
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Frank-Wolfe, LP alternating with Nesterov acceleration,
    cutting-plane QCQP, low-temp L-BFGS, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        # Try multiple initializations
        candidates = []
        # Cosine
        x = np.linspace(0, 1, n_init)
        c1 = np.cos(np.pi * x / 2) ** 2
        c1 = np.maximum(c1, 0)
        c1 = c1 / c1.sum()
        candidates.append(c1)
        # Uniform
        c2 = np.ones(n_init) / n_init
        candidates.append(c2)
        # Triangle
        c3 = 1 - np.abs(2 * x - 1)
        c3 = c3 / c3.sum()
        candidates.append(c3)

        for c in candidates:
            sc = evaluate_local(c)
            if sc < best_score:
                best_score = sc
                best_seq = c.copy()
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Frank-Wolfe (fast initial progress) ===
    print(f"\n=== Phase 1: Frank-Wolfe n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_fw = min(rem * 0.15, 200.0)
    if t_fw > 30:
        print(f"  Budget: {t_fw:.0f}s")
        a_fw, _ = frank_wolfe_refine(best_seq, t_fw)
        update_best(a_fw, "fw_phase1")

    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP Alternating n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.35, 550.0)
    if t_lp > 60:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=500000)
        update_best(a_lp, "lp_phase2")

    # === Phase 3: Cutting-plane QCQP ===
    print(f"\n=== Phase 3: Cutting-Plane QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_cp = min(rem * 0.15, 200.0)
    if t_cp > 20:
        print(f"  Budget: {t_cp:.0f}s")
        a_cp, sc_cp = cutting_plane_qcqp(best_seq, solver, t_cp, max_rounds=80)
        update_best(a_cp, "cp_qcqp")
        # Also try direct active-set with various tolerances
        a_q1, _ = solve_qcqp_active_set(best_seq, solver, rel_tol=1e-3, max_active=30)
        update_best(a_q1, "qcqp_wide")
        a_q2, _ = solve_qcqp_active_set(best_seq, solver, rel_tol=1e-5, max_active=10)
        update_best(a_q2, "qcqp_mid")
        a_q3, _ = solve_qcqp_active_set(best_seq, solver, rel_tol=1e-8, max_active=5)
        update_best(a_q3, "qcqp_tight")

    # === Phase 4: LP round 2 (from polished) ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.15, 200.0)
    if t_lp2 > 30:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=200000)
        update_best(a_lp2, "lp_phase4")

    # === Phase 5: L-BFGS low temperature ===
    print(f"\n=== Phase 5: L-BFGS Low-Temp (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lb = min(rem * 0.10, 120.0)
    if t_lb > 15:
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_lb = lbgfs_refine(best_seq, t_lb, T_stages, maxiter_per_stage=5000)
        update_best(a_lb, "lbgfs")

    # === Phase 6: QCQP final polish ===
    print(f"\n=== Phase 6: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_qf, _ = solve_qcqp_active_set(best_seq, solver, rel_tol=1e-8, max_active=5)
        update_best(a_qf, "qcqp_final")
        a_cpf, _ = cutting_plane_qcqp(best_seq, solver, min(rem * 0.3, 30.0), max_rounds=20)
        update_best(a_cpf, "cp_final")

    # === Phase 7: Try n-1 ===
    print(f"\n=== Phase 7: Try n={n_best-1} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80 and n_best > 100:
        t_alt = min(rem * 0.12, 120.0)
        print(f"  Budget: {t_alt:.0f}s")
        init_alt = interpolate_sequence(best_seq, n_best - 1)
        a_alt, _ = nesterov_lp_refine(init_alt, t_alt * 0.6, solver, n_max_iters=100000)
        update_best(a_alt, f"lp_n{n_best-1}")
        rem = deadline - time.monotonic()
        if rem > 15 and best_seq is not None and len(best_seq) == n_best - 1:
            a_qa, _ = solve_qcqp_active_set(best_seq, solver, rel_tol=1e-6, max_active=10)
            update_best(a_qa, f"qcqp_n{n_best-1}")

    # === Phase 8: Try n+1 ===
    print(f"\n=== Phase 8: Try n={n_best+1} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alt2 = min(rem * 0.10, 100.0)
        print(f"  Budget: {t_alt2:.0f}s")
        # Re-interpolate from the best n_best solution (not the modified one)
        # Use the best_seq which might have changed length, so interpolate from original
        init_alt2 = interpolate_sequence(best_seq, n_best + 1)
        a_alt2, _ = nesterov_lp_refine(init_alt2, t_alt2 * 0.6, solver, n_max_iters=100000)
        update_best(a_alt2, f"lp_n{n_best+1}")
        rem = deadline - time.monotonic()
        if rem > 15 and best_seq is not None and len(best_seq) == n_best + 1:
            a_qa2, _ = solve_qcqp_active_set(best_seq, solver, rel_tol=1e-6, max_active=10)
            update_best(a_qa2, f"qcqp_n{n_best+1}")

    # === Phase 9: Final multiplicative subgradient ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.3, 25.0), n_steps=100000)
        update_best(a_ms, "final_mult")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_qf2, _ = solve_qcqp_active_set(best_seq, solver, rel_tol=1e-9, max_active=3)
            update_best(a_qf2, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
