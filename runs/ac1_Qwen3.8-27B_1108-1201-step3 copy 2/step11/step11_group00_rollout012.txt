    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def score_from_conv(a):

    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Subgradient peak-shaving refinement

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        if a_arr.size == 0:
            return float('inf')
        a_list = [float(max(0.0, min(1000.0, x))) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def score_direct(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2 directly (fast, no clipping)."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_grad_a_fft(a, w):
    """
    Compute grad_a[i] = 2 * sum_k w[k] * a[k-i] where a[m]=0 for m outside [0,n-1].
    w has length 2n-1 (LogSumExp weights over convolution lags).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    w_flipped = w[::-1]  # length 2n-1
    conv_len = n + len(w_flipped) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a, fft_size)
    WF = rfft(w_flipped, fft_size)
    conv = irfft(A * WF, fft_size)[:conv_len]
    # grad_a[i] = 2 * conv[2n-2-i] for i=0,...,n-1
    grad_a = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad_a


def lse_obj_and_grad_z(z, T):
    """
    Compute objective value and gradient in z-space for the LogSumExp smoothed objective.
    a = softmax(z), objective = (1/T) * log(sum_k exp(T * (a*a)_k))
    """
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    grad_a = compute_grad_a_fft(a, w)

    # Transform gradient to z-space: grad_z = a * (grad_a - a . grad_a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def lse_func_wrapper(z, T):
    """Wrapper for L-BFGS: returns objective."""
    val, _ = lse_obj_and_grad_z(z, T)
    return val


def lse_jac_wrapper(z, T):
    """Wrapper for L-BFGS: returns gradient."""
    _, g = lse_obj_and_grad_z(z, T)
    return g


def logsumexp_optimize(a0, time_limit, T_start=80.0, T_end=0.01, n_stages=20, verbose=True):
    """
    Multi-stage LogSumExp-annealed L-BFGS-B optimization.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    best_a = a_init.copy()
    best_score = score_direct(a_init)
    if verbose:
        print(f"    LSE start: n={n}, T=[{T_start:.1f}..{T_end:.4f}], stages={n_stages}, score={best_score:.6f}")

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Scale iterations: more for early stages (higher T = smoother), fewer for late

        max_iter = max(10, int(60 - 30 * frac))
        maxfun = max_iter * 300

                fun=lse_func_wrapper, x0=z, args=(stage_T,),

        except Exception:

def subgradient_polish(a0, time_limit, max_iters=50000, verbose=True):
    """
    Subgradient-based peak shaving: at each step, find the argmax lag k*,
    compute the subgradient, and take a projected step in the negative subgradient direction.
    Uses diminishing step sizes and running average tracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_direct(a)
    running_avg = a.copy().astype(np.float64)
    n_avg = 1
    t0 = time.monotonic()

    if verbose:
        print(f"    SG polish start: n={n}, score={best_score:.6f}")

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        # Subgradient: g[i] = 2 * a[k_star - i] (zero if out of range)
        g = np.zeros(n)
        i_arr = np.arange(n)
        j_arr = k_star - i_arr
        valid = (j_arr >= 0) & (j_arr < n)
        g[valid] = 2.0 * a[j_arr[valid]]

        # Projected gradient step on simplex {a >= 0, sum(a) = 1}
        # a_new = proj_simplex(a - lr * g)
        lr = 2.0 / (it + 10.0)

        a_raw = a - lr * g
        # Project onto simplex: find lambda such that sum(max(a_raw - lambda, 0)) = 1
        sorted_a = np.sort(a_raw)[::-1]
        css = np.cumsum(sorted_a)
        rho = np.nonzero(sorted_a * np.arange(1, n + 1) > (css - 1))[0]
        if len(rho) > 0:
            rho = rho[0]
            theta = (css[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_raw - theta, 0)
        else:
            a_new = np.ones(n) / n

        ss = a_new.sum()
        if ss > 1e-15:
            a_new = a_new / ss
        else:
            a_new = a.copy()

        sc = score_direct(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        # Running average (Likhachev-style)
        running_avg += a_new
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = running_avg / n_avg
            sc_avg = score_direct(avg_check)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()

        a = a_new

        if it % 5000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it}, elapsed={elapsed:.1f}s, best={best_score:.6f}, current={sc:.6f}")

    return best_a, best_score


def coordinate_refine(a0, time_limit, seed=123, verbose=True):
    """
    Coordinate descent: try moving mass between pairs of indices.
    Focus on pairs that affect the top peaks most.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_direct(a)
    t0 = time.monotonic()
    stall = 0

    if verbose:
        print(f"    Coord refine start: n={n}, score={best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]

        improved = False
        for _ in range(20):
            # Pick a peak
            k = top_k[rng.integers(len(top_k))]
            # Find the highest contributor to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Pick destination: random index far from peak
            dst = rng.integers(n)
            if dst == src:
                dst = (dst + 1) % n

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = score_direct(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                amt = a[i] * rng.uniform(0.003, 0.02)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = a_trial / a_trial.sum()
                sc = score_direct(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def structured_init(n, seed=0):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.1
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
        a = np.maximum(a, 0)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.6 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    else:
        x = np.linspace(0, 1, n)
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: LogSumExp at parent length (n~1600)
    Phase 2: Interpolate to n=2000, LogSumExp
    Phase 3: Interpolate to n=2400, LogSumExp
    Phase 4: Subgradient polishing
    Phase 5: Coordinate refinement
    Phase 6: Final verification
    """
    t_global = time.monotonic()
    budget = min(float(time_budget) - 90.0, 1700.0)
    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, eval_score={p_score:.6f}, direct_score={score_direct(parent):.6f}")
            best_seq = parent.copy()
            best_score = min(p_score, score_direct(parent))
    except Exception as e:
        print(f"  Could not load parent: {e}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc_direct = score_direct(a_norm)
        sc_eval = evaluate_safe(a_norm)
        sc = min(sc_direct, sc_eval) if np.isfinite(sc_eval) else sc_direct
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600
    if parent is None:
        best_seq = structured_init(n_base, seed)
        best_score = score_direct(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # ============ Phase 1: LogSumExp at base length ============
    remaining = budget - (time.monotonic() - t_global)
    print(f"\n{'='*60}")
    print(f"Phase 1: LogSumExp at n={n_base} (remaining={remaining:.0f}s)")
    print(f"{'='*60}")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 20:
        a_out, _ = logsumexp_optimize(best_seq, t1, T_start=100.0, T_end=0.01, n_stages=25)
        try_update(a_out, f"lse_n{n_base}")

    # ============ Phase 2: LogSumExp with different schedule ============
    remaining = budget - (time.monotonic() - t_global)
    print(f"\n{'='*60}")
    print(f"Phase 2: LogSumExp (alt schedule) at n={n_base} (remaining={remaining:.0f}s)")
    print(f"{'='*60}")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t2, T_start=200.0, T_end=0.005, n_stages=30)
        try_update(a_out, f"lse_alt_n{n_base}")

    # ============ Phase 3: Subgradient at base length ============
    remaining = budget - (time.monotonic() - t_global)
    print(f"\n{'='*60}")
    print(f"Phase 3: Subgradient polish at n={n_base} (remaining={remaining:.0f}s)")
    print(f"{'='*60}")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 10:
        a_out, _ = subgradient_polish(best_seq, t3, max_iters=100000)
        try_update(a_out, f"sg_n{n_base}")

    # ============ Phase 4: Interpolate to n=2000 ============
    n_2000 = 2000
    n_long1 = max(n_base, 2000)
    remaining = budget - (time.monotonic() - t_global)
    print(f"\n{'='*60}")
    print(f"Phase 4: LogSumExp at n={n_long1} (remaining={remaining:.0f}s)")
    print(f"{'='*60}")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20:
        init_long = interpolate_sequence(best_seq, n_long1)
        a_out, _ = logsumexp_optimize(init_long, t4, T_start=80.0, T_end=0.01, n_stages=25)
        try_update(a_out, f"lse_n{n_long1}")

    # ============ Phase 5: Subgradient at longer length ============
    remaining = budget - (time.monotonic() - t_global)
    print(f"\n{'='*60}")
    print(f"Phase 5: Subgradient at n={n_long1} (remaining={remaining:.0f}s)")
    print(f"{'='*60}")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        a_out, _ = subgradient_polish(best_seq, t5, max_iters=100000)
        try_update(a_out, f"sg_n{n_long1}")

    # ============ Phase 6: Try n=2400 ============
    n_long2 = 2400
    remaining = budget - (time.monotonic() - t_global)
    print(f"\n{'='*60}")
    print(f"Phase 6: LogSumExp at n={n_long2} (remaining={remaining:.0f}s)")
    print(f"{'='*60}")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 20:
        init_long2 = interpolate_sequence(best_seq, n_long2)
        a_out, _ = logsumexp_optimize(init_long2, t6, T_start=60.0, T_end=0.01, n_stages=20)
        try_update(a_out, f"lse_n{n_long2}")

    # ============ Phase 7: Coordinate refinement on best ============
    remaining = budget - (time.monotonic() - t_global)
    print(f"\n{'='*60}")
    print(f"Phase 7: Coordinate refine at n={len(best_seq)} (remaining={remaining:.0f}s)")
    print(f"{'='*60}")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 5:
        a_out, _ = coordinate_refine(best_seq, t7, seed=777)
        try_update(a_out, "coord_refine")

    # ============ Phase 8: Final subgradient polish ============
    remaining = budget - (time.monotonic() - t_global)
    print(f"\n{'='*60}")
    print(f"Phase 8: Final SG polish (remaining={remaining:.0f}s)")
    print(f"{'='*60}")
    t8 = max(0, remaining - 30.0)
    if t8 > 5:
        a_out, _ = subgradient_polish(best_seq, t8, max_iters=100000)
        try_update(a_out, "final_sg")

    # ============ Final verification ============
    print(f"\n{'='*60}")
    print("FINAL VERIFICATION")
    print(f"{'='*60}")
    if best_seq is not None:
        fs_direct = score_direct(best_seq)
        fs_eval = evaluate_safe(best_seq)
        print(f"  Direct score: {fs_direct:.8f}")
        print(f"  Eval score:   {fs_eval:.8f}")
        print(f"  n = {len(best_seq)}")
        final_score = min(fs_direct, fs_eval) if np.isfinite(fs_eval) else fs_direct
        print(f"  FINAL: {final_score:.8f}")
    else:
        print("  ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
