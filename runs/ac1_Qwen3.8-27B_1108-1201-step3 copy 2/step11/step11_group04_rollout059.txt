```python
"""

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fft_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fft_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_best_response(a_fixed):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Uses scipy linprog with HiGHS."""
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    
    # LP: min M s.t. T @ b - M <= 0, sum(b) = 1, b >= 0, M >= 0
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # A_ub @ x <= b_ub:  T @ b - M <= 0
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq @ x = b_eq:  sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            b_val = res.x[:n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s
        else:
            return None
    except Exception:
        return None


def spectral_flatten(a, n_iters=100, blend=0.3, deadline=None):
    """Flatten the power spectrum |A(f)|^2 using iterative spectral factorization."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = normalize(a)
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L, dtype=np.complex128)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        # Target: geometric mean of |F|^2 for non-DC bins
        P = np.abs(F) ** 2
        non_dc = P[1:L // 2 + 1]
        if non_dc.min() < 1e-30:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        # Blend |F|^2 toward target
        phases = np.angle(F)
        mags = np.abs(F)
        F_new = F.copy()
        
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * np.sqrt(target)
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a

    # Convert to gradient w.r.t. z (log-domain)
    # a = softmax(z), so da[i]/dz[j] = a[i](delta_ij - a[j])
    # grad_z[j] = sum_i grad_a[i] * da[i]/dz[j] = sum_i grad_a[i] * a[i] * (delta_ij - a[j])
    #           = a[j] * grad_a[j] - a[j] * sum_i grad_a[i] * a[i]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages=None):
    """L-BFGS refinement with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    
    if T_stages is None:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-300))
    
    t0 = time.monotonic()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        max_iter = min(2000, max(50, int(remaining * 10)))
        
        def func(z_in):
            return logsumexp_score_grad(z_in, n, T)[0]
        
        def jac(z_in):
            return logsumexp_score_grad(z_in, n, T)[1]
        
        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15, 'disp': False})
            z = res.x
        except Exception:
            pass
    
    # Convert back
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a = a / a.sum()
    return a


def multiplicative_subgrad(a0, time_limit, n_steps=500000):
    """Multiplicative weights subgradient method targeting active peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    eta = 0.05
    scale = 1.0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fft_autocorr(a)
        max_b = b.max()
        
        # Find active peaks (within 0.5% of max)
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Project gradient to sum(a)=1 hyperplane
        g = g - np.dot(g, a) * a  # project to tangent space of simplex
        
        # Multiplicative update (mirror descent)
        a_new = a * np.exp(-eta * scale * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        sc = score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            scale *= 0.75
            no_improve = 0
            if scale < 0.01:
                break
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based autocorrelation
2. LP alternating (best-response) with Nesterov acceleration via HiGHS
3. Spectral pre-conditioning (flat |F|^2 target)
4. L-BFGS refinement with LogSumExp smoothing
5. Multiplicative subgradient for peak shaving
6. Multi-start diversification and adaptive time allocation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fft_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score(a):
    """Compute the objective value 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fft_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix_fast(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b) using vectorized ops."""
    n = len(a)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    T = np.where((idx >= 0) & (idx < n), a[idx], 0.0)
    return T


def solve_best_response(a_fixed):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    T = build_conv_matrix_fast(a_fixed)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            b_val = res.x[:n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s
    except Exception:
        pass
    return None


def spectral_flatten(a, n_iters=100, blend=0.3, deadline=None):
    """Flatten the power spectrum |F|^2 using iterative spectral projection."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = normalize(a)

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L, dtype=np.complex128)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        P = np.abs(F) ** 2
        non_dc = P[1:L // 2 + 1]
        if non_dc.min() < 1e-30:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        mags = np.abs(F)
        F_new = F.copy()

        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * np.sqrt(target)
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def nesterov_lp(a0, time_limit, n_max_iters=1000000):
    """Nesterov-accelerated LP alternating with stall detection and restarts."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()

    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(777)
    n_iters_done = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_best_response(y)
        if b_new is None:
            continue
        n_iters_done += 1

        alpha = 2.0 / (it + 2)
        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        a_new = ((it + 1) * a + (it + 2) * b_new) / ((it + 1) + (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / sa

        sc_a = score(a_new)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a_new.copy()

        sc_y = score(y_new)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y_new.copy()

        a = a_new
        y = y_new

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 12:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                amp = 0.01 + 0.01 * min(stall, 5)
                y = y + amp * np.cos(freq * x + phase)
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                a = y.copy()
                stall = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} iters_done={n_iters_done}")

    return best_a, best_score


def lse_obj_grad(z, n, T):
    """LogSumExp smoothed objective and gradient in log-domain."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()

    b = fft_autocorr(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages=None):
    """L-BFGS refinement with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        max_iter = min(3000, max(30, int(remaining * 8)))

        def func(z_in):
            return lse_obj_grad(z_in, n, T)[0]

        def jac(z_in):
            return lse_obj_grad(z_in, n, T)[1]

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxls': 50, 'disp': False})
            z = res.x
        except Exception:
            pass

    z_shift = z - z.max()
    a = np.exp(z_shift)
    a = a / a.sum()
    return np.maximum(a, 0)


def multiplicative_subgrad(a0, time_limit, n_steps=500000):
    """Multiplicative weights subgradient targeting active peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()

    eta = 0.04
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fft_autocorr(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        a_new = a * np.exp(-eta * scale * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc = score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 60:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating (Nesterov-accelerated via HiGHS), spectral flattening,
    L-BFGS with LogSumExp, multiplicative subgradient, and multi-start.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = normalize(best_seq)
        best_score = score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # Phase 1: Spectral pre-conditioning
    print(f"\n[Phase 1] Spectral pre-conditioning")
    t_spec = min(rem * 0.03, 30.0)
    if t_spec > 5:
        a_spec = spectral_flatten(best_seq, n_iters=200, blend=0.25,
                                  deadline=time.monotonic() + t_spec)
        local_update(a_spec, "spectral")
        a_spec2 = spectral_flatten(best_seq, n_iters=100, blend=0.5,
                                   deadline=time.monotonic() + min(t_spec, 15) + 15)
        local_update(a_spec2, "spectral_strong")

    # Phase 2: Main LP alternating (primary driver)
    print(f"\n[Phase 2] LP alternating n={n_best} (score={best_score:.8f})")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.45, 600.0)
    if t_lp > 30:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp(best_seq, t_lp, n_max_iters=2000000)
        local_update(a_lp, "lp_main")

    # Phase 3: L-BFGS refinement
    print(f"\n[Phase 3] L-BFGS refinement (score={best_score:.8f})")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.20, 250.0)
    if t_lbfgs > 20:
        print(f"  Budget: {t_lbfgs:.0f}s")
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages)
        local_update(a_lb, "lbgfs")

    # Phase 4: Multiplicative subgradient
    print(f"\n[Phase 4] Multiplicative subgradient (score={best_score:.8f})")
    rem = deadline - time.monotonic()
    t_ms = min(rem * 0.08, 80.0)
    if t_ms > 10:
        a_ms, _ = multiplicative_subgrad(best_seq, t_ms)
        local_update(a_ms, "subgrad")

    # Phase 5: Second LP round
    print(f"\n[Phase 5] LP round 2 (score={best_score:.8f})")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.15, 180.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp(best_seq, t_lp2, n_max_iters=1000000)
        local_update(a_lp2, "lp_2")

    # Phase 6: Perturbation escape + re-optimize
    print(f"\n[Phase 6] Perturbation escape (score={best_score:.8f})")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(5):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            strength = 0.008 + 0.008 * trial
            x = np.linspace(0, 2 * np.pi, n_best)
            freq = rng.integers(1, 20)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = best_seq + strength * np.cos(freq * x + phase)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            a_sp = spectral_flatten(a_pert, n_iters=15, blend=0.3)
            t_sub = min(rem_sub * 0.18, 40.0)
            if t_sub > 10:
                a_opt, _ = nesterov_lp(a_sp, t_sub, n_max_iters=500000)
                local_update(a_opt, f"pert{trial}")

    # Phase 7: Try different sequence lengths
    print(f"\n[Phase 7] Different n (score={best_score:.8f})")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [800, 1000, 1500]:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            old_n = len(best_seq)
            old_x = np.linspace(0, 1, old_n)
            new_x = np.linspace(0, 1, n_try)
            a_interp = np.interp(new_x, old_x, best_seq)
            a_interp = np.maximum(a_interp, 0)
            a_interp = a_interp / a_interp.sum()

            t_sub = min(rem_sub * 0.35, 80.0)
            if t_sub > 15:
                a_sp = spectral_flatten(a_interp, n_iters=30, blend=0.3)
                a_opt, _ = nesterov_lp(a_sp, t_sub * 0.7, n_max_iters=500000)
                local_update(a_opt, f"n{n_try}")

    # Phase 8: Final polish
    print(f"\n[Phase 8] Final polish (score={best_score:.8f})")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.35, 40.0)
        if t_fin > 10:
            T_stages_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin2 = min(rem * 0.6, 25.0)
        if t_fin2 > 8:
            a_fin, _ = nesterov_lp(best_seq, t_fin2, n_max_iters=200000)
            local_update(a_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.1f}s")

        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
