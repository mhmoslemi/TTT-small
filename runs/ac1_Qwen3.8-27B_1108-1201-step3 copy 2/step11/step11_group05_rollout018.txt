    size = 2 * n - 1

        return float('inf')
    n = len(a)

def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if s <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def mirror_descent_optimize(a0, time_limit, T_schedule, eta_init=0.1):
    """
    Mirror descent (multiplicative weights) with temperature annealing.
    Optimizes the LogSumExp surrogate of max(conv(a,a)).
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # Work with sum = 1, rescale at end
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()  # With sum=1, score = 2n*max_b
    
    t0 = time.monotonic()
    steps = 0
    no_improve = 0
    eta = eta_init
    
    # Nesterov momentum
    a_omo = a.copy()  # a_{t-1}
    
    for T_idx, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        for step in range(max(100, int(remaining * 2000))):
            if time.monotonic() - t0 > time_limit:
                break
            steps += 1

            alpha_mom = (steps - 1) / (steps + 2) if steps > 1 else 0
            y = a + alpha_mom * (a - a_omo)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:

                w = exp_terms / exp_terms.sum()

            a = a_new

def solve_b_dual(a, active_set, solver):
    """Solve min max_{k in active_set} conv(a,b)[k] using the dual LP."""
    n = len(a)
    S = len(active_set)
    
    # Build A matrix: A[k, j] = a[active_set[k] - j] if valid
    A = np.zeros((S, n))
    for i, k in enumerate(active_set):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = a[idx]
    
    # Dual LP: max lambda s.t. A^T p >= lambda * 1, sum(p) = 1, p >= 0
    p_var = cp.Variable(S, nonneg=True)
    lambda_var = cp.Variable()
    
    constraints = [
        cp.sum(p_var) == 1,
        A.T @ p_var >= lambda_var  # This is n constraints
    ]
    
    prob = cp.Problem(cp.Maximize(lambda_var), constraints)
    prob.solve(solver=solver)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, float('inf')
    
    p_val = np.asarray(p_var.value, dtype=np.float64)
    lambda_val = float(lambda_var.value)
    
    # Recover b: it's concentrated where A^T p is minimal
    Ap = A.T @ p_val
    min_Ap = Ap.min()
    b = np.zeros(n)
    # b is on the face where A^T p = lambda (complementary slackness)
    # Actually, from the primal-dual relationship:
    # The primal optimal b satisfies: (A @ b)_k = M for k in active set (by strong duality)
    # And b_j = 0 where (A^T p)_j > lambda (by complementary slackness)
    tolerance = 1e-8
    support = np.where(Ap <= min_Ap + tolerance)[0]
    if len(support) == 0:
        support = np.array([np.argmin(Ap)])
    # On the support, we need to find b that satisfies A @ b = M * 1 (for active set)
    # This is: A_support @ b_support = M * 1_S
    # With sum(b) = 1 and b >= 0
    # If |support| == S, we can solve directly
    A_sub = A[:, support]
    # A_sub @ b_sub = M (vector of length S)
    # sum(b_sub) = 1
    # This is (S+1) equations in |support| unknowns
    # If |support| == S+1, unique solution
    # If |support| > S+1, use LP or flag
    if len(support) <= S + 1:
        # Solve the system
        B_mat = np.vstack([A_sub, np.ones((1, len(support)))])
        rhs = np.concatenate([np.full(S, lambda_val), [1.0]])
        if len(support) == S + 1:
            b_sub = np.linalg.solve(B_mat, rhs)
            if np.all(b_sub >= -1e-10):
                b_sub = np.maximum(b_sub, 0)
                b_sub = b_sub / b_sub.sum()
                b[support] = b_sub
                return b, lambda_val
        elif len(support) < S + 1:
            # Underdetermined, use LP
            b_sub_var = cp.Variable(len(support), nonneg=True)
            constraints_sub = [A_sub @ b_sub_var <= lambda_var * cp.ones(S),  # Should be ==
                              cp.sum(b_sub_var) == 1]
            # Actually use equality:
            constraints_sub = [A_sub @ b_sub_var == lambda_var * cp.ones(S),
                              cp.sum(b_sub_var) == 1]
            prob_sub = cp.Problem(cp.Minimize(0), constraints_sub)
            prob_sub.solve(solver=solver)
            if prob_sub.status in ('optimal', 'optimal_inaccurate') and b_sub_var.value is not None:
                b_sub = np.asarray(b_sub_var.value, dtype=np.float64)
                b_sub = np.maximum(b_sub, 0)
                b_sub = b_sub / b_sub.sum()
                b[support] = b_sub
                return b, lambda_val
    
    # Fallback: just put mass uniformly on support
    b[support] = 1.0 / len(support)
    return b, float(max(A @ b))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_gradient_logsumexp(a, w):
    """Compute gradient: 2 * correlate(w, a)[n-1:2n-1] using FFT."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    # correlate(w, a) = convolve(w[::-1], a)
    w_rev = w[::-1]
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    corr_full = irfft(W * A, fft_size)[:2 * n - 1]
    return 2.0 * corr_full[n - 1:2 * n - 1]

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if s <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_toeplitz(a):
    """Build (2n-1) x n Toeplitz matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Fast Multiplicative Weights (quick improvement) ===
    print(f"\n=== Phase 1: MW-Nesterov n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_mw, _ = multiplicative_weights_nesterov(best_seq, t_alloc, T_stages, eta_init=0.04)
        local_update(a_mw, "mw_phase1")

    # === Phase 2: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP-Nesterov n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating_nesterov(best_seq, t_alloc, solver)
        local_update(a_lp, "lp_phase2")

    # === Phase 3: Fine MW polishing ===
    print(f"\n=== Phase 3: MW Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_fine = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        a_mw2, _ = multiplicative_weights_nesterov(best_seq, t_alloc, T_fine, eta_init=0.02)
        local_update(a_mw2, "mw_fine")

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternating_nesterov(best_seq, t_alloc, solver)
        local_update(a_lp2, "lp_phase4")

    # === Phase 5: Perturbation escape ===
    print(f"\n=== Phase 5: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            strength = 0.005 + 0.005 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.3, 60.0)
            T_esc = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]
            a_opt, _ = multiplicative_weights_nesterov(a_pert, t_sub, T_esc, eta_init=0.03)
            local_update(a_opt, f"pert_mw_t{trial}")

            # Quick LP on perturbed result
            rem_sub2 = deadline - time.monotonic()
            if rem_sub2 > 20:
                t_lp_sub = min(rem_sub2 * 0.3, 40.0)
                a_lp_pert, _ = lp_alternating_nesterov(a_opt, t_lp_sub, solver)
                local_update(a_lp_pert, f"pert_lp_t{trial}")

    # === Phase 6: Try n=1500 ===
    print(f"\n=== Phase 6: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100 and len(best_seq) != 1500:
        t_n1500 = min(rem * 0.25, 200.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        # Quick MW
        t_mw_1500 = t_n1500 * 0.4
        a_1500_mw, _ = multiplicative_weights_nesterov(init_1500, t_mw_1500,
                                                        [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                                        eta_init=0.03)
        local_update(a_1500_mw, "mw_n1500")
        # LP polish
        rem_sub = deadline - time.monotonic()
        if rem_sub > 30:
            t_lp_1500 = min(rem_sub * 0.5, 100.0)
            a_1500_lp, _ = lp_alternating_nesterov(a_1500_mw, t_lp_1500, solver)
            local_update(a_1500_lp, "lp_n1500")

    # === Phase 7: Try n=1000 ===
    print(f"\n=== Phase 7: Try n=1000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60 and len(best_seq) != 1000:
        t_n1000 = min(rem * 0.15, 100.0)
        print(f"  Budget: {t_n1000:.0f}s")
        init_1000 = interpolate_sequence(best_seq, 1000)
        a_1000_mw, _ = multiplicative_weights_nesterov(init_1000, t_n1000,
                                                        [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                                        eta_init=0.03)
        local_update(a_1000_mw, "mw_n1000")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        n_fin = len(best_seq)
        t_fin = min(rem * 0.4, 30.0)
        T_final = [0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
        a_fin, _ = multiplicative_weights_nesterov(best_seq, t_fin, T_final, eta_init=0.01)
        local_update(a_fin, "final_mw")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_final_lp, _ = lp_alternating_nesterov(best_seq, min(rem * 0.7, 20.0), solver)
        local_update(a_final_lp, "final_lp")

    # Final verification
    print(f"\n=== FINAL RESULTS ===")
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"Best local score: {best_score:.8f} (n={len(best_seq)})")
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"evaluate_sequence: {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
