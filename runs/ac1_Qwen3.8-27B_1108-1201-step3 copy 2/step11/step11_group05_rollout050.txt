def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]:
        try:

        except Exception:

def solve_lp(a_fixed, solver, tol=1e-10):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    obj = cp.Minimize(M_var)
    prob = cp.Problem(obj, constraints)
    
    for s in [solver, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            kwargs = {}
            if s == cp.CLARABEL:
                kwargs = {'tol_gap_abs': tol, 'tol_feas': tol, 'tol_infeas': tol}
            elif s == cp.PDLP:
                kwargs = {'tol_res_abs': tol, 'tol_res_dual_abs': tol, 'tol_res_rel': tol}
            prob.solve(solver=s, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    continue
                b_val = b_val / s_sum
                return b_val, float(max(0.0, M_var.value))
        except Exception:
            continue
    return None, None

    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    candidates = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]
    for solver_name in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_tight(a_fixed, solver, tol=1e-11):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [solver]
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        if s not in solvers:
            solvers.append(s)

    for s in solvers:
        try:
            kwargs = {}
            if s == cp.CLARABEL:
                kwargs = {'tol_gap_abs': tol, 'tol_feas': tol, 'tol_infeas': tol, 'verbose': False}
            elif s == cp.PDLP:
                kwargs = {'tol_res_abs': tol, 'tol_res_dual_abs': tol, 'tol_res_rel': tol}
            prob.solve(solver=s, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    continue
                b_val = b_val / s_sum
                return b_val, float(max(0.0, M_var.value))
        except Exception:
            continue
    return None, None


def evaluate_local(a):
    """Evaluate sequence score locally (fast, no clamping)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_alternate_nesterov(a_init, time_limit, solver, max_iters=100000):
    """LP alternating with Nesterov dual averaging."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    consecutive_lp_fail = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val = solve_lp_tight(y, solver, tol=1e-11)
        if b_new is None:
            consecutive_lp_fail += 1
            if consecutive_lp_fail > 20:
                break
            continue
        consecutive_lp_fail = 0

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y /= sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa

        score_a = evaluate_local(a)
        score_y = evaluate_local(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                break

            print(f"      LP iter={it+1} ({elapsed:.0f}s): best={best_score:.10f} M={M_val:.6e}")

    return best_a, best_score


def compute_subgradient(a, active_set):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a, given active set.
    
    d/d a[j] of conv(a,a)[k] = 2*a[k-j] if 0<=k-j<n, else 0.
    Subgradient is average over active set.
    """
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # g[j] += 2*a[k-j] for j in [j_start, j_end]
            # a[k-j] for j=j_start is a[k-j_start], for j=j_end is a[k-j_end]
            g[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
    g /= len(active_set)
    return g


def nesterov_subgrad_logspace(a_init, time_limit, max_steps=200000):
    """Nesterov-accelerated subgradient in log-space (softmax parameterization)."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    step_count = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Evaluate at current point
        a_eval = np.exp(z - z.max())
        a_eval /= a_eval.sum()

        b = fast_convolve(a_eval)
        max_b = b.max()
        score = 2.0 * n * max_b

        step_count += 1

        if score < best_score:
            best_score = score
            best_a = a_eval.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-6:
                break

        # Find active set
        threshold = max_b * (1.0 - 1e-5)
        active_mask = b >= threshold
        active_set = np.where(active_mask)[0]
        if len(active_set) == 0:
            active_set = np.array([np.argmax(b)])

        # Compute subgradient in a-space
        g_a = compute_subgradient(a_eval, active_set)

        # Convert to z-space (softmax Jacobian)
        S = np.dot(g_a, a_eval)
        grad_z = a_eval * (g_a - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()

        # Nesterov momentum on z
        # (implicit in the Adam above)

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NSubGrad {elapsed:.0f}s: best={best_score:.10f} lr={lr:.6f} |active|={len(active_set)}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    """Create a structured initial sequence with various shapes."""
    x = np.linspace(0, 1, n)
    if style == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 2
    elif style == 2:
        a = np.exp(-4 * x * (1 - x))
    elif style == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    elif style == 4:
        a = np.exp(-2 * (x - 0.5) ** 2 / 0.15)
    elif style == 5:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 5 * x) + 0.15 * np.cos(2 * np.pi * 11 * x + 0.7)
        a = np.maximum(a, 0)
    elif style == 6:
        a = (1 - np.abs(2 * x - 1)) ** 0.7 + 0.1
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_equalization(a_init, time_limit, solver, n_peaks=5):
    """Targeted peak reduction: iteratively solve LP focusing on top peaks."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    y = a.copy()

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()

        # Focus on top peaks
        top_k = min(n_peaks, 2 * n - 1)
        peak_indices = np.argsort(b)[-top_k:]

        # Solve LP with emphasis on peak reduction
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T_mat = build_conv_matrix(y)
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for s_solver in [solver, cp.CLARABEL, cp.PDLP]:
            try:
                prob.solve(solver=s_solver)
                if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                    b_val = np.asarray(b_var.value, dtype=np.float64)
                    b_val = np.maximum(b_val, 0)
                    sv = b_val.sum()
                    if sv > 1e-15:
                        b_val /= sv
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y /= sy

        a = ((it + 1) * a + alpha * (it + 2) * b_val) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa

        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            print(f"      PeakEq iter={it+1}: score={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating, Nesterov subgradient, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, style=2)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}) ***")

    # === Phase 1: LP alternating from parent (n=1200) ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: LP Alternating n={n_cur} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternate_nesterov(best_seq, t_alloc, solver)
        local_update(a_lp, "lp_n1200")

    # === Phase 2: Nesterov subgradient refinement (n=1200) ===
    print(f"\n=== Phase 2: Nesterov SubGrad n={len(best_seq)} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ns, _ = nesterov_subgrad_logspace(best_seq, t_alloc)
        local_update(a_ns, "nsubgrad_n1200")

    # === Phase 3: LP again from improved point ===
    print(f"\n=== Phase 3: LP Round 2 n={len(best_seq)} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternate_nesterov(best_seq, t_alloc, solver)
        local_update(a_lp2, "lp2_n1200")

    # === Phase 4: Try n=1400 ===
    print(f"\n=== Phase 4: n=1400 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 200:
        n_target = 1400
        init_1400 = interpolate_seq(best_seq, n_target)
        t_sub = min(rem * 0.25, 300.0)
        print(f"  Budget: {t_sub:.0f}s")

        # LP on n=1400
        t_lp_1400 = t_sub * 0.5
        a_lp_1400, _ = lp_alternate_nesterov(init_1400, t_lp_1400, solver)
        local_update(a_lp_1400, "lp_n1400")

        # Subgradient on n=1400
        rem_sub = deadline - time.monotonic()
        if rem_sub > 30:
            t_ns_1400 = min(rem_sub * 0.4, 150.0)
            a_ns_1400, _ = nesterov_subgrad_logspace(best_seq if len(best_seq) == 1400 else init_1400, t_ns_1400)
            local_update(a_ns_1400, "nsubgrad_n1400")

    # === Phase 5: Try n=1600 ===
    print(f"\n=== Phase 5: n=1600 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        n_target = 1600
        init_1600 = interpolate_seq(best_seq, n_target)
        t_sub = min(rem * 0.20, 200.0)
        print(f"  Budget: {t_sub:.0f}s")
        a_lp_1600, _ = lp_alternate_nesterov(init_1600, t_sub, solver)
        local_update(a_lp_1600, "lp_n1600")

    # === Phase 6: Peak equalization on best ===
    print(f"\n=== Phase 6: Peak Equalization (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pe = min(rem * 0.15, 100.0)
        print(f"  Budget: {t_pe:.0f}s")
        a_pe, _ = peak_equalization(best_seq, t_pe, solver, n_peaks=5)
        local_update(a_pe, "peak_eq")

    # === Phase 7: Diverse restart ===
    print(f"\n=== Phase 7: Diverse Restart (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_best = len(best_seq)
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            style = trial * 2 + 1
            a_div = structured_init(n_best, rng, style=style)
            # Blend with current best
            blend = 0.3 * a_div + 0.7 * best_seq
            blend = np.maximum(blend, 0)
            blend /= blend.sum()

            t_sub = min(rem_sub * 0.25, 60.0)
            a_div_opt, _ = lp_alternate_nesterov(blend, t_sub, solver)
            local_update(a_div_opt, f"diverse_{trial}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 60.0)
        a_fin, _ = nesterov_subgrad_logspace(best_seq, t_fin)
        local_update(a_fin, "final_subgrad")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin2 = min(rem * 0.5, 30.0)
        a_fin2, _ = lp_alternate_nesterov(best_seq, t_fin2, solver)
        local_update(a_fin2, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.10f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.10f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
