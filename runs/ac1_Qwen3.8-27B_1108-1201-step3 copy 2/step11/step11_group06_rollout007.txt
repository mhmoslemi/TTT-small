        rows_list.append(ks)
        cols_list.append(np.full(len(ks), j, dtype=np.int32))
        vals_list.append(a[ks - j])
    rows = np.concatenate(rows_list).astype(np.int32)
    cols = np.concatenate(cols_list).astype(np.int32)
    vals = np.concatenate(vals_list).astype(np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            dt = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"      Solver {solver_name}: {dt:.3f}s on test")
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val = b_val / s_val
    return b_val, max(0.0, float(M_var.value))


def lp_alternating_nesterov(a0, time_limit, solver, max_iters=1000000, verbose=False):
    """LP alternating with Nesterov dual averaging."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_start = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)
    lp_times = []
    T_mat = build_conv_matrix_sparse(y)

    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_dt = time.monotonic() - lp_t0
        lp_times.append(lp_dt)
        if b_new is None:
            continue
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 10.0:
                if verbose:
                    print(f"      LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = normalize(a)
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        T_mat = build_conv_matrix_sparse(y)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 4:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall = 0
            if verbose and (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t_start
                print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = logsumexp_obj_grad(z, self.n, self.T)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = logsumexp_obj_grad(z, self.n, self.T)
        return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t_start = time.monotonic()
    deadline = t_start + time_limit
    for stage_T in T_stages:
        if time.monotonic() - t_start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 5:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-28,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return exp_z / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses projected subgradient, MW, LP alternating, LSE L-BFGS,
    and multi-length exploration with aggressive time management.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = np.ones(n_init) / n_init
        best_score = local_eval(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: score={sc:.8f} (n={len(a)})")

    # === Phase 1: Quick MW warmup on parent ===
    print(f"\n=== Phase 1: MW warmup n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.03, 40.0)
    if t1 > 10:
        a_mw, _ = mw_descent(best_seq, t1, c_init=1.0, top_k=30, verbose=False)
        try_update(a_mw, "mw_1")

    # === Phase 2: Projected subgradient refinement ===
    print(f"\n=== Phase 2: PGD n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.05, 60.0)
    if t2 > 10:
        a_pgd, _ = projected_subgradient(best_seq, t2, top_k=15, c_init=0.5, verbose=False)
        try_update(a_pgd, "pgd_1")

    # === Phase 3: LP alternating (main workhorse) ===
    print(f"\n=== Phase 3: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.30, 500.0)
    if t3 > 60:
        print(f"  Budget: {t3:.0f}s")
        a_lp, _ = lp_alternating_nesterov(best_seq, t3, solver, max_iters=2000000, verbose=True)
        try_update(a_lp, "lp_1")

    # === Phase 4: PGD polish after LP ===
    print(f"\n=== Phase 4: PGD polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.05, 50.0)
    if t4 > 10:
        a_pgd2, _ = projected_subgradient(best_seq, t4, top_k=10, c_init=0.3, verbose=False)
        try_update(a_pgd2, "pgd_2")

    # === Phase 5: Palindromic LP ===
    print(f"\n=== Phase 5: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.10, 120.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = lp_alternating_nesterov(a_pal, t5, solver, max_iters=200000, verbose=False)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    lengths_to_try = [1500, 2000, 1000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.12, 180.0)
        if t_ln < 30:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)

        # Quick MW
        t_mw = min(t_ln * 0.25, 40.0)
        a_ln_mw, _ = mw_descent(init_ln, t_mw, c_init=1.0, top_k=30, verbose=False)
        try_update(a_ln_mw, f"mw_n{ln}")

        # PGD
        rem = deadline - time.monotonic()
        t_pgd = min(rem * 0.2, 50.0)
        if t_pgd > 10:
            a_ln_pgd, _ = projected_subgradient(a_ln_mw, t_pgd, top_k=15, c_init=0.5, verbose=False)
            try_update(a_ln_pgd, f"pgd_n{ln}")

        # LP (only if fast enough)
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, 120.0)
        if t_lp > 20 and ln <= 1500:
            a_ln_lp, _ = lp_alternating_nesterov(a_ln_mw, t_lp, solver, max_iters=100000, verbose=False)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 7: LSE L-BFGS fine on best ===
    print(f"\n=== Phase 7: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t7 = min(rem * 0.20, 250.0)
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05,
                    0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        print(f"  {len(T_stages)} stages, budget={t7:.0f}s")
        a_lse = lbgfs_refine(best_seq, t7, T_stages, maxiter_per_stage=400)
        try_update(a_lse, "lse_fine")

    # === Phase 8: Perturbation + re-optimize ===
    print(f"\n=== Phase 8: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.01 * trial)
            t_sub = min(rem * 0.25, 50.0)
            a_opt, _ = projected_subgradient(a_pert, t_sub, top_k=15, c_init=0.5, verbose=False)
            try_update(a_opt, f"pert_pgd{trial}")
            rem = deadline - time.monotonic()
            if rem > 30:
                t_lp = min(rem * 0.4, 60.0)
                a_opt2, _ = lp_alternating_nesterov(a_pert, t_lp, solver, max_iters=50000, verbose=False)
                try_update(a_opt2, f"pert_lp{trial}")

    # === Phase 9: Final LP polish ===
    print(f"\n=== Phase 9: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.30, 120.0)
        a_fin_lp, _ = lp_alternating_nesterov(best_seq, t9, solver, max_iters=200000, verbose=False)
        try_update(a_fin_lp, "final_lp")

    # === Phase 10: Final PGD + LSE polish ===
    print(f"\n=== Phase 10: Final PGD+LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t10a = min(rem * 0.3, 30.0)
        a_fin_pgd, _ = projected_subgradient(best_seq, t10a, top_k=10, c_init=0.3, verbose=False)
        try_update(a_fin_pgd, "fin_pgd")

        rem = deadline - time.monotonic()
        if rem > 20:
            t10b = min(rem * 0.5, 60.0)
            T_fine = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_fin_lse = lbgfs_refine(best_seq, t10b, T_fine, maxiter_per_stage=300)
            try_update(a_fin_lse, "fin_lse")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")
    else:
        print("WARNING: No valid sequence found!")
        best_seq = np.ones(1200) / 1200
        best_score = local_eval(best_seq)

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
