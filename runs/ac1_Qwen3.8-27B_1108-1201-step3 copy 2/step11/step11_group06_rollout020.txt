**Frequency-domain amplitude optimization:**

The autocorrelation `b = IDFT(|A|^2)` where `A = FFT(a_padded)`. We want to minimize `max(b)`.

In the frequency domain, `|A[m]|` are the amplitudes we can control. The constraint `a_i >= 0` translates to a constraint on the phases and amplitudes of `A`.

A simpler approach: 
1. Compute `A = FFT(a)` (zero-padded to length N)
2. The power spectrum `P[m] = |A[m]|^2`
3. We want `IDFT(P)` to be flat
4. Gradient of `max_k b[k]` w.r.t. `P[m]`: if `b[k*]` is the max, then `∂b[k*]/∂P[m] = Re(exp(2πi mk*/N))` (with appropriate normalization)
5. Update `P` to reduce the peak, maintaining `P >= 0`
6. Recover `a` from `A` (keeping phase, updating amplitude): `A_new[m] = A[m] * sqrt(P_new[m]/P[m])`
7. Take IFFT, project to non-negative, zero-pad appropriately

This is essentially a projected gradient method in the frequency domain!

def freq_domain_optimize(a0, time_limit, n_steps=10000):
    """
    Optimize in frequency domain: minimize max(b) by adjusting |A[m]|.
    Constraint: a_i >= 0 enforced by projection in spatial domain.
    """
    n = len(a0)
    N = 1 << (2*n-2).bit_length()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Pre-compute phase
    A = np.fft.fft(a, N)
    phase = np.angle(A)
    
    # Initial power
    P = np.abs(A)**2
    # DC bin: P[0] = S^2 = 1 (fixed since we normalize)
    
    eta = 0.1
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current spatial
        A_curr = np.abs(A_curr) ... 
        # Actually, let me keep A updated
        a_curr = np.fft.ifft(A, N).real[:n]
        a_curr = np.maximum(a_curr, 0)
        s = a_curr.sum()
        if s < 1e-15:
            break
        a_curr = a_curr / s
        
        b = fast_convolve(a_curr)
        max_b = b.max()
        k_star = np.argmax(b)
        
        # Gradient of b[k_star] w.r.t. P[m]
        # b[k] = (1/N) * sum_m P[m] * exp(2*pi*i*m*k/N)
        # db[k_star]/dP[m] = (1/N) * cos(2*pi*m*k_star/N)  (real part, since P is real)
        # Actually: b = IDFT(P), so b[k] = (1/N) sum_m P[m] exp(2πikm/N)
        # db[k_star]/dP[m] = (1/N) exp(2πik_star m / N)
        # For real b: = (1/N) cos(2πk_star m / N)
        
        grad_P = np.cos(2 * np.pi * k_star * np.arange(N) / N) / N
        
        # Update P: reduce P[m] where grad_P[m] > 0 (increases peak)
        P_new = P - eta * grad_P
        P_new = np.maximum(P_new, 1e-10)
        # Keep DC fixed
        P_new[0] = P[0]
        P_new[N//2] = P[N//2] if N % 2 == 0 else P_new[0]
        
        # Update A amplitude
        A = np.abs(A) * np.sqrt(P_new / np.maximum(P, 1e-30))
        A = A * np.exp(1j * phase)
        
        # Project to non-negative in spatial domain
        a = np.fft.ifft(A, N).real[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        # Update phase from new a
        A = np.fft.fft(a, N)
        phase = np.angle(A)
        P = np.abs(A)**2
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            eta = min(eta * 1.01, 0.5)
        else:
            eta *= 0.95
    
    return best_a, best_score

def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with pre-built sparse matrix."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)

    T_mat = build_conv_matrix(n, vals=None)
    fill_conv_matrix(T_mat, y)

    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        fill_conv_matrix(T_mat, y)
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_dt = time.monotonic() - lp_t0
        lp_times.append(lp_dt)
        if b_new is None:
            continue
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 5.0:
                break

        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 5e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 4:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                    fill_conv_matrix(T_mat, y)
                stall_count = 0
            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={np.mean(lp_times[-20:]):.3f}s")

    return best_a, best_score


def multiplicative_weights(a0, time_limit, n_steps, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.3)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break
        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")
    return best_a, best_score


def gs_flat_spectrum(a0, time_limit, n_iters=5000):
    """
    Gerchberg-Saxton style frequency domain iterations.
    Alternates between spatial (non-negativity) and frequency (flat spectrum).
    """
    n = len(a0)
    N = 1 << (2 * n - 2).bit_length()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    damping = 0.3  # how aggressively to flatten

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Frequency domain
        A = np.fft.fft(a, N)
        power = np.abs(A) ** 2
        phase = np.angle(A)

        # Target: flatten power spectrum (but keep DC fixed)
        # Mean power excluding DC
        non_dc = np.arange(N) != 0
        if N % 2 == 0:
            non_dc[N // 2] = False
        mean_power = power[non_dc].mean() if non_dc.any() else power[0] / N

        # Blend power toward flat
        for m in range(N):
            if m == 0:
                continue
            if N % 2 == 0 and m == N // 2:
                continue
            target = mean_power
            current = power[m]
            new_power = (1 - damping) * current + damping * target
            new_power = max(new_power, 1e-20)
            A[m] = np.sqrt(new_power) * np.exp(1j * phase[m])
            if N - m < N and N - m != m:
                A[N - m] = np.conj(A[m])

        # Spatial domain: project
        a_new = np.fft.ifft(A, N).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T_temp, deadline):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_obj_grad(z, self.n, self.T, self.deadline)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_obj_grad(z, self.n, self.T, self.deadline)
        return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': maxiter_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.02):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    else:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LP (main), MW, GS frequency domain, LSE L-BFGS, multi-length.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    # Phase 1: Quick MW warmup
    print(f"\n=== Phase 1: MW warmup n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, n_steps=500000, eta0=0.05)
        sc = evaluate_local(a_mw)
        if sc < best_score:
            best_score = sc
            best_seq = a_mw
            n_best = len(best_seq)
            print(f"  MW improved to {best_score:.8f}")

    # Phase 2: Main Nesterov LP
    print(f"\n=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 500.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, n_max_iters=1000000)
        sc = evaluate_local(a_lp)
        if sc < best_score:
            best_score = sc
            best_seq = a_lp
            n_best = len(best_seq)
            print(f"  LP improved to {best_score:.8f}")

    # Phase 3: Multi-length LP (n=1000)
    print(f"\n=== Phase 3: LP n=1000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120 and n_best != 1000:
        t3 = min(rem * 0.2, 250.0)
        print(f"  Budget: {t3:.0f}s")
        init_1000 = interpolate_sequence(best_seq, 1000)
        sc_init = evaluate_local(init_1000)
        print(f"  Interp score: {sc_init:.8f}")
        # Quick MW first
        a_1000_mw, _ = multiplicative_weights(init_1000, t3 * 0.15, n_steps=100000, eta0=0.05)
        sc_mw = evaluate_local(a_1000_mw)
        start_1000 = a_1000_mw if sc_mw < sc_init else init_1000
        rem = deadline - time.monotonic()
        t3_lp = min(rem * 0.6, t3 * 0.75)
        if t3_lp > 30:
            a_1000, _ = nesterov_lp_refine(start_1000, t3_lp, solver, n_max_iters=500000)
            sc_1000 = evaluate_local(a_1000)
            if sc_1000 < best_score:
                best_score = sc_1000
                best_seq = a_1000
                n_best = 1000
                print(f"  n=1000 improved to {best_score:.8f}")

    # Phase 4: GS frequency domain (novel)
    print(f"\n=== Phase 4: GS flat spectrum n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t4 = min(rem * 0.08, 100.0)
        a_gs, _ = gs_flat_spectrum(best_seq, t4, n_iters=3000)
        sc_gs = evaluate_local(a_gs)
        print(f"  GS direct score: {sc_gs:.8f}")
        # LP from GS result
        if sc_gs < best_score + 0.01:
            rem = deadline - time.monotonic()
            t4_lp = min(rem * 0.3, 80.0)
            if t4_lp > 20:
                a_gs_lp, _ = nesterov_lp_refine(a_gs, t4_lp, solver, n_max_iters=100000)
                sc_gs_lp = evaluate_local(a_gs_lp)
                if sc_gs_lp < best_score:
                    best_score = sc_gs_lp
                    best_seq = a_gs_lp
                    n_best = len(best_seq)
                    print(f"  GS+LP improved to {best_score:.8f}")

    # Phase 5: LSE L-BFGS fine-tuning
    print(f"\n=== Phase 5: LSE L-BFGS n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t5 = min(rem * 0.18, 300.0)
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lbgfs_refine(best_seq, t5, T_stages, maxiter_per_stage=600)
        sc_lse = evaluate_local(a_lse)
        if sc_lse < best_score:
            best_score = sc_lse
            best_seq = a_lse
            n_best = len(best_seq)
            print(f"  LSE improved to {best_score:.8f}")

    # Phase 6: Perturbation + LP escape
    print(f"\n=== Phase 6: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.008 * trial)
            t_sub = min(rem * 0.3, 60.0)
            if t_sub < 20:
                continue
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=100000)
            sc_opt = evaluate_local(a_opt)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt
                n_best = len(best_seq)
                print(f"  Pert{trial} improved to {best_score:.8f}")

    # Phase 7: Final LP polish
    print(f"\n=== Phase 7: Final LP polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t7 = min(rem * 0.35, 150.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t7, solver, n_max_iters=500000)
        sc_fin = evaluate_local(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin
            n_best = len(best_seq)
            print(f"  Final LP improved to {best_score:.8f}")

    # Phase 8: Final LSE micro-polish
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        t8 = min(rem * 0.4, 60.0)
        T_micro = [0.001, 0.0005, 0.0002, 0.0001]
        a_micro = lbgfs_refine(best_seq, t8, T_micro, maxiter_per_stage=300)
        sc_micro = evaluate_local(a_micro)
        if sc_micro < best_score:
            best_score = sc_micro
            best_seq = a_micro
            n_best = len(best_seq)
            print(f"  Micro LSE improved to {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
