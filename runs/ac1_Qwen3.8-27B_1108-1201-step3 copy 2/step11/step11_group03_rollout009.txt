def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_q_sparse(k, n):
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

def get_solver():

    exp_z = np.exp(z_shifted)

    S_val = np.dot(grad_a, a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Coarse n-search to find optimal sequence length
2. LP alternating with Nesterov acceleration for coarse optimization
3. L-BFGS on LogSumExp-smoothed objective with FFT-based gradient for fine optimization
4. Active-set QCQP (constraint generation) for exact polishing
5. Multi-n refinement around the best length

Key innovation: FFT-based gradient computation for the smoothed objective,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_solver():
    """Find a working LP/QCQP solver, preferring fast ones."""
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def evaluate_local(a):
    """Compute score locally (assumes a >= 0, sum > 0)."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k] s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s2 = b_val.sum()
                if s2 < 1e-15:
                    return None
                return b_val / s2
        except Exception:
            continue
    return None


def nesterov_lp_refine(a0, time_limit, solver, max_iters=500000):
    """Nesterov-accelerated LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        sc_y = 2.0 * n * fast_convolve(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

    return best_a, best_score


def smooth_obj_grad(a, T):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. a (not z)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    lse = max_b + T * np.log(exp_shifted.sum())
    obj = 2.0 * n * lse

    # Gradient: sum_k p_k * 2 * (Q_k @ a)_j
    # (Q_k @ a)_j = a_{k-j} * (1.0 if j==k-j else 0.5) for valid k-j
    # grad_j = sum_k p_k * 2 * a_{k-j} * (1.0 if j==k-j else 0.5)
    #        = sum_k p_k * a_{k-j} * (2.0 if j==k-j else 1.0)
    #        = cross_corr(a, p)[j] + p[2j] * a[j]  (the diagonal correction)

    # Cross-correlation: sum_i a[i] * p[i+j]
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    cross = irfft(A * P, fft_size)[:n]

    # Diagonal correction: for position j, if 2j < 2n-1, add p[2j]*a[j]
    diag_idx = 2 * np.arange(n)
    diag_mask = diag_idx < (2 * n - 1)
    cross[diag_mask] += p[diag_idx[diag_mask]] * a[diag_mask]

    grad = 2.0 * cross  # factor of 2 from the objective
    return obj, grad


def lbgfs_refine(a0, time_limit, T_stages):
    """L-BFGS on LogSumExp-smoothed objective with temperature annealing."""
    n = len(a0)
    t0 = time.monotonic()

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def fun(a_x):
            a_x = np.maximum(a_x, 0)
            s = a_x.sum()
            if s < 1e-15:
                return 1e15
            a_x = a_x / s
            obj, _ = smooth_obj_grad(a_x, T)
            return obj

        def jac(a_x):
            a_x = np.maximum(a_x, 0)
            s = a_x.sum()
            if s < 1e-15:
                return np.zeros(n)
            a_x = a_x / s
            _, g = smooth_obj_grad(a_x, T)
            return g

        try:
            bounds = [(0, None)] * n
            res = minimize(fun, a0, jac=jac, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-30, 'gtol': 1e-15,
                                   'maxfun': 30000, 'disp': False})
            a0 = np.maximum(res.x, 0)
            s = a0.sum()
            if s > 1e-15:
                a0 = a0 / s
        except Exception:
            pass

    return a0


def solve_qcqp_active(a_init, active_set, solver):
    """Solve min t s.t. a^T Q_k a <= t for k in active, a>=0, sum(a)=1."""
    n = len(a_init)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    for k in active_set:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                return a_val / s_sum, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def active_set_refine(a_init, solver, max_iters=100, tol=1e-8):
    """
    Growing active set (constraint generation) for the convex QCQP:
    min t s.t. a^T Q_k a <= t for all k, a>=0, sum(a)=1.
    """
    n = len(a_init)
    a = a_init.copy()
    b = fast_convolve(a)
    max_b = b.max()

    # Initialize active set with top positions
    sorted_idx = np.argsort(b)[::-1]
    active_set = set(int(k) for k in sorted_idx[:5])

    for it in range(max_iters):
        a_sol, t_val = solve_qcqp_active(a, list(active_set), solver)
        if a_sol is None:
            break

        # Check for violated constraints
        b_new = fast_convolve(a_sol)
        worst_violation = 0.0
        worst_k = -1
        for k in range(len(b_new)):
            if k not in active_set:
                viol = b_new[k] - t_val
                if viol > worst_violation:
                    worst_violation = viol
                    worst_k = k

        if worst_violation < tol * max(t_val, 1e-10):
            return a_sol, t_val

        active_set.add(worst_k)
        a = a_sol

        # Limit active set size by keeping most binding
        if len(active_set) > 40:
            b_check = fast_convolve(a)
            top = set(int(k) for k in np.argsort(b_check)[::-1][:30])
            active_set = top

    return a, t_val


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_update(a0, time_limit, n_steps=20000):

        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 50:
            eta *= 0.85
            no_improve = 0
            if eta < 0.001:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases:
    1. Load parent / initialize
    2. Coarse n-search (try several lengths)
    3. LP alternating for coarse optimization
    4. L-BFGS on LogSumExp for fine optimization
    5. Active-set QCQP for exact polishing
    6. Refine near best n
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_solver()
    print(f"Using solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: Coarse n-search ===
    print(f"\n=== Phase 1: Coarse n-search ===")
    rem = deadline - time.monotonic()
    # Try a few nearby n values with quick LP
    n_candidates = list(range(max(2, n_best - 8), n_best + 9))
    n_scores = {}
    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        if n_try <= 1:
            continue
        if n_try == n_best:
            n_scores[n_try] = best_score
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        # Quick LP for ~15 seconds
        t_quick = min(12.0, rem * 0.08)
        a_try, sc_try = nesterov_lp_refine(init_try, t_quick, solver, max_iters=5000)
        s_try = a_try.sum()
        if s_try > 1e-15:
            a_try = a_try / s_try
            sc_try = 2.0 * n_try * fast_convolve(a_try).max()
        n_scores[n_try] = sc_try
        print(f"    n={n_try}: score={sc_try:.10f}")

    # Pick best n
    if n_scores:
        best_n = min(n_scores, key=n_scores.get)
        if n_scores[best_n] < best_score * 1.001:  # allow small margin
            print(f"  Best n from search: {best_n} (score={n_scores[best_n]:.10f})")
            if best_n != n_best:
                best_seq = interpolate_sequence(best_seq, best_n)
                best_score = n_scores[best_n]
                n_best = best_n

    # === Phase 2: LP Alternating (main coarse optimization) ===
    print(f"\n=== Phase 2: LP Alternating n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.35, 500.0)
    if t_lp > 30:
        print(f"  Time budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, max_iters=200000)
        update_best(a_lp, "lp_phase2")

    # === Phase 3: L-BFGS LogSumExp refinement ===
    print(f"\n=== Phase 3: L-BFGS LogSumExp (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.25, 400.0)
    if t_lbfgs > 20:
        print(f"  Time budget: {t_lbfgs:.0f}s")
        T_stages = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages)
        update_best(a_lb, "lbfgs_lse")

    # === Phase 4: Active-set QCQP ===
    print(f"\n=== Phase 4: Active-Set QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_qcqp, t_val = active_set_refine(best_seq, solver, max_iters=80, tol=1e-9)
        update_best(a_qcqp, "qcqp_refine")

    # === Phase 5: L-BFGS again after QCQP ===
    print(f"\n=== Phase 5: L-BFGS Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs2 = min(rem * 0.15, 200.0)
    if t_lbfgs2 > 15:
        a_lb2 = lbgfs_refine(best_seq, t_lbfgs2, [0.01, 0.005, 0.002, 0.001])
        update_best(a_lb2, "lbfgs_polish")

    # === Phase 6: Final QCQP ===
    print(f"\n=== Phase 6: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_qcqp2, _ = active_set_refine(best_seq, solver, max_iters=60, tol=1e-10)
        update_best(a_qcqp2, "qcqp_final")

    # === Phase 7: Multiplicative update polish ===
    print(f"\n=== Phase 7: Multiplicative Update (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_mult = min(rem * 0.1, 60.0)
    if t_mult > 5:
        a_mult, _ = multiplicative_update(best_seq, t_mult, n_steps=50000)
        update_best(a_mult, "mult_update")

    # === Phase 8: Try best_n +/- 1 with full pipeline ===
    print(f"\n=== Phase 8: Nearby n refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    for dn in [-1, 1, -2, 2]:
        if time.monotonic() > deadline - 30:
            break
        n_try = n_best + dn
        if n_try < 2:
            continue
        t_near = min(rem * 0.2, 120.0)
        if t_near < 10:
            continue
        print(f"  Trying n={n_try}, budget={t_near:.0f}s")
        init_near = interpolate_sequence(best_seq, n_try)
        a_near, _ = nesterov_lp_refine(init_near, t_near * 0.4, solver, max_iters=30000)
        update_best(a_near, f"lp_n{n_try}")
        if best_seq is not None and len(best_seq) == n_try:
            a_near2 = lbgfs_refine(best_seq, t_near * 0.3, [0.01, 0.005, 0.002])
            update_best(a_near2, f"lbfgs_n{n_try}")
            if rem > 15:
                a_near3, _ = active_set_refine(best_seq, solver, max_iters=40, tol=1e-9)
                update_best(a_near3, f"qcqp_n{n_try}")

    # Final QCQP on absolute best
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Final QCQP (score={best_score:.10f}) ===")
        a_final, _ = active_set_refine(best_seq, solver, max_iters=50, tol=1e-10)
        update_best(a_final, "qcqp_ultimate")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
