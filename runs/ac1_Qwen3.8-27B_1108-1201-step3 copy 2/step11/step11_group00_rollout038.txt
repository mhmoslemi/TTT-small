Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Nesterov-accelerated iteration with running average tracking
3. Multi-length exploration (1600, 2000, 2500)
4. FFT-based LogSumExp gradient polishing with multi-stage annealing
5. Subgradient peak shaving for final improvement
Key improvements over previous: much faster LP via HiGHS + constraint reduction,
enabling 10-50x more iterations; better time allocation; multiple length attempts.

import numpy as np

from scipy.optimize import linprog, minimize


def fast_convolution(a, b=None):
    """Full convolution a*b (or a*a if b is None) using FFT."""
    if b is None:
        b = a
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix_row(a_fixed, k):
    """Build row k of the convolution matrix T where (T@b)[k] = conv(a_fixed, b)[k]."""
    n = len(a_fixed)
    row = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        row[j_start:j_end + 1] = a_fixed[j_start:j_end + 1]
    return row


def solve_lp_fast(a_fixed, K_max=150):
    """
    Solve: min M s.t. conv(a_fixed, b)[k] <= M for all k,
           sum(b) = 1, b >= 0
    Using constraint reduction and scipy HiGHS.
    Returns optimal b.
    """
    n = len(a_fixed)
    # Initial active set: top K lags by conv(a_fixed, uniform)
    uniform_b = np.ones(n) / n
    conv_uniform = fast_convolution(a_fixed, uniform_b)
    all_lags = np.argsort(conv_uniform)[::-1]
    active_set = list(all_lags[:min(K_max, 2 * n - 1)])
    
    b_opt = np.ones(n) / n
    for iteration in range(15):
        K = len(active_set)
        # Build T_K
        T_K = np.zeros((K, n))
        for i, k in enumerate(active_set):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                T_K[i, j_start:j_end + 1] = a_fixed[j_start:j_end + 1]
        
        # LP: min M, vars = [b_0,...,b_{n-1}, M]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        A_ub = np.hstack([T_K, -np.ones((K, 1))])
        b_ub = np.zeros(K)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(None, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            return np.ones(n) / n
        
        b_opt = res.x[:n]
        b_opt = np.maximum(b_opt, 0)
        s = b_opt.sum()
        if s < 1e-15:
            return np.ones(n) / n
        b_opt = b_opt / s
        M_val = res.x[-1]
        
        # Check violations outside active set using FFT
        conv_b = fast_convolution(a_fixed, b_opt)
        max_lag = int(np.argmax(conv_b))
        if conv_b[max_lag] > M_val * (1 + 1e-7) + 1e-12:
            if max_lag not in active_set:
                active_set.append(max_lag)
                if len(active_set) > K_max * 3:
                    # Keep only top K_max*3
                    conv_vals = conv_b
                    top_lags = np.argsort(conv_vals)[::-1][:K_max * 3]
                    active_set = [int(x) for x in top_lags]
                continue
        else:
            return b_opt
    
    return b_opt


def nesterov_lp_optimize(a0, time_limit, K_max=150):
    """

    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    if s < 1e-15:

    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolution(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star = solve_lp_fast(y, K_max)
        
        alpha_t = 2.0 / (t + 2.0)
        
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        # Check score
        conv_a = fast_convolution(a)
        score_a = 2.0 * n * conv_a.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        conv_y = fast_convolution(y)
        score_y = 2.0 * n * conv_y.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg >= 5 and n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_convolution(avg_check).max()
            if score_avg < best_score:

def compute_lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a using FFT.
    Returns (value, gradient).
    """
    n = len(a)
    b = fast_convolution(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (valid k-i)
    # This is 2 * cross-correlation of w and a
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)
    # grad[i] = 2 * conv[n-1+i]
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return lse_val, grad


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad_z(z, T):
    """Objective and gradient in z-space (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    lse_val, grad_a = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return lse_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=15):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_convolution(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        iters_per_stage = max(10, min(60, int(30 + 30 * (si / n_stages))))
        
        def func_lse(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val
        
        def jac_lse(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 100,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_convolution(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Subgradient-based peak shaving: reduce top peaks by moving mass.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolution(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolution(a)
        top_k = np.argsort(b)[::-1][:12]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            
            # Find highest contributor
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            
            # Find target: low-value position not in peak areas
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for p in range(js2, je2 + 1):
                    peak_areas.add(p)
            
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if not candidates:
                continue
            
            # Pick low-value candidates
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = int(dst_pool[rng.integers(len(dst_pool))])
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolution(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolution(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 300:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic method."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):

        x = np.linspace(0, 1, n)

    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)

    else:
        x = np.linspace(0, 1, n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def subgradient_refine(a0, time_limit, n_steps_max=50000):
    """
    Subgradient method for directly minimizing max_k (a*a)[k].
    Uses the subgradient of the max function.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolution(a).max()
    t0 = time.monotonic()
    
    # Polyak step size: c / (f(a) - f*) where f* is lower bound
    # Use adaptive step size
    step = 0.1 / np.sqrt(n)
    
    for t in range(1, n_steps_max + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolution(a)
        k_star = int(np.argmax(b))
        
        # Subgradient of max_k (a*a)[k] at a is 2*a shifted
        # d/da_i [a*a][k] = 2*sum_j a[j] * delta(i + j, k) (for the relevant terms)
        # More precisely: d/da_i (a*a)[k] = a[k-i] + a[i-k+n-1-i]... 
        # Actually for full convolution (a*a)[k] = sum_i a[i]*a[k-i]
        # d/da_i (a*a)[k] = a[k-i] (if 0<=k-i<n) + a[i-k+n-1]... 
        # Let me be careful: (a*a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
        # d/da_i (a*a)[k] = a[k-i] if 0<=k-i<n, plus a[i'] where i' satisfies k-i'=i, i.e., i'=k-i
        # So d/da_i (a*a)[k] = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=i<n and we also consider the other term)
        # Wait, (a*a)[k] = sum_i a[i]*a[k-i]. If I differentiate w.r.t. a[m]:
        # = a[k-m] (from the first factor being a[m]) + a[m-k] (from the second factor, if k-m is valid... )
        # Actually: (a*a)[k] = sum_i a[i]*a[k-i]. Setting i=m gives a[m]*a[k-m].
        # Also, if we set k-i=m, i.e., i=k-m, we get a[k-m]*a[m].
        # So if both m and k-m are valid indices, the derivative is 2*a[k-m].
        # If only one is valid, it's a[k-m] or a[m-k+...]. 
        
        # Simpler: the gradient of (a*a)[k] w.r.t. a is:
        # grad[m] = a[k-m] (if 0<=k-m<n) + a[m-k+n-1] (if 0<=m-k+n-1<n)... 
        # No, let me think again. The full convolution (a*a)[k] = sum_{i=0}^{n-1} a[i]*a[k-i] where a[j]=0 for j outside [0,n-1].
        # d/da[m] (a*a)[k] = a[k-m] + a[m-k] (where a[j]=0 outside [0,n-1])
        # But a[m-k] is nonzero only if 0<=m-k<n, i.e., k<=m<k+n.
        # And a[k-m] is nonzero only if 0<=k-m<n, i.e., k-n<m<=k.
        
        # So grad[m] = (a[k-m] if 0<=k-m<n else 0) + (a[m-k] if 0<=m-k<n else 0)
        # But a[m-k] for m>=k: a[m-k] where m-k is in [0,n-1]
        # And a[k-m] for m<=k: a[k-m] where k-m is in [0,n-1]
        
        # In vector form: grad = a[k::-1] (padded) + a[k+1:] (padded)
        # More precisely:
        # For m in [0, k]: grad[m] includes a[k-m] (from first term)
        # For m in [k, 2k]: ... this is getting confusing.
        
        # Let me just compute it directly:
        grad = np.zeros(n)
        # Term 1: a[k-m] for m where 0 <= k-m < n, i.e., max(0,k-n+1) <= m <= min(k, n-1)
        m_start1 = max(0, k - n + 1)
        m_end1 = min(k, n - 1)
        if m_start1 <= m_end1:
            grad[m_start1:m_end1 + 1] += a[k - m_end1:k - m_start1 + 1][::-1]
        
        # Term 2: a[m-k] for m where 0 <= m-k < n, i.e., k <= m <= min(k+n-1, n-1)
        m_start2 = max(k, 0)
        m_end2 = min(k + n - 1, n - 1)
        if m_start2 <= m_end2 and k >= 0:
            grad[m_start2:m_end2 + 1] += a[m_start2 - k:m_end2 - k + 1]
        
        # Actually I think for the full convolution of a with itself:
        # (a*a)[k] = sum_i a[i]*a[k-i]
        # The gradient w.r.t. a[m] is: a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n and m!=k-m to avoid double counting)
        # But when m = k-m (i.e., 2m=k), it's just a[m] (one term, not two).
        # Actually no: the sum is over i, and the term with i=m is a[m]*a[k-m].
        # There's no "other term" - each i gives exactly one product a[i]*a[k-i].
        # So d/da[m] (a*a)[k] = a[k-m] if 0<=k-m < n, else 0.
        
        # Wait, that's only correct if we don't also have the term where k-i = m, i.e., i = k-m.
        # But i and k-i are the two factors. If i = m, the contribution is a[m]*a[k-m], derivative is a[k-m].
        # If k-i = m (i.e., i = k-m), the contribution is a[k-m]*a[m], derivative is a[m].
        # These are the same term! i=m and i=k-m give the same product a[m]*a[k-m].
        # So the derivative is just a[k-m] if 0<=k-m<n.
        
        # Hmm wait, that's only true if m != k-m (i.e., m != k/2). If m = k/2, then i=m=k-m and the term is a[m]^2, derivative is 2*a[m].
        
        # Let me just redo this properly:
        # (a*a)[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (where a[j]=0 for j<0 or j>=n)
        # d/da[m] (a*a)[k] = sum_i [delta(i,m)*a[k-i] + a[i]*delta(k-i,m)]
        # = a[k-m] (if 0<=k-m<n) + (a[k-m] if 0<=k-m<n) [from the second term, i=k-m]
        # Wait, the second term: a[i]*delta(k-i,m) is nonzero when i=k-m, giving a[k-m]*1.
        # But this is the same as the first term when i=m! So we get a[k-m] + a[k-m] = 2*a[k-m]?
        # No: the first term gives a[k-m] when i=m (requires 0<=k-m<n).
        # The second term gives a[k-m] when i=k-m (requires 0<=k-m<n, i.e., m is a valid index for the "other factor").
        # These are the same condition! So the total derivative is 2*a[k-m] if 0<=k-m<n.
        # Unless m = k-m (i.e., 2m=k), in which case we're double-counting the same i, and it's just 2*a[m].
        
        # Actually I think the correct answer is:
        # d/da[m](a*a)[k] = 2*a[k-m] if 0<=k-m<n, else 0
        # (The factor of 2 comes from the two ways m can appear in the product)
        # But wait, when 2m=k, the term in the sum is a[m]*a[m] = a[m]^2, and d/da[m] of that is 2*a[m].
        # And 2*a[k-m] = 2*a[m] in this case. So it's consistent.
        
        # So: grad[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
        # In vector form: grad is a shifted and reversed version of a, times 2.
        
        grad = np.zeros(n)
        # k-m in [0, n-1] means m in [k-n+1, k]
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            # grad[m] = 2*a[k-m] for m in [m_lo, m_hi]
            # k-m ranges from k-m_lo to k-m_hi
            grad[m_lo:m_hi + 1] = 2.0 * a[k - m_hi:k - m_lo + 1][::-1]
        
        # Projected gradient step
        alpha = 0.5 / np.sqrt(t + 1.0) * 0.1
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolution(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            alpha2 = alpha * 0.1
            a_try = a - alpha2 * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                score_try = 2.0 * n * fast_convolution(a_try).max()
                if score_try < best_score:
                    best_score = score_try
                    best_a = a_try.copy()
                    a = a_try
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization with:
    - Fast LP Nesterov (HiGHS + constraint reduction) at multiple lengths
    - LogSumExp gradient refinement
    - Subgradient refinement
    - Peak shaving for final polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    print(f"Time budget: {time_budget:.0f}s")
    
    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Determine starting length
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    remaining = time_budget - (time.monotonic() - t_global_start)
    
    # Phase 1: LP Nesterov at base length
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 450.0)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, K_max=150)
        try_update(a_out, f"lp_n{n_base}")
    
    # Phase 2: LP at longer length n=2000
    n_long1 = max(n_base + 200, 2000)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LP Nesterov at n={n_long1} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        init_long = interpolate_sequence(best_seq, n_long1)
        a_out, _ = nesterov_lp_optimize(init_long, t2, K_max=150)
        try_update(a_out, f"lp_n{n_long1}")
    
    # Phase 3: LSE gradient at best length
    n_current = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LSE gradient at n={n_current} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t3, T_start=50.0, T_end=0.05, n_stages=12)
        try_update(a_out, f"lse_n{n_current}")
    
    # Phase 4: Subgradient refinement
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Subgradient at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 10:
        a_out, _ = subgradient_refine(best_seq, t4)
        try_update(a_out, f"subgrad_n{len(best_seq)}")
    
    # Phase 5: Try even longer length n=2500
    n_long2 = 2500
    if n_long2 != n_current and n_long2 != n_long1:
        remaining = time_budget - (time.monotonic() - t_global_start)
        print(f"\n=== Phase 5: LP at n={n_long2} (remaining={remaining:.0f}s) ===")
        t5 = min(remaining * 0.15, 300.0)
        if t5 > 30:
            init_long2 = interpolate_sequence(best_seq, n_long2)
            a_out, _ = nesterov_lp_optimize(init_long2, t5, K_max=150)
            try_update(a_out, f"lp_n{n_long2}")
    
    # Phase 6: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 5:
        a_out, _ = peak_shaving(best_seq, t6, seed=456)
        try_update(a_out, "peak_shave_1")
    
    # Phase 7: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final LSE polish (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 150.0)
    if t7 > 10:
        a_out, _ = logsumexp_optimize(best_seq, t7, T_start=5.0, T_end=0.02, n_stages=10)
        try_update(a_out, "final_lse")
    
    # Phase 8: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 5:
            a_out, _ = peak_shaving(best_seq, t8, seed=789)
            try_update(a_out, "peak_shave_2")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
