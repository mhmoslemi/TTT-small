```python
    n = len(a)
    conv = fast_convolve(a)
    # LogSumExp gradient weights
    max_conv = conv.max()
    exp_T = np.exp(T * conv - T * max_conv)
    w = exp_T / exp_T.sum()
    
    # Cross-correlation: sum_j w[i+j] * a[j] for j=0..n-1
    # = conv(w, flip(a))[i+n-1]
    a_flip = a[::-1]
    # Use FFT for convolution
    fft_size = 1 << (len(w) + len(a_flip) - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_flip, fft_size)
    cross = irfft(W * A, fft_size)[:len(w) + len(a_flip) - 1]
    # cross[i + n - 1] for i = 0..n-1
    grad_corr = cross[n-1:2*n-1]

    diag = np.zeros(n)
```

But I also need to handle the constraint sum(a) = 1, a >= 0. I'll use L-BFGS-B with the non-negativity constraint and handle the sum constraint by working in a parameterization.

Actually, for simplicity and robustness, let me just use Adam with projection onto {a >= 0, sum(a) = 1} after each step. The projection is:
1. Clip to [0, inf]
2. If sum > 1, scale down (but this might make some elements negative)
3. Better: use the "Euclidean projection onto the simplex" algorithm

Actually, the simplest approach: after each gradient step, clip to [0, inf] and normalize to sum 1. This isn't exactly the Euclidean projection but works well in practice.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-Adam gradient descent
with multi-n exploration and LP polish.

Key innovations vs previous:
- LSE (LogSumExp) smoothed gradient as main workhorse (no LP in main loop)
- FFT-accelerated gradient computation via cross-correlation
- Systematic multi-n scan to find optimal dimension
- Adam optimizer with cosine annealing on LSE temperature T
- LP used only for final polish
- Simplex projection after each step
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the score 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_obj_grad(a, T):
    """
    Compute LogSumExp-smoothed objective and gradient for f(a) = 2n*max(conv(a,a)).
    Assumes sum(a) = 1.
    
    f_T(a) = (2n/T) * log(sum_k exp(T * conv_k))
    grad[i] = (4n/T) * sum_{j=0}^{n-1} w_{i+j} * a_j
    where w = softmax(T * conv)
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    exp_T = np.exp(T * (conv - max_c))
    sum_exp = exp_T.sum()
    w = exp_T / sum_exp
    
    # cross_corr[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # Computed as conv(w, flip(a))[n-1 : 2n-1]
    a_flip = a[::-1]
    # Linear conv of w (len 2n-1) and a_flip (len n) has length 3n-2
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_flip, fft_size)
    cross_full = irfft(W * A, fft_size)
    cross = cross_full[n - 1:2 * n - 1]
    
    grad = (4.0 * n / T) * cross
    
    # Objective: (2n/T) * (T*max_c + log(sum_exp))
    obj = 2.0 * n * max_c + (2.0 * n / T) * np.log(sum_exp)
    
    return obj, grad


def simplex_project(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_adam_optimize(a0, time_limit, T_start=50.0, T_end=2.0, 
                       lr_start=0.01, lr_end=0.001, total_iters=None):
    """
    Adam optimizer on LSE-smoothed objective with temperature annealing.
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    if total_iters is None:
        # Estimate based on time
        # Each iteration: ~2 FFTs of size ~4n, plus overhead
        # For n=1200, fft_size ~ 4096, each FFT ~ 10us
        # Total per iter ~ 50us, so ~20000 iters/second
        # But Python overhead is significant, more like 2000-5000 iters/sec
        total_iters = int(time_limit * 3000)  # conservative estimate
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t_count = 0
    
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature and learning rate
        frac = (it + 1) / total_iters
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute objective and gradient
        obj, grad = lse_obj_grad(a, T)
        
        # Adam update
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a_new = simplex_project(a_new)
        a = a_new
        
        # Check score
        if (it + 1) % 100 == 0 or it == 0:
            sc = score_of(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    # Final check
    sc = score_of(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def lse_lbfgs_optimize(a0, time_limit, T_stages=None):
    """L-BFGS on LSE-smoothed objective."""
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1]
    
    n_stages = len(T_stages)
    
    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        iters_per_stage = max(10, int(remaining * 30 / (n_stages - si)))
        
        def func(x):
            a_trial = simplex_project(x)
            obj, _ = lse_obj_grad(a_trial, T)
            sc = score_of(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
            return obj
        
        def jac(x):
            a_trial = simplex_project(x)
            _, grad = lse_obj_grad(a_trial, T)
            return grad
        
        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': iters_per_stage * 100,
                                   'disp': False})
            a = simplex_project(res.x)
        except Exception:
            pass
        
        elapsed = time.monotonic() - t0
        print(f"    LSE-LBFGS T={T:.2f} ({elapsed:.0f}s): score={score_of(a):.8f}")
    
    return best_a


def build_sparse_conv_matrix(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # For each output position k, T[k, i] = a[k-i] if 0 <= k-i < n
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        # T[k, i] = a[k-i] for k = i, i+1, ..., i+n-1
        rows = np.arange(i, i + n)
        cols = np.full(n, i)
        vals = a.copy()
        row_ind.append(rows)
        col_ind.append(cols)
        data.append(vals)
    T = csr_matrix((np.concatenate(data), 
                    (np.concatenate(row_ind), np.concatenate(col_ind))), 
                   shape=(size, n))
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, T_mat, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_mat @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=10000):
    """Nesterov LP alternating method."""
    n = len(a0)
    a = simplex_project(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        T_mat = build_sparse_conv_matrix(y)
        b_new = solve_lp_best_response(y, T_mat, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = simplex_project(y)
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = simplex_project(a)
        
        for cand in [a, y]:
            sc = score_of(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
        
        if (it + 1) % 50 == 0:
            print(f"    LP iter={it+1}: score={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1 - 1e-4)
        peaks = np.where(b >= threshold)[0]
        
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])
        
        improved = False
        for peak_k in peaks[:5]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            
            for pos in idx[np.argsort(contribs)[::-1][:10]]:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / (others_sum + 1e-30)
                    a_trial = simplex_project(a_trial)
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Try simultaneous reduction
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:5]]
                a_trial = a.copy()
                for pos in top_pos:
                    a_trial[pos] *= 0.999
                a_trial = simplex_project(a_trial)
                sc = score_of(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                elif no_improve > 1000:
                    break
        else:
            no_improve = 0
        
        if (int(time.monotonic() - t0)) % 30 == 0 and (int(time.monotonic() - t0) > 0):
            print(f"    CD {time.monotonic()-t0:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_to(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    return a_new


def structured_init(n, seed=0):
    """Create structured initial sequence."""
    rng = np.random.default_rng(seed)
    choice = seed % 5
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-Adam gradient descent
with systematic multi-n exploration and LP polish.

Key innovations vs previous:
- LSE (LogSumExp) smoothed gradient via FFT cross-correlation as main workhorse
- Systematic multi-n scan to find optimal dimension
- Adam optimizer with cosine annealing and restarts on stagnation
- LP used only for final polish (frees up time for gradient steps)
- Peak-rotation exploration to escape local optima
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the score 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def simplex_project(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_obj_grad(a, T):
    """
    Compute LogSumExp-smoothed objective and gradient for f(a) = 2n*max(conv(a,a)).
    Assumes a >= 0, sum(a) = 1.
    Returns (objective_value, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    exp_T = np.exp(T * (conv - max_c))
    sum_exp = exp_T.sum()
    w = exp_T / sum_exp

    # cross_corr[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # = conv(w, flip(a))[n-1 : 2n-1]
    a_flip = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_flip, fft_size)
    cross_full = irfft(W * A, fft_size)
    cross = cross_full[n - 1:2 * n - 1]

    grad = (4.0 * n / T) * cross
    obj = 2.0 * n * max_c + (2.0 * n / T) * np.log(sum_exp)

    return obj, grad


def lse_adam_optimize(a0, time_limit, T_start=30.0, T_end=3.0,
                      lr_start=0.005, lr_end=0.0005, restart_interval=5000):
    """
    Adam optimizer on LSE-smoothed objective with temperature annealing and restarts.
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t_count = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1e-6))
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        obj, grad = lse_obj_grad(a, T)

        t_count += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = simplex_project(a_new)

        # Check score periodically
        if t_count % 500 == 0:
            sc = score_of(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 500

            # Restart on stagnation
            if no_improve > restart_interval:
                # Perturb toward reducing the second peak
                b = fast_convolve(a)
                peak_idx = np.argmax(b)
                b2 = b.copy()
                b2[peak_idx] = -np.inf
                second_peak = int(np.argmax(b2))
                js = max(0, second_peak - n + 1)
                je = min(n - 1, second_peak)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[second_peak - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:5]]
                for pos in top_pos:
                    a[pos] *= 0.95
                noise = rng.standard_normal(n) * 0.01
                a = simplex_project(a + noise)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

    sc = score_of(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def lse_lbfgs_optimize(a0, time_limit, T_stages=None):
    """L-BFGS on LSE-smoothed objective with staged temperature."""
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1]

    n_stages = len(T_stages)

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit - 3:
            break
        T = T_stages[si]
        remaining = time_limit - (time.monotonic() - t0)
        iters = max(10, int(remaining * 25 / (n_stages - si)))

        def func(x):
            a_trial = simplex_project(x)
            obj, _ = lse_obj_grad(a_trial, T)
            sc = score_of(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
            return obj

        def jac(x):
            a_trial = simplex_project(x)
            _, grad = lse_obj_grad(a_trial, T)
            return grad

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters * 100,
                                    'disp': False})
            a = simplex_project(res.x)
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        print(f"    LSE-LBFGS T={T:.2f} ({elapsed:.0f}s): score={score_of(a):.8f}")

    return best_a, best_score


def build_sparse_conv_matrix(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    row_parts = []
    col_parts = []
    data_parts = []
    for i in range(n):
        rows = np.arange(i, i + n)
        cols = np.full(n, i, dtype=np.int32)
        vals = a.copy()
        row_parts.append(rows)
        col_parts.append(cols)
        data_parts.append(vals)
    T = csr_matrix((np.concatenate(data_parts),
                    (np.concatenate(row_parts), np.concatenate(col_parts))),
                   shape=(size, n))
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, T_mat, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_mat @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=10000):
    """Nesterov LP alternating method for final polish."""
    n = len(a0)
    a = simplex_project(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        T_mat = build_sparse_conv_matrix(y)
        b_new = solve_lp_best_response(y, T_mat, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = simplex_project(y)

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = simplex_project(a)

        for cand in [a, y]:
            sc = score_of(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if (it + 1) % 50 == 0:
            print(f"    LP iter={it + 1}: score={best_score:.8f}")

    return best_a, best_score

def interpolate_to(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    return a_new


def structured_init(n, seed=0):

    choice = seed % 6

    elif choice == 4:

    else:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + np.exp(-8.0 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a_best_scan = pick_best([a1, a2])
        a3 = lse_lbfgs_optimize(a_best_scan, t_sub * 0.20,
                                T_stages=[5.0, 3.0, 2.0, 1.0, 0.5])

        final_cand = pick_best([a1, a2, a3])
        sc = score_of(final_cand)
        scan_results.append((sc, n_try, final_cand))
        print(f"    n={n_try}: score={sc:.8f}")

    # Update best from scan
    scan_results.sort(key=lambda x: x[0])
    for sc, n_val, seq in scan_results[:5]:
        try_update(seq, f"scan_n{n_val}")

    # === Phase 2: Deep LSE-Adam with multiple temperature stages ===
    print(f"\n=== Phase 2: Deep LSE-Adam (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        a_cur = best_seq.copy()
        n_passes = 4
        t_per_pass = t_alloc / n_passes
        T_sched = [(20.0, 8.0), (10.0, 4.0), (5.0, 2.0), (3.0, 1.0)]
        lr_sched = [(0.003, 0.0008), (0.002, 0.0005), (0.001, 0.0003), (0.0005, 0.0001)]

        for p in range(n_passes):
            if time.monotonic() > deadline - 60:
                break
            T_s, T_e = T_sched[p]
            lr_s, lr_e = lr_sched[p]
            t_left = min(t_per_pass, deadline - time.monotonic() - 30)
            if t_left < 5:
                break
            a_deep, _ = lse_adam_optimize(a_cur, t_left,
                                           T_start=T_s, T_end=T_e,
                                           lr_start=lr_s, lr_end=lr_e)
            try_update(a_deep, f"deep_p{p}")
            a_cur = best_seq.copy()

    # === Phase 3: LSE-LBFGS refinement ===
    print(f"\n=== Phase 3: LSE-LBFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        a_lb = lse_lbfgs_optimize(best_seq, t_alloc,
                                   T_stages=[15.0, 8.0, 5.0, 3.0, 2.0, 1.5,
                                             1.0, 0.7, 0.5, 0.3, 0.2, 0.1])
        try_update(a_lb, "lbfgs")

    # === Phase 4: LP Polish ===
    print(f"\n=== Phase 4: LP Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        solver = get_lp_solver()
        print(f"  LP solver: {solver}")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=5000)
        try_update(a_lp, "lp_polish")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc)
        try_update(a_cd, "cd")

    # === Phase 6: Final LP squeeze ===
    print(f"\n=== Phase 6: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem, 100.0)
    if t_alloc > 15:
        solver = get_lp_solver()
        a_fin, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=5000)
        try_update(a_fin, "final_lp")

    # === Phase 7: Final coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_cd2, _ = coordinate_descent_refine(best_seq, rem - 5)
        try_update(a_cd2, "cd_final")

    # Final verification
    if best_seq is not None:
        fs = score_of(best_seq)
        print(f"\nFINAL SCORE: {fs:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
