import numpy as np
import cvxpy as cp
import time

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For column j: non-zero in rows j to j+n-1, with values a[0], a[1], ..., a[n-1]
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = j + i
            if k < size:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver, n_iters_extra=0):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver]
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]:

            prob.solve(solver=s, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val = b_val / s_val
    return b_val, float(max(0.0, M_var.value if M_var.value is not None else 0.0))

where `S = {k : (conv(a,a))_k = max}` and `∇(conv(a,a))_k = 2 Q_k a` (as computed earlier).

So `g(a)_i = (4n/|S|) * sum_{k in S} a_{k-i}` (with out-of-bounds = 0).

This is the gradient of the max function. The function is not smooth (it's the max of smooth functions), so we use subgradient methods.

For the **accelerated** method, we need a Lipschitz constant `L` for the gradient. Since `|g(a)_i| <= 4n * max(a) <= 4n` (since `a_i <= 1`), we have `||g(a)||_2 <= 4n * sqrt(n) = 4n^{3/2}`. This is a very loose bound.

A tighter bound: `||g(a)||_2^2 = (4n/|S|)^2 * ||sum_{k in S} Q_k a||_2^2`. The vector `sum_{k in S} Q_k a` has `i`-th component `sum_{k in S} a_{k-i}`, and its squared norm is `sum_i (sum_{k in S} a_{k-i})^2`. If `|S|` is small (say 1), this is `sum_i a_{k*-i}^2 = sum_i a_i^2 <= 1` (since `a` is a probability vector). So `||g(a)||_2 <= 4n` in this case.

With Lipschitz constant `L = 4n`, the optimal step size for accelerated gradient is `1/(t*L)`, giving convergence rate `O(1/t^2)` for smooth convex problems. For non-smooth, it's `O(1/t)`.

But again, our problem is non-convex, so these bounds don't apply. Still, using `step_t = c/(t+1)` for some constant `c` should work in practice.

Hmm, wait. Let me reconsider the Fessler-Birge update. The standard scheme is:

For minimizing $f(x) = \max_i f_i(x)$ over $x \in C$:

1. $y_0 \in C$
2. For $k = 0, 1, 2, \ldots$:
   - $x_{k+1} \in \arg\min_{x \in C} \langle g(y_k), x \rangle$ where $g(y_k)$ is a subgradient of $f$ at $y_k$
   - $t_{k+1} = \frac{1 + \sqrt{1 + 4t_k^2}}{2}$
   - $y_{k+1} = x_{k+1} + \frac{t_k}{t_{k+1}}(x_{k+1} - x_k)$
   - (Optional) $z_{k+1} = \frac{t_k z_k + x_{k+1}}{t_{k+1}}$

In our case, the "subgradient" step is replaced by the LP best response: instead of taking a gradient step, we find the exact minimizer of $\max_k \text{conv}(y, b)_k$ over the simplex.

So:
- $x_{k+1} = b_k = \text{best\_response}(y_k)$
- $y_{k+1} = b_k + \frac{t_k}{t_{k+1}}(b_k - b_{k-1})$
- $a_{k+1} = \frac{t_k a_k + b_k}{t_{k+1}}$

Let me verify: $y_{k+1} = b_k + \frac{t_k}{t_{k+1}}(b_k - b_{k-1})$. If $b_k = b_{k-1}$ (converged), then $y_{k+1} = b_k$. Good.

The extrapolation factor $\frac{t_k}{t_{k+1}}$ starts at $\frac{1}{(1+\sqrt{5})/2} \approx 0.618$ and approaches 1 as $k \to \infty$. So it's a Nesterov-like momentum that starts moderate and becomes stronger.

This looks correct. Let me proceed.

Now, one issue: the LP best response minimizes $\max_k \text{conv}(y, b)_k$, which is a cross-convolution. But our objective is $\max_k \text{conv}(b, b)_k$ (auto-convolution). The LP gives us a good direction, but the actual improvement in the objective depends on how the auto-convolution changes.

This is the fundamental limitation of the alternating method. To overcome it, I can also check the auto-convolution of the new point and keep track of the best.

Let me now think about the Adam log-sum-exp approach. The key idea is to approximate the max with a log-sum-exp:

$\max_k b_k \approx \frac{1}{T} \log \sum_k e^{T b_k}$

For small $T$, this is a tight approximation. The gradient of this smoothed objective is:

$\nabla_a \text{LSE}(b) = W \cdot \nabla_a b$ where $W = \frac{e^{T b}}{\sum_k e^{T b_k}}$

And $\nabla_a b_k = 2 Q_k a$ (as before), so:

$(\nabla_a \text{LSE})_i = 2 \sum_k W_k a_{k-i}$

In the z-space (log-domain), the gradient is:
$(\nabla_z \text{LSE})_i = a_i ((\nabla_a \text{LSE})_i - \sum_j (\nabla_a \text{LSE})_j a_j)$

This is what the previous code does. Let me keep it but with better hyperparameters.

For the temperature schedule, I'll use:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

Key improvements over previous:
- Vectorized sparse matrix construction via scipy.sparse.diags (O(n) not O(n^2) Python loops)
- Tighter LP tolerances (1e-10) for more accurate best responses

import numpy as np
import cvxpy as cp
import time
import math
from scipy.fft import rfft, irfft
from scipy.sparse import diags as sparse_diags


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    offsets = list(range(n))
    diagonals = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    T = sparse_diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fastest/most accurate."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver]
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]:
        if s not in solvers_to_try:
            solvers_to_try.append(s)

        try:
            if s == cp.CLARABEL:
                prob.solve(solver=s, tol_gap_abs=1e-10, tol_gap_rel=1e-10,
                           tol_feas=1e-10, max_iter=10000)
            elif s == cp.PDLP:
                prob.solve(solver=s, tol_gap=1e-9, tol_feas=1e-9, max_iterations=10000)
            elif s == cp.SCS:
                prob.solve(solver=s, eps=1e-9, max_iters=20000)
            elif s == cp.ECOS:
                prob.solve(solver=s, abstol=1e-9, reltol=1e-9)
            else:
                prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue

        return None, None

    b_val = b_val / s_val
    M_val = float(M_var.value) if M_var.value is not None else 0.0

    stray = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b, M = solve_lp_best_response(y, solver)
        if b is None:
            stray += 1
            if stray > 30:
                break
            continue
        stray = 0

        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t * t))

        # Fessler-Birge extrapolation
        alpha = t / t_new - 1.0
        y = b + alpha * (b - prev_b)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = b.copy()
        else:
            y = y / sy

        # Fessler-Birge averaging
        a = (t * a + b) / t_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = b.copy()
        else:
            a = a / sa

        prev_b = b
        t = t_new

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FesslerBirge iter={it+1} ({elapsed:.0f}s): best={best_score:.10f} M_lp={M:.10f}")

        # Stall detection: perturb every 500 iterations if stuck
        if it > 200 and it % 500 == 0:
            score_check = 2.0 * n * fast_convolve(y).max()
            if score_check > best_score * 1.001:
                x_arr = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 20)
                noise = 0.005 * np.cos(freq * x_arr + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-10:
                    y = y / sy2
                a = a + 0.5 * noise
                a = np.maximum(a, 0)
                sa2 = a.sum()
                if sa2 > 1e-10:
                    a = a / sa2

    return best_a, best_score


def adam_logsumexp(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1, seed=42):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    total_stages = len(T_stages)
    steps_per_stage = max(3000, int(time_limit * 80000 / total_stages))

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > t0 + time_limit:
            break
        remaining = t0 + time_limit - time.monotonic()
        if remaining < 1:
            break

        top_k = max(5, int(top_k_frac * (2 * n - 1)))

        for step in range(steps_per_stage):
            if time.monotonic() > t0 + time_limit:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute weights for log-sum-exp subgradient
            if T_stage < 0.005:
                # Pure active set
                threshold = max_b * (1.0 - 1e-5)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(2 * n - 1)
                w[active] = 1.0 / len(active)
            else:
                if top_k < (2 * n - 1):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(2 * n - 1)
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()

            # Gradient via FFT: g[i] = 2 * sum_k w[k] * a[k-i]
            fft_size = 1 << (4 * n - 1).bit_length()
            W_full = np.zeros(fft_size)
            W_full[:2 * n - 1] = w
            A_full = np.zeros(fft_size)
            A_full[:n] = a
            g_full = irfft(np.conj(rfft(W_full, fft_size)) * rfft(A_full, fft_size), fft_size)
            g_a = 2.0 * g_full[:n]

            # Convert to z-space gradient
            S_dot = np.dot(g_a, a)
            grad_z = a * (g_a - S_dot)

            # Adam update
            lr = lr_init * (lr_decay ** stage_idx)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            step_num = stage_idx * steps_per_stage + step
            m_hat = m / (1.0 - beta1 ** (step_num + 1))
            v_hat = v / (1.0 - beta2 ** (step_num + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Center z for numerical stability
            z -= z.mean()

            if (step + 1) % 20000 == 0:
                elapsed = time.monotonic() - t0
                a_cur = np.exp(z - z.max())
                a_cur = a_cur / a_cur.sum()
                cur_score = 2.0 * n * fast_convolve(a_cur).max()
                print(f"    Adam stage {stage_idx} (T={T_stage:.4f}) step {step+1} ({elapsed:.0f}s): best={best_score:.10f} cur={cur_score:.10f}")

    return best_a, best_score


def direct_subgradient(a0, time_limit, step_scale=0.01, seed=42):
    """FFT-based direct subgradient on active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    scale = 1.0

    fft_size = 1 << (4 * n - 1).bit_length()
    A_full = np.zeros(fft_size)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Adaptive active set
        tighten = min(0.01, 0.001 * (1 + it / 5000))
        threshold = max_b * (1.0 - tighten)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # FFT-based gradient
        w = np.zeros(2 * n - 1)
        w[active] = 2.0 / len(active)
        W_full = np.zeros(fft_size)
        W_full[:2 * n - 1] = w
        A_full[:] = 0
        A_full[:n] = a
        g_full = irfft(np.conj(rfft(W_full, fft_size)) * rfft(A_full, fft_size), fft_size)
        g = g_full[:n]

        # Step
        eta = step_scale * scale / (1.0 + 0.01 * it)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            scale *= 0.85
            no_improve = 0
            if scale < 0.003:
                # Perturb and reset
                x_arr = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 25)
                noise = 0.01 * np.cos(freq * x_arr + rng.uniform(0, 2 * np.pi))
                a = a + noise
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-10:
                    a = a / sa
                scale = 1.0
                step_scale *= 0.9

        if it % 50000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    DirectSubGrad {elapsed:.0f}s: score={best_score:.10f} scale={scale:.4f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x + 0.7)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 0.5 + 0.2 * np.cos(2 * np.pi * 5 * x + 1.0) + 0.15 * np.cos(2 * np.pi * 11 * x + 2.0)
        a = np.maximum(a, 0)
    elif choice == 8:
        a = np.exp(-1.5 * (x - 0.5) ** 2) + 0.3
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 20)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
