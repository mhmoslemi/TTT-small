This IS a standard LP! The variables are `a[0], ..., a[n-1], M` (total n+1 variables) and the constraints are `2n-1` (for the convolution) + 1 (for the sum) + n (for non-negativity).

But the constraint matrix is dense in a specific pattern. For `conv(a,a)[k] <= M`, this is `sum_j a[j]*a[k-j] <= M`, which is QUADRATIC in `a`, not linear!

So it's actually a quadratic program (QP), not an LP. The previous program's LP alternating works because it linearizes by fixing one of the two sequences in the convolution.

So the full problem is a QP:
```
minimize 2n * M
subject to a^T T_k a <= M for k = 0, ..., 2n-2
           sum(a) = 1
           a >= 0
```

where `T_k` is the matrix such that `a^T T_k a = conv(a,a)[k]`.

This is a non-convex QP (since the constraints are quadratic). However, for a fixed `M`, the feasibility problem is also non-convex.

The LP alternating method is a heuristic that works well in practice. Let me stick with it.

Another idea: what about using **semidefinite relaxation**? Relax `a a^T` to a PSD matrix `X`, and solve:
```
minimize 2n * M
subject to trace(X T_k) <= M for all k
           X @ 1 = 0 (or some linearization of sum(a) = 1)
           X >= 0
           X_ii = a_i^2 (but this is also nonlinear)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative weights with adaptive step size for fast initial convergence
2. Nesterov-accelerated LP alternating (main workhorse)
3. Multi-stage LogSumExp L-BFGS with ultra-fine geometric cooling
4. Multi-length exploration (n=1200 parent, n=2000, n=2500)
5. Palindromic variant LP for reduced-dimension search
6. Genetic perturbation for escaping local minima
Key improvements: ultra-fine LSE cooling (40 stages), larger length exploration,
better time management, and palindromic LP variant.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    total_nnz = 0
    for j in range(n):
        count = min(n, size - j)
        total_nnz += count
    rows = np.zeros(total_nnz, dtype=np.int32)
    cols = np.zeros(total_nnz, dtype=np.int32)
    vals = np.zeros(total_nnz, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_best_solver():
    """Find the fastest working LP solver."""
    results = []
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(100, nonneg=True)
            c = np.ones(100)
            prob = cp.Problem(cp.Minimize(c @ x), [cp.sum(x) == 1, x <= 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                results.append((time.monotonic() - t0, solver_name))
        except Exception:
            continue
    if results:
        results.sort()
        return results[0][1]
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.CLARABEL]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                M_val = float(max(0, M_var.value)) if M_var.value is not None else None
                return b_val, M_val
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_and_grad(z, n, temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / temp
    scaled_max = scaled.max()
    exp_terms = np.exp(scaled - scaled_max)
    log_sum = np.log(exp_terms.sum() + 1e-300)
    lse = max_b + temp * (scaled_max + log_sum)
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class LSEFunc:
    def __init__(self, n, temp, deadline):
        self.n = n
        self.temp = temp
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_obj_and_grad(z, self.n, self.temp)
        return obj


class LSEJac:
    def __init__(self, n, temp, deadline):
        self.n = n
        self.temp = temp
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_obj_and_grad(z, self.n, self.temp)
        return g


def multiplicative_weights_adaptive(a0, time_limit, n_steps=500000, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-5)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active]
        w_active = w_active / w_active.sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.7
            if eta < 1e-10:
                break
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating with sparse matrices."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    T_mat = build_conv_matrix_sparse(y)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        T_mat = build_conv_matrix_sparse(y)
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 6:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.01 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
    return best_a, best_score


def lbfgs_refine(a0, time_limit, T_stages, maxiter_per_stage=600):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            if time.monotonic() > deadline:
                break
        except Exception:
            pass
    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-10)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def multi_start_init(n, rng):
    """Generate a diverse initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2 + 0.1
    elif choice == 2:
        a = 1.0 / (1.0 + 5.0 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 7 * x) + 0.2 * np.cos(2 * np.pi * 13 * x)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = np.exp(-3.0 * x) + np.exp(-3.0 * (1 - x))
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
