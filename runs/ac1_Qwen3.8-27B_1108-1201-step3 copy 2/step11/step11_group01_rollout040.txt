    # In numpy: this is np.correlate(w, a, mode='full')[n-1:] ... let me check.

    # Actually np.correlate(w, a, 'full') gives the full cross-correlation.
    # (w ⊛ a)_j = sum_i w[i+j] * a[i] = np.correlate(w, a, 'full')[j]

    corr = irfft(W * np.conj(A), fft_size)[:n]

def lse_lbfgs_optimize(a0, time_limit, tau_schedule, max_iter_per_stage=100):
    """Multi-stage LSE L-BFGS optimization."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    for si, tau in enumerate(tau_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z_val):
            obj, _ = lse_obj_grad_z(z_val, tau, n)
            return obj
        
        def jac(z_val):
            _, g = lse_obj_grad_z(z_val, tau, n)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': max_iter_per_stage * 50,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            if (si + 1) % 3 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage {si+1}/{len(tau_schedule)} tau={tau:.4f}: score={best_score:.8f} ({elapsed:.0f}s)")
        except Exception as e:
            print(f"    LSE stage {si+1} error: {e}")
    
    return best_a, best_score


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    # Build sparse representation: for each k, the non-zero j's are max(0,k-n+1) to min(n-1,k)
    # We'll build it as a dense matrix for simplicity (n=1200 gives 2399x1200 = 2.9M entries)
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed):
    import cvxpy as cp
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp(a0, time_limit, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_best_response(y)
        if b_new is None:
            continue
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

def structured_init(n, rng, method=0):
    x = np.linspace(0, 1, n)
    if method == 0:
        a = np.sin(np.pi * x) ** 2
    elif method == 1:
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif method == 2:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif method == 3:
        # Alternating pattern
        a = np.ones(n)
        a[::2] = 1.5
        a[1::2] = 0.5
    elif method == 4:
        # Random with smoothing
        a = rng.random(n)
        # Simple smoothing
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif method == 5:
        # Saker-type: 1 + alpha * cos with random phase
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 7 * x + rng.random() * 2 * np.pi)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_z(z, tau, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    tb = tau * b
    max_tb = tb.max()
    exp_tb = np.exp(tb - max_tb)
    w = exp_tb / exp_tb.sum()
    logsumexp = max_tb + np.log(exp_tb.sum())
    obj = 2.0 * n * logsumexp
    fft_size = 1 << (len(b) + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 2.0 * tau * corr
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs_optimize(a0, time_limit, tau_schedule, max_iter_per_stage=100, seed=0):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    for si, tau in enumerate(tau_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_val):
            obj, _ = lse_obj_grad_z(z_val, tau, n)
            return obj

        def jac(z_val):
            _, g = lse_obj_grad_z(z_val, tau, n)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': max_iter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            if (si + 1) % 3 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage {si+1}/{len(tau_schedule)} tau={tau:.4f}: score={best_score:.8f} ({elapsed:.0f}s)")
        except Exception:
            pass

    return best_a, best_score


def build_conv_matrix(a_fixed):

    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def nesterov_lp(a0, time_limit, max_iters=1000000, solver=None):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        a = a_new
        y = y_new
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score

def structured_init(n, rng, method=0):
    x = np.linspace(0, 1, n)

    elif method == 2:

        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x + rng.random() * 2 * np.pi)
    elif method == 3:
        a = np.ones(n)
        period = rng.integers(3, 10)
        pattern = np.abs(np.sin(np.pi * np.arange(period)))
        a = np.tile(pattern, n // period + 1)[:n]
    elif method == 4:
        a = rng.random(n)
        kernel = np.ones(min(11, n)) / min(11, n)
        a = np.convolve(a, kernel, mode='same')
    elif method == 5:
        freq1 = rng.integers(5, 30)
        freq2 = rng.integers(3, freq1)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq1 * x) + 0.3 * np.cos(2 * np.pi * freq2 * x + np.pi / 3)
    elif method == 6:
        # Two-lobe: high at edges, low in center
        a = (x - 0.5) ** 2 + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:

    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
