The QP can be solved using projected gradient method or by exploiting the structure. Since H_{k*} has a very specific structure (anti-diagonal), the gradient of the objective is:
∇(a^T H_{k*} a)[m] = 2*a[k*-m] (if valid)

So the projected gradient step is:
a_new = project_simplex(a - eta * 2 * reflect_{k*}(a))

where reflect_{k*}(a)[m] = a[k*-m] if 0 <= k*-m < n, else 0.

This is O(n) per step! And we can do many steps of projected gradient on this QP before returning to the Nesterov outer loop.

So the improved Nesterov iteration is:
1. Compute k* = argmax_k conv(a)[k]
2. Solve (approximately) the QP: min a^T H_{k*} a s.t. a in simplex
   - Use projected gradient: a_temp = project_simplex(a - eta * 2 * reflect_{k*}(a))
   - Do 10-50 steps of this
3. Nesterov update: a_new = combine(a_old, a_temp)

This is much faster per iteration than the LP (O(n * steps) vs O(n^2) for the LP) and gives a better direction.

def solve_qp_best_response(a_init, k_star, n_inner_steps=20):
    """Solve min a^T H_k a s.t. a in simplex using projected gradient."""
    n = len(a_init)
    a = a_init.copy()
    # Step size: L = 2 * ||H_k||_2 = 2 (since H_k has eigenvalues in [-1, 1] for the anti-diagonal)
    # Actually, H_k has exactly 2 non-zero eigenvalues: +2 and -2 (for the pairs), or +1 for the diagonal
    # So ||H_k||_2 <= 2. Use eta = 1/2.
    eta = 0.5  # Conservative
    
    for _ in range(n_inner_steps):
        # Gradient of a^T H_k a at a:
        # ∂/∂a_m = 2 * a[k*m] if 0 <= k-m < n, else 0
        # (the factor of 2 accounts for both the i and j terms)
        grad = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]
        
        a = project_simplex(a - eta * grad)
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated QP best-response (fast O(n) inner steps)
2. LSE L-BFGS multi-stage annealing for fine polishing
3. Multi-length exploration
Key innovation: Instead of LP-based best response, solve the actual QP
(a^T H_k a minimization) via projected gradient, which is much faster per step.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely evaluate using the external evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project vector onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def solve_qp_best_response(a_init, k_star, n_inner_steps=30):
    """
    Solve min a^T H_k a s.t. a in simplex using projected gradient.
    H_k has H_k[i,j]=1 if i+j=k. Gradient at a is 2*a[k-m] (reflected).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    # Lipschitz constant of gradient: ||2*H_k||_2 = 2 (anti-diagonal matrix has spectral norm 2)
    # Actually for the single anti-diagonal, eigenvalues are +-2 (for symmetric case) or +-1
    # Use eta = 1/2 for safety
    eta = 1.0  # Aggressive since H_k is very structured
    
    for _ in range(n_inner_steps):
        # Gradient: grad[m] = 2*a[k-m] if 0 <= k-m < n
        # Use FFT for speed when n is large
        j = k_star - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = np.zeros(n)
        grad[mask] = 2.0 * a[j[mask]]
        
        a = project_simplex(a - eta * grad)
    
    return a


def nesterov_qp(a0, time_limit, n_inner=30, check_interval=50):
    """
    Nesterov-accelerated method with QP best-response inner loop.
    Each outer step: find k*=argmax conv, solve QP for k*, Nesterov update.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_check = t0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute convolution of y (Nesterov extrapolated point)
        b = fast_convolve(y)
        k_star = int(np.argmax(b))
        
        # Solve QP best response
        a_br = solve_qp_best_response(y, k_star, n_inner_steps=n_inner)
        
        # Nesterov update
        t_k = 1.0 + it
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * a_br
        a_new = project_simplex(a_new)
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = project_simplex(y_new)
        
        a = a_new
        y = y_new
        
        # Periodic check
        if it % check_interval == 0 or (time.monotonic() - last_check > 5.0):
            sc_a = score_normalized(a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            sc_y = score_normalized(y)
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
            last_check = time.monotonic()
            if it % (check_interval * 10) == 0:
                print(f"    NesterovQP iter={it}: score={best_score:.8f} k*={k_star} conv_max={b.max():.6f}")

    # Final check
    sc_a = score_normalized(a)
    if sc_a < best_score:
        best_score = sc_a
        best_a = a.copy()
    sc_y = score_normalized(y)
    if sc_y < best_score:
        best_score = sc_y
        best_a = y.copy()

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LSE objective and gradient in log-domain."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = T_val * (b - max_b)
    shifted = np.clip(shifted, -700, 0)
    exp_b = np.exp(shifted)
    w = exp_b / exp_b.sum()
    log_sum_exp = max_b + np.log(exp_b.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient w.r.t. a: g[m] = 2 * sum_k w[k] * a[k-m] (valid)
    g = np.zeros(n)
    # Use the fact that g = 2 * correlate(w, a)[:n]
    # Compute via FFT for speed
    fft_size = 1 << ((2 * n - 1) + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    C = irfft(W * np.conj(A), fft_size)
    g = 2.0 * C[:n]
    
    S = np.dot(g, a)
    grad_z = a * (g - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=12):
    """L-BFGS optimization of LSE approximation with temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 40

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 4 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f}: score={sc:.8f}")
        except Exception as e:
            print(f"    LSE stage {si+1} failed: {e}")
            break

    return softmax_from_z(z)


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T@b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """LP-based best response: min M s.t. conv(a_fixed, b) <= M, b in simplex."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def nesterov_lp(a0, time_limit, n_inner_lp=1, check_interval=100):
    """Nesterov with LP best-response (fallback/supplement)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_check = t0
    it = 0
    solver = None

    # Find a working solver
    for s_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s_name, verbose=False)
            if p.status in ('optimal', 'optimal_inaccurate'):
                solver = s_name
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            b = fast_convolve(y)
            k_star = int(np.argmax(b))
            b_new = solve_qp_best_response(y, k_star, n_inner_steps=n_inner_lp)

        t_k = 1.0 + it
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = project_simplex(a_new)
        y_new = a_new + momentum * (a_new - a_old)
        y_new = project_simplex(y_new)
        a = a_new
        y = y_new

        if it % check_interval == 0 or (time.monotonic() - last_check > 5.0):
            sc_a = score_normalized(a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            last_check = time.monotonic()

    sc_final = score_normalized(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2)
    else:
        # Uniform
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
