        y = y_new

    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fsize)
    return irfft(A * A, fsize)[:size]


def score(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * fast_autocorr(a).max() / (s * s))


def build_T(a_fixed):
    """Build (2n-1) x n matrix T where T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp(T):
    """Solve min M s.t. T@b <= M, 1^T b = 1, b >= 0, M free."""
    n = T.shape[1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([T, -np.ones((T.shape[0], 1))])
    b_ub = np.zeros(T.shape[0])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    except Exception:
        return None, None
    if res.status != 0:
        return None, None
    b = np.maximum(res.x[:n], 0)
    s = b.sum()
    if s < 1e-15:
        return None, None
    return b / s, float(res.x[-1])


def nesterov_lp(a0, time_limit):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / max(s, 1e-15)
    
    y = a.copy()
    y_prev = a.copy()
    best_a = a.copy()
    best_sc = score(a)
    
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    t = 0
    fails = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        T = build_T(y)
        b_star, M_val = solve_lp(T)
        if b_star is None:
            fails += 1
            if fails > 20:
                break
            continue
        fails = 0
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        yn = y_new.sum()
        if yn < 1e-15:
            y_new = y.copy()
        else:
            y_new /= yn
        
        if t == 1:
            a_new = b_star.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            a_new = y_new + mom * (y_new - y)
            a_new = np.maximum(a_new, 0)
            an = a_new.sum()
            a_new = a_new / max(an, 1e-15)
        
        y_prev = y
        y = y_new
        a = a_new
        avg_a += a
        n_avg += 1
        
        # Periodic checks
        check_interval = max(1, t // 100)
        if t % check_interval == 0:
            sc = score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            sc_y = score(y)
            if sc_y < best_sc:
                best_sc = sc_y
                best_a = y.copy()
        
        if n_avg >= 20 and t % max(1, n_avg // 50) == 0:
            sc_avg = score(avg_a / n_avg)
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = avg_a / n_avg
        
        if t % 500 == 0:
            print(f"      NLP t={t} ({time.monotonic()-t0:.0f}s): best={best_sc:.6f}, M={M_val:.6f}, it/s={t/(time.monotonic()-t0):.1f}")
    
    return best_a, best_sc


def lse_optimize(a0, time_limit, T_start=100.0, T_end=1.0, n_stages=10):
    """LogSumExp-smoothed optimization via projected gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / max(s, 1e-15)
    
    best_a = a.copy()
    best_sc = score(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Adaptive learning rate
        lr = 1.0 / (T * np.sqrt(n))
        
        max_iters = max(20, int(remaining * 50 / (n_stages - si)))
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_autocorr(a)
            Tb = T * b
            maxTb = Tb.max()
            expTb = np.exp(Tb - maxTb)
            w = expTb / expTb.sum()
            
            # Gradient: 2 * cross-correlation of a and w
            # grad[j] = 2 * sum_k w[k] * a[k-j] (with valid indices)
            # This is 2 * cross_correlation(a, w)[n-1+j]... 
            # Actually: grad[j] = 2 * sum_i w[i+j] * a[i] = 2 * (a * w_shifted)[j]
            # Let's compute: conv(a[::-1], w) gives cross-correlation
            # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
            # = 2 * convolve(a, w)[j + (2n-2) - j]... 
            # Let me just use the formula: grad = 2 * ifft(FFT(a) * conj(FFT(w))) shifted appropriately
            # Actually simpler: grad[j] = 2 * sum_{i: 0<=i<n, 0<=k-j<n} ... 
            # Let me use the direct formula via FFT:
            # The k-th element of conv(a,a) is sum_i a[i]*a[k-i]
            # d/d a[j] of sum_i a[i]*a[k-i] = 2*a[k-j] (for fixed k)
            # So the gradient of the LSE approximation is:
            # grad[j] = 2 * sum_k w[k] * a[k-j] (where a[k-j] = 0 if k-j not in [0,n-1])
            
            # Compute via FFT: this is the cross-correlation of w with a
            # cross_corr[j] = sum_k w[k] * a[k-j]
            # = convolve(w, a[::-1])[j + n - 1]... 
            # Let me just do it directly for clarity (O(n^2) but n is only 1600)
            # Actually for n=1600, O(n^2) = 2.5M operations, which is fine per iteration
            
            # Use FFT for efficiency
            fsize = 1 << (2 * n - 1 - 1).bit_length()
            A = rfft(a, fsize)
            W = rfft(w, fsize)
            # cross_correlation[j] = sum_k w[k] * a[k-j]
            # = ifft(conj(FFT(w)) * FFT(a))[j + (fsize - n + 1)]... 
            # Actually, let me use the convolution approach:
            # sum_k w[k] * a[k-j] = (w * a)[j] where * is cross-correlation
            # cross_corr = ifft(FFT(w) * conj(FFT(a)))... 
            # Hmm, let me think again.
            # FFT cross-correlation: if x = cross_correlation(w, a), then
            # X[k] = W[k] * conj(A[k]) (in frequency domain)
            # where W = FFT(w), A = FFT(a)
            # So x = ifft(W * conj(A))
            # And x[j] = sum_k w[k] * conj(a[k-j]) = sum_k w[k] * a[k-j] (since a is real)
            
            x = irfft(W * np.conj(A), fsize)
            # x[j] = sum_k w[k] * a[k-j] for j = 0, ...
            # But we need to handle the indexing correctly.
            # The full cross-correlation has length 2n-1.
            # x[0] = sum_k w[k] * a[k] (front part)
            # x[n-1] = sum_k w[k] * a[k-(n-1)] (back part)
            # We want grad[j] = 2 * sum_k w[k] * a[k-j] for j = 0, ..., n-1
            # This is x[0:n] ... wait, that's not right either.
            
            # Let me be very explicit:
            # x = ifft(FFT(w_padded) * conj(FFT(a_padded)))
            # x[m] = sum_{k=0}^{fsize-1} w[k] * a[(k-m) mod fsize]
            # For m = 0: x[0] = sum_k w[k] * a[k mod fsize]
            #   = sum_k w[k] * a[k] (for k < n, a[k]=0 for k >= n)
            # For m = j (0 <= j < n): x[j] = sum_k w[k] * a[(k-j) mod fsize]
            #   = sum_k w[k] * a[k-j] (where a[k-j] = 0 if k-j < 0 or k-j >= n)
            #   = sum_{k=j}^{j+n-1} w[k] * a[k-j] (if we extend a to be 0 outside [0,n-1])
            #   = sum_{i=0}^{n-1} w[i+j] * a[i]
            
            # So x[j] = sum_i w[i+j] * a[i] for j = 0, ..., n-1
            # And grad[j] = 2 * x[j]
            
            grad = 2.0 * x[:n]
            
            # Projected gradient step with projection onto simplex
            a_new = a - lr * grad
            # Project to simplex (Duchi et al. algorithm)
            a_new = project_simplex(a_new)
            
            sc = score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            
            # Backtracking if score increased
            if sc > score(a) * 1.01:
                lr *= 0.5
            else:
                a = a_new
        
    return best_a, best_sc


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.argmax(cond)
    lambda_ = cssv[k] / (k + 1)
    return np.maximum(v - lambda_, 0)


def peak_shave(a0, time_limit, seed=123):
    """Coordinate descent peak shaving."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / max(s, 1e-15)
    best_a = a.copy()
    best_sc = score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]
        improved = False
        
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            src = idx[np.argmax(contrib)]
            
            # Find destination: low-value position far from peaks
            peak_set = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j2 in range(js2, je2 + 1):
                    peak_set.add(j2)
            
            candidates = [j for j in range(n) if j not in peak_set and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Sort by value (ascending) and pick from the lowest
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            
            for trial in range(3):
                dst = dst_pool[rng.integers(len(dst_pool))]
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = a_t / a_t.sum()
                    sc = score(a_t)
                    if sc < best_sc - 1e-14:
                        a = a_t
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_t = a.copy()
                a_t[i] -= amt
                a_t[j] += amt
                a_t = a_t / max(a_t.sum(), 1e-15)
                sc = score(a_t)
                if sc < best_sc:
                    a = a_t
                    best_sc = sc
                    best_a = a.copy()
                    stall = 0
            
            if stall > 500:
                break
        
        if t0 > 0 and (time.monotonic() - t0) % 30 < 0.1:
            print(f"      Peak: {best_sc:.6f} (t={time.monotonic()-t0:.0f}s, stall={stall})")
    
    return best_a, best_sc


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    a_norm = a / max(s, 1e-15)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ns = a_new.sum()
    return a_new / max(ns, 1e-15)


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.2 * np.cos(5 * x)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.7 * np.exp(-5 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)
        a[:n//4] *= 0.5
        a[-n//4:] *= 0.5
    a = np.maximum(a, 0)
    s = a.sum()
    return a / max(s, 1e-15)


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
