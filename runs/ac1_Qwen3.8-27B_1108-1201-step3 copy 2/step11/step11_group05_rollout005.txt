def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient(a, w):
    """Compute g[i] = 2 * sum_j a[j] * w[i+j] using FFT."""
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    X = rfft(a, fft_size)
    Y = rfft(w, fft_size)
    cross = irfft(np.conj(X) * Y, fft_size)
    return 2.0 * cross[:n]

def evaluate_local(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def mirror_descent_optimize(a0, time_limit, seed=0):
    """Mirror descent with adaptive temperature and step size."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    eta = 0.05
    T = 5.0
    momentum = 0.85
    v = np.zeros(n)
    step = 0
    no_improve = 0
    patience = 200
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        max_c = c.max()
        score = 2.0 * n * max_c
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        if no_improve > patience:
            # Restart with perturbation
            T = min(10.0, T * 3.0)
            perturb = 0.02 * rng.standard_normal(n)
            a_pert = a + perturb * a
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a = a_pert / sp
            v = np.zeros(n)
            eta = 0.05
            no_improve = 0
            patience = max(50, patience * 2)
        else:
            # Cool temperature
            T = max(0.005, T * 0.9995)
        
        # Compute weights
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted / T)
        w = exp_c / exp_c.sum()
        
        # Compute gradient
        g = compute_gradient(a, w)
        
        # Momentum
        v = momentum * v + (1 - momentum) * g
        
        # Mirror descent update
        scale = -eta * v
        scale -= scale.max()  # Numerical stability
        a_new = a * np.exp(scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate
        c_new = fast_convolve(a_new)
        score_new = 2.0 * n * c_new.max()
        
        if score_new <= score * (1 + 1e-10):
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta = max(eta * 0.7, 0.001)
        
        step += 1
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD step {step} ({elapsed:.0f}s): T={T:.4f}, eta={eta:.4f}, best={best_score:.8f}")
    
    return best_a, best_score

```python
def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Binary search for threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n+1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    proj = np.maximum(v - theta, 0)
    return proj / proj.sum()  # Ensure exact normalization
```

def solve_qp_best_response(lam, n, n_iters=100):
    """Solve min a^T Q a s.t. sum(a)=1, a>=0 where Q[i,j]=lam[i+j]."""
    a = np.ones(n) / n
    eta = 1.0 / (2.0 * n)  # Step size
    
    for _ in range(n_iters):
        # Compute gradient: 2 * Q @ a
        # Q[a] = sum_j lam[i+j] * a[j] for each i
        # This is the same as compute_gradient but without the factor of 2
        fft_size = 1 << (3 * n - 2).bit_length()
        A = rfft(a, fft_size)
        L = rfft(lam, fft_size)
        Qa = irfft(np.conj(A) * L, fft_size)[:n]
        grad = 2.0 * Qa
        
        a = project_simplex(a - eta * grad)
    
    return a
