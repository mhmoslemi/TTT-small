2. FISTA-style Nesterov LP alternating with adaptive restarts

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    n = len(a)
    size = 2 * n - 1
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # So T @ b = convolve(a, b)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if k < size:
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue

def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        problem = LSEProblem(n, stage_T, deadline)
        try:
            res = minimize(
                fun=problem, x0=z, jac=problem.jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)
