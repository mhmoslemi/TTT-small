"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max smoothed gradient descent in log-space with Nesterov momentum
2. Very small temperatures (T ~ 0.001-0.1) for fine-tuning near the optimum
3. Backtracking line search for robust step size selection
4. Peak-directed coordinate moves for final polishing
5. Multi-start with diverse initializations and multi-length exploration
Key improvements: much finer temperature range for local optimization,
adaptive step sizes, and targeted peak-reducing moves.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)

def compute_score_normalized(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def smoothed_obj_and_grad(z, T, n, fft_size_conv, fft_size_corr):
    """
    Compute soft-max smoothed objective and gradient in log-space.
    f_smooth = T * log(sum_k exp(conv_k / T))
    grad_z = a * (grad_a - dot(grad_a, a))
    """
    a = softmax_from_z(z)
    conv = fast_convolve(a)  # length 2n-1
    
    # Softmax weights
    conv_shifted = conv / T
    conv_max = conv_shifted.max()
    exp_conv = np.exp(conv_shifted - conv_max)
    w = exp_conv / exp_conv.sum()
    
    # Objective
    obj = T * (conv_max + np.log(exp_conv.sum()))
    
    # Gradient w.r.t. a: grad_a[j] = 2 * sum_k w[k] * a[k-j]
    # This is 2 * cross_correlation(w, a) [:n]
    W = rfft(w, fft_size_corr)
    A = rfft(a, fft_size_corr)
    grad_a = 2.0 * irfft(W * np.conj(A), fft_size_corr)[:n]
    
    # Transform to log-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def nesterov_softmax_descent(z0, T, n, fft_size_corr, time_limit, 
                               max_iter=100000, lr_init=0.1, momentum=0.9,
                               lr_decay=0.99999, verbose=False):
    """
    Nesterov accelerated gradient descent for soft-max smoothed objective.
    Works in log-space z. Uses backtracking line search.
    """
    z = z0.copy()
    v = np.zeros(n)
    best_z = z.copy()
    
    a0 = softmax_from_z(z)
    best_obj = T * np.log(np.sum(np.exp(fast_convolve(a0) / T)))
    # More stable:
    conv0 = fast_convolve(a0)
    conv0_max = (conv0 / T).max()
    best_obj = T * (conv0_max + np.log(np.exp(conv0 / T - conv0_max).sum()))
    
    t0 = time.monotonic()
    lr = lr_init
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Nesterov lookahead
        z_la = z - momentum * v
        obj_la, grad_la = smoothed_obj_and_grad(z_la, T, n, fft_size_conv, fft_size_corr)
        
        # Update velocity
        v = momentum * v + lr * grad_la
        
        # Proposed step
        z_new = z - v
        
        # Clip to prevent numerical issues
        z_new = np.clip(z_new, -30, 30)
        
        # Evaluate at new point
        a_new = softmax_from_z(z_new)
        conv_new = fast_convolve(a_new)
        cn_max = (conv_new / T).max()
        obj_new = T * (cn_max + np.log(np.exp(conv_new / T - cn_max).sum()))
        
        # If objective increased significantly, do linear search
        if obj_new > obj_la + 1e-10:
            # Backtracking: reduce step
            for shrink in range(10):
                z_try = z - (v / (2 ** shrink))
                z_try = np.clip(z_try, -30, 30)
                a_try = softmax_from_z(z_try)
                conv_try = fast_convolve(a_try)
                ct_max = (conv_try / T).max()
                obj_try = T * (ct_max + np.log(np.exp(conv_try / T - ct_max).sum()))
                if obj_try <= obj_la + 1e-10:
                    z_new = z_try
                    obj_new = obj_try
                    v = v / (2 ** shrink)
                    break
            else:
                # All backtracking failed, skip this step
                z = z_la
                continue
        
        z = z_new
        
        # Update best
        a_best_check = softmax_from_z(z)
        conv_best = fast_convolve(a_best_check)
        cb_max = (conv_best / T).max()
        obj_best = T * (cb_max + np.log(np.exp(conv_best / T - cb_max).sum()))
        
        if obj_best < best_obj:
            best_obj = obj_best
            best_z = z.copy()
        
        # Decay learning rate
        lr *= lr_decay
        
        if verbose and iters % 1000 == 0:
            a_check = softmax_from_z(z)
            true_score = compute_score_normalized(a_check)
            print(f"    iter={iters}, T={T:.4f}, smooth_obj={obj_best:.8f}, true_score={true_score:.8f}")
    
    a_out = softmax_from_z(best_z)
    score = compute_score_normalized(a_out)
    return a_out, score

def peak_directed_moves(a0, time_limit, n_top=5, verbose=False):
    """
    Directly target the top peaks of the convolution and try to reduce them
    by adjusting the contributing coefficients.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score_normalized(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        
        # Find top peaks
        top_indices = np.argsort(conv)[::-1][:n_top]
        
        improved = False
        for k in top_indices:
            k = int(k)
            # Pairs contributing to conv[k]: (i, k-i) for i where both are valid
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            
            if i_start >= i_end:
                continue
                
            idxs = np.arange(i_start, i_end + 1)
            # Contribution of pair (i, k-i) is a[i] * a[k-i]
            # (or a[i]^2 if i = k-i)
            contribs = np.array([a[i] * a[k - i] for i in idxs])
            
            # Find the largest contributor
            max_ci = np.argmax(contribs)
            i_max = idxs[max_ci]
            j_max = k - i_max
            
            if i_max == j_max:
                # Single element squared
                # Try reducing a[i_max]
                for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                    delta = a[i_max] * frac
                    a_trial = a.copy()
                    a_trial[i_max] -= delta
                    # Redistribute to others proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[i_max] = False
                    others = a[mask]
                    os = others.sum()
                    if os < 1e-15:
                        continue
                    a_trial[mask] = a[mask] + delta * a[mask] / os
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score_normalized(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Try transferring mass from the larger to the smaller
                if a[i_max] > a[j_max]:
                    src, dst = i_max, j_max
                else:
                    src, dst = j_max, i_max
                
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    delta = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score_normalized(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            # Try random small perturbations
            for _ in range(20):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = rng.standard_normal() * 0.005 * max(a[i], 1e-8)
                a_trial = a.copy()
                a_trial[i] -= delta
                a_trial[j] += delta
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if verbose and iters % 100 == 0:
            print(f"    PeakDir iter={iters}, best={best_score:.10f}")
    
    return best_a, best_score

def coordinate_hill_climbing(a0, time_limit, n_pairs=500, verbose=False):
    """
    Fast coordinate-wise hill climbing: for each coordinate, try small
    perturbations and keep the best.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try several transfer amounts
            base = max(a[i], a[j], 1e-8)
            for sign in [1, -1]:
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    t = sign * frac * base
                    if a[i] + (t if sign < 0 else -t) < -1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] += t
                    a_trial[j] -= t
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score_normalized(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved and iters > 20:
            # Perturb to escape
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            # Reset to best if perturbation didn't help
            if compute_score_normalized(a) > best_score * 1.01:
                a = best_a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

**Simplex projection algorithm:**

def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u - (cssv - 1) / ind > 0
    rho = ind[cond][-1]
    theta = (cssv[cond][-1] - 1) / rho
    return np.maximum(v - theta, 0)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute cross-correlation (w star a)[j] = sum_k w[k]*a[k-j] for j=0..len(a)-1."""
    n = len(a)
    lw = len(w)
    size = lw + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(W * np.conj(A), fft_size)[:n]


def project_simplex(v):
    """Project vector v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = (cssv[cond][-1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / s

    lr = lr_init
    iters = 0
    fft_size_corr = 1 << (((2 * n - 1) + n - 1) - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Nesterov lookahead
        a_la = project_simplex(a - momentum * v)
        
        # Compute gradient at lookahead point
        conv_la = fast_convolve(a_la)
        w = softmax_weights(conv_la, T)
        grad_a = 2.0 * fast_cross_corr(w, a_la)
        
        # Update velocity
        v = momentum * v + lr * grad_a
        
        # Proposed step
        a_new = project_simplex(a - v)
        
        # Evaluate new point
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()
        
        # Smoothed objective for acceptance
        cn_shift = conv_new / T
        cn_mx = cn_shift.max()
        smooth_new = T * (cn_mx + np.log(np.exp(cn_shift - cn_mx).sum()))
        
        # Backtracking if smoothed objective increased
        if smooth_new > best_smooth + 1e-12:
            accepted = False
            for shrink in range(8):
                factor = 1.0 / (2.0 ** (shrink + 1))
                a_try = project_simplex(a - v * factor)
                conv_try = fast_convolve(a_try)
                ct_shift = conv_try / T
                ct_mx = ct_shift.max()
                smooth_try = T * (ct_mx + np.log(np.exp(ct_shift - ct_mx).sum()))
                if smooth_try <= best_smooth + 1e-12:
                    a_new = a_try
                    score_new = 2.0 * n * conv_try.max()
                    smooth_new = smooth_try
                    v = v * factor
                    accepted = True
                    break
            if not accepted:
                # Skip this iteration
                a = a_la
                continue
        
        a = a_new
        
        # Update best
        if smooth_new < best_smooth:
            best_smooth = smooth_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
            elif score_new < best_score + 1e-10:
                # Keep tracking even if smooth improved but score didn't quite
                best_a = a.copy()
                if score_new < best_score:
                    best_score = score_new
        
        lr *= lr_decay
        
        if verbose and iters % 2000 == 0:
            print(f"      P


GD iter={iters}, score={best_score:.10f}, smooth={best_smooth:.8e}")
    
    return best_a, best_score


def fictitious_play_step(a, M):
    """
    Solve the LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0.
    Returns the optimal b (normalized to sum=1).
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A: A[k,j] = a[k-j] if 0 <= k-j <= n-1
    # For efficiency, build it as a sum of shifted diagonals
    A = np.zeros((num_constraints, n))
    for j in range(n):
        # For variable b[j], it contributes to constraints k where 0<=k-j<=n-1
        # i.e., k ranges from j to j+n-1
        k_start = j
        k_end = min(num_constraints, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    # LP: maximize sum(b) = minimize -sum(b)
    # subject to: A @ b <= M, b >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 10, 'presolve': True})
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    
    return None
