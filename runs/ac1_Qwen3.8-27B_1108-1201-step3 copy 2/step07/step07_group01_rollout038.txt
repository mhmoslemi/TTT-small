"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the convex LP approach: at each step, solve an LP to find the best direction
g that maximizes sum(g) while keeping conv(f,g) bounded. Update f towards g.
Combined with adaptive step size, multi-start, and final polishing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, solver=cp.GLPK):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.asarray(g.value, dtype=np.float64)
    g_val = np.maximum(g_val, 0)
    if g_val.sum() < 1e-15:
        return None
    return g_val


def lp_iterations(f_init, time_budget, verbose=True):
    """Run LP-based iterations."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 0).astype(np.float64)
    s = f.sum()
    f = f / s  # normalize to sum=1
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    best_score = 2.0 * n * best_M  # sum=1, so score = 2n*max_b
    
    t0 = time.monotonic()
    step = 0.1  # initial step size
    it = 0
    
    # Try different solvers
    solvers_to_try = [cp.CLARABEL, cp.ECOS, cp.GLPK, cp.CBC]
    working_solver = None
    
    while time.monotonic() - t0 < time_budget:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Solve LP
        if working_solver is None:
            for solver in solvers_to_try:
                try:
                    g = solve_lp(f, M, solver=solver)
                    if g is not None:
                        working_solver = solver
                        break
                except Exception:
                    continue
        else:
            g = solve_lp(f, M, solver=working_solver)
        
        if g is None:
            # Try next solver
            for solver in solvers_to_try:
                if solver != working_solver:
                    try:
                        g = solve_lp(f, M, solver=solver)
                        if g is not None:
                            working_solver = solver
                            break
                    except Exception:
                        continue
            if g is None:
                if verbose:
                    print(f"  LP failed at iter {it}, breaking")
                break
        
        # Rescale g to sum=1
        g_sum = g.sum()
        g = g / g_sum
        
        # Line search for step size
        improved = False
        for trial_step in [step, step * 0.5, step * 0.25, step * 0.1]:
            f_trial = (1 - trial_step) * f + trial_step * g
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            M_trial = fast_convolve(f_trial).max()
            if M_trial < M * (1 - 1e-10):  # strict improvement
                f = f_trial
                M = M_trial
                improved = True
                step = min(step * 1.1, 0.5)  # increase step slightly
                break
        
        if not improved:
            # Try even smaller step
            for trial_step in [0.01, 0.005, 0.001]:
                f_trial = (1 - trial_step) * f + trial_step * g
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                M_trial = fast_convolve(f_trial).max()
                if M_trial < M * (1 - 1e-10):
                    f = f_trial
                    M = M_trial
                    step = trial_step
                    break
        
        # Track best
        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={it} ({elapsed:.0f}s): M={M:.8f} best_M={best_M:.8f} step={step:.4f}")
    
    # Return normalized
    best_f = best_f / best_f.sum()
    final_score = 2.0 * n * fast_convolve(best_f).max()
    return best_f, final_score

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def score_normalized(a):
    """Score for a normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(2.0 * len(a) * b.max())


def solve_lp_single(f, M, solver):
    """Solve LP: max 1^T g s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=solver)
    if g.value is None or prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    return np.maximum(np.asarray(g.value, dtype=np.float64), 0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    M = fast_convolve(f).max()
    best_M = M

    t0 = time.monotonic()
    step = 0.15
    it = 0
    stall = 0
    last_solve_time = 0.0

    solver_candidates = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]
    solver = None
    for sol in solver_candidates:
        t_solve_start = time.monotonic()
        try:
            g = solve_lp_direction(f, best_M, sol)
            last_solve_time = time.monotonic() - t_solve_start
            if g is not None:
                solver = sol
                if verbose:
                    print(f"    [{label}] Solver: {sol} (solve time: {last_solve_time:.2f}s)")
                break
        except Exception:
            continue

    if solver is None:
        if verbose:
            print(f"    [{label}] No LP solver available, returning init.")
        return best_f, 2.0 * n * best_M

    while time.monotonic() - t0 < time_budget:
        it += 1
        elapsed_total = time.monotonic() - t0
        remaining = time_budget - elapsed_total
        if remaining < 2.0:
            break

        b = fast_convolve(f)
        M = b.max()

        t_solve_start = time.monotonic()
        g = solve_lp_direction(f, M, solver)
        last_solve_time = time.monotonic() - t_solve_start
        if g is None:
            stall += 20
            if stall > 200:
                break
            continue

        # Line search: try multiple steps, keep best
        best_trial_M = M
        best_trial_f = None
        candidates = [step, step * 0.7, step * 0.5, step * 0.3, 0.08, 0.04, 0.02, 0.01]
        for t_val in candidates:
            if t_val >= 0.99 or t_val <= 0:
                continue
            f_try = (1.0 - t_val) * f + t_val * g
            f_try = np.maximum(f_try, 0.0)
            ss = f_try.sum()
            if ss < 1e-10:
                continue
            f_try = f_try / ss
            M_try = fast_convolve(f_try).max()
            if M_try < best_trial_M - 1e-14:
                best_trial_M = M_try
                best_trial_f = f_try.copy()

        if best_trial_f is not None:
            f = best_trial_f
            M = best_trial_M
            stall = 0
            step = min(step * 1.02, 0.4)
        else:
            stall += 1
            step = max(step * 0.6, 0.01)
            if stall > 60:
                rng_loc = np.random.default_rng(it * 7919 + 31)
                noise = rng_loc.standard_normal(n) * 0.015
                f = f + noise
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                stall = 0
                step = 0.15

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()

        if verbose and it % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{label}] it={it} ({elapsed:.0f}/{time_budget:.0f}s) "
                  f"M={M:.9f} best={best_M:.9f} step={step:.4f} solve={last_solve_time:.2f}s")

    best_f = best_f / best_f.sum()
    return best_f, 2.0 * n * fast_convolve(best_f).max()


def mirror_descent(a_init, time_budget, T_start=0.008, T_end=0.0005,
                   eta_start=0.01, eta_end=0.001, label="MD"):
    """Mirror descent with temperature annealing."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_budget:
            break
        progress = min(1.0, it / 50000.0)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        M = b.max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()

        scaled = b / T
        mx = scaled.max()
        exp_s = np.exp(scaled - mx)
        w = exp_s / exp_s.sum()

        fft_sz = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_sz)
        A = rfft(a, fft_sz)
        correl = irfft(np.conj(W) * A, fft_sz)[:n]
        grad = 4.0 * n * correl
        grad = grad - np.dot(a, grad)

        log_upd = np.clip(-eta * grad, -3.0, 3.0)
        a = a * np.exp(log_upd)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

    best_a = best_a / best_a.sum()
    return best_a, 2.0 * n * fast_convolve(best_a).max()


def coord_descent(a_init, time_budget, label="CD"):
    """Coordinate descent: reduce peak by adjusting individual coordinates."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    factors = [0.95, 0.9, 0.85, 0.8, 0.7, 0.6, 0.5, 0.4]

    while time.monotonic() - t0 < time_budget and no_improve < 400:
        b = fast_convolve(a)
        M = b.max()
        no_improve += 1
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:10]

        improved = False
        for ci in top:
            if improved:
                break
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            for fct in factors:
                a_try = a.copy()
                old = a_try[i]
                a_try[i] = old * fct
                diff = old - a_try[i]
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_try[mask]
                ot = others.sum()
                if ot < 1e-15:
                    continue
                a_try[mask] = others + diff * others / ot
                a_try = np.maximum(a_try, 0.0)
                ss = a_try.sum()
                if ss < 1e-15:
                    continue
                a_try = a_try / ss
                sc = 2.0 * n * fast_convolve(a_try).max()
                if sc < best_score - 1e-13:
                    a = a_try
                    best_a = a.copy()
                    best_score = sc
                    no_improve = 0
                    improved = True
                    break

        if not improved and no_improve > 80:
            rng_pd = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            mask_r = rng_pd.random(n) < 0.08
            if mask_r.any():
                a[mask_r] *= (1.0 + rng_pd.standard_normal(int(mask_r.sum())) * 0.04)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0

    return best_a, best_score


def peak_equalize(a_init, time_budget, label="PEQ"):
    """Peak equalization: iteratively reduce the highest peak by adjusting its contributors."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(5000):
        if time.monotonic() - t0 > time_budget:
            break
        b = fast_convolve(a)
        M = b.max()
        if 2.0 * n * M < best_score - 1e-14:
            best_score = 2.0 * n * M
            best_a = a.copy()

        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top3 = np.argsort(contribs)[::-1][:3]

        improved = False
        for ci in top3:
            if improved:
                break
            i = int(idx[ci])
            for fct in [0.92, 0.85, 0.75, 0.6]:
                a_try = a.copy()
                old = a_try[i]
                a_try[i] = old * fct
                diff = old - a_try[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_try[mask]
                ot = others.sum()
                if ot < 1e-15:
                    continue
                a_try[mask] = others + diff * others / ot
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try / a_try.sum()
                sc = 2.0 * n * fast_convolve(a_try).max()
                if sc < best_score - 1e-13:
                    a = a_try
                    best_a = a.copy()
                    best_score = sc
                    improved = True
                    break
        if not improved:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase LP-based + complementary optimization.
    Phase 1: LP iterations from parent (primary)
    Phase 2: LP from perturbed start
    Phase 3: Mirror descent
    Phase 4: Peak equalization + CD
    Phase 5: Final LP polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            else:
                print("Parent score not finite, using uniform")
                best_seq = np.ones(768) / 768
                best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Uniform init: n=768, score={best_score:.7f}")

    n_best = len(best_seq)

    # === Phase 1: Main LP from parent ===
    rem = deadline - time.monotonic() - 350
    if rem > 90:
        t_alloc = min(rem * 0.55, 700.0)
        print(f"\n=== Phase 1: LP main (budget={t_alloc:.0f}s, n={n_best}) ===")
        a1, s1 = lp_optimize(best_seq, t_alloc, label="LP_main")
        if s1 < best_score:
            best_score = s1
            best_seq = a1
            print(f"  IMPROVED: {best_score:.7f}")

    # === Phase 2: LP from perturbed start ===
    rem = deadline - time.monotonic() - 200
    if rem > 60:
        t_alloc = min(rem * 0.35, 250.0)
        rng2 = np.random.default_rng(seed + 100)
        # Create perturbation: blend with a smooth profile
        smooth = np.linspace(1.0, 1.0, n_best)
        noise = rng2.standard_normal(n_best) * 0.4
        a_div = 0.5 * best_seq + 0.3 * smooth + 0.2 * noise
        a_div = np.maximum(a_div, 0)
        a_div = a_div / a_div.sum()
        print(f"\n=== Phase 2: LP perturbed (budget={t_alloc:.0f}s) ===")
        a2, s2 = lp_optimize(a_div, t_alloc, label="LP_pert")
        if s2 < best_score:
            best_score = s2
            best_seq = a2
            print(f"  IMPROVED: {best_score:.7f}")

    # === Phase 3: Mirror descent ===
    rem = deadline - time.monotonic() - 100
    if rem > 40:
        t_alloc = min(rem * 0.35, 120.0)
        print(f"\n=== Phase 3: Mirror Descent (budget={t_alloc:.0f}s) ===")
        a3, s3 = mirror_descent(best_seq, t_alloc,
                                T_start=0.005, T_end=0.0003,
                                eta_start=0.008, eta_end=0.0005, label="MD")
        if s3 < best_score:
            best_score = s3
            best_seq = a3
            print(f"  IMPROVED: {best_score:.7f}")

    # === Phase 4: Peak equalize + CD ===
    rem = deadline - time.monotonic() - 50
    if rem > 20:
        t_alloc1 = min(rem * 0.4, 50.0)
        print(f"\n=== Phase 4a: Peak Equalize (budget={t_alloc1:.0f}s) ===")
        a4, s4 = peak_equalize(best_seq, t_alloc1, label="PEQ")
        if s4 < best_score:
            best_score = s4
            best_seq = a4
            print(f"  IMPROVED: {best_score:.7f}")

        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc2 = min(rem * 0.5, 30.0)
            print(f"=== Phase 4b: CD (budget={t_alloc2:.0f}s) ===")
            a5, s5 = coord_descent(best_seq, t_alloc2, label="CD")
            if s5 < best_score:
                best_score = s5
                best_seq = a5
                print(f"  IMPROVED: {best_score:.7f}")

    # === Phase 5: Final LP polish ===
    rem = deadline - time.monotonic() - 5
    if rem > 20:
        t_alloc = min(rem * 0.7, 80.0)
        print(f"\n=== Phase 5: Final LP (budget={t_alloc:.0f}s) ===")
        a6, s6 = lp_optimize(best_seq, t_alloc, label="LP_final")
        if s6 < best_score:
            best_score = s6
            best_seq = a6
            print(f"  IMPROVED: {best_score:.7f}")

    # === Phase 6: Tiny final CD ===
    rem = deadline - time.monotonic() - 3
    if rem > 5:
        a7, s7 = coord_descent(best_seq, min(rem * 0.7, 8.0), label="CD_fin")
        if s7 < best_score:
            best_score = s7
            best_seq = a7
            print(f"  IMPROVED: {best_score:.7f}")

    # Final verification
    result = [float(max(0.0, x)) for x in best_seq]
    fs = float(evaluate_sequence(result))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Fallback
        result = [1.0 / 768] * 768
        best_score = 2.0

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(result)})")
    return result
