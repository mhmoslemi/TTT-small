def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix_indices(n):
    """Precompute sparse structure of the convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_best_response(a, M, row_idx, col_idx):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0 using LP."""
    n = len(a)
    # Build values for the constraint matrix
    vals = np.empty(len(row_idx), dtype=np.float64)
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        j = col_idx[idx]
        vals[idx] = a[k - j]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    
    # Target: flat power spectrum
    # |A(f)|^2 should be as uniform as possible
    # We want to iteratively adjust a to make |A|^2 flat
    
    t = 0.1  # step size for soft update
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        A = rfft(a, 2 * n - 1)
        power = np.abs(A) ** 2
        total_power = power.sum()
        
        # Target: uniform power
        # But we need to be careful about DC component
        target_power = total_power / len(power)
        
        # New amplitude: blend current with target
        # |A_new| = |A| * (target_power / power)^0.5 * t + |A| * (1-t)
        # = |A| * (1 - t + t * sqrt(target_power / power))
        ratio = np.sqrt(target_power / (power + 1e-30))
        new_amp = np.abs(A) * (1 - t + t * np.minimum(ratio, 3.0))
        new_phase = np.angle(A)
        
        A_new = new_amp * np.exp(1j * new_phase)
        a_new = irfft(A_new, 2 * n - 1)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            t *= 0.9
            if t < 0.001:
                break
        
        t = min(t * 1.01, 0.15)  # Slowly increase step
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spec iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate using the actual evaluator and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):

    Multi-phase optimization:
    Phase 1: LP-based fictitious play (primary)
    Phase 2: Multiplicative weights (approximate, faster)
    Phase 3: Spectral refinement (novel approach)
    Phase 4: Peak-focused coordinate descent
    Phase 5: Try nearby lengths with LP
    Phase 6: Final coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = 2.0
        print(f"Initialized uniform: n={n_init}, score=2.0")
    
    n_parent = len(parent)
    
    # === Phase 1: LP-based Fictitious Play (main workhorse) ===
    print(f"\n=== Phase 1: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400.0, rem * 0.30)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")

    # === Phase 3: Multiplicative Weights ===
    print(f"\n=== Phase 3: Multiplicative Weights (best={best_score:.8f}) ===")

        best_seq, best_score = check_and_update(a_mw, "MW", best_seq, best_score)
        
        # Try with different T
        if rem - (time.monotonic() - (deadline - rem)) > 100:
            rem = deadline - time.monotonic()
            if rem > 50:
                t3b = min(100.0, rem * 0.15)
                print(f"  MW T=2.0, budget: {t3b:.0f}s")

    # === Phase 8: Final CD Polish ===
    print(f"\n=== Phase 8: Final CD Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:

    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")
    return [float(x) for x in best_seq]
