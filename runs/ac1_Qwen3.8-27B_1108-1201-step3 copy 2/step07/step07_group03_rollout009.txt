```python
        # = (2/T) * np.correlate(w, a, 'valid')

            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Anneal temperature
        T *= T_decay
        lr *= lr_decay
    
    return best_a, best_score
```

```python
"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. Soft-max gradient descent with temperature annealing (fast, many iterations)
2. LP-based fictitious play for exact best responses
3. Coordinate descent and peak-shaving for final polish
4. Multi-length exploration and diverse initialization
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_soft_max_grad(a, T):
    """Compute gradient of soft-max auto-conv objective.
    Returns gradient of (1/T)*log(sum_k exp(conv(a,a)[k]/T)).
    """
    n = len(a)
    b = fast_convolve(a)
    # Softmax weights with numerical stability
    b_shifted = b - b.max()
    w = np.exp(b_shifted / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    # Gradient: (2/T) * cross-correlation of w and a (valid part)
    # g[i] = sum_j a[j] * w[i+j]
    fft_size = 1 << (3 * n).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g_full = irfft(W * np.conj(A), fft_size)
    g = g_full[:n]
    return (2.0 / T) * g, b, w


def soft_max_descent(a0, time_limit, T_init=2e-3, T_min=1e-6, T_decay=0.9995,
                     lr_init=0.002, lr_decay=0.9999, reheat_every=500,
                     reheat_factor=5.0, verbose=False):
    """Soft-max gradient descent with temperature annealing and reheating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it >= reheat_every and T < best_score / (2.0 * n) * 0.1:
            # Reheat to escape local minima
            T = min(T * reheat_factor, T_init * 0.5)
            lr = lr_init * 0.5
            no_improve = 0
            it = 0

        grad, b, w = compute_soft_max_grad(a, T)
        a_new = simplex_project(a - lr * grad)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive step size
        if no_improve > 50:
            lr *= 0.9
            no_improve = 0
        elif no_improve > 200:
            # Try a small random perturbation
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.05 * best_a.max()
            perturbed = best_a + rng.standard_normal(n) * scale
            a = simplex_project(perturbed)
            lr = lr_init * 0.8
            no_improve = 0

        T = max(T * T_decay, T_min)
        lr = max(lr * lr_decay, 1e-8)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMGD it={it} ({elapsed:.0f}s): best={best_score:.6f} T={T:.2e} lr={lr:.2e}")

    return best_a, best_score


def topk_subgrad_descent(a0, time_limit, top_k=10, lr_init=0.001, verbose=False):
    """Subgradient descent using top-k averaged subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    momentum = 0.0
    mom_decay = 0.85

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.01 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            sg[j] += 2.0 * w[idx] * a[k - j]

        # Momentum
        momentum = mom_decay * momentum + sg
        norm_m = np.linalg.norm(momentum)
        if norm_m > 1e-15:
            momentum = momentum / norm_m

        a_new = simplex_project(a - lr * momentum)
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 100:
            lr *= 0.85
            no_improve = 0
        if no_improve > 400:
            rng = np.random.default_rng(it * 13 + 3)
            a = simplex_project(best_a + rng.standard_normal(n) * 0.02)
            momentum = 0.0
            lr = lr_init
            no_improve = 0

        lr = max(lr * 0.9998, 1e-8)

        if verbose and it % 100 == 0:
            print(f"    SubGrad it={it}: best={best_score:.6f}")

    return best_a, best_score


def coord_descent_polish(a0, time_limit, n_sweeps=500):
    """Coordinate descent: try adjusting each coordinate to reduce peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.7, 0.85, 0.93, 1.07, 1.15, 1.4]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss < 1e-15:
                    a[i] = old_val
                    a = a / a.sum()
                    continue
                a = a / ss
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()

        if not improved and sweep > 10:
            # Try finer adjustments
            any_fine = False
            for i in rng.choice(n, size=min(n, 200), replace=False):
                old_val = a[i]
                if old_val < 1e-14:
                    continue
                for factor in [0.97, 0.985, 1.015, 1.03]:
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss < 1e-15:
                        a[i] = old_val
                        continue
                    a = a / ss
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-15:
                        best_score = score
                        best_a = a.copy()
                        old_val = a[i]
                        improved = True
                        any_fine = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()
            if not any_fine and sweep > 30:
                break

    return best_a, best_score


def peak_shave(a0, time_limit):
    """Targeted peak reduction by adjusting contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(3000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        top_k = min(8, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]

        improved = False
        for pk in peaks:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_c = np.argsort(contribs)[::-1][:4]

            for ci in top_c:
                i = int(idx[ci])
                j = peak_k - i
                if a[i] < 1e-14:
                    continue
                for factor in [0.85, 0.75, 0.6]:
                    a_t = a.copy()
                    a_t[i] *= factor
                    a_t = np.maximum(a_t, 0)
                    a_t = a_t / a_t.sum()
                    score = 2.0 * n * fast_convolve(a_t).max()
                    if score < best_score - 1e-14:
                        a = a_t
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """LP: minimize M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CBC, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    solvers_to_try = []
    for s in ['GUROBI', 'MOSEK', 'PDLP', 'ECOS', 'SCS', 'GLPK', 'CBC', 'CVXOPT']:
        try:
            sol = getattr(cp, s, None)
            if sol is not None:
                solvers_to_try.append(sol)
        except Exception:
            pass
    for solver_name in solvers_to_try:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def fictitious_play(a0, time_limit, solver=None, verbose=False):
    """Alternating LP best responses."""
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            break
        a_new, _ = solve_lp_best_response(b_new, solver)
        if a_new is None:
            break

        for cand in [a_new, 0.5 * (a + a_new), b_new]:
            cand = np.maximum(cand, 0)
            ss = cand.sum()
            if ss < 1e-10:
                continue
            cand = cand / ss
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```
