import numpy as np

# === Core Functions ===

def fast_auto_conv(a):
    """Compute full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, n=fft_size)
    return irfft(A * A, n=fft_size)[:size]

def compute_conv_and_grad(a, w):
    """
    Compute auto-convolution and weighted subgradient.
    w has length 2n-1.
    Returns (b, grad) where b is auto-conv (len 2n-1), grad is subgradient (len n).
    """
    n = len(a)
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    # Need fft_size >= 3n-2 for the gradient (cross-correlation)
    grad_fft_size = 1 << (3 * n - 2).bit_length()
    
    # Convolution
    if grad_fft_size >= conv_size:
        A = rfft(a, n=grad_fft_size)
    else:
        A = rfft(a, n=fft_size)
    
    # Use the larger FFT for both
    use_size = max(fft_size, grad_fft_size)
    A = rfft(a, n=use_size)
    b_full = irfft(A * A, n=use_size)
    b = b_full[:conv_size]
    
    # Subgradient: grad[i] = sum_t w[t] * a[t-i] for valid indices
    W = rfft(w, n=use_size)
    grad = irfft(np.conj(A) * W, n=use_size)[:n]
    
    return b, grad

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.nonzero(cond)[0][-1] + 1
    theta = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - theta, 0.0)

def eval_max_b(a):
    """Max of auto-convolution for sum-normalized sequence."""
    s = a.sum()
    if s < 1e-15:
        return 1e10
    a_n = a / s
    b = fast_auto_conv(a_n)
    return float(b.max())

            break
