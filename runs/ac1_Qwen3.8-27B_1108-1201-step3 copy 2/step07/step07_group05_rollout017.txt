For the Hessian, since c[k] is quadratic in a, the Hessian of c[k] is just 2*H_k (the matrix I defined earlier). The Hessian of the smoothed objective is:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)

def adaptive_lbgfs(a0, time_limit, T_start=10000, T_end=50, maxiter=10000):
    """
    L-BFGS with adaptive temperature that decreases over iterations.
    """
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}
    
    # Store current T in a mutable container
    state = {'T': T_start, 'iter': 0}
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        state['iter'] += 1
        # Adaptive temperature: decrease over iterations
        progress = min(1.0, state['iter'] / (maxiter * 0.8))
        state['T'] = T_start * (T_end / T_start) ** progress
        T = state['T']
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        exp_shifted = np.exp(shifted - max_shifted)
        logsumexp = max_shifted + np.log(exp_shifted.sum())
        return (2.0 * n / T) * np.exp(T * logsumexp)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        T = state['T']
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        exp_shifted = np.exp(shifted - max_shifted)
        w = exp_shifted / exp_shifted.sum()
        
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        cross = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        grad_a = 4.0 * n * cross
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    result['seconds'] = time.monotonic() - t0
    return result


def peak_guided_cd(a0, time_limit, verbose=False):
    """
    Coordinate descent that focuses on coordinates contributing to the top peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        
        # Find top peaks
        top_k = min(5, len(conv))
        peak_amps = np.sort(conv)[::-1][:top_k]
        threshold = peak_amps[-1] * 0.99  # peaks within 1% of the top
        active_peaks = np.where(conv >= threshold)[0]
        
        # Find coordinates that contribute to active peaks
        active_coords = set()
        for pk in active_peaks:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for i in range(js, je + 1):
                if a[i] > 1e-10:
                    active_coords.add(i)
        
        # Also include some random coordinates for exploration
        active_coords_list = list(active_coords)
        for _ in range(min(20, n // 10)):
            active_coords_list.append(rng.integers(0, n))
        
        improved = False
        for i in active_coords_list:
            if a[i] < 1e-14:
                continue
            
            # Try several transfer amounts
            for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = a[i] * frac
                # Find where to put the transferred mass
                # Try a few random targets
                for _ in range(3):
                    j = rng.integers(0, n)
                    if j == i:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if time.monotonic() - t0 > time_limit:
                break
        
        if not improved and iters > 100:
            # Perturb
            idx = rng.integers(0, n, size=min(50, n//10))
            a = best_a.copy()
            for i in idx:
                a[i] *= (1 + 0.05 * rng.standard_normal())
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and iters % 100 == 0:
            print(f"  PGCD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def fine_local_search(a0, time_limit, verbose=False):
    """
    Very fine-grained local search: try tiny perturbations of individual coordinates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    
    # Find which coordinates matter (contribute to peaks)
    conv = fast_convolve(a)
    top_peaks = np.where(conv >= 0.99 * conv.max())[0]
    
    # Find relevant coordinates
    relevant = set()
    for pk in top_peaks:
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        relevant.update(range(js, je + 1))
    relevant = list(relevant)
    
    # Sort by magnitude (try larger coordinates first)
    relevant.sort(key=lambda i: -a[i])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in relevant:
            if a[i] < 1e-12:
                continue
            
            # Try very small adjustments
            for scale in [0.001, 0.005, 0.01, 0.02]:
                for sign in [1, -1]:
                    delta = sign * scale * a[i]
                    a_trial = a.copy()
                    a_trial[i] += delta
                    if a_trial[i] < 0:
                        continue
                    # Redistribute the change to keep sum = 1
                    diff = -delta  # the amount that needs to be redistributed
                    if abs(diff) < 1e-18:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask].copy()
                    os = others.sum()
                    if os < 1e-15:
                        continue
                    a_trial[mask] = others + diff * others / os
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if time.monotonic() - t0 > time_limit:
                break
        
        if not improved:
            # Try a different set of relevant coordinates
            conv = fast_convolve(best_a)
            top_peaks = np.where(conv >= 0.995 * conv.max())[0]
            relevant = set()
            for pk in top_peaks:
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                relevant.update(range(js, je + 1))
            relevant = list(relevant)
            rng.shuffle(relevant)
            # Add some random ones
            for _ in range(20):
                relevant.append(rng.integers(0, n))
            relevant.sort(key=lambda i: -best_a[i])
            a = best_a.copy()
            # Small perturbation
            for i in relevant[:20]:
                a[i] *= (1 + 0.01 * rng.standard_normal())
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def chirp_init(n, f0=3, f1=7):
    """Create a chirp-modulated non-negative initialization."""
    t = np.linspace(0, 1, n)
    freq = f0 + (f1 - f0) * t
    phase = 2 * np.pi * np.cumsum(freq) / n
    sig = 0.5 + 0.5 * np.cos(phase)  # Always in [0, 1]
    sig = np.maximum(sig, 1e-6)
    return sig / sig.sum()


def multilobe_init(n, nlobes=4):
    """Create a multi-lobe initialization."""
    t = np.linspace(0, 1, n)
    sig = np.ones(n) * 0.1
    for l in range(nlobes):
        center = (l + 0.5) / nlobes
        width = 0.3 / nlobes
        sig += np.exp(-0.5 * ((t - center) / width) ** 2)
    return sig / sig.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:

    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
