So: d(c[k])/da[t] = 2*a[k-t] if 0 <= k-t <= n-1, else 0.
(And when k-t = t, this gives 2*a[t], which is correct.)

Great, so the gradient is simple:
grad_k[t] = 2*a[k-t] if max(0, k-n+1) <= t <= min(n-1, k), else 0.

This is just a shifted (and possibly truncated) copy of a, scaled by 2!

So for the subgradient of max(c), at the maximizing set S:
subgrad = (1/|S|) * sum_{k in S} 2 * shifted_a(k)

where shifted_a(k)[t] = a[k-t] (valid range).

This is very efficient to compute!

"""
Multi-phase optimization for minimizing 2*n*max(auto_conv)/sum^2.
Uses: exact subgradient descent, peak-targeted coordinate descent,
L-BFGS with log-sum-exp smoothing at multiple temperatures,
length exploration, and peak balancing heuristics.
Key innovations over previous approach:
- Exploits the simple structure of auto-convolution gradients
- Peak-targeted moves instead of random pairs
- Wider length exploration (up to 4096)
- Asymmetric initializations
- Hybrid subgradient + L-BFGS with adaptive switching

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def autoconv_subgradient(k, a):
    """
    Gradient of conv[k] w.r.t. a for auto-convolution.
    d(conv[k])/da[t] = 2*a[k-t] if 0<=k-t<=n-1, else 0.
    """
    n = len(a)
    t_min = max(0, k - n + 1)
    t_max = min(n - 1, k)
    g = np.zeros(n)
    if t_min <= t_max:
        j_indices = k - np.arange(t_min, t_max + 1)
        g[t_min:t_max + 1] = 2.0 * a[j_indices]
    return g


def smoothed_obj_and_grad(z, T, n):
    """Compute log-sum-exp smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    log_sum_exp = max_shifted + np.log(w.sum())
    obj = (2.0 * n / T) * np.exp(T * log_sum_exp)
    # Gradient: (2n/T) * sum_k w_k * grad(conv[k])
    # Each grad(conv[k]) is a shifted copy of a
    grad_a = np.zeros(n)
    for k_idx in range(len(conv)):
        if w[k_idx] < 1e-15:
            continue
        grad_k = autoconv_subgradient(int(k_idx), a)
        grad_a += w[k_idx] * grad_k
    # Project to log-space (variance-preserving)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with log-sum-exp smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = smoothed_obj_and_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = smoothed_obj_and_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass

    result['seconds'] = time.monotonic() - t0
    return result


def subgradient_descent(a0, time_limit, eps_rel=1e-5, alpha_init=0.05, decay=0.995, verbose=False):
    """Exact subgradient descent targeting the active peak set."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    alpha = alpha_init
    iters = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        tol = max(eps_rel * max_val, 1e-14)
        active = np.where(conv >= max_val - tol)[0]

        # Compute subgradient (average gradient over active set)
        sg = np.zeros(n)
        for k_idx in active:
            sg += autoconv_subgradient(int(k_idx), a)
        sg /= len(active)

        # Remove mean to stay on sum=1 hyperplane
        sg -= np.mean(sg)

        # Step
        a_new = a - alpha * sg
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            alpha *= 0.5
            continue
        a_new = a_new / s_new

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        a = a_new
        iters += 1

        # Adaptive step size
        if iters % 50 == 0:
            alpha *= decay
            if no_improve_count > 200:
                alpha = alpha_init * 0.5
                # Perturb
                rng = np.random.default_rng(iters)
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve_count = 0

        if verbose and iters % 500 == 0:
            print(f"    SGD iter={iters}, alpha={alpha:.6f}, best={best_score:.8f}")

    return best_a, best_score

            indices = np.arange(t_min, t_max + 1)

            contributions = a[indices] * a[partners]

                j = int(partners[ci])

                        contrib_pos2 = ak - contrib_pos

                    others = a_trial[mask]
                    os = others.sum()
                    if os < 1e-15:
                        continue
                    a_trial[mask] = others + delta * others / os
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()

                    break
            if improved:
                break

        if not improved and iters > 20:

            for _ in range(5):

                j = rng.integers(0, n)
                if i == j:
                    continue
                t_move = rng.standard_normal() * 0.02 * max(a[i], a[j], 1e-6)
                a_trial = a.copy()
                a_trial[i] -= t_move
                a_trial[j] += t_move
                a_trial = np.maximum(a_trial, 0)

                if ss < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_asymmetric_init(n, rng):
    """Generate various asymmetric initializations."""
    inits = []
    # Linear ramp
    a = np.linspace(0.01, 1.0, n)
    inits.append(a / a.sum())
    # Quadratic ramp
    x = np.linspace(0, 1, n)
    a = x ** 2 + 0.01
    inits.append(a / a.sum())
    # Exponential decay
    a = np.exp(-x * 3) + 0.01
    inits.append(a / a.sum())
    # Bump in the middle
    a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.01
    inits.append(a / a.sum())
    # Two bumps
    a = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02) + 0.01
    inits.append(a / a.sum())
    # Random smooth
    a = np.ones(n) + 0.3 * np.sin(2 * np.pi * rng.random() * 3 * x) + 0.2 * np.cos(2 * np.pi * rng.random() * 5 * x)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    # Random
    a = rng.random(n) + 0.01
    inits.append(a / a.sum())
    return inits

def nesterov_logspace(a0, time_limit, T=500, lr_init=0.03, momentum=0.92):
    """Nesterov accelerated gradient in log-space."""
    n = len(a0)
    z = z_from_a(a0)
    v = np.zeros(n)
    best_z = z.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    lr = lr_init
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        z_la = z - momentum * v
        a_la = softmax_from_z(z_la)
        _, grad = smoothed_obj_and_grad(z_la, T, n)
        v = momentum * v + lr * grad
        z_new = z - v
        z_new = np.clip(z_new, -20, 20)
        a_new = softmax_from_z(z_new)
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_z = z_new.copy()
        z = z_new
        lr *= 0.9997

    a_out = softmax_from_z(best_z)
    return a_out, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Length exploration with quick L-BFGS
    Phase 2: Focused optimization (subgradient + L-BFGS + peak targeting)
    Phase 3: Cross-length refinement
    Phase 4: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi * 2, n_init)
        parent = parent + 0.1 * np.cos(t)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Length exploration (300s) ===
    print(f"\n=== Phase 1: Length Exploration (best={best_score:.8f}) ===")
    p1_deadline = time.monotonic() + min(300.0, (deadline - time.monotonic()) * 0.2)

    # Lengths to try
    lengths_to_try = [n_parent]
    for alt_n in [512, 640, 896, 1024, 1280, 1536, 2048, 3072, 4096]:
        if alt_n != n_parent and 128 <= alt_n <= 4096:
            lengths_to_try.append(alt_n)
    # Add a few random lengths near parent
    for _ in range(3):
        alt_n = int(n_parent * (1 + rng.standard_normal() * 0.1))
        if 128 <= alt_n <= 4096 and alt_n not in lengths_to_try:
            lengths_to_try.append(alt_n)

    lengths_to_try.sort(key=lambda x: abs(x - n_parent))

    for alt_n in lengths_to_try:
        if time.monotonic() > p1_deadline:
            break
        rem = p1_deadline - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.4, 45.0)
        # Initialize: interpolate from best
        init_seq = interpolate_sequence(best_seq, alt_n)
        # Also try an asymmetric init
        inits = [init_seq]
        if alt_n <= 2048:
            asym_inits = generate_asymmetric_init(alt_n, rng)
            inits.extend(asym_inits[:2])

        for idx, init in enumerate(inits):
            if time.monotonic() > p1_deadline:
                break
            rem2 = p1_deadline - time.monotonic()
            if rem2 < 10:
                break
            t_alloc2 = min(rem2 * 0.6, t_alloc / len(inits), 30.0)
            label = f"len{alt_n}_{'interp' if idx==0 else 'asym'}"
            res = lbgfs_optimize(init, t_alloc2, T=500, maxiter=2000)
            best_seq, best_score = check_and_update(res['a'], label, best_seq, best_score, verbose=True)

    print(f"  After Phase 1: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 2: Focused optimization at best length (500s) ===
    print(f"\n=== Phase 2: Focused Optimization (best={best_score:.8f}) ===")
    p2_deadline = time.monotonic() + min(500.0, (deadline - time.monotonic()) * 0.3)
    n_best = len(best_seq)

    # 2a: L-BFGS with multiple temperatures
    for T_val in [200, 500, 1000, 3000]:
        if time.monotonic() > p2_deadline:
            break
        rem = p2_deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.25, 80.0)
        print(f"  L-BFGS T={T_val}, n={n_best}, budget={t_alloc:.0f}s")
        res = lbgfs_optimize(best_seq, t_alloc, T=T_val, maxiter=5000)
        best_seq, best_score = check_and_update(res['a'], f"lbgfs_T{T_val}", best_seq, best_score)

    # 2b: Subgradient descent
    if time.monotonic() < p2_deadline - 30:
        rem = p2_deadline - time.monotonic() - 20
        t_alloc = min(rem * 0.25, 60.0)
        print(f"  Subgradient, n={n_best}, budget={t_alloc:.0f}s")
        a_sg, sc_sg = subgradient_descent(best_seq, t_alloc, eps_rel=1e-4, alpha_init=0.03)
        best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # 2c: Peak-targeted coordinate descent
    if time.monotonic() < p2_deadline - 20:
        rem = p2_deadline - time.monotonic() - 10
        t_alloc = min(rem * 0.3, 80.0)
        print(f"  Peak-targeted CD, n={n_best}, budget={t_alloc:.0f}s")
        a_pc, sc_pc = peak_targeted_cd(best_seq, t_alloc)
        best_seq, best_score = check_and_update(a_pc, "peak_cd", best_seq, best_score)

    # 2d: Peak balancing
    if time.monotonic() < p2_deadline - 15:
        rem = p2_deadline - time.monotonic() - 5
        t_alloc = min(rem * 0.4, 60.0)
        print(f"  Peak balancing, n={n_best}, budget={t_alloc:.0f}s")
        a_pb, sc_pb = peak_balancing(best_seq, t_alloc, n_peaks=12)
        best_seq, best_score = check_and_update(a_pb, "peak_bal", best_seq, best_score)

    print(f"  After Phase 2: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 3: Cross-length refinement (250s) ===
    print(f"\n=== Phase 3: Cross-Length Refinement (best={best_score:.8f}) ===")
    p3_deadline = time.monotonic() + min(250.0, (deadline - time.monotonic()) * 0.2)
    n_best = len(best_seq)

    # Try lengths close to the current best
    nearby = []
    for delta in [-2, -1, 1, 2, 4, 8, -4, -8, 16, -16, 32, -32]:
        alt = n_best + delta
        if 128 <= alt <= 4096 and alt not in nearby:
            nearby.append(alt)

    for alt_n in nearby[:6]:
        if time.monotonic() > p3_deadline:
            break
        rem = p3_deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.3, 40.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Refine n={alt_n}, budget={t_alloc:.0f}s")
        res = lbgfs_optimize(init_alt, t_alloc, T=1000, maxiter=3000)
        best_seq, best_score = check_and_update(res['a'], f"cross_n{alt_n}", best_seq, best_score)

    # If we found a better length, do more optimization at that length
    if len(best_seq) != n_best:
        print(f"  Length changed to {len(best_seq)}, doing more optimization...")
        if time.monotonic() < p3_deadline:
            rem = p3_deadline - time.monotonic()
            t_alloc = min(rem * 0.5, 50.0)
            res = lbgfs_optimize(best_seq, t_alloc, T=500, maxiter=5000)
            best_seq, best_score = check_and_update(res['a'], "cross_refine", best_seq, best_score)

    print(f"  After Phase 3: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 4: Final polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.8f}) ===")
    p4_deadline = deadline
    n_best = len(best_seq)

    # 4a: High-temperature L-BFGS (close to exact)
    for T_val in [5000, 20000, 100000]:
        rem = p4_deadline - time.monotonic() - 30
        if rem < 10:
            break
        t_alloc = min(rem * 0.3, 40.0)
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        res = lbgfs_optimize(best_seq, t_alloc, T=T_val, maxiter=5000)
        best_seq, best_score = check_and_update(res['a'], f"final_T{T_val}", best_seq, best_score)

    # 4b: Nesterov
    rem = p4_deadline - time.monotonic() - 20
    if rem > 15:
        t_alloc = min(rem * 0.25, 40.0)
        print(f"  Nesterov, budget={t_alloc:.0f}s")
        a_ng, sc_ng = nesterov_logspace(best_seq, t_alloc, T=1000, lr_init=0.02, momentum=0.93)
        best_seq, best_score = check_and_update(a_ng, "nesterov", best_seq, best_score)

    # 4c: Subgradient with finest tuning
    rem = p4_deadline - time.monotonic() - 15
    if rem > 10:
        t_alloc = min(rem * 0.3, 40.0)
        print(f"  Fine subgradient, budget={t_alloc:.0f}s")
        a_sg2, sc_sg2 = subgradient_descent(best_seq, t_alloc, eps_rel=1e-6, alpha_init=0.01, decay=0.999)
        best_seq, best_score = check_and_update(a_sg2, "fine_sg", best_seq, best_score)

    # 4d: Peak-targeted CD
    rem = p4_deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem * 0.35, 35.0)
        print(f"  Final peak CD, budget={t_alloc:.0f}s")
        a_pc2, sc_pc2 = peak_targeted_cd(best_seq, t_alloc)
        best_seq, best_score = check_and_update(a_pc2, "final_pcd", best_seq, best_score)

    # 4e: Peak balancing final
    rem = p4_deadline - time.monotonic() - 5
    if rem > 8:
        t_alloc = min(rem * 0.4, 25.0)
        a_pb2, sc_pb2 = peak_balancing(best_seq, t_alloc, n_peaks=20)
        best_seq, best_score = check_and_update(a_pb2, "final_pb", best_seq, best_score)

    # 4f: One more L-BFGS
    rem = p4_deadline - time.monotonic() - 2
    if rem > 5:
        t_alloc = min(rem * 0.8, 20.0)
        res = lbgfs_optimize(best_seq, t_alloc, T=50000, maxiter=3000)
        best_seq, best_score = check_and_update(res['a'], "final_lbgfs", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
