"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Sequential Convex Approximation (SCA)
with LP subproblems, combined with peak-shaving coordinate descent and random restarts.

Key innovation: Uses SCA framework where each iteration solves a linear program
that linearizes the quadratic convolution constraints around the current point.
This provides much stronger descent directions than gradient-based methods,
especially near the uniform distribution where gradients are nearly flat.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def sca_iteration(a, tol=1e-12):
    """
    One SCA iteration: solve LP to minimize t subject to
    linearized convolution constraints.
    Returns the new point.
    """
    n = len(a)
    b0 = fast_convolve(a)  # Current convolution
    # Linearized constraint: 2 * conv(a, a_new)[k] - b0[k] <= t
    # conv(a, a_new)[k] = sum_i a[i] * a_new[k-i]
    # This is linear in a_new!
    
    # The LP is:
    # minimize t
    # subject to: for each k in 0..2n-2:
    #   sum_i a[i] * a_new[max(0, k-i)..] ... 
    #   
    # Actually, conv(a, a_new)[k] = sum_{i: valid} a[i] * a_new[k-i]
    # where valid means 0 <= i < n and 0 <= k-i < n.
    # So conv(a, a_new)[k] = sum_{j: valid} a_new[j] * a[k-j]
    # (by re-indexing with j = k-i)
    
    # Constraint: 2 * sum_{j} a[k-j] * a_new[j] - b0[k] <= t
    # i.e., sum_j (2*a[k-j]) * a_new[j] - t <= b0[k]
    
    # Let's set up the LP matrix.
    # Variables: a_new[0..n-1], t
    # Constraints: for each k: sum_j (2*a[k-j]) * a_new[j] - t <= b0[k]
    #              sum_j a_new[j] = 1
    #              a_new[j] >= 0
    
    # For efficiency, let's use scipy.optimize.linprog or cvxpy.
    # For n=768, the constraint matrix is (2n-1) x (n+1) = 1535 x 769.
    # This is manageable.
    
    # Let's use scipy's linprog for speed.
    from scipy.optimize import linprog
    
    # c = [0, 0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: (2n-1) x (n+1)
    # Row k: [2*a[k-0], 2*a[k-1], ..., 2*a[k-(n-1)], -1] (with a[j]=0 for j<0 or j>=n)
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = b0.copy()
    
    for k in range(2*n-1):
        for j in range(n):
            # a[k-j] is the coefficient for a_new[j]
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = 2.0 * a[idx]
        A_ub[k, n] = -1.0  # coefficient of t
    
    # Equality: sum a_new[j] = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a_new[j] >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    else:
        return a.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Sequential Convex Approximation (SCA)
with LP subproblems, top-K peak-guided multiplicative weights, and coordinate descent.
Key innovation: SCA linearizes the non-convex quadratic convolution constraints
around the current point, producing strong descent directions via LP solves
that outperform pure gradient methods especially near flat regions of the landscape.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_subgradient_topk(a, b, K=5):
    """Compute subgradient using top-K peaks of the convolution."""
    n = len(a)
    # Find top K peaks
    sorted_idx = np.argsort(b)[::-1]
    top_k = sorted_idx[:min(K, len(b))]
    max_val = b[top_k[0]]
    
    # Weight peaks by proximity to max (exponential decay)
    weights = np.exp(-(b[top_k] - max_val) * n * 0.1)
    weights /= weights.sum()
    
    # Compute subgradient: g[i] = sum_k w_k * 2*a[k-i] (if valid)
    g = np.zeros(n)
    for k_idx in range(len(top_k)):
        k = int(top_k[k_idx])
        w = weights[k_idx]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += 2.0 * w * a[k - j]
    
    return g


def sca_lp_solve(a, K=20):
    """
    Solve SCA LP subproblem: minimize t subject to
    2*conv(a, a_new)[k] - b0[k] <= t for top-K peaks,
    sum(a_new) = 1, a_new >= 0.
    Returns candidate a_new or None if failed.
    """
    n = len(a)
    b0 = fast_convolve(a)
    
    # Get top K peak indices
    sorted_idx = np.argsort(b0)[::-1]
    top_k = sorted_idx[:K]
    
    # Build constraint matrix: K x (n+1)
    # Row k: [2*a[k-0], 2*a[k-1], ..., 2*a[k-(n-1)], -1] <= b0[k]
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for row, k in enumerate(top_k):
        b_ub[row] = b0[int(k)]
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, le := je + 1):
            A_ub[row, j] = 2.0 * a[k - j]
        A_ub[row, n] = -1.0
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Equality: sum a_new = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            a_new = res.x[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-12:
                a_new /= s
            return a_new
    except Exception:
        pass
    
    return None


def sca_optimize(a_init, time_limit, label="SCA"):
    """
    SCA optimization loop with backtracking line search.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 50
    K_start = 10
    K_end = 50
    
    print(f"    {label}: n={n}, initial score={best_score:.8f}")
    
    it = 0
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break
        
        # Adaptive K: increase over iterations
        progress = min(1.0, it / 200.0)
        K = int(K_start + (K_end - K_start) * progress)
        
        # Solve LP
        a_candidate = sca_lp_solve(a, K=K)
        if a_candidate is None:
            no_improve += 5
            continue
        
        # Backtracking line search
        current_score = score_from_a(a)
        improvement = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_trial = (1.0 - alpha) * a + alpha * a_candidate
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                continue
            a_trial /= s
            sc = score_from_a(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a_trial.copy()
                no_improve = 0
                improvement = True
                break
            elif sc < current_score:
                a = a_trial
                improvement = True
                break
        
        if not improvement:
            no_improve += 1
            # Try a small random perturbation to escape
            if no_improve > 10 and no_improve % 15 == 0:
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.standard_normal(n) * 0.01
                a = best_a + noise
                a = np.maximum(a, 1e-10)
                a /= a.sum()
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): K={K} "
                  f"current={score_from_a(a):.8f} best={best_score:.8f}")
    
    elapsed = time.monotonic() - t0
    print(f"    {label} done: iter={it} ({elapsed:.0f}s) best={best_score:.8f}")
    return best_a, best_score


def multiplicative_weights_topk(a_init, time_limit, K=5, eta_start=0.05, label="MW"):
    """
    Multiplicative weights using top-K peak subgradient with geometric decay.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = eta_start
    no_improve = 0
    
    print(f"    {label}: n={n}, initial score={best_score:.8f}")
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient
        g = compute_subgradient_topk(a, b, K=K)
        
        # Project to simplex tangent space
        g = g - np.dot(a, g)
        
        # Multiplicative update
        update = -eta * g
        update = np.clip(update, -3.0, 3.0)
        a = a * np.exp(update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            continue
        a /= s
        
        # Decay eta
        eta *= 0.9995
        if no_improve > 200:
            eta *= 0.5
            if eta < 1e-6:
                # Random restart from best
                rng = np.random.default_rng(it * 13)
                noise = rng.standard_normal(n) * 0.02
                a = best_a + noise
                a = np.maximum(a, 1e-10)
                a /= a.sum()
                no_improve = 0
                eta = eta_start * 0.3
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): eta={eta:.6f} "
                  f"score={current_score:.8f} best={best_score:.8f}")
    
    elapsed = time.monotonic() - t0
    print(f"    {label} done: iter={it} ({elapsed:.0f}s) best={best_score:.8f}")
    return best_a, best_score


def coordinate_descent_polish(a_init, time_limit, label="CD"):
    """
    Coordinate descent: try multiplicative changes to individual coordinates,
    focusing on those contributing to top peaks.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 30
    
    factors = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]
    
    print(f"    {label}: n={n}, initial score={best_score:.8f}")
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve_pass = 0
        b = fast_convolve(a)
        
        # Identify coordinates contributing to top peaks
        top_peak = int(np.argmax(b))
        k = top_peak
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        contribs = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
        top_coords = np.argsort(contribs)[::-1][:min(20, len(contribs))]
        top_coords = top_coords + js  # Convert to absolute indices
        
        # Try modifying top-contributing coordinates first, then others
        order = np.concatenate([top_coords, np.setdiff1d(np.arange(n), top_coords)])
        np.random.shuffle(order[len(top_coords):])
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if no_improve_pass > 50:
                break
            
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < 0 or new_val > 1.0:
                    continue
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask_nonzero = mask & (a_trial > 1e-15)
                others_sum = a_trial[mask_nonzero].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask_nonzero] -= diff * a_trial[mask_nonzero] / others_sum
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-12:
                    continue
                a_trial /= s
                
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a_trial.copy()
                    no_improve = 0
                    no_improve_pass = 0
                    break
        
        else:
            no_improve += 1
        
        if time.monotonic() - t0 > time_limit:
            break
    
    elapsed = time.monotonic() - t0
    print(f"    {label} done: ({elapsed:.0f}s) best={best_score:.8f}")
    return best_a, best_score


def pair_shave(a_init, time_limit, label="PS"):
    """
    Targeted pair shaving: reduce both elements of the top-contributing pair
    to the peak, redistributing to other positions.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    print(f"    {label}: n={n}, initial score={best_score:.8f}")
    
    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        if b[peak_k] * 2.0 * n < best_score * (1 - 1e-10):
            break
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]
        
        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if i == j or a[j] < 1e-14 or a[i] < 1e-14:
                continue
            for f1 in [0.85, 0.9, 0.95]:
                for f2 in [0.85, 0.9, 0.95]:
                    a_trial = a.copy()
                    old_i = a_trial[i]
                    old_j = a_trial[j]
                    a_trial[i] = old_i * f1
                    a_trial[j] = old_j * f2
                    diff = (old_i - a_trial[i]) + (old_j - a_trial[j])
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    mask |= (a_trial < 1e-15)
                    mask[i] = True
                    mask[j] = True
                    mask_nonzero = mask & (a_trial > 1e-15)
                    if mask_nonzero.sum() < 2:
                        continue
                    others_sum = a_trial[mask_nonzero].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask_nonzero] += diff * a_trial[mask_nonzero] / others_sum
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-12:
                        continue
                    a_trial /= s
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try single-coordinate reduction
            for ci in top_idx[:5]:
                i = int(idx[ci])
                if a[i] < 1e-14:
                    continue
                for f in [0.8, 0.9, 0.95]:
                    a_trial = a.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * f
                    diff = old_val - a_trial[i]
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask_nonzero = mask & (a_trial > 1e-15)
                    others_sum = a_trial[mask_nonzero].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask_nonzero] += diff * a_trial[mask_nonzero] / others_sum
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial /= a_trial.sum()
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved:
                break
    
    elapsed = time.monotonic() - t0
    print(f"    {label} done: ({elapsed:.0f}s) best={best_score:.8f}")
    return best_a, best_score
