def multiplicative_step(f, eta, momentum_buf, momentum_alpha, n_top):
    """
    Multiplicative gradient descent step with momentum.
    
    f: current sequence (sum to 1, non-negative)
    eta: learning rate
    momentum_buf: momentum buffer
    momentum_alpha: momentum coefficient
    n_top: number of top peaks to consider
    """
    n = len(f)
    b = fast_convolve(f)
    
    # Get top peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))
    
    # Compute subgradient contribution from top peaks
    # pressure[i] = sum_{k in top} w_k * (f[k-i] + f[i]) for valid indices
    # where w_k = b[k] (weight by peak height)
    pressure = np.zeros(n)
    
    for k in top_idx:
        w = b[k]  # weight by peak height
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For i in [lo, hi]: contribution is f[k-i] + f[i]
        # But f[k-i] for i in [lo, hi] is f[hi:lo-1:-1] (reversed f[lo:hi+1])
        idx = np.arange(lo, hi + 1)
        pressure[idx] += w * (f[k - idx] + f[idx])
    
    # Normalize pressure
    p_max = pressure.max()
    if p_max > 0:
        pressure /= p_max
    
    # Momentum
    momentum_buf = momentum_alpha * momentum_buf + (1 - momentum_alpha) * pressure
    
    # Multiplicative update
    f_new = f * np.exp(-eta * momentum_buf)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    
    return f_new, momentum_buf

"""
Optimized sequence search combining multiplicative gradient descent,
LP-based direction finding, and coordinate descent for minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. Multiplicative (mirror) gradient descent with momentum - naturally handles
   non-negativity and often converges faster for this type of problem.
2. Soft-peak weighting for smoother gradients.
3. Adaptive learning rate with backtracking.
4. Combined LP + multiplicative search.
5. Better multi-length exploration with strategic interpolation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_pressure(f, n_top, temperature=0.0):
    """
    Compute the pressure vector from top peaks.
    pressure[i] = sum of weighted contributions to peaks from f[i].
    
    If temperature > 0, uses soft weighting (softmax of peak heights).
    """
    n = len(f)
    b = fast_convolve(f)
    
    if temperature > 0:
        # Soft weighting
        scaled = b / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        # Use all peaks (or top ones for efficiency)
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_masked = np.zeros_like(w)
            w_masked[top_idx] = w[top_idx]
            w_masked /= w_masked.sum()
            w = w_masked
    else:
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        w = np.zeros(len(b))
        # Weight by peak height
        w[top_idx] = b[top_idx]
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
    
    pressure = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        pressure[idx] += w[k] * (f[k - idx] + f[idx])
    
    return pressure, b.max()

"""
Optimized sequence search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Combines multiplicative (mirror) gradient descent with momentum, LP-based
direction finding, adaptive peak weighting, coordinate descent, and
multi-length exploration. Key innovation over previous LP-only approach:
multiplicative updates naturally handle non-negativity and provide
smoother convergence for this peak-minimization problem.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_pressure(f, n_top, temperature=0.0):
    """
    Compute pressure vector from peaks.
    pressure[i] measures how much f[i] contributes to the peaks.
    Returns (pressure, max_peak).
    """
    n = len(f)
    b = fast_convolve(f)
    max_peak = b.max()

    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))

    if temperature > 0:
        # Soft weighting via softmax
        scaled = b[top_idx] / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= w_sum
    else:
        # Hard weighting: proportional to peak height
        w = b[top_idx].copy()
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.ones(len(top_idx)) / len(top_idx)

    pressure = np.zeros(n)
    for i, k in enumerate(top_idx):
        if w[i] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        pressure[idx] += w[i] * (f[k - idx] + f[idx])

    return pressure, max_peak


def multiplicative_gd(f_init, time_limit, eta_init=0.5, n_top=80,
                      momentum_alpha=0.7, temperature=0.0, label="mgd"):
    """
    Multiplicative gradient descent with momentum and adaptive learning rate.
    Uses: f_new = f * exp(-eta * momentum_buf), normalized.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    momentum_buf = np.zeros(n)
    eta = eta_init
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(9999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        pressure, cur_peak = compute_pressure(f, n_top, temperature)

        # Normalize pressure for stable updates
        p_max = pressure.max()
        if p_max > 1e-15:
            pressure /= p_max
        else:
            break

        # Update momentum
        momentum_buf = momentum_alpha * momentum_buf + (1.0 - momentum_alpha) * pressure

        # Multiplicative update
        f_new = f * np.exp(-eta * momentum_buf)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new /= s

        # Evaluate
        new_peak = fast_convolve(f_new).max()

        if new_peak < best_peak - 1e-14:
            best_peak = new_peak
            best_f = f_new.copy()
            f = f_new.copy()
            no_improve = 0
            # Slightly increase eta on improvement (with cap)
            eta = min(eta * 1.02, 2.0)
        else:
            f = f_new.copy()  # Still move, just track best
            no_improve += 1
            if no_improve > 30:
                eta *= 0.85
                no_improve = 0
            if no_improve > 200:
                # Perturb from best
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-15:
                    f /= s
                momentum_buf = np.zeros(n)
                no_improve = 0
                eta = eta_init * 0.5

        # Gradually increase temperature for diversity
        if it % 200 == 0:
            temperature = min(temperature + 0.0002, 0.05)
            # Gradually increase n_top
            n_top = min(n_top + 5, 3 * n // 2)

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} best_peak={best_peak:.10f} "
                  f"score={2.0*n*best_peak:.6f} eta={eta:.3f} t={elapsed:.0f}s")

    return best_f, best_peak


def lp_solve_direction(f, top_k):
    """
    Solve LP: maximize sum(g) s.t. g>=0, (f*g)[k] <= M for top-k peaks.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
    except Exception:
        return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def lp_phase(f_init, time_limit, top_k_init=40, top_k_max=200, label="lp"):
    """LP-based optimization with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 60 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)

        g, M = lp_solve_direction(f, top_k)
        if g is None:
            g, M = lp_solve_direction(f, min(2 * top_k + 20, 2 * n - 1))
            if g is None:
                break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g
        best_t = 0.05
        best_t_peak = fast_convolve(f).max()

        for t in [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_peak:
                best_t_peak = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_peak = fast_convolve(f).max()
        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 60:
                break

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} peak={best_peak:.10f} "
                  f"score={2.0*n*best_peak:.6f} tk={top_k} t={elapsed:.0f}s")

    return best_f, best_peak


def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements with golden ratio search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_peak = fast_convolve(f).max()
    multipliers = [0.6, 0.75, 0.85, 0.9, 1.1, 1.15, 1.25, 1.5, 2.0]

    improved_any = True
    passes = 0
    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_peak = fast_convolve(f).max()
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_peak - 1e-13:
                    best_j_peak = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_peak < best_peak - 1e-13:
                    best_peak = best_j_peak
                    improved_any = True
        if passes % 5 == 0:
            print(f"    cd pass={passes} peak={best_peak:.10f} "
                  f"score={2.0*n*best_peak:.6f} t={time.monotonic()-t0:.0f}s")

    return f, best_peak
