import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def cross_corr_fft(w, a):
    """Compute corr[j] = sum_k w[k]*a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    # w has length 2n-1, a has length n
    # We need conv of w (len 2n-1) and a (len n): result has len 3n-2
    N = 1 << (3 * n).bit_length()
    Wf = rfft(w, N)
    Af = rfft(a, N)
    # corr[j] = sum_k w[k]*a[k-j]
    # Using: ifft(conj(fft(w)) * fft(a))[j] gives circular cross-correlation
    # For linear: need N >= len(w) + len(a) - 1 = 3n-2
    circ = irfft(np.conj(Wf) * Af, N)
    return circ[:n]

def compute_smoothed_score_and_grad(x, t):
    """
    Compute smoothed score and its gradient w.r.t. x (log-parameters).
    a = softmax(x), score = 2n * logsumexp(t * conv(a,a)) / t
    """
    n = len(x)
    a = scipy_special_softmax(x)
    
    conv_a = fast_convolve(a)  # length 2n-1
    
    # Smoothed max: (1/t) * log(sum(exp(t * conv_a)))
    # Use log-sum-exp for numerical stability
    max_conv = conv_a.max()
    t_conv = t * (conv_a - max_conv)
    log_sum = max_conv + (1.0 / t) * np.log(np.sum(np.exp(t_conv)))
    
    score = 2.0 * n * log_sum
    
    # Weights: w[k] = exp(t * conv_a[k]) / sum(exp(t * conv_a))
    w = np.exp(t_conv)
    w /= w.sum()
    
    # grad_a[j] = 2n * 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # = 4n * crosscorr(w, a)[j]
    corr = cross_corr_fft(w, a)
    grad_a = 4.0 * n * corr
    
    # grad_x = a * (grad_a - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return score, grad_x

def scipy_special_softmax(x):
    m = x.max()
    e = np.exp(x - m)
    return e / e.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax-parameterized L-BFGS with progressive temperature annealing (main novel contribution)
2. Analytic gradient computation via FFT-based cross-correlation
3. LP-based fictitious play for initialization and alternative refinement
4. Coordinate descent and peak-targeted shaving for final polishing
5. Multi-length exploration with interpolation

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_func(x):
    """Numerically stable softmax."""
    m = x.max()
    e = np.exp(x - m)
    return e / e.sum()

def cross_corr_fft(w, a):
    """Compute corr[j] = sum_k w[k]*a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    N = 1 << (3 * n).bit_length()
    Wf = rfft(w, N)
    Af = rfft(a, N)
    circ = irfft(np.conj(Wf) * Af, N)
    return circ[:n]

def compute_smoothed_score_and_grad(x, t):
    """
    Compute smoothed score 2n*(1/t)*log(sum(exp(t*conv(a,a)))) and gradient w.r.t. x.
    a = softmax(x).
    """
    n = len(x)
    a = softmax_func(x)
    conv_a = fast_convolve(a)
    
    max_conv = conv_a.max()
    t_conv = t * (conv_a - max_conv)
    log_sum = max_conv + (1.0 / t) * np.log(np.sum(np.exp(t_conv)))
    score = 2.0 * n * log_sum
    
    w = np.exp(t_conv)
    w /= w.sum()
    
    corr = cross_corr_fft(w, a)
    dot_ac = np.dot(corr, a)
    grad_x = 4.0 * n * a * (corr - dot_ac)
    
    return score, grad_x

    x0 = np.log(a0_safe)
    
    best_a = a0_safe.copy()
    best_score = 2.0 * n * fast_convolve(a0_safe).max()
    t_start = time.monotonic()
    
    t_schedule = [0.08, 0.05, 0.03, 0.02, 0.012, 0.007, 0.004, 0.0025, 0.0015, 0.001, 0.0006, 0.0004, 0.0002]
    
    for t in t_schedule:
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break
        time_rem = time_limit - elapsed
        if time_rem < 10:
            break
        time_this = min(time_rem * 0.7, 120.0)
        
        def fun(x):
            s, _ = compute_smoothed_score_and_grad(x, t)
            return s
        
        def grad(x):
            _, g = compute_smoothed_score_and_grad(x, t)
            return g
        
        try:
            res = scipy_minimize(fun, x0, method='L-BFGS-B', jac=grad,
                               options={'maxiter': 300, 'maxfun': 2000,
                                       'ftol': 1e-12, 'gtol': 1e-10,
                                       'maxls': 50, 'iprint': -1})
            
            if res.x is not None:
                a_new = softmax_func(res.x)
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    L-BFGS t={t:.5f}: {best_score:.6f} (iter={res.nit})")
                x0 = res.x.copy()
            else:
                break
        except Exception as e:
            if verbose:
                print(f"    L-BFGS t={t:.5f} error: {e}")
            break
        
        a_cur = softmax_func(x0)
        cur_score = 2.0 * n * fast_convolve(a_cur).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()
    
    return best_a, best_score

def lbfgs_multi_restart(a0, time_limit, n_restarts=5, verbose=True):
    """L-BFGS with multiple restarts from perturbed points."""
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(999)
    
    time_per_restart = time_limit / (n_restarts + 1)
    
    # First run: direct
    t_alloc = min(time_per_restart * 1.5, time_limit * 0.3)
    if t_alloc > 10:
        best_a, best_score = lbfgs_anneal(a0, t_alloc, verbose=verbose)
    
    for r in range(1, n_restarts + 1):
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break
        time_rem = time_limit - elapsed
        if time_rem < 15:
            break
        t_alloc = min(time_per_restart, time_rem * 0.8)
        if t_alloc < 10:
            break
        
        # Perturb the current best
        sigma = 0.05 / np.sqrt(r)
        a_pert = best_a + rng.standard_normal(n) * sigma * best_a.mean()
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            break
        a_pert = a_pert / s
        
        a_new, score_new = lbfgs_anneal(a_pert, t_alloc, verbose=verbose)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
    
    return best_a, best_score

def solve_lp_best_response(a_fixed, solver=None):
    """LP: given fixed a, find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n
    
    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        a_avg = (iters * a_avg + a) / (iters + 1)
        
        for cand in [a, b, 0.5 * (a + b), a_avg]:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        if verbose and iters % 10 == 0:
            print(f"    FP it={iters}: best={best_score:.6f}")
    
    return best_a, best_score, iters

def damped_lp_refine(a0, time_limit, solver=None):
    """Damped alternating LP with decreasing step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 5.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

def coord_descent_polish(a0, time_limit, n_sweeps=300):
    """Coordinate descent: try scaling each coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for factor in [0.7, 0.85, 0.93, 1.07, 1.15, 1.35]:
                a[i] = old_val * factor
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s
                score = 2.0 * n * fast_convolve(a_norm).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a_norm.copy()
                    a = a_norm
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
        
        if not improved and sweep > 10:
            # Try smaller steps
            any_improve = False
            for i in rng.choice(n, size=min(n, 80), replace=False):
                old_val = a[i]
                if old_val < 1e-15:
                    continue
                for factor in [0.96, 0.98, 1.02, 1.04]:
                    a[i] = old_val * factor
                    s = a.sum()
                    if s < 1e-15:
                        a[i] = old_val
                        continue
                    a_norm = a / s
                    score = 2.0 * n * fast_convolve(a_norm).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a_norm.copy()
                        a = a_norm
                        old_val = a[i]
                        improved = True
                        any_improve = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()
            if not any_improve and sweep > 20:
                break
    
    return best_a, best_score

def peak_shave(a0, time_limit):
    """Target the highest peaks for reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(3000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        top_k = min(5, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        improved = False
        
        for peak_k in peaks:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            if contribs.max() < 1e-15:
                continue
            top_contrib = np.argsort(contribs)[::-1][:4]
            for ci in top_contrib:
                i = int(idx[ci])
                if a[i] < 1e-14:
                    continue
                for factor in [0.85, 0.75, 0.65]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    s = a_trial.sum()
                    if s < 1e-15:
                        break
                    a_trial = a_trial / s
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try pairs
            peak_k = int(peaks[0])
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top2 = np.argsort(contribs)[::-1][:3]
            pair_improved = False
            for ci1 in range(len(top2)):
                for ci2 in range(ci1 + 1, len(top2)):
                    i1 = int(idx[top2[ci1]])
                    i2 = int(idx[top2[ci2]])
                    if a[i1] < 1e-14 or a[i2] < 1e-14:
                        continue
                    for f1 in [0.8, 0.7]:
                        for f2 in [0.8, 0.7]:
                            a_trial = a.copy()
                            a_trial[i1] *= f1
                            a_trial[i2] *= f2
                            s = a_trial.sum()
                            if s < 1e-15:
                                continue
                            a_trial = a_trial / s
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-13:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                pair_improved = True
                                break
                        if pair_improved:
                            break
                    if pair_improved:
                        break
                if pair_improved:
                    break
            if not pair_improved:
                break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def nesterov_refine(a0, time_limit, top_k=30, lr_init=0.005):
    """Nesterov-accelerated projected subgradient as backup."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    
    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k
        
        b = fast_convolve(y)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            sg[j] += 2.0 * w[idx] * y[k - j]
        
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v
        y_new = y - lr * v
        y = simplex_project(y_new)
        
        score = 2.0 * n * fast_convolve(y).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
        if no_improve > 400:
            noise = rng.standard_normal(n) * 0.02
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.7, lr * 1.5)
            no_improve = 0
        lr *= 0.9998
    
    return best_a, best_score

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.CBC, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

    """
    Multi-phase optimization:
    1. Diverse initialization + short LP
    2. L-BFGS smoothed optimization with annealing (main)
    3. LP fictitious play
    4. Coordinate descent + peak shaving
    5. Multi-length exploration
    6. Final polish
    """

    rng = np.random.default_rng(2024)
    
    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    solver = None
    try:
        solver = get_lp_solver()
        print(f"LP solver: {solver}")

        print("No LP solver found, will use non-LP methods only")

            if rem < 40:
                break
            t_alt = min(60.0, rem * 0.4)
            print(f"  Alt n={alt_n}, time={t_alt:.0f}s")
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt = lbfgs_anneal(init_alt, t_alt, verbose=False)
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")
            
            # Quick CD
            rem = deadline - time.monotonic()
            if rem > 15:
                a_cd2, s_cd2 = coord_descent_polish(a_alt, min(20.0, rem * 0.3), n_sweeps=50)
                best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, f"alt_cd_n{alt_n}")
    
    # === Phase 7: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Final L-BFGS with very low temperature
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            a_fin, s_fin = lbfgs_anneal(best_seq, min(60.0, rem * 0.5), verbose=True)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_lbfgs")
        
        # Final CD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_fin2, _ = coord_descent_polish(best_seq, min(30.0, rem * 0.6), n_sweeps=100)
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_cd")
        
        # Final peak shave
        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_fin3, _ = peak_shave(best_seq, min(15.0, rem * 0.7))
            best_seq, best_score = check_and_update(a_fin3, best_seq, best_score, "final_ps")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
