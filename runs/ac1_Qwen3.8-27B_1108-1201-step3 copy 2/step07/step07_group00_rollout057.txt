            vals.append(f[k - m])
    
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K, n))
    
    c = -np.ones(n)
    b_ub = np.full(K, M)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None
    
    if not res.success:
        return None
    
    g_val = np.maximum(res.x, 0.0)
    S = g_val.sum()
    if S <= 1.0 + 1e-6:
        return None
    return g_val


def run_lp_optimization(f_init, time_limit, top_k_init=50, top_k_max=300, label="lp"):
    """LP-based optimization with adaptive K and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 40 == 0 and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)
        
        g = solve_lp_direction(f, top_k)
        
        if g is None:
            # Try with fewer constraints (relax)
            g = solve_lp_direction(f, max(10, top_k // 2))
            if g is None:
                stuck += 1
                if stuck > 15:
                    # Perturb and retry
                    f = project_simplex(f + rng.standard_normal(n) * 0.001)
                    f = np.maximum(f, 0)
                    s = f.sum()
                    if s > 1e-10: f /= s
                    top_k = top_k_init
                    stuck = 0
                continue
        
        S_g = g.sum()
        h = g / S_g
        
        # Line search
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()
        for t in [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10: continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-15:
                best_t_obj = obj
                best_t = t
        
        if best_t == 0.05 and best_t_obj >= fast_convolve(f).max() - 1e-15:
            # No improvement found in line search
            stuck += 1
            if stuck > 10:
                f = project_simplex(f + rng.standard_normal(n) * 0.003)
                f = np.maximum(f, 0)
                s = f.sum()
                if s > 1e-10: f /= s
                top_k = top_k_init
                stuck = 0
            continue
        
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10: f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break
        
        if it % 100 == 0:
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} K={top_k} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def run_fista_smooth(a_init, time_limit, tau_init=None, label="fista"):
    """FISTA on smooth approximation with adaptive tau."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    if tau_init is None:
        tau_init = best_obj / 20.0
    tau = tau_init
    
    momentum = 0.0
    lr = 0.5 * tau  # Conservative initial step
    it = 0
    no_improve = 0
    rng = np.random.default_rng(99)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = a + ((momentum - 1.0) / t_k) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = a.copy()
        
        g, b = compute_smooth_grad(y, tau)
        
        # Adaptive step size
        a_trial = project_simplex(y - lr * g)
        
        a_prev = a
        a = a_trial
        momentum = t_k
        
        if it % 100 == 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.1, 2.0 * tau)
            else:
                no_improve += 1
                if no_improve % 5 == 0:
                    lr *= 0.8
            
            if it % 500 == 0:
                tau = max(tau * 0.85, best_obj / 100.0)
                score = 2.0 * n * best_obj
                print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} tau={tau:.6f} t={time.monotonic()-t0:.0f}s")
            
            if no_improve > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                a_prev = a.copy()
                momentum = 0.0
                no_improve = 0
                lr = 0.3 * tau
    
    return best_a, best_obj


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    
    multipliers = [0.5, 0.7, 0.85, 1.0/0.85, 1.3, 2.0]
    
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f[j] = new_val
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_j_val
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            f[j] = best_j_val
            if best_j_obj < best_obj - 1e-13:
                s = f.sum()
                if s > 1e-10: f /= s
                best_obj = best_j_obj
                improved = True
        
        if not improved:
            break
    
    return f, best_obj
