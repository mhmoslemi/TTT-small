        result = minimize(
            objective, v, jac=grad_func, method='L-BFGS-B',
            bounds=bounds, options={'maxiter': 200, 'gtol': 1e-8, 'maxls': 40}
        )
        
        v = result.x
        a_new = softmax_params_to_seq(v)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Decrease temperature
        b_now = fast_convolve(a_new)
        tau_new = tau * 0.5
        if tau_new < b_now.max() * 0.02:
            tau_new = b_now.max() * 0.02
        tau_schedule.append(tau_new)
        
        if len(tau_schedule) > 30:
            break
    
    return best_a, best_score

def mirror_descent(a0, time_limit, lr_init=0.05, tau_init=None, verbose=False):
    """Mirror descent with multiplicative weights and adaptive step size."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if tau_init is None:
        b0 = fast_convolve(a)
        tau_init = max(b0.max() / 3.0, 1e-6)
    tau = tau_init
    
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        
        # Compute softmax weights
        b_scaled = b / tau
        b_max = b_scaled.max()
        exp_b = np.exp(b_scaled - b_max)
        p = exp_b / exp_b.sum()
        S = np.dot(p, b)
        
        # h[i] = 2 * sum_k p[k] * a[k-i]
        pa_conv = conv_general(p, a[::-1])
        h = 2.0 * pa_conv[n - 1:n - 1 + n]
        
        # Gradient direction (what we want to minimize)
        grad = h - 2.0 * S
        
        # Multiplicative weights update: decrease a where gradient is high
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.8
            no_improve = 0
        
        # Slowly decrease temperature
        if it % 50 == 0:
            b_now = fast_convolve(a)
            tau = max(tau * 0.95, b_now.max() * 0.05)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.4f} tau={tau:.6f}")
    
    return best_a, best_score

def coord_descent_polish(a0, time_limit, max_sweeps=500):
    """Coordinate descent: try scaling individual coordinates to reduce peak."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for factor in [0.7, 0.85, 0.93, 1.07, 1.15, 1.4]:
                new_val = old_val * factor
                a[i] = new_val
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
        
        if not improved and sweep > 20:
            # Try finer perturbations
            any_small = False
            for i in rng.choice(n, size=min(n, 200), replace=False):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[i]
                if old_val < 1e-15:
                    continue
                for factor in [0.97, 0.985, 1.015, 1.03]:
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-15:
                        a[i] = old_val
                        continue
                    a = a / s
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        any_small = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()
            if not any_small and not improved and sweep > 50:
                break
    
    return best_a, best_score

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b>=0, sum(b)=1, minimizing max conv(a,b)."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
    
    b = a.copy()
    best_a = a.copy()
    best_score = float('inf')
    iters = 0
    a_avg = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        a_avg = (iters * a_avg + a) / (iters + 1)
        
        candidates = [a, b, 0.5 * (a + b), a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()
        
        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={iters} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, iters

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def init_cosine(n, side='middle'):
    """Cosine window initialization."""
    t = np.arange(n)
    if side == 'middle':
        a = np.cos(np.pi * t / (n - 1)) ** 2
    elif side == 'edges':
        a = np.sin(np.pi * t / (n - 1)) ** 2
    else:
        a = (1 + np.cos(2 * np.pi * t / (n - 1))) / 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_two_peak(n, rng):
    """Two Gaussian peaks at n/4 and 3n/4."""
    t = np.arange(n, dtype=np.float64)
    sigma = n * rng.uniform(0.05, 0.15)
    a = np.exp(-((t - n * 0.25) ** 2) / (2 * sigma ** 2)) + np.exp(-((t - n * 0.75) ** 2) / (2 * sigma ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_comb(n, rng):
    """Comb pattern: 1 + alpha*cos(2*pi*m*i/n)."""
    m = int(rng.integers(2, max(3, n // 10)))
    alpha = rng.uniform(0.3, 0.9)
    t = np.arange(n)
    a = 1.0 + alpha * np.cos(2 * np.pi * m * t / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_fourier_structured(n, rng):
    """Structured Fourier: few harmonics with specific frequencies."""
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    n_harm = int(rng.integers(2, 8))
    for _ in range(n_harm):
        freq = int(rng.integers(1, n // 4))
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_power_law(n, rng):
    """Power law: a[i] proportional to (distance to center)^beta."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(t - center) / center + 1e-6
    beta = rng.uniform(-1.5, 1.5)
    a = dist ** beta
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_rngh(n, rng):
    """Random with smoothing."""
    a = rng.random(n)
    ksize = int(rng.integers(3, 15))
    kernel = np.ones(ksize) / ksize
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def evaluate_score(a):
    """Compute the score for a normalized sequence."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * n * b.max()

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

            t_fin = min(rem * 0.5, 60.0)
            a_fin, s_fin = coord_descent_polish(best_seq, t_fin, max_sweeps=500)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")
        
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_fin2, s_fin2 = mirror_descent(best_seq, min(rem * 0.6, 20.0), lr_init=0.005, verbose=False)
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_md")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
