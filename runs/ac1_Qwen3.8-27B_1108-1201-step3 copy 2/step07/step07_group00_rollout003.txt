def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Golden section search for t in [0, beta] minimizing obj_fn((1-t)*f + t*h)."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a = 0.0
    b = beta
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    def evaluate(t):
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:

        f_trial = f_trial / s
        return fast_convolve(f_trial).max()

    fd = evaluate(d)

        if fc < fd:

            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = evaluate(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = evaluate(d)
        if abs(b - a) < 1e-6:
            break

    Uses all peaks within n_active_frac of the max."""

    active_vals = b[active]

"""
Hybrid softmax-subgradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: Nesterov-accelerated gradient descent in softmax (logit) space with
importance-weighted subgradients. Secondary: LP-based direction finding with adaptive
constraint growth. Exploration across multiple sequence lengths.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, b, threshold_frac=0.05):
    """
    Compute weighted subgradient of max(b) w.r.t. a, where b = conv(a,a).
    Uses peaks within threshold_frac of max, weighted by softmax.
    """
    n = len(a)
    b_max = b.max()
    if b_max < 1e-15:
        return np.zeros(n)

    threshold = b_max * (1.0 - threshold_frac)
    active = np.where(b >= threshold)[0]

    active_vals = b[active]
    spread = active_vals.max() - active_vals.min()
    if spread > 1e-10:
        weights = np.exp((active_vals - active_vals.max()) / (spread * 0.05))
        weights /= weights.sum()
    else:
        weights = np.ones(len(active)) / len(active)

    grad = np.zeros(n)
    for idx in range(len(active)):
        k = active[idx]
        w = 2.0 * weights[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        segment = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += w * segment

    return grad


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def softmax_grad(g_a, a):
    """Convert gradient w.r.t. a (simplex) to gradient w.r.t. logit x."""
    return a * (g_a - np.dot(g_a, a))


def run_softmax_nesterov(a_init, time_limit, lr_init=0.01, momentum=0.9,
                         threshold_frac=0.05, label="softmax"):
    """
    Nesterov-accelerated gradient descent in softmax/logit space.
    Optimizes F(a) = max(conv(a,a)) with a = softmax(x), sum(a)=1 automatically.
    """
    t0 = time.monotonic()
    n = len(a_init)
    # Initialize logit from a_init
    a_clipped = np.clip(a_init, 1e-15, None)
    a_clipped = a_clipped / a_clipped.sum()
    x = np.log(a_clipped)
    x_prev = x.copy()
    v = np.zeros(n)

    best_a = a_init.copy()
    best_obj = fast_convolve(a_init).max()
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    # Adaptive threshold: start narrow, widen if stuck
    cur_threshold = threshold_frac
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov: lookahead step
        a = softmax_fn(x)
        b = fast_convolve(a)
        g_a = compute_subgradient(a, b, cur_threshold)
        g_x = softmax_grad(g_a, a)

        # Momentum update
        v = momentum * v + g_x
        x = x - lr * v

        # Evaluate
        a_new = softmax_fn(x)
        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            stuck_count = 0
            # Slightly increase lr when making progress
            lr = min(lr * 1.01, lr_init * 2.0)
        else:
            no_improve += 1
            stuck_count += 1
            if no_improve > 20:
                lr *= 0.9
                no_improve = 0
            if stuck_count > 100:
                # Widen threshold to include more peaks
                cur_threshold = min(cur_threshold * 1.5, 0.3)
                stuck_count = 0
                # Perturb to escape local minima
                x = x + rng.standard_normal(n) * 0.001
                v = np.zeros(n)
                lr = lr_init * 0.5

        x_prev = x.copy()

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={score:.6f} lr={lr:.6f} thr={cur_threshold:.3f} t={elapsed:.0f}s")
            # Anneal lr
            lr = max(lr * 0.99, lr_init * 0.1)

    return best_a, best_obj


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve(f, top_idx, M):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in top_idx.
    Returns g or None.
    """
    n = len(f)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-9)
    except Exception:
        return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def line_search(f, h, max_obj, max_steps=8):
    """Find best step size t in (0,1] such that obj((1-t)*f + t*h) < max_obj."""
    best_t = 0.0
    best_obj_trial = max_obj
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj_trial:
            best_obj_trial = obj
            best_t = t
    return best_t, best_obj_trial


def run_lp_phase(f_init, time_limit, top_k_init=50, top_k_max=400, label="lp"):
    """
    LP-based optimization with adaptive constraint set and progressive tightening.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()
    it = 0
    stuck = 0
    top_k = top_k_init

    # Adaptive constraint level: start at current max, tighten over time
    M_cur = best_obj * 0.999
    tighten_rate = 0.9995  # tighten M by this factor each successful iteration

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)

        b = fast_convolve(f)
        cur_max = b.max()

        # Update M: tighten if we've improved, else reset to current max
        if cur_max < M_cur:
            M_cur = cur_max * 0.999
        else:
            M_cur = cur_max * 0.999

        # Select top peaks
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)

        g = lp_solve(f, top_idx, M_cur)

        if g is None:
            # Relax constraint
            M_cur = cur_max * 1.001
            g = lp_solve(f, top_idx, M_cur)
            if g is None:
                stuck += 5
                if stuck > 20:
                    break
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > 15:

            M_cur *= 1.002

        h = g / S_g
        best_t, best_t_obj = line_search(f, h, cur_max)

        if best_t <= 0:

            if stuck > 15:
                break
            M_cur *= 1.003
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
            M_cur = cur_obj * 0.999
        else:
            stuck += 1
            if stuck > 40:
                break

        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={score:.6f} K={top_k} t={elapsed:.0f}s")

    return best_f, best_obj

def interpolate_sequence(f, target_n):
    """Interpolate a sequence to a new length."""
    old_x = np.linspace(0.0, 1.0, len(f))
    new_x = np.linspace(0.0, 1.0, target_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-10:
        f_interp /= s
    else:
        f_interp = np.ones(target_n) / target_n
    return f_interp


def coordinate_polish(f_init, time_limit, max_rounds=5):
    """Brief coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)

    multipliers = [0.7, 0.85, 1.15, 1.3]

    for _round in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        for j in range(0, n, max(1, n // 200)):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()

                s = f_trial.sum()
                if s < 1e-10:

                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-12:

        f7, obj7 = coordinate_polish(best_seq, time_limit=t7, max_rounds=3)
        try_update(f7, "cd")

    # === Final: one more LP burst if time ===
    rem = deadline - time.monotonic()
    if rem > 40:
        print(f"\n=== Final LP burst (remaining {rem:.0f}s) ===")
        f_final, obj_final = run_lp_phase(best_seq, time_limit=min(rem - 10, 80.0),
                                          top_k_init=200, top_k_max=400,
                                          label="lp_burst")
        try_update(f_final, "lp_burst")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
