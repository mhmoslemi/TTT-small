def fast_convolve(a):

    A = rfft(a, fft_size)

def project_simplex(v, target_sum=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = target_sum}."""
    # First ensure non-negativity
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-30:
        return np.full(len(v), target_sum / len(v))
    return v * (target_sum / s)

def optimize_with_smoothed_gradient(a0, time_limit, T_init=100.0, T_decay=0.999, 
                                      lr_init=0.01, momentum=0.9, verbose=False):
    """
    Optimize using smoothed gradient descent with adaptive temperature.
    Works in the space of non-negative vectors with fixed sum.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    iters = 0
    stall_count = 0
    min_improvement = 1e-12
    
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Compute gradient
        conv = fast_convolve(a)
        shifted = conv * T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        
        # Gradient: g[i] = sum_k w[k] * a[k-i] (cross-correlation)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad = irfft(np.conj(W) * A, fft_size)[:n]
        
        # Scale gradient
        grad = 2.0 * n * grad
        
        # Momentum
        v = momentum * v + lr * grad
        a_new = a - v
        
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate
        score = compute_score(a_new)
        
        if score < best_score - min_improvement:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            a = a_new  # still accept for momentum continuity
            stall_count += 1
        
        # Adaptive learning rate
        if stall_count > 50:
            lr *= 0.9
            v *= 0.5
            stall_count = 0
        elif stall_count > 100:
            # Perturb
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = lr_init * 0.5
            stall_count = 0
        
        # Decay temperature (make smoothing sharper over time)
        T = max(T_init * T_decay ** iters, 1.0)
        
        if verbose and iters % 100 == 0:
            print(f"  SGD iter={iters}, T={T:.1f}, lr={lr:.2e}, score={best_score:.8f}")
    
    return best_a, best_score


def active_lag_optimization(a0, time_limit, n_active=5, verbose=False):

    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

        iters += 1
        conv = fast_convolve(a)

        # Find top-k lags

        top_indices = np.argpartition(conv, -top_k)[-top_k:]

        # For lag k, dc_k/da_i = a[k-i] (if valid)

        a_new = a_new / s_new

def block_coordinate_descent(a0, time_limit, n_blocks=8, verbose=False):
    """
    Divide the sequence into blocks and optimize each block while keeping others fixed.
    Within each block, use L-BFGS to minimize the max convolution.
    """
    from scipy.optimize import minimize_scalar
    
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    block_size = max(4, n // n_blocks)
    n_actual_blocks = (n + block_size - 1) // block_size
    block_indices = []
    for b in range(n_actual_blocks):
        start = b * block_size
        end = min(n, start + block_size)
        block_indices.append((start, end))
    
    improved = True
    rounds = 0
    while time.monotonic() - t0 < time_limit and improved:
        rounds += 1
        improved = False
        
        for (start, end) in block_indices:
            if time.monotonic() - t0 > time_limit - 10:
                break
            
            # Optimize a[start:end] while keeping rest fixed
            # The objective is 2n * max_k c_k where c depends on the block
            # Use a simple 1D optimization: scale the block
            block = a[start:end].copy()
            block_sum = block.sum()
            if block_sum < 1e-15:
                continue
            
            # Try different scalings and redistributions
            best_block = block.copy()
            best_local = compute_score(a)
            
            # Try uniform scaling
            for scale in [0.8, 0.9, 1.0, 1.1, 1.2]:
                a_trial = a.copy()
                a_trial[start:end] = block * scale
                # Redistribute the difference to maintain sum=1
                diff = block_sum * (1 - scale)
                mask = np.ones(n, dtype=bool)
                mask[start:end] = False
                others_sum = a_trial[mask].sum()
                if others_sum > 1e-15:
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_block = a_trial[start:end].copy()
            
            # Try internal redistribution within the block
            for _ in range(5):
                i_off = rng.integers(0, end - start)
                j_off = rng.integers(0, end - start)
                if i_off == j_off:
                    continue
                t = rng.standard_normal() * 0.05 * block.sum()
                a_trial = a.copy()
                a_trial[start + i_off] -= t
                a_trial[start + j_off] += t
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_block = a_trial[start:end].copy()
            
            if best_local < best_score - 1e-14:
                a[start:end] = best_block
                best_score = best_local
                best_a = a.copy()
                improved = True
    
    if verbose:
        print(f"  BCD rounds={rounds}, best={best_score:.8f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Corrected smoothed gradient descent (log-sum-exp parametrization) with proper FFT cross-correlation
2. Adaptive temperature annealing with backtracking line search
3. Active peak equalization (focus on top-k convolution peaks)
4. Local search with tiny perturbations for final polish
5. Multi-length exploration around the parent length
Key fix: correct gradient computation using W * conj(A) for cross-correlation.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_conv_and_w(a, T):
    """Compute convolution and softmax weights."""
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv * T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones_like(conv) / len(conv)
    else:
        w = w / w_sum
    return conv, w


def compute_gradient(a, w):
    """
    Compute gradient of 2n*max(c_k) smoothed by log-sum-exp.
    grad[m] = 4n * sum_j w[m+j] * a[j]  (cross-correlation of w and a)
    """
    n = len(a)
    len_w = len(w)
    size = n + len_w - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: (w cross a)[m] = sum_j w[m+j]*a[j] = ifft(FFT(w) * conj(FFT(a)))
    grad = irfft(W * np.conj(A), fft_size)[:n]
    return 4.0 * n * grad


def smoothed_gd(a0, time_limit, T_init=50.0, T_final=1.0, lr_init=0.005,
                momentum_init=0.85, verbose=False):
    """
    Smoothed gradient descent with adaptive temperature annealing.
    Optimizes in the non-negative simplex (sum=1, a>=0).
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    iters = 0
    stall_count = 0
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        remaining_frac = 1.0 - (time.monotonic() - t0) / time_limit

        # Anneal temperature: start high (smooth), end low (sharp)
        T = T_final + (T_init - T_final) * (remaining_frac ** 2)
        T = max(T, 0.5)

        # Compute gradient
        conv, w = compute_conv_and_w(a, T)
        grad = compute_gradient(a, w)

        # Normalize gradient to prevent explosion
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e10:
            grad = grad / grad_norm * 1e10

        # Momentum update
        v = momentum_init * v + lr * grad
        a_new = a - v

        # Project to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        score = compute_score(a_new)

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            # Reduce learning rate slightly on success
            lr *= 0.999
        else:
            a = a_new
            stall_count += 1

        # Adaptive learning rate
        if stall_count > 100:
            lr *= 0.5
            v *= 0.3
            stall_count = 0
        elif stall_count > 300:

            a = np.maximum(a, 0.0)
            s_p = a.sum()
            if s_p > 1e-15:

            v = np.zeros(n)
            lr = max(lr_init * 0.3, lr * 0.5)
            stall_count = 0
        elif stall_count > 150:
            lr *= 0.8
            v *= 0.5

        if verbose and iters % 200 == 0:
            print(f"    SGD iter={iters}, T={T:.1f}, lr={lr:.2e}, score={best_score:.8f}")

        n_conv = len(conv)

        # Find top-k indices
        top_k = min(n_active, n_conv)
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
        top_indices = sorted(top_indices.tolist())

        # Build weighted indicator: give more weight to the max
        max_val = conv[top_indices].max()
        min_top = conv[top_indices].min()
        spread = max_val - min_top
        if spread < 1e-12:
            break

        w = np.zeros(n_conv)
        for k in top_indices:
            # Weight proportional to how close to max
            w[k] = 1.0 + 5.0 * (conv[k] - min_top) / (spread + 1e-15)
        w = w / w.sum()

        # Compute gradient
        grad = compute_gradient(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Try step
        stepped = False
        for step_mult in [1.0, 0.5, 0.1, 0.01]:
            alpha = lr * step_mult
            a_new = a - alpha * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            score = compute_score(a_new)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                stall = 0
                stepped = True
                break
            elif score < compute_score(a) + 1e-10:
                # Accept even small non-improvement for exploration
                a = a_new
                stall += 1

        if not stepped:
            stall += 1
            if stall > 50:
                # Perturb
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                lr = max(0.0001, lr * 0.7)
                stall = 0

        # Decay lr
        lr *= 0.9998
        if lr < 1e-6:
            lr = 1e-4
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        if verbose and iters % 500 == 0:
            print(f"    APR iter={iters}, best={best_score:.8f}, lr={lr:.1e}")

    return best_a, best_score


def local_search(a0, time_limit, n_trials_per_iter=200, verbose=False):
    """
    Very local search: try tiny perturbations and accept improvements.
    This is the final polish step.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(888)
    scale = 0.005
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n_trials_per_iter):
            if time.monotonic() - t0 > time_limit - 5:
                break

            # Random tiny perturbation
            a_trial = a + rng.standard_normal(n) * scale
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t

            score = compute_score(a_trial)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                no_improve = 0
            else:
                no_improve += 1

        if not improved:
            # Reduce scale
            scale *= 0.7
            if scale < 1e-7:
                # Try a slightly larger jump
                a = best_a + rng.standard_normal(n) * 0.01
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                scale = 0.003

    if verbose:
        print(f"    LS iters={iters}, best={best_score:.8f}")
    return best_a, best_score


def pair_swap_optimize(a0, time_limit, n_swaps_per_iter=500, verbose=False):
    """
    Optimize by trying pairwise transfers between coordinates.
    For each pair (i,j), find the optimal amount to transfer.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(31415)
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n_swaps_per_iter):
            if time.monotonic() - t0 > time_limit - 5:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try transferring from i to j (and vice versa)
            base = min(a[i], a[j]) if a[j] > 0 else a[i]
            if base < 1e-10:
                # Still try small transfers
                base = max(a[i], 1e-6)

            # Try several fractions
            for frac in [0.1, 0.3, 0.5, 0.01, 0.05]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] = max(0, a[i] - t)
                a_trial[j] = a[j] + t
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break

            # Also try reverse direction
            if not improved and a[j] > 1e-10:
                for frac in [0.1, 0.3, 0.5]:
                    t = frac * a[j]
                    a_trial = a.copy()
                    a_trial[j] = max(0, a[j] - t)
                    a_trial[i] = a[i] + t
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = compute_score(a_trial)
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        break

        if not improved:
            stall += 1
            if stall > 50:
                a = best_a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stall = 0

        if verbose and iters % 100 == 0:
            print(f"    PS iter={iters}, best={best_score:.8f}")

    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def propagate_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Smoothed GD at parent length (adaptive T)
    Phase 2: Active peak reduction
    Phase 3: Pair swap optimization
    Phase 4: Local search polish
    Phase 5: Multi-length exploration
    Phase 6: Final polish
    """
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)
    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Smoothed GD (main optimization) ===
    rem = deadline - time.monotonic()
    if rem > 60:
        t1 = min(rem * 0.35, 600.0)
        print(f"\n=== Phase 1: Smoothed GD (n={n_parent}, {t1:.0f}s) ===")
        a1, s1 = smoothed_gd(best_seq, t1, T_init=30.0, T_final=0.5,
                             lr_init=0.003, momentum_init=0.85, verbose=True)
        try_update(a1, "SGD1")

    # === Phase 2: Active Peak Reduction ===
    rem = deadline - time.monotonic()
    if rem > 40:
        t2 = min(rem * 0.25, 400.0)
        print(f"\n=== Phase 2: Active Peak Reduction (n={n_parent}, {t2:.0f}s) ===")
        a2, s2 = active_peak_reduction(best_seq, t2, n_active=3, verbose=True)
        try_update(a2, "APR")

    # === Phase 3: More Smoothed GD (different params) ===
    rem = deadline - time.monotonic()
    if rem > 40:
        t3 = min(rem * 0.25, 350.0)
        print(f"\n=== Phase 3: Smoothed GD v2 (n={n_parent}, {t3:.0f}s) ===")
        a3, s3 = smoothed_gd(best_seq, t3, T_init=10.0, T_final=0.3,
                             lr_init=0.001, momentum_init=0.9, verbose=True)
        try_update(a3, "SGD2")

    # === Phase 4: Pair Swap Optimization ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t4 = min(rem * 0.2, 250.0)
        print(f"\n=== Phase 4: Pair Swap (n={n_parent}, {t4:.0f}s) ===")
        a4, s4 = pair_swap_optimize(best_seq, t4, n_swaps_per_iter=300, verbose=True)
        try_update(a4, "PSWAP")

    # === Phase 5: Multi-length exploration ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 5: Multi-length ({rem:.0f}s) ===")
        for alt_n in [n_parent - 1, n_parent + 1, n_parent - 2, n_parent + 2,
                      n_parent - 3, n_parent + 3]:
            if alt_n < 64 or alt_n > 2048:
                continue
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t5 = min(rem * 0.25, 120.0)
            a_alt = interpolate_to_length(best_seq, alt_n)
            print(f"  Trying n={alt_n} ({t5:.0f}s)...")
            a5, s5 = smoothed_gd(a_alt, t5, T_init=20.0, T_final=1.0,
                                 lr_init=0.002, momentum_init=0.88)
            try_update(a5, f"n={alt_n}")

    # === Phase 6: Local search polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t6 = min(rem * 0.4, 150.0)
        print(f"\n=== Phase 6: Local Search ({t6:.0f}s) ===")
        a6, s6 = local_search(best_seq, t6, n_trials_per_iter=300, verbose=True)
        try_update(a6, "LOCAL")

    # === Phase 7: Final active peak + local search ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.4, 100.0)
        print(f"\n=== Phase 7: Final APR ({t7:.0f}s) ===")
        a7, s7 = active_peak_reduction(best_seq, t7, n_active=5, verbose=True)
        try_update(a7, "APR_final")

    rem = deadline - time.monotonic()
    if rem > 15:
        t8 = min(rem * 0.6, 60.0)
        print(f"\n=== Phase 8: Final Polish ({t8:.0f}s) ===")
        a8, s8 = local_search(best_seq, t8, n_trials_per_iter=500)
        try_update(a8, "FINAL")

    # Final verification
    if best_seq is not None:
        final = [float(max(0.0, x)) for x in best_seq]
        fs = float(evaluate_sequence(final))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq / best_seq.sum()

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")

    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(max(0.0, x)) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main entry point. Multi-strategy optimization to minimize
    2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    return propagate_candidate(seed=seed, budget_s=budget_s)
