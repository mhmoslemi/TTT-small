def softmax_from_z(z):

def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.5, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 5
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages

        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 3:
            break

                fun=make_func(stage_T),

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-centric approach with:
1. Best-response LP alternating with Nesterov acceleration (main workhorse)
2. Periodic L-BFGS soft-max refinement to escape slow LP convergence

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached

    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:

    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

            a = np.maximum(a, 0)
            sa = a.sum()
            if sa < 1e-15:
                a = y.copy()
            else:
                a = a / sa
        score_a = evaluate_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if nesterov:
            score_y = evaluate_score(y)
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP alt iter={it+1} ({elapsed:.0f}s): best={best_score:.6f} (LPs={lp_count})")
    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=10000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()
    stuck = 0
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:4]
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.92, 0.87]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_score(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stuck = 0
                    break
            if improved:
                break
        if not improved:
            if len(top_contrib) >= 1:
                i = idx[top_contrib[0]]
                j = peak_k - i
                if i != j and a[j] > 1e-14:
                    for f1 in [0.96, 0.92]:
                        for f2 in [0.96, 0.92]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            score = evaluate_score(a_trial)
                            if score < best_score - 1e-12:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                stuck = 0
                                break
                        if improved:
                            break
            if not improved:
                stuck += 1
                if stuck > 200:
                    break
    return best_a, best_score


def mirror_descent_step(a, lr, threshold_frac=0.05):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b - threshold_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    a_new = np.maximum(a - lr * g, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def mirror_descent_optimize(a0, time_limit, lr_init=0.02, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        a_new = mirror_descent_step(a, lr)
        score_new = evaluate_score(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
        lr = max(lr, 1e-7)
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    LP-centric optimization with periodic L-BFGS refinement.
    Key innovations: much more LP time, periodic restarts, better Nesterov,
    multi-start with different lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # Phase 1: Quick L-BFGS warmup (120s)
    print(f"\n=== Phase 1: L-BFGS Warmup (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 120.0)
    if t_alloc > 20:
        print(f"  L-BFGS: T=50->1, n={n_best}, budget={t_alloc:.0f}s")
        a_warm = lbgfs_refine(best_seq, t_alloc, T_start=50.0, T_end=1.0, maxiter=6000)
        sc = evaluate_score(a_warm)
        if sc < best_score:
            best_score = sc
            best_seq = a_warm.copy()
            print(f"  Warmup improved: {sc:.6f}")

    # Phase 2: LP Alternating with periodic L-BFGS (850s)
    print(f"\n=== Phase 2: LP Alternating (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.47, 850.0)
    if t_alloc > 60:
        print(f"  LP+Nesterov: n={len(best_seq)}, budget={t_alloc:.0f}s")
        # Run in chunks with periodic L-BFGS
        n_chunks = 3
        chunk_budget = t_alloc / n_chunks * 0.8
        for chunk in range(n_chunks):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_chunk = min(chunk_budget, rem * 0.45)
            if t_chunk < 30:
                break
            print(f"  LP chunk {chunk+1}/{n_chunks}: budget={t_chunk:.0f}s, current best={best_score:.6f}")
            a_lp, lp_score = lp_alternating(best_seq, t_chunk, solver=solver, nesterov=True)
            sc = evaluate_score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
                print(f"  LP chunk {chunk+1} improved: {sc:.6f}")
            # Periodic L-BFGS restart
            rem = deadline - time.monotonic()
            if rem > 40 and chunk < n_chunks - 1:
                t_lb = min(rem * 0.12, 60.0)
                if t_lb > 15:
                    print(f"  L-BFGS restart (T=5->0.5): budget={t_lb:.0f}s")
                    a_restart = lbgfs_refine(best_seq, t_lb, T_start=5.0, T_end=0.5, maxiter=3000)
                    sc = evaluate_score(a_restart)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a_restart.copy()
                        print(f"  Restart improved: {sc:.6f}")
        # Final LP push
        rem = deadline - time.monotonic()
        if rem > 60:
            t_final_lp = min(rem * 0.4, 120.0)
            print(f"  Final LP push: budget={t_final_lp:.0f}s")
            a_lp_f, _ = lp_alternating(best_seq, t_final_lp, solver=solver, nesterov=True)
            sc = evaluate_score(a_lp_f)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp_f.copy()
                print(f"  Final LP improved: {sc:.6f}")

    # Phase 3: Fine L-BFGS (250s)
    print(f"\n=== Phase 3: Fine L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 30:
        print(f"  L-BFGS fine: T=2->0.1, n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_fine = lbgfs_refine(best_seq, t_alloc, T_start=2.0, T_end=0.1, maxiter=8000)
        sc = evaluate_score(a_fine)
        if sc < best_score:
            best_score = sc
            best_seq = a_fine.copy()
            print(f"  Fine L-BFGS improved: {sc:.6f}")

    # Phase 4: Mirror descent (100s)
    print(f"\n=== Phase 4: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        print(f"  MirrorDesc: n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_md, _ = mirror_descent_optimize(best_seq, t_alloc, lr_init=0.01)
        sc = evaluate_score(a_md)
        if sc < best_score:
            best_score = sc
            best_seq = a_md.copy()
            print(f"  MirrorDesc improved: {sc:.6f}")

    # Phase 5: Peak shaving (150s)
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  PeakShave: n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave(best_seq, t_alloc, n_steps=30000)
        sc = evaluate_score(a_ps)
        if sc < best_score:
            best_score = sc
            best_seq = a_ps.copy()
            print(f"  PeakShave improved: {sc:.6f}")

    # Phase 6: Alternative lengths (150s)
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for factor in [1.25, 1.5, 0.8, 2.0]:
        alt_n = int(round(n_best * factor))
        if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)
    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(rem * 0.35, 80.0)
        if t_alloc < 25:
            break
        print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt = lbgfs_refine(init_alt, t_alloc, T_start=20.0, T_end=1.0, maxiter=3000)
        sc = evaluate_score(a_alt)
        if sc < best_score:
            best_score = sc
            best_seq = a_alt.copy()
            print(f"  Alt n={alt_n} improved: {sc:.6f}")
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_lp = min(rem * 0.3, 40.0)
            if t_lp > 15:
                a_al, _ = lp_alternating(best_seq, t_lp, solver=solver, nesterov=True)
                sc = evaluate_score(a_al)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_al.copy()
                    print(f"  LP alt n={alt_n} improved: {sc:.6f}")

    # Phase 7: Final polish (remaining)
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t1 = min(rem * 0.4, 30.0)
        a_f1 = lbgfs_refine(best_seq, t1, T_start=1.0, T_end=0.05, maxiter=3000)
        sc = evaluate_score(a_f1)
        if sc < best_score:

            best_seq = a_f1.copy()

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
