import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def interp_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    sn = f_new.sum()
    if sn > 1e-15:
        f_new = f_new / sn
    return f_new

def build_sparse_conv_matrix(a, n):
    """Build sparse matrix C where C[k,i] = a[k-i] if 0<=k-i<n."""
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # C[k, i] = a[k-i] for k-i in [0, n-1], i.e., k in [i, i+n-1]
        k_range = range(i, i + n)
        rows.extend(k_range)
        cols.extend([i] * n)
        vals.extend(a.tolist())
    C = sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return C

def lp_fictitious_play(a0, time_limit, max_iter=100, step_schedule='constant'):

        w = np.exp((c - mc) / T)
        w = w / w.sum()
        
        # grad_a[i] = 2 * sum_k w[k] * d(c[k])/d(a[i])
        # d(c[k])/d(a[i]) = 2*a[k-i] (if 0<=k-i<n, i≠k-i) + a[i] (if k=2i)
        # = 2*(w ⊛ a)[i] - w[2i]*a[i] (correction for diagonal)
        # where (w ⊛ a)[i] = sum_m w[m+i]*a[m]
        
        corr = np.correlate(w, a, mode='full')
        # corr[k + n - 1] = sum_m w[m + k]*a[m] for k = 0, ..., n-1
        cross_corr = corr[n-1 : n-1+n]  # shape (n,)
        
        # Diagonal correction: subtract w[2i]*a[i] for valid 2i
        diag_correction = np.zeros(n)
        valid_2i = 2 * np.arange(n)
        valid_mask = valid_2i < 2*n-1
        diag_correction[valid_mask] = w[valid_2i[valid_mask]] * a[valid_mask]
        
        grad_a = 2.0 * cross_corr - diag_correction
        grad_a *= 2.0 * n  # Chain rule for 2n factor
        
        # Transform to z-space
        S = np.dot(a, grad_a)
        return a * (grad_a - S)
    
    try:
        res = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                               'maxfun': maxiter * 30, 'disp': False})
        a_out = log_softmax(res.x)
    except Exception:
        a_out = a_init.copy()
    
    a_out = np.maximum(a_out, 0)
    a_out = a_out / a_out.sum()
    return a_out

def peak_shave_v2(a0, time_limit, n_trials=5000):
    """
    Improved peak shaving: identify top peaks, find contributing pairs,
    and try to reduce them by redistributing mass.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        # Find top peaks (excluding borders)
        c_search = c.copy()
        c_search[0] = -np.inf
        c_search[-1] = -np.inf
        peak_k = int(np.argmax(c_search))
        
        if c[peak_k] <= best_M * 1.0001:
            if trial > 50:
                # Try random perturbation to escape
                if trial % 100 == 0:
                    idx = np.random.randint(n)
                    a_trial = a.copy()
                    diff = a[idx] * 0.1
                    a_trial[idx] = a[idx] - diff
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    if fast_convolve(a_trial).max() < best_M:
                        a = a_trial
                        best_M = fast_convolve(a).max()
                        best_a = a.copy()
                continue
            else:
                continue
        
        # Elements contributing to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        top_idx = np.argsort(contribs)[::-1][:8]
        improved = False
        
        for ci in top_idx:
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j or a[i] < 1e-15 or a[j] < 1e-15:
                continue
            
            # Try various adjustments
            for f1 in [0.85, 0.8, 0.7, 0.6]:
                for f2 in [1.0, 1.05, 1.1, 1.15]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    sn = a_trial.sum()
                    if sn < 1e-15:
                        continue
                    a_trial = a_trial / sn
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_M - 1e-12:
                        a = a_trial
                        best_M = M_trial
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved:
                # Try single element reduction with redistribution
                for factor in [0.85, 0.7, 0.5]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    # Redistribute to elements far from peak
                    mask = np.ones(n, dtype=bool)
                    # Exclude elements that contribute to top peaks
                    top_peaks = np.argsort(c)[::-1][:10]
                    for pk in top_peaks:
                        for ii in range(n):
                            jj = int(pk) - ii
                            if 0 <= jj < n:
                                mask[ii] = False
                                mask[jj] = False
                    if mask.sum() > 0 and a_trial[mask].sum() > 1e-15:
                        a_trial[mask] += diff * a_trial[mask] / a_trial[mask].sum()
                        M_trial = fast_convolve(a_trial).max()
                        if M_trial < best_M - 1e-12:
                            a = a_trial
                            best_M = M_trial
                            best_a = a_trial.copy()
                            improved = True
                            break
                if improved:
                    break
        
        if not improved and trial > 200 and trial % 50 == 0:
            # Random restart from best
            a = best_a + np.random.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_M

def adam_optimize(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.008):
    """Adam optimizer in log-space with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    a = a / a.sum()
    z = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (3*n-3).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** progress
        
        a_curr = np.exp(z - np.max(z))
        a_curr = a_curr / a_curr.sum()
        
        c = fast_convolve(a_curr)
        mc = c.max()
        exp_c = np.exp((c - mc) / T)
        w = exp_c / exp_c.sum()
        
        # Gradient via cross-correlation
        corr = np.correlate(w, a_curr, mode='full')
        cross_corr = corr[n-1 : n-1+n]
        valid_2i = 2 * np.arange(n)
        valid_mask = valid_2i < 2*n-1
        diag = np.zeros(n)
        diag[valid_mask] = w[valid_2i[valid_mask]] * a_curr[valid_mask]
        
        grad_a = (2.0 * cross_corr - diag) * 2.0 * n
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Adam
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1-beta1) * grad_z
        v = beta2 * v + (1-beta2) * (grad_z * grad_z)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = np.exp(z - np.max(z))
        a_new = a_new / a_new.sum()
        score = fast_convolve(a_new).max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            z = np.log(np.maximum(best_a, 1e-10)) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr *= 0.8
    
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")
    
    def track(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    
    n_parent = len(best_seq)
    print(f"Starting: score={best_score:.6f}, n={n_parent}")
    
    # === Phase 1: LP Fictitious Play ===
    print(f"\n=== Phase 1: LP Fictitious Play (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t_p1 = min(rem * 0.3, 300.0)
    if t_p1 > 30:
        a_fp, _ = lp_fictitious_play(best_seq, t_p1, max_iter=200)
        track(a_fp, "fp_main")
    
    # === Phase 2: Multi-start Adam ===
    print(f"\n=== Phase 2: Multi-Start Adam ===")
    rem = deadline - time.monotonic()
    t_p2 = min(rem * 0.3, 400.0)
    
    # Generate diverse starts
    starts = []
    starts.append(best_seq.copy())
    
    # Perturbed
    for scale in [0.02, 0.05, 0.1]:
        pert = best_seq + rng.standard_normal(n_parent) * scale
        pert = np.maximum(pert, 0)
        pert = pert / pert.sum()
        starts.append(pert)
    
    # Different shapes
    x = np.linspace(0, 1, n_parent)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        shape = x ** alpha
        shape = np.maximum(shape, 0.001)
        shape = shape / shape.sum()
        mix = 0.5 * best_seq + 0.5 * shape
        mix = mix / mix.sum()
        starts.append(mix)
    
    # Cosine modulation
    for freq in [2, 4, 8]:
        mod = best_seq * (1 + 0.1 * np.cos(2*np.pi*freq*x))
        mod = np.maximum(mod, 0)
        mod = mod / mod.sum()
        starts.append(mod)
    
    time_per_start = t_p2 / min(len(starts), 6)
    n_starts = min(6, len(starts))
    
    for idx in range(n_starts):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_sub = min(time_per_start, rem * 0.4)
        if t_sub < 20:
            continue
        print(f"  Adam start {idx+1}/{n_starts}, budget={t_sub:.0f}s")
        a_res, _ = adam_optimize(starts[idx], t_sub, T_init=0.04, T_final=0.002, lr=0.007)
        track(a_res, f"adam_{idx}")
    
    # === Phase 3: L-BFGS Polish ===
    print(f"\n=== Phase 3: L-BFGS Polish (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for T in [100, 30, 10]:
            rem = deadline - time.monotonic() - 50
            if rem < 20:
                break
            t_sub = min(rem * 0.3, 80.0)
            a_lb = smooth_lbgfs(best_seq, t_sub, T=T, maxiter=2000)
            track(a_lb, f"lbgfs_T{T}")
    
    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_sub = min(rem * 0.4, 120.0)
        a_ps, _ = peak_shave_v2(best_seq, t_sub, n_trials=10000)
        track(a_ps, "peak_shave")
    
    # === Phase 5: Final Adam + Peak Shave ===
    print(f"\n=== Phase 5: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_sub = min(rem * 0.4, 60.0)
