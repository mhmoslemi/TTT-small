```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Soft-max L-BFGS with geometric temperature annealing (efficient FFT-based gradients)
2. Hinge-loss joint optimization (simultaneously optimize sequence and target peak)
3. Projected subgradient with backtracking line search
4. Smart peak shaving with safe-coordinate redistribution
5. Final micro-polish
Key innovations: O(n log n) gradient via FFT convolution, hinge-loss objective,
and adaptive peak redistribution strategy.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def simplex_project(x):
    """Project x onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def compute_grad_a(a, w):
    """
    Compute grad_a where grad_a[i] = sum_k w[k] * db[k]/da[i]
    and db[k]/da[i] = 2*a[k-i] if 0<=k-i<n else 0.
    Uses FFT-based convolution for O(n log n) efficiency.
    """
    n = len(a)
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (with a[j]=0 for j outside [0,n-1])
    # This equals 2 * conv(w_reversed, a)[n-1:2n-1] reversed
    w_rev = w[::-1]
    conv_result = np.convolve(w_rev, a, mode='full')
    return 2.0 * conv_result[n-1:2*n-1][::-1]


def softmax_obj_and_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    Tb = T * (b - max_b)
    exp_Tb = np.exp(Tb)
    log_sum_exp = max_b * T + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = compute_grad_a(a, w)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def hinge_obj_and_grad(z, M, n, lam=0.1):
    """Hinge-loss objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    excess = np.maximum(b - M, 0.0)
    obj = np.sum(excess**2) + lam * M
    w = 2.0 * excess
    grad_a = compute_grad_a(a, w)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    grad_M = -np.sum(excess) + lam
    return obj, grad_z, grad_M


def hinge_joint_wrapper(x, n, lam):
    """Wrapper for L-BFGS: x = [z_0, ..., z_{n-1}, M]."""
    z = x[:-1]
    M = x[-1]
    obj, grad_z, grad_M = hinge_obj_and_grad(z, M, n, lam)
    return obj, np.concatenate([grad_z, [grad_M]])


def lbgfs_softmax(a0, time_limit, T_start=100.0, T_end=1.0, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing and geometric temperature."""
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()
    
    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(10, maxiter // n_stages)
    
    for stage_idx, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_val):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            max_b = b.max()
            Tb = T_val * (b - max_b)
            exp_Tb = np.exp(Tb)
            log_sum = max_b * T_val + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum
        
        def jac(z_val):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            max_b = b.max()
            Tb = T_val * (b - max_b)
            exp_Tb = np.exp(Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = compute_grad_a(a, w)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-10, 'maxfun': iters_per_stage * 30, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS stage {stage_idx} error: {e}")
    
    return softmax_from_z(z)


def hinge_lbgfs(a0, time_limit, lam=0.1, M_init_margin=0.0):
    """Hinge-loss L-BFGS: jointly optimize sequence (via softmax) and target M."""
    n = len(a0)
    z = z_from_a(a0)
    b0 = fast_convolve(a0)
    M = b0.max() + M_init_margin
    
    x0 = np.concatenate([z, [M]])
    t0 = time.monotonic()
    
    def func(x_val):
        return hinge_joint_wrapper(x_val, n, lam)[0]
    
    def jac(x_val):
        return hinge_joint_wrapper(x_val, n, lam)[1]
    
    try:
        res = minimize(
            fun=func, x0=x0, jac=jac, method='L-BFGS-B',
            bounds=[(None, None)] * n + [(b0.min(), None)],
            options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-10,
                     'maxfun': 150000, 'disp': False}
        )
        z_result = res.x[:-1]
    except Exception as e:
        print(f"    Hinge L-BFGS error: {e}")
        z_result = z
    
    return softmax_from_z(z_result)


def subgradient_optimize(a0, time_limit, lr_init=0.01, n_steps=50000):
    """
    Projected subgradient descent on the simplex.
    Uses exact max subgradient with backtracking line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: peaks within 2% of max
        threshold = max_b - 0.02 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient: average gradient over active peaks
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        grad_a = compute_grad_a(a, w)
        
        # Line search: try a few step sizes
        best_lr = lr
        best_trial = a
        best_trial_score = best_score
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            a_trial = simplex_project(a - lr_try * grad_a)
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_trial_score:
                best_trial_score = score
                best_trial = a_trial
                best_lr = lr_try
        
        if best_trial_score < best_score - 1e-15:
            best_score = best_trial_score
            best_a = best_trial.copy()
            a = best_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            lr *= 0.8
            no_improve = 0
        if no_improve > 200:
            lr *= 0.9
            no_improve = 100
        lr = max(lr, 1e-8)
        
        if step % 5000 == 0:
            print(f"    Subgrad {time.monotonic()-t0:.0f}s: score={best_score:.6f} lr={lr:.8f} active={len(active)}")
    
    return best_a, best_score


def peak_shave_smart(a0, time_limit, n_steps=5000):
    """
    Smart peak shaving: identifies the peak, finds contributing coordinates,
    and redistributes mass to "safe" coordinates (those not contributing to peaks).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the main peak (excluding edges which are naturally smaller)
        b_inner = b.copy()
        b_inner[:3] = -np.inf
        b_inner[-3:] = -np.inf
        peak_k = int(np.argmax(b_inner))
        
        if 2.0 * n * b[peak_k] >= best_score - 1e-14:
            # This peak determines the score
            pass
        
        # Find contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Top contributors
        top_n = min(5, len(idx))
        top_contrib = np.argsort(contribs)[::-1][:top_n]
        
        improved = False
        
        # Strategy 1: Reduce individual top contributor
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.9, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                
                # Try proportional redistribution
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = a[mask] + diff * a[mask] / a[mask].sum()
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if improved:
            continue
        
        # Strategy 2: Reduce two contributors simultaneously
        if len(top_contrib) >= 2:
            for ci1 in range(min(3, len(top_contrib))):
                for ci2 in range(ci1 + 1, min(3, len(top_contrib))):
                    i1 = idx[top_contrib[ci1]]
                    i2 = idx[top_contrib[ci2]]
                    if i1 == i2 or a[i1] < 1e-14 or a[i2] < 1e-14:
                        continue
                    for f1 in [0.95, 0.9]:
                        for f2 in [0.95, 0.9]:
                            a_trial = a.copy()
                            diff1 = a_trial[i1] * (1 - f1)
                            diff2 = a_trial[i2] * (1 - f2)
                            a_trial[i1] *= f1
                            a_trial[i2] *= f2
                            mask = np.ones(n, dtype=bool)
                            mask[i1] = False
                            mask[i2] = False
                            total_diff = diff1 + diff2
                            a_trial[mask] = a[mask] + total_diff * a[mask] / a[mask].sum()
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if improved:
            continue
        
        # Strategy 3: Reduce top-3 contributors uniformly
        a_trial = a.copy()
        top3 = top_contrib[:3]
        for ci in top3:
            a_trial[idx[ci]] *= 0.93
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            a = a_trial
            best_score = score
            best_a = a.copy()
            continue
        
        # Strategy 4: Random perturbation (annealing-like)
        if step % 100 == 0:
            rng = np.random.default_rng(step)
            a_trial = a.copy()
            perturbation = rng.normal(0, 0.01 * a.max(), n)
            a_trial = a + perturbation
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            # Accept with some probability if worse (simulated annealing)
            if score < best_score * 1.01:
                a = a_trial  # Accept slightly worse for exploration
        
        # If completely stuck, break
        if step > 500 and not improved and step % 500 == 0:
            # Try a larger perturbation
            rng = np.random.default_rng(12345 + step)
            a_trial = a + rng.normal(0, 0.05 * a.max(), n)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score * 1.05:
                a = a_trial
    
    return best_a, best_score
```
