def lp_step(f, M, t=0.5):
    """One LP-based iteration step."""
    n = len(f)
    # Build constraint matrix
    # (f*b)[k] = sum_j f[k-j]*b[j] for valid j
    # For k=0: only j=0, coefficient f[0]
    # For k=1: j=0,1, coefficients f[1], f[0]
    # ...
    
    # Use scipy sparse matrix for efficiency
    from scipy.sparse import lil_matrix, csr_matrix
    
    F = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n-1)
        for j in range(j_start, j_end+1):
            F[k, j] = f[k-j]
    F = F.tocsr()
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [F @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL)  # or ECOS, GLOP
    
    if b_var.value is None:
        return f.copy()
    
    b_val = np.maximum(b_var.value, 0)
    s = b_val.sum()
    if s < 1e-10:
        return f.copy()
    g = b_val * (f.sum() / s)
    return g

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterations (following 
the state-of-the-art approach), combined with mirror descent polishing, 
Nesterov-accelerated subgradient method, and coordinate descent refinement.
The LP approach finds the optimal direction to move while keeping the 
convolution peak bounded, providing a principled and efficient optimization.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csr_matrix
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def solve_lp(f, M):
    """
    Solve LP: maximize sum(b) subject to (f*b)[k] <= M for all k, b >= 0.
    Returns the optimal b (not rescaled).
    """
    n = len(f)
    
    # Build sparse constraint matrix F where (F @ b)[k] = (f*b)[k]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    F = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    b_var = cp.Variable(n)
    constraints = [F @ b_var <= M, b_var >= 0]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if b_var.value is not None:
                return np.maximum(b_var.value, 0)
        except Exception:
            continue
    
    return f.copy()


def lp_iteration(f, t=0.5, max_t_trials=5):
    """
    One LP-based iteration: solve LP, interpolate, return best result.
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    sum_f = f.sum()
    
    b_opt = solve_lp(f, M)
    s = b_opt.sum()
    if s < 1e-10:
        return f.copy(), 0.0
    
    g = b_opt * (sum_f / s)
    
    # Try different t values and pick the best
    best_f = f.copy()
    best_score = score_from_a(f)
    
    for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
        f_trial = (1 - t) * f + t * g
        f_trial = np.maximum(f_trial, 0)
        s_trial = f_trial.sum()
        if s_trial < 1e-10:
            continue
        f_trial = f_trial / s_trial
        sc = score_from_a(f_trial)
        if sc < best_score:
            best_score = sc
            best_f = f_trial.copy()
    
    return best_f, best_score


def mirror_descent(a_init, time_limit, T_start=0.01, T_end=0.001,
                   eta_start=0.03, eta_end=0.001, max_iters=50000,
                   top_k=5, momentum=0.9):
    """Mirror descent with top-K peak weighting and Nesterov momentum."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    # Nesterov momentum
    v = np.zeros(n)
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
        
        # Top-K peak weighting
        # Get indices of top K peaks
        top_indices = np.argsort(b)[::-1][:top_k]
        # Exponential weighting
        peak_vals = b[top_indices]
        weights = np.exp(peak_vals / T - peak_vals[0] / T)
        weights = weights / weights.sum()
        
        # Compute gradient as weighted sum of subgradients at top peaks
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_indices):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w_k * 2.0 * a[k - idx]
        
        # Project to tangent space
        grad = grad - np.dot(a, grad)
        
        # Mirror update with momentum
        update = v + eta * grad
        v = momentum * v + eta * grad  # simplified momentum
        # Actually let me use proper Nesterov:
        # v = momentum * v + grad
        # a = a * exp(-eta * v)
        
        log_update = -eta * (v + grad)
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        
        v = 0.9 * v + 0.1 * grad  # slow momentum
        a = a_new
    
    return best_a, best_score


def coordinate_descent(a_init, time_limit=30.0):
    """Coordinate descent with optimized step for each coordinate."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        no_improve = 0
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-10:
                continue
            for f_fac in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f_fac
                if a_trial[i] < 0:
                    continue
                # Redistribute
                diff = a_trial.sum() - 1.0
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
    
    return best_a, best_score

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterations (SOTA approach),
Nesterov-accelerated mirror descent with top-K peak subgradients, and coordinate
descent polishing. The LP finds the optimal direction to increase total mass while
keeping the convolution peak bounded, providing principled and efficient optimization.
Key improvements: sparse LP with precomputed structure, adaptive step sizes,
momentum-based mirror descent, and strategic restarts to escape local optima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_lp_indices(n):
    """Precompute row/col indices for the sparse LP constraint matrix."""
    rows = []
    cols = []
    kj_pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            kj_pairs.append((k, j))
    return np.array(rows), np.array(cols), kj_pairs


def solve_lp(f, M, row_idx, col_idx, kj_pairs):
    """Solve LP: maximize sum(b) s.t. (f*b)[k] <= M, b >= 0. Returns optimal b."""
    n = len(f)
    vals = np.array([f[k - j] for k, j in kj_pairs])
    F = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    
    b_var = cp.Variable(n)
    constraints = [F @ b_var <= M, b_var >= 0]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    last_err = None
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if b_var.value is not None and np.all(np.isfinite(b_var.value)):
                return np.maximum(b_var.value, 0.0)
        except Exception as e:
            last_err = e
            continue
    
    return f.copy()


def lp_step(f, M, row_idx, col_idx, kj_pairs, t_values=None):
    """One LP iteration: solve LP, interpolate, return best trial."""
    n = len(f)
    sum_f = f.sum()
    b_opt = solve_lp(f, M, row_idx, col_idx, kj_pairs)
    s_opt = b_opt.sum()
    if s_opt < 1e-10:
        return f.copy(), score_from_a(f)
    
    g = b_opt * (sum_f / s_opt)
    best_trial = f.copy()
    best_sc = score_from_a(f)
    
    if t_values is None:
        t_values = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    
    for t in t_values:
        f_trial = (1.0 - t) * f + t * g
        f_trial = np.maximum(f_trial, 0.0)
        st = f_trial.sum()
        if st < 1e-10:
            continue
        f_trial = f_trial / st
        sc = score_from_a(f_trial)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_trial = f_trial.copy()
    
    return best_trial, best_sc


def mirror_descent_topk(a_init, time_limit, T_start=0.005, T_end=0.0005,
                        eta_start=0.02, eta_end=0.0005, top_k=5,
                        momentum=0.85, max_iters=50000):
    """Mirror descent with top-K peak subgradient and Nesterov momentum."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    best_iter = 0
    patience = 0
    patience_limit = 5000
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(a)
        cur_sc = 2.0 * n * b.max()
        
        if cur_sc < best_score - 1e-15:
            best_score = cur_sc
            best_a = a.copy()
            best_iter = it
            patience = 0
        else:
            patience += 1
        
        if patience > patience_limit:
            break
        
        # Top-K peak subgradient
        top_indices = np.argsort(b)[::-1][:top_k]
        peak_vals = b[top_indices]
        weights = np.exp(peak_vals / T - peak_vals[0] / T)
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_indices):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w_k * 2.0 * a[k - idx]
        
        # Tangent space projection
        grad = grad - np.dot(a, grad)
        
        # Nesterov momentum update
        v = momentum * v + grad
        log_update = -eta * v
        log_update = np.clip(log_update, -4.0, 4.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    return best_a, best_score


def coordinate_descent_polish(a_init, time_limit=30.0):
    """Coordinate descent focusing on peak-contributing coordinates."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    factors = [0.4, 0.6, 0.75, 0.85, 0.92, 1.08, 1.15, 1.3, 1.6]
    no_improve = 0
    max_no_improve = 30
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved = False
        
        # Focus on coordinates that contribute to the peak
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        peak_coords = set(range(js, je + 1))
        
        # Also include top contributing coordinates
        contribs = np.zeros(n)
        for k in np.argsort(b)[::-1][:3]:
            k = int(k)
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            idx = np.arange(js2, je2 + 1)
            contribs[idx] += b[k] * a[idx]
        top_coords = set(np.argsort(contribs)[::-1][:20])
        
        focus_coords = list(peak_coords | top_coords)
        
        rng = np.random.default_rng(hash(str(int(time.monotonic()))) % 2**31)
        rng.shuffle(focus_coords)
        
        for i in focus_coords:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            for f_fac in factors:
                a_trial = a.copy()
                a_trial[i] *= f_fac
                if a_trial[i] < 0:
                    continue
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
        
        if not improved:
            no_improve += 1
    
    return best_a, best_score


def fourier_perturb(a, eps=0.05):
    """Add structured perturbation in Fourier domain."""
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
    noise = rng.standard_normal(A.shape) * eps
    noise[0] = 0  # preserve DC
    A_pert = A + noise
    a_pert = irfft(A_pert, fft_size)[:n]
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s
