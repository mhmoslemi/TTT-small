Novel spectral-iterative approach to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral relaxation, multiplicative weight updates, and anti-peak initialization.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def spectral_relaxation(a0, time_limit, alpha_init=0.3, naming="sr"):
    """Spectral relaxation: iterate between time and frequency domains to flatten auto-convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    fft_size = 1 << (2 * n - 2).bit_length()
    A = fft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    
    # Target magnitude (flat spectrum)
    target_mag = np.full_like(mag, np.sqrt(np.mean(mag**2)))
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    alpha = alpha_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Interpolate magnitude toward flat
        mag_new = (1 - alpha) * mag + alpha * target_mag
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            # Recompute spectrum for next iteration
            A = fft(a_new, fft_size)
            phase = np.angle(A)
            mag = np.abs(A)
            target_mag = np.full_like(mag, np.sqrt(np.mean(mag**2)))
        else:
            # Revert and reduce step size
            mag = (1 - alpha) * mag + alpha * target_mag
            # Revert to previous
            A = fft(a, fft_size)
            phase = np.angle(A)
            mag = np.abs(A)
            alpha *= 0.8
        
        # Slowly decrease alpha
        alpha = max(alpha * 0.9995, 0.01)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpectRelax it={it} ({elapsed:.0f}s): best={best_score:.6f} alpha={alpha:.4f}")
    
    return best_a, best_score

"""
Novel smoothed-gradient (log-sum-exp) approach to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses temperature-annealed log-sum-exp smoothing with projected gradient descent,
multiplicative weight updates, spectral diversification, and coordinate descent polishing.
Key innovation: working with a smooth surrogate (LSE) of the max function allows
efficient gradient-based optimization with proper step-size control, combined with
adaptive temperature scheduling and diverse restarts.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_lse_gradient(a, t):
    """
    Compute gradient of 2n * LSE_t(conv(a,a)) with respect to a.
    Returns (gradient, lse_value, b).
    """
    n = len(a)
    b = fast_convolve(a)
    # Numerically stable softmax
    tb = t * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    p = exp_tb / exp_tb.sum()
    # q_i = sum_j a_j * p_{i+j} = correlate(p, a, mode='valid')
    q = np.correlate(p, a, mode='valid')
    grad = 4.0 * n * q
    # LSE value
    lse_val = (tb_max + np.log(exp_tb.sum())) / t
    return grad, lse_val, b, p


def smoothed_grad_descent(a0, t_init, t_end, n_steps, time_limit, lr_init=0.005, momentum=0.9, verbose=False):
    """
    Projected gradient descent on 2n * LSE_t(conv(a,a)) with annealing of t.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_lse = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum
    lr = lr_init
    no_improve = 0
    it = 0

    while it < n_steps and (time.monotonic() - t0) < time_limit:
        it += 1
        # Anneal temperature
        frac = min(1.0, it / max(n_steps, 1))
        t_cur = t_init * (t_end / t_init) ** frac

        grad, lse_val, b, p = compute_lse_gradient(a, t_cur)
        obj = 2.0 * n * lse_val

        # Momentum update
        v = momentum * v + grad
        # Adaptive step size
        grad_norm = np.linalg.norm(v)
        if grad_norm > 1e-15:
            step = lr * v / grad_norm
        else:
            step = np.zeros(n)

        a_new = simplex_project(a - step)

        # Check actual (non-smoothed) score
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_lse - 1e-14:
            best_lse = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Reduce learning rate if stuck
        if no_improve > 80:
            lr *= 0.8
            no_improve = 0
        if no_improve > 400:
            # Restart from best with noise
            noise = np.random.default_rng(it * 31 + 7).standard_normal(n) * 0.02
            a = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0

        lr *= 0.9998

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_lse:.6f} t={t_cur:.1f} lr={lr:.5f}")

    return best_a, best_lse


def multiplicative_weights(a0, time_limit, lr_init=0.02, verbose=False):
    """
    Multiplicative weight update (exponential gradient) on the simplex.
    a_i(t+1) ∝ a_i(t) * exp(-lr * grad_i(t))
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while (time.monotonic() - t0) < time_limit:
        it += 1
        b = fast_convolve(a)
        # Softmax weights (moderate temperature)
        t_temp = 10.0 / b.max() if b.max() > 1e-15 else 1.0
        tb = t_temp * b
        tb_max = tb.max()
        p = np.exp(tb - tb_max)
        p = p / p.sum()

        q = np.correlate(p, a, mode='valid')

        # Multiplicative update
        a_new = a * np.exp(-lr * q)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.999

        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
        if no_improve > 800:
            break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.5f}")

    return best_a, best_score


def spectral_diversify(a0, time_limit, n_restarts=5):
    """
    Use spectral methods to generate diverse candidates near a0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(999)

    for restart in range(n_restarts):
        if (time.monotonic() - t0) >= time_limit:
            break

        # Random spectral perturbation
        A = fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Perturb magnitudes
        perturb_mag = 0.1 + 0.3 * rng.random()
        mag_new = mag * (1.0 + perturb_mag * rng.standard_normal(fft_size + 1))
        mag_new = np.maximum(mag_new, 0)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

    return best_a, best_score

def init_bimodal(n, w, rng):
    """Bimodal initialization: high at edges, low in middle."""
    a = np.zeros(n)
    q1 = n // 4
    q3 = 3 * n // 4
    a[:q1] = w
    a[q3:] = w
    a[q1:q3] = 1.0 - w
    # Add some noise
    a = a + rng.standard_normal(n) * 0.05 * w
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_fourier(n, rng, n_harm=None):
    """Fourier-based initialization."""
    if n_harm is None:
        n_harm = int(rng.integers(3, 15))
    amplitudes = rng.random(n_harm) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_trimodal(n, rng):
    """Trimodal initialization."""
    a = np.zeros(n)
    third = n // 3
    amps = rng.random(3) + 0.3

    s = a.sum()
    if s < 1e-10:

    else:

    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Local evaluation (same formula as evaluate_sequence but avoids function call overhead)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * b.max()


def propose_candidate():
    """
    Multi-phase optimization using smoothed gradient descent (log-sum-exp),
    multiplicative weights, spectral diversification, and coordinate descent.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Generate diverse candidates at multiple lengths ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t1_start = time.monotonic()
    t_phase1 = 120.0

    n_values = [512, 640, 768, 896, 1024, 1280, 1536]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values:
        if (time.monotonic() - t1_start) > t_phase1:
            break
        # Generate several inits for this n
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_bimodal(n, 0.7, rng))
        inits.append(init_bimodal(n, 0.5, rng))
        inits.append(init_fourier(n, rng))
        inits.append(init_trimodal(n, rng))
        inits.append(init_bimodal(n, 0.8, rng))

        for init_a in inits:
            if (time.monotonic() - t1_start) > t_phase1:
                break
            # Quick smoothed GD to get a starting point
            t_bd = deadline - time.monotonic()
            alloc = min(8.0, t_bd * 0.02)
            if alloc < 2:
                break
            a_opt, s_opt = smoothed_grad_descent(
                init_a, t_init=5.0, t_end=50.0,
                n_steps=200, time_limit=alloc,
                lr_init=0.008, momentum=0.85
            )
            # Evaluate with actual function
            actual_score = evaluate_local(a_opt)
            candidates.append((actual_score, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best: {candidates[0][0]:.6f} (n={candidates[0][2]})")

    # Update best
    for s_c, a_c, n_c in candidates[:5]:
        if s_c < best_score:
            actual = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_seq = a_c.copy()
                print(f"  NEW BEST [p1]: {best_score:.6f} (n={n_c})")

    # === Phase 2: Main optimization - Smoothed GD with annealing ===
    print(f"\n=== Phase 2: Smoothed Gradient Descent (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    t_phase2 = 600.0

    # Focus on top 3 candidate lengths
    seen_lengths = set()
    focus_candidates = []
    for s_c, a_c, n_c in candidates:
        if n_c not in seen_lengths:
            seen_lengths.add(n_c)
            focus_candidates.append((s_c, a_c, n_c))
        if len(focus_candidates) >= 3:
            break

    # Also ensure we try the parent length
    if parent is not None and len(parent) not in seen_lengths:
        focus_candidates.append((best_score, parent, len(parent)))

    time_per_candidate = t_phase2 // max(len(focus_candidates), 1)

    for s_c, a_c, n_c in focus_candidates:
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        t_alloc = min(time_per_candidate, rem * 0.9)
        if t_alloc < 20:
            continue

        print(f"  SGD n={n_c}, budget={t_alloc:.0f}s, init_score={s_c:.6f}")

        # Multiple t schedules
        t_schedules = [
            (3.0, 30.0),    # mild annealing
            (5.0, 80.0),    # stronger annealing
            (2.0, 200.0),   # very strong annealing
            (10.0, 50.0),   # start hot
        ]

        best_local = a_c.copy()
        best_local_score = s_c

        for ti, te in t_schedules:
            rem = deadline - time.monotonic() - 400
            if rem < 15:
                break
            t_this = min(t_alloc * 0.25, rem * 0.8)
            a_opt, s_opt = smoothed_grad_descent(
                best_local, t_init=ti, t_end=te,
                n_steps=5000, time_limit=t_this,
                lr_init=0.006, momentum=0.9, verbose=False
            )
            if s_opt < best_local_score:
                best_local = a_opt
                best_local_score = s_opt

        # Follow with MWU
        rem = deadline - time.monotonic() - 350
        if rem > 15:
            t_mwu = min(60.0, rem * 0.3)
            a_mwu, s_mwu = multiplicative_weights(best_local, t_mwu, lr_init=0.01)
            if s_mwu < best_local_score:
                best_local = a_mwu
                best_local_score = s_mwu

        actual = float(evaluate_sequence(best_local.tolist()))
        if np.isfinite(actual) and actual < best_score:
            best_score = actual
            best_seq = best_local.copy()
            print(f"  NEW BEST [p2_n{n_c}]: {best_score:.6f}")

    # === Phase 3: Spectral diversification ===
    print(f"\n=== Phase 3: Spectral Diversification (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_spec = min(60.0, rem * 0.25)
            a_spec, s_spec = spectral_diversify(best_seq, t_spec, n_restarts=20)
            if s_spec < best_score:
                actual = float(evaluate_sequence(a_spec.tolist()))
                if np.isfinite(actual):
                    best_score = actual
                    best_seq = a_spec.copy()
                    print(f"  NEW BEST [spectral]: {best_score:.6f}")

            # Try spectral from different inits
            n_best = len(best_seq)
            for w in [0.6, 0.75, 0.9]:
                rem = deadline - time.monotonic() - 250
                if rem < 15:
                    break
                a_bi = init_bimodal(n_best, w, rng)
                a_opt, s_opt = smoothed_grad_descent(
                    a_bi, t_init=5.0, t_end=100.0,
                    n_steps=2000, time_limit=min(40, rem * 0.3),
                    lr_init=0.007, momentum=0.9
                )
                actual = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [bimodal_w{w}]: {best_score:.6f}")

    # === Phase 4: Intensive polishing ===
    print(f"\n=== Phase 4: Intensive Polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_cd = min(120.0, rem * 0.4)
            print(f"  CD n={n_best}, budget={t_cd:.0f}s")
            a_cd, s_cd = coord_descent_polish(best_seq, t_cd)
            if s_cd < best_score:
                actual = float(evaluate_sequence(a_cd.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best_seq = a_cd.copy()
                    print(f"  NEW BEST [cd]: {best_score:.6f}")

        # Peak directed search
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_ps = min(60.0, rem * 0.4)
            print(f"  PeakSearch budget={t_ps:.0f}s")
            a_ps, s_ps = peak_directed_search(best_seq, t_ps)
            if s_ps < best_score:
                actual = float(evaluate_sequence(a_ps.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best_seq = a_ps.copy()
                    print(f"  NEW BEST [peak]: {best_score:.6f}")

        # Another round of SGD from the polished result
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_sg2 = min(80.0, rem * 0.4)
            print(f"  SGD2 budget={t_sg2:.0f}s")
            a_sg2, s_sg2 = smoothed_grad_descent(
                best_seq, t_init=20.0, t_end=500.0,
                n_steps=5000, time_limit=t_sg2,
                lr_init=0.003, momentum=0.92
            )
            if s_sg2 < best_score:
                actual = float(evaluate_sequence(a_sg2.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best_seq = a_sg2.copy()
                    print(f"  NEW BEST [sgd2]: {best_score:.6f}")

        # MWU from polished
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_mwu2 = min(40.0, rem * 0.4)
            a_mwu2, s_mwu2 = multiplicative_weights(best_seq, t_mwu2, lr_init=0.005)
            if s_mwu2 < best_score:
                actual = float(evaluate_sequence(a_mwu2.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best_seq = a_mwu2.copy()
                    print(f"  NEW BEST [mwu2]: {best_score:.6f}")

    # === Phase 5: Try neighboring lengths ===
    print(f"\n=== Phase 5: Neighboring Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        neighbors = []
        for delta in [64, 128, -64, -128, 192, -192]:
            alt_n = n_best + delta
            if 256 <= alt_n <= 2560:
                neighbors.append(alt_n)

        for alt_n in neighbors[:3]:
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            init_alt = interpolate_seq(best_seq, alt_n)
            t_alt = min(60.0, rem * 0.4)
            a_alt, s_alt = smoothed_grad_descent(
                init_alt, t_init=5.0, t_end=80.0,
                n_steps=3000, time_limit=t_alt,
                lr_init=0.006, momentum=0.9
            )
            actual = float(evaluate_sequence(a_alt.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_seq = a_alt.copy()
                print(f"  NEW BEST [n{alt_n}]: {best_score:.6f}")

            # Quick CD from there
            rem = deadline - time.monotonic() - 50
            if rem > 10 and actual < best_score + 0.01:
                a_cd_alt, s_cd_alt = coord_descent_polish(a_alt, min(20, rem * 0.4))
                actual2 = float(evaluate_sequence(a_cd_alt.tolist()))
                if np.isfinite(actual2) and actual2 < best_score:
                    best_score = actual2
                    best_seq = a_cd_alt.copy()
                    print(f"  NEW BEST [n{alt_n}_cd]: {best_score:.6f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Multiple short CD rounds
        for round_i in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 5:
                break
            t_cd = min(30.0, rem * 0.3)
            a_fin, s_fin = coord_descent_polish(best_seq, t_cd)
            actual = float(evaluate_sequence(a_fin.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_seq = a_fin.copy()
                print(f"  FINAL cd{round_i}: {best_score:.6f}")

        # Peak search on final
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_fin2, s_fin2 = peak_directed_search(best_seq, min(15, rem * 0.5))
            actual = float(evaluate_sequence(a_fin2.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_seq = a_fin2.copy()
                print(f"  FINAL peak: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
