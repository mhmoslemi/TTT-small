"""
Hybrid optimizer for the nonnegative linear autocorrelation problem.

Minimize:
    2 * n * max_k conv(a, a)_k / (sum(a))^2

The code combines:
- FFT-based fast autocorrelation and gradients.
- Smooth p-norm log-sum-exp style optimization with L-BFGS on the simplex.
- Exact pair-transfer coordinate descent with local 1D line search.
- Thermal minimax-style alternating dynamics.
- LP fictitious play using scipy HiGHS when available.
- Length exploration around the supplied parent sequence.

The global height_sequence_1 is used as the starting point when available.
"""

import numpy as np
import time

try:
    from scipy.fft import rfft as _rfft, irfft as _irfft
    _HAS_RFFT = True
except Exception:
    _HAS_RFFT = False

try:
    from scipy.optimize import minimize as _minimize
    _HAS_MINIMIZE = True
except Exception:
    _HAS_MINIMIZE = False

try:
    from scipy.optimize import linprog as _linprog
    _HAS_LINPROG = True
except Exception:
    _HAS_LINPROG = False

# Globals used by the L-BFGS callback.
_OPT_N = 0
_OPT_P = 2.0
_OPT_SCALE = 1.0

_PAIR_FRACTIONS = (0.01, 0.025, 0.05, 0.09, 0.15, 0.25, 0.40, 0.60, 0.80, 0.97)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(512, dtype=np.float64) / 512.0
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _local_score(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return float('inf')
    c = _fast_convolve(a)
    return float(2.0 * n * c.max())


def _softmax(w):
    w = np.asarray(w, dtype=np.float64).reshape(-1)
    if w.size == 0:
        return w
    if not np.all(np.isfinite(w)):
        return np.ones(w.size, dtype=np.float64) / float(w.size)
    w = w - float(np.max(w))
    a = np.exp(w)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _fast_convolve(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = 2 * n - 1
    if not _HAS_RFFT or n < 64:
        c = np.convolve(a, a, mode='full')
    else:
        size = 1 << (m - 1).bit_length()
        A = _rfft(a, size)
        c = _irfft(A * A, size)[:m]
    c = np.real_if_close(c, tol=1000)
    c = np.asarray(c, dtype=np.float64)
    # True convolution is nonnegative for nonnegative input; clip roundoff.
    c = np.maximum(c, 0.0)
    return c


def _fast_grad_cross(weights, a):
    """
    For weights w_k, k=0..2n-2, and vector a_i, compute

        g_i = 2 * sum_{j=0}^{n-1} w_{i+j} a_j

    which is the gradient contribution of sum_k w_k conv(a,a)_k.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    weights = np.asarray(weights, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if not _HAS_RFFT or n < 64:
        g = np.empty(n, dtype=np.float64)
        for i in range(n):
            g[i] = 2.0 * float(np.dot(weights[i:i + n], a))
        return np.maximum(g, 0.0)

    L = 4 * n - 2
    size = 1 << (L - 1).bit_length()
    wr = weights[::-1]
    A = _rfft(a, size)
    B = _rfft(wr, size)
    conv = _irfft(A * B, size)[:L]
    conv = np.real_if_close(conv, tol=1000)
    conv = np.asarray(conv, dtype=np.float64)
    h = conv[n - 1:2 * n - 1][::-1]
    g = 2.0 * h
    return np.maximum(g, 0.0)


def _pnorm_objective(w):
    n = int(_OPT_N)
    p = float(_OPT_P)
    scale = float(_OPT_SCALE)
    if n <= 0:
        return 1e100, np.zeros(0, dtype=np.float64)

    w = np.asarray(w, dtype=np.float64).reshape(-1)
    if w.size != n:
        w = w[:n] if w.size > n else np.pad(w, (0, n - w.size), mode='edge')
    if not np.all(np.isfinite(w)):
        return 1e100, np.zeros(n, dtype=np.float64)

    a = _softmax(w)
    c = _fast_convolve(a)
    if c.size != 2 * n - 1:
        return 1e100, np.zeros(n, dtype=np.float64)

    if p <= 1.0:
        f = 1.0
        ga = np.zeros(n, dtype=np.float64)
    else:
        pos = c > 0.0
        if not np.any(pos):
            return 1e100, np.zeros(n, dtype=np.float64)

        log_c = np.full(c.shape, -np.inf, dtype=np.float64)
        log_c[pos] = np.log(c[pos])
        log_cp = p * log_c
        mcp = float(np.max(log_cp))

        if not np.isfinite(mcp):
            return 1e100, np.zeros(n, dtype=np.float64)

        with np.errstate(over='ignore', under='ignore', invalid='ignore'):
            exp_diff = np.exp(log_cp - mcp)
        sum_exp = float(np.sum(exp_diff))
        if not np.isfinite(sum_exp) or sum_exp <= 0.0:
            return 1e100, np.zeros(n, dtype=np.float64)

        log_sum_cp = mcp + np.log(sum_exp)

        with np.errstate(over='ignore', under='ignore', invalid='ignore'):
            f_val = np.exp(log_sum_cp / p)

        log_gamma = (1.0 / p - 1.0) * log_sum_cp
        weights = np.zeros_like(c, dtype=np.float64)

        log_w = (p - 1.0) * log_c[pos] + log_gamma
        if log_w.size == 0:
            return 1e100, np.zeros(n, dtype=np.float64)
        lw_max = float(np.max(log_w))
        if not np.isfinite(lw_max):
            return 1e100, np.zeros(n, dtype=np.float64)

        # Scale for numerical safety. In normal operation lw_max is near zero.
        safe_max = max(-700.0, min(700.0, lw_max))
        with np.errstate(over='ignore', under='ignore', invalid='ignore'):
            wpos = np.exp(log_w - safe_max) * np.exp(safe_max)
        weights[pos] = wpos

        if not np.all(np.isfinite(weights)):
            weights = np.where(np.isfinite(weights), weights, 0.0)

        ga = _fast_grad_cross(weights, a)
        if not np.all(np.isfinite(ga)):
            ga = np.where(np.isfinite(ga), ga, 0.0)

        f = float(f_val)
        if not np.isfinite(f):
            f = float(c.max())

    dot = float(np.dot(ga, a))
    gw = a * (ga - dot)
    gw = np.where(np.isfinite(gw), gw, 0.0)
    return float(f * scale), gw * scale


def _pnorm_polish(a, p, maxiter):
    if not _HAS_MINIMIZE:
        return a
    a = _normalize(a)
    n = a.size
    if n == 0:
        return a
    global _OPT_N, _OPT_P, _OPT_SCALE
    _OPT_N = n
    _OPT_P = float(p)
    _OPT_SCALE = 1000.0

    a_safe = np.maximum(a, 1e-14)
    w = np.log(a_safe / float(a_safe.sum()))
    w -= float(np.mean(w))

    try:
        res = _minimize(
            _pnorm_objective,
            w,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': int(max(maxiter, 1)),
                'ftol': 1e-12,
                'gtol': 1e-7,
                'maxcor': 30,
                'maxls': 50,
                'iprint': 0
            }
        )
        if res is not None and hasattr(res, 'x') and res.x is not None:
            a2 = _normalize(np.asarray(res.x, dtype=np.float64))
            return a2
    except Exception:
        pass
    return a


def _pnorm_multi(a, time_budget, deadline, seed=1, p_list=(8, 16, 32)):
    if not _HAS_MINIMIZE or time_budget <= 0:
        return a
    a = _normalize(a)
    n = a.size
    best = a.copy()
    best_score = _local_score(best)

    # Rough timing for one objective/gradient call.
    global _OPT_N, _OPT_P, _OPT_SCALE
    _OPT_N = n
    _OPT_P = 16.0
    _OPT_SCALE = 1.0
    w = np.log(np.maximum(a, 1e-14))
    w -= float(np.mean(w))
    t0 = time.monotonic()
    try:
        _pnorm_objective(w)
        dt = max(time.monotonic() - t0, 1e-4)
    except Exception:
        dt = 0.05
    est_iter = max(0.005, dt * 15.0)

    rng = np.random.default_rng(seed)
    per = max(1.0, time_budget / float(len(p_list)))

    for p in p_list:
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        budget = min(per, rem - 15.0)
        if budget <= 0.0:
            continue
        maxiter = int(budget / est_iter)
        maxiter = max(30, min(int(maxiter), 600))

        a2 = _pnorm_polish(best, p, maxiter)
        s2 = _local_score(a2)
        if s2 < best_score - 1e-12:
            best_score = s2
            best = a2.copy()

        # Optional small-noise restart at this p.
        rem = deadline - time.monotonic()
        if rem > 35:
            noise = rng.standard_normal(n) * (0.03 / np.sqrt(n))
            a3 = _normalize(best + noise)
            p2 = max(8.0, 0.75 * float(p))
            maxiter2 = max(20, maxiter // 3)
            a3 = _pnorm_polish(a3, p2, maxiter2)
            s3 = _local_score(a3)
            if s3 < best_score - 1e-12:
                best_score = s3
                best = a3.copy()

        print(f"    pnorm stage p={p} local={best_score:.8f}")

    return best


def _pair_eval(L, c, q1, q2, q3, delta):
    val = c + delta * L
    d2 = delta * delta
    val[q1] += d2
    val[q2] += d2
    val[q3] -= 2.0 * d2
    return float(val.max())


def _pair_line_search(a, c, i, j, max_delta):
    """
    Minimize max conv(a',a') over a_i += delta, a_j -= delta, delta >= 0.
    """
    n = a.size
    cur = float(c.max())
    if max_delta <= 1e-14:
        return 0.0, cur
    if i == j:
        return 0.0, cur

    U = min(float(max_delta), 0.999999 * float(a[j]))
    if U <= 1e-14:
        return 0.0, cur

    m = 2 * n - 1
    L = np.zeros(m, dtype=np.float64)
    L[i:i + n] += 2.0 * a
    L[j:j + n] -= 2.0 * a

    q1 = 2 * i
    q2 = 2 * j
    q3 = i + j

    pts_x = [0.0]
    pts_y = [cur]
    best_delta = 0.0
    best_val = cur
    best_idx = 0

    for idx, fr in enumerate(_PAIR_FRACTIONS, start=1):
        d = fr * U
        val = _pair_eval(L, c, q1, q2, q3, d)
        pts_x.append(d)
        pts_y.append(val)
        if val < best_val - 1e-18:
            best_val = val
            best_delta = d
            best_idx = idx

    # Quadratic interpolation around the best sampled point.
    if 0 < best_idx < len(pts_x) - 1:
        x0 = pts_x[best_idx - 1]
        y0 = pts_y[best_idx - 1]
        x1 = pts_x[best_idx]
        y1 = pts_y[best_idx]
        x2 = pts_x[best_idx + 1]
        y2 = pts_y[best_idx + 1]
        if abs(x1 - x0) > 1e-30 and abs(x2 - x1) > 1e-30 and abs(x2 - x0) > 1e-30:
            a2 = ((y2 - y0) / (x2 - x0) - (y1 - y0) / (x1 - x0)) / (x2 - x1)
            b1 = (y1 - y0) / (x1 - x0) - a2 * (x1 + x0)
            if abs(a2) > 1e-30:
                xv = -b1 / (2.0 * a2)
                if 0.0 < xv < U:
                    fv = _pair_eval(L, c, q1, q2, q3, xv)
                    if fv < best_val - 1e-18:
                        best_val = fv
                        best_delta = xv

    return best_delta, best_val


def _pair_descent(a, time_limit, max_rounds=2000, low_count=12, high_count=12, max_tries=6000):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return a
    c = _fast_convolve(a)
    best = a.copy()
    best_score = float(2.0 * n * c.max())
    t_end = time.monotonic() + time_limit
    round_no = 0
    tries = 0
    no_improve = 0
    perturbations = 0

    while time.monotonic() < t_end and round_no < max_rounds and tries < max_tries:
        round_no += 1
        c = np.maximum(c, 0.0)
        peg = float(c.max())
        temp = max(peg * 0.10, 1e-15)

        u = np.exp(-np.maximum(peg - c, 0.0) / temp)
        su = float(u.sum())
        if su <= 0.0 or not np.isfinite(su):
            u = np.zeros(c.size, dtype=np.float64)
            u[int(np.argmax(c))] = 1.0
        else:
            u /= su

        D = 0.5 * _fast_grad_cross(u, a)
        low_idx = np.argsort(D)[:low_count]
        high_idx = np.argsort(D)[::-1][:high_count]

        accepted = False
        for j in high_idx:
            if a[j] <= 1e-12:
                continue
            U = min(float(a[j]), max(0.75 / float(n), 0.35 * float(a[j])))
            if U <= 1e-13:
                continue
            for i in low_idx:
                i = int(i)
                j = int(j)
                if i == j:
                    continue
                tries += 1
                if time.monotonic() >= t_end or tries >= max_tries:
                    break
                if D[j] - D[i] <= 1e-12:
                    continue
                delta, f = _pair_line_search(a, c, i, j, U)
                if delta <= 0.0 or f >= peg - 1e-15:
                    continue

                old_a = a.copy()
                old_c = c.copy()
                old_peg = peg

                a_try = old_a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                a_try = np.maximum(a_try, 0.0)
                s = float(a_try.sum())
                if s <= 1e-15:
                    a = old_a
                    c = old_c
                    break
                a_try /= s
                c_try = _fast_convolve(a_try)
                peg_try = float(c_try.max())

                if peg_try < old_peg - 1e-14:
                    a = a_try
                    c = c_try
                    score_try = float(2.0 * n * peg_try)
                    if score_try < best_score - 1e-12:
                        best_score = score_try
                        best = a.copy()
                    no_improve = 0
                    accepted = True
                    break
                else:
                    a = old_a
                    c = old_c
            if accepted or tries >= max_tries or time.monotonic() >= t_end:
                break

        if not accepted:
            no_improve += 1
            if no_improve > 8:
                if perturbations < 5 and (t_end - time.monotonic()) > 20.0:
                    rngp = np.random.default_rng(12345 + round_no)
                    noise = rngp.standard_normal(n) * (0.02 / np.sqrt(n))
                    a = _normalize(a + noise)
                    c = _fast_convolve(a)
                    perturbations += 1
                    no_improve = 0
                    continue
                break

    return best


def _thermal_alternate(a, time_limit, max_iter=20000, beta=8.0, temp_frac=0.10):
    a = _normalize(a)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a
    a_cur = a.copy()
    avg = a.copy()
    best = a.copy()
    best_score = _local_score(best)
    t_end = time.monotonic() + time_limit
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() >= t_end:
            break
        c = _fast_convolve(a_cur)
        peg = float(c.max())
        temp = max(peg * float(temp_frac), 1e-15)

        u = np.exp(-np.maximum(peg - c, 0.0) / temp)
        su = float(u.sum())
        if su <= 0.0 or not np.isfinite(su):
            u = np.zeros(c.size, dtype=np.float64)
            u[int(np.argmax(c))] = 1.0
        else:
            u /= su

        D = 0.5 * _fast_grad_cross(u, a_cur)
        d_scale = float(D.max()) + 1e-15
        a_new = np.exp(-float(beta) * D / d_scale)
        sn = float(a_new.sum())
        if sn <= 0.0 or not np.isfinite(sn):
            a_new = a_cur
        else:
            a_new = a_new / sn

        lam = min(1.0, 2.0 / float(it + 1))
        a_cur = (1.0 - lam) * a_cur + lam * a_new
        a_cur = _normalize(a_cur)
        avg = float(it) * avg + a_cur
        avg /= float(it + 1)

        sc_cur = float(2.0 * n * float(c.max()))
        # c belongs to previous a_cur; reusing it is only a lower-cost proxy.
        # Compute accurate current score occasionally or when close to best.
        if sc_cur < best_score + 1e-7 or it % 20 == 0:
            sc_cur = _local_score(a_cur)
        if sc_cur < best_score - 1e-12:
            best_score = sc_cur
            best = a_cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 50 == 0:
            sc_avg = _local_score(avg)
            if sc_avg < best_score - 1e-12:
                best_score = sc_avg
                best = avg.copy()
                no_improve = 0

        if no_improve > 4000:
            break

    return best


def _lp_precompute(n):
    m = 2 * n - 1
    mcol = -np.ones((m, 1), dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * n + [(0.0, None)]
    obj = np.zeros(n + 1, dtype=np.float64)
    obj[n] = 1.0
    b_ub = np.zeros(m, dtype=np.float64)
    return mcol, A_eq, b_eq, bounds, obj, b_ub


def _solve_best_response(a, lp_data):
    if not _HAS_LINPROG:
        return None, None
    a = _normalize(a)
    n = a.size
    m, A_eq, b_eq, bounds, obj, b_ub = lp_data
    T = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = a
    A_ub = np.hstack((T, mcol))

    res = None
    try:
        res = _linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs')
    except Exception:
        try:
            res = _linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds)
        except Exception:
            res = None

    if res is None:
        return None, None
    x = getattr(res, 'x', None)
    if x is None:
        return None, None
    x = np.asarray(x, dtype=np.float64)
    if x.size < n + 1 or not np.all(np.isfinite(x)):
        return None, None

    b = x[:n]
    b = np.maximum(b, 0.0)
    sb = float(b.sum())
    if sb <= 1e-15:
        return None, None
    b /= sb
    M = float(x[n])
    if not np.isfinite(M) or M < 0.0:
        M = 0.0
    return b, M

def _interpolate_seq(f, new_n):
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    new_n = int(max(1, new_n))
    if f.size == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if f.size == new_n:
        return _normalize(f)
    s = float(f.sum())
    if s <= 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    fn = f / s
    old_x = np.linspace(0.0, 1.0, f.size, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    out = np.interp(new_x, old_x, fn)
    return _normalize(out)


def _init_fourier(n, rng):
    n = int(max(16, n))
    n_h = int(rng.integers(6, 24))
    amps = rng.random(n_h) * 0.35
    phases = rng.uniform(0.0, 2.0 * np.pi, n_h)
    freqs = np.arange(1, n_h + 1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.65, dtype=np.float64)
    for j in range(n_h):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / float(n) + phases[j])
    return _normalize(a)


def _quick_optimize(a, time_budget, deadline, seed=3, p_list=(16, 32)):
    a = _normalize(a)
    rem = deadline - time.monotonic()
    t1 = min(0.35 * time_budget, rem - 5.0)
    if t1 > 5.0:
        a = _thermal_alternate(a, t1, max_iter=5000, beta=8.0, temp_frac=0.12)

    rem = deadline - time.monotonic()
    if _HAS_MINIMIZE:
        t2 = min(0.40 * time_budget, rem - 10.0)
        if t2 > 10.0:
            a = _pnorm_multi(a, t2, deadline, seed=seed, p_list=p_list)

    rem = deadline - time.monotonic()
    t3 = min(0.25 * time_budget, rem - 3.0)
    if t3 > 5.0:
        a = _pair_descent(a, t3, max_rounds=250, low_count=8, high_count=8, max_tries=1500)

    return a


def _check_update(cand, best, best_score, label=""):
    old_best_score = float(best_score)
    a = _normalize(cand)
    if a.size == 0:
        return best, best_score

    s_loc = _local_score(a)
    if s_loc < float(best_score) - 1e-9:
        best = a.copy()
        best_score = float(s_loc)
        try:
            s_eval = float(evaluate_sequence([float(x) for x in a]))
            if np.isfinite(s_eval):
                # Use the official evaluator when it confirms or improves.
                if s_eval < float(best_score) + 1e-9:
                    best_score = min(float(best_score), s_eval)
                if float(best_score) < old_best_score - 1e-7:
                    print(f"  NEW BEST [{label}] score={float(best_score):.8f} n={a.size}")
    else:
        # Even if not better than current local best, make sure bounds are finite.
        pass

    # Try simple reversal symmetrization.
    if a.size > 2:
        sym = 0.5 * (a + a[::-1])
        sym = _normalize(sym)
        ss_loc = _local_score(sym)
        if ss_loc < float(best_score) - 1e-9:
            best = sym.copy()
            best_score = float(ss_loc)
            try:
                s_eval = float(evaluate_sequence([float(x) for x in sym]))
                if np.isfinite(s_eval):
                    if s_eval < float(best_score) + 1e-9:
                        best_score = min(float(best_score), s_eval)
                    if float(best_score) < old_best_score - 1e-7:
                        print(f"  NEW BEST [sym_{label}] score={float(best_score):.8f} n={sym.size}")
    return best, float(best_score)

                pn = p / sp
                best = pn.copy()
                try:
                    best_score = float(evaluate_sequence([float(x) for x in pn]))
                except Exception:
                    best_score = _local_score(pn)
                print(f"Start from parent: n={pn.size}, score={best_score:.8f}")
    except Exception as e:
        print(f"No usable parent: {e}")

    if best is None:
        best = np.ones(768, dtype=np.float64) / 768.0
        best_score = 2.0
        print("Using uniform fallback n=768, score=2.0")

    best, best_score = _check_update(best, best, best_score, "initial")

    # Phase 1: exact pair-transfer local descent.
    rem = deadline - time.monotonic()
    if rem > 100.0:
        t = min(250.0, rem * 0.35)
        print(f"Phase 1: pair descent budget={t:.0f}s")
        a1 = _pair_descent(best, t, max_rounds=3000, low_count=12, high_count=12, max_tries=8000)
        best, best_score = _check_update(a1, best, best_score, "pair1")

    # Phase 2: smooth p-norm polishing.
    rem = deadline - time.monotonic()
    if rem > 200.0:
        t = min(380.0, rem * 0.35)
        print(f"Phase 2: pnorm budget={t:.0f}s")
        a2 = _pnorm_multi(best, t, deadline, seed=101, p_list=(8, 16, 32))
        best, best_score = _check_update(a2, best, best_score, "pnorm1")

    # Phase 3: pair descent again.
    rem = deadline - time.monotonic()
    if rem > 150.0:
        t = min(200.0, rem * 0.22)
        print(f"Phase 3: pair descent budget={t:.0f}s")
        a3 = _pair_descent(best, t, max_rounds=2000, low_count=14, high_count=14, max_tries=6000)
        best, best_score = _check_update(a3, best, best_score, "pair2")

    # Phase 4: thermal minimax dynamics.
    rem = deadline - time.monotonic()
    if rem > 150.0:
        t = min(240.0, rem * 0.18)
        print(f"Phase 4: thermal budget={t:.0f}s")
        a4 = _thermal_alternate(best, t, max_iter=50000, beta=8.0, temp_frac=0.10)
        best, best_score = _check_update(a4, best, best_score, "thermal")

    # Phase 5: short pair polish.
    rem = deadline - time.monotonic()
    if rem > 90.0:
        t = min(90.0, rem * 0.12)
        a5 = _pair_descent(best, t, max_rounds=1000, low_count=10, high_count=10, max_tries=2500)
        best, best_score = _check_update(a5, best, best_score, "pair3")

    # Phase 6: LP fictitious play, if time and LP solver are available.
    rem = deadline - time.monotonic()
    if rem > 220.0 and _HAS_LINPROG and len(best) <= 1200:
        t = min(300.0, rem * 0.25)
        print(f"Phase 6: LP budget={t:.0f}s")
        a6 = _lp_fictitious(best, t)
        best, best_score = _check_update(a6, best, best_score, "lp")

    # Phase 7: final p-norm sharpening.
    rem = deadline - time.monotonic()
    if rem > 160.0:
        t = min(170.0, rem * 0.20)
        a7 = _pnorm_multi(best, t, deadline, seed=4242, p_list=(32, 48))
        best, best_score = _check_update(a7, best, best_score, "pnorm2")

    # Phase 8: final pair polish.
    rem = deadline - time.monotonic()
    if rem > 80.0:
        t = min(100.0, rem * 0.18)
        a8 = _pair_descent(best, t, max_rounds=1500, low_count=16, high_count=16, max_tries=5000)
        best, best_score = _check_update(a8, best, best_score, "pair4")

    # Phase 9: explore nearby and longer lengths.
    rem = deadline - time.monotonic()
    if rem > 240.0:
        n0 = len(best)
        trial_lengths = []
        for nn in (n0 - 64, n0 + 64, n0 + 128, int(n0 * 1.25), int(n0 * 1.5)):
            nn = int(nn)
            if 256 <= nn <= 1536 and nn != n0 and nn not in trial_lengths:
                trial_lengths.append(nn)

        max_trials = 3 if rem > 420.0 else 2
        rng = np.random.default_rng(999)

        for trial_index, nn in enumerate(trial_lengths[:max_trials]):
            rem = deadline - time.monotonic()
            if rem < 110.0:
                break
            budget = min(80.0, rem * 0.35)
            print(f"Phase 9: length trial n={nn}, budget={budget:.0f}s")

            candA = _interpolate_seq(best, nn)
            candA = _quick_optimize(candA, budget, deadline, seed=1000 + trial_index, p_list=(16, 32))
            best, best_score = _check_update(candA, best, best_score, f"interp_n{nn}")

            rem = deadline - time.monotonic()
            if rem < 80.0:
                break

            candB = _init_fourier(nn, rng)
            budget2 = min(60.0, rem * 0.35)
            candB = _quick_optimize(candB, budget2, deadline, seed=2000 + trial_index, p_list=(16, 32))
            best, best_score = _check_update(candB, best, best_score, f"fourier_n{nn}")

    # Final symmetrization and verification.
    rem = deadline - time.monotonic()
    if rem > 20.0:
        best, best_score = _check_update(best, best, best_score, "final")

    best = _normalize(np.maximum(best, 0.0))
    try:
        final_score = float(evaluate_sequence([float(x) for x in best]))
        if np.isfinite(final_score) and final_score < best_score + 1e-12:
            best_score = final_score
    except Exception:
        final_score = _local_score(best)
        if np.isfinite(final_score) and final_score < best_score + 1e-12:
            best_score = final_score

    print(f"Final local score ~= {best_score:.8f}, n={best.size}")
    return [float(x) for x in best]
