SCA (Sequential Convex Approximation) based optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Key ideas:
1. Normalize sum(a)=1, minimize max_k conv(a,a)[k].
2. SCA: linearize quadratic constraints at current point, solve LP.
3. Line search to ensure descent.
4. Subgradient polish with log-sum-exp smoothing.
5. Multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(a, n):
    """Build the sparse constraint matrix for the SCA LP.
    
    Row k: 2*a^t[k-j] for valid j, 0 otherwise, -1 for M column.
    Total: (2n-1) x (n+1) sparse matrix.
    """
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        # M column
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))

    A = build_constraint_matrix(a, n)
    
    a_var = cp.Variable(n)
    M_var = cp.Variable()
    
    constraints = [
        a_var >= 0,
        cp.sum(a_var) == 1,
        A @ cp.vstack([a_var, M_var]) <= c
    ]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    # Try multiple solvers
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, **get_solver_params(solver))
            if a_var.value is not None:
                a_new = np.maximum(a_var.value, 0.0)
                s = a_new.sum()
                if s > 1e-10:
                    a_new /= s
                    return a_new, float(M_var.value)
        except Exception:
            continue
    
    return None, None

"""
SCA (Sequential Convex Approximation) optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Strategy:
1. Normalize sum(a)=1, minimize max_k conv(a,a)[k].
2. SCA: linearize quadratic constraints at current point, solve LP with all constraints.
3. Line search with backtracking to ensure descent.
4. Log-sum-exp subgradient polishing with adaptive temperature.
5. Coordinate descent for final micro-optimization.
6. Multi-length exploration and symmetric variant.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sca_matrix(a, n, k_indices=None):
    """Build sparse constraint matrix for SCA LP.
    Row k: entries 2*a[k-j] at columns j where 0<=k-j<n, and -1 at column n (for M).
    """
    if k_indices is None:
        k_indices = range(2 * n - 1)
    k_indices = list(k_indices)
    K = len(k_indices)
    rows = []
    cols = []
    vals = []
    rhs = np.zeros(K)
    for row_i, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(row_i)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        rows.append(row_i)
        cols.append(n)
        vals.append(-1.0)
        rhs[row_i] = fast_convolve(a)[k] if k_indices is None or k == k_indices[row_i] else 0.0
    # We need rhs = c[k_indices]
    # Will be set by caller
    A_mat = csr_matrix((vals, (rows, cols)), shape=(K, n + 1))
    return A_mat, rhs

    cols = []

    for k in range(2 * n - 1):

def run_sca_phase(f_init, time_limit, use_full=True, top_k_reduced=60, label="sca", verbose=True):
    """Main SCA optimization loop with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    use_full_current = use_full

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Choose which LP to solve
        a_new = None
        if use_full_current:
            a_new, _ = solve_sca_lp_full(f)
            if a_new is None:
                use_full_current = False
                a_new = solve_sca_lp_reduced(f, top_k=top_k_reduced)
        else:
            a_new = solve_sca_lp_reduced(f, top_k=top_k_reduced)
            if a_new is None and it % 10 == 0:
                # Occasional retry with full
                a_new, _ = solve_sca_lp_full(f)
                if a_new is not None:
                    use_full_current = True

        if a_new is None:
            stuck += 1
            if stuck > 30:
                break
            continue

        # Line search: try several step sizes
        best_t = 0.0
        best_t_obj = best_obj
        cur_obj = fast_convolve(f).max()

        for t in [0.3, 0.5, 0.7, 1.0, 0.15, 0.1]:
            f_trial = (1.0 - t) * f + t * a_new
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-12:
                best_t_obj = obj
                best_t = t

        if best_t == 0.0:
            # Try very small steps
            for t in [0.03, 0.01, 0.05]:
                f_trial = (1.0 - t) * f + t * a_new
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_t_obj - 1e-12:
                    best_t_obj = obj
                    best_t = t

        if best_t == 0.0:
            stuck += 1
            if stuck > 50:
                break
            continue

        # Apply step
        f = (1.0 - best_t) * f + best_t * a_new
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if verbose and it % 50 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"t={elapsed:.0f}s stuck={stuck}")

    return best_f, best_obj


def run_subgradient_phase(f_init, time_limit, lr_init=0.01, tau_init=0.001, label="sg"):
    """Log-sum-exp subgradient polishing with adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    tau = tau_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute LSE gradient
        c = fast_convolve(f)
        c_shifted = c - c.max()
        w = np.exp(c_shifted / tau)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum

        # Gradient via cross-correlation (FFT)
        fft_size = 1 << (2 * n).bit_length()
        W = rfft(w, fft_size)
        A = rfft(f, fft_size)
        corr = irfft(W * np.conj(A), fft_size)
        grad = 2.0 * corr[:n]

        # Projected subgradient step
        f_new = project_simplex(f - lr * grad)

        if it % 50 == 0:
            obj = fast_convolve(f_new).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f_new.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.05)
            else:
                no_improve += 1
                if no_improve > 10:
                    lr *= 0.9
                    no_improve = 0

            if it % 200 == 0:
                tau = min(tau * 1.1, 0.05)
                print(f"    {label} iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                      f"lr={lr:.5f} tau={tau:.5f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 30:
                # Restart with perturbation
                f_new = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                lr = lr_init
                no_improve = 0

        f = f_new

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]

    improved = True
    rounds = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        rounds += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
        if improved and rounds % 5 == 0:
            print(f"    {label} round={rounds} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")

    return f, best_obj


def try_symmetric_sca(f_init, time_limit, label="sym"):
    """SCA with symmetry constraint a[i] = a[n-1-i]."""
    n = len(f_init)
    # Symmetrize
    f_sym = (f_init + f_init[::-1]) / 2.0
    f_sym = project_simplex(f_sym)

    t0 = time.monotonic()
    best_f = f_sym.copy()
    best_obj = fast_convolve(f_sym).max()
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit and it < 2000:
        it += 1
        n_half = (n + 1) // 2
        c = fast_convolve(f_sym)

        # Build LP with symmetry: a[i] = a[n-1-i]
        # Free variables: a[0], ..., a[n_half-1]
        # a[n-1-i] = a[i]
        rows = []
        cols = []
        vals = []
        rhs = np.zeros(2 * n - 1)

        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                # a[j] = a[min(j, n-1-j)]
                j_sym = min(j, n - 1 - j)
                if j_sym < n_half:
                    rows.append(k)
                    cols.append(j_sym)
                    vals.append(2.0 * f_sym[k - j])
            rows.append(k)
            cols.append(n_half)
            vals.append(-1.0)
            rhs[k] = c[k]

        A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n_half + 1))

        a_var = cp.Variable(n_half)
        M_var = cp.Variable()

            a_var >= 0,
            # sum(a) = 1 with symmetry: sum of first half * 2 (minus middle if n odd)
            # For even n: 2 * sum(a_var) = 1
            # For odd n: 2 * sum(a_var) - a_var[n_half-1] = 1
        ]
        if n % 2 == 0:
            constraints.append(2.0 * cp.sum(a_var) == 1)
        else:
            constraints.append(2.0 * cp.sum(a_var) - a_var[n_half - 1] == 1)
        constraints.append(A_mat @ var_vec <= rhs)

        prob = cp.Problem(cp.Minimize(M_var), constraints)

        a_new_half = None
        for solver in [cp.ECOS, cp.SCS]:
            try:
                if solver == cp.ECOS:
                    prob.solve(solver=solver, max_iter=30000, abstol=1e-9, reltol=1e-9)
                else:
                    prob.solve(solver=solver, max_iters=100000, eps=1e-9)
                if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    a_new_half = np.maximum(a_var.value, 0.0)
                    break
            except Exception:
                continue

        if a_new_half is None:
            stuck += 1
            if stuck > 20:
                break
            continue

        # Reconstruct full sequence from half
        a_new = np.zeros(n)
        for i in range(n_half):
            a_new[i] = a_new_half[i]
            a_new[n - 1 - i] = a_new_half[i]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            stuck += 5
            continue
        a_new /= s

        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.3, 0.5, 0.7, 1.0, 0.15]:
            f_trial = (1.0 - t) * f_sym + t * a_new
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-12:
                best_t_obj = obj
                best_t = t

        if best_t == 0.0:
            stuck += 1
            if stuck > 30:
                break
            continue

        f_sym = (1.0 - best_t) * f_sym + best_t * a_new
        f_sym = project_simplex(f_sym)
        cur_obj = fast_convolve(f_sym).max()

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_f = f_sym.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 50 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization using SCA (linearized LP) as primary engine,
    with LSE subgradient polishing, coordinate descent, and multi-length search.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent found, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    n_parent = len(parent)

    # === Phase 1: SCA full constraints on parent length ===
    print(f"\n{'='*60}")
    print(f"Phase 1: SCA full (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 700.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_sca_phase(parent, time_limit=t1, use_full=True, label="sca_full")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [sca_full]: {score1:.6f} (n={len(f1)})")

    # === Phase 2: SCA reduced (more iterations, faster per iter) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: SCA reduced (n={len(best_seq)}, current best={best_score:.6f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s, top_k=80")
        f2, obj2 = run_sca_phase(best_seq, time_limit=t2, use_full=False,
                                  top_k_reduced=80, label="sca_red")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [sca_red]: {score2:.6f}")

    # === Phase 3: Symmetric SCA ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Symmetric SCA (n={len(best_seq)}, current best={best_score:.6f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 200.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        f3, obj3 = try_symmetric_sca(best_seq, time_limit=t3, label="sym_sca")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [sym_sca]: {score3:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5]:
        tl = int(n_base * factor)
        tl = max(256, min(4096, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t4 = min(rem * 0.12, 120.0)
        if t4 < 20:
            break
        # Interpolate
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  SCA n={target_n}, budget={t4:.0f}s")
        f4, obj4 = run_sca_phase(f_interp, time_limit=t4, use_full=True,
                                  label=f"sca_n{target_n}", verbose=False)
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [n={target_n}]: {score4:.6f}")

    # === Phase 5: Subgradient polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Subgradient polish (best={best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    if t5 > 20:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = run_subgradient_phase(best_seq, time_limit=t5, lr_init=0.008,
                                         tau_init=0.0005, label="sg_polish")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg]: {score5:.6f}")

    # === Phase 6: Final SCA burst ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final SCA burst (best={best_score:.6f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.20, 200.0)
    if t6 > 20:
        print(f"  Budget: {t6:.0f}s, n={len(best_seq)}")
        f6, obj6 = run_sca_phase(best_seq, time_limit=t6, use_full=True,
                                  label="sca_final", verbose=False)
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [sca_final]: {score6:.6f}")

    # === Phase 7: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Coordinate descent (best={best_score:.6f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.15, 100.0)
    if t7 > 10:
        print(f"  Budget: {t7:.0f}s")
        f7, obj7 = coordinate_polish(best_seq, time_limit=t7, label="cd")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {score7:.6f}")

    # === Final verification ===
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
