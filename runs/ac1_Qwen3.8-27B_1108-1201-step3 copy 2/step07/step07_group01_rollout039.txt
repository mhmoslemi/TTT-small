```python
import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    b = fast_convolve(a_norm)
    return float(2.0 * n * b.max())

def project_simplex(v):
    """Project v onto the probability simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fista(a_init, time_limit, step_size=None, N=1000):
    """FISTA for minimizing max(conv(a,a)) subject to sum(a)=1, a>=0."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    b = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    
    # Lipschitz constant estimate
    L = 4.0 * n * a.max() * 2  # rough estimate
    
    t = 1.0
    a_prev = a.copy()
    
    t0 = time.monotonic()
    for k in range(N):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient of max(conv(a,a)) at a
        ck = int(np.argmax(fast_convolve(a)))
        grad = np.zeros(n)
        for i in range(n):
            j = ck - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # FISTA update
        y = a + ((t - 1) / (t + 1)) * (a - a_prev)
        a_new = project_simplex(y - grad / L)
        
        # Update t
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t * t))
        
        # Check improvement
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a_prev = a
        a = a_new
        t = t_new
    
    return best_a, best_score
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-length search with mirror descent,

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgrad(a):
    """Compute subgradient of max(conv(a,a)) at point a."""
    b = fast_convolve(a)
    k = int(np.argmax(b))
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad, b


def cosine_window(n):
    """Generate a raised-cosine (Hann) window of length n."""
    if n <= 1:
        return np.ones(n)
    t = np.linspace(0, np.pi, n)
    return 0.5 * (1 + np.cos(t))


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving non-negativity."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

def subgradient_polish(a_init, time_limit, label=""):
    """Subgradient method with Polyak step size and averaging."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = norm_score(a)
    t0 = time.monotonic()

    # Reference point for Polyak step
    a_ref = a.copy()
    t_ref = best_score / (2.0 * n)  # target peak level

    running_sum = np.zeros(n)
    count = 0
    eta = 1.0 / np.sqrt(n * 50.0)

    while time.monotonic() - t0 < time_limit:
        grad, b = compute_subgrad(a)
        score = float(2.0 * n * b.max())

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()

        # Polyak-like step: use distance to "target"
        current_peak = b.max()
        gap = current_peak - t_ref
        if gap < 1e-10:
            eta = 1e-8
        else:
            grad_norm_sq = np.dot(grad, grad)
            if grad_norm_sq > 1e-15:
                eta = max(gap * 1.5 / grad_norm_sq, 1e-10)
            else:
                eta = 1e-8

        # Projected subgradient step
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
        a = a_new

        # Running average
        running_sum += a
        count += 1
        a_avg = running_sum / count
        score_avg = norm_score(a_avg)
        if score_avg < best_score - 1e-15:
            best_score = score_avg
            best_a = a_avg.copy()

    return best_a, best_score


def lbfgs_smooth(a_init, time_limit, T=50.0, label=""):
    """L-BFGS optimization using log-sum-exp smoothing."""
    from scipy.optimize import minimize

    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def func_z(z):
        if time.monotonic() - t0 > time_limit * 0.9:
            return 1e15
        # Softmax to get a
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_local = exp_z / exp_z.sum()
        b = fast_convolve(a_local)
        scaled = b / T
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        return 2.0 * n * T * lse

    def jac_z(z):
        if time.monotonic() - t0 > time_limit * 0.9:
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_local = exp_z / exp_z.sum()
        b = fast_convolve(a_local)
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()
        # Gradient wrt a: 2n * cross-correlation of w with a
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_local, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 2.0 * n * correl[:n]
        # Chain rule through softmax
        mean_grad = np.dot(a_local, grad_a)
        grad_z = a_local * (grad_a - mean_grad)
        return grad_z

    try:
        res = minimize(fun=func_z, x0=z0, jac=jac_z, method='L-BFGS-B',
                       options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': 50000, 'disp': False})
        z_opt = res.x
        z_shifted = z_opt - z_opt.max()
        exp_z = np.exp(z_shifted)
        a_out = exp_z / exp_z.sum()
    except Exception:
        a_out = a_init.copy()

    return a_out


def multi_phase_optimize(a_init, time_limit, label=""):
    """Multi-phase optimization: mirror descent -> L-BFGS -> subgradient polish."""
    n = len(a_init)
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = norm_score(a_init)

    # Phase 1: Mirror descent (quick convergence)
    t_alloc1 = min(time_limit * 0.4, time_limit - 10)
    if t_alloc1 > 5:
        a1, s1 = mirror_descent_optimize(a_init, t_alloc1, label=label + "_MD")
        if s1 < best_score:
            best_a, best_score = a1, s1

    # Phase 2: L-BFGS smoothing (refine)
    elapsed = time.monotonic() - t0
    t_alloc2 = min(time_limit * 0.3, time_limit - elapsed - 10)
    if t_alloc2 > 5:
        for T in [30, 10]:
            elapsed = time.monotonic() - t0
            t_rem = time_limit - elapsed - 10
            if t_rem < 5:
                break
            t_sub = min(t_rem * 0.5, 60.0)
            a2 = lbfgs_smooth(best_a, t_sub, T=T, label=label + f"_LB{T}")
            s2 = norm_score(a2)
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()

    # Phase 3: Subgradient polish (final refinement)

        if s3 < best_score:

    def expand(h):
        if n % 2 == 0:
            return np.concatenate([h, h[::-1]])
        else:
            return np.concatenate([h[:-1], [h[-1]], h[-2::-1]])

    best_h = half.copy()
    a_full = expand(half)
    a_full = np.maximum(a_full, 1e-15)
    a_full = a_full / a_full.sum()
    best_h = a_full[:half_n].copy()
    best_score = norm_score(a_full)
    t0 = time.monotonic()

    eta = 2.0 / np.sqrt(half_n * 200.0)
    eta_init = eta
    decay = 0.99995
    stuck = 0

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        a_full = expand(half)
        a_full = np.maximum(a_full, 1e-15)
        a_full = a_full / a_full.sum()

        grad, b = compute_subgrad(a_full)
        score = float(2.0 * n * b.max())
        if score < best_score - 1e-15:
            best_score = score
            best_h = a_full[:half_n].copy()
            stuck = 0
        else:
            stuck += 1

        # Fold gradient back to half
        if n % 2 == 0:
            half_grad = (grad[:half_n] + grad[:half_n][::-1]) / 2.0
        else:
            half_grad = grad[:half_n].copy()
            half_grad[:-1] += grad[:n-2][::-1][:half_n-1][:half_n-1] if half_n > 1 else 0

        half_grad_centered = half_grad - np.dot(half, half_grad)
        half = half * np.exp(-eta * half_grad_centered)
        half = np.maximum(half, 1e-300)
        half = half / half.sum()

        eta *= decay
        if stuck > 500:
            rng = np.random.default_rng(it * 13 + 3)
            noise = rng.standard_normal(half_n) * 0.01
            half = best_h + noise
            half = np.maximum(half, 1e-15)
            half = half / half.sum()
            eta = eta_init
            stuck = 0

    best_a = expand(best_h)
    best_a = np.maximum(best_a, 0)
    best_a = best_a / best_a.sum()
    return best_a, norm_score(best_a)


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate using actual evaluate_sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-length, multi-method search for optimal non-negative sequence.
    Strategy:
    1. Try multiple lengths from 256 to 2048
    2. Use cosine window and interpolated parent as initializations
    3. Apply mirror descent + L-BFGS + subgradient polish
    4. Exploit palindromic symmetry for speed
    5. Search over length to find the globally best n
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    parent_n = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_n = len(parent)
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={parent_n}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Fallback: use cosine window at a reasonable length
        parent = cosine_window(512)
        parent = parent / parent.sum()
        parent_n = 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Generated parent: n={parent_n}, score={best_score:.7f}")

    # === Phase 1: Quick scan over multiple lengths ===
    print("\n=== Phase 1: Multi-length Quick Scan ===")
    # Generate candidate lengths centered around parent_n
    candidate_lengths = set()
    for factor in [0.5, 0.625, 0.75, 0.875, 1.0, 1.125, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0]:
        cn = int(parent_n * factor)
        if 128 <= cn <= 4096:
            candidate_lengths.add(cn)
    # Add some "nice" lengths
    for nl in [256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]:
        if 128 <= nl <= 4096:
            candidate_lengths.add(nl)

    candidate_lengths = sorted(candidate_lengths)
    print(f"  Testing {len(candidate_lengths)} candidate lengths: {candidate_lengths}")

    # Quick pass: short optimization for each length
    quick_time = min(20.0, (deadline - time.monotonic()) / max(1, len(candidate_lengths)) * 0.7)
    scan_results = {}

    for cn in candidate_lengths:
        if time.monotonic() > deadline - 300:
            break
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break

        t_alloc = min(quick_time, rem * 0.8 / max(1, 1.0))
        if t_alloc < 3:
            break

        # Initialize: interpolate parent if available, else cosine window
        if parent is not None:
            a_init = interpolate_to_length(parent, cn)
        else:
            a_init = cosine_window(cn) / cosine_window(cn).sum()

        a_sc, s_sc = mirror_descent_optimize(a_init, t_alloc, label=f"scan_n{cn}")
        scan_results[cn] = (a_sc, s_sc)
        print(f"  n={cn}: quick score={s_sc:.8f}")

    # Sort by score and pick top candidates
    if scan_results:
        sorted_lens = sorted(scan_results.items(), key=lambda x: x[1][1])
        print(f"  Top lengths: {[(n, f'{s:.6f}') for n, (a, s) in sorted_lens[:5]]}")

        # Update best from scan
        for cn, (a_sc, s_sc) in sorted_lens[:5]:
            best_seq, best_score = check_and_update(a_sc, f"scan_n{cn}", best_seq, best_score)

        # Phase 2: Deep optimization on top lengths
        print("\n=== Phase 2: Deep Optimization on Top Lengths ===")
        top_lengths = [n for n, _ in sorted_lens[:5]]

        for cn in top_lengths:
            if time.monotonic() > deadline - 100:
                break
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break

            t_alloc = min(rem * 0.35, 180.0)
            if t_alloc < 10:
                break

            # Start from the scan result
            a_init = scan_results[cn][0].copy()

            # Try multiple initializations
            inits = [a_init]
            if parent is not None:
                inits.append(interpolate_to_length(parent, cn))
            inits.append(cosine_window(cn) / cosine_window(cn).sum())

            for ii, init in enumerate(inits[:2]):
                rem = deadline - time.monotonic() - 60
                if rem < 20:
                    break
                t_sub = min(rem * 0.4, 120.0)
                if t_sub < 10:
                    break

                label = f"deep_n{cn}_init{ii}"
                a_opt, s_opt = multi_phase_optimize(init, t_sub, label=label)
                best_seq, best_score = check_and_update(a_opt, label, best_seq, best_score)

    # === Phase 3: Symmetry exploitation on best length ===
    if best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Phase 3: Symmetric Optimization (n={n_best}) ===")
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(rem * 0.4, 120.0)
            a_sym, s_sym = symmetric_optimize(best_seq, t_alloc, label=f"sym_n{n_best}")
            best_seq, best_score = check_and_update(a_sym, f"sym_n{n_best}", best_seq, best_score)

    # === Phase 4: Targeted refinement on current best ===
    if best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Phase 4: Targeted Refinement (n={n_best}, score={best_score:.7f}) ===")

        # Try nearby lengths around the best
        nearby_lengths = []
        for delta in [-50, -30, -20, -10, 10, 20, 30, 50, 75, 100]:
            nl = n_best + delta
            if 128 <= nl <= 4096 and nl != n_best and nl not in nearby_lengths:
                nearby_lengths.append(nl)

        for nl in nearby_lengths[:4]:
            if time.monotonic() > deadline - 40:
                break
            rem = deadline - time.monotonic() - 40
            if rem < 15:
                break
            t_alloc = min(rem * 0.35, 60.0)
            if t_alloc < 5:
                break

            a_init = interpolate_to_length(best_seq, nl)
            a_opt, s_opt = multi_phase_optimize(a_init, t_alloc, label=f"near_n{nl}")
            best_seq, best_score = check_and_update(a_opt, f"near_n{nl}", best_seq, best_score)

    # === Phase 5: Final polishing ===
    if best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Phase 5: Final Polish (n={n_best}, score={best_score:.7f}) ===")

        # L-BFGS with very low temperature
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            for T in [5, 2, 1]:
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                t_sub = min(rem * 0.4, 20.0)
                a_fin = lbfgs_smooth(best_seq, t_sub, T=T, label=f"fin_T{T}")
                best_seq, best_score = check_and_update(a_fin, f"fin_T{T}", best_seq, best_score)

        # Final mirror descent with very small step
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            t_sub = min(rem * 0.5, 20.0)
            a_md, s_md = mirror_descent_optimize(
                best_seq, t_sub, eta=0.001, decay=0.99999,
                stuck_threshold=200, label="final_MD"
            )
            best_seq, best_score = check_and_update(a_md, "final_MD", best_seq, best_score)

        # Subgradient polish
        rem = deadline - time.monotonic() - 5
        if rem > 3:
            a_sg, s_sg = subgradient_polish(best_seq, min(rem * 0.7, 10), label="final_SG")
            best_seq, best_score = check_and_update(a_sg, "final_SG", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = cosine_window(512)
        best_seq = best_seq / best_seq.sum()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
