```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def smooth_max_and_grad(a, T):

    n = len(a)

    scaled = b / T
    mx = scaled.max()
    exp_s = np.exp(scaled - mx)

    w_matrix = w[idx]  # (n,n)
    grad = 2.0 * (w_matrix @ a)
```

```python
"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS with temperature continuation in softmax space,
subgradient descent with momentum, peak shaving, and length exploration.
Combines efficient FFT-based convolution with vectorized O(n^2) gradients.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def smooth_max_and_grad(a, T):
    """
    Compute T*log(sum_k exp(conv(a,a)[k]/T)) and gradient wrt a.
    Returns (value, gradient) where gradient[j] = 2*sum_k w[k]*a[k-j].
    """
    n = len(a)
    b = fast_convolve(a)
    scaled = b / T
    mx = scaled.max()
    exp_s = np.exp(scaled - mx)
    w = exp_s / exp_s.sum()
    value = b.max() + T * np.log(exp_s.sum())

    # grad[j] = 2 * sum_m w[j+m] * a[m], using Hankel matrix
    idx = np.add.outer(np.arange(n), np.arange(n))
    H = w[idx]
    grad = 2.0 * (H @ a)
    return value, grad


def project_simplex(v):
    """Project vector v onto the probability simplex (a>=0, sum(a)=1)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lbgfs_continue(a0, time_limit, T_list=None, maxiter_per_T=3000):
    """L-BFGS refinement in softmax space with temperature continuation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    z = np.log(a) - np.log(a).mean()

    if T_list is None:
        T_list = [100, 50, 25, 12, 6, 3, 1.5, 0.75, 0.4]

    t0 = monotonic()
    best_z = z.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    print(f"    Start score: {best_score:.6f}")

    for T in T_list:
        if monotonic() - t0 > time_limit:
            break
        t_phase_start = monotonic()
        n_iter = min(maxiter_per_T, max(200, int(3000 * (100 / T))))

        def func(z_in):
            z_s = z_in - z_in.max()
            e = np.exp(z_s)
            a_curr = e / e.sum()
            val, _ = smooth_max_and_grad(a_curr, T)
            return 2.0 * n * val

        def jac(z_in):
            z_s = z_in - z_in.max()
            e = np.exp(z_s)
            a_curr = e / e.sum()
            val, grad_a = smooth_max_and_grad(a_curr, T)
            S = a_curr @ grad_a
            grad_z = a_curr * (grad_a - S * a_curr)
            grad_z *= 2.0 * n
            return grad_z

        try:
            res = minimize(
                func, z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': n_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': n_iter * 12, 'disp': False}
            )
            z = res.x
            z_s = z - z.max()
            a_curr = np.exp(z_s)
            a_curr = a_curr / a_curr.sum()
            score = float(evaluate_sequence(a_curr.tolist()))
            if score < best_score:
                best_score = score
                best_z = z.copy()
            elapsed = monotonic() - t_phase_start
            print(f"    T={T}: score={score:.6f} ({elapsed:.1f}s, niter={res.niter})")
        except Exception as e:
            print(f"    T={T}: error {e}")

    z_s = best_z - best_z.max()
    a_out = np.exp(z_s)
    a_out = a_out / a_out.sum()
    return a_out

def subgradient_opt(a0, time_limit, lr_init=0.003, momentum=0.92):
    """Subgradient method with momentum and adaptive step on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    t0 = monotonic()
    no_improve = 0
    step = 0

    while monotonic() - t0 < time_limit:
        elapsed_frac = (monotonic() - t0) / time_limit
        lr = lr_init * 0.5 * (1 + np.cos(np.pi * min(1.0, elapsed_frac)))
        lr = max(lr, 1e-5)

        b = fast_convolve(a)
        peak = b.max()
        top_k = min(n, max(20, n // 5))
        peaks_idx = np.argpartition(b, -top_k)[-top_k:]
        pv = b[peaks_idx]
        T_l = 0.005 * max(peak, 1e-10)
        w = np.exp((pv - pv.max()) / T_l)
        w = w / w.sum()

        # Subgradient (vectorized)
        sg = np.zeros(n)
        for i in range(len(peaks_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peaks_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * sg
        a_trial = a + v
        a_trial = project_simplex(a_trial)

        score = 2.0 * n * fast_convolve(a_trial).max()
        step += 1
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 800:
            scale = 0.001 * (1 + no_improve / 1000.0)
            a = best_a + np.random.standard_normal(n) * scale
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            no_improve = 0

        if step % 5000 == 0:
            print(f"    SubGrad step={step} ({monotonic()-t0:.0f}s): {best_score:.6f}")

    return best_a


def peak_shave_opt(a0, time_limit):
    """Local peak shaving: reduce peak by adjusting top contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0
    step = 0

    while monotonic() - t0 < time_limit and step < 50000:
        step += 1
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        if peak_val <= best_score / (2.0 * n) + 1e-14:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total = contribs.sum()
        if total < 1e-20:
            break

        improved = False
        # Try reducing top contributors
        top_c = np.argsort(contribs)[::-1][:8]
        for ci in top_c:
            if improved:
                break
            i = int(idx[ci])
            old = a[i]
            if old < 1e-13:
                continue
            for factor in [0.97, 0.94, 0.9, 0.85]:
                new_val = old * factor
                diff = old - new_val
                if diff < 1e-15:
                    continue
                a_t = a.copy()
                a_t[i] = new_val
                # Redistribute to non-peak indices
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                # Also mask the peak partner
                partner = peak_k - i
                if partner != i and 0 <= partner < n:
                    mask[partner] = False
                others = a_t[mask]
                osum = others.sum()
                if osum < 1e-13:
                    break
                a_t[mask] = others + diff * others / osum
                a_t = a_t / a_t.sum()
                sc = 2.0 * n * fast_convolve(a_t).max()
                if sc < best_score - 1e-13:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Random perturbation
            for _ in range(3):
                a_t = a + np.random.standard_normal(n) * 0.0003
                a_t = np.maximum(a_t, 0)
                a_t = a_t / a_t.sum()
                sc = 2.0 * n * fast_convolve(a_t).max()
                if sc < best_score - 1e-14:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 100:
                break
        else:
            failed = 0

    return best_a


def interp_seq(a_old, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    a_norm = a_old / a_old.sum()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_norm)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new


def check_update(a, label, best_seq, best_score):
    """Evaluate candidate and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases: L-BFGS continuation, palindromic L-BFGS, subgradient, peak shaving,
    length exploration, and final polish.
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 15.0)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: L-BFGS continuation (primary) ===
    print(f"\n=== Phase 1: L-BFGS Continuation (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        a1 = lbgfs_continue(best_seq, t1)
        best_seq, best_score = check_update(a1, "L-BFGS", best_seq, best_score)

    # === Phase 2: Palindromic L-BFGS ===
    print(f"\n=== Phase 2: Palindromic L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t2 = min(rem * 0.2, 300.0)
    if t2 > 20:
        a2 = lbgfs_palin(best_seq, t2)
        best_seq, best_score = check_update(a2, "Palin", best_seq, best_score)

    # === Phase 3: Subgradient with momentum ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t3 = min(rem * 0.25, 400.0)
    if t3 > 20:
        a3 = subgradient_opt(best_seq, t3)
        best_seq, best_score = check_update(a3, "SubGrad", best_seq, best_score)

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t4 = min(rem * 0.2, 300.0)
    if t4 > 10:
        a4 = peak_shave_opt(best_seq, t4)
        best_seq, best_score = check_update(a4, "PeakShave", best_seq, best_score)

    # === Phase 5: L-BFGS second pass on best ===
    print(f"\n=== Phase 5: L-BFGS Pass 2 (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t5 = min(rem * 0.25, 300.0)
    if t5 > 20:
        # Use tighter temperatures
        n5 = len(best_seq)
        a5 = lbgfs_continue(best_seq, t5, T_list=[20, 10, 5, 2, 1, 0.5, 0.25])
        best_seq, best_score = check_update(a5, "L-BFGS2", best_seq, best_score)

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for factor in [0.75, 1.25, 1.5, 2.0]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    # Deduplicate and limit
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - monotonic()
        if rem < 40:
            break
        t_alt = min(rem * 0.25, 200.0)
        if t_alt < 20:
            continue
        print(f"  Trying n={alt_n} (budget={t_alt:.0f}s)")
        init_alt = interp_seq(best_seq, alt_n)
        a_alt = lbgfs_continue(init_alt, t_alt, T_list=[50, 20, 10, 5, 2, 1])
        best_seq, best_score = check_update(a_alt, f"Len_{alt_n}", best_seq, best_score)

    # === Phase 7: Multiple restarts at best length ===
    print(f"\n=== Phase 7: Restarts (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for restart_i in range(5):
        rem = deadline - monotonic()
        if rem < 30:
            break
        t_r = min(rem * 0.15, 150.0)
        if t_r < 15:
            break
        # Add noise
        noise_scale = [0.002, 0.001, 0.003, 0.0005, 0.0015][restart_i]
        a_r = best_seq + np.random.standard_normal(n_best) * noise_scale
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        a_r = lbgfs_continue(a_r, t_r, T_list=[10, 5, 2, 1, 0.5])
        best_seq, best_score = check_update(a_r, f"Restart{restart_i}", best_seq, best_score)

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 15:
        t_ps = min(rem * 0.4, 60.0)
        a_ps = peak_shave_opt(best_seq, t_ps)
        best_seq, best_score = check_update(a_ps, "FinalPS", best_seq, best_score)

    rem = deadline - monotonic()
    if rem > 10:
        t_final = min(rem * 0.5, 30.0)
        a_final = lbgfs_continue(best_seq, t_final, T_list=[5, 2, 1, 0.5, 0.2])
        best_seq, best_score = check_update(a_final, "FinalLBFGS", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
```
