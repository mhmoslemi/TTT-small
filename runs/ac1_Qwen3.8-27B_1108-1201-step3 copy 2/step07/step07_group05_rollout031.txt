import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_weights(c, T):
    """Compute softmax weights for the log-sum-exp smoothing."""
    c_shifted = c / T
    c_max = c_shifted.max()
    exp_c = np.exp(c_shifted - c_max)
    return exp_c / exp_c.sum()

    b = a[::-1]
    w_rev = w[::-1]
    fft_size = 1 << (len(w_rev) + len(b) - 1 - 1).bit_length()
    Wr = rfft(w_rev, fft_size)
    B = rfft(b, fft_size)
    conv_wb = irfft(Wr * B, fft_size)
    # conv_wb[m] = sum_k w_rev[k] * b[m-k]
    # We want sum_m w[n-1-j+m]*b[m] = conv of flip(w) and b at index n-1-j
    # conv_wb[n-1-j] = sum_k w_rev[k]*b[n-1-j-k] = sum_k w[2n-2-k]*b[n-1-j-k]
    # Let l = 2n-2-k: sum_l w[l]*b[2n-2-l-n+1+j] = sum_l w[l]*b[n-3+j-l]... this is getting messy.
    
    # Let me just use the direct method but vectorized:
    # For each j, grad[j] = dot(w[j:j+n], a[0:n])
    # This is a sliding dot product - can be done with a Toeplitz matrix or FFT.
    
    # Actually, let's construct it as: create a matrix M where M[j,k] = 1 if k is in [j, j+n-1]
    # Then grad[j] = sum_k w[k]*a[k-j]*M[j,k]
    
    # For n up to 2048, the direct O(n^2) is fine (4M operations)
    # But for larger n, let's use FFT.
    
    # Approach: pad a to length 2n-1 with zeros, compute correlation
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    # correlation[j] = sum_k w[k] * a_pad[k + j] ... no
    # We want sum_k w[k] * a[k-j] where a is only defined for 0..n-1
    # = sum_{k=j}^{j+n-1} w[k] * a[k-j]
    
    # This is: (w * a_pad)[j] where * is correlation (not convolution)
    # corr[j] = sum_k w[k] * a_pad[k-j]
    # = conv(flip(w), a_pad)[j + len(w) - 1] ... 
    
    # Let me just use np.correlate or manual FFT
    # corr = np.correlate(w, a_pad, mode='full') gives corr[m] = sum_k w[k]*a_pad[k-m]
    # We want sum_k w[k]*a_pad[k-j] = corr[j] for the right indexing
    
    # Actually: np.correlate(a, v, mode='full')[m] = sum_n a[n+m]*v[n]
    # So np.correlate(w, a_pad, 'full')[j] = sum_n w[n+j]*a_pad[n]
    # That's sum_n w[n+j]*a_pad[n], not what we want.
    
    # What we want: sum_k w[k]*a_pad[k-j] = sum_k w[k]*a_pad[k-j]
    # Let l = k-j: sum_l w[l+j]*a_pad[l] = np.correlate(w, a_pad, 'full')[j + n - 1]... 
    
    # This is getting too complicated. Let me just use the O(n^2) direct method
    # which for n=768 is about 600K multiplies per gradient eval - very fast.
    
    grad = np.zeros(n)
    for j in range(n):
        k_end = min(j + n, len(w))
        if j < k_end:
            grad[j] = np.dot(w[j:k_end], a[:k_end - j])
    
    return (2.0 / T) * grad, conv


def lse_objective(a, T):
    """Log-sum-exp smoothed objective."""
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv / T
    max_s = shifted.max()
    return (2.0 * n / T) * np.exp(max_s + np.log(np.exp(shifted - max_s).sum()) - max_s)


def projected_gradient_descent(a0, time_limit, T_start=0.05, T_end=0.001,
                                alpha=0.9995, verbose=True):
    """
    Projected gradient descent with log-sum-exp smoothed max objective.
    Uses backtracking line search and adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.1
    iters = 0
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute gradient
        grad, conv = gradient_smoothed_fft(a, T, n)
        current_score = compute_score(a)
        
        if not np.isfinite(grad).all():
            break
            
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            stale += 1
            if stale > 50:
                break
            # Perturb
            a = project_simplex(a + 0.001 * np.random.standard_normal(n))
            continue
        
        # Backtracking line search
        a_old = a.copy()
        score_old = current_score
        accepted = False
        for backtrack in range(20):
            a_trial = project_simplex(a_old - lr * grad)
            score_trial = compute_score(a_trial)
            if score_trial < score_old - 1e-10:
                a = a_trial
                accepted = True
                break
            lr *= 0.5
        
        if not accepted:
            # Try a simpler step
            a_trial = project_simplex(a_old - 0.001 * grad / grad_norm)
            score_trial = compute_score(a_trial)
            if score_trial < score_old:
                a = a_trial
            else:
                stale += 1
                if stale > 100:
                    # Restart with perturbation
                    a = project_simplex(best_a + 0.01 * np.random.standard_normal(n))
                    stale = 0
                continue
        
        stale = 0
        lr = min(lr * 1.5, 1.0)  # Increase step size if successful
        
        if score_trial < best_score - 1e-12:
            best_score = score_trial
            best_a = a.copy()
        
        if verbose and iters % 200 == 0:
            print(f"    PGD iter={iters}, T={T:.6f}, lr={lr:.4f}, score={best_score:.8f}")
    
    return best_a, best_score


def peak_shaving(a0, time_limit, verbose=True):
    """
    Iteratively reduce the single highest peak of the autocorrelation.
    At each step, take a gradient step against the highest peak only.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.5
    iters = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))
        
        # Gradient of c[max_idx] w.r.t. a: 2 * flip_{max_idx}(a)
        # flip_k(a)[j] = a[k-j] if 0 <= k-j < n
        j_start = max(0, max_idx - n + 1)
        j_end = min(n, max_idx + 1)
        grad = np.zeros(n)
        for j in range(j_start, j_end):
            grad[j] = 2.0 * a[max_idx - j]
        
        # Try to reduce this peak
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
            
        # Line search
        a_old = a.copy()
        accepted = False
        for factor in [lr, lr*0.5, lr*0.25, 0.1, 0.05, 0.01]:
            a_trial = project_simplex(a_old - factor * grad / grad_norm)
            sc_trial = compute_score(a_trial)
            sc_old = compute_score(a_old)
            if sc_trial < sc_old - 1e-12:
                a = a_trial
                accepted = True
                lr = min(lr * 1.1, 2.0)
                break
        
        if not accepted:
            # Try random perturbation focused on peak contributors
            idx = np.arange(j_start, j_end)
            for _ in range(5):
                p = idx[rng.integers(0, len(idx))]
                delta = a[p] * 0.1 * rng.standard_normal()
                a_trial = a.copy()
                a_trial[p] -= delta
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        accepted = True
                        break
            if not accepted:
                lr *= 0.8
                if lr < 0.001:
                    break
        else:
            sc = compute_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
        
        if verbose and iters % 100 == 0:
            print(f"    PS iter={iters}, peak_idx={max_idx}, score={best_score:.8f}")
    
    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T, maxiter=3000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    a0_safe = np.maximum(a0, 1e-15)
    a0_safe = a0_safe / a0_safe.sum()
    z0 = np.log(a0_safe)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        val = (2.0 * n / T) * np.exp(max_s + np.log(np.exp(shifted - max_s).sum()) - max_s)
        return val
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)
        w = softmax_weights(conv, T)
        # Gradient w.r.t. a: (2/T) * sum_k w[k] * flip_k(a)
        grad_a = np.zeros(n)
        for j in range(n):
            k_end = min(j + n, len(w))
            if j < k_end:
                grad_a[j] = np.dot(w[j:k_end], a[:k_end - j])
        grad_a = (2.0 / T) * grad_a
        # Project to log-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        z_max = res.x.max()
        a_out = np.exp(res.x - z_max)
        a_out = a_out / a_out.sum()
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass
    
    result['seconds'] = time.monotonic() - t0
    return result


def coordinate_descent(a0, time_limit, n_pairs=300, verbose=False):
    """Fast coordinate descent with pairwise moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            best_local = best_score
            best_delta = 0.0
            base = max(a[i], a[j], 1e-8)
            
            for frac in [0.05, 0.1, 0.2, 0.4, 0.1, 0.02]:
                for sign in [1, -1]:
                    t = sign * frac * base
                    if a[i] + t < -1e-15 or a[j] - t < -1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] += t
                    a_trial[j] -= t
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_local - 1e-13:
                        best_local = sc
                        best_delta = t
            
            if best_delta != 0.0:
                a[i] += best_delta
                a[j] -= best_delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        
        if not improved:
            stale += 1
            if stale > 20:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(best_a + noise)
                best_score = compute_score(a)
                stale = 0
        else:
            stale = 0
        
        if verbose and iters % 50 == 0:
            print(f"    CD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def nesterov_accelerated(a0, time_limit, T=200, lr_init=0.02, momentum=0.93):
    """Nesterov accelerated gradient in log-space."""
    n = len(a0)
    a0_safe = np.maximum(a0, 1e-15)
    a0_safe = a0_safe / a0_safe.sum()
    z = np.log(a0_safe)
    v = np.zeros(n)
    best_z = z.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        z_la = z - momentum * v
        z_max = z_la.max()
        a_la = np.exp(z_la - z_max)
        a_la = a_la / a_la.sum()
        
        conv = fast_convolve(a_la)
        w = softmax_weights(conv, T)
        grad_a = np.zeros(n)
        for j in range(n):
            k_end = min(j + n, len(w))
            if j < k_end:
                grad_a[j] = np.dot(w[j:k_end], a_la[:k_end - j])
        grad_a = (2.0 / T) * grad_a
        S = np.dot(grad_a, a_la)
        grad_z = a_la * (grad_a - S)
        
        v = momentum * v + lr * grad_z
        z_new = z - v
        z_new = np.clip(z_new, -30, 30)
        
        z_max_new = z_new.max()
        a_new = np.exp(z_new - z_max_new)
        a_new = a_new / a_new.sum()
        score = compute_score(a_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_z = z_new.copy()
        
        z = z_new
        lr *= 0.9999
    
    z_max = best_z.max()
    a_out = np.exp(best_z - z_max)
    a_out = a_out / a_out.sum()
    return a_out, best_score


def make_chirp_init(n, freq=3.0, amplitude=0.3):
    """Create a chirp-like initialization."""
    t = np.linspace(0, 1, n)
    a = 1.0 + amplitude * np.cos(2 * np.pi * freq * t)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_bump_init(n, width=0.3, center=0.5):
    """Create a Gaussian bump initialization."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - center) / width) ** 2)
    a = a + 0.01
    return a / a.sum()


def make_bimodal_init(n, sep=0.3):
    """Create a bimodal initialization."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5 + sep) / 0.15) ** 2) + np.exp(-((t - 0.5 - sep) / 0.15) ** 2)
    a = a + 0.01
    return a / a.sum()


def make_random_sparse(n, density=0.3, rng=None):
    """Create a random sparse initialization."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = rng.random(n)
    mask = rng.random(n) < density
    a[~mask] = 0.0
    a = a + 1e-4
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer:
    1. Diversified initialization (chirps, bumps, sparse, parent)
    2. Projected gradient with adaptive temperature (main workhorse)
    3. Peak shaving (direct peak reduction)
    4. L-BFGS in log-space (polish)
    5. Coordinate descent (fine-tune)
    6. Nesterov acceleration
    7. Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    n_main = 768
    if parent is None:
        parent = np.ones(n_main) / n_main
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        n_main = len(parent)
        print(f"Initialized uniform: n={n_main}, score={best_score:.8f}")
    else:
        n_main = len(parent)
    
    # === Phase 1: Diversified initialization (100s) ===
    print(f"\n=== Phase 1: Init Diversity (best={best_score:.8f}) ===")
    t1_budget = 100.0
    t1_end = time.monotonic() + t1_budget
    
    initializations = {
        'chirp1': make_chirp_init(n_main, freq=2.0, amplitude=0.4),
        'chirp2': make_chirp_init(n_main, freq=5.0, amplitude=0.3),
        'bump': make_bump_init(n_main, width=0.4, center=0.5),
        'bimodal': make_bimodal_init(n_main, sep=0.25),
        'sparse1': make_random_sparse(n_main, density=0.4, rng=np.random.default_rng(1)),
        'sparse2': make_random_sparse(n_main, density=0.2, rng=np.random.default_rng(2)),
    }
    
    for name, init in initializations.items():
        if time.monotonic() > t1_end:
            break
        sc = compute_score(init)
        print(f"  Init {name}: score={sc:.6f}")
        if sc < best_score:
            # Quick refinement
            t_alloc = min(10.0, t1_end - time.monotonic())
            if t_alloc > 2:
                a_ref, sc_ref = projected_gradient_descent(init, t_alloc, T_start=0.1, T_end=0.01)
                best_seq, best_score = check_and_update(a_ref, f"init_{name}", best_seq, best_score)
            else:
                best_seq, best_score = check_and_update(init, f"init_{name}", best_seq, best_score)
    
    # Also try the parent with quick PGD
    if time.monotonic() < t1_end:
        t_alloc = min(20.0, t1_end - time.monotonic())
        if t_alloc > 5:
            print(f"  PGD on parent, budget={t_alloc:.0f}s")
            a_pg, sc_pg = projected_gradient_descent(best_seq, t_alloc, T_start=0.05, T_end=0.005)
            best_seq, best_score = check_and_update(a_pg, "pgd_parent", best_seq, best_score)
    
    # === Phase 2: Main projected gradient (500s) ===
    print(f"\n=== Phase 2: Projected Gradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 800
    t2_alloc = min(500.0, max(100.0, rem))
    print(f"  Budget: {t2_alloc:.0f}s")
    
    # Multiple runs with different temperature schedules
    schedules = [
        (0.1, 0.005),
        (0.05, 0.002),
        (0.03, 0.001),
    ]
    
    t_per_sched = t2_alloc / len(schedules)
    for T_s, T_e in schedules:
        rem = deadline - time.monotonic() - 800
        if rem < 20:
            break
        t_alloc = min(t_per_sched, rem * 0.5)
        print(f"  PGD T={T_s}->{T_e}, budget={t_alloc:.0f}s")
        a_pg, sc_pg = projected_gradient_descent(best_seq, t_alloc, T_start=T_s, T_end=T_e, verbose=True)
        best_seq, best_score = check_and_update(a_pg, f"pgd_{T_s}", best_seq, best_score)
    
    # === Phase 3: Peak shaving (300s) ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t3_alloc = min(300.0, rem)
        print(f"  Budget: {t3_alloc:.0f}s")
        a_ps, sc_ps = peak_shaving(best_seq, t3_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)
    
    # === Phase 4: L-BFGS multi-temperature (400s) ===
    print(f"\n=== Phase 4: L-BFGS Multi-T (best={best_score:.8f}) ===")
    temps = [100, 300, 1000, 3000]
    t4_total = 350.0
    t_per_T = t4_total / len(temps)
    for T_val in temps:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(t_per_T, rem * 0.6)
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        res = lbgfs_optimize(best_seq, t_alloc, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(res['a'], f"lbgfs_T{T_val}", best_seq, best_score)
    
    # === Phase 5: Coordinate descent (250s) ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t5_alloc = min(250.0, rem)
        print(f"  Budget: {t5_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t5_alloc, n_pairs=400, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)
    
    # === Phase 6: Nesterov + another peak shave (200s) ===
    print(f"\n=== Phase 6: Nesterov + Peak Shave (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t6a = min(100.0, rem * 0.5)
        if t6a > 10:
            print(f"  Nesterov budget={t6a:.0f}s")
            a_ne, sc_ne = nesterov_accelerated(best_seq, t6a, T=300, lr_init=0.01, momentum=0.94)
            best_seq, best_score = check_and_update(a_ne, "nesterov", best_seq, best_score)
    
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t6b = min(100.0, rem)
        print(f"  Peak Shave budget={t6b:.0f}s")
        a_ps2, sc_ps2 = peak_shaving(best_seq, t6b, verbose=True)
        best_seq, best_score = check_and_update(a_ps2, "peak_shave2", best_seq, best_score)
    
    # === Phase 7: Length exploration (150s) ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    nearby_ns = [n_main - 1, n_main + 1, n_main - 2, n_main + 2,
                 n_main + 4, n_main - 4]
    for alt_n in nearby_ns[:4]:
        if alt_n < 128 or alt_n > 2048:
            continue
        rem = deadline - time.monotonic() - 80
        if rem < 30:
            break
        t_alloc = min(60.0, rem * 0.4)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  L-BFGS n={alt_n}, budget={t_alloc:.0f}s")
        res = lbgfs_optimize(init_alt, t_alloc, T=500, maxiter=2000)
        best_seq, best_score = check_and_update(res['a'], f"lbgfs_n{alt_n}", best_seq, best_score)
    
    # === Phase 8: Final polish (100s) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_alloc = min(40.0, rem * 0.5)
        res = lbgfs_optimize(best_seq, t_alloc, T=2000, maxiter=2000)
        best_seq, best_score = check_and_update(res['a'], "final_lbgfs", best_seq, best_score)
    
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_alloc = min(30.0, rem * 0.6)
        a_cd2, _ = coordinate_descent(best_seq, t_alloc, n_pairs=500)
        best_seq, best_score = check_and_update(a_cd2, "final_cd", best_seq, best_score)
    
    rem = deadline - time.monotonic() - 10
    if rem > 5:
        t_alloc = min(20.0, rem * 0.7)
        a_ps3, _ = peak_shaving(best_seq, t_alloc)
        best_seq, best_score = check_and_update(a_ps3, "final_ps", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
