"""
Optimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP refinement with
sparse matrices, peak shaving via targeted coordinate swaps, multi-length
exploration, and adaptive time allocation. Builds on FFT-accelerated
subgradient descent and adds new peak-shaving and M-shrinking LP strategies.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_and_subgradient(a, w):
    """
    Compute auto-convolution of a and subgradient g[i] = sum_j a[j]*w[i+j].
    Returns (b, g) where b is length 2n-1 and g is length n.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    b_full = irfft(A * A, fft_size)
    b = b_full[:2 * n - 1]
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return b, g


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def subgradient_optimize_fft(a0, time_limit, lr_init=0.004, momentum=0.92,
                             n_top=None, seed=42, restart_interval=2000,
                             kick_size=0.002, decel_factor=0.93):
    """FFT-accelerated projected subgradient with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_local_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    it_count = 0

    if n_top is None:
        n_top = max(10, n // 10)

    while time.monotonic() - t0 < time_limit:
        b, _ = convolve_and_subgradient(a, np.zeros(2 * n - 1))
        b_max = b.max()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        top_vals = b[top_idx]
        threshold = b_max * 0.9995
        active_mask = top_vals >= threshold
        if not active_mask.any():
            active_mask[0] = True

        w = np.zeros(2 * n - 1)
        active_indices = top_idx[active_mask]
        w[active_indices] = 1.0 / len(active_indices)

        _, g = convolve_and_subgradient(a, w)
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        obj_new = eval_local_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it_count += 1

            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.05)
            no_improve = 0
            kick_size *= 0.6
        elif no_improve >= restart_interval // 2:
            cur_lr *= decel_factor
        elif no_improve >= restart_interval // 4:

def build_lp_matrix_sparse(a, M):
    """Build sparse LP constraint matrix for conv(a,g) <= M. Returns (A_sparse, b_vec)."""
    n = len(a)
    num_constraints = 2 * n - 1
    rows = []
    cols = []
    vals = []

    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(k - j)
            vals.append(a[j])

    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    b = np.full(num_constraints, M)
    return A, b


def lp_refine_aggressive(f, time_limit=60.0, max_iters=40, m_shrink=0.999, verbose=False):
    """
    Aggressive LP refinement with M shrinking. Each iteration:
    1. Compute current M = max(conv(a,a))
    2. Slightly reduce M target to M * m_shrink
    3. Solve LP: max sum(g) s.t. conv(a,g) <= M_target, g >= 0
    4. Blend and update
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_local_max_b(a)
    current_M = best_obj

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Rebuild sparse matrix with current a
        try:
            M_target = current_M * m_shrink
            A_sparse, b_vec = build_lp_matrix_sparse(a, M_target)
            c = -np.ones(n)

            res = linprog(c, A_ub=A_sparse, b_ub=b_vec,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={"presolve": True, "time_limit": min(10, time_limit - (time.monotonic() - t0))})

            if not res.success or res.x is None:
                # Try without shrinking M
                M_target = current_M
                A_sparse, b_vec = build_lp_matrix_sparse(a, M_target)
                res = linprog(c, A_ub=A_sparse, b_ub=b_vec,
                              bounds=[(0.0, None)] * n, method='highs',
                              options={"presolve": True, "time_limit": 5})
                if not res.success or res.x is None:
                    break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            # Adaptive blending
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]

            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = eval_local_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_local_max_b(a)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    current_M = obj
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.3f}, max_b={best_obj:.10f}")
                else:
                    a = best_a.copy()
                    current_M = best_obj
            else:
                # Try very small blend
                found = False
                for t in [0.01, 0.005, 0.002]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    blend = blend / blend.sum()
                    obj_blend = eval_local_max_b(blend)
                    if obj_blend < best_obj - 1e-14:
                        best_obj = obj_blend
                        best_a = blend.copy()
                        current_M = obj_blend
                        found = True
                        break
                if not found:
                    break

        except Exception as e:
            if verbose:
                print(f"    LP error: {e}")
            break

    return best_a, best_obj


def peak_shaving(a0, time_limit=30.0, n_peaks=5, verbose=False):
    """
    Targeted peak shaving: identify top peaks in convolution, then find
    coordinate pairs that contribute most to each peak and redistribute mass.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_local_max_b(a)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        if b_max < best_obj:
            best_obj = b_max
            best_a = a.copy()
            if verbose:
                print(f"    Peak shave: max_b={best_obj:.10f}")

        # Find top peaks
        top_idx = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            if b[k] < best_obj * 1.001:
                continue

            # Find the pair (i, j) with i+j=k that maximizes a[i]*a[j]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue

            best_pair = None
            best_contrib = 0
            for i in range(lo, hi + 1):
                j = k - i
                contrib = a[i] * a[j]
                if contrib > best_contrib:
                    best_contrib = contrib
                    best_pair = (i, j)

            if best_pair is None:
                continue

            i, j = best_pair
            # Try reducing the larger of a[i], a[j] and adding to a random low-contribution index
            if a[i] >= a[j]:
                src = i
            else:
                src = j

            # Find a target index with low contribution to all top peaks
            # Use an index near the edges (less likely to contribute to center peaks)
            candidates = []
            for cand in [0, 1, 2, n - 3, n - 2, n - 1, n // 2]:
                if cand != src and a[cand] < a[src] * 0.5:
                    candidates.append(cand)

            if not candidates:
                continue

            # Try small transfers
            for frac in [0.01, 0.02, 0.05, 0.1]:
                transfer = a[src] * frac
                a_tr = a.copy()
                a_tr[src] -= transfer
                a_tr[candidates[0]] += transfer
                a_tr = np.maximum(a_tr, 0.0)
                s = a_tr.sum()
                if s < 1e-10:
                    continue
                a_tr = a_tr / s
                obj_tr = eval_local_max_b(a_tr)
                if obj_tr < best_obj - 1e-14:
                    a = a_tr
                    improved = True
                    break

            if improved:
                break

        if not improved:
            # Try a broader search: random small perturbations
            rng_local = np.random.default_rng(hash(tuple(a[:10].round(6))) % 2**31)
            found_improve = False
            for _ in range(50):
                idx = rng_local.integers(0, n)
                shift = rng_local.integers(0, n)
                if shift >= n:
                    continue
                delta = a[idx] * 0.05 * rng_local.choice([-1, 1])
                a_tr = a.copy()
                a_tr[idx] += delta
                a_tr[(idx + shift) % n] -= delta
                a_tr = np.maximum(a_tr, 0.0)
                s = a_tr.sum()
                if s < 1e-10:
                    continue
                a_tr = a_tr / s
                obj_tr = eval_local_max_b(a_tr)
                if obj_tr < best_obj - 1e-14:
                    a = a_tr
                    found_improve = True
                    improved = True
                    break

            if not improved:
                break

    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit, multipliers=None):
    """Fast coordinate descent with FFT evaluation."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_local_max_b(a)

    if multipliers is None:
        multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0.0 or scale > 50.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_local_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    LP: {best_obj:.10f}")

    # Phase 3: Peak shaving
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.15, 30.0)
    if t3 > 3:
        a_ps, obj_ps = peak_shaving(best_a, time_limit=t3, n_peaks=5, verbose=verbose)
        if obj_ps < best_obj:
            best_obj = obj_ps
            best_a = a_ps.copy()
            if verbose:
                print(f"    PS: {best_obj:.10f}")

    # Phase 4: Second round SG + LP
    rem = time_budget - (time.monotonic() - t0)
    if rem > 15:
        t4_sg = min(rem * 0.4, 40.0)
        t4_lp = min(rem * 0.5, 60.0)
        if t4_sg > 5:
            a_sg2, obj_sg2 = subgradient_optimize_fft(best_a, time_limit=t4_sg,
                                                      seed=seed + 777,
                                                      lr_init=0.001, momentum=0.95,
                                                      n_top=max(20, n // 8))
            if obj_sg2 < best_obj:
                best_obj = obj_sg2
                best_a = a_sg2.copy()
        rem2 = time_budget - (time.monotonic() - t0)
        if rem2 > 5:
            a_lp2, obj_lp2 = lp_refine_aggressive(best_a, time_limit=min(rem2, t4_lp),
                                                  max_iters=20, m_shrink=0.9998,
                                                  verbose=verbose)
            if obj_lp2 < best_obj:
                best_obj = obj_lp2
                best_a = a_lp2.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses FFT subgradient, aggressive sparse LP, peak shaving,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_uniform_init(1152)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Aggressive optimization on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Aggressive optimization on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        print(f"  Time budget: {rem:.0f}s")
        a_opt, obj_opt = combined_optimize(parent, time_budget=rem, seed=seed, verbose=True)
        update_best(a_opt, "phase1")

    # === Phase 2: Different initializations at same length ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Alternative initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {
        "gauss_0.2": make_gaussian_init(n_base, 0.2),
        "gauss_0.35": make_gaussian_init(n_base, 0.35),
        "cosine": make_cosine_init(n_base),
        "tukey_0.4": make_tukey_init(n_base, 0.4),
    }

    # Also try perturbed parent
    perturbed = parent + rng.standard_normal(n_base) * 0.02
    perturbed = np.maximum(perturbed, 0)
    perturbed = perturbed / perturbed.sum()
    inits["perturbed_parent"] = perturbed

    total_time_diverse = min(250.0, (deadline - time.monotonic() - 400.0) * 0.35)
    time_per_init = total_time_diverse / len(inits)

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.25)
        if t_alloc < 8:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = combined_optimize(init_a, time_budget=t_alloc,
                                           seed=seed + hash(name) % 100000, verbose=False)
        update_best(a_div, f"div_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 0.6, 1.33, 0.8, 1.1]:
            tl = int(n_parent * factor)
            tl = max(256, min(3000, tl))
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 50.0)
            if t_alloc < 8:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = combined_optimize(f_interp, time_budget=t_alloc,
                                               seed=seed + target_n, verbose=False)
            update_best(a_len, f"len_n{target_n}")

    # === Phase 4: Deep optimization on best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep optimization (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_deep = min(rem * 0.6, 180.0)
            print(f"  Deep: n={n_best}, t={t_deep:.0f}s")
            a_deep, obj_deep = combined_optimize(best_seq, time_budget=t_deep,
                                                 seed=seed + 99999, verbose=True)
            update_best(a_deep, "deep")

            # Extra LP pass
            rem = deadline - time.monotonic() - 80
            if rem > 10:
                t_lp_extra = min(rem, 40.0)
                a_lp_extra, obj_lp_extra = lp_refine_aggressive(
                    best_seq, time_limit=t_lp_extra, max_iters=25,
                    m_shrink=0.9999, verbose=True)
                update_best(a_lp_extra, "deep_lp")

    # === Phase 5: Peak shaving + CD polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Peak shaving & CD (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_ps = min(rem * 0.4, 25.0)
            a_ps, obj_ps = peak_shaving(best_seq, time_limit=t_ps, n_peaks=8, verbose=True)
            update_best(a_ps, "ps")

            rem = deadline - time.monotonic() - 30
            if rem > 5:
                t_cd = min(rem, 20.0)
                a_cd, obj_cd = coordinate_descent_fast(best_seq, time_limit=t_cd)
                update_best(a_cd, "cd")

    # === Phase 6: Final micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            t_micro = min(rem * 0.4, 15.0)
            n_f = len(best_seq)
            a_micro, obj_micro = subgradient_optimize_fft(
                best_seq, time_limit=t_micro,
                lr_init=0.0002, momentum=0.96,
                n_top=max(30, n_f // 5), seed=seed + 777777)
            update_best(a_micro, "micro_sg")

            rem = deadline - time.monotonic() - 8
            if rem > 3:
                a_micro_lp, obj_micro_lp = lp_refine_aggressive(
                    best_seq, time_limit=min(rem, 10.0), max_iters=8, m_shrink=0.99999)
                update_best(a_micro_lp, "micro_lp")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
