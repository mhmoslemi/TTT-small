def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():

    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:

            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

    if s < 1e-15:

    rho = rho_candidates[-1]

        temp = 1.0 / (1 + 0.05 * t)
        w = np.exp((conv_b - conv_b.max()) / max(temp, 1e-6))
        dual = 0.95 * dual + 0.05 * w / w.sum()
        
        # Use weighted subgradient (weighted by dual)
        sg = np.zeros(n)
        # Use top-k peaks for efficiency
        k_top = min(20, n_peaks)
        top_k = np.argpartition(conv_b, -k_top)[-k_top:]
        weights = dual[top_k]
        weights = weights / weights.sum()
        for i in range(len(top_k)):
            k = int(top_k[i])
            if weights[i] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += 2.0 * weights[i] * b_new[k - j]
        
        # FTRL multiplicative update
        cum_grad += sg
        # Play: a_i ∝ exp(-cum_grad_i) / sum
        # But use the LP best response mixed with FTRL
        ftrl_a = np.exp(-0.5 * cum_grad)
        ftrl_a = np.maximum(ftrl_a, 1e-30)
        ftrl_a = ftrl_a / ftrl_a.sum()
        
        # Mix with LP best response
        alpha = 2.0 / (t + 2)  # Polyak step
        a = (1 - alpha) * a + alpha * b_new
        a = project_simplex(a)
        
        # Update running average
        cum_sum += a
        cum_count += 1
        running_avg = cum_sum / cum_count
        
        # Evaluate candidates
        candidates = [a, b_new, running_avg, ftrl_a, 0.5*(a+b_new)]
        for cand in candidates:
            s = cand.sum()
            if s < 1e-12:
                continue
            c = cand / s
            sc = fft_score(c)
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = c.copy()
            if sc < best_avg_score - 1e-15:
                best_avg_score = sc
                best_avg = c.copy()
        
        if verbose and t % 50 == 0:
            elapsed = monotonic() - t0
            print(f"  FP n={n} t={t} ({elapsed:.0f}s/M{monotonic()-t0/max(t,1)*1000:.1f}ms/it): best={best_score:.6f} avg={best_avg_score:.6f}")
    
    return best_a, best_score, best_avg, best_avg_score

def polyak_subgrad(a0, time_limit, f_star_est=None, max_iters=100000):
    """
    Projected subgradient descent with Polyak step size.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = fft_score(a)
    t0 = monotonic()
    
    if f_star_est is None:
        f_star_est = best_score * 0.9  # optimistic estimate
    
    t = 0
    no_improve = 0
    
    while monotonic() - t0 < time_limit and t < max_iters:
        t += 1
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Subgradient of max(conv(a,a)) at a
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            sg[j] += 2.0 * a[k_star - j]
        
        # Full objective gradient: 2*n*sg (since we minimize 2*n*max_conv)
        grad = 2.0 * n * sg
        
        # Polyak step
        current_val = 2.0 * n * b[k_star]
        gap = current_val - f_star_est
        sq_grad = np.dot(grad, grad)
        if sq_grad < 1e-30:
            break
        eta = gap / sq_grad
        eta = min(eta, 10.0 / n)  # cap the step
        
        a_new = project_simplex(a - eta * grad)
        score = fft_score(a_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            # Update f_star estimate
            f_star_est = min(f_star_est, score * 0.95)
        else:
            no_improve += 1
        
        a = a_new
        
        if t % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"  PolyakSG t={t} ({elapsed:.0f}s): score={best_score:.6f}")
        
        if no_improve > 5000:
            # Reset with perturbation
            a = best_a + np.random.standard_normal(n) * 0.001
            a = project_simplex(a)
            f_star_est = fft_score(a) * 0.9
            no_improve = 0
    
    return best_a, best_score

def peak_shave_refined(a0, time_limit, max_rounds=5000):
    """
    Refined peak shaving: identify contributing pairs at the peak,
    and systematically reduce them while redistributing mass.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = fft_score(a)
    t0 = monotonic()
    
    for rnd in range(max_rounds):
        if monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        peak_val = b[k_peak]
        
        if peak_val <= best_score / (2.0 * n) + 1e-16:
            break
        
        # Get contributing pairs
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        pair_vals = a[idx] * a[k_peak - idx]
        total = pair_vals.sum()
        
        if total < 1e-20:
            break
        
        # Sort by contribution (descending)
        order = np.argsort(pair_vals)[::-1]
        
        improved = False
        for ci in order[:10]:
            i = int(idx[ci])
            j = k_peak - i
            if i == j:
                # Single element contributes a[i]^2
                # Reduce a[i]
                for frac in [0.9, 0.8, 0.7, 0.6]:
                    a_trial = a.copy()
                    old = a_trial[i]
                    a_trial[i] = old * frac
                    diff = old - a_trial[i]
                    # Redistribute to elements far from peak
                    mask = np.ones(n, dtype=bool)
                    # Exclude indices that contribute to the peak
                    for jj in range(js, je + 1):
                        if a[jj] > 1e-10:
                            mask[jj] = False
                    # Also exclude i and j from receiving
                    mask[i] = False
                    if j < n:
                        mask[j] = False
                    if mask.any():
                        a_trial[mask] += diff * a[mask] / a[mask].sum()
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = fft_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Pair (i, j)
                for f1 in [0.9, 0.8, 0.7]:
                    for f2 in [0.9, 0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s < 1e-15:
                            break
                        a_trial /= s
                        sc = fft_score(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            # Try random perturbation
            for _ in range(3):
                a_trial = a + np.random.standard_normal(n) * 0.0003
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = fft_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                if rnd > 100:
                    break
    
    return best_a, best_score

def lbgfs_smooth(a0, time_limit, T=50, maxiter=3000):
    """L-BFGS with log-sum-exp smoothing of the max."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 -= z0.max()
    t0 = monotonic()
    stopped = [False]
    
    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        az = z - z.max()
        ez = np.exp(az)
        a = ez / ez.sum()
        b = fast_convolve(a)
        mb = b.max()
        sb = (b - mb) / T
        eb = np.exp(sb)
        lse = mb + T * np.log(eb.sum())
        return 2.0 * n * lse / T
    
    def jac(z):
        if stopped[0]:
            return np.zeros(n)
        az = z - z.max()
        ez = np.exp(az)
        a = ez / ez.sum()
        b = fast_convolve(a)
        mb = b.max()
        sb = (b - mb) / T
        eb = np.exp(sb)
        w = eb / eb.sum()
        
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            wk = w[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * wk * a[k - idx]
        
        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        return a * (grad_a - S * a)
    
    try:
        res = minimize(func, z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxfun': maxiter * 10})
        if not stopped[0]:
            zf = res.x
            zf -= zf.max()
            a_out = np.exp(zf)
            a_out = a_out / a_out.sum()
            return a_out
    except Exception:
        pass
    return a0

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi
