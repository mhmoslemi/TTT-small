import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_local_score(a):
    """Fast score assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_gradient_fft(a, w, n, fft_size):
    """
    Compute gradient of sum_k w[k]*(a*a)[k] w.r.t. a.
    grad[p] = 2 * sum_j a[j] * w[j+p] for p=0..n-1
    Using FFT: grad = 2 * IFFT(conj(FFT(a)) * FFT(w))[:n]
    """
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * corr


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='cosine',
                  verbose=False, verbose_interval=5000):
    """Adam optimizer in log-space with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'cosine':
            T = T_final + 0.5 * (T_init - T_final) * (1 + np.cos(np.pi * progress))
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.2:
                T = T_init
            else:
                p2 = (progress - 0.2) / 0.8
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init + (T_final - T_init) * progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        grad_a = compute_gradient_fft(a_curr, w, n, fft_size)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

            best_a = a_new.copy()
            no_improve = 0

            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.0001, lr * 0.8)

    return best_a, best_score


def peak_pair_search(a0, time_limit, max_trials=20000):
    """
    Peak-pair transport local search.
    Identifies convolution peaks and redistributes mass from peak-contributing
    positions to 'cold' positions to reduce the max convolution value.
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), get_local_score(a0)
    a = a / s

    best_a = a.copy()
    best_score = get_local_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(999)

    for trial in range(max_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        target = best_score / (2.0 * n)

        if max_b <= target * 1.0001:
            continue

        # Find hot positions (above 95% of max)
        threshold = max_b * 0.95
        hot_k = np.where(b > threshold)[0]

        # Find cold positions (below 50% of mean)
        mean_b = 1.0 / (2 * n - 1)
        cold_k = np.where(b < mean_b * 0.5)[0]
        if len(cold_k) == 0:
            cold_k = np.where(b < mean_b)[0]
        if len(cold_k) == 0:
            continue

        improved = False
        for k in hot_k[:3]:
            if improved:
                break

            # Find contributing pairs for this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue

            # Top contributors
            top_frac = np.argsort(contribs)[::-1][:8]

            for ci in top_frac:
                i = int(idx[ci])
                j = int(k - i)
                if a[i] < 1e-10 or a[j] < 1e-10:
                    continue

                # Try moving mass from a[i] (or a[j]) to a cold position
                # Choose the one to reduce (the larger one that's more "peak-contributing")
                if a[i] >= a[j]:
                    donor = i
                    partner = j
                else:
                    donor = j
                    partner = i

                # Choose a cold position that doesn't create new peaks
                # Prefer cold positions far from both donor and partner
                dist_to_donor = np.abs(cold_k - donor)
                dist_to_partner = np.abs(cold_k - partner)
                min_dist = np.minimum(dist_to_donor, dist_to_partner)
                # Weight by distance (prefer far) and by how cold the position is
                coldness = mean_b - b[cold_k]
                score_cold = min_dist / (max_dist := min_dist.max() + 1) * 0.5 + coldness / (coldness.max() + 1e-30) * 0.5
                cold_sorted = cold_k[np.argsort(score_cold)[::-1]][:5]

                for cold_pos in cold_sorted:
                    # Try different transfer amounts
                    for frac in [0.05, 0.1, 0.15, 0.2]:
                        delta = a[donor] * frac
                        if delta < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[donor] -= delta
                        a_trial[cold_pos] += delta
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        sc = get_local_score(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Periodic random perturbation to escape local minima
        if not improved and trial % 200 == 199:
            n_pert = rng.integers(1, max(2, n // 50))
            perturb_positions = rng.choice(n, size=n_pert, replace=False)
            deltas = rng.standard_normal(n_pert) * 0.02
            for pos, d in zip(perturb_positions, deltas):
                a[pos] = max(0, a[pos] + d)
            a = a / a.sum()

    return best_a, best_score
