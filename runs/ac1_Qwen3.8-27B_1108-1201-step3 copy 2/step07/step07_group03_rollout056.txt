def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def normalized_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    b = fast_convolve(a_norm)
    return 2.0 * n * b.max()

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.ECOS, cp.PDLP, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):

    s = b_val.sum()

    iters = 0

    for t_val in t_schedule:
        if time.monotonic() - t0_time > time_limit:
            break
        
        def obj(z):
            return smoothed_objective(z, t_val)
        
        def grad(z):
            return smoothed_gradient(z, t_val)
        
        try:
            result = minimize(obj, z0, jac=grad, method='L-BFGS-B',
                             options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8})
            z_new = result.x
            a_new = softmax_from_z(z_new)
            score_new = 2.0 * n * fast_convolve(a_new).max()
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            z0 = z_new
            if verbose:
                elapsed = time.monotonic() - t0_time
                print(f"    SmoothLBFGS t={t_val} ({elapsed:.0f}s): score={score_new:.6f}")
        except Exception as e:
            if verbose:
                print(f"    SmoothLBFGS t={t_val} failed: {e}")
            continue
    
    return best_a, best_score

def subgradient_descent(a0, time_limit, top_k=30, lr_init=0.003, verbose=False):
    """Projected subgradient method with softmax-tempered subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0_time = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    lr = lr_init
    
    it = 0
    while time.monotonic() - t0_time < time_limit:
        it += 1
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            sg[j] += 2.0 * w[idx] * a[k - j]
        
        # Project subgradient onto simplex tangent space
        sg_proj = sg - np.mean(sg)
        
        # Nesterov momentum
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2.0
        beta_k = (t_k - 1) / t_k
        if not hasattr(subgradient_descent, '_v'):
            subgradient_descent._v = np.zeros(n)
        v = subgradient_descent._v
        v = beta_k * v + sg_proj
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v
        
        a_new = a - lr * v
        # Project onto simplex
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
        if no_improve > 400:
            noise = rng.standard_normal(n) * 0.02
            a = np.maximum(best_a + noise, 0)
            a = a / a.sum()
            subgradient_descent._v = np.zeros(n)
            lr = max(lr * 1.5, 0.002)
            no_improve = 0
        
        lr *= 0.9998
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0_time
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")
    
    del subgradient_descent._v
    return best_a, best_score

def coord_descent_analytical(a0, time_limit, n_sweeps=100):
    """Coordinate descent with golden-section line search on each coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0_time = time.monotonic()
    rng = np.random.default_rng(777)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0_time > time_limit:
            break
        improved_any = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0_time > time_limit:
                break
            if a[i] < 1e-15:
                # Try setting it to a small positive value
                for trial_x in [0.001 / n, 0.005 / n, 0.01 / n]:
                    trial = a.copy()
                    trial[i] = trial_x
                    trial = trial / trial.sum()
                    score = 2.0 * n * fast_convolve(trial).max()
                    if score < best_score - 1e-13:
                        a = trial
                        best_score = score
                        best_a = a.copy()
                        improved_any = True
                continue
            
            # Golden section search on coordinate i
            # We try to find optimal x = a[i] in range [0, 2*a[i]]
            x_lo = 0.0
            x_hi = 2.0 * a[i] + 1.0 / n
            
            # Evaluate at golden section points
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            f_c1 = best_score  # cache
            f_c2 = best_score
            c1 = x_hi - gr * (x_hi - x_lo)
            c2 = x_lo + gr * (x_hi - x_lo)
            
            trial1 = a.copy()
            trial1[i] = c1
            trial1 = trial1 / trial1.sum()
            f1 = 2.0 * n * fast_convolve(trial1).max()
            
            trial2 = a.copy()
            trial2[i] = c2
            trial2 = trial2 / trial2.sum()
            f2 = 2.0 * n * fast_convolve(trial2).max()
            
            for _ in range(15):
                if f1 < f2:
                    x_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = x_hi - gr * (x_hi - x_lo)
                    t_c = a.copy()
                    t_c[i] = c1
                    t_c = t_c / t_c.sum()
                    f1 = 2.0 * n * fast_convolve(t_c).max()
                else:
                    x_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = x_lo + gr * (x_hi - x_lo)
                    t_c = a.copy()
                    t_c[i] = c2
                    t_c = t_c / t_c.sum()
                    f2 = 2.0 * n * fast_convolve(t_c).max()
            
            # Best point is the midpoint
            x_best = (x_lo + x_hi) / 2
            trial_best = a.copy()
            trial_best[i] = x_best
            trial_best = trial_best / trial_best.sum()
            f_best = 2.0 * n * fast_convolve(trial_best).max()
            
            if f_best < best_score - 1e-13:
                a = trial_best
                best_score = f_best
                best_a = a.copy()
                improved_any = True
        
        if not improved_any and sweep > 10:
            break
    
    return best_a, best_score

def active_set_newton(a0, time_limit, m_max=10):
    """Active set refinement: enforce equality of top m convolution values."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0_time = time.monotonic()
    
    for iteration in range(200):
        if time.monotonic() - t0_time > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        
        # Identify active set: lags within 2% of max
        threshold = b_max * 0.98
        active_mask = b >= threshold
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) < 2:
            active_lags = np.argpartition(b, -5)[-5:]
        active_lags = active_lags[:m_max]
        
        m = len(active_lags)
        
        # Solve: find a on simplex such that (a*a)[k_1] = ... = (a*a)[k_m] = M (minimized)
        # Use scipy minimize with constraints
        a_flat = a.copy()
        
        def obj_fn(x):
            x = np.maximum(x, 0)
            x = x / x.sum()
            b_x = fast_convolve(x)
            # Maximize the minimum gap from max (i.e., minimize max)
            return b_x.max()
        
        def constraint_eq(x):
            """Equality constraints: conv values at active lags are equal."""
            x = np.maximum(x, 0)
            x = x / x.sum()
            b_x = fast_convolve(x)
            vals = b_x[active_lags]
            return vals - vals[0]
        
        constraints = []
        for j in range(1, min(m, 5)):
            constraints.append({'type': 'eq', 'fun': lambda x, j=j: fast_convolve(
                np.maximum(x, 0) / np.maximum(np.sum(np.maximum(x, 0)), 1e-15))[active_lags[j]] -
                fast_convolve(np.maximum(x, 0) / np.maximum(np.sum(np.maximum(x, 0)), 1e-15))[active_lags[0]]
            })
        
        bounds = [(0, 1)] * n
        
        try:
            result = minimize(obj_fn, a_flat, method='SLSQP', bounds=bounds,
                            constraints=constraints, options={'maxiter': 50, 'ftol': 1e-10})
            if result.success:
                a_new = np.maximum(result.x, 0)
                a_new = a_new / a_new.sum()
                score_new = 2.0 * n * fast_convolve(a_new).max()
                if score_new < best_score - 1e-12:
                    best_score = score_new
                    best_a = a_new.copy()
                    a = a_new
        except Exception:
            pass
        
        # If no improvement for a while, try with smaller active set
        if iteration % 20 == 19:
            active_lags = active_lags[:max(2, m // 2)]
    
    return best_a, best_score

def peak_shave_refined(a0, time_limit):
    """Targeted peak shaving: reduce the highest peak by adjusting contributing coords."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0_time = time.monotonic()
    
    for step in range(2000):
        if time.monotonic() - t0_time > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.max() < 1e-15:
            break
        
        # Try reducing top contributors
        top_c = np.argsort(contribs)[::-1][:8]
        improved = False
        
        for ci in top_c:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            # Try reducing i
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                trial = a.copy()
                trial[i] *= factor
                trial = trial / trial.sum()
                score = 2.0 * n * fast_convolve(trial).max()
                if score < best_score - 1e-12:
                    a = trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
            
            # Try reducing j (if different)
            if i != j and a[j] >= 1e-14:
                for factor in [0.95, 0.9, 0.85]:
                    trial = a.copy()
                    trial[j] *= factor
                    trial = trial / trial.sum()
                    score = 2.0 * n * fast_convolve(trial).max()
                    if score < best_score - 1e-12:
                        a = trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            # Try simultaneous reduction of top 2 contributors
            if len(top_c) >= 2:
                i1 = int(idx[top_c[0]])
                i2 = int(idx[top_c[1]])
                for f1 in [0.9, 0.8]:
                    for f2 in [0.9, 0.8]:
                        trial = a.copy()
                        trial[i1] *= f1
                        trial[i2] *= f2
                        trial = trial / trial.sum()
                        score = 2.0 * n * fast_convolve(trial).max()
                        if score < best_score - 1e-12:
                            a = trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_blocks(n, rng):
    """Initialize with random block amplitudes."""
    n_blocks = int(rng.integers(4, 30))
    block_size = max(2, n // n_blocks)
    amplitudes = rng.random(n_blocks) + 0.1
    a = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        a[start:end] = amplitudes[b]
    kernel = np.ones(min(5, block_size)) / min(5, block_size)
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_noise(n, base, rng, sigma=0.08):
    """Add Gaussian noise to a base sequence."""
    a = base.copy() + rng.standard_normal(n) * sigma
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

            a_f1, _ = subgradient_descent(best_seq, min(rem * 0.4, 30.0), top_k=20, lr_init=0.002)
            best_seq, best_score = check_and_update(a_f1, best_seq, best_score, "final_sg")
        
        # Quick coord descent
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_f2, _ = coord_descent_analytical(best_seq, min(rem * 0.5, 20.0), n_sweeps=15)
            best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
