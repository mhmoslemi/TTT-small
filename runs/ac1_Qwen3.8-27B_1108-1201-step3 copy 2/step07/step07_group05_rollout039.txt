```python
def simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. (2008)
    if v.sum() <= 1 and (v >= 0).all():
        # Already in simplex (if sum <= 1, we need to add to make sum = 1)
        # Actually we need sum = 1 exactly
        pass
    # Standard projection
    mu = 0.0
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.arange(1, n+1)
    cond = sorted_v - (cssv - 1) / rho > 0
    if not cond.any():
        mu = (v.sum() + 1) / n  # fallback
    else:
        rho_max = rho[cond][-1]
        mu = (cssv[cond][-1] - 1) / rho_max
    result = np.maximum(v - mu, 0)
    # Ensure sum = 1 (numerical fix)
    result = result / result.sum()
    return result
```

```python
    cond = u - (cssv - 1.0) / rho > 0
    if not cond.any():
        return np.ones(n) / n

    theta = (cssv[cond][-1] - 1.0) / rho_max
    result = np.maximum(v - theta, 0)
    result = result / result.sum()
    return result
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):

    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not cond.any():
        return np.ones(n) / n
    rho_max = rho[cond][-1]
    theta = (cssv[cond][-1] - 1.0) / rho_max
    result = np.maximum(v - theta, 0)
    result = result / result.sum()
    return result

def lp_best_response(f, M):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns normalized g (sum=1) or None if infeasible.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build sparse constraint matrix
    A = lil_matrix((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    A = A.tocsr()
    
    b = M * np.ones(num_constraints)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            return None
        return g / g_sum
    except Exception:
        return None
