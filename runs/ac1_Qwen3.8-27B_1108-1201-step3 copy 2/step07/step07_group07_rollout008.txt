Let me code this up:

```python
def active_set_lp(a, M, max_iters=100, time_limit=30.0):
    """
    Solve max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Use active-set/cutting-plane method.
    """
    n = len(a)
    t0 = time.monotonic()
    
    # Start with no constraints (just g >= 0)
    # The initial solution is g = 0 (objective = 0)
    # Add constraints one by one
    
    active = set()  # indices of active constraints
    g = np.zeros(n)
    
    for _ in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP with current active constraints
        # max sum(g) s.t. conv(a,g)[k] <= M for k in active, g >= 0
        num_active = len(active)
        if num_active == 0:
            # No constraints: g can be anything non-negative
            # This is unbounded, so we need at least one constraint
            # Add the constraint at the peak of conv(a, ones)
            conv_ones = np.convolve(a, np.ones(n))
            peak_k = np.argmax(conv_ones)
            active.add(peak_k)
            continue
        
        # Build constraint matrix for active constraints
        # Row k: sum_j a[j] * g[k-j] <= M (with appropriate bounds on k-j)
        A = np.zeros((num_active, n))
        active_list = sorted(active)
        for idx, k in enumerate(active_list):
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    A[idx, j] = a[i]
        
        c = -np.ones(n)
        b_constraints = np.full(num_active, M)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_constraints,
                         bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
        except Exception:
            break
        
        g = res.x
        
        # Check for violated constraints
        conv_ag = np.convolve(a, g)
        violations = np.where(conv_ag > M + 1e-10)[0]
        
        if len(violations) == 0:
            break
        
        # Add the most violated constraint
        worst = violations[np.argmax(conv_ag[violations])]
        active.add(worst)
    
    return g
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP refinement,
FFT-accelerated subgradient descent with dual-guided directions,
mirror descent, and coordinate descent polishing.

Key innovations:
- Active-set (cutting-plane) LP for fast and accurate direction finding
- Dual-based subgradient for principled optimization
- Mirror descent (KL projection) for better simplex geometry
- Multi-length exploration with interpolation
- Adaptive time allocation across phases

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_var(a, v):
    """Compute conv(a, v) via FFT for v of length n."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    V = rfft(v, fft_size)
    return irfft(A * V, fft_size)[:size]


def project_simplex(v):
    """Projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_max_b(a):
    """Return max of auto-convolution."""
    return float(fast_conv(a).max())


def build_conv_row(a, k):
    """Build row k of the convolution constraint matrix."""
    n = len(a)
    row = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for j in range(lo, hi + 1):
        row[j] = a[k - j]
    return row


def active_set_lp_solve(a, M, time_limit=5.0, max_constraints=300):
    """
    Solve max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Uses cutting-plane (active set) method.
    Returns (g, converged).
    """

    t0 = time.monotonic()
    
    if n == 0:
        return np.array([]), True
    
    # Start with the most "binding" constraint
    conv_ones = fast_conv(np.concatenate([a, np.zeros(n-1)])) 
    # Actually, conv(a, ones(n)) is just the sliding sum
    # Let me compute it properly
    ones = np.ones(n)
    conv_ones = conv_var(a, ones)
    peak_k = int(np.argmax(conv_ones))
    
    active = [peak_k]
    g = np.zeros(n)
    
    for _ in range(max_constraints + 10):
        if time.monotonic() - t0 > time_limit:
            return g, False
        if len(active) > max_constraints:
            return g, False
        
        m = len(active)
        A = np.zeros((m, n))
        for idx, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[idx, j] = a[k - j]
        
        c = -np.ones(n)
        b_ub = np.full(m, M)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                         bounds=[(0.0, None)] * n, method='highs',
                         options={'presolve': True, 'time_limit': max(0.1, time_limit)})
            if not res.success:
                return g, False
        except Exception:
            return g, False
        
        g_new = res.x
        g_sum = g_new.sum()
        
        if g_sum < 1e-12:
            return g_new, True
        
        # Check for violated constraints
        conv_ag = conv_var(a, g_new)
        max_violation = conv_ag.max() - M

                break

def lp_refine(a, time_limit=60.0, max_outer=50):
    """
    LP-based refinement: iteratively solve LP and blend.
    Returns (best_a, best_obj).
    """
    t0 = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_obj = get_max_b(a)
    
    for outer in range(max_outer):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = get_max_b(best_a)
        
        # Solve LP
        g, converged = active_set_lp_solve(best_a, M,
                                           time_limit=min(5.0, time_limit - (time.monotonic() - t0)),
                                           max_constraints=200)
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum
        
        # Check if g is close to best_a
        if np.abs(g_norm - best_a).max() < 1e-8:
            break
        
        # Line search for blending
        best_t = 0.0
        best_blend_obj = best_obj
        
        # Try golden section or just a grid
        t_vals = np.concatenate([
            np.array([0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0])
        ])
        
        for t in t_vals:
            blend = (1.0 - t) * best_a + t * g_norm
            blend = np.maximum(blend, 0.0)
            bs = blend.sum()
            if bs < 1e-10:
                continue
            blend = blend / bs
            obj = get_max_b(blend)
            if obj < best_blend_obj:
                best_blend_obj = obj
                best_t = t
        
        if best_t > 0.0:
            a_new = (1.0 - best_t) * best_a + best_t * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            obj_new = get_max_b(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    LP iter {outer}: t={best_t:.3f}, max_b={best_obj:.10f}")
        else:
            break
    
    return best_a, best_obj
