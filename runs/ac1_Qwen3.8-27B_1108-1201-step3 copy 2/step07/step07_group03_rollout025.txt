        i_end = min(n - 1, int(k))
        if i_start <= i_end:

def mirror_descent_update(a, subgrad, eta):
    """Mirror descent step using negative entropy on the simplex."""
    log_a = np.log(np.maximum(a, 1e-300))
    logits = log_a - eta * subgrad
    logits -= logits.max()
    a_new = np.exp(logits)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def symmetrize_normalize(a):

    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a_sym / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_blocks(n, rng):
    """Initialize with random block amplitudes and smoothing."""

    amplitudes = rng.random(n_blocks) + 0.1
    a = np.zeros(n)

        end = min(start + block_size, n)
        a[start:end] = amplitudes[b]

        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:

    mask = rng.random(n) < density
    a[mask] = rng.random(int(max(mask.sum(), 1))) + 0.1
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_noise(n, base, rng, sigma=0.1):
    """Add Gaussian noise to a base sequence."""
    a = base.copy() + rng.standard_normal(n) * sigma
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def evaluate_score(a):
    """Compute the objective value for a normalized sequence (sum=1)."""
    n = len(a)
    if n == 0 or a.sum() < 0.01:
        return float('inf')
    a_norm = a / a.sum()
    return float(2.0 * n * fast_convolve(a_norm).max())


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using the official evaluator and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def lp_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False, symmetrize_each=True):
    """LP-based fictitious play with optional symmetrization."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = evaluate_score(a)
    iters = 0
    last_best_iter = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
            if symmetrize_each:
                a = symmetrize_normalize(a)

        a_avg = (iters * a_avg + a) / (iters + 1)

        candidates = [a, b, 0.5 * (a + b), a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = csum = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = evaluate_score(cand_norm)
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()
                last_best_iter = iters

        if verbose and iters % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def lp_mirror_hybrid(n, time_limit, initial=None, solver=None, verbose=False):
    """Hybrid: LP best response + mirror descent with dual-guided subgradient."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    best_a = a.copy()
    best_score = evaluate_score(a)
    it = 0
    temperature = 0.05
    eta = 0.005
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # LP best response
        b_opt, M_opt, dual_w = solve_lp_best_response(a, solver)
        if b_opt is not None:
            # Move towards LP solution with damped step
            alpha = min(1.0, 5.0 / (it + 10))
            a_lp = (1 - alpha) * a + alpha * b_opt
            a_lp = np.maximum(a_lp, 0)
            a_lp = a_lp / a_lp.sum()

            # Mirror descent step using dual-guided subgradient
            if dual_w is not None:
                sg = compute_subgrad_from_dual(a, dual_w)
            else:
                sg = compute_subgrad_all_peaks(a, temperature)

            a_md = mirror_descent_update(a, sg, eta)

            # Combine LP and MD steps
            a_new = 0.6 * a_lp + 0.4 * a_md
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

            # Symmetrize
            a_new = symmetrize_normalize(a_new)

            score = evaluate_score(a_new)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            a = a_new

        # Anneal temperature and eta
        temperature *= 0.995
        eta = 5.0 / (it + 20)

        if no_improve > 80:
            # Add small noise to escape
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.005
            a = simplex_project(a + noise)
            a = symmetrize_normalize(a)
            no_improve = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP+MD it={it} ({elapsed:.0f}s): best={best_score:.6f} temp={temperature:.4f}")

    return best_a, best_score


def nesterov_optimize(a0, time_limit, top_k=25, lr_init=0.006, use_symmetry=True):
    """Nesterov-accelerated projected subgradient with all-peak softmax."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temperature = 0.03

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        sg = compute_subgrad_all_peaks(y, temperature)
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        y_new = y - lr * v
        y_projected = simplex_project(y_new)

        if use_symmetry:
            y_projected = symmetrize_normalize(y_projected)

        score = evaluate_score(y_projected)
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_projected

        if no_improve > 100:
            lr *= 0.85
            no_improve = 0

        if no_improve > 350:
            rng = np.random.default_rng(it * 13 + 5)
            noise = rng.standard_normal(n) * 0.008
            y = simplex_project(best_a + noise)
            if use_symmetry:
                y = symmetrize_normalize(y)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 1.5)
            no_improve = 0

        lr *= 0.9995
        temperature = 0.03 * (0.999 ** it)

    return best_a, best_score


def coord_descent_polish(a0, time_limit, n_sweeps=300):
    """Full coordinate descent with adaptive factors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[i] = new_val
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    a = a / a.sum()
                    continue
                a = a / s
                score = evaluate_score(a)
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()

        if not improved and sweep > 10:
            # Try finer perturbations
            any_small = False
            for i in rng.choice(n, size=min(n, 150), replace=False):
                old_val = a[i]
                if old_val < 1e-14:
                    continue
                for factor in [0.97, 0.985, 1.015, 1.03]:
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-15:
                        a[i] = old_val
                        continue
                    a = a / s
                    score = evaluate_score(a)
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        old_val = a[i]
                        improved = True
                        any_small = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()
                if not improved and i != n - 1 - i:
                    j = n - 1 - i
                    old_val_j = a[j]
                    if old_val_j >= 1e-14:
                        for f1 in [0.93, 0.96]:
                            for f2 in [0.93, 0.96]:
                                a[i] = a[i] * f1
                                a[j] = old_val_j * f2
                                a = np.maximum(a, 0)
                                s = a.sum()
                                if s < 1e-15:
                                    a[i] = a[i] / f1
                                    a[j] = old_val_j
                                    continue
                                a = a / s
                                score = evaluate_score(a)
                                if score < best_score - 1e-14:
                                    best_score = score
                                    best_a = a.copy()
                                    improved = True
                                    any_small = True
                                    break
                            if improved:
                                break
                        if not improved:
                            a[i] = a[i]
                            a[j] = old_val_j
                            a = a / a.sum()
            if not any_small and sweep > 30:
                break

    return best_a, best_score


def peak_targeted_shave(a0, time_limit):
    """Focus on reducing the highest peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()

    for step in range(3000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        top_k = min(8, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        peak_k = int(peaks[0])

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.max() < 1e-15:
            break

        top_contrib = np.argsort(contribs)[::-1][:6]
        improved = False

        for ci in top_contrib:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.92, 0.85, 0.75, 0.65]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_score(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if not improved and i != j and a[j] >= 1e-14:
                for f1 in [0.88, 0.8, 0.7]:
                    for f2 in [0.88, 0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        score = evaluate_score(a_trial)
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score


def propose_candidate():

    Multi-phase optimization combining LP fictitious play, mirror descent,
    coordinate descent, and multi-length exploration.

        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    print(f"\n=== Phase 1: Diverse Init (best={best_score:.6f}) ===")

                cand = init_noise(n, base, rng, sigma=0.1)

    print(f"  Generated {len(candidates)} candidates, top: {candidates[0][0]:.6f}")

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
