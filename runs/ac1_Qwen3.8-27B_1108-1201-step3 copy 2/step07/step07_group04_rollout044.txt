import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_solver_cache():
    global _lp_solver
    if '_lp_solver' not in globals():
        for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS]:
            try:
                x = cp.Variable(3, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    _lp_solver = solver_name
                    return solver_name
            except Exception:
                continue
        _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = _lp_solver_cache()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    a_avg = a.copy().astype(np.float64)
    iters = 0

    while monotonic() - t0 < time_limit and iters < max_iters:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:

        a_new, _ = solve_lp_best_response(b_new, solver)
        if a_new is None:

        a_avg = (iters * a_avg + 0.5 * (a_new + b_new)) / (iters + 1)

def compute_smoothed_obj_and_grad(z, T, n):
    """Compute smoothed objective and gradient in log-space.
    z: log-parameters, a = softmax(z)
    Returns: objective value, gradient
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled_b = (b - max_b) / T
    exp_b = np.exp(scaled_b)
    w = exp_b / exp_b.sum()
    lse = max_b + T * np.log(exp_b.sum())
    obj = 2.0 * n * lse

    # Gradient: grad_a = 2 * sum_k w_k * a_shifted_k
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    # Transform to z-space: grad_z = a * (grad_a - a @ grad_a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def adam_optimize(a0, time_limit, T_schedule=None, max_iter=50000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on smoothed objective in log-space with temperature annealing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z = np.log(a_safe)
    z = z - z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = monotonic()

    if T_schedule is None:
        T_schedule = [200.0, 80.0, 30.0, 10.0, 5.0, 2.0, 1.0]

    # Divide time budget among temperature phases
    n_phases = len(T_schedule)
    time_per_phase = time_limit / n_phases * 1.3  # overlap

    for phase, T in enumerate(T_schedule):
        phase_end = min(monotonic() - t0 + time_per_phase, time_limit)
        while monotonic() - t0 < phase_end and t < max_iter:
            t += 1
            obj, grad = compute_smoothed_obj_and_grad(z, T, n)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Normalize z for numerical stability
            if t % 100 == 0:
                z = z - z.mean()

            # Compute actual sequence and score
            z_shifted = z - z.max()
            a = np.exp(z_shifted)
            a = a / a.sum()
            actual_score = 2.0 * n * fast_convolve(a).max()

            if actual_score < best_score - 1e-12:
                best_score = actual_score
                best_a = a.copy()

            if t % 5000 == 0:
                elapsed = monotonic() - t0
                print(f"    Adam T={T:.0f} iter={t} ({elapsed:.0f}s): smooth={obj:.6f} actual={best_score:.6f}")

    return best_a, best_score

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0

    for rnd in range(max_rounds):
        if monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_val = b.max()
        target = best_score / (2.0 * n)

        if peak_val <= target + 1e-14:
            break

        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()

        if total_contrib < 1e-20:
            break

        improved = False

        # Strategy 1: Reduce top contributor
        top_idx = idx[np.argmax(contribs)]
        for factor in [0.97, 0.95, 0.92, 0.9, 0.85, 0.8, 0.7]:
            a_trial = a.copy()
            old = a_trial[top_idx]
            a_trial[top_idx] = old * factor
            diff = old - a_trial[top_idx]
            mask = np.ones(n, dtype=bool)
            mask[top_idx] = False
            others_sum = a_trial[mask].sum()
            if others_sum > 1e-15:
                a_trial[mask] += diff * a_trial[mask] / others_sum
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        # Strategy 2: Reduce top-2 contributors simultaneously
        if not improved:
            top2 = np.argsort(contribs)[::-1][:2]
            i1, i2 = idx[top2[0]], idx[top2[1]]
            for f1 in [0.95, 0.9, 0.85]:
                for f2 in [0.95, 0.9, 0.85]:
                    a_trial = a.copy()
                    a_trial[i1] *= f1
                    a_trial[i2] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial /= ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Strategy 3: Move mass from peak contributor to a "quiet" position
        if not improved:
            top_c = idx[np.argmax(contribs)]
            # Find position that doesn't contribute to top-5 peaks
            top_peaks = np.argpartition(b, -5)[-5:]
            quiet_positions = []
            for j in range(n):
                if any(abs(j - k) < 3 for k in top_peaks):
                    continue
                quiet_positions.append(j)
            if quiet_positions:
                # Pick the quietest position
                q = quiet_positions[np.argmax(a)]  # move from larger to keep smoothness
                for frac in [0.05, 0.1, 0.15, 0.2]:
                    a_trial = a.copy()
                    amt = a_trial[top_c] * frac
                    a_trial[top_c] -= amt
                    a_trial[q] += amt
                    if a_trial[top_c] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        # Strategy 4: Small Gaussian perturbation
        if not improved:
            for sigma in [0.001, 0.002, 0.005]:
                a_trial = a + np.random.standard_normal(n) * sigma
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Reset with larger perturbation
                a = best_a + np.random.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if (rnd + 1) % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.6f} peak={peak_val:.8f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, max_iter=100000):
    """Mirror descent (multiplicative weights) on the simplex with subgradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    lr0 = 0.5
    no_improve = 0

    for t in range(1, max_iter + 1):
        if monotonic() - t0 > time_limit:
            break
        if t % 10000 == 0:
            elapsed = monotonic() - t0
            print(f"    MirDesc iter={t} ({elapsed:.0f}s): score={best_score:.6f}")

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Subgradient of (a*a)_{peak_k}
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        sg = np.zeros(n)
        sg[idx] = 2.0 * a[peak_k - idx]

        # Mirror descent step: a_new_j proportional to a_j * exp(-eta * sg_j)
        eta = lr0 / np.sqrt(t)
        z = np.log(np.maximum(a, 1e-30)) - eta * sg
        z = z - z.max()
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                a = best_a + np.random.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

    return best_a, best_score


def lbgfs_final(a0, time_limit, T=3.0):
    """Final L-BFGS polish with very low temperature."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        obj, _ = compute_smoothed_obj_and_grad(z, T, n)
        return obj

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        _, grad = compute_smoothed_obj_and_grad(z, T, n)
        return grad

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': 75000, 'disp': False})
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0
    return a_out


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_update(a, label, best_seq, best_score):
    """Evaluate candidate and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: LP fictitious play initialization
    Phase 2: Adam with temperature annealing (main workhorse)
    Phase 3: Mirror descent
    Phase 4: Advanced peak shaving
    Phase 5: L-BFGS final polish
    Phase 6: Multi-restart
    Phase 7: Final peak shaving
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    n = len(best_seq)
    solver = _lp_solver_cache()
    print(f"Using LP solver: {solver}")

    # === Phase 1: LP Fictitious Play Initialization ===
    print(f"\n=== Phase 1: FP Init (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 60:
        t1 = min(rem * 0.08, 80.0)
        a_fp, sc_fp, it = fictitious_play_quick(n, t1, best_seq, solver=solver)
        best_seq, best_score = evaluate_and_update(a_fp, f"FP_init({it}it)", best_seq, best_score)

    # === Phase 2: Adam with Temperature Annealing ===
    print(f"\n=== Phase 2: Adam Annealing (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 100:
        t2 = min(rem * 0.45, 700.0)
        print(f"  Budget: {t2:.0f}s")
        T_schedule = [300.0, 100.0, 40.0, 15.0, 6.0, 3.0, 1.5]
        a_adam, sc_adam = adam_optimize(best_seq, t2, T_schedule=T_schedule, lr=0.008)
        best_seq, best_score = evaluate_and_update(a_adam, "adam_anneal", best_seq, best_score)

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 60:
        t3 = min(rem * 0.15, 200.0)
        a_md, sc_md = mirror_descent(best_seq, t3)
        best_seq, best_score = evaluate_and_update(a_md, "mir_desc", best_seq, best_score)

    # === Phase 4: Advanced Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 40:
        t4 = min(rem * 0.25, 350.0)
        a_ps, sc_ps = peak_shave_advanced(best_seq, t4)
        best_seq, best_score = evaluate_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 5: L-BFGS Final Polish ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 30:
        t5 = min(rem * 0.2, 200.0)
        a_lb = lbgfs_final(best_seq, t5, T=2.0)
        best_seq, best_score = evaluate_and_update(a_lb, "lbgfs_T2", best_seq, best_score)

        rem = deadline - monotonic()
        if rem > 15:
            t5b = min(rem * 0.3, 80.0)
            a_lb2 = lbgfs_final(best_seq, t5b, T=1.0)
            best_seq, best_score = evaluate_and_update(a_lb2, "lbgfs_T1", best_seq, best_score)

    # === Phase 6: Multi-restart ===
    print(f"\n=== Phase 6: Multi-restart (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 60:
        n_restarts = max(2, min(5, int(rem / 40)))
        for r in range(n_restarts):
            rem = deadline - monotonic()
            if rem < 20:
                break
            t_r = min(rem * 0.25, 60.0)
            # Perturb best and re-optimize
            a_pert = best_seq + np.random.standard_normal(n) * 0.002
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a_new, _ = adam_optimize(a_pert, t_r, T_schedule=[5.0, 2.0, 1.0], lr=0.005)
            best_seq, best_score = evaluate_and_update(a_new, f"restart_{r}", best_seq, best_score)

    # === Phase 7: Final Peak Shaving ===
    print(f"\n=== Phase 7: Final Shave (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 15:
        t7 = min(rem * 0.6, 60.0)
        a_final, _ = peak_shave_advanced(best_seq, t7)
        best_seq, best_score = evaluate_and_update(a_final, "final_shave", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
