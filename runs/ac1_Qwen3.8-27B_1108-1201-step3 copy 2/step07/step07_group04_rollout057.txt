def conv_gradient(a, k):
    """Gradient of c[k] = sum_j a[j]*a[k-j] with respect to a.
    Returns vector g where g[i] = 2*a[k-i] (for valid i).
    """
    n = len(a)
    g = np.zeros(n)
    # g[i] = 2*a[k-i] if 0 <= k-i < n
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        g[j] += 2.0 * a[k - j]
    # Note: when k-j == j (i.e., 2j == k), the term is counted only once
    # But in the sum c[k] = sum_j a[j]*a[k-j], the term a[j]*a[k-j] for j=k/2
    # is a[j]^2, and its derivative is 2*a[j], which is what we get here.
    # So this is correct as-is.
    return g


def conv_gradient_fast(a, peaks):

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(peaks)
    return g

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)  # running average
    t0 = monotonic()
    lr = lr_init
    no_improve = 0
    step = 0
    
    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Get top-k peaks
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        
        # Weight peaks by their values (softmax-like)
        peak_vals = b[peaks]
        temp = 0.001 * max(peak_vals.max(), 1e-10)
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w = w / w.sum()

        g = np.zeros(n)
        for i, k in enumerate(peaks):
            if w[i] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]
        
        # Mirror descent update: a_new[i] proportional to a[i] * exp(-lr * g[i])
        log_a = np.log(a) - lr * g
        log_a -= log_a.max()  # numerical stability

        if s < 1e-15:

        step += 1

            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate
            no_improve = 0
        else:
            no_improve += 1
            
        lr *= decay
        if no_improve > 200:

            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            lr = max(lr * 3, 0.005)
            no_improve = 0
            
        if step % 1000 == 0:
            elapsed = monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None, warm_start=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        b_var.value = np.maximum(warm_start, 0)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

        # Mirror descent update
        log_a = np.log(a) - lr * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        step += 1

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= decay
        if no_improve > 300:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            lr = max(lr * 3, 0.003)
            no_improve = 0

        if verbose and step % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"    MD {elapsed:.0f}s step={step}: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def subgradient_momentum(a0, time_limit, lr_init=0.003, momentum=0.9, top_k=30,
                         temp_scale=0.003, verbose=False):
    """Subgradient method with momentum for the max-conv objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t0 = monotonic()
    no_improve = 0
    step = 0

    def project_simplex(v):
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
        if len(rho) > 0:
            rho = rho[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
        else:
            theta = 0.0
        return np.maximum(v - theta, 0.0)

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        peak_vals = b[peaks]
        max_pv = peak_vals.max()
        if max_pv < 1e-15:
            break
        temp = temp_scale * max_pv
        w = np.exp((peak_vals - max_pv) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / ws

        g = subgradient_of_max_conv(a, peaks, w)
        v = momentum * v - lr * g
        a_candidate = a + v
        a_candidate = project_simplex(a_candidate)
        s = a_candidate.sum()
        if s < 1e-15:
            break
        a_candidate = a_candidate / s

        score = 2.0 * n * fast_convolve(a_candidate).max()
        step += 1

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        if no_improve > 400:
            perturbation = 0.003 * np.random.standard_normal(n)
            a = best_a + perturbation
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2.5, 0.002)
            no_improve = 0

        if verbose and step % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s step={step}: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True,
                    avg_rate_init=1.0, avg_rate_decay=0.999):
    """LP-based fictitious play with adaptive averaging."""
    t0 = monotonic()
    if initial is None:
        t = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.1 * np.sin(t) + 0.05 * np.sin(2 * t + 0.5)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    iters = 0
    stagnation = 0
    avg_rate = avg_rate_init
    b_warm = None
    a_warm = None

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver, warm_start=b_warm)
        if b_new is not None:
            b = b_new
            b_warm = b_new

        a_new, M_a = solve_lp_best_response(b, solver, warm_start=a_warm)
        if a_new is not None:
            a = a_new
            a_warm = a_new

        # Adaptive averaging: faster early, slower later
        a_avg = (avg_rate * a_avg + a) / (avg_rate + 1)
        b_avg = (avg_rate * b_avg + b) / (avg_rate + 1)
        avg_rate *= avg_rate_decay

        candidates = [a, b, 0.5 * (a + b), a_avg, 0.5 * (a_avg + b_avg),
                      a_avg * b_avg / (a_avg * b_avg).sum() if (a_avg * b_avg).sum() > 1e-15 else a]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_n = cand / s
            score = 2.0 * n * fast_convolve(cand_n).max()
            if score < best_score - 1e-13:
                best_score = score
                best_a = cand_n.copy()
                stagnation = 0
            else:
                stagnation += 1

        if stagnation > 150:
            noise = np.random.standard_normal(n) * 0.003
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            b = np.maximum(b + np.random.standard_normal(n) * 0.003, 0)
            b = b / b.sum()
            stagnation = 0
            avg_rate = max(avg_rate * 2, 1.0)

        if verbose and iters % 20 == 0:
            elapsed = monotonic() - t0
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def lbgfs_smooth(a0, time_limit, T=80, maxiter=3000):
    """L-BFGS refinement in log-space with soft-max smoothing of the max."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_s = z_final - z_final.max()
            a_out = np.exp(z_s)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()

    return a_out


def coordinate_descent(a0, time_limit, max_passes=200):
    """Coordinate descent: try adjusting each element to reduce the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    peak = best_score / (2.0 * n)

    for p in range(max_passes):
        if monotonic() - t0 > time_limit:
            break
        improved = False
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        # Try adjusting each of the top contributors at the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        order = np.argsort(contribs)[::-1][:8]

        for ci in order:
            i = int(idx[ci])
            for factor in [0.92, 0.85, 0.75]:
                old = a[i]
                if old < 1e-14:
                    break
                new = old * factor
                diff = old - new
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new
                a_trial[mask] = others + diff * others / os
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    peak = best_score / (2.0 * n)
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try small random perturbations
            for _ in range(3):
                a_trial = a + np.random.standard_normal(n) * 0.0005
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and p > 20:
            break

    return best_a, best_score


def make_asymmetric_window(n, eps=0.15, freq=1):
    """Create an asymmetric window for initialization."""
    t = np.arange(n) / n
    base = np.cos(np.pi * t) ** 2
    mod = 1.0 + eps * np.sin(2 * np.pi * freq * t + 0.3)
    a = base * mod
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def make_skewed_gaussian(n, skew=0.3, sigma=0.3):
    """Create a skewed Gaussian window."""
    t = np.arange(n) / n
    x = t - 0.5
    width = sigma * (1.0 + skew * x)
    a = np.exp(-0.5 * (x / max(width, 0.01)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def make_product_window(n, e1=0.1, e2=0.05):
    """Product of two sine modulations."""
    t = np.arange(n) / n
    a = 1.0 + e1 * np.sin(2 * np.pi * t) + e2 * np.sin(4 * np.pi * t + 1.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: mirror descent, LP fictitious play,
    L-BFGS smoothing, coordinate descent, and multi-start exploration.
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Try multiple asymmetric initializations ===
    print(f"\n=== Phase 1: Asymmetric Initializations ===")
    inits = [
        ("asym_window", make_asymmetric_window(n_best, 0.15, 1)),
        ("asym_window2", make_asymmetric_window(n_best, 0.2, 2)),
        ("skew_gauss", make_skewed_gaussian(n_best, 0.3, 0.35)),
        ("product", make_product_window(n_best, 0.12, 0.06)),
        ("best_seq", best_seq.copy()),
    ]
    for label, init in inits:
        best_seq, best_score = check_and_update(init, f"init_{label}", best_seq, best_score)

    # === Phase 2: Mirror Descent (main workhorse) ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 600
    if rem > 30:
        t_md = min(rem * 0.6, 300.0)
        print(f"  MD budget={t_md:.0f}s, n={n_best}")
        a_md, _ = mirror_descent(best_seq, t_md, lr_init=0.004, decay=0.9997,
                                 top_k=25, temp_scale=0.002, verbose=True)
        best_seq, best_score = check_and_update(a_md, "mirror_descent", best_seq, best_score)

    # === Phase 3: LP Fictitious Play ===
    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 500
    if rem > 40:
        t_fp = min(rem * 0.5, 250.0)
        print(f"  FP budget={t_fp:.0f}s, n={n_best}")
        a_fp, _, iters = fictitious_play(n_best, t_fp, initial=best_seq,
                                         solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"fp_n{n_best}", best_seq, best_score)

    # === Phase 4: Subgradient with momentum ===
    print(f"\n=== Phase 4: Subgradient+Momentum (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 400
    if rem > 30:
        t_sg = min(rem * 0.5, 150.0)
        print(f"  SubGrad budget={t_sg:.0f}s, n={n_best}")
        a_sg, _ = subgradient_momentum(best_seq, t_sg, lr_init=0.002, momentum=0.92,
                                       top_k=30, temp_scale=0.003, verbose=True)
        best_seq, best_score = check_and_update(a_sg, "subgrad_mom", best_seq, best_score)

    # === Phase 5: Mirror Descent round 2 (different params) ===
    print(f"\n=== Phase 5: MD Round 2 (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 300
    if rem > 30:
        t_md2 = min(rem * 0.5, 150.0)
        print(f"  MD2 budget={t_md2:.0f}s")
        a_md2, _ = mirror_descent(best_seq, t_md2, lr_init=0.008, decay=0.999,
                                  top_k=10, temp_scale=0.001, verbose=True)
        best_seq, best_score = check_and_update(a_md2, "md_round2", best_seq, best_score)

    # === Phase 6: L-BFGS with multiple temperatures ===
    print(f"\n=== Phase 6: L-BFGS Polish (best={best_score:.6f}) ===")
    for T in [50, 120, 300]:
        rem = deadline - monotonic() - 200
        if rem < 20:
            break
        t_lb = min(rem * 0.3, 60.0)
        print(f"  L-BFGS T={T}, budget={t_lb:.0f}s")
        a_lb = lbgfs_smooth(best_seq, t_lb, T=T, maxiter=2000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 150
    if rem > 15:
        t_cd = min(rem * 0.5, 60.0)
        print(f"  CoordDescent budget={t_cd:.0f}s")
        a_cd, _ = coordinate_descent(best_seq, t_cd)
        best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)

    # === Phase 8: LP FP on refined sequence ===
    print(f"\n=== Phase 8: FP Refined (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 100
    if rem > 40:
        t_fp2 = min(rem * 0.5, 120.0)
        print(f"  FP2 budget={t_fp2:.0f}s")
        a_fp2, _, _ = fictitious_play(n_best, t_fp2, initial=best_seq,
                                      solver=solver, verbose=False)
        best_seq, best_score = check_and_update(a_fp2, "fp_refined", best_seq, best_score)

    # === Phase 9: Try different lengths ===
    print(f"\n=== Phase 9: Length Exploration (best={best_score:.6f}) ===")
    n_curr = len(best_seq)
    alt_lengths = []
    for factor in [0.75, 1.25, 1.5, 0.5]:
        alt_n = int(n_curr * factor)
        if 128 <= alt_n <= 4096 and alt_n != n_curr and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:2]:
        rem = deadline - monotonic() - 60
        if rem < 30:
            break
        t_alt = min(rem * 0.4, 80.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Quick MD at new length
        a_alt, _ = mirror_descent(init_alt, t_alt, lr_init=0.005, decay=0.9997,
                                  top_k=20, temp_scale=0.002, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"md_n{alt_n}", best_seq, best_score)

    # === Phase 10: Final polish ===
    print(f"\n=== Phase 10: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 20
    if rem > 15:
        t_final = min(rem * 0.4, 30.0)
        a_f1 = lbgfs_smooth(best_seq, t_final, T=100, maxiter=1500)
        best_seq, best_score = check_and_update(a_f1, "final_lbgfs", best_seq, best_score)

    rem = deadline - monotonic() - 10
    if rem > 10:
        t_cd_final = min(rem * 0.5, 20.0)
        a_cd2, _ = coordinate_descent(best_seq, t_cd_final)
        best_seq, best_score = check_and_update(a_cd2, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
