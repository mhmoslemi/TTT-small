import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_conv(a, b):
    """Compute score given sequence and its convolution."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * b.max() / (s * s)


def normalize_sum1(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_direction(f, top_k, M, solver=cp.SCS):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns g or None.
    """
    n = len(f)
    b = fast_convolve(f)

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build constraint matrix (K x n)
    # (f*g)[k] = sum_{j=lo}^{hi} f[k-j] * g[j]
    A_mat = np.zeros((K, n))
    for i in range(K):
        k = top_idx[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        A_mat[i, idx] = f[k - idx]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A_mat @ g <= M])

        return None
    g_val = np.maximum(g.value, 0.0)

def subgradient_polish(f_init, time_limit, n_top=80):
    """Subgradient method with momentum for polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    b = fast_convolve(f)
    best_score = score_from_conv(f, b)
    lr = 0.001
    momentum = 0.9
    temp = 1.0 / max(n, 10)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Compute gradient of max conv (subgradient)
        # Use softmax-weighted average of active peaks
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_top = np.zeros(len(b))
            w_top[top_idx] = w[top_idx]
            w_top /= w_top.sum()
            w = w_top

        # Subgradient of max_k a^T Q_k a wrt a
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * w[k] * f[k - idx]

        v = momentum * v - lr * g
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-15:
            f = f_new / s
        else:
            f = best_f.copy()

        if it % 50 == 0:
            b = fast_convolve(f)
            cur_score = score_from_conv(f, b)
            if cur_score < best_score - 1e-10:
                best_score = cur_score
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.01)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.92
                    no_improve = 0

            if it % 200 == 0:
                temp = min(temp * 1.2, 1.0)

def coordinate_polish(f_init, time_limit):
    """Coordinate descent polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    b = fast_convolve(f)
    best_score = score_from_conv(f, b)
    multipliers = [0.7, 0.85, 1.0, 1.15, 1.3, 1.5]

    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj = best_score
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = normalize_sum1(f_trial)
                b_trial = fast_convolve(f_trial)
                obj = score_from_conv(f_trial, b_trial)
                if obj < best_obj - 1e-11:
                    best_obj = obj
                    best_val = new_val
            if best_val != f[j]:
                f[j] = best_val
                f = normalize_sum1(f)
                if best_obj < best_score - 1e-11:
                    best_score = best_obj
                    improved = True

    return f, best_score


def generate_structured_init(n, kind="uniform", seed=0):
    """Generate various structured initializations."""
    rng = np.random.default_rng(seed)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.5 * np.sin(x)
        return a / a.sum()
    elif kind == "cosine_window":
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        a = np.abs(rng.standard_normal(n)) + 0.1
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        return a / a.sum()
    elif kind == "bimodal":
        a = np.zeros(n)
        a[:n//4] = 1.0
        a[3*n//4:] = 1.0
        a += 0.1
        return a / a.sum()
    elif kind == "tapered":
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.5 * np.abs(2*x - 1)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-phase approach with structured initializations,
    LP cutting planes, subgradient polishing, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_b = fast_convolve(parent)
            p_score = score_from_conv(parent, p_b)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Evaluate parent with actual function
    if parent is not None:
        parent_actual = float(evaluate_sequence(parent.tolist()))
        print(f"Parent actual score: {parent_actual:.8f}")
    else:
        parent_actual = float('inf')

    best_seq = parent.copy() if parent is not None else np.ones(1024) / 1024
    best_score = parent_actual if np.isfinite(parent_actual) else float('inf')

    # === Phase 0: Try structured initializations ===
    print(f"\n=== Phase 0: Structured initializations ===")
    n_base = len(best_seq) if parent is not None else 1024
    init_kinds = ["uniform", "sinusoidal", "cosine_window", "random_smooth", "tapered", "bimodal"]
    init_lengths = [n_base, max(512, n_base // 2), min(4096, n_base * 2)]

    candidates = []
    for kind in init_kinds:
        for n_try in init_lengths:
            if time.monotonic() > deadline:
                break
            a = generate_structured_init(n_try, kind)
            b = fast_convolve(a)
            s = score_from_conv(a, b)
            actual = float(evaluate_sequence(a.tolist()))
            print(f"  {kind} n={n_try}: score={s:.6f} actual={actual:.6f}")
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_seq = a.copy()
                print(f"  NEW BEST: {actual:.8f}")
            candidates.append((actual, a.copy()))

    # Sort candidates by score
    candidates = [c for c in candidates if np.isfinite(c[0])]
    candidates.sort(key=lambda x: x[0])
    print(f"  Best candidate in phase 0: {candidates[0][0]:.8f}" if candidates else "  No valid candidates")

    # === Phase 1: LP optimization on best candidates ===
    print(f"\n=== Phase 1: LP optimization ===")
    # Take top 3 distinct candidates
    top_candidates = []
    seen_n = set()
    for score_c, seq_c in candidates:
        if len(seq_c) not in seen_n:
            top_candidates.append(seq_c)
            seen_n.add(len(seq_c))
        if len(top_candidates) >= 3:
            break

    if parent is not None and parent not in [c[1] for c in top_candidates]:
        top_candidates.insert(0, parent)

    for ci, f_init in enumerate(top_candidates):
        if time.monotonic() > deadline - 60:
            break
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.3, 400.0)
        if t_alloc < 20:
            break
        n_c = len(f_init)
        print(f"\n  LP candidate {ci}: n={n_c}, budget={t_alloc:.0f}s")
        f_opt, s_opt = run_lp_optimize(f_init, time_limit=t_alloc,
                                         top_k_init=40, top_k_max=250,
                                         label=f"lp_c{ci}")
        actual = float(evaluate_sequence(f_opt.tolist()))
        if np.isfinite(actual) and actual < best_score:
            best_score = actual
            best_seq = f_opt.copy()
            print(f"  NEW BEST [lp_c{ci}]: {actual:.8f}")
        else:
            print(f"  lp_c{ci} result: {actual:.8f} (no improvement)")

    # === Phase 2: Deeper LP on current best ===
    print(f"\n=== Phase 2: Deep LP (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 60:
        rem = deadline - time.monotonic()
        t2 = min(rem * 0.25, 350.0)
        if t2 > 30:
            n_b = len(best_seq)
            print(f"  n={n_b}, budget={t2:.0f}s, top_k=100->300")
            f2, s2 = run_lp_optimize(best_seq, time_limit=t2,
                                      top_k_init=100, top_k_max=300,
                                      label="lp_deep")
            actual2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(actual2) and actual2 < best_score:
                best_score = actual2
                best_seq = f2.copy()
                print(f"  NEW BEST [deep]: {actual2:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    n_cur = len(best_seq)
    target_lengths = []
    for factor in [0.6, 0.8, 1.25, 1.5, 2.0, 3.0]:
        tl = int(n_cur * factor)
        tl = max(256, min(8000, tl))
        if tl != n_cur and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_cur))

    for target_n in target_lengths[:4]:
        if time.monotonic() > deadline - 40:
            break
        rem = deadline - time.monotonic()
        t3 = min(rem * 0.2, 120.0)
        if t3 < 15:
            break

        # Interpolate or decimate
        if target_n > n_cur:
            old_x = np.linspace(0.0, 1.0, n_cur)
            new_x = np.linspace(0.0, 1.0, target_n)
            f_interp = np.interp(new_x, old_x, best_seq)
        else:
            # Average blocks
            blocks = n_cur // target_n
            f_interp = np.zeros(target_n)
            valid = blocks * target_n
            for i in range(target_n):
                f_interp[i] = best_seq[i*blocks:(i+1)*blocks].mean()
            if valid < n_cur:
                f_interp[-1] += best_seq[valid:].sum() / target_n

        f_interp = np.maximum(f_interp, 0.0)
        s_int = f_interp.sum()
        if s_int > 1e-10:
            f_interp /= s_int
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  LP n={target_n}, budget={t3:.0f}s")
        f3, s3 = run_lp_optimize(f_interp, time_limit=t3,
                                  top_k_init=50, top_k_max=200,
                                  label=f"lp_n{target_n}")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [n{target_n}]: {score3:.8f}")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient polish (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 30:
        rem = deadline - time.monotonic()
        t4 = min(rem * 0.2, 120.0)
        if t4 > 15:
            n_b = len(best_seq)
            print(f"  n={n_b}, budget={t4:.0f}s")
            f4, s4 = subgradient_polish(best_seq, time_limit=t4,
                                         n_top=max(50, n_b // 5))
            score4 = float(evaluate_sequence(f4.tolist()))
            if np.isfinite(score4) and score4 < best_score:
                best_score = score4
                best_seq = f4.copy()
                print(f"  NEW BEST [sg]: {score4:.8f}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 20:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.15, 60.0)
        if t5 > 10:
            print(f"  budget={t5:.0f}s, n={len(best_seq)}")
            f5, s5 = coordinate_polish(best_seq, time_limit=t5)
            score5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(score5) and score5 < best_score:
                best_score = score5
                best_seq = f5.copy()
                print(f"  NEW BEST [cd]: {score5:.8f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 10:
        rem = deadline - time.monotonic()
        t6 = min(rem * 0.5, 100.0)
        if t6 > 10:
            n_b = len(best_seq)
            print(f"  n={n_b}, budget={t6:.0f}s")
            f6, s6 = run_lp_optimize(best_seq, time_limit=t6,
                                      top_k_init=150, top_k_max=350,
                                      label="lp_final")
            score6 = float(evaluate_sequence(f6.tolist()))
            if np.isfinite(score6) and score6 < best_score:
                best_score = score6
                best_seq = f6.copy()
                print(f"  NEW BEST [lp_final]: {score6:.8f}")

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Fallback to parent
        if parent is not None:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))
        else:
            best_seq = np.ones(1024) / 1024
            best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    # Ensure all values are valid
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
