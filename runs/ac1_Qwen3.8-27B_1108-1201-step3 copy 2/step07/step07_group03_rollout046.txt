def fast_convolve(a):
    n = len(a)

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def logsumexp_anneal(a0, time_limit, T_start=0.5, T_end=0.01, n_iters=2000):
    """Optimize log-sum-exp smooth approximation with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()

    # Precompute the convolution matrix structure for gradient
    # grad of c_k w.r.t. a_j: dc_k/da_j = 2*a_{k-j} (for valid j)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        T = T_start * (T_end / T_start) ** (it / max(n_iters - 1, 1))
        c = fast_convolve(a)

        # Softmax weights on peaks
        c_max = c.max()
        w = np.exp((c - c_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum

        # Gradient: grad_j = sum_k w_k * 2 * a_{k-j}
        grad = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[k] * a[k - idx]

        # Mirror descent step (multiplicative update)
        eta = 0.1 * T
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s

        # Also try Euclidean projected gradient
        a_eucl = a - eta * grad
        a_eucl = simplex_project(a_eucl)

        sc_mirror = obj_normalized(a_new)
        sc_eucl = obj_normalized(a_eucl)

        if sc_mirror < sc_eucl:
            a = a_new
            sc = sc_mirror
        else:
            a = a_eucl
            sc = sc_eucl

        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    n_harm = int(rng.integers(3, 20))
    amps = rng.random(n_harm) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: LP fictitious play at multiple lengths
    Phase 2: Peak-targeted coordinate descent with line search
    Phase 3: Log-sum-exp annealing
    Phase 4: Pair optimization
    Phase 5: Final intensive polish
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(2024)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # Phase 1: LP fictitious play at multiple lengths
    print("\n=== Phase 1: LP Fictitious Play ===")
    t1_start = time.monotonic()
    n_values = list(range(512, 1300, 128))
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
    n_values = n_values[:6]

    time_per_n = 180.0
    for n_fp in n_values:
        rem = deadline - time.monotonic() - 600
        if rem < 60:
            break
        t_alloc = min(time_per_n, rem * 0.8)
        if t_alloc < 30:
            continue

        print(f"  FP n={n_fp}, budget={t_alloc:.0f}s")
        initial = interpolate_seq(best_seq, n_fp) if best_seq is not None else None
        a_fp, score_fp, iters = fictitious_play(n_fp, t_alloc, initial=initial,
                                                  solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score,
                                                  f"fp_n{n_fp}_it{iters}")

        # Try averaged versions
        if best_seq is not None and len(best_seq) == n_fp:
            # Slightly perturb and re-run briefly
            a_pert = best_seq + rng.standard_normal(n_fp) * 0.03
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a_fp2, score_fp2, _ = fictitious_play(n_fp, 30.0, initial=a_pert,
                                                    solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_fp2, best_seq, best_score,
                                                      f"fp2_n{n_fp}")

    # Phase 2: Peak-targeted coordinate descent
    print(f"\n=== Phase 2: Peak CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_cd = min(rem * 0.5, 400.0)
            print(f"  PeakCD n={len(best_seq)}, budget={t_cd:.0f}s")
            a_cd, s_cd = peak_targeted_cd(best_seq, t_cd, top_peaks=3)
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "peak_cd")

    # Phase 3: Log-sum-exp annealing
    print(f"\n=== Phase 3: LogSumExp Anneal (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_ls = min(rem * 0.5, 200.0)
            print(f"  LSE budget={t_ls:.0f}s")
            a_ls, s_ls = logsumexp_anneal(best_seq, t_ls, T_start=0.3, T_end=0.005,
                                           n_iters=5000)
            best_seq, best_score = check_and_update(a_ls, best_seq, best_score, "lse")

            # Follow with CD
            if best_seq is not None:
                rem = deadline - time.monotonic() - 150
                if rem > 20:
                    a_cd2, _ = peak_targeted_cd(best_seq, min(rem * 0.6, 100.0), top_peaks=2)
                    best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "cd_after_lse")

    # Phase 4: Pair optimization
    print(f"\n=== Phase 4: Pair Opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_pair = min(rem * 0.5, 150.0)
            print(f"  Pair budget={t_pair:.0f}s")
            a_pair, s_pair = pair_optimization(best_seq, t_pair, n_pairs=5)
            best_seq, best_score = check_and_update(a_pair, best_seq, best_score, "pair")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Multiple rounds of CD with decreasing peak count
        for n_peaks in [1, 2, 3]:
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            t_fin = min(rem * 0.4, 80.0)
            a_fin, s_fin = peak_targeted_cd(best_seq, t_fin, top_peaks=n_peaks)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score,
                                                      f"final_cd_pk{n_peaks}")

        # One more LP round
        rem = deadline - time.monotonic() - 15
        if rem > 30 and best_seq is not None:
            n_best = len(best_seq)
            t_lp = min(rem * 0.6, 60.0)
            a_lp, _, _ = fictitious_play(n_best, t_lp, initial=best_seq,
                                          solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, "final_lp")

            # Quick CD after
            rem = deadline - time.monotonic() - 5
            if rem > 10:
                a_cd3, _ = peak_targeted_cd(best_seq, min(rem * 0.8, 30.0), top_peaks=1)
                best_seq, best_score = check_and_update(a_cd3, best_seq, best_score,
                                                          "final_cd_after_lp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
