def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score_normalized(a):
    """Score for a probability vector (sum=1, non-negative)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def convolve_at_k(a, k):

    je = min(n - 1, k)
    idx = np.arange(js, je + 1)

def lbgfs_soft_max_v2(a0, time_limit, T_start=50.0, T_end=0.5, maxiter=10000):
    """
    Improved L-BFGS with soft-max smoothing.
    Uses lower temperatures and more iterations at low T.
    """
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    def make_obj(T_val):
        def func(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            lse = max_Tb + np.log(np.exp(Tb - max_Tb).sum())
            return (2.0 * n / T_val) * lse
        return func
    
    def make_grad(T_val):
        def jac(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            # Convert to log-space
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    # Temperature schedule: more stages at low T
    temps = [T_start, T_start*0.3, T_start*0.1, T_start*0.03, T_end]
    z = z0.copy()
    
    for stage_i, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 10:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        # Allocate more iterations at lower temperature
        if stage_i < 2:
            iters = maxiter // 10
        else:
            iters = maxiter // 3
        
        try:
            res = minimize(
                fun=make_obj(stage_T),
                x0=z,
                jac=make_grad(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        a_check = np.exp(z - z.max())
        a_check = a_check / a_check.sum()
        sc = eval_score_normalized(a_check)
        print(f"    L-BFGS T={stage_T:.2f}: score={sc:.6f}")
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def nesterov_subgradient(a0, time_limit, n_steps=50000, lr_init=0.1):
    """
    Nesterov-accelerated subgradient method with Polyak step size.
    Uses the true subgradient of the max function.
    """
    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = eval_score_normalized(a)
    
    # Nesterov momentum
    y = a.copy()  # virtual iterate
    x_avg = a.copy()  # averaged iterate
    
    t0 = time.monotonic()
    lr = lr_init
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        b = fast_convolve(y)
        M = b.max()
        
        # Active set (top peaks)
        top_k = np.argsort(b)[::-1][:max(1, len(b) // 100)]
        # Use weighted average for smoothness
        threshold = M * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * y[k - idx]
        g /= len(active)
        
        # Project gradient onto tangent space (sum=0)
        g -= np.mean(g)
        
        # Nesterov update
        momentum = 2.0 / (step + 2)
        y_new = y - lr * g
        y_new = np.maximum(y_new, 0)
        y_new = y_new / y_new.sum()
        
        # Update averaged iterate
        x_avg = (step / (step + 1)) * x_avg + (1 / (step + 1)) * y_new
        x_avg = np.maximum(x_avg, 0)
        x_avg = x_avg / x_avg.sum()
        
        y = y_new
        
        # Adaptive learning rate
        if step % 1000 == 0 and step > 0:
            lr *= 0.95
        
        # Check improvement
        sc_avg = eval_score_normalized(x_avg)
        sc_y = eval_score_normalized(y)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = x_avg.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if step % 5000 == 0:
            print(f"    NesterovSG step={step}: best={best_score:.6f}")
    
    return best_a, best_score

                a = a_try
                best_score = score
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try reducing both i and j
            if i_star != j_star and a[j_star] > 1e-14:
                for f1 in [0.8, 0.6, 0.4]:
                    for f2 in [0.8, 0.6, 0.4]:
                        a_try = a.copy()
                        delta1 = a_try[i_star] * (1 - f1)
                        delta2 = a_try[j_star] * (1 - f2)
                        a_try[i_star] *= f1
                        a_try[j_star] *= f2
                        total_delta = delta1 + delta2
                        mask = np.ones(n, dtype=bool)
                        mask[idx] = False
                        if mask.sum() > 0 and total_delta > 0:
                            a_try[mask] += total_delta * a_try[mask] / a_try[mask].sum()
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        score = eval_score_normalized(a_try)
                        if score < best_score - 1e-14:
                            a = a_try
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            
            if not improved:
                # Random perturbation
                a_try = a + 0.005 * np.random.randn(n)
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                score = eval_score_normalized(a_try)
                if score < best_score:
                    a = a_try
                    best_score = score
                    best_a = a.copy()
                elif step % 100 == 0:
                    a = best_a.copy()  # reset to best
        
        if step % 500 == 0:
            print(f"    PeakEq step={step}: score={best_score:.6f}")
    
    return best_a, best_score


def random_restart_optimize(a0, time_limit, n_restarts=20, optimize_time_each=5.0):
    """
    Generate random perturbations of a0 and quickly optimize each.
    """
    n = len(a0)
    best_a = np.array(a0, dtype=np.float64).copy()
    best_a = np.maximum(best_a, 0)
    best_a = best_a / best_a.sum()
    best_score = eval_score_normalized(best_a)
    t0 = time.monotonic()
    
    rng = np.random.default_rng(12345)
    
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Generate perturbation
        scale = rng.uniform(0.01, 0.1)
        a_try = best_a + scale * rng.standard_normal(n)
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()
        
        # Quick L-BFGS optimization
        z = np.log(np.maximum(a_try, 1e-300))
        T = 10.0
        
        def func(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            lse = max_Tb + np.log(np.exp(Tb - max_Tb).sum())
            return (2.0 * n / T) * lse
        
        def jac(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10, 'disp': False})
            a_opt = np.exp(res.x - res.x.max())
            a_opt = a_opt / a_opt.sum()
            score = eval_score_normalized(a_opt)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    Restart {r}: improved to {best_score:.6f}")
        except Exception:
            pass
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
