import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_gradient_softmax(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv(a,a)[k])) at point a."""
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    # Softmax weights
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    p = exp_b / exp_b.sum()  # length 2n-1
    
    # Gradient: g[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    # This is cross-correlation of a and p
    fft_size = 1 << (2 * (2*n-1) - 1).bit_length()  # need N >= n + (2n-1) - 1
    # Actually for cross-correlation of length n and length (2n-1), we need N >= n + 2n-1 - 1 = 3n-2
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.fft(a, fft_size)
    P = np.fft.fft(p, fft_size)
    # cross_corr[j] = sum_m a[m] * p[m+j] = ifft(conj(A) * P)[j]
    cc = np.fft.ifft(np.conj(A) * P)[:n]
    g = 2.0 * np.real(cc)
    return g, b, p

def softmax_gradient_descent(a0, time_limit, beta_init=50.0, beta_max=500.0, 
                               verbose=False):
    """Soft-max gradient descent with adaptive temperature and line search."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    beta = beta_init
    no_improve = 0
    lr = 0.01
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g, b, p = compute_gradient_softmax(a, beta)
        
        # Line search: try to find step size that reduces objective
        best_step = None
        best_f = None
        f_current = (1.0/beta) * np.log(np.exp(beta * (b - b.max())).sum() + 1e-300) + b.max()
        # Actually the soft-max value is: max(b) + (1/beta)*log(sum(exp(beta*(b-max(b)))))
        
        for factor in [lr, lr*0.5, lr*0.25, lr*0.125, lr*0.0625, lr*0.03125]:
            a_trial = simplex_project(a - factor * g)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            b_trial = fast_convolve(a_trial)
            f_trial = b_trial.max() + (1.0/beta) * np.log(np.exp(beta * (b_trial - b_trial.max())).sum() + 1e-300)
            if best_f is None or f_trial < best_f:
                best_f = f_trial
                best_step = factor
                best_a_trial = a_trial.copy()
            
            if f_trial < f_current - 1e-10:
                break
        
        if best_step is not None:
            a = best_a_trial
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adaptive beta: increase to tighten the approximation
        if no_improve % 20 == 0:
            beta = min(beta_max, beta * 1.1)
        
        # Learning rate adjustment
        if no_improve > 30:

            no_improve = 0
        elif no_improve > 100:

            a = simplex_project(best_a + noise)
            lr = 0.01

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max smoothed gradient with mirror descent on the simplex
2. Adaptive temperature (increasing softmax sharpness over time)
3. FFT-based fast gradient computation
4. Interleaved LP-based fictitious play for global structure
5. Pair-coordinate descent for active peak refinement
6. Multi-length exploration with diverse initializations

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0]) if v[0] > 0 else np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def compute_softmax_gradient(a, beta):
    """
    Compute gradient of soft_max_beta(conv(a,a)) at point a.
    
    F(a) = (1/beta) * log(sum_k exp(beta * conv(a,a)[k]))
    
    Gradient: g[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    where p = softmax(beta * conv(a,a))
    
    Returns: (gradient vector of length n, convolution vector, softmax weights)
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    
    # Compute softmax weights with numerical stability
    b_shifted = b - b.max()
    exp_b = np.exp(beta * b_shifted)
    p = exp_b / exp_b.sum()
    
    # Gradient via cross-correlation: g[j] = 2 * sum_m a[m] * p[j+m]
    # Use FFT for efficiency
    # Cross-correlation: (a ⋆ p)[j] = sum_m a[m] * p[j+m]
    # = ifft(conj(FFT(a)) * FFT(p))[j]
    total_len = n + len(p) - 1  # = n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    P = np.fft.fft(p, fft_size)
    cc = np.fft.ifft(np.conj(A) * P)
    g = 2.0 * np.real(cc[:n])
    
    return g, b, p

def soft_max_value(b, beta):
    """Compute (1/beta)*log(sum exp(beta*b)) with numerical stability."""
    b_max = b.max()
    val = b_max + (1.0 / beta) * np.log(np.exp(beta * (b - b_max)).sum())
    return val

def mirror_descent(a0, time_limit, beta_init=30.0, beta_max=300.0,

    Mirror descent (multiplicative weights) on the simplex for the soft-max objective.
    """

    best_a = a.copy()

    t0 = time.monotonic()

        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
            a_new = np.maximum(a_new, 0.0)

            else:

        if score < best_score - 1e-13:

            best_a = a_new.copy()
            a = a_new

        else:
            no_improve += 1
            # Try a smaller step

                a_new2 = a * np.exp(-0.1 * scaled_lr * g)

                if s2 > 1e-15:

                        best_score = score2

                    else:

            lr = lr_init

            no_improve = 0

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None, timeout=30.0):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    t0 = time.monotonic()
    try:
        if solver is not None:
            prob.solve(solver=solver)
        else:
            prob.solve()
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None
    b_val = b_val / sv
    return b_val

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return None

                    if st < 1e-15:

                    a_trial = a_trial / st

def coord_descent_fast(a0, time_limit, verbose=False):

    n = len(a0)

def interpolate_seq(f, new_n):
    """Interpolate a sequence (normalized) to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_cosine(n, freq, phase, amplitude, rng):
    """Initialize with a cosine-based pattern."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n) * (1.0 - amplitude)
    a += amplitude * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_multi_cosine(n, rng):
    """Initialize with multiple cosine components."""
    n_harmonics = int(rng.integers(2, 8))
    a = np.ones(n) * 0.8
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_harmonics):
        freq = int(rng.integers(1, n // 4 + 1))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.25)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_from_parent(parent, n, rng, sigma=0.08):
    """Initialize by interpolating parent and adding noise."""
    base = interpolate_seq(parent, n)
    noise = rng.standard_normal(n) * sigma
    a = np.maximum(base + noise, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def check_score(a):
    """Compute the score (without calling evaluate_sequence)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    b = fast_convolve(a_norm)
    return float(2.0 * n * b.max())
