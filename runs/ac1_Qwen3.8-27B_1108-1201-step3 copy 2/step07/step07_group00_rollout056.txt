from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgradient(a, top_k=10):
    """Compute subgradient of max_k (a*a)[k]."""
    n = len(a)
    b = fast_convolve(a)
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    
    g = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # g[j] += 2 * a[k-j] for j in [lo, hi]
        # a[k-j] for j in [lo, hi] is a[k-hi : k-lo+1][::-1]
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(top_idx)
    return g, b.max(), top_idx

    log_weights -= log_weights.max()  # stability
    a_new = np.exp(log_weights)
    a_new /= a_new.sum()
