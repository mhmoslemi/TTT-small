import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def get_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def compute_softmax_grad(a, w, n, fft_size):
    """Compute gradient of n * soft_max(conv(a,a), T) w.r.t. a"""
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr

def nesterov_logspace(a0, time_limit, T_init=0.05, T_final=0.001,
                       lr=0.01, momentum=0.9, schedule='exp'):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    v = np.zeros(n)
    fft_size = 1 << (2*n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Temperature
        T = T_init * (T_final / T_init) ** progress
        
        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp((b - max_b) / T)
        w = exp_b / exp_b.sum()
        
        grad_a = compute_softmax_grad(a_curr, w, n, fft_size)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Nesterov momentum
        v = momentum * v + grad_z
        lr_t = lr / (1.0 + 0.01 * t)  # slow decay
        z = z - lr_t * v
        
        a_new = softmax_from_z(z)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
    
    return best_a, best_score

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:

    return exp_z / s

    """Score for a sequence with sum=1: 2*n*max_conv"""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def cross_corr_fft(a, b, fft_size):
    """Compute cross-correlation of a and b (length 2n-1 result, take first n for gradient)."""
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * np.conj(B), fft_size)

def gradient_softmax_conv(a, w, n, fft_size):
    """
    Gradient of 2n * T * log(sum(exp(conv(a,a)/T))) w.r.t. a.
    = 2n * cross_correlation(a, w)
    where w[k] = exp(conv[k]/T) / sum(exp(conv/T))

    A = rfft(a, fft_size)

    return 2.0 * n * corr[:n]

                       lr=0.01, momentum=0.9):

    Nesterov-accelerated gradient descent in log-space with adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated gradient
descent in log-space with adaptive temperature, L-BFGS polishing, and
structured local search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# --- Helper functions ---

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def get_normalized_score(a):
    """Score = 2*n*max_conv for normalized (sum=1) sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

def grad_soft_conv(a, w, n, fft_size):
    """
    Gradient of 2n * soft_max(conv(a,a)) w.r.t. a.
    Result has length n (only the first n terms of the full cross-correlation).
    """
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    return 2.0 * n * corr[:n]

        b = fast_convolve(a_curr)
