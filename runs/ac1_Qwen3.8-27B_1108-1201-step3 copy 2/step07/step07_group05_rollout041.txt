def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)

        return np.ones(len(z)) / len(z)
    return exp_z / s

def cross_corr_a_w(a, w):
    """Compute result[m] = sum_j a[j]*w[j+m] for m=0..n-1 using FFT."""
    n = len(a)
    total_size = n + len(w) - 1
    fft_size = 1 << (total_size - 1).bit_length()
    a_flip = a[::-1]
    AF = rfft(a_flip, fft_size)
    WF = rfft(w, fft_size)
    result = irfft(AF * WF, fft_size)
    return result[n - 1:n - 1 + n]
