import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_two(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    return b_val, float(M_var.value), None


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def evaluate_safe(a):

        if len(a_list) < 1:
            return float('inf')
        if sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:

def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    a = a / s
    n = len(a)
    b = fast_convolve(a)

def interpolate_sequence(f, new_n):

    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lp_alternate(a_init, time_limit, solver=None, n_windows=5):
    """
    LP-based alternating best-response with multi-window time averaging.
    Alternates between solving LP for a given b and LP for b given a.
    Keeps time averages over multiple windows and tracks the best.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    b = a.copy()

    best_score = score_normalized(a)
    best_a = a.copy()
    t0 = time.monotonic()

    # Multi-window averages
    window_sizes = [10, 50, 100, 500, 1000]
    window_sizes = [w for w in window_sizes if w >= 5]
    n_win = len(window_sizes)
    # For each window, maintain a running sum and count
    win_sums_a = [np.zeros(n) for _ in range(n_win)]
    win_sums_b = [np.zeros(n) for _ in range(n_win)]
    win_counts = [0] * n_win

    # Global average
    global_sum = np.zeros(n)
    global_count = 0

    iteration = 0
    no_improve = 0
    last_best = best_score

    for iteration in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP: find best b given current a
        b_new, M_ab, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Solve LP: find best a given current b
        a_new, M_ba, _ = solve_lp_best_response(b_new, solver)
        if a_new is None:
            continue

        # Update iterates
        a = a_new
        b = b_new

        # Update windows
        for wi in range(n_win):
            win_sums_a[wi] += a
            win_sums_b[wi] += b
            win_counts[wi] += 1

        global_sum += a + b
        global_count += 2

        # Evaluate periodically
        if iteration % 5 == 0:
            # Check global average
            if global_count > 10:
                ga = global_sum / global_count
                ga = np.maximum(ga, 0)
                gs = ga.sum()
                if gs > 1e-15:
                    ga = ga / gs
                    gs_score = score_normalized(ga)
                    if gs_score < best_score:
                        best_score = gs_score
                        best_a = ga.copy()
                        no_improve = 0
                    else:
                        no_improve += 5

            # Check windows
            for wi in range(n_win):
                if win_counts[wi] >= window_sizes[wi]:
                    wavg = win_sums_a[wi] / win_counts[wi]
                    wavg = np.maximum(wavg, 0)
                    ws = wavg.sum()
                    if ws > 1e-15:
                        wavg = wavg / ws
                        w_score = score_normalized(wavg)
                        if w_score < best_score:
                            best_score = w_score
                            best_a = wavg.copy()
                            no_improve = 0

            # Also check current iterates
            a_score = score_normalized(a)
            b_score = score_normalized(b)
            if a_score < best_score:
                best_score = a_score
                best_a = a.copy()
                no_improve = 0
            if b_score < best_score:
                best_score = b_score
                best_a = b.copy()
                no_improve = 0

            # If stuck, reset averages and restart from best
            if no_improve > 200:
                for wi in range(n_win):
                    win_sums_a[wi] = np.zeros(n)
                    win_sums_b[wi] = np.zeros(n)
                    win_counts[wi] = 0
                global_sum = np.zeros(n)
                global_count = 0
                a = best_a.copy()
                b = best_a.copy()
                no_improve = 0

            if iteration % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LPA iter={iteration} ({elapsed:.0f}s): best={best_score:.6f}, M_ab={M_ab:.6f}, M_ba={M_ba:.6f}")

    return best_a, best_score


def lp_dual_mirror_descent(a_init, time_limit, solver=None, lr_init=0.1):
    """
    Hybrid method: use LP dual variables to guide mirror descent.
    The LP dual gives a distribution over active convolution positions,
    which we use as a "smart" subgradient for the max-conv objective.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = z_from_a(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP to get dual variables (distribution over active positions)
        T = build_conv_matrix(a)
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        y_var = cp.Variable(2 * n - 1, nonneg=True)
        constraints = [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1,
            # Dual: max sum(y) s.t. T^T y <= M, y >= 0
            # We extract y from the dual
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        # Extract dual variables y (on the T @ b <= M constraints)
        try:
            y_val = np.asarray(T @ b_var <= M_var)
            # Actually, let's get the dual from the constraint
        except Exception:
            pass

        # Alternative: compute the uniform distribution over near-max positions
        b_conv = convolve_two(a, a)
        max_b = b_conv.max()
        threshold = max_b - 0.01 * max_b
        active = np.where(b_conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b_conv)])
        y_dist = np.zeros(2 * n - 1)
        y_dist[active] = 1.0 / len(active)

        # Compute smart gradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Mirror descent update
        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = score_normalized(a_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.9
            no_improve = 0
        lr = max(lr, 1e-8)

        if step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-MD step={step} ({elapsed:.0f}s): best={best_score:.6f}, lr={lr:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.5, maxiter=10000):
    """L-BFGS refinement with soft-max smoothing and geometric temperature."""
    from scipy.optimize import minimize
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()

    def func_factory(T_val):
        def func(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def jac_factory(T_val):
        def jac(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=func_factory(stage_T),
                x0=z,
                jac=jac_factory(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def peak_shave_anneal(a0, time_limit, n_restarts=10):
    """
    Peak shaving with simulated annealing restarts.
    At each step, try to reduce the peak by modifying coordinates.
    Periodically do a "restart" with small random perturbation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()
    step = 0
    max_steps = 100000

    while step < max_steps:
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_mid = b.copy()
        b_mid[0] = -np.inf
        b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_n = min(8, len(idx))
        top_contrib = np.argsort(contribs)[::-1][:top_n]

        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            j = int(peak_k - i)
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.90, 0.85, 0.75]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    step += 1
                    break
            if improved:
                break

        # Also try reducing j
        if not improved:
            for ci in top_contrib:
                j = int(peak_k - int(idx[ci]))
                if a[j] < 1e-14 or j == int(idx[ci]):
                    continue
                for factor in [0.98, 0.95, 0.90]:
                    new_val = a[j] * factor
                    diff = a[j] - new_val
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others = a[mask]
                    if others.sum() < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        step += 1
                        break
                if improved:
                    break

        if not improved:
            # Stuck - do a restart with perturbation
            step += 1
            a = best_a.copy()
            # Small Gaussian perturbation in log-space
            z = z_from_a(a)
            noise_level = 0.01
            z += noise_level * rng.standard_normal(n)
            a = softmax_from_z(z)
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

            if step % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    PeakShave step={step} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, n_passes=50):
    """Faster coordinate descent with golden section search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            # Try to optimize coordinate i by shifting mass
            base = a.copy()
            base[i] = 0
            base_sum = base.sum()
            if base_sum < 1e-15:
                continue
            # Golden section search for optimal t = a[i]
            lo, hi = 0.0, min(1.0, a[i] * 3)
            phi = (1 + np.sqrt(5)) / 2
            for _ in range(8):
                m1 = hi - (hi - lo) / phi
                m2 = lo + (hi - lo) / phi
                a1 = base * (1 - m1) + m1 * np.zeros(n)
                a1[i] = m1
                a1 = a1 / a1.sum()
                a2 = base * (1 - m2) + m2 * np.zeros(n)
                a2[i] = m2
                a2 = a2 / a2.sum()
                s1 = score_normalized(a1)
                s2 = score_normalized(a2)
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2
            a_new = base * (1 - t_opt) + t_opt * np.zeros(n)
            a_new[i] = t_opt
            a_new = a_new / a_new.sum()
            sc = score_normalized(a_new)
            if sc < best_score - 1e-14:
                a = a_new
                best_score = sc
                best_a = a.copy()

        if p % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc pass={p} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP alternating, LP-dual mirror descent,
    L-BFGS refinement, peak shaving with annealing, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = min(p_score, score_normalized(parent))
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_normalized(a)
        sc_eval = evaluate_safe(a)
        sc = min(sc, sc_eval) if np.isfinite(sc_eval) else sc
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: LP Alternating with Multi-Window Averaging (550s) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 60:
        print(f"  LP Alt n={n_best}, budget={t_alloc:.0f}s")
        a_lp, sc_lp = lp_alternate(best_seq, t_alloc, solver=solver)
        try_update(a_lp, "lp_alt")

    # === Phase 2: L-BFGS Soft-Max with Aggressive Annealing (400s) ===
    print(f"\n=== Phase 2: L-BFGS Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 50->0.5, n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=50.0, T_end=0.5, maxiter=12000)
        try_update(a_lb, "lbgfs_main")

        # Second pass at lower temperature
        rem = deadline - time.monotonic()
        t_alloc2 = min(rem * 0.12, 150.0)
        if t_alloc2 > 20:
            print(f"  L-BFGS T: 3->0.1, budget={t_alloc2:.0f}s")
            a_lb2 = lbgfs_refine(best_seq, t_alloc2, T_start=3.0, T_end=0.1, maxiter=8000)
            try_update(a_lb2, "lbgfs_lowT")

    # === Phase 3: LP-Dual Mirror Descent (250s) ===
    print(f"\n=== Phase 3: LP-Dual Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  LP-MD n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_md, sc_md = lp_dual_mirror_descent(best_seq, t_alloc, solver=solver, lr_init=0.05)
        try_update(a_md, "lp_mirror")

    # === Phase 4: Peak Shaving with Annealing (250s) ===
    print(f"\n=== Phase 4: Peak Shaving + Annealing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  PeakShave n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_ps, sc_ps = peak_shave_anneal(best_seq, t_alloc)

    # === Phase 5: Coordinate Descent (150s) ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  CoordDesc n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent_fast(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")

    # === Phase 6: Alternative Lengths (200s) ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    # Try lengths around the current best
    candidates = [n_best - 64, n_best - 32, n_best + 32, n_best + 64,
                  n_best + 128, n_best - 128, int(n_best * 1.5), int(n_best * 0.75)]
    candidates = [x for x in candidates if 256 <= x <= 4096 and x != n_best]
    candidates = list(dict.fromkeys(candidates))  # Remove duplicates preserving order

    for alt_n in candidates[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.3, 80.0)
        if t_alloc < 20:
            break
        print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)

        t_lp = t_alloc * 0.4

        # Quick L-BFGS

        t_lb = min(rem * 0.2, 50.0)
        if t_lb > 15:
            a_alt = lbgfs_refine(best_seq if len(best_seq) == alt_n else interpolate_sequence(best_seq, alt_n),
                                  t_lb, T_start=20.0, T_end=1.0, maxiter=4000)
            try_update(a_alt, f"lbgfs_n{alt_n}")

    # === Phase 7: Final Polish (100s) ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_final = min(rem * 0.4, 50.0)
        if t_final > 10:
            a_fin = lbgfs_refine(best_seq, t_final, T_start=2.0, T_end=0.1, maxiter=3000)
            try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.4, 40.0)
        if t_ps > 10:
            a_ps2, _ = peak_shave_anneal(best_seq, t_ps)
            try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_final = min(rem * 0.5, 30.0)
        if t_lp_final > 10:
            a_lp_fin, _ = lp_alternate(best_seq, t_lp_final, solver=solver)
            try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")

    if best_seq is None:
        best_seq = np.ones(768) / 768.0

    return [float(x) for x in best_seq]
