def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:

                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_start_score = best_score

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = i + n - 1
            if k_end > 2 * n - 2:
                k_end = 2 * n - 2

            # Compute unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # Current affected values
            affected = conv[k_start:k_end + 1].copy()
            # Sensitivity: for position k in [k_start, k_end], the coefficient is a[k-i]
            # i.e., a_shifted[m] = a[m] where m = k - k_start = k - i
            a_shifted = a[0:n].copy()  # a[k-i] for k = i, i+1, ..., i+n-1

            # Find best delta
            best_delta = 0.0
            best_new_max = M
            base_score = best_score

            for trial in range(n_trials):
                if trial == 0:
                    delta = -0.4 * a[i]
                elif trial == 1:
                    delta = -0.2 * a[i]
                elif trial == 2:
                    delta = -0.1 * a[i]
                elif trial == 3:
                    delta = -0.05 * a[i]
                elif trial == 4:
                    delta = 0.05 * a[i]
                elif trial == 5:
                    delta = 0.1 * a[i]
                elif trial == 6:
                    delta = 0.2 * a[i]
                else:
                    delta = 0.4 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                # Compute new affected values
                # Change = 2*delta*a[k-i] for k != 2i, 2*delta*a[i]+delta^2 for k=2i
                new_affected = affected + 2.0 * delta * a_shifted
                # Handle k = 2i special case
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    # Replace the linear approximation with the exact quadratic
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                new_sum = sum_a + delta
                if new_sum < 0.01:
                    continue
                new_score = 2.0 * n * new_max / (new_sum * new_sum)

                if new_score < best_new_max if False else new_max < best_new_max:
                    pass
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                # Apply update
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                sum_a += best_delta

                # Update conv incrementally
                ks = np.arange(k_start, k_end + 1)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 50:
                # Try smaller steps
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with smaller step sizes for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            for factor in [0.97, 0.94, 0.91, 0.985, 1.015, 1.03, 1.06, 1.09]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                else:
                    new_affected = new_affected.copy()

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-17:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score, sweeps


def mirror_descent(a0, time_limit, lr_init=0.01, top_k=20, verbose=False):
    """Mirror descent (multiplicative weight update) for the convex problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-6:
                break

        # Compute subgradient: weight the top-k peaks
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.01 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w = w / ws

        # Subgradient of max(conv) at a
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        # Mirror descent: a_new = a * exp(-lr * grad), then project to simplex
        scaled = a * np.exp(-lr * grad)
        a = simplex_project(scaled)

        lr *= 0.9999

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses (fictitious play)."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)

        for cand in [a, b, 0.5 * (a + b), a_avg]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def damped_lp_refine(a0, time_limit, solver=None):
    """Damped alternating LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 3.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 18))
    amplitudes = rng.random(n_harmonics) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    """Initialize with a cosine bell shape (concentrated in middle)."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.7)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    """Initialize with U-shape (more mass at edges)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_noise(n, base, rng, sigma=0.1):
    """Add Gaussian noise to a base sequence."""
    a = base.copy() + rng.standard_normal(n) * sigma
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def peak_pattern_break(a0, rng, magnitude=0.05):
    """Deliberately perturb the sequence to escape a local optimum pattern."""
    n = len(a0)
    a = a0.copy()
    # Find peak positions
    conv = fast_convolve(a)
    top_k = min(5, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    # Perturb coordinates that contribute to peaks
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        # Pick a few random contributing indices
        n_pick = max(1, len(idx) // 4)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a[i] *= (1.0 + rng.standard_normal() * magnitude)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s

        if rem > 40:
            # Try rounding to a "nice" length
            alt_n = int(round(n_best / 128.0) * 128)
            if alt_n != n_best and 256 <= alt_n <= 2560:
                t_alt = min(rem * 0.6, 100.0)
                init_alt = interpolate_seq(best_seq, alt_n)
                a_alt, s_alt, _ = fast_coord_descent(init_alt, t_alt, n_trials=10, label=f"alt{alt_n}")
                best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_final = min(rem * 0.7, 80.0)
            a_fin, s_fin, _ = fine_coord_descent(best_seq, t_final, label="final")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_fine")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_fin2, s_fin2, _ = fast_coord_descent(best_seq, min(rem * 0.8, 40.0), n_trials=16, label="last")
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "last_fcd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
