```python
def lp_direction(a, M):
    """Solve LP to find best direction b."""
    n = len(a)
    # Variables: b_0, ..., b_{n-1}
    # Maximize: sum(b)
    # Subject to: conv(a,b)_j <= M for j=0,...,2n-2
    #             b_i >= 0
    
    # c = -ones(n) (minimize -sum(b))

    # A_ub * b <= b_ub
    # For each j in 0,...,2n-2:
    #   sum_k a_k * b_{j-k} <= M
    # where k ranges over valid indices
    
    A_rows = []
    b_ub = []
    for j in range(2*n-1):
        k_min = max(0, j - (n-1))
        k_max = min(j, n-1)
        row = np.zeros(n)
        for k in range(k_min, k_max+1):
            b_idx = j - k
            row[b_idx] = a[k]
        A_rows.append(row)
        b_ub.append(M)
    
    A_ub = np.array(A_rows)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
```

```python
from scipy.optimize import linprog
import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lp_step(a, M):
    """Solve LP to find best direction b."""
    n = len(a)
    # Maximize sum(b) s.t. conv(a,b)_j <= M, b >= 0
    # In linprog form: minimize -sum(b) s.t. A_ub @ b <= M_vec
    
    c = -np.ones(n)
    
    # Build A_ub matrix (2n-1 x n)
    # Row j: conv(a,b)_j = sum_k a[k] * b[j-k] <= M
    # For row j, the coefficient of b[m] is a[j-m] if 0 <= j-m <= n-1, else 0
    # So A_ub[j, m] = a[j-m] if max(0, j-n+1) <= m <= min(j, n-1), else 0
    
    # This is a Toeplitz-like matrix
    A_ub = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        m_min = max(0, j - n + 1)
        m_max = min(j, n - 1)
        for m in range(m_min, m_max + 1):
            A_ub[j, m] = a[j - m]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None
```

The minimum possible value is `1/(2n-1)` (if we could achieve a perfectly flat convolution), giving a score of `2n/(2n-1)`.

Now, `conv(c,c)_j = sum_k c[k]*c[j-k]`. This is a bilinear form. The optimization is:
- Minimize `max_j F_j(c)` where `F_j(c) = sum_k c[k]*c[j-k]`
- Subject to `c >= 0`, `sum(c) = 1`.

This is a non-convex optimization problem (minimize the max of quadratic functions). However, we can use the following approach:

**Bisection on M**: Binary search on M (the maximum convolution value). For a given M, check if there exists `c >= 0`, `sum(c) = 1` such that `conv(c,c)_j <= M` for all j. This is a feasibility problem with quadratic constraints, which is non-convex in general.

But wait, if we fix one of the variables (say, fix c and ask for b such that `conv(c,b) <= M`), it becomes a linear constraint. This is exactly the LP we described!

So the algorithm is:
1. Start with some c (e.g., uniform).
2. Compute M = max(conv(c,c)).
3. Solve LP: max sum(b) s.t. conv(c,b) <= M, b >= 0.
4. Let b* be the solution. Rescale to sum(b*) = 1.
5. Update c = (1-t)*c + t*b*.
6. Repeat.

At convergence, we have a fixed point where c is optimal for the LP with the current M, and M = max(conv(c,c)).

This is a "best response" iteration, similar to the method of alternating minimization or the Sinkhorn algorithm.

The key question is: does this converge to the global optimum? Not necessarily, but it should give good results in practice.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based best-response iteration
as the primary method, combined with soft-max gradient polishing and multi-start
diversity. The LP approach finds the optimal direction at each step subject to
the convolution peak constraint, which is more principled than pure gradient methods.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ b = conv(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        m_min = max(0, j - n + 1)
        m_max = min(j, n - 1)
        if m_max >= m_min:
            cols = np.arange(m_min, m_max + 1)
            A[j, cols] = a[j - cols]
    return A


def lp_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Return b or None."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a_init, time_limit, verbose=True, label=""):
    """LP-based best-response iteration with adaptive step size."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init, score_from_a(a_init)
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 200

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        current_score = 2.0 * n * M

        b_new = lp_direction(a, M)
        if b_new is None:
            break
        s_new = b_new.sum()
        if s_new < 1e-15:
            break
        b_new = b_new / s_new

        # Line search: try various step sizes
        improved = False
        for t in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * b_new
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = score_from_a(a_trial)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                stuck = 0
                break

        if not improved:
            stuck += 1
            if stuck > max_stuck:
                # Add small random perturbation
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.standard_normal(n) * 0.01
                a = best_a + noise
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stuck = 0
                if time.monotonic() - t0 > time_limit:
                    break
                continue

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} LP iter={it} ({elapsed:.0f}s): score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def soft_max_objective(z, T, n):
    """Objective: 2n*T*log(sum(exp(b_j/T))) in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return 2.0 * n * T * lse


def soft_max_grad(z, T, n):
    """Gradient of soft-max objective in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()

    # Gradient w.r.t. a: 2n * w-convolved with a (cross-correlation)
    # d/da_i [sum_j w_j * (a*a)_j] = 2n * sum_j w_j * a[j-i] (with appropriate bounds)
    # This is 2n * cross_correlation(w, a) truncated to length n
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    grad_a = 2.0 * n * correl[:n]

    # Chain rule through softmax: grad_z = a * (grad_a - a . grad_a)
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return grad_z


def lbgfs_soft_max(a_init, time_limit, T=50, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        return soft_max_grad(z, T, n)

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14,
                                'maxfun': maxiter * 20, 'disp': False})
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a_init.copy()
    return a_out


def gradient_descent_smooth(a_init, time_limit, T_start=1.0, T_end=0.01,
                            lr_start=0.1, lr_end=0.01, max_iters=50000,
                            verbose=True, label=""):
    """Gradient descent on soft-max approximation with annealing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        # Compute gradient of soft-max
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()

        # Gradient: 2n * sum_j w_j * a[j-i] (cross-correlation, truncated)
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad = 2.0 * n * correl[:n]

        # Update: multiplicative (mirror) step on simplex
        log_update = -lr * grad / (a + 1e-30)
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} GD iter={it} ({elapsed:.0f}s): T={T:.4f} lr={lr:.4f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score

        # Wait, if j=2i, then the single term in the sum is a[i]*a[i]=a[i]^2, and its derivative is 2*a[i] = 2*a[j-i]. So the formula 2*a[j-i] works universally!
        
        # Let me verify: j=1, i=0: 0<=1<n, grad[0] = 2*a[1]. But the actual derivative should be a[1] (from a[0]*a[1], derivative w.r.t a[0] is a[1]).
        # Hmm, the issue is that for j=1, the sum has terms: k=0 gives a[0]*a[1], k=1 gives a[1]*a[0].
        # d/da[0] of a[0]*a[1] = a[1]. d/da[0] of a[1]*a[0] = a[1]. Total: 2*a[1].
        # But wait, a[0]*a[1] and a[1]*a[0] are the SAME term! The sum is over k, and for j=1 with n>=2:
        # sum_k a[k]*a[1-k] where k ranges so that 0<=k<n and 0<=1-k<n.
        # k=0: a[0]*a[1]. k=1: a[1]*a[0]. These are indeed two separate terms in the sum (different k values).
        # So the derivative w.r.t a[0] is: from k=0 term: a[1]. From k=1 term: a[1]. Total: 2*a[1].
        # And 2*a[j-i] = 2*a[1-0] = 2*a[1]. ✓!
        
        # Great, so the formula is simply: grad[i] = 2*a[j-i] if 0<=j-i<n, else 0.
        grad = np.zeros(n)
        valid = np.arange(n)
        targets = j_peak - valid
        mask = (targets >= 0) & (targets < n)
        grad[mask] = 2.0 * a[targets[mask]]

        # Project to tangent space of simplex (subtract mean weighted by a)
        # For multiplicative updates on the simplex, we just do:
        a = a * np.exp(-lr * grad / (a.sum() / n + 1e-10))
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        # Anneal learning rate
        lr = max(0.001, 0.05 * (0.999 ** it))

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} PT iter={it} ({elapsed:.0f}s): lr={lr:.5f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True, label=""):
    """Fast coordinate descent for final polishing."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5:
        no_improve += 1
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old = a[i]
            if old < 1e-10:
                continue
            for f in factors:
                a[i] = old * f
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss < 1e-15:
                    a[i] = old
                    break
                a = a / ss
                sc = score_from_a(a)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] = old
                    a = a / a.sum()
        if not improved:
            no_improve += 10

        if verbose and no_improve <= 5:
            elapsed = time.monotonic() - t0
            print(f"    {label} CD pass ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def evaluate_candidate(a):
    """Safely evaluate using the provided evaluate_sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    try:
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based best-response iteration,
    soft-max gradient descent, targeted peak reduction, and coordinate descent.
    Uses multiple initializations and sequence lengths for diversity.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_candidate(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.7f} (n={len(a)})")

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_candidate(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    # Generate diverse initializations
    def make_inits(n, count=5):
        inits = []
        # Uniform
        inits.append(np.ones(n) / n)
        # Random
        for _ in range(2):
            r = rng.random(n)
            r = r / r.sum()
            inits.append(r)
        # Sinusoidal (non-negative)
        x = np.linspace(0, 2 * np.pi, n)
        s = 1.0 + 0.5 * np.cos(x)
        s = s / s.sum()
        inits.append(s)
        # Edge-heavy
        edge = np.ones(n)
        center = n // 2
        for i in range(n):
            edge[i] = 1.0 + 0.8 * abs(i - center) / center
        edge = edge / edge.sum()
        inits.append(edge)
        return inits

    # Try multiple lengths
    if parent is not None:
        base_n = len(parent)
    else:
        base_n = 768

    lengths_to_try = [base_n]
    for factor in [0.75, 1.0, 1.25, 1.5, 2.0]:
        alt_n = int(base_n * factor)
        if 64 <= alt_n <= 4096 and alt_n not in lengths_to_try:
            lengths_to_try.append(alt_n)
    # Only keep a few to manage time
    lengths_to_try = lengths_to_try[:4]

    print(f"Lengths to try: {lengths_to_try}")

    # === Phase 1: LP-based optimization from parent ===
    if parent is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 60:
            t_alloc = min(rem * 0.4, 400.0)
            print(f"\n=== Phase 1: LP from parent (n={base_n}, budget={t_alloc:.0f}s) ===")
            a_lp, _ = lp_optimize(parent, t_alloc, label="LP_parent")
            update_best(a_lp, "LP_parent")

    # === Phase 2: LP from random/structured inits at best length ===
    n_work = base_n if best_seq is None else len(best_seq)
    inits = make_inits(n_work, count=4)
    
    for idx, init in enumerate(inits):
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alloc = min(rem * 0.25, 150.0)
        print(f"\n=== Phase 2.{idx+1}: LP from init[{idx}] (n={n_work}, budget={t_alloc:.0f}s) ===")
        a_lp, _ = lp_optimize(init, t_alloc, label=f"LP_init{idx}")
        update_best(a_lp, f"LP_init{idx}")

    # === Phase 3: Gradient smoothing from current best ===
    if best_seq is not None:
        n_best = len(best_seq)
        for T in [50, 20, 10, 5]:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem * 0.2, 60.0)
            print(f"\n=== Phase 3: GD T={T} (n={n_best}, budget={t_alloc:.0f}s) ===")
            a_gd, _ = gradient_descent_smooth(best_seq, t_alloc, T_start=float(T),
                                              T_end=float(T) * 0.05, label=f"GD_T{T}")
            update_best(a_gd, f"GD_T{T}")

    # === Phase 4: L-BFGS soft-max polishing ===
    if best_seq is not None:
        n_best = len(best_seq)
        for T in [30, 10, 5, 2]:
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 50.0)
            print(f"\n=== Phase 4: L-BFGS T={T} (n={n_best}, budget={t_alloc:.0f}s) ===")
            a_lb = lbgfs_soft_max(best_seq, t_alloc, T=float(T), maxiter=3000)
            update_best(a_lb, f"lbfgs_T{T}")

    # === Phase 5: Peak-targeted descent ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.35, 100.0)
            print(f"\n=== Phase 5: Peak Targeted (n={len(best_seq)}, budget={t_alloc:.0f}s) ===")
            a_pt, _ = peak_targeted_descent(best_seq, t_alloc, label="PT")
            update_best(a_pt, "peak_targeted")

    # === Phase 6: Coordinate descent polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_alloc = min(rem * 0.5, 60.0)
            print(f"\n=== Phase 6: Coord Descent (n={len(best_seq)}, budget={t_alloc:.0f}s) ===")
            a_cd, _ = coordinate_descent(best_seq, t_alloc, label="CD")
            update_best(a_cd, "coord_descent")

    # === Phase 7: Second round LP from current best ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            t_alloc = min(rem * 0.5, 120.0)
            print(f"\n=== Phase 7: LP round 2 (n={len(best_seq)}, budget={t_alloc:.0f}s) ===")
            a_lp2, _ = lp_optimize(best_seq, t_alloc, label="LP_r2")
            update_best(a_lp2, "LP_round2")

    # === Phase 8: Try alternative length ===
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 40:
            alt_n = int(n_best * 1.3) if n_best < 2000 else int(n_best * 0.8)
            if alt_n != n_best and 64 <= alt_n <= 4096:
                t_alloc = min(rem * 0.5, 80.0)
                init_alt = interpolate_sequence(best_seq, alt_n)
                print(f"\n=== Phase 8: Alt length n={alt_n} (budget={t_alloc:.0f}s) ===")
                a_alt, _ = lp_optimize(init_alt, t_alloc * 0.6, label=f"LP_n{alt_n}")
                update_best(a_alt, f"alt_n{alt_n}")
                if best_seq is not None and len(best_seq) == alt_n:
                    rem2 = deadline - time.monotonic() - 15
                    if rem2 > 10:
                        a_cd2, _ = coordinate_descent(best_seq, min(rem2 * 0.6, 20.0), label="CD_alt")
                        update_best(a_cd2, f"cd_alt_n{alt_n}")

    # === Phase 9: Final polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_fin, _ = coordinate_descent(best_seq, min(rem * 0.7, 10.0), label="final")
            update_best(a_fin, "final_cd")
            
            rem = deadline - time.monotonic() - 2
            if rem > 3:
                a_gdf, _ = gradient_descent_smooth(best_seq, min(rem * 0.8, 5.0),
                                                   T_start=1.0, T_end=0.1,
                                                   lr_start=0.01, lr_end=0.001,
                                                   max_iters=5000, label="final_gd")
                update_best(a_gdf, "final_gd")

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = evaluate_candidate(best_seq)

    # Final verification
    fs = evaluate_candidate(best_seq)
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
