import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, line_search
from time import monotonic
import cvxpy as cp
from itertools import product


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_star] - 1.0) / (rho_star + 1.0)
    return np.maximum(v - theta, 0.0)

    b = fast_convolve(a)
    max_val = b.max()
    # Find all peaks within tolerance
    tol = 1e-10 * max_val + 1e-15
    peaks = np.where(b >= max_val - tol)[0]
    
    # For each peak, the gradient is 2 * a reversed and shifted
    # grad_i = 2 * a[k - i] for peak at k
    grad = np.zeros(n)
    for k in peaks:
        # grad_i = 2 * a[k-i] (with a[j]=0 for j out of range)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(peaks)
    return grad, max_val


def nesterov_subgradient(a0, time_limit, c=1.0, verbose=False):
    """
    Nesterov-accelerated subgradient method for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    x = np.maximum(a0, 0).astype(np.float64)
    s = x.sum()
    x = x / s
    
    y = x.copy()
    best_x = x.copy()
    best_score = 2.0 * n * fast_convolve(x).max()
    t = 1
    t0 = monotonic()
    
    while monotonic() - t0 < time_limit:
        # Nesterov momentum
        alpha_t = (1 + np.sqrt(1 + 4 * t * t)) / 2
        t_new = t + 1
        alpha_new = (1 + np.sqrt(1 + 4 * t_new * t_new)) / 2
        beta = (alpha_t - 1) / alpha_new
        
        # Compute subgradient at y
        grad, max_val = subgradient_of_max_conv(y)
        
        # Step size
        epsilon = 0.01
        eta = c * epsilon / np.sqrt(t_new)
        
        # Update
        x_new = y - eta * grad
        x_new = project_simplex(x_new)
        
        # Nesterov update
        y = (1 - beta) * x_new + beta * y
        
        t = t_new
        
        # Evaluate
        max_conv = fast_convolve(x_new).max()
        score = 2.0 * n * max_conv
        if score < best_score - 1e-14:
            best_score = score
            best_x = x_new.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = monotonic() - t0
            print(f"    Nesterov t={t} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_x, best_score


def lbgfs_continuation(a0, time_limit, temperatures=None, verbose=False):
    """
    L-BFGS continuation method: start with high T, decrease to low T.
    """
    n = len(a0)
    if temperatures is None:
        temperatures = [500, 200, 100, 50, 25, 10, 5, 2, 1]
    
    a = np.maximum(a0, 1e-30).astype(np.float64)
    s = a.sum()
    a = a / s
    t0 = monotonic()
    
    for T in temperatures:
        if monotonic() - t0 > time_limit:
            break
        time_for_T = min(60.0, (time_limit - (monotonic() - t0)) / max(1, len(temperatures)))
        
        z0 = np.log(np.maximum(a, 1e-30))
        z0 = z0 - z0.mean()
        stopped = [False]
        
        def func(z):
            if monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_curr = exp_z / exp_z.sum()
            b = fast_convolve(a_curr)
            max_b = b.max()
            scaled_b = (b - max_b) / T
            exp_b = np.exp(scaled_b)
            lse = max_b + T * np.log(exp_b.sum())
            return 2.0 * n * lse / T
        
        def jac(z):
            if monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros(n)
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_curr = exp_z / exp_z.sum()
            b = fast_convolve(a_curr)
            max_b = b.max()
            scaled_b = (b - max_b) / T
            exp_b = np.exp(scaled_b)
            w = exp_b / exp_b.sum()
            
            # Gradient of sum_k w[k] * (a*a)[k] wrt a
            grad_a = np.zeros(n)
            for k in range(len(w)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_curr[k - idx]
            
            grad_a *= 2.0 * n
            S = np.dot(grad_a, a_curr)
            grad_z = a_curr * (grad_a - a_curr * S)
            return grad_z
        
        try:
            res = minimize(
                fun=func, x0=z0, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': min(2000, int(time_for_T * 50)), 
                         'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': int(time_for_T * 100), 'disp': False}
            )
            if not stopped[0]:
                z_final = res.x
                z_shifted = z_final - z_final.max()
                a = np.exp(z_shifted)
                a = a / a.sum()
        except Exception:
            pass
        
        if verbose:
            elapsed = monotonic() - t0
            score = 2.0 * n * fast_convolve(a).max()
            print(f"  L-BFGS T={T} ({elapsed:.0f}s): score={score:.6f}")
    
    return a


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent with line search for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0
    
    order = np.random.permutation(n)
    idx = 0
    
    while monotonic() - t0 < time_limit:
        i = order[idx % n]
        idx += 1
        
        # Try changing a[i] while keeping sum = 1
        # If we change a[i] to a[i] + delta, we need to adjust other elements
        # Simplest: move delta from a[i] to a[j] for some j, or spread it
        
        # Strategy: try reducing a[i] and spreading the excess uniformly
        old_val = a[i]
        if old_val < 1e-10:
            continue
        
        for factor in [0.99, 0.98, 0.95, 0.9, 0.8]:
            new_val = old_val * factor
            diff = old_val - new_val
            if diff < 1e-12:
                continue
            
            a_trial = a.copy()
            a_trial[i] = new_val
            # Spread diff to all other elements proportionally
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            others_sum = a_trial[mask].sum()
            if others_sum < 1e-15:
                continue
            a_trial[mask] += diff * a_trial[mask] / others_sum
            a_trial = a_trial / a_trial.sum()
            
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve = 0
                break
        else:
            no_improve += 1
        
        if no_improve > 3 * n:
            # Reset with perturbation
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            order = np.random.permutation(n)
            
        if verbose and idx % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def fictitious_play(a0, time_limit, solver=None, verbose=False):
    """LP-based fictitious play."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    iters = 0
    avg_a = a.copy()
    avg_b = b.copy()
    stagnation = 0
    
    while monotonic() - t0 < time_limit:
        iters += 1
        
        # LP best response for b given a
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
            avg_b = (iters * avg_b + b) / (iters + 1)
        
        # LP best response for a given b
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
            avg_a = (iters * avg_a + a) / (iters + 1)
        
        # Evaluate candidates
        for cand in [a, b, avg_a, avg_b, 0.5*(a+b)]:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand.copy()
                stagnation = 0
            else:
                stagnation += 1
        
        # Perturbation if stagnating
        if stagnation > 100:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = a.copy()
            stagnation = 0
        
        if verbose and iters % 20 == 0:
            elapsed = monotonic() - t0
            print(f"  FP iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, iters


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def score_fn(a):
    """Compute the score for a normalized sequence."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def check_and_update(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def peak_shave_advanced(a0, time_limit):
    """Advanced peak shaving with multiple strategies."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    step = 0
    while monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        if peak_val <= best_score / (2.0 * n) + 1e-15:
            break
        
        # Get contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        partners = peak_k - idx
        contribs = a[idx] * a[partners]
        
        improved = False
        
        # Strategy 1: Reduce top contributor
        top_idx = np.argsort(contribs)[::-1][:10]
        for ci in top_idx:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-12:
                continue
            for factor in [0.97, 0.95, 0.9, 0.85, 0.8]:
                a_trial = a.copy()
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Strategy 2: Reduce pair
        if not improved:
            for ci in top_idx[:5]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or j < 0 or j >= n or a[j] < 1e-12:
                    continue
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        # Strategy 3: Small random perturbation
        if not improved:
            for _ in range(3):
                a_trial = a + np.random.standard_normal(n) * 0.0003
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and step > 100:
            break
    
    return best_a, best_score
