"""
Minimizes 2*n*max(auto_conv)/sum(a)^2 using adaptive multi-peak subgradient
descent with exact line search, direct pairwise swap exploitation of quadratic
structure, active-set peak balancing, and simulated annealing restarts.
Key novelties vs previous approach:
1. Additive updates with exact 1D line search (vs fixed-step multiplicative)
2. Direct pairwise 2-variable optimization exploiting b[k] quadratic structure
3. Active-set multi-peak balancing (equalize top-K peaks)
4. Simulated annealing for global exploration
5. Multi-length exploration with early pruning
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_and_conv(a):
    """Compute 2*n*max(conv(a,a)) and return (score, b). Assumes sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max()), b


def subgradient_at_peak(a, k):
    """Compute subgradient of b[k] w.r.t. a: grad[i] = 2*a[k-i] (zero-padded)."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def multi_peak_subgradient(a, b, top_k=5, temperature=0.01):
    """Weighted subgradient from top-K peaks with softmax weighting."""
    n = len(a)
    b_top_idx = np.argsort(b)[::-1][:top_k]
    weights = np.exp(b[b_top_idx] / temperature - b[b_top_idx].max())
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, b_top_idx):
        grad += w * subgradient_at_peak(a, int(k))
    return grad


def line_search_score(a, direction, step_sizes=None):
    """Try several step sizes in direction, return best (a_new, score)."""
    n = len(a)
    if step_sizes is None:
        step_sizes = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
    best_score = float('inf')
    best_a = a.copy()
    s_a = a.sum()
    for eta in step_sizes:
        a_trial = a + eta * direction
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        sc, _ = score_and_conv(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
    return best_a, best_score


def refine_line_search(a, direction, eta_lo, eta_hi):
    """Bisection-style refinement of step size in [eta_lo, eta_hi]."""
    best_eta = (eta_lo + eta_hi) / 2
    best_score = float('inf')
    for _ in range(12):
        for eta in [best_eta, (eta_lo + best_eta) / 2, (best_eta + eta_hi) / 2]:
            a_trial = a + eta * direction
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc, _ = score_and_conv(a_trial)
            if sc < best_score:
                best_score = sc
                best_eta = eta
    a_new = a + best_eta * direction
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new, best_score


def pairwise_swap_optimize(a, b, max_pairs=50, max_delta_frac=0.3):
    """Try moving mass from position p to position q to reduce the peak.
    Returns improved a and score."""
    n = len(a)
    k_peak = int(np.argmax(b))
    best_score = float(2.0 * n * b.max())
    best_a = a.copy()

    # Get contributing indices to the peak
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    idx = np.arange(js, je + 1)
    contribs = a[idx] * a[k_peak - idx]
    
    # Sort by contribution, take top
    top_local = np.argsort(contribs)[::-1][:max_pairs]
    candidates_p = [int(idx[t]) for t in top_local if a[idx[t]] > 1e-10]

    # For each p, try moving to various q
    for p in candidates_p:
        if a[p] < 1e-10:
            continue
        # The "natural" target: positions that don't contribute to the peak
        # or positions where moving mass reduces the peak
        grad_p = 2.0 * (a[k_peak - p] if 0 <= k_peak - p < n else 0.0)
        
        # Try moving to positions with low subgradient value
        grad_all = np.zeros(n)
        js2 = max(0, k_peak - n + 1)
        je2 = min(n - 1, k_peak)
        idx2 = np.arange(js2, je2 + 1)
        grad_all[idx2] = 2.0 * a[k_peak - idx2]
        
        # Positions where grad is low (good targets)
        target_candidates = np.argsort(grad_all)[:20]
        for q in target_candidates:
            q = int(q)
            if q == p:
                continue
            if a[q] > 0.5:  # don't over-concentrate
                continue
            grad_q = grad_all[q]
            # Moving from p to q changes b[k] by ~ delta*(grad_q - grad_p)
            if grad_q >= grad_p:
                continue
            # Try several delta values
            max_d = a[p] * max_delta_frac
            for frac in [0.1, 0.25, 0.5, 0.75]:
                delta = max_d * frac
                if delta < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[p] -= delta
                a_trial[q] += delta
                if a_trial.min() < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc, b_new = score_and_conv(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    b = b_new  # update for subsequent pairs
                    a = a_trial.copy()
                    break

    return best_a, best_score


def active_set_balance(a, b, top_k=5):
    """Find a direction that reduces all top-K peaks simultaneously.
    Uses the idea: direction = -weighted_sum of subgradients, projected."""
    n = len(a)
    top_idx = np.argsort(b)[::-1][:top_k]
    # Weight proportional to how close to max
    b_max = b[top_idx[0]]
    weights = np.ones(top_k)
    for i, k in enumerate(top_idx):
        weights[i] = np.exp((b[k] - b_max) / 0.01)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        grad += w * subgradient_at_peak(a, int(k))
    
    # Project to tangent space (sum=0)
    grad = grad - np.dot(a, grad)
    
    # The direction is -grad (projected)
    direction = -grad
    return direction


def adaptive_subgradient_method(a_init, time_limit, top_k=5, T_init=0.05,
                                 T_final=0.001, verbose=True, label=""):
    """Main optimization loop: adaptive subgradient + line search + pairwise swaps."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score, _ = score_and_conv(a)
    t0 = time.monotonic()
    stuck_count = 0
    prev_score = best_score
    iter_count = 0
    swap_every = 20
    anneal_temp = 0.1

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        b = fast_convolve(a)
        current_score = float(2.0 * n * b.max())

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Adaptive temperature for multi-peak weighting
        progress = min(1.0, iter_count / 5000)
        T = T_init * (T_final / T_init) ** progress

        # Compute multi-peak subgradient
        grad = multi_peak_subgradient(a, b, top_k=top_k, temperature=T)
        # Project to tangent space
        grad = grad - np.dot(a, grad)
        # Direction
        direction = -grad
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        direction = direction / dir_norm

        # Line search: coarse first
        a_new, sc_new = line_search_score(a, direction)

        # If stuck, try active set balance direction
        if stuck_count > 50:
            dir_balance = active_set_balance(a, b, top_k=top_k + 2)
            dn = np.linalg.norm(dir_balance)
            if dn > 1e-15:
                dir_balance = dir_balance / dn
                a_bal, sc_bal = line_search_score(a, dir_balance)
                if sc_bal < sc_new:
                    a_new, sc_new = a_bal, sc_bal

        # Periodic pairwise swaps
        if iter_count % swap_every == 0:
            a_swapped, sc_swapped = pairwise_swap_optimize(a_new, fast_convolve(a_new),
                                                            max_pairs=30)
            if sc_swapped < sc_new:
                a_new, sc_new = a_swapped, sc_swapped

        # Accept/reject (simulated annealing)
        if sc_new < current_score:
            a = a_new
        elif stuck_count > 100 and np.random.random() < np.exp(-(current_score - sc_new) * 5.0 / max(anneal_temp, 0.01)):
            a = a_new  # accept worse with some probability
        else:
            # Try smaller steps
            a_new2, sc_new2 = refine_line_search(a, direction, 0.001, 0.01)
            if sc_new2 < current_score:
                a = a_new2
            else:
                # Random perturbation if very stuck
                if stuck_count > 200:
                    rng = np.random.default_rng(iter_count * 31 + 7)
                    noise = rng.standard_normal(n) * 0.03
                    a_pert = a + noise
                    a_pert = np.maximum(a_pert, 1e-15)
                    a_pert = a_pert / a_pert.sum()
                    sc_pert, _ = score_and_conv(a_pert)
                    if sc_pert < current_score * 1.1:  # accept if not too bad
                        a = a_pert
                    stuck_count = 0
                    anneal_temp *= 0.95

        anneal_temp *= 0.999

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iter_count} ({elapsed:.0f}s): "
                  f"cur={current_score:.7f} best={best_score:.7f} T={T:.5f} stuck={stuck_count}")

    return best_a, best_score

                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc, _ = score_and_conv(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        if not improved:
            # Try multi-peak balancing step
            direction = active_set_balance(a, b, top_k=7)
            dn = np.linalg.norm(direction)
            if dn > 1e-14:
                direction /= dn
                a_bal, sc_bal = line_search_score(a, direction,
                                                  step_sizes=[0.005, 0.01, 0.02, 0.05])
                if sc_bal < best_score - 1e-13:
                    best_score = sc_bal
                    best_a = a_bal.copy()
                    a = a_bal.copy()
                    no_improve = 0
                    improved = True

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    {label} polish done ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.arange(n) - n / 2
    sigma = sigma_frac * n
    a = np.exp(-x**2 / (2 * sigma**2))
    a = a / a.sum()
    return a


def sinc_init(n, width_frac=0.5):
    """Sinc-like initialization (IFFT of rectangular window)."""
    half = n // 2
    spectrum = np.zeros(n)
    w = int(width_frac * half)
    spectrum[half - w:half + w + 1] = 1.0
    a = np.real(irfft(spectrum, n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_structured_init(n, seed):
    """Random initialization with smooth structure."""
    rng = np.random.default_rng(seed)
    # Generate smooth random signal
    n_modes = max(3, n // 32)
    freqs = rng.integers(1, n_modes + 1, size=max(3, n // 16))
    phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
    amps = rng.uniform(0.5, 1.0, size=len(freqs))
    x = np.arange(n)
    a = np.ones(n)
    for f, ph, am in zip(freqs, phases, amps):
        a += am * np.cos(2 * np.pi * f * x / n + ph)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Explore multiple (length, init) combinations quickly
    Phase 2: Focused adaptive subgradient on best candidates
    Phase 3: Intensive local polishing
    Phase 4: Final micro-optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # === Phase 1: Quick exploration of (length, init) combinations ===
    print("\n=== Phase 1: Exploration ===")
    n_parent = len(parent) if parent is not None else 768

    # Define candidates to try
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(n_parent)
    for factor in [0.5, 0.625, 0.75, 1.25, 1.5, 2.0]:
        alt_n = int(n_parent * factor)
        if 128 <= alt_n <= 4096 and alt_n not in lengths_to_try:
            lengths_to_try.append(alt_n)
    # Also try some "round" numbers
    for rn in [256, 384, 512, 640, 768, 1024, 1280, 1536, 2048]:
        if 128 <= rn <= 4096 and rn not in lengths_to_try:
            lengths_to_try.append(rn)
    lengths_to_try = sorted(set(lengths_to_try))[:10]

    # Quick evaluation: for each length, try a couple inits with short optimization
    candidates = []  # list of (initialization, length, quick_score)
    
    for n_try in lengths_to_try[:6]:
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_per = min(rem / max(1, len(lengths_to_try[:6]) - (lengths_to_try.index(n_try))), 60.0)
        t_per = max(t_per, 15.0)
        
        inits = []
        if parent is not None:
            inits.append(interpolate_sequence(parent, n_try))
        inits.append(gaussian_init(n_try, 0.25))
        inits.append(random_structured_init(n_try, seed + n_try))

        for ii, init_a in enumerate(inits[:2]):
            if time.monotonic() > deadline - 350:
                break
            t_alloc = min(t_per / 2, 30.0)
            lbl = f"exp_n{n_try}_i{ii}"
            a_res, sc_res = adaptive_subgradient_method(
                init_a, t_alloc, top_k=5, T_init=0.03, T_final=0.005,
                verbose=False, label=lbl
            )
            candidates.append((a_res, n_try, sc_res))
            best_seq, best_score = check_and_update(a_res, lbl, best_seq, best_score)
            print(f"    {lbl}: score={sc_res:.7f}")

    # Sort candidates by score
    candidates.sort(key=lambda x: x[2])
    print(f"  Top candidates: {[f'{c[2]:.6f}' for c in candidates[:5]]}")

    # === Phase 2: Focused optimization on best candidates ===
    print(f"\n=== Phase 2: Focused Optimization (best={best_score:.7f}) ===")
    
    # Take the best 2-3 candidates and optimize harder
    focus_candidates = []
    for a_c, n_c, sc_c in candidates[:3]:
        if sc_c < best_score + 0.05:  # only focus on competitive ones
            focus_candidates.append((a_c, n_c))
    
    if not focus_candidates and best_seq is not None:
        focus_candidates.append((best_seq, len(best_seq)))

    for ci, (a_focus, n_focus) in enumerate(focus_candidates[:2]):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(rem * 0.5, 400.0)
        lbl = f"focus_{ci}_n{n_focus}"
        print(f"  {lbl}: budget={t_alloc:.0f}s, start_score={score_and_conv(a_focus)[0]:.7f}")
        
        # Run adaptive subgradient with more aggressive settings
        for attempt in range(3):
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_attempt = min(rem * 0.35, 150.0)
            # Vary top_k and temperature
            tk = [5, 8, 12][attempt % 3]
            Ti = [0.05, 0.02, 0.01][attempt % 3]
            Tf = [0.001, 0.003, 0.005][attempt % 3]
            a_res, sc_res = adaptive_subgradient_method(
                a_focus, t_attempt, top_k=tk, T_init=Ti, T_final=Tf,
                verbose=True, label=f"{lbl}_a{attempt}"
            )
            best_seq, best_score = check_and_update(a_res, f"{lbl}_a{attempt}",
                                                     best_seq, best_score)
            if a_res is not None:
                a_focus = a_res  # warm start for next attempt

    # === Phase 3: Intensive polishing ===
    print(f"\n=== Phase 3: Intensive Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem * 0.6, 300.0)
            print(f"  Polish: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_pol, sc_pol = intensive_polish(best_seq, t_alloc, verbose=True,
                                              label="polish")
            best_seq, best_score = check_and_update(a_pol, "polish",
                                                     best_seq, best_score)

    # === Phase 4: Final micro-optimization ===
    print(f"\n=== Phase 4: Final Micro-opt (best={best_score:.7f}) ===")
    if best_seq is not None:
        # Another round of adaptive subgradient with very low temperature
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem * 0.5, 100.0)
            a_fin, sc_fin = adaptive_subgradient_method(
                best_seq, t_alloc, top_k=3, T_init=0.005, T_final=0.0005,
                verbose=True, label="micro"
            )
            best_seq, best_score = check_and_update(a_fin, "micro",
                                                     best_seq, best_score)

        # Final polish
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.6, 60.0)
            a_fin2, sc_fin2 = intensive_polish(best_seq, t_alloc, verbose=True,
                                                label="final_polish")
            best_seq, best_score = check_and_update(a_fin2, "final_polish",
                                                     best_seq, best_score)

        # One more micro subgradient
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_fin3, sc_fin3 = adaptive_subgradient_method(
                best_seq, min(rem * 0.5, 30.0), top_k=3,
                T_init=0.002, T_final=0.0002, verbose=False, label="final_micro"
            )
            best_seq, best_score = check_and_update(a_fin3, "final_micro",
                                                     best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)}, time={elapsed:.0f}s)")
    return [float(x) for x in best_seq]
