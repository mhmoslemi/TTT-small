∂(a*a)[k] / ∂a[i] = a[k-i] + a[i-k] (with appropriate boundary conditions, i.e., only terms where the indices are valid)

This is essentially a "windowed" version of a.

Now, for the mirror descent approach on the simplex:
- Update: a_new[i] = a[i] * exp(-eta * g[i]) / Z
- Where Z is the normalization constant

This is the multiplicative weight update. It's natural because it automatically maintains non-negativity and we can normalize to maintain the simplex constraint.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

    A = rfft(a, fft_size)

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)]) * target_sum
    # Scale to sum = 1 first, then scale back
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # Compute gradient of smooth max at y
        b = fast_convolve(y)
        b_max = b.max()
        log_w = beta * (b - b_max)
        log_w = np.clip(log_w, -50, 50)
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        
        # Gradient: g[i] = sum_k w[k] * (a[k-i] + a[i-k])
        fft_size = 1 << (2 * n - 1).bit_length()
        w_ext = np.zeros(fft_size)
        w_ext[:len(w)] = w
        a_ext = np.zeros(fft_size)
        a_ext[:n] = y
        
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a_ext, fft_size)
        
        g1 = irfft(W_ext * np.conj(A_ext), fft_size)[:n]
        g2 = irfft(A_ext * np.conj(W_ext), fft_size)[:n]
        g = (g1 + g2) / beta  # Divide by beta since d/dT log_sum_exp = softmax weighted
        
        # Actually, the gradient of (1/beta)*log(sum exp(beta*b_k)) with respect to a[i]
        # = sum_k (beta * db_k/da[i] * exp(beta*b_k)) / (beta * sum_j exp(beta*b_j))
        # = sum_k (db_k/da[i] * w[k])
        # = sum_k w[k] * (a[k-i] + a[i-k])
        # So no division by beta needed. Let me fix.
        g = g1 + g2
        
        # Gradient step
        y_new = y - step * g
        # Project onto simplex
        a_new = project_simplex(y_new, target_sum=1.0)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        y_new_fista = a_new + ((t_k - 1) / t_new) * (a_new - a)
        y = y_new_fista
        a = a_new
        t_k = t_new
        
        # Periodic evaluation and adaptation
        if it % 200 == 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                # Decrease step size
                step *= 0.95
                # Increase beta
                beta = min(beta * 1.1, 100.0)
            else:
                no_improve += 1
                if no_improve > 10:
                    step *= 0.8
                    no_improve = 0
                if no_improve > 30:
                    # Restart
                    rng = np.random.default_rng(it)
                    a = best_a + 0.001 * rng.standard_normal(n)
                    a = project_simplex(a, target_sum=1.0)
                    y = a.copy()
                    t_k = 1.0
                    no_improve = 0
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            obj = fast_convolve(a).max()
            print(f"    fista iter={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"step={step:.5f} beta={beta:.1f} t={elapsed:.0f}s")
    
    obj = fast_convolve(best_a).max()
    if obj < best_obj:
        best_obj = obj
    return best_a, best_obj


def lp_solve_reduced(f, n, top_k, eps=1e-8):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None, 0.0
    
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def run_lp_phase(f_init, time_limit, top_k_init=30, top_k_max=100):
    """LP-based optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 50 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)

        g, M = lp_solve_reduced(f, n, top_k)
        if g is None:
            break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 10:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()

        for t in [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    lp iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"top_k={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]

    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(0, n, max(1, n // 500)):  # Subsample for speed
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj
