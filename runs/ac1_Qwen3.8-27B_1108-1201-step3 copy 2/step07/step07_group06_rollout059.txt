    """
    Compute corr[j] = sum_k w[k] * a[k-j] for j=0..n-1.
    Uses FFT cross-correlation.
    """

    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    return corr[:n]


def softmax_forward(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s

                   beta1=0.9, beta2=0.999, eps=1e-8,

    and adaptive restarts.

    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(a)
    
    m = np.zeros(n)
    v = np.zeros(n)

    no_improve = 0

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / budget)
        
        # Temperature schedule
        if progress < 0.2:
            T = T_init
        else:
            p = (progress - 0.2) / 0.8
            T = T_init * (T_final / T_init) ** p
        
        # Adaptive K (number of peaks to focus on)

        b = fast_convolve(a_curr)

                # Accept (always in SA mode)

                a_curr = softmax_forward(z)

        a_new = softmax_forward(z)

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
(1) Top-K peak-aware soft-max gradient descent in log-space with Adam,
(2) Iterative water-filling for targeted peak equalization,

Key innovation: combines smooth gradient flow with discrete water-filling

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_corr(a, w, fft_size):
    """Linear cross-correlation: corr[j] = sum_k w[k]*a[k-j] for j=0..n-1."""
    n = len(a)
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    return corr[:n]


def softmax_forward(z):
    """Stable softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def score_normalized(a):
    """Score for sum-normalized non-negative sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def waterfill_step(a, step_size, n_peaks=5):
    """
    One water-filling step: for top peaks, move contributing pairs
    toward equalization.
    """
    n = len(a)
    a = a.copy()
    b = fast_convolve(a)
    peak_indices = np.argsort(b)[::-1][:n_peaks]

    for k in peak_indices:
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            if i >= j:
                continue
            diff = a[i] - a[j]
            if diff != 0:
                a[i] -= step_size * diff
                a[j] += step_size * diff

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def freq_restart(a, rng, n_modes=5, amplitude=0.05):
    """Perturb in frequency domain for global exploration."""
    n = len(a)
    a = a.copy()
    A = rfft(a)
    for _ in range(n_modes):
        k = rng.integers(1, min(n // 2, 200))
        phase = rng.uniform(0, 2 * np.pi)
        A[k] += amplitude * np.exp(1j * phase)
        if k < len(A) - 1 - k:
            A[-k] += amplitude * np.exp(-1j * phase)
    a_new = irfft(A, 2 * n - 1)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

def coordinate_descent_fast(a0, time_budget, max_sweeps=100):
    """Fast coordinate descent with adaptive step sizes."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_budget:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            if a[i] < 1e-15:
                continue
            best_local = a[i]
            best_local_score = best_score

            for factor in [0.75, 0.85, 0.92, 1.08, 1.15, 1.3]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = score_normalized(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = a[i] * factor

            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def main_adam(a0, budget, T_init=0.04, T_final=0.0008, lr=0.008,
              beta1=0.9, beta2=0.999, eps=1e-8,
              wf_interval=50, wf_step=0.05, wf_peaks=5,
              restart_every=3000, freq_restart_every=8000,
              max_no_improve=4000, verbose=False):
    """
    Main optimization: Adam in log-space with top-K soft-max,
    interleaved water-filling, and frequency-domain restarts.
    """
    n = len(a0)
    rng = np.random.default_rng(123)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)
    no_improve = 0
    current_lr = lr

    while time.monotonic() - t0 < budget:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / budget)

        # Temperature: hold then anneal
        if progress < 0.15:
            T = T_init
        else:
            p = (progress - 0.15) / 0.85
            T = T_init * (T_final / T_init) ** p

        # Adaptive K for top-K emphasis
        K = max(1, int(n * 0.05 * max(0.3, 1.0 - progress)))

        a_curr = softmax_forward(z)
        b = fast_convolve(a_curr)
        max_b = b.max()

        # Soft-max weights with top-K boosting
        b_shifted = b - max_b
        exp_b = np.exp(b_shifted / T)
        w = exp_b / exp_b.sum()

        # Top-K boost
        if K < len(b) and K > 1:
            top_K_idx = np.argsort(b)[::-1][:K]
            w[top_K_idx] *= (1.0 + 0.3 * K)
            w = w / w.sum()

        # Gradient
        corr = compute_corr(a_curr, w, fft_size)
        grad_a = 2.0 * n * corr
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - current_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic water-filling
        if t % wf_interval == 0:
            a_curr = softmax_forward(z)
            a_wf = waterfill_step(a_curr, wf_step, wf_peaks)
            if score_normalized(a_wf) < best_score:
                best_score = score_normalized(a_wf)
                best_a = a_wf.copy()
                z = np.log(np.maximum(a_wf, 1e-15))
                no_improve = 0
            else:
                # Blend: partially apply water-filling
                a_blend = 0.7 * a_curr + 0.3 * a_wf
                a_blend = np.maximum(a_blend, 1e-15)
                a_blend = a_blend / a_blend.sum()
                if score_normalized(a_blend) < best_score:
                    best_score = score_normalized(a_blend)
                    best_a = a_blend.copy()
                    z = np.log(a_blend)
                    no_improve = 0

        # Periodic frequency-domain restart
        if t % freq_restart_every == 0:
            a_curr = softmax_forward(z)
            a_freq = freq_restart(a_curr, rng, n_modes=3, amplitude=0.03)
            sc_freq = score_normalized(a_freq)
            if sc_freq < best_score:
                best_score = sc_freq
                best_a = a_freq.copy()
                z = np.log(np.maximum(a_freq, 1e-15))
                no_improve = 0

        # Evaluate current
        a_new = softmax_forward(z)
        score = score_normalized(a_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart when stuck
        if no_improve > max_no_improve:
            perturbation = 0.01 + 0.02 * (1.0 - progress)
            z = np.log(np.maximum(best_a, 1e-15)) + np.random.standard_normal(n) * perturbation
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            current_lr = max(0.001, current_lr * 0.75)

        # Slow LR decay
        current_lr = max(0.001, current_lr * (1.0 - 0.00005))

        if verbose and t % 5000 == 0:
            print(f"      Adam t={t} T={T:.5f} lr={current_lr:.5f} best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        t_sub = min(t_per_start, rem * 0.9)
        if t_sub < 15:
            continue
        sc_init, label, init_a = inits_scored[idx]
        print(f"  Start {idx+1}: {label} (init={sc_init:.6f})")
        a_res, sc_res = main_adam(
            init_a, t_sub, T_init=0.05, T_final=0.001,
            lr=0.01, wf_interval=30, wf_step=0.08, wf_peaks=5,
            max_no_improve=2000
        )
        best_seq, best_score = evaluate_and_track(a_res, f"init_{label}", best_seq, best_score)

    # === Phase 2: Main optimization (55% time) ===
    print(f"\n=== Phase 2: Main Optimization (best={best_score:.6f}) ===")
    t_p2 = min((deadline - time.monotonic()) * 0.55, 900.0)

    # Run multiple Adam configurations
    configs = [
        {'T_init': 0.04, 'T_final': 0.0008, 'lr': 0.008, 'wf_interval': 50, 'wf_step': 0.05, 'wf_peaks': 5},
        {'T_init': 0.06, 'T_final': 0.001, 'lr': 0.012, 'wf_interval': 80, 'wf_step': 0.03, 'wf_peaks': 3},
        {'T_init': 0.03, 'T_final': 0.0005, 'lr': 0.006, 'wf_interval': 30, 'wf_step': 0.07, 'wf_peaks': 8},
    ]

    n_configs = len(configs)
    t_per_config = t_p2 / n_configs

    for ci, cfg in enumerate(configs):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_sub = min(t_per_config, rem * 0.95)
        if t_sub < 20:
            continue
        print(f"  Config {ci+1}/{n_configs}: T=[{cfg['T_init']}->{cfg['T_final']}], lr={cfg['lr']}, wf={cfg['wf_interval']}")
        a_res, sc_res = main_adam(
            best_seq, t_sub,
            T_init=cfg['T_init'], T_final=cfg['T_final'],
            lr=cfg['lr'], wf_interval=cfg['wf_interval'],
            wf_step=cfg['wf_step'], wf_peaks=cfg['wf_peaks'],
            restart_every=4000, freq_restart_every=10000,
            max_no_improve=5000
        )
        best_seq, best_score = evaluate_and_track(a_res, f"cfg{ci+1}", best_seq, best_score)
        print(f"    After config {ci+1}: best={best_score:.6f}")

    # === Phase 3: Local polishing (25% time) ===
    print(f"\n=== Phase 3: Local Polishing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100

    # Aggressive peak shaving
    if rem > 30:
        t_sub = min(rem * 0.35, 120.0)
        a_ps, _ = peak_shave_aggressive(best_seq, t_sub, n_trials=8000)
        best_seq, best_score = evaluate_and_track(a_ps, "peak_shave", best_seq, best_score)
        print(f"  After peak shave: {best_score:.6f}")

    # Coordinate descent
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_sub = min(rem * 0.3, 80.0)
        a_cd, _ = coordinate_descent_fast(best_seq, t_sub, max_sweeps=50)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_desc", best_seq, best_score)
        print(f"  After coord descent: {best_score:.6f}")

    # Another round of peak shaving
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t_sub = min(rem * 0.4, 60.0)
        a_ps2, _ = peak_shave_aggressive(best_seq, t_sub, n_trials=5000)
        best_seq, best_score = evaluate_and_track(a_ps2, "peak_shave_2", best_seq, best_score)
        print(f"  After peak shave 2: {best_score:.6f}")

    # === Phase 4: Fine-tuning with very low temperature (7% time) ===
    print(f"\n=== Phase 4: Fine-Tuning (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        t_sub = min(rem * 0.6, 80.0)
        a_fin, _ = main_adam(
            best_seq, t_sub, T_init=0.01, T_final=0.0003,
            lr=0.003, wf_interval=20, wf_step=0.02, wf_peaks=3,
            max_no_improve=3000
        )
        best_seq, best_score = evaluate_and_track(a_fin, "fine_tune", best_seq, best_score)
        print(f"  After fine-tune: {best_score:.6f}")

    # Final peak shave
    rem = deadline - time.monotonic() - 15
    if rem > 10:
        t_sub = min(rem * 0.7, 30.0)
        a_fin_ps, _ = peak_shave_aggressive(best_seq, t_sub, n_trials=3000)
        best_seq, best_score = evaluate_and_track(a_fin_ps, "final_ps", best_seq, best_score)

    # === Final verification ===
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.6f}")

    return [float(x) for x in best_seq]
