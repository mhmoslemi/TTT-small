2. L-BFGS fine polishing
3. Simultaneous multi-peak shaving with backtracking line search
4. Multi-length exploration

Key insight: The problem is non-convex (the Hessian of each quadratic form (a*a)[k]
is indefinite), so we use smooth approximations and gradient-based methods with

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smoothing_weights(conv_vals, T):
    """Compute softmax weights over convolution values."""
    shifted = T * (conv_vals - conv_vals.max())
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    return w

def compute_gradient_softmax(a, w, n):
    """Compute gradient of soft_max(conv(a,a), T) wrt a.
    grad_a[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    
    This is 2 * cross_correlation(w, a)[i]
    """
    # Cross-correlation: corr[i] = sum_j w[j+i] * a[j]
    # Using FFT
    N = 1
    while N < (2*n-1) + n - 1:
        N *= 2
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)[:n]
    return 2.0 * corr

def softmax_normalize(z):
    """Convert log-space to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

        # Compute convolution
        conv = fast_convolve(a)
        
        # Compute weights
        w = compute_smoothing_weights(conv, T)

def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smoothing_weights(conv_vals, T):
    """Compute softmax weights over convolution values."""
    shifted = T * (conv_vals - conv_vals.max())
    exp_shifted = np.exp(shifted)
    s = exp_shifted.sum()
    if s < 1e-300:
        w = np.zeros(len(conv_vals))
        w[np.argmax(conv_vals)] = 1.0
        return w
    return exp_shifted / s


def compute_gradient_a(a, w, n):
    """Compute gradient of soft_max(conv(a,a), T) wrt a.
    grad_a[i] = 4n * sum_j w[j+i] * a[j]  (cross-correlation)
    """
    N = 1
    while N < 3 * n - 2:
        N *= 2
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)[:n]
    return 4.0 * n * corr


def compute_variance_gradient(a, conv, n):
    """Gradient of var(conv(a,a)) wrt a.
    var = mean(conv^2) - mean(conv)^2
    d var / d a_i = 2*(mean(conv)-conv_term) * d(conv_term)/d a_i  ... simplified:
    d var / d a_i = (2/len(conv)) * sum_k conv[k] * dconv_k/da_i - (1/len(conv)) * sum_k dconv_k/da_i * sum_m conv[m] / len... 
    Actually let's use: var = (1/L) sum_k conv[k]^2 - ((1/L) sum_k conv[k])^2
    d var / d a_i = (2/L) sum_k conv[k] * dconv_k/da_i - 2*(mean_conv/L) * sum_k dconv_k/da_i
    where dconv_k/da_i = 2*a[k-i] (same as before)
    So d var / d a_i = (4/L) * sum_k conv[k]*a[k-i] - (4*mean_conv/L) * sum_k a[k-i]
    The second term: sum_k a[k-i] for valid k... this is a convolution of a with ones.
    Let's just compute it directly.
    """
    L = len(conv)
    # dconv_k/da_i = 2*a[k-i] for valid k-i
    # First term: (4/L) * sum_k conv[k] * 2*a[k-i] / 2 ... 
    # Actually: d var / d a_i = (2/L) * sum_k conv[k] * (2*a[k-i]) - 2*(1/L * sum_m conv[m]) * (1/L) * sum_k (2*a[k-i])
    # = (4/L) * sum_k conv[k]*a[k-i] - (4*mean_conv/L^2) * sum_k a[k-i]
    # The second sum: sum_k a[k-i] for k where 0<=k-i<n and 0<=k<2n-1
    # This equals sum of a over a window of length min(n, k+1, 2n-1-k) ... it's the convolution of a with ones of appropriate length
    # Simpler: just compute the full gradient vector
    # sum_k conv[k] * a[k-i] is cross-correlation of conv and a (same structure as before)
    N = 1
    while N < 3 * n - 2:
        N *= 2
    C = rfft(conv, N)
    A = rfft(a, N)
    first_term = 4.0 / L * irfft(C * np.conj(A), N)[:n]
    
    # sum_k a[k-i] = (ones * a) cross-correlation... actually it's sum over valid j of a[j] where k=i+j
    # = sum_{j=0}^{n-1} a[j] * 1[0 <= i+j < 2n-1]
    # For i in [0, n-1]: this is sum of a[j] for j=0 to n-1 (since i+j can be at most 2n-2 < 2n-1)
    # Wait: i+j < 2n-1 means j < 2n-1-i. Since j < n always, and 2n-1-i >= n for i <= n-1.
    # So sum_k a[k-i] = sum_j a[j] = 1 (since a is normalized to sum 1)
    # Wait that's not right. k ranges over [0, 2n-2]. k-i = j means k = i+j.
    # Valid k: 0 <= i+j <= 2n-2, so j ranges from max(0, -i) to min(n-1, 2n-2-i).
    # Since i >= 0 and i <= n-1: j ranges from 0 to min(n-1, 2n-2-i).
    # 2n-2-i >= n-1 when i <= n-1. So j ranges from 0 to n-1.
    # Therefore sum_k a[k-i] = sum_j a[j] = 1 for all i!
    
    mean_conv = conv.sum() / L
    second_term = (4.0 * mean_conv / (L * L)) * np.ones(n)
    
    return first_term - second_term


def softmax_normalize(z):
    """Convert log-space to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def adam_optimize(a0, time_limit, T_start=100.0, T_end=3.0, lr=0.05,
                  beta1=0.9, beta2=0.999, var_penalty=0.0, verbose=True,
                  lr_decay=0.95, lr_decay_interval=100):
    """Adam optimization in log-space with annealed temperature and optional variance penalty."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z = np.log(a_safe / a_safe.sum())
    z = z - z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    t0 = monotonic()

    while monotonic() - t0 < time_limit:
        t += 1
        elapsed = monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Anneal temperature: fast decay
        T = T_end + (T_start - T_end) * np.exp(-6.0 * progress)

        # Current a
        a = softmax_normalize(z)

        # Compute convolution
        conv = fast_convolve(a)

        # Compute weights
        w = compute_smoothing_weights(conv, T)

        # Gradient in a-space
        grad_a = compute_gradient_a(a, w, n)

        # Add variance penalty if requested
        if var_penalty > 0:
            grad_a = grad_a + var_penalty * compute_variance_gradient(a, conv, n)

        # Transform to z-space
        S = np.dot(a, grad_a)
        grad_z = a * (grad_a - S)

        # Clip gradient for stability
        gz_norm = np.linalg.norm(grad_z)
        if gz_norm > 1.0:
            grad_z = grad_z / gz_norm

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = z - z.mean()

        # Evaluate
        a_new = softmax_normalize(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Learning rate decay
        if t % lr_decay_interval == 0:
            lr *= lr_decay

        if verbose and t % 300 == 0:
            print(f"    Adam t={t} ({elapsed:.0f}s): T={T:.1f} score={best_score:.8f} lr={lr:.5f}")

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=5.0, maxiter=5000, var_penalty=0.0):
    """L-BFGS polish in log-space with optional variance penalty."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe / a_safe.sum())
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = softmax_normalize(z)
        conv = fast_convolve(a)
        shifted = T * (conv - conv.max())
        lse = conv.max() + (1.0 / T) * np.log(np.exp(shifted).sum())
        val = 2.0 * n * lse
        if var_penalty > 0:
            val += var_penalty * np.var(conv)
        return val

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        a = softmax_normalize(z)
        conv = fast_convolve(a)
        w = compute_smoothing_weights(conv, T)
        grad_a = compute_gradient_a(a, w, n)
        if var_penalty > 0:
            grad_a = grad_a + var_penalty * compute_variance_gradient(a, conv, n)
        S = np.dot(a, grad_a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            a_out = softmax_normalize(res.x)
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out


def peak_shave_multi(a0, time_limit, top_k=5, use_flat_weights=True):
    """Multi-peak shaving with backtracking line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0
    step = 0

    while monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)

        # Get top-k peaks (excluding the very first and last which are trivially small)
        k_start = max(0, n // 4)
        k_end = len(conv) - n // 4
        conv_focus = conv.copy()
        conv_focus[:k_start] = -np.inf
        conv_focus[k_end:] = -np.inf
        peak_indices = np.argpartition(conv_focus, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices][::-1])]

        peak_vals = conv[peak_indices]

        if use_flat_weights:
            # Weight peaks by how much they exceed a threshold (the median of all peaks)
            threshold = np.median(conv)
            excess = np.maximum(0, peak_vals - threshold)
            if excess.sum() < 1e-15:
                w = np.ones(top_k) / top_k
            else:
                w = excess / excess.sum()
        else:
            temp = 0.005 * (peak_vals.max() - peak_vals.min() + 1e-10)
            w = np.exp((peak_vals - peak_vals.max()) / temp)
            w = w / w.sum()

        # Subgradient direction (sum of weighted peak gradients)
        grad = np.zeros(n)
        for idx in range(top_k):
            k = int(peak_indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idxs = np.arange(js, je + 1)
            grad[idxs] += 2.0 * w[idx] * a[k - idxs]

        # Backtracking line search
        best_alpha = 0.0
        a_trial_best = None
        alpha = 0.005

        for _ in range(25):
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                break
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                best_alpha = alpha
                a_trial_best = a_trial.copy()
                best_score = score
            alpha *= 0.6
            if alpha < 1e-6:
                break

        if a_trial_best is not None:
            a = a_trial_best
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try small random perturbation
            if no_improve % 20 == 0:
                for _ in range(5):
                    a_trial = a + np.random.standard_normal(n) * 0.001
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                        break

            if no_improve > 200:
                a = best_a + np.random.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if step % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: step={step} score={best_score:.8f}")

    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Adam with annealed temperature,
    L-BFGS polishing, multi-peak shaving, and length exploration.
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Primary Adam optimization (long run, annealed T) ===
    print(f"\n=== Phase 1: Adam Primary (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 500
    if rem > 80:
        t_alloc = min(rem * 0.65, 900.0)
        print(f"  Adam n={n_parent}, budget={t_alloc:.0f}s, T: 200->5")
        a1, _ = adam_optimize(best_seq, t_alloc, T_start=200.0, T_end=5.0,
                              lr=0.08, lr_decay=0.93, lr_decay_interval=50, verbose=True)
        best_seq, best_score = check_and_update(a1, "adam_primary", best_seq, best_score)

    # === Phase 2: Second Adam with different parameters ===
    print(f"\n=== Phase 2: Adam Secondary (best={best_score:.8f}) ===")
    for (T_s, T_e, lr, vp, label) in [
        (80, 3, 0.04, 0.0, "adam_lowT"),
        (150, 8, 0.10, 0.01, "adam_varpen"),
    ]:
        rem = deadline - monotonic() - 300
        if rem < 40:
            break
        t_alloc = min(rem * 0.45, 250.0)
        print(f"  Adam T={T_s}->{T_e} lr={lr} vp={vp}, budget={t_alloc:.0f}s")
        a2, _ = adam_optimize(best_seq, t_alloc, T_start=T_s, T_end=T_e,
                              lr=lr, var_penalty=vp, verbose=False)
        best_seq, best_score = check_and_update(a2, label, best_seq, best_score)

    # === Phase 3: L-BFGS fine polish ===
    print(f"\n=== Phase 3: L-BFGS Polish (best={best_score:.8f}) ===")
    for T in [8, 4, 2]:
        rem = deadline - monotonic() - 200
        if rem < 25:
            break
        t_alloc = min(rem * 0.3, 80.0)
        print(f"  L-BFGS T={T}, budget={t_alloc:.0f}s")
        a3 = lbgfs_polish(best_seq, t_alloc, T=T, maxiter=5000)
        best_seq, best_score = check_and_update(a3, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 4: Multi-peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 150
    if rem > 25:
        t_alloc = min(rem * 0.5, 150.0)
        a4, _ = peak_shave_multi(best_seq, t_alloc, top_k=5, use_flat_weights=True)
        best_seq, best_score = check_and_update(a4, "peak_shave", best_seq, best_score)

    # === Phase 5: Adam round 2 (refined) ===
    print(f"\n=== Phase 5: Adam Round 2 (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 120
    if rem > 40:
        t_alloc = min(rem * 0.5, 180.0)
        print(f"  Adam R2 T=30->2, budget={t_alloc:.0f}s")
        a5, _ = adam_optimize(best_seq, t_alloc, T_start=30.0, T_end=2.0,
                              lr=0.03, lr_decay=0.97, verbose=False)
        best_seq, best_score = check_and_update(a5, "adam_r2", best_seq, best_score)

        # Quick L-BFGS after
        rem = deadline - monotonic() - 60
        if rem > 20:
            a5b = lbgfs_polish(best_seq, min(rem * 0.5, 40.0), T=2, maxiter=3000)
            best_seq, best_score = check_and_update(a5b, "lbgfs_r2", best_seq, best_score)

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for factor in [1.25, 0.8, 1.5, 0.7]:
        alt_n = int(n_best * factor)
        if 384 <= alt_n <= 3072 and alt_n != n_best and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:2]:
        rem = deadline - monotonic() - 80
        if rem < 40:
            break
        t_alloc = min(rem * 0.4, 120.0)
        init_alt = interpolate_to_length(best_seq, alt_n)
        print(f"  Explore n={alt_n}, budget={t_alloc:.0f}s")
        a6, _ = adam_optimize(init_alt, t_alloc, T_start=100.0, T_end=5.0,
                              lr=0.06, verbose=False)
        best_seq, best_score = check_and_update(a6, f"explore_n{alt_n}", best_seq, best_score)

    # === Phase 7: Final peak shave + L-BFGS ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 15
    if rem > 15:
        t7a = min(rem * 0.4, 30.0)
        a7, _ = peak_shave_multi(best_seq, t7a, top_k=3, use_flat_weights=True)
        best_seq, best_score = check_and_update(a7, "final_shave", best_seq, best_score)

    rem = deadline - monotonic() - 5
    if rem > 10:
        a7b = lbgfs_polish(best_seq, min(rem * 0.6, 20.0), T=2, maxiter=2000)
        best_seq, best_score = check_and_update(a7b, "final_lbgfs", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
