"""

descent in log-space, with multiple initialization strategies (ramp, exponential,

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute score assuming a is non-negative with sum > 0."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * n * b.max()


def softmax_fn(z):
    """Convert log-domain z to probability vector."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_obj_and_grad(z, T, n, n_top=80):
    """Compute smooth objective and its gradient w.r.t. z."""
    a = softmax_fn(z)
    b = fast_convolve(a)
    x = b / T
    m = x.max()
    p = np.exp(x - m)
    p = p / p.sum()
    
    # Objective: 2n * T * logsumexp(b/T)
    obj = 2.0 * n * T * (m + np.log(np.exp(x - m).sum()))
    
    # Gradient computation using top peaks
    top_idx = np.argsort(p)[::-1][:n_top]
    
    grad_a = np.zeros(n)
    for k in top_idx:
        k = int(k)
        pk = p[k]
        if pk < 1e-15:
            continue
        # db_k/da_j = a[k-j] + a[j-k] (with a_i=0 outside [0,n-1])
        # First term: a[k-j] valid for j in [max(0,k-n+1), min(k, n-1)]
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:
            js = np.arange(lo1, hi1 + 1)
            grad_a[js] += pk * a[k - js]
        # Second term: a[j-k] valid for j in [max(0,k), min(k+n-1, n-1)]
        lo2 = max(0, k)
        hi2 = min(k + n - 1, n - 1)
        if lo2 <= hi2:
            js = np.arange(lo2, hi2 + 1)
            grad_a[js] += pk * a[js - k]
    
    # Project to log-space
    S = np.dot(grad_a, a)
    grad_z = 2.0 * n * a * (grad_a - S)
    
    return obj, grad_z


def lbgfs_smooth(z0, T, n, time_limit, maxiter=5000, n_top=80):
    """L-BFGS optimization of smooth objective in log-space."""
    t0 = time.monotonic()
    best_z = z0.copy()
    best_obj = compute_obj_and_grad(z0, T, n, n_top)[0]
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return compute_obj_and_grad(z, T, n, n_top)[0]
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return compute_obj_and_grad(z, T, n, n_top)[1]
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_fn(res.x)
        sc = score_normalized(a_out)
        if sc < best_obj / (2.0 * n * T):
            best_z = res.x.copy()
            best_obj = sc * 2.0 * n * T
    except Exception:
        pass
    
    return best_z, softmax_fn(best_z)

        _, a_ng = nesterov_smooth(z0, T=300, n=n, time_limit=t_phase3, lr_init=0.01, momentum=0.91)
        sc_ng = score_normalized(a_ng)
        print(f"  Nesterov: score={sc_ng:.8f}")
        if sc_ng < best_score - 1e-12:
            best_score = sc_ng
            best_a = a_ng.copy()
    
    # Phase 4: Peak-targeted coordinate descent
    print(f"\nPhase 4: Peak-targeted coordinate descent (best={best_score:.8f})")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_phase4 = min(rem * 0.2, 250.0)
        a_cd, sc_cd = coordinate_descent_peak(best_a, t_phase4, n_pairs=500)
        print(f"  CD: score={sc_cd:.8f}")
        if sc_cd < best_score - 1e-12:
            best_score = sc_cd
            best_a = a_cd.copy()
    
    # Phase 5: Simulated annealing
    print(f"\nPhase 5: Simulated annealing (best={best_score:.8f})")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_phase5 = min(rem * 0.2, 200.0)
        a_sa, sc_sa = simulated_annealing_peak(best_a, t_phase5, n_moves=300)
        print(f"  SA: score={sc_sa:.8f}")
        if sc_sa < best_score - 1e-12:
            best_score = sc_sa
            best_a = a_sa.copy()
    
    # Phase 6: Final L-BFGS polish
    print(f"\nPhase 6: Final L-BFGS polish (best={best_score:.8f})")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_phase6 = min(rem * 0.3, 100.0)
        z0 = np.log(np.maximum(best_a, 1e-30))
        z, a_final = lbgfs_smooth(z0, 100, n, t_phase6, maxiter=5000, n_top=100)
        sc_final = score_normalized(a_final)
        print(f"  Final L-BFGS: score={sc_final:.8f}")
        if sc_final < best_score - 1e-12:
            best_score = sc_final
            best_a = a_final.copy()
    
    # Phase 7: Final coordinate descent
    rem = deadline - time.monotonic()
    if rem > 10:
        t_phase7 = min(rem * 0.5, 60.0)
        a_final2, sc_final2 = coordinate_descent_peak(best_a, t_phase7, n_pairs=800)
        print(f"  Final CD: score={sc_final2:.8f}")
        if sc_final2 < best_score - 1e-12:
            best_score = sc_final2
            best_a = a_final2.copy()
    
    # Final verification with evaluate_sequence
    best_a = np.maximum(best_a, 0)
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score):
        print(f"\nFINAL SCORE (evaluated): {final_score:.8f} (n={len(best_a)})")
    else:
        print(f"\nWARNING: evaluate_sequence returned inf, using fallback")
        best_a = np.ones(n) / n
        final_score = float(evaluate_sequence(best_a.tolist()))
        print(f"  Fallback uniform: {final_score:.8f}")
    
    return [float(x) for x in best_a]
