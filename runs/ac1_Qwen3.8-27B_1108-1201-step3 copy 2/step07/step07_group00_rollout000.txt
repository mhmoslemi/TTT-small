import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, beta, n_top=200):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*b_k))) where b = conv(a,a).
    Uses top-n_top peaks for efficiency.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    b = fast_convolve(a)
    # Use top peaks for the softmax (rest are negligible)
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        b_top = b[top_idx]
    else:
        top_idx = np.arange(len(b))
        b_top = b

    # Softmax with temperature beta
    b_shifted = b_top - b_top.max()
    w_top = np.exp(beta * b_shifted)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(len(top_idx)) / len(top_idx)
    else:
        w_top /= w_sum

    # Gradient: grad_i = 2 * sum_k w_k * a_{k-i}
    grad = np.zeros(n)
    for idx_i in range(len(top_idx)):
        k = top_idx[idx_i]
        w = w_top[idx_i]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # a_{k-i} for i in [lo, hi] means a[k - i] where k-i ranges from k-hi to k-lo
        # So grad[lo:hi+1] += 2 * w * a[k-hi : k-lo+1][::-1]... 
        # Actually: grad_i = 2 * sum_k w_k * a_{k-i}
        # For fixed k: contributes 2*w_k*a_{k-i} to grad_i for i in [max(0,k-n+1), min(n-1,k)]
        # a_{k-i} where i ranges from lo to hi, so k-i ranges from k-hi to k-lo
        # grad[lo:hi+1] += 2*w * a[k-hi : k-lo+1][::-1]
        # Wait: when i=lo, k-i = k-lo; when i=hi, k-i = k-hi
        # So grad[lo:hi+1] += 2*w * a[k-hi:k-lo+1] reversed
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * w * seg

    return grad

def lp_solve(f, n, peaks_idx, M):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peaks_idx.
    Returns (g, dual_vars) or (None, None) on failure.
    """
    K = len(peaks_idx)
    if K == 0 or K > 2 * n:
        return None, None

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for row_i in range(K):
        k = peaks_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(row_i)
            cols.append(j)
            vals.append(f[j])

    if len(vals) == 0:
        return None, None

    A = cp.SparseMatrix((vals, (rows, cols)), shape=(K, n))

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-8)
    except Exception:
        return None, None

    if g.value is None:
        return None, None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, None

    # Try to get dual variables
    dual = None
    try:
        for con in prob.constraints:
            if con.dual_value is not None:
                dual = np.asarray(con.dual_value)
                break
    except Exception:
        pass

    return g_val, dual


def run_lp_phase(f_init, time_limit, top_k_init=40, top_k_max=300, label="lp"):
    """
    LP-based optimization with adaptive constraint set and dual-guided peak selection.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive top_k: grow gradually
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        b = fast_convolve(f)
        M = b.max()

        # Select peaks: top-K plus any peaks above 0.95*M
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:].copy()
        else:
            top_idx = np.arange(len(b)).copy()
        # Add peaks above threshold
        threshold = 0.95 * M
        above_thresh = np.where(b >= threshold)[0]
        all_idx = np.unique(np.concatenate([top_idx, above_thresh]))
        all_idx = np.sort(all_idx)

        g, dual = lp_solve(f, n, all_idx, M)

        if g is None:
            # Fallback: try with just top-K
            g, dual = lp_solve(f, n, top_idx, M)
            if g is None:
                # Perturb and retry
                if stuck > 5:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                    stuck = 0
                    continue
                break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                # Try with fewer constraints (relax)
                relaxed_idx = all_idx[:max(10, len(all_idx) // 2)]
                g, dual = lp_solve(f, n, relaxed_idx, M)
                if g is not None and g.sum() > 1.0 + 1e-5:
                    S_g = g.sum()
                    stuck = 0
                else:
                    # Perturb
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    stuck = 0
                    continue
            continue

        # Direction
        h = g / S_g

        # Line search - more aggressive
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()
        for t in [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 60:
                break

        if it % 40 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def peak_shave(f_init, time_limit):
    """
    Direct peak shaving: identify worst peak, reduce contributing elements.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(42)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        k_max = b.argmax()
        M = b[k_max]

        # Find top contributors to peak k_max
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        contributions = np.zeros(hi - lo + 1)
        for j in range(lo, hi + 1):
            contributions[j - lo] = f[j] * f[k_max - j]

        # Get top contributors
        top_contrib_idx = np.argsort(contributions)[::-1][:min(10, len(contributions))]

        # Reduce the largest contributor
        reduced = False
        for tc in top_contrib_idx[:5]:
            j = lo + tc
            partner = k_max - j
            if partner < 0 or partner >= n:
                continue
            # Reduce f[j] by a small amount
            reduction = f[j] * 0.02
            if reduction < 1e-10:
                continue
            f_trial = f.copy()
            f_trial[j] = max(0.0, f_trial[j] - reduction)
            # Redistribute to maintain sum
            s = f_trial.sum()
            if s > 1e-10:
                # Add to a random low element
                f_trial += reduction / n
                f_trial /= f_trial.sum()
            obj = fast_convolve(f_trial).max()
            if obj < M - 1e-12:
                f = f_trial
                reduced = True
                break

        if not reduced:
            # Try increasing a random low element (spreading mass)
            low_idx = np.argmin(f)
            f_trial = f.copy()
            f_trial[low_idx] += 0.001
            f_trial /= f_trial.sum()
            obj = fast_convolve(f_trial).max()
            if obj < M - 1e-12:
                f = f_trial
            else:
                # Perturb
                f = project_simplex(f + rng.standard_normal(n) * 0.001)

        if it % 100 == 0:
            cur_obj = fast_convolve(f).max()
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_f = f.copy()
            if it % 500 == 0:
                print(f"    peak_shave iter={it} obj={best_obj:.10f} "
                      f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.6, 0.8, 0.9, 1.1, 1.3, 1.7])

    improved = True
    rounds = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        rounds += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj


def resample_sequence(f, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
