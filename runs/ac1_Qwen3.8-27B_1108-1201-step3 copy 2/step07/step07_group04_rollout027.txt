"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth-max L-BFGS in log-space with adaptive annealing
2. Coordinate descent peak shaving
3. Multi-length exploration
4. Active-set Newton refinement

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def smooth_max_and_grad(a, T):
    """
    Compute the smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient_vector).
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    
    # Softmax weights
    c_max = c.max()
    scaled = (c - c_max) / T
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    smooth_val = c_max + T * np.log(exp_scaled.sum())
    
    # Gradient: grad[p] = 2 * sum_{m=0}^{n-1} a[m] * w[p+m]
    # This is cross-correlation of w and a, first n elements
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad = 2.0 * irfft(W * np.conj(A), fft_size)[:n]
    
    return smooth_val, grad, w


def log_space_objective(z, n, T):
    """
    Objective in log-space: 2*n*smooth_max(softmax(z))
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    val, _, _ = smooth_max_and_grad(a, T)
    return 2.0 * n * val


def log_space_gradient(z, n, T):
    """
    Gradient of 2*n*smooth_max(softmax(z)) w.r.t. z.
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    _, grad_a, _ = smooth_max_and_grad(a, T)
    
    # Chain rule: df/dz[i] = a[i] * (grad_a[i] - <a, grad_a>)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return 2.0 * n * grad_z


def lbgfs_optimize(a0, n, T, time_limit, maxiter=20000, verbose=True):
    """
    L-BFGS optimization in log-space with given temperature T.
    """
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    
    t0 = monotonic()
    stopped = [False]
    
    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        return log_space_objective(z, n, T)
    
    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        return log_space_gradient(z, n, T)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0] and res.x is not None:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
            return a_out, res.success
        else:
            return a0, False
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")
        return a0, False


def coordinate_descent(a0, time_limit, max_passes=10000, verbose=True):
    """
    Coordinate descent: for each coordinate, try small adjustments.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

        improved = False
        # Compute current convolution
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        
        for i in range(n):
            if monotonic() - t0 > time_limit:
                break
            
            # Try reducing a[i] by a small amount and redistributing
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # Try several reduction factors
            for factor in [0.99, 0.98, 0.95, 0.9]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-16:
                    continue
                
                # Redistribute diff proportionally to others
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to maintain sum
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = a[mask] + diff * a[mask] / others_sum
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        passes += 1
        if verbose and passes % 100 == 0:
            elapsed = monotonic() - t0
            print(f"    CD pass={passes} ({elapsed:.0f}s): score={best_score:.8f}")
        
        if not improved:
            # Try increasing a low coordinate
            min_i = int(np.argmin(a))
            if a[min_i] < best_a.mean() * 0.5:
                for factor in [1.1, 1.5, 2.0]:
                    new_val = a[min_i] * factor
                    diff = new_val - a[min_i]
                    a_trial = a.copy()
                    a_trial[min_i] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[min_i] = False
                    others = a[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    # Scale others down
                    scale = (others_sum - diff) / others_sum
                    if scale < 0:
                        continue
                    a_trial[mask] = others * scale
                    
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if not improved:
                # Try random perturbation
                for _ in range(3):
                    idx = np.random.randint(n, size=2)
                    a_trial = a.copy()
                    i, j = idx[0], idx[1]
                    if a[i] < 1e-15 or a[j] < 1e-15:
                        continue
                    transfer = a[j] * 0.01
                    a_trial[i] += transfer
                    a_trial[j] -= transfer
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                
                if not improved:
                    break
    
    return best_a, best_score


def peak_shave_aggressive(a0, time_limit, verbose=True):

        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.sum() < 1e-20:
            break

        order = np.argsort(contribs)[::-1]

                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + red * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_fast(a):
    """Fast local evaluation (avoids list conversion overhead)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)
