    A = rfft(a, fft_size)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with FFT-accelerated soft-max gradient and geometric temperature annealing
2. LP-based alternating optimization with Nesterov averaging
3. Aggressive pairwise peak-shaving local search
4. Multi-start with different sequence lengths and perturbations
Key improvements: O(n log n) gradient via FFT, Adam adaptive optimizer,
pairwise local search, and more aggressive low-temperature refinement.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_grad_fft(w, a, n):
    """
    Compute grad_a[i] = 2 * sum_j w[i+j] * a[j] using FFT.
    w has length 2n-1, a has length n.
    """
    w_rev = w[::-1]
    size = 1 << (3 * n).bit_length()
    conv_aw = irfft(rfft(a, size) * rfft(w_rev, size), size)
    grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    return grad_a


def evaluate_safe(a):

        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))

def adam_optimize(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=10,
                  lr_init=0.005, beta1=0.9, beta2=0.999, eps=1e-8, verbose=True):
    """
    Adam optimizer on log-space parameterization with soft-max smoothed objective.
    Uses FFT-accelerated gradient and geometric temperature schedule.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    for stage_idx, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining_stages = n_stages - stage_idx
        stage_budget = (time_limit - (time.monotonic() - t0)) / remaining_stages
        stage_end = time.monotonic() + stage_budget
        iters_in_stage = 0
        prev_obj = float('inf')
        stall_count = 0

        while time.monotonic() < stage_end:
            t += 1
            iters_in_stage += 1

            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            obj = (2.0 * n / T_val) * log_sum_exp

            # Check for stall
            if obj < prev_obj - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_obj = obj

            # Compute gradient via FFT
            w = exp_Tb / exp_Tb.sum()
            grad_a = compute_grad_fft(w, a, n)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr_init / (1.0 + 0.01 * stage_idx)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if iters_in_stage % 1000 == 0:
                a_check = softmax_from_z(z)
                sc = 2.0 * n * fast_convolve(a_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                    if verbose:
                        elapsed = time.monotonic() - t0
                        print(f"    Adam T={T_val:.2f} iter={iters_in_stage}: score={best_score:.8f} ({elapsed:.0f}s)")

        # End of stage: update best
        a_check = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_check).max()
        if sc < best_score:
            best_score = sc
            best_a = a_check.copy()

    a_out = softmax_from_z(z)
    sc = 2.0 * n * fast_convolve(a_out).max()
    if sc < best_score:
        best_score = sc
        best_a = a_out.copy()

    return best_a, best_score


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_alternating(a0, time_limit, solver=None, n_max_iters=2000):
    """
    LP-based alternating optimization with Nesterov averaging.
    Alternates between finding best response b for fixed a, and updating a.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    history = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Alt iter={it+1}: score={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def pairwise_peak_shave(a0, time_limit, n_steps=20000, verbose=True):
    """
    Aggressive pairwise peak shaving.
    For the current peak, identify top contributing pairs and try
    joint adjustments of pairs of coordinates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 500:
            # Try a random perturbation to escape local minimum
            mask = np.random.random(n) > 0.9
            a[mask] *= (0.9 + 0.2 * np.random.random(mask.sum()))
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            continue

        b = fast_convolve(a)
        # Exclude edges (they are single terms)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        sorted_contribs = np.argsort(contribs)[::-1]
        top_m = sorted_contribs[:8]

        improved = False

        # Strategy 1: Reduce single top contributor, redistribute
        for ci in top_m[:4]:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.9]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Reduce pair of contributors
        if not improved and len(top_m) >= 2:
            for ci in range(min(5, len(top_m))):
                for cj in range(ci + 1, min(5, len(top_m))):
                    i = idx[top_m[ci]]
                    j = idx[top_m[cj]]
                    if i == j or a[i] < 1e-14 or a[j] < 1e-14:
                        continue
                    for f1 in [0.97, 0.93]:
                        for f2 in [0.97, 0.93]:
                            a_trial = a.copy()
                            old_i = a_trial[i]
                            old_j = a_trial[j]
                            a_trial[i] = old_i * f1
                            a_trial[j] = old_j * f2
                            diff_i = old_i - a_trial[i]
                            diff_j = old_j - a_trial[j]
                            total_diff = diff_i + diff_j
                            if total_diff < 1e-16:
                                continue
                            mask = np.ones(n, dtype=bool)
                            mask[i] = False
                            mask[j] = False
                            others = a_trial[mask]
                            others_sum = others.sum()
                            if others_sum < 1e-15:
                                continue
                            a_trial[mask] = others + total_diff * others / others_sum
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Reduce all top contributors slightly
        if not improved:
            a_trial = a.copy()
            for ci in top_m[:4]:
                i = idx[ci]
                a_trial[i] *= 0.97
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step}: score={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def multi_peak_shave(a0, time_limit, n_steps=10000, verbose=True):
    """
    Multi-peak shaving: simultaneously reduce multiple peaks.
    Identifies top K peaks and tries to reduce them together.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 300:
            break

        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        # Find top 3 peaks
        peak_indices = np.argsort(b_search)[::-1][:3]

        improved = False
        # Try reducing contributions to all top peaks simultaneously
        all_top_coords = set()
        for pk in peak_indices:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_c = np.argsort(contribs)[::-1][:3]
            for tc in top_c:
                all_top_coords.add(idx[tc])

        all_top_coords = list(all_top_coords)
        if not all_top_coords:
            no_improve += 1
            continue

        for factor in [0.98, 0.96, 0.93]:
            a_trial = a.copy()
            total_diff = 0.0
            for i in all_top_coords:
                total_diff += a_trial[i] * (1.0 - factor)
                a_trial[i] *= factor
            a_trial = np.maximum(a_trial, 0)
            mask = np.ones(n, dtype=bool)
            for i in all_top_coords:
                mask[i] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum > 1e-15:
                a_trial[mask] = others + total_diff * others / others_sum
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and step % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiPeak step={step}: score={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n + 1)
    f_full = np.zeros(old_n + 1)
    f_full[1:] = f_norm
    new_x = np.linspace(0, 1, new_n + 1)
    f_interp = np.interp(new_x, old_x, f_full)
    f_interp = np.maximum(f_interp[1:], 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, sigma=0.1, rng=None):
    """Add multiplicative noise to a sequence."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    noise = 1.0 + sigma * rng.standard_normal(n)
    a_pert = a * np.maximum(noise, 0.1)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

    """

    Combines Adam with FFT gradient, LP alternating, and aggressive local search.
    """
    budget = 1800.0
    t_start = time.monotonic()
    deadline = t_start + budget - 30.0  # leave 30s buffer

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)
    rng = np.random.default_rng(12345)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, {elapsed:.0f}s) ***")

    # === Phase 1: Adam optimization (400s) ===
    print(f"\n=== Phase 1: Adam T=100->1 (best={best_score:.8f}) ===")

        print(f"  LP-Alt n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t_alloc, solver=solver)

    # === Phase 3: Adam at low temperature (350s) ===
