def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent (log-sum-exp) with FISTA + Adam hybrid
2. Geometric temperature annealing for escaping local minima
3. Diverse parametric initializations (exponential, cosine, Fourier, power-law)
4. Multi-length exploration around the best length
5. Coordinate descent + peak-shaving polishing
6. LP-based fictitious play as complement

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_smooth_grad(b, T):
    """Compute smooth max objective value and gradient.
    
    f_T(b) = T * log(sum_k exp(conv(b,b)_k / T))
    grad_i = 2 * sum_j b[j] * weights[i+j] where weights = softmax(conv/T)
    """
    n = len(b)
    conv_b = fast_convolve(b)
    shift = conv_b.max()
    exp_vals = np.exp((conv_b - shift) / T)
    total = exp_vals.sum()
    obj = shift + T * np.log(total)
    weights = exp_vals / total

    # Gradient: grad_i = 2 * sum_j b[j] * weights[i+j]
    # Computed via: grad[i] = 2 * conv(b, weights_rev)[2n-2-i]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    U = rfft(b, fft_size)
    V_rev = rfft(weights[::-1], fft_size)
    conv_result = irfft(U * V_rev, fft_size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return obj, grad


def fista_smooth(b0, T, max_iters=2000, tol=1e-12, time_limit=None, t_start=None):
    """FISTA optimization on smooth objective at fixed temperature."""
    n = len(b0)
    x = b0.copy()
    y = b0.copy()
    t_k = 1.0
    prev_best = fast_convolve(y).max()
    best_x = y.copy()
    t0 = t_start if t_start is not None else time.monotonic()

    for it in range(max_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        obj_y, grad_y = compute_smooth_grad(y, T)

        # Adaptive step size based on gradient norm
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-15:
            break
        alpha = 0.5 / gnorm

        x_new = y - alpha * grad_y
        x_new = simplex_project(x_new)

        # Compute actual objective at new point
        actual_max = fast_convolve(x_new).max()
        if actual_max < prev_best - 1e-14:
            prev_best = actual_max
            best_x = x_new.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y_new = x_new + beta * (x_new - x)
        y_new = simplex_project(y_new)

        if abs(y_new - y).max() < tol * max(1.0, y.max()):
            break

        x, y, t_k = x_new, y_new, t_new

    # Check y as well
    actual_max_y = fast_convolve(y).max()
    if actual_max_y < prev_best:
        prev_best = actual_max_y
        best_x = y.copy()

    return best_x, prev_best


def adam_smooth_anneal(b0, time_budget, T_ratio=0.75, min_T_frac=0.005,
                       max_iters_per_T=500, verbose=False):
    """Adam optimization on smooth objective with geometric temperature annealing."""
    n = len(b0)
    b = b0.copy()
    conv_b = fast_convolve(b)
    T0 = conv_b.max() * 0.5
    T_min = conv_b.max() * min_T_frac

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    adam_t = 0
    lr_base = 0.02

    best_b = b.copy()
    best_max_b = conv_b.max()
    t0 = time.monotonic()
    T = T0
    total_iters = 0
    stagnation = 0

    while time.monotonic() - t0 < time_budget and T > T_min:
        # Run Adam at current T
        for it in range(max_iters_per_T):
            if time.monotonic() - t0 > time_budget:
                break

            conv_b = fast_convolve(b)
            shift = conv_b.max()
            exp_vals = np.exp((conv_b - shift) / T)
            total = exp_vals.sum()
            weights = exp_vals / total

            # Gradient
            fft_size = 1 << (3 * n - 2 - 1).bit_length()
            U = rfft(b, fft_size)
            V_rev = rfft(weights[::-1], fft_size)
            conv_result = irfft(U * V_rev, fft_size)
            grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]

            # Adam update
            adam_t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** adam_t)
            v_hat = v / (1.0 - beta2 ** adam_t)
            lr = lr_base / (np.sqrt(v_hat) + eps)

            b_new = b - lr * m_hat
            b_new = simplex_project(b_new)

            # Check actual improvement
            conv_new = fast_convolve(b_new)
            max_new = conv_new.max()
            if max_new < best_max_b - 1e-14:
                best_max_b = max_new
                best_b = b_new.copy()
                b = b_new
                stagnation = 0
            elif max_new < conv_b.max() + 0.01 * T:
                # Accept if not too bad (annealing)
                b = b_new
                stagnation += 1
            else:
                stagnation += 1

            total_iters += 1

        # Check convergence at this T
        if stagnation > max_iters_per_T * 0.8:
            # Reset momentum
            m = np.zeros(n)
            v = np.zeros(n)
            stagnation = 0

        # Decrease temperature
        T *= T_ratio

        if verbose and total_iters % 2000 < max_iters_per_T:
            elapsed = time.monotonic() - t0
            print(f"      T={T:.6f} ({elapsed:.0f}s): best_max={best_max_b:.8f}")

    # Final: return best found
    return best_b, best_max_b


def coord_descent_polish(a0, time_limit, n_sweeps=500):
    """Full coordinate descent with adaptive factors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

                if new_val < 0:

                a[i] = new_val
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True

                else:

            else:
                a = best_a.copy()
        else:

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = _find_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def _find_solver():

    global _solver_cache

    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_exponential(n, rng, alpha_range=(0.5, 3.0)):

def init_cosine_window(n, rng, beta_range=(0.5, 2.0)):

        inits.append(init_exponential(n, rng))
        # Cosine window
        inits.append(init_cosine_window(n, rng))
        # Power law
        inits.append(init_power_law(n, rng))
        # Fourier
        inits.append(init_fourier(n, rng))
        # Skewed
        inits.append(init_skewed(n, rng))

        for init in inits:
            # Quick FISTA + Adam
            a1, s1 = fista_smooth(init, T=fast_convolve(init).max() * 0.3,
                                  max_iters=50, time_limit=3.0)
            a2, s2 = adam_smooth_anneal(a1, time_budget=5.0, T_ratio=0.7,
                                        max_iters_per_T=100)
            score = 2.0 * n * fast_convolve(a2).max()
            all_cands.append((score, a2, n))

    all_cands.sort(key=lambda x: x[0])
    print(f"  {len(all_cands)} candidates generated, best fast-score: {all_cands[0][0]:.6f}")

    # Update best from top candidates
    for sc, ac, nc in all_cands[:5]:
        best_seq, best_score = check_and_update(ac, best_seq, best_score,
                                                f"p1_n{nc}_{sc:.4f}")

    # === Phase 2: Main smooth gradient annealing (800s) ===
    print(f"\n=== Phase 2: Smooth Gradient Annealing (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    t_phase2 = 800.0

    n_best = len(best_seq)
    # Use the best candidate as starting point
    start = best_seq.copy()

    # Run Adam annealing with multiple temperature schedules
    schedules = [
        (0.8, 0.003, 800),   # (T_ratio, min_T_frac, iters_per_T)
        (0.75, 0.005, 600),
        (0.85, 0.01, 1000),
    ]

    for sched_idx, (t_ratio, t_min_frac, iters_pt) in enumerate(schedules):
        rem = deadline - time.monotonic() - 500
        if rem < 60:
            break
        t_alloc = min(rem * 0.5, t_phase2 / len(schedules))
        if t_alloc < 30:
            continue

        print(f"  Schedule {sched_idx}: ratio={t_ratio}, budget={t_alloc:.0f}s")

        # Start from current best with slight noise for diversity
        if sched_idx > 0:
            noise = rng.standard_normal(n_best) * 0.05
            start_noisy = simplex_project(best_seq + noise)
        else:
            start_noisy = start.copy()

        a_opt, max_opt = adam_smooth_anneal(start_noisy, time_budget=t_alloc,
                                            T_ratio=t_ratio, min_T_frac=t_min_frac,
                                            max_iters_per_T=iters_pt, verbose=True)
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score,
                                                f"adam_s{sched_idx}")

        # Follow with FISTA at low temperature
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            T_low = fast_convolve(a_opt).max() * 0.02
            a_fista, _ = fista_smooth(a_opt, T=T_low, max_iters=1000,
                                      time_limit=min(rem * 0.3, 60.0))
            best_seq, best_score = check_and_update(a_fista, best_seq, best_score,
                                                    f"fista_after_s{sched_idx}")

    # === Phase 3: Coordinate descent + Peak shaving (400s) ===
    print(f"\n=== Phase 3: Polishing (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_cd = min(rem * 0.4, 150.0)
        print(f"  CoordDescent n={n_best}, budget={t_cd:.0f}s")
        a_cd, s_cd = coord_descent_polish(best_seq, t_cd, n_sweeps=500)
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd")

    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_ps = min(rem * 0.4, 80.0)
        print(f"  PeakShave budget={t_ps:.0f}s")
        a_ps, s_ps = peak_shave(best_seq, t_ps)
        best_seq, best_score = check_and_update(a_ps, best_seq, best_score, "peak")

    # Another round of Adam with lower temperature
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t_a2 = min(rem * 0.5, 100.0)
        print(f"  Adam round 2, budget={t_a2:.0f}s")
        a_a2, _ = adam_smooth_anneal(best_seq, time_budget=t_a2, T_ratio=0.8,
                                     min_T_frac=0.002, max_iters_per_T=200)
        best_seq, best_score = check_and_update(a_a2, best_seq, best_score, "adam2")

    # === Phase 4: LP Fictitious Play (200s) ===
    print(f"\n=== Phase 4: LP Fictitious Play (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 80
    if rem > 40:
        t_fp = min(rem * 0.6, 150.0)
        print(f"  FP n={n_best}, budget={t_fp:.0f}s")
        try:
            solver = _find_solver()
            a_fp, sf_fp, iters = fictitious_play(n_best, t_fp, initial=best_seq, solver=solver)
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score,
                                                    f"fp_it{iters}")
        except Exception as e:
            print(f"  FP failed: {e}")

    # === Phase 5: Multi-length exploration (100s) ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 40
    if rem > 30:
        alt_lengths = []
        for delta in [-3, -2, -1, 1, 2, 3, 5, -5, 7, -7]:
            alt_n = n_best + delta
            if 256 <= alt_n <= 2048 and alt_n != n_best:
                alt_lengths.append(alt_n)
        # Also try some significantly different lengths
        for factor in [0.85, 1.15]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 2048 and alt_n not in alt_lengths and alt_n != n_best:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:6]:
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            t_alt = min(rem * 0.4, 30.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            # Quick optimization
            a_alt, _ = fista_smooth(init_alt, T=fast_convolve(init_alt).max() * 0.2,
                                    max_iters=200, time_limit=t_alt * 0.5)
            a_alt2, _ = adam_smooth_anneal(a_alt, time_budget=t_alt * 0.5,
                                           T_ratio=0.75, max_iters_per_T=200)
            # Quick coord descent
            rem2 = deadline - time.monotonic() - 25
            if rem2 > 5:
                a_alt3, _ = coord_descent_polish(a_alt2, min(rem2 * 0.5, 10.0),
                                                 n_sweeps=50)
            else:
                a_alt3 = a_alt2
            best_seq, best_score = check_and_update(a_alt3, best_seq, best_score,
                                                    f"alt_n{alt_n}")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 20:
        # Repeated short coordinate descent passes
        n_passes = max(1, int(rem // 15))
        for p in range(n_passes):
            rem = deadline - time.monotonic() - 8
            if rem < 5:
                break
            t_pass = min(rem * 0.5, 20.0)
            a_fin, _ = coord_descent_polish(best_seq, t_pass, n_sweeps=100)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, f"fin_cd{p}")

    # Final FISTA at very low temperature
    rem = deadline - time.monotonic() - 5
    if rem > 5:
        T_very_low = fast_convolve(best_seq).max() * 0.01
        a_final, _ = fista_smooth(best_seq, T=T_very_low, max_iters=500,
                                  time_limit=min(rem * 0.8, 10.0))
        best_seq, best_score = check_and_update(a_final, best_seq, best_score, "final_fista")

    # Final coordinate descent
    rem = deadline - time.monotonic() - 2
    if rem > 3:
        a_last, _ = coord_descent_polish(best_seq, min(rem * 0.9, 8.0), n_sweeps=30)
        best_seq, best_score = check_and_update(a_last, best_seq, best_score, "last_cd")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
