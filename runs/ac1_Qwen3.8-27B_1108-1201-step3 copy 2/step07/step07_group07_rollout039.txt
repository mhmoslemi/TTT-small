            continue
        left = dp[j - 1] if j >= 1 else 0.0
        val = dp[j + n - 1] - left

            active[j] = True

        return a.copy()

    g = np.zeros(n, dtype=np.float64)

def subgradient_step(a, n_peaks=20):
    """
    One subgradient step: compute subgradient of max(conv(a,a)) and take a step.
    Returns updated a (not normalized).
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    threshold = b_max * 0.995
    active_mask = b >= threshold
    if active_mask.sum() > n_peaks:
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        active_mask = np.zeros(len(b), dtype=bool)
        active_mask[top_idx] = True
    elif not active_mask.any():
        active_mask[np.argmax(b)] = True

    # Subgradient: average gradient over active peaks
    # d(conv(a,a)[k])/da[j] = a[k-j] + (a[k-j] if k-j != j and valid)
    # Use cross-convolve: g[j] = sum_k w[k] * a[k-j] (approximately)
    w = active_mask.astype(np.float64) / active_mask.sum()
    # The subgradient of conv(a,a)[k] w.r.t. a is 2 * cross_convolve(a, e_k) shifted
    # More precisely: d/d a[j] conv(a,a)[k] = a[k-j] + (a[j] if k-2j==0 and 0<=k-j<n else a[k-j])
    # Approximation: g = cross_convolve(w, a) + cross_convolve(a, w) (both give similar)
    g = cross_convolve(a, w) + cross_convolve(w, a)
    g = g[:n]
    return g


def subgradient_optimize(a_init, time_limit, lr_init=0.003, momentum=0.9, seed=42, n_top=20):
    """
    Subgradient descent with Nesterov momentum for max(conv(a,a)).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a /= a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_local_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        g = subgradient_step(a, n_top)
        # Normalize subgradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g /= g_norm

        v = momentum * v - cur_lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= ss

        obj_new = eval_local_max_b(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= 200:
            cur_lr *= 0.8
            if cur_lr < lr_init * 0.01:
                a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0)
                a /= a.sum()
                v = np.zeros(n)
                cur_lr = lr_init * 0.5
                no_improve = 0
        elif no_improve >= 100:
            cur_lr *= 0.95

    return best_a, best_obj


def coordinate_descent(a_init, time_limit, n_multipliers=7):
    """
    Coordinate descent: scale one element, rescale others to maintain sum=1.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_obj = eval_local_max_b(a)

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        order = np.random.RandomState(int(time.monotonic() * 1000) % 100000).permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue
                a_tr = a.copy()
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[j] = new_val
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr /= ss
                val = eval_local_max_b(a_tr)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[j] = best_j_val
                a[mask] *= scale
                a = np.maximum(a, 0)
                a /= a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def make_shifted_init(n, shift=0.1, sigma=0.3):
    """Gaussian centered at non-0.5 position to break symmetry."""
    x = np.linspace(0, 1, n)
    center = 0.5 + shift
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_ramp_init(n, asym=0.5):
    """Asymmetric ramp initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + asym * np.sin(np.pi * x)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_power_law_init(n, alpha=1.5):
    """Power-law shaped initialization (asymmetric)."""
    x = np.linspace(0.01, 1.0, n)
    a = x ** alpha + (1 - x) ** (alpha * 0.7)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_cosine_init(n, phase=0.0):
    """Raised cosine with optional phase shift for asymmetry."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x + phase)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def combined_optimize(a_init, time_limit, n, seed=42):
    """
    Combined optimization: LP iteration + subgradient + polish.
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = eval_local_max_b(a)

    # Phase A: LP optimization (primary)
    t_lp = min(time_limit * 0.5, time_limit - 5)
    if t_lp > 2:
        a_lp, obj_lp = lp_optimize(a, t_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase B: Subgradient from LP result
    rem = time_limit - (time.monotonic() - t0)
    t_sg = min(rem * 0.4, 30.0)
    if t_sg > 3:
        a_sg, obj_sg = subgradient_optimize(best_a, t_sg, lr_init=0.002, momentum=0.92, seed=seed, n_top=20)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()

    # Phase C: More LP
    rem = time_limit - (time.monotonic() - t0)
    t_lp2 = min(rem * 0.5, 20.0)
    if t_lp2 > 2:
        a_lp2, obj_lp2 = lp_optimize(best_a, t_lp2)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()

    # Phase D: Coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    t_cd = min(rem, 15.0)
    if t_cd > 3:
        a_cd, obj_cd = coordinate_descent(best_a, t_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses O(n) LP direction finding as the core optimization, with
    multi-length exploration and diverse asymmetric initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_shifted_init(1152, shift=0.08, sigma=0.3)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using shifted init n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: LP optimization on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP optimization on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(rem * 0.45, 180.0)
        print(f"  LP+SG+CD: n={n_parent}, t={t_alloc:.0f}s")
        a_opt, obj_opt = combined_optimize(parent, t_alloc, n_parent, seed=seed)
        update_best(a_opt, "phase1_parent")

    # === Phase 2: Diverse asymmetric initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse Initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {}
    for shift in [0.05, -0.05, 0.1, -0.1, 0.15]:
        inits[f"shift_{shift}"] = make_shifted_init(n_base, shift=shift, sigma=0.3)
    inits["ramp_3"] = make_ramp_init(n_base, asym=0.3)
    inits["ramp_5"] = make_ramp_init(n_base, asym=0.5)
    inits["power_15"] = make_power_law_init(n_base, alpha=1.5)
    inits["power_2"] = make_power_law_init(n_base, alpha=2.0)
    inits["cos_phase"] = make_cosine_init(n_base, phase=0.05)
    inits["cos_phase2"] = make_cosine_init(n_base, phase=0.1)

    total_time_diverse = min(400.0, (deadline - time.monotonic() - 400.0) * 0.5)
    time_per_init = total_time_diverse / len(inits)

    for idx, (name, init_a) in enumerate(inits.items()):
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.3)
        if t_alloc < 8:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = combined_optimize(init_a, t_alloc, n_base, seed=seed + idx * 1000 + 7)
        update_best(a_div, f"div_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length Exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.6, 0.75, 0.9, 1.1, 1.25, 1.5, 2.0]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 200
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            # Add asymmetry to break local minima
            x = np.linspace(0, 1, target_n)
            f_interp = f_interp * (1.0 + 0.1 * np.sin(2 * np.pi * x))
            f_interp = np.maximum(f_interp, 0)
            f_interp /= f_interp.sum()

            t_alloc = min(rem * 0.3, 60.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = combined_optimize(f_interp, t_alloc, target_n, seed=seed + target_n)
            update_best(a_len, f"len_n{target_n}")

    # === Phase 4: Deep optimization on current best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep Optimization (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_deep = min(rem * 0.6, 120.0)
            print(f"  Deep LP: n={n_best}, t={t_deep:.0f}s")
            a_deep, obj_deep = lp_optimize(best_seq, t_deep, max_stall=200)
            update_best(a_deep, "deep_lp")

            # Continue with combined
            rem = deadline - time.monotonic() - 60
            if rem > 20:
                t_comb = min(rem * 0.5, 60.0)
                a_comb, obj_comb = combined_optimize(best_seq, t_comb, n_best, seed=seed + 99999)
                update_best(a_comb, "deep_comb")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Coordinate Descent Polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd = min(rem, 45.0)
            print(f"  CD: n={len(best_seq)}, t={t_cd:.0f}s")
            a_cd, obj_cd = coordinate_descent(best_seq, t_cd)
            update_best(a_cd, "cd")

    # === Phase 6: Final micro-optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Micro-optimization (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            t_micro = min(rem * 0.6, 20.0)
            a_micro, obj_micro = lp_optimize(best_seq, t_micro, max_stall=100)
            update_best(a_micro, "micro_lp")

            rem = deadline - time.monotonic() - 5
            if rem > 3:
                a_mcd, obj_mcd = coordinate_descent(best_seq, rem)
                update_best(a_mcd, "micro_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
