"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best response algorithm
combined with gradient polishing. The LP approach solves maximize sum(b) s.t. b>=0, 
(f*b)[k]<=M for all k, which gives a direction that provably reduces the peak when
mixed with the current iterate. Combined with coordinate descent and peak shaving
for final refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def lp_best_response(f, M, solver='CLARABEL'):
    """Solve the LP: maximize sum(b) s.t. b >= 0, (f*b)[k] <= M for all k.
    Returns the optimal b vector."""
    n = len(f)
    b = cp.Variable(n)
    
    # Cross-correlation constraint matrix
    # (f * b)[k] = sum_i f[i] * b[k-i] for valid i
    # We need to express this as A @ b <= M * ones
    
    # For efficiency, build the constraint matrix
    # (f*b)[k] = sum_{i=0}^{n-1} f[i] * b[k-i] where b[j] = 0 for j<0 or j>=n
    
    # Let's build it row by row
    # Row k: sum_i f[i] * b[k-i] where 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
    
    # Actually, for CVXPY, let's just use the matrix form
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # So A[k,j] = f[k-j] if j <= k and k-j < n
    
    # This is a convolution matrix (Toeplitz-like)
    n_constraints = 2 * n - 1
    # A is n_constraints x n
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1 and 0 <= j <= n-1
    
    # For CVXPY, we can use the constraint directly
    # (convolve(f, b))[k] <= M
    
    # Let's use the explicit form with CVXPY
    # Actually, let's build the matrix
    
    # A[k, j] = f[k-j] if max(0, k-n+1) <= j <= min(n-1, k)
    A = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[k, j] = f[k - j]
    
    constraints = [b >= 0, A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if b.value is not None:
            return np.maximum(b.value, 0)
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None


def lp_best_response_sparse(f, M, solver='CLARABEL'):
    """More efficient LP using sparse matrix."""
    from scipy.sparse import csr_matrix
    n = len(f)
    n_constraints = 2 * n - 1
    
    # Build sparse representation
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    # For each k, the non-zero entries are at j = max(0,k-n+1) to min(n-1,k)
    
    rows = []
    cols = []
    vals = []
    
    for k in range(n_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    
    b = cp.Variable(n)
    constraints = [b >= 0, A_sparse @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if b.value is not None:
            return np.maximum(b.value, 0)
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None

    t0 = time.monotonic()

            break
        
        b = fast_convolve(f)
        M = float(b.max())

        if b_lp is None:
            # Try dense
            b_lp = lp_best_response(f, M)
        if b_lp is None:
            break
        
        b_sum = b_lp.sum()
        if b_sum < 1e-15:
            break
        
        g = b_lp / b_sum  # Normalize so sum(g) = 1
        
        # Try different step sizes
        best_t_score = current_score
        best_t = 0
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:

            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 1e-15:

            sc = score_from_a(f_trial)
            if sc < best_t_score:
                best_t_score = sc
                best_t = t
        
        if best_t == 0 or best_t_score >= best_score - 1e-14:
            # Try smaller steps
            for t in [0.05, 0.1, 0.2]:
                f_trial = (1 - t) * f + t * g
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s < 1e-15:
                    continue
                f_trial = f_trial / s
                sc = score_from_a(f_trial)
                if sc < best_t_score:
                    best_t_score = sc
                    best_t = t
        
        if best_t > 0 and best_t_score < best_score - 1e-14:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)

                print(f"    {label} iter={it} ({elapsed:.1f}s): no improvement, stopping")
            break
    
    return best_f, best_score

def peak_shave(a_init, time_limit, verbose=True, label=""):
    """Targeted peak shaving: reduce contributions to the top peaks."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for step in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        if b[peak_k] <= best_score / (2.0 * n) + 1e-16:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_count = min(8, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_count]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing both i and j
            for ci in top_idx[:3]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-14 or i == j:
                    continue
                for f1 in [0.9, 0.8, 0.7]:
                    for f2 in [0.9, 0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_from_a(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                # Random restart from best
                rng = np.random.default_rng(step)
                a = best_a + rng.standard_normal(n) * 0.01
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                if score_from_a(a) > best_score + 0.01:
                    break

    return best_a, best_score


def multiplicative_weights_exact(a_init, time_limit, eta_start=0.005,
                                 decay=0.9995, max_iters=50000, verbose=True, label=""):
    """Multiplicative weights using exact peak subgradient (no smoothing)."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = eta_start

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        k_peak = int(np.argmax(b))

        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_peak - idx]

        grad = grad - np.dot(a, grad)

        log_update = -eta * grad
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        eta *= decay

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MW iter={it} ({elapsed:.0f}s): eta={eta:.6f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def coordinate_descent_polish(a_init, time_limit, verbose=True, label=""):
    """Faster coordinate descent with batch factor updates."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    factors = [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]
    no_improve = 0
    max_no_improve = 50

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break
        if not improved_pass:
            no_improve += 1

    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best response algorithm
combined with mirror descent, peak shaving, and coordinate descent polishing.
The LP approach (maximize sum(b) s.t. b>=0, (f*b)[k]<=M) gives a provably
improving direction when mixed with the current iterate. Combined with
multi-phase refinement and adaptive step sizing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def lp_solve(f, M, solver='CLARABEL'):
    """Solve LP: maximize sum(b) s.t. b >= 0, (f*b)[k] <= M for all k."""
    n = len(f)
    n_con = 2 * n - 1

    # Build sparse constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n
    rows = []
    cols = []
    vals = []
    for k in range(n_con):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])

    A_sparse = csr_matrix((vals, (rows, cols)), shape=(n_con, n))

    b_var = cp.Variable(n)
    constraints = [b_var >= 0, A_sparse @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=solver)
        if b_var.value is not None:
            return np.maximum(b_var.value, 0.0)
    except Exception:
        pass

    # Fallback: try ECOS
    try:
        prob2 = cp.Problem(objective, constraints)
        prob2.solve(solver='ECOS')
        if b_var.value is not None:
            return np.maximum(b_var.value, 0.0)
    except Exception:
        pass

    return None


def lp_iterate(f_init, time_limit, max_iters=80, verbose=True, label=""):
    """LP-based best response iteration with adaptive step sizing."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()

    if verbose:
        print(f"  {label}: n={n}, initial score={best_score:.7f}")

    no_improve_count = 0
    max_no_improve = 5

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(f)
        M = float(b.max())
        current_score = 2.0 * n * M

        # Solve LP
        t_lp_start = time.monotonic()
        b_lp = lp_solve(f, M)
        t_lp = time.monotonic() - t_lp_start

        if b_lp is None:
            if verbose:
                print(f"    {label} iter={it}: LP failed, stopping")
            break

        b_sum = b_lp.sum()
        if b_sum < 1e-15:
            break

        g = b_lp / b_sum  # Normalize sum(g) = 1

        # Try multiple step sizes via line search
        best_t_score = current_score
        best_t = 0.0
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-15:
                continue
            f_trial = f_trial / s
            sc = score_from_a(f_trial)
            if sc < best_t_score:
                best_t_score = sc
                best_t = t

        if best_t > 0 and best_t_score < best_score - 1e-14:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            best_score = best_t_score
            best_f = f.copy()
            no_improve_count = 0

            if verbose and (it % 10 == 0 or it <= 5):
                elapsed = time.monotonic() - t0
                print(f"    {label} iter={it} ({elapsed:.1f}s, lp={t_lp:.1f}s): "
                      f"t={best_t:.2f} score={best_score:.7f}")
        else:
            no_improve_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    {label} iter={it} ({elapsed:.1f}s): no improvement "
                      f"(no_improve={no_improve_count})")
            if no_improve_count >= max_no_improve:
                break

    return best_f, best_score


def mirror_descent_optimize(a_init, time_limit, T_start=0.01, T_end=0.0005,
                            eta_start=0.02, eta_end=0.0005, max_iters=20000,
                            verbose=True, label=""):
    """Adaptive mirror descent with temperature annealing on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stuck_count = 0
    stuck_threshold = 500

    fft_size = 1 << (3 * n - 1).bit_length()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Soft-max weights
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()

        # Gradient computation via FFT cross-correlation
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 4.0 * n * correl[:n]

        # Project to tangent space
        mean_grad = np.dot(a, grad_a)
        grad_a = grad_a - mean_grad

        # Multiplicative update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        # Stuck detection
        if stuck_count > stuck_threshold:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.03
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.0f}s): T={T:.5f} "
                  f"eta={eta:.6f} score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def multiplicative_weights_exact(a_init, time_limit, eta_start=0.005,
                                 decay=0.9995, max_iters=50000, verbose=True, label=""):
    """Multiplicative weights using exact peak subgradient."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = eta_start

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        k_peak = int(np.argmax(b))

        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_peak - idx]

        grad = grad - np.dot(a, grad)

        log_update = -eta * grad
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        eta *= decay

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MW iter={it} ({elapsed:.0f}s): eta={eta:.6f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=True, label=""):
    """Aggressive peak shaving with multiple strategies."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find the peak (exclude endpoints which are always small)
        b_work = b.copy()
        if n > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_count = min(10, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_count]

        improved = False

        # Strategy 1: Single coordinate reduction
        for ci in top_idx:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7, 0.6]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Pair coordinate reduction
        if not improved:
            for ci in top_idx[:5]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8, 0.7]:
                    for f2 in [0.9, 0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_from_a(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Reduce top contributor and redistribute to low-contrib positions
        if not improved:
            i = int(idx[top_idx[0]])
            a_trial = a.copy()
            old_val = a_trial[i]
            a_trial[i] = old_val * 0.8
            diff = old_val - a_trial[i]
            # Redistribute to positions that don't contribute to this peak
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            # Prioritize positions far from the peak
            others_idx = np.where(mask)[0]
            dist = np.abs(others_idx - peak_k) + np.abs(others_idx + peak_k)
            order = np.argsort(dist)[::-1][:n // 3]
            target_positions = others_idx[order]
            weights = a_trial[target_positions]
            w_sum = weights.sum()
            if w_sum > 1e-15:
                a_trial[target_positions] += diff * weights / w_sum
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Random perturbation to escape
            rng = np.random.default_rng(step * 31 + 7)
            a = best_a + rng.standard_normal(n) * 0.015
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            if score_from_a(a) > best_score * 1.01:
                break

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} PS step={step} ({elapsed:.1f}s): score={best_score:.7f}")

    return best_a, best_score


def coordinate_descent_polish(a_init, time_limit, verbose=True, label=""):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    factors = [0.7, 0.8, 0.85, 0.9, 1.1, 1.2, 1.3]
    no_improve = 0
    max_no_improve = 30

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break
        if not improved_pass:
            no_improve += 1

        if verbose and no_improve % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} CD ({elapsed:.0f}s): best={best_score:.7f} "
                  f"no_improve={no_improve}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP best response (primary novel method)
    2. Mirror descent with annealing
    3. Additional LP rounds from improved points
    4. Multiplicative weights
    5. Peak shaving (multiple rounds)
    6. Coordinate descent polish
    7. Alternative length exploration
    8. Final refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # === Phase 1: LP Best Response (main novelty) ===
    print("\n=== Phase 1: LP Best Response ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t_alloc = min(rem * 0.45, 250.0)
            print(f"  LP1: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lp, _ = lp_iterate(best_seq, t_alloc, max_iters=100, label="LP1")
            best_seq, best_score = check_and_update(a_lp, "LP1", best_seq, best_score)

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        configs = [
            (0.015, 0.0005, 0.025, 0.001, "MD1"),
            (0.008, 0.0003, 0.015, 0.0005, "MD2"),
        ]
        for T_s, T_e, eta_s, eta_e, lbl in configs:
            rem = deadline - time.monotonic() - 350
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 150.0)
            print(f"  {lbl}: budget={t_alloc:.0f}s")
            a_md, _ = mirror_descent_optimize(
                best_seq, t_alloc, T_start=T_s, T_end=T_e,
                eta_start=eta_s, eta_end=eta_e, label=lbl
            )
            best_seq, best_score = check_and_update(a_md, lbl, best_seq, best_score)

    # === Phase 3: LP Round 2 ===
    print(f"\n=== Phase 3: LP Round 2 (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 60:
            t_alloc = min(rem * 0.45, 200.0)
            print(f"  LP2: budget={t_alloc:.0f}s")
            a_lp2, _ = lp_iterate(best_seq, t_alloc, max_iters=60, label="LP2")
            best_seq, best_score = check_and_update(a_lp2, "LP2", best_seq, best_score)

    # === Phase 4: Multiplicative Weights ===
    print(f"\n=== Phase 4: MW (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem * 0.4, 100.0)
            print(f"  MW: budget={t_alloc:.0f}s")
            a_mw, _ = multiplicative_weights_exact(
                best_seq, t_alloc, eta_start=0.006, decay=0.9996, label="MW"
            )
            best_seq, best_score = check_and_update(a_mw, "MW", best_seq, best_score)

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.5, 100.0)
            print(f"  PS1: budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc, label="PS1")
            best_seq, best_score = check_and_update(a_ps, "PS1", best_seq, best_score)

    # === Phase 6: LP Round 3 ===
    print(f"\n=== Phase 6: LP Round 3 (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 50:
            t_alloc = min(rem * 0.5, 100.0)
            print(f"  LP3: budget={t_alloc:.0f}s")
            a_lp3, _ = lp_iterate(best_seq, t_alloc, max_iters=40, label="LP3")
            best_seq, best_score = check_and_update(a_lp3, "LP3", best_seq, best_score)

    # === Phase 7: Peak Shaving Round 2 ===

        if rem > 15:
            t_alloc = min(rem * 0.5, 60.0)
            print(f"  PS2: budget={t_alloc:.0f}s")

    print(f"\n=== Phase 8: Coord Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem * 0.5, 50.0)
            print(f"  CD: budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent_polish(best_seq, t_alloc, label="CD")
            best_seq, best_score = check_and_update(a_cd, "CD", best_seq, best_score)

    print(f"\n=== Phase 9: LP Final (best={best_score:.7f}) ===")

        if rem > 40:

            a_lp4, _ = lp_iterate(best_seq, t_alloc, max_iters=25, label="LP4")
            best_seq, best_score = check_and_update(a_lp4, "LP4", best_seq, best_score)

    # === Phase 10: Final PS + CD ===
    print(f"\n=== Phase 10: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(rem * 0.5, 25.0)
            a_ps_fin, _ = peak_shave(best_seq, t_alloc, label="PS_fin")
            best_seq, best_score = check_and_update(a_ps_fin, "PS_fin", best_seq, best_score)

        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem * 0.6, 15.0)
            a_cd_fin, _ = coordinate_descent_polish(best_seq, t_alloc, label="CD_fin")
            best_seq, best_score = check_and_update(a_cd_fin, "CD_fin", best_seq, best_score)

    # === Phase 11: Alternative length exploration ===
    print(f"\n=== Phase 11: Alt Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.0, 0.75, 1.25, 1.5]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 3072 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:1]:
            rem = deadline - time.monotonic() - 30
            if rem < 40:
                break
            t_alloc = min(rem * 0.6, 40.0)
            print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = mirror_descent_optimize(
                init_alt, t_alloc, T_start=0.01, T_end=0.0005,
                eta_start=0.02, eta_end=0.0005, label=f"alt_n{alt_n}"
            )
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}",
                                                    best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
