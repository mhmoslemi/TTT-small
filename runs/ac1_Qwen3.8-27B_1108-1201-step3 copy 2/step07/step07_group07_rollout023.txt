def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def eval_local_max_b(a):

def compute_subgradient(a, b, n_top=50, threshold_ratio=0.999):

    Compute subgradient of max(conv(a,a)) with respect to a.
    Uses peak-weighted average.
    """
    n = len(a)
    b_max = b.max()
    
    # Select peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))
    
    top_vals = b[top_idx]
    threshold = b_max * threshold_ratio
    active_mask = top_vals >= threshold
    if not active_mask.any():
        active_mask[0] = True
    
    active_indices = top_idx[active_mask]
    active_vals = b[active_indices]
    
    # Weight by excess over mean
    mean_b = b.mean()
    weights = np.maximum(active_vals - mean_b, 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(len(active_indices))
    weights = weights / weights.sum()
    
    # Compute subgradient: g[i] = sum_k w_k * a[k-i] (for valid k-i)

    w = np.zeros(2 * n - 1)

    g = irfft(np.conj(A) * W, fft_size)[:n]

def build_lp_matrix(a):
    """Build the LP constraint matrix for conv(f,g) <= M."""
    n = len(a)
    num_constraints = 2 * n - 1
    # Use the structure: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Hankel-like matrix
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, k - j] = a[j]
    return A

def make_gaussian_init(n, sigma=0.3):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on the simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_local_max_b(a)
    
    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue
                
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_local_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:

def solve_kkt(S, n):
    """
    Solve the KKT system for a fully supported solution given peak set S.
    Returns (a, c) or None if no valid solution exists.
    """
    # Build matrix B where B[i, j] = 1 if i+j in S, else 0
    # System: B @ a = c * ones, sum(a) = 1
    # Augmented: [B, -1; ones^T, 0] @ [a; c] = [0; 1]
    
    S_set = set(S)
    B = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if (i + j) in S_set:
                B[i, j] = 1.0
    
    # Augmented system
    A_aug = np.zeros((n+1, n+1))
    A_aug[:n, :n] = B
    A_aug[:n, n] = -1.0
    A_aug[n, :n] = 1.0
    A_aug[n, n] = 0.0
    
    rhs = np.zeros(n+1)
    rhs[n] = 1.0
    
    try:
        sol = np.linalg.solve(A_aug, rhs)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(A_aug, rhs, rcond=None)[0]
        except:
            return None
    
    a = sol[:n]
    c = sol[n]
    
    # Check non-negativity
    if np.any(a < -1e-10):
        return None
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return None
    a = a / s
    
    return a

def kkt_solve(S_set, n):
    """
    Solve the KKT system for fully supported solution.
    S_set: set of peak indices (integers)
    n: sequence length
    Returns (a, c) where a is the solution (sum=1, a>=0) or None.
    """
    # Build B: B[i,j] = 1 if (i+j) in S_set
    B = np.zeros((n, n))
    S_arr = np.array(sorted(S_set))
    for i in range(n):
        for k in S_arr:
            j = k - i
            if 0 <= j < n:
                B[i, j] = 1.0

    A_aug = np.zeros((n+1, n+1))
    A_aug[:n, :n] = B
    A_aug[:n, n] = -1.0
    A_aug[n, :n] = 1.0

    rhs[n] = 1.0
    
    try:
        sol = np.linalg.solve(A_aug, rhs)
    except np.linalg.LinAlgError:
        return None
    
    a = sol[:n]
    c = sol[n]
    
    if np.any(a < -1e-8):
        return None
    
    a = np.maximum(a, 0.0)

    if s < 1e-10:

    return a
