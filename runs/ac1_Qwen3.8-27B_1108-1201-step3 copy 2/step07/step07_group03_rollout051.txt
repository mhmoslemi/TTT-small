If `i = k-i` (i.e., `k = 2i`), these are the same term, so it's just `a_i^2` becoming `(a_i+d)^2 = a_i^2 + 2*a_i*d + d^2`.

If `k != 2i`, the two terms are distinct: `a_i * a_{k-i} + a_{k-i} * a_i = 2 * a_i * a_{k-i}` becoming `2*(a_i+d) * a_{k-i} = 2*a_i*a_{k-i} + 2*d*a_{k-i}`.

So:
- If `k = 2i`: change is `2*a_i*d + d^2`
- If `k != 2i` and `0 <= k-i < n`: change is `2*d*a_{k-i}`
- Otherwise: change is 0

So the subgradient with respect to `a_i` is:
`d/d a_i [conv(a,a)_k] = 2*a_{k-i}` (for `0 <= k-i < n` and `k != 2i`)
`d/d a_i [conv(a,a)_k] = 2*a_i` (for `k = 2i`)

More compactly: the subgradient component for coordinate `i` of `conv(a,a)_k` is `2 * a_{k-i}` if `0 <= k-i < n`, with a special case that when `k=2i`, it's `2*a_i` (which is the same formula since `a_{k-i} = a_i` when `k=2i`).

def build_conv_matrix(a):
    """Build matrix T (2n-1 x n) such that T @ b = conv(a, b)."""
    n = len(a)

def simplex_project(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1)
    cond = u * rho_arr > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def projected_subgradient(a0, time_limit, lr_init=0.005, top_k=20):
    """Projected subgradient with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = eval_score_fast(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4*(it-1)**2)) / 2
        beta = (t_k - 1) / t_k

        # Compute subgradient of max_k conv(y,y)_k
        b = fast_autoconv(y)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.01 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Subgradient: g_i = 2 * sum_k w_k * y_{k-i} for valid indices
        g = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = np.arange(i_start, i_end + 1)
                g[j] += 2.0 * w[idx] * y[k - j]

        v = beta * v + g
        nv = np.linalg.norm(v)
        if nv > 1e-15:
            v = v / nv

        y_new = y - lr * v
        y_new = simplex_project(y_new)

        score = eval_score_fast(y_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_new

        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
        if no_improve > 600:
            rng = np.random.default_rng(it * 31 + 7)
            y = simplex_project(best_a + rng.standard_normal(n) * 0.03)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0

        lr *= 0.9998

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    f_interp = np.interp(x_new, x_old, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def diverse_init(n, rng, seed_type=0):
    """Generate diverse initial sequences."""
    if seed_type == 0:
        # Fourier-based
        n_harm = int(rng.integers(3, 20))
        amps = rng.random(n_harm) * 0.3
        phases = rng.uniform(0, 2*np.pi, n_harm)
        freqs = np.arange(1, n_harm+1)
        t = np.arange(n, dtype=np.float64)
        a = np.full(n, 0.5)
        for j in range(n_harm):
            a += amps[j] * np.cos(2*np.pi*freqs[j]*t/n + phases[j])
    elif seed_type == 1:
        # Block-based
        n_blocks = int(rng.integers(4, 30))
        a = np.zeros(n)
        for b in range(n_blocks):
            s_start = b * n // n_blocks
            s_end = (b+1) * n // n_blocks
            a[s_start:s_end] = rng.random() + 0.1
        # Smooth
        ksize = max(3, n // 100)
        if ksize > 1:
            kernel = np.ones(ksize) / ksize
            a = np.convolve(a, kernel, mode='same')
    elif seed_type == 2:
        # Sparse
        density = rng.uniform(0.2, 0.5)
        a = np.zeros(n)
        mask = rng.random(n) < density
        a[mask] = rng.random(int(mask.sum())) + 0.05
    else:
        # Smoothed noise
        a = np.ones(n) * 0.5 + rng.standard_normal(n) * 0.2

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a

def damped_lp(a0, time_limit, solver, alpha_init=1.0):
    """Damped alternating LP with geometrically decreasing step size."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_score = eval_score_fast(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new, _ = solve_lp_response(a, solver)
        if b_new is None:
            continue
        alpha = alpha_init / (1 + 0.5 * it)
        alpha = max(alpha, 0.01)
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-14:
            break
        a_new /= sc
        score = eval_score_fast(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

def peak_shave(a0, time_limit):
    """Target the highest convolution peaks by reducing contributing coordinates."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_score = eval_score_fast(a)
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_autoconv(a)
        peak_k = int(np.argmax(b))
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = idx[np.argsort(contribs)[::-1][:8]]

        improved = False
        for i in top_idx:
            if a[i] < 1e-14:
                continue
            for f in [0.8, 0.6, 0.4]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial = np.maximum(a_trial, 0)
                sc = a_trial.sum()
                if sc < 1e-14:
                    continue
                a_trial /= sc
                score = eval_score_fast(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reductions
            for i in top_idx[:5]:
                if a[i] < 1e-14:
                    continue
                j = peak_k - i
                if j < 0 or j >= n or i == j:
                    continue
                if a[j] < 1e-14:
                    continue
                for f1 in [0.8, 0.6]:
                    for f2 in [0.8, 0.6]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        sc = a_trial.sum()
                        if sc < 1e-14:
                            continue
                        a_trial /= sc
                        score = eval_score_fast(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score

def propose_candidate():
    """
    Multi-phase optimizer:
    1. Evaluate parent and uniform baselines
    2. LP-based alternating optimization at multiple lengths
    3. Coordinate descent polishing
    4. Projected subgradient refinement
    5. Peak-targeted shaving
    6. Iterative improvement loops
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(2024)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        score = float(evaluate_sequence(a_n.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_n.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_n)})")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            try_update(parent, "parent")
            print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    # === Phase 1: Quick baselines and diverse starts ===
    print("\n=== Phase 1: Baselines ===")
    # Uniform
    for n_u in [512, 768, 1024, 1280, 1536]:
        try_update(np.ones(n_u) / n_u, f"uniform_n{n_u}")

    # Diverse initializations, quick optimize
    n_main = len(parent) if parent is not None else 768
    n_candidates = [n_main]
    for alt in [512, 640, 768, 896, 1024, 1280, 1536]:
        if alt != n_main:
            n_candidates.append(alt)

    quick_results = []
    for n_c in n_candidates[:5]:
        for st in range(4):
            if time.monotonic() > deadline - 1600:
                break
            if st == 0 and parent is not None:
                init = interpolate_seq(parent, n_c)
            else:
                init = diverse_init(n_c, rng, seed_type=st)
            # Quick subgradient (5s)
            a_q, s_q = projected_subgradient(init, 5.0, lr_init=0.006, top_k=10)
            quick_results.append((s_q, a_q, n_c))

    quick_results.sort(key=lambda x: x[0])
    for s_r, a_r, n_r in quick_results[:5]:
        try_update(a_r, f"quick_n{n_r}")

    # === Phase 2: Main LP alternating at best length ===
    print(f"\n=== Phase 2: LP Alternating (best={best_score:.8f}) ===")
    n_focus = len(best_seq) if best_seq is not None else n_main
    t2_budget = min(800.0, deadline - time.monotonic() - 600)
    if t2_budget > 30:
        print(f"  Focus n={n_focus}, budget={t2_budget:.0f}s")
        init_fp = best_seq.copy() if best_seq is not None else np.ones(n_focus) / n_focus
        a_lp, s_lp = alternating_lp(init_fp, t2_budget, solver, verbose=True)
        try_update(a_lp, f"altlp_n{n_focus}")

        # Damped LP follow-up
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_dl = min(rem * 0.3, 100.0)
            a_dl, s_dl = damped_lp(best_seq, t_dl, solver)
            try_update(a_dl, f"damped_n{n_focus}")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_cd = min(rem * 0.5, 200.0)
            print(f"  BD budget={t_cd:.0f}s")
            a_cd, s_cd = coord_descent(best_seq, t_cd, max_sweeps=3000)
            try_update(a_cd, "coord_descent")

    # === Phase 4: Subgradient + restarts ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        for lr_try in [0.003, 0.008, 0.015]:
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            t_sg = min(rem * 0.3, 80.0)
            print(f"  SG lr={lr_try}, budget={t_sg:.0f}s")
            a_sg, s_sg = projected_subgradient(best_seq, t_sg, lr_init=lr_try, top_k=30)
            try_update(a_sg, f"sg_lr{lr_try}")

            # Follow with coord descent
            rem = deadline - time.monotonic() - 150
            if rem > 10:
                a_cd2, _ = coord_descent(best_seq, min(rem * 0.3, 40.0), max_sweeps=500)
                try_update(a_cd2, "cd_after_sg")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shave (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 10:
            t_ps = min(rem * 0.4, 80.0)
            a_ps, s_ps = peak_shave(best_seq, t_ps)
            try_update(a_ps, "peak_shave")

    # === Phase 6: Second LP round with refined start ===
    print(f"\n=== Phase 6: LP Round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_fp2 = min(rem * 0.6, 120.0)
            a_fp2, s_fp2 = alternating_lp(best_seq, t_fp2, solver, verbose=False)
            try_update(a_fp2, "lp_round2")

            rem = deadline - time.monotonic() - 30
            if rem > 10:
                a_dl2, _ = damped_lp(best_seq, min(rem * 0.5, 40.0), solver)
                try_update(a_dl2, "damped_round2")

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for factor in [1.5, 0.7, 2.0]:
            alt_n = int(n_best * factor)
            if alt_n < 128 or alt_n > 3000 or alt_n == n_best:
                continue
            rem = deadline - time.monotonic() - 40
            if rem < 20:
                break
            t_alt = min(rem * 0.5, 60.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt = alternating_lp(init_alt, t_alt, solver, verbose=False)
            try_update(a_alt, f"alt_n{alt_n}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_fcd = min(rem * 0.4, 40.0)
            a_fin, _ = coord_descent(best_seq, t_fcd, max_sweeps=1000)
            try_update(a_fin, "final_cd")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_fin2, _ = peak_shave(best_seq, min(rem * 0.5, 15.0))
            try_update(a_fin2, "final_ps")

        rem = deadline - time.monotonic() - 3
        if rem > 3:
            a_fin3, _ = projected_subgradient(best_seq, min(rem * 0.7, 10.0), lr_init=0.001, top_k=50)
            try_update(a_fin3, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
