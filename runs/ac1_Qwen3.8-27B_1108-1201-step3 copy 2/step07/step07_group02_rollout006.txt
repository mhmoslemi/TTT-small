The Frank-Wolfe method would be:
- At each step, solve the linearization: minimize `<g, v>` over the simplex, where `g` is a subgradient
- The minimizer over the simplex is a vertex: `v = e_i` where `i = argmin g[i]`
- Update: `a <- (1-t) * a + t * e_i`

This is very fast! Each step is O(n) for the subgradient computation (if we precompute the active set) plus O(n) for the update.

Wait, the subgradient of `max_k (a*a)[k]` at point `a` is:
- If `S = {k : (a*a)[k] = max}`, then `g = (1/|S|) * sum_{k in S} ∂(a*a)[k]/∂a`
- `∂(a*a)[k]/∂a[i] = a[k-i]` (when valid indices)

So the subgradient computation requires knowing the active set and computing a weighted sum of shifts of `a`.

For Frank-Wolfe, the linear minimization oracle over the simplex gives us a vertex. So the update is:
`a_new = (1-t) * a + t * e_{i*}` where `i* = argmin_i g[i]`.

This is extremely fast - each iteration is O(n) for the argmin plus O(n) for the update. The convolution computation is O(n log n) with FFT.

```python
def frank_wolfe(a, time_limit):
    n = len(a)
    a = a / a.sum()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2*n*fast_convolve(a).max()
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        score = 2*n*max_b
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find active set (within tolerance of max)
        tol = 1e-4 * max_b
        active = np.where(b >= max_b - tol)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n-1, k)
            idx = np.arange(js, je+1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Frank-Wolfe step: move towards vertex minimizing g
        i_star = np.argmin(g)
        
        # Step size: standard FW is t = 2/(iter+2), or line search
        t = 2.0 / (iter_count + 2)
        
        a = (1 - t) * a + t * np.eye(n)[i_star]
        # a is already on the simplex
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        iter_count += 1
        
        if iter_count % 10000 == 0:
            print(f"  FW iter={iter_count} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    
    return best_a
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0)


def compute_score(a):
    """Compute the objective value (assuming sum(a) = 1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def softmax_stable(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def subgradient(a, b, rel_tol=0.05):
    """Compute subgradient of max_k (a*a)[k] at point a.
    
    Parameters:
    a: probability vector (length n)
    b: auto-convolution of a (length 2n-1)
    rel_tol: relative tolerance for active set
    
    Returns:
    g: subgradient vector (length n)
    """
    n = len(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.ones(n) / n
    
    threshold = max_b * (1 - rel_tol)
    active = np.where(b >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    
    g /= len(active)
    return g


def fast_subgradient_fft(a, b, rel_tol=0.05):
    """FFT-accelerated subgradient computation for large active sets."""
    n = len(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.ones(n) / n
    
    threshold = max_b * (1 - rel_tol)
    active_mask = (b >= threshold).astype(np.float64)
    n_active = active_mask.sum()
    
    if n_active < 50:
        # Direct method is faster for small active sets
        active = np.where(active_mask)[0]
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        return g / n_active
    
    # FFT-based: g[i] = sum_k w[k] * a[k-i] where w = active_mask / n_active
    # = convolve(w_padded, a, 'valid') where w_padded has length 2n-1
    w = active_mask / n_active
    # g[i] = sum_{j=0}^{n-1} w[i+j] * a[j] = convolve(w, a, 'valid')[i]
    g = np.convolve(w, a, mode='valid')
    # g should have length (2n-1) - n + 1 = n
    if len(g) != n:
        # Fallback
        active = np.where(active_mask)[0]
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= n_active
    return g


def nesterov_mirror_descent(a0, time_limit, lr_init=0.1, rel_tol=0.05, verbose=True):

    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)

            best_score = score_new

        a = a_new
        z = z_y_new

        # Standard Nesterov: y_{t+1} = x_{t+1} + (t/(t+3)) * (x_{t+1} - x_t)
        # In log-space: z_y_{t+1} = z_{t+1} + (t/(t+3)) * (z_{t+1} - z_t)
        momentum = step / (step + 3.0) if step > 0 else 0.0
        z_y = z + momentum * (z - z_y)
        
        # Actually, the standard Nesterov update is:
        # y_k = x_k + (k-1)/(k+2) * (x_k - x_{k-1})
        # x_{k+1} = proj(y_k - eta * grad(y_k))
        # 
        # But in mirror descent, we work in the dual space:
        # z_y = z + momentum * (z - z_prev)
        # z_new = z_y - lr * g(z_y)
        # 
        # Let me redo this properly:
        # We maintain z (current) and z_prev (previous)
        # Virtual: z_y = z + (t/(t+3)) * (z - z_prev)
        # Step: z_new = z_y - lr * g(softmax(z_y))

def frank_wolfe_linesearch(a0, time_limit, verbose=True):
    """Frank-Wolfe with strong step size (line search along the edge)."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    
    for step in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        g = subgradient(a, b, rel_tol=0.02)
        
        # FW direction: move towards vertex minimizing g
        i_star = np.argmin(g)
        if a[i_star] > 1 - 1e-10:
            # Already at a vertex
            break
        
        # Line search: minimize f((1-t)*a + t*e_i) for t in [0,1]
        # Use ternary search (f is convex in t)
        e_i = np.zeros(n)
        e_i[i_star] = 1.0
        
        def f_of_t(t):
            a_t = (1 - t) * a + t * e_i
            return compute_score(a_t)
        
        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(40):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            if f_of_t(m1) < f_of_t(m2):
                hi = m2
            else:
                lo = m1
        
        t_opt = (lo + hi) / 2
        a = (1 - t_opt) * a + t_opt * e_i
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and (step + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            n_nonzero = np.sum(a > 1e-10)
            print(f"    FW {elapsed:.0f}s step={step+1}: best={best_score:.8f} support={n_nonzero}")
        
        # Check convergence
        if best_score < 1.50 and step > 100:
            # We're in a good region, slow down
            pass
    
    return best_a, best_score


def lp_best_response(a_fixed, solver=None):
    """Solve LP to find b minimizing max_k conv(a,b)[k] subject to sum(b)=1, b>=0."""
    if not HAS_CVXPY:
        return None, None
    
    if solver is None:
        solver = _find_lp_solver()
    
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution constraints: conv(a_fixed, b_var)[k] <= M_var for all k
    # conv(a_fixed, b_var)[k] = sum_j a_fixed[j] * b_var[k-j]
    # This is a linear function of b_var
    
    # Use a sparse matrix representation
    # For efficiency, build the constraint matrix
    constraints = [cp.sum(b_var) == 1]
    
    # For each k in 0..2n-2:
    #   sum_{j: 0<=k-j<n} a_fixed[j] * b_var[k-j] <= M_var
    #   i.e., sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_fixed[j] * b_var[k-j] <= M_var
    
    # Build as a matrix: T[k, i] = a_fixed[k-i] if 0 <= k-i < n, else 0
    # Then T @ b_var <= M_var * ones
    
    # For n=768, T is 1535 x 768. Let's build it.
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints.append(T @ b_var <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def _find_lp_solver():
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def soft_max_lbgfs(a0, time_limit, T_start=5.0, T_end=0.5, maxiter=3000, verbose=True):
    """L-BFGS with soft-max smoothing at low temperature."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    stopped = [False]
    
    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            a_loc = softmax_stable(z)
            b = fast_convolve(a_loc)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(z)
            a_loc = softmax_stable(z)
            b = fast_convolve(a_loc)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_loc[k - idx]
            S = np.dot(grad_a, a_loc)
            return a_loc * (grad_a - S)
        return jac
    
    # Temperature stages
    n_stages = 3
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': iters_per_stage * 30, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = softmax_stable(z)
    if verbose:
        print(f"    L-BFGS done ({time.monotonic()-t0:.0f}s): score={compute_score(a_out):.8f}")
    return a_out


def peak_shave(a0, time_limit, n_steps=5000, verbose=True):
    """Targeted peak shaving: reduce the largest contributors to the peak."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)

        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:8]
        
        improved = False

            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] *= factor
                # Redistribute to maintain sum
                others_sum = a_trial.sum()
                a_trial = a_trial / others_sum
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing top 2 simultaneously
            if len(top) >= 2:
                for i1 in top[:4]:
                    for i2 in top[:4]:
                        i = idx[i1]
                        j = idx[i2]
                        if i == j or a[i] < 1e-14 or a[j] < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[i] *= 0.95
                        a_trial[j] *= 0.95
                        a_trial = a_trial / a_trial.sum()
                        score = compute_score(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            # Try small random perturbation
            if step % 100 == 0:
                rng = np.random.default_rng(step)
                a_trial = a.copy()
                idxs = rng.choice(n, size=5, replace=False)
                for i in idxs:
                    a_trial[i] *= rng.uniform(0.8, 1.2)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
        
        if not improved and step > 500:
            # Try accepting small improvements with noise
            rng = np.random.default_rng(step + 10000)
            a_trial = a * (1 + 0.01 * rng.standard_normal(n))
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                break
            a_trial = a_trial / s
            score = compute_score(a_trial)
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
            elif step % 500 == 0:
                # No improvement in a while, break
                if step > 2000:
                    break
    
    if verbose:
        print(f"    PeakShave done ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    return best_a, best_score

def interpolate_to_length(f, new_n):
    """Interpolate a probability vector to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
```
