"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent.
Key insight: For fixed other coordinates, the objective as a function of a_i is
a convex piecewise-linear (plus one quadratic) function, enabling exact 1D minimization
via ternary search per coordinate. Multiple strategies are combined:
1. Exact coordinate descent (primary)
2. Perturbation restarts to escape local minima
3. Multiple sequence lengths
4. Proportionate norm updates as a complementary method
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def eval_sequence_safe(a):
    """Evaluate using the provided evaluate_sequence function."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    return float(evaluate_sequence(a_norm.tolist()))


def ternary_search_1d(conv, e, f, lo, hi, num_iters=30):
    """
    Find t in [lo, hi] minimizing max_k(conv_k + t*e_k + t^2*f_k).
    The function is convex, so ternary search works.
    """
    for _ in range(num_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        g1 = np.max(conv + m1 * e + m1 * m1 * f)
        g2 = np.max(conv + m2 * e + m2 * m2 * f)
        if g1 < g2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def exact_coordinate_descent(a, time_limit, verbose=False, restart_interval=50,
                             noise_scale=0.01, rng=None):
    """
    Exact coordinate descent for minimizing max(conv(a,a)) with a_i >= 0.
    The objective is scale-invariant, so we don't need to fix sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return a, float('inf')
    a = a / s  # Normalize for numerical stability

    if rng is None:
        rng = np.random.default_rng(42)

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    sweep = 0
    n_conv = 2 * n - 1

    # Precompute index arrays
    k_idx = np.arange(n_conv)

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        sweep_start = time.monotonic()
        conv = fast_convolve(a)
        prev_max = conv.max()
        total_improvement = 0.0

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-18:
                # Only can increase
                lo = 0.0
                hi = max(1e-6, a.max() * 0.1)
            else:
                lo = -a[i]
                hi = a[i] + max(1e-6, a.max() * 0.5)

            # Compute e and f for this coordinate
            k_minus_i = k_idx - i
            valid = (k_minus_i >= 0) & (k_minus_i < n)

            e = np.zeros(n_conv)
            f = np.zeros(n_conv)

            # Linear terms: for k where k-i is valid and k != 2i
            mask = valid & (k_idx != 2 * i)
            e[mask] = 2.0 * a[k_minus_i[mask]]

            # Quadratic term at k = 2i
            if 2 * i < n_conv:
                e[2 * i] = 2.0 * a[i]
                f[2 * i] = 1.0

            # Find optimal t
            t_opt = ternary_search_1d(conv, e, f, lo, hi, num_iters=25)

            # Apply update
            if abs(t_opt) > 1e-15:
                a[i] = max(0.0, a[i] + t_opt)
                # Update conv incrementally (faster than recomputing full conv)
                conv += t_opt * e + t_opt * t_opt * f

        # After full sweep, normalize and check score
        s = a.sum()
        if s < 1e-30:
            break
        a = a / s
        score = compute_score_normalized(a)

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            total_improvement = best_score - score

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep}, score={best_score:.10f}, elapsed={elapsed:.1f}s")

        # Check for convergence
        conv = fast_convolve(a)
        new_max = conv.max()
        if abs(new_max - prev_max) < 1e-12:
            # Stuck - add perturbation
            if sweep > 5:
                noise = rng.standard_normal(n) * noise_scale * a.max()
                a = np.maximum(a + noise, 0)
                a = a / a.sum()
                noise_scale *= 0.7
            else:
                noise_scale = max(0.001, noise_scale * 0.5)

        # Periodic restart with larger perturbation

            noise = rng.standard_normal(n) * 0.05 * a.max()
            a_pert = np.maximum(a + noise, 0)
            a_pert = a_pert / a_pert.sum()
            pert_score = compute_score_normalized(a_pert)
            if pert_score < best_score:
                a = a_pert
                best_score = pert_score
                best_a = a.copy()

    return best_a, best_score


def proportionate_norm(a0, time_limit, mu=0.1, verbose=False, rng=None):
    """
    Proportionate norm algorithm: multiplicative updates based on convolution values.
    a_i <- a_i * (1 - mu * (conv_{2i} - target) / target)
    This is inspired by RMA in array signal processing.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return a0, float('inf')
    a = a / s

    if rng is None:
        rng = np.random.default_rng(123)

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        target = (a.sum() ** 2) / (2 * n - 1)  # Average value
        max_conv = conv.max()

        # Proportionate update: reduce coordinates that contribute to high peaks
        for i in range(n):
            k_2i = 2 * i
            if k_2i < len(conv):
                excess = (conv[k_2i] - target) / (target + 1e-30)
                factor = 1.0 - mu * max(0, excess)
                factor = max(0.01, min(2.0, factor))
                a[i] *= factor

        # Also do a compensating increase on low-conv coordinates to maintain diversity
        low_mask = conv[2 * np.arange(n)] < target
        if low_mask.any():
            a[low_mask] *= (1.0 + mu * 0.5)

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-30:
            break
        a = a / s

        score = compute_score_normalized(a)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()

        if verbose and iters % 50 == 0:
            print(f"    PNA iter={iters}, score={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
    return best_seq, best_score
