    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    Returns the optimal g.

    objective = cp.Maximize(cp.sum(g))

    constraints = [g >= 0]

"""
LP-based fictitious play optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy HiGHS solver for the LP subproblem, spectral initialization for
good starting points, and coordinate descent for final polishing.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.ndimage import uniform_filter1d


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * fast_convolve(a).max() / (s * s)


def build_conv_indices(n):
    """Precompute row/col indices for the conv matrix A where A@g = f*g."""
    rows = []
    cols = []
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        rows.extend(range(j, k_end))
        cols.extend([j] * (k_end - j))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_step(f, M, row_idx, col_idx, n):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns optimal g or None if infeasible.
    """
    vals = f[col_idx]
    A = sp.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def spectral_init(n, iterations=50, seed=42):
    """Generate a sequence with relatively flat auto-correlation via spectral method."""
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum()
    
    N = 1 << (2 * n).bit_length()
    
    for _ in range(iterations):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        target = mag.mean()
        phase = np.angle(A)
        A_new = target * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s
        else:
            break
    
    return a


def lp_fictitious_play(f_init, time_limit, row_idx, col_idx, n,
                       alpha_init=0.3, alpha_min=0.01, verbose=True):
    """
    LP-based fictitious play to minimize max(conv(f,f)) subject to sum(f)=1, f>=0.
    """
    f = np.maximum(f_init.copy(), 0)
    s = f.sum()
    if s < 1e-15:
        return f_init
    f = f / s  # normalize to sum = 1
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    M = best_M
    alpha = alpha_init
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 30
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed = time.monotonic() - t0
        
        # Solve LP
        g = solve_lp_step(f, M, row_idx, col_idx, n)
        
        if g is None or g.sum() < 1e-10:
            # LP failed or trivial - try reducing M
            M *= 0.995
            no_improve += 5
            continue
        
        # Rescale g to sum = 1
        g = g / g.sum()
        
        # Check if g is meaningfully different from f
        diff = np.abs(g - f).max()
        
        # Update
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f
        else:
            f_new = f_new / ss
        
        # Evaluate
        M_new = fast_convolve(f_new).max()
        
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update M: use current max but allow slight reduction
        M = max(M_new, M * 0.998)
        
        # Adaptive alpha
        if no_improve >= max_no_improve:
            alpha = max(alpha_min, alpha * 0.8)
            no_improve = 0
            if alpha < alpha_min * 1.1:
                # Reset to best and restart
                f = best_f.copy()
                M = best_M
                alpha = alpha_init * 0.5
                no_improve = 0
        
        f = f_new
        
        if verbose and (iteration % 20 == 1 or best_M < 0.752401):
            score = 2.0 * n * best_M
            print(f"  FP iter={iteration} t={elapsed:.0f}s "
                  f"M={M:.6f} best_M={best_M:.6f} score={score:.6f} "
                  f"alpha={alpha:.4f} diff={diff:.6f}")
    
    return best_f


def coordinate_descent_polish(a0, time_limit, n_sweeps=50):
    """Coordinate descent with multiplicative updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = compute_score(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a, best_score


def peak_shave_polish(a0, time_limit, n_trials=2000):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
LP-based fictitious play optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy HiGHS solver for the LP subproblem, spectral initialization for good
starting points, multi-start diversity, and coordinate descent for final polishing.
Fundamentally different from the previous Adam-based approach: leverages the
linear-program structure of the active-set method from the state-of-the-art
description to take principled large steps toward optimal solutions.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * fast_convolve(a).max() / (s * s)


def build_conv_indices(n):
    """Precompute row/col indices for the matrix A where A@g = f*g (linear conv)."""
    rows = []
    cols = []
    for j in range(n):
        k_end = j + n
        if k_end > 2 * n - 1:
            k_end = 2 * n - 1
        rows.extend(range(j, k_end))
        cols.extend([j] * (k_end - j))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_step(f, M, row_idx, col_idx, n):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k=0..2n-2.
    Returns optimal g vector or None.
    """
    vals = f[col_idx]
    A = sp.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs',
                      options={'time_limit': 10.0, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def spectral_init(n, iterations=80, seed=42):
    """Generate a sequence with relatively flat auto-correlation via spectral flattening."""
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n)) + 0.05
    a = a / a.sum()

    N = 1 << (2 * n).bit_length()

    for it in range(iterations):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        target = mag.mean()
        phase = np.angle(A)
        A_new = target * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s
        else:
            break

    return a


def lp_fictitious_play(f_init, time_limit, row_idx, col_idx, n,
                       alpha_init=0.3, alpha_min=0.005, verbose=True):
    """
    LP-based fictitious play to minimize max(conv(f,f)) subject to sum(f)=1, f>=0.
    Uses the active-set LP from the state-of-the-art description.
    """
    f = np.maximum(np.asarray(f_init, dtype=np.float64), 0)
    s = f.sum()
    if s < 1e-15:
        return f_init
    f = f / s

    best_f = f.copy()
    best_M = fast_convolve(f).max()
    M = best_M
    alpha = alpha_init
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 40

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed = time.monotonic() - t0

        # Solve LP
        g = solve_lp_step(f, M, row_idx, col_idx, n)

        if g is None or g.sum() < 1e-10:
            M *= 0.99
            no_improve += 10
            continue

        # Rescale g to sum = 1
        g = g / g.sum()

        # Measure difference
        diff = np.abs(g - f).max()

        # Update: convex combination
        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / ss

        # Evaluate
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt M: track current peak but allow gentle reduction
        M = max(M_new, M * 0.997)

        # Adaptive step size
        if no_improve >= max_no_improve:
            alpha = max(alpha_min, alpha * 0.7)
            no_improve = 0
            if alpha <= alpha_min * 1.2:
                # Restart from best
                f = best_f.copy()
                M = best_M
                alpha = alpha_init * 0.5
                no_improve = 0

        f = f_new

        if verbose:
            if iteration % 10 == 1 or (best_M < 0.751 and iteration % 5 == 0):
                score = 2.0 * n * best_M
                print(f"    FP n={n} iter={iteration} t={elapsed:.0f}s "
                      f"M={M:.6f} best_M={best_M:.6f} score={score:.6f} "
                      f"alpha={alpha:.4f} diff={diff:.5f}")

    return best_f


def lp_fictitious_play_minimize_m(f_init, time_limit, row_idx, col_idx, n,
                                   alpha_init=0.2, verbose=True):
    """
    Variant: directly minimize M by targeting a decreasing M schedule.
    At each step, set M_target slightly below current best, solve LP with that M.
    """
    f = np.maximum(np.asarray(f_init, dtype=np.float64), 0)
    s = f.sum()
    if s < 1e-15:
        return f_init
    f = f / s

    best_f = f.copy()
    best_M = fast_convolve(f).max()
    M = best_M
    alpha = alpha_init
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed = time.monotonic() - t0

        # Target: slightly below current M
        M_target = M * (1.0 - 0.002)

        g = solve_lp_step(f, M_target, row_idx, col_idx, n)

        if g is None or g.sum() < 1e-10:
            # Can't achieve the target, relax slightly
            M_target = M * 0.999
            g = solve_lp_step(f, M_target, row_idx, col_idx, n)
            if g is None or g.sum() < 1e-10:
                M = M_target  # accept the relaxation
                no_improve += 5
                continue

        g = g / g.sum()
        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / ss

        M_new = fast_convolve(f_new).max()

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        M = max(M_new, M_target)

        if no_improve >= 30:
            alpha = max(0.003, alpha * 0.8)
            no_improve = 0
            if alpha < 0.004:
                f = best_f.copy()
                M = best_M
                alpha = alpha_init * 0.5

        f = f_new

        if verbose and iteration % 15 == 1:
            score = 2.0 * n * best_M
            print(f"    LPmin n={n} iter={iteration} t={elapsed:.0f}s "
                  f"M={M:.6f} best_M={best_M:.6f} score={score:.6f} alpha={alpha:.4f}")

    return best_f


def coordinate_descent_polish(a0, time_limit, n_sweeps=40):
    """Coordinate descent with multiplicative updates for fine polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            for factor in [0.88, 0.93, 0.97, 1.03, 1.07, 1.12]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-14:
                    continue
                a_trial = a_trial / s
                score = compute_score(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                    improved = True
                    break

        if not improved:
            break

    return best_a, best_score


def peak_shave_polish(a0, time_limit, n_trials=3000):
    """Targeted peak reduction by adjusting element pairs contributing to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        peak_k = int(np.argmax(b_search))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_i = np.argsort(contribs)[::-1][:4]

        improved = False
        for ci in top_i:
            i = idx[ci]
            j = int(peak_k - i)
            if i == j or a[i] < 1e-14 or a[j] < 1e-14:
                continue

            for f1 in [0.92, 0.87, 0.82, 0.75]:
                for f2 in [1.02, 1.05, 1.08, 1.12]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-14:
                        continue
                    a_trial = a_trial / s
                    score = compute_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and trial > 100 and trial % 80 == 0:
            a = best_a + rng.standard_normal(n) * 0.008
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
