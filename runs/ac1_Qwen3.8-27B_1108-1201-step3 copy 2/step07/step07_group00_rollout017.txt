Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses cutting-plane LP, IAA diversification, and adaptive multi-phase search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_direction(f, g):
    """Compute f*g (cross-convolution) via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_constraint_matrix(f, indices, n):
    """Build the constraint matrix A where row k has f values for conv constraint k."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A

def solve_lp_direction(f, n, M, active_idx, eps=1e-8):
    """Solve the LP direction-finding problem."""
    K = len(active_idx)
    if K == 0:
        return None
    
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception as e:
        # Try ECOS for smaller problems
        if n <= 2000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=10000)
            except Exception:
                return None
        else:
            return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

                S_g = g.sum()
        
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                break
            continue
        
        # Line search (golden section)
        h = g / S_g
        cur_obj = M
        best_t = 0.05
        best_t_obj = cur_obj
        
        # Quick scan
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        # Refine around best_t
        if best_t > 0:
            lo = max(0, best_t * 0.5)
            hi = min(1.0, best_t * 2.0)
            for _ in range(5):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                f1 = np.maximum((1-m1)*f + m1*h, 0); s1 = f1.sum()
                f1 /= s1 if s1 > 1e-10 else 1
                f2 = np.maximum((1-m2)*f + m2*h, 0); s2 = f2.sum()
                f2 /= s2 if s2 > 1e-10 else 1
                o1 = fast_convolve(f1).max()
                o2 = fast_convolve(f2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2
            f_mid = np.maximum((1-t_mid)*f + t_mid*h, 0)
            s_mid = f_mid.sum()
            f_mid /= s_mid if s_mid > 1e-10 else 1
            o_mid = fast_convolve(f_mid).max()
            if o_mid < best_t_obj:
                best_t_obj = o_mid
                best_t = t_mid
        
        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n
        
        cur_obj = fast_convolve(f).max()
        M = cur_obj
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break
        
        # Update active set periodically
        if it % 20 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k:
                # Keep only the most important constraints
                active_scores = b[active_idx]
                # Also include recently violated ones

                active_idx = np.unique(np.concatenate([active_idx, top_new]))
                # Limit size
                if len(active_idx) > max_k:
                    active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]
        
        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} t={elapsed:.0f}s")
    
    return best_f, best_obj

def iaa_iterate(f_init, n_iters=200, flat_level=0.0, clip_neg=True):
    """
    Iterative Amplitude Adjustment: try to flatten the auto-convolution sidelobes.
    """
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(n_iters):
        # Compute auto-convolution
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        
        # Desired spectrum: keep peak, flatten rest
        R_desired = R.copy()
        peak_idx = R.argmax()
        peak_val = R[peak_idx]
        # Set sidelobes to flat_level (relative to peak)
        R_desired[peak_idx] = peak_val
        mask = np.ones(size, dtype=bool)
        # Zero out the main lobe region (around the peak)
        main_lobe_half = max(3, n // 10)
        lobe_start = max(0, peak_idx - main_lobe_half)
        lobe_end = min(size - 1, peak_idx + main_lobe_half)
        R_desired[lobe_start:lobe_end+1] = 0  # Let the main lobe be free
        # Actually, for auto-convolution of a non-negative sequence, the peak is at the edge
        # The "main lobe" is the entire thing... let me just target the peak
        
        # Better: target = peak for the max, flat_level * peak for others
        R_target = np.full(size, flat_level * peak_val)
        R_target[peak_idx] = peak_val
        
        # Update amplitude
        R_fft = rfft(R_target, fft_size)
        # Take square root (amplitude)
        new_amp = np.sqrt(np.maximum(R_fft.real, 0))
        
        # But R_fft might be complex... For real autocorrelation, the FFT is real and non-negative
        # if the autocorrelation is PSD. But our R_target might not be PSD.
        # Let's just use the magnitude
        new_F = np.sqrt(np.maximum(R_fft, 0)) * (F / np.abs(F + 1e-30)) * np.abs(F + 1e-30)
        # This is getting complicated. Let me use a simpler approach.
        
        # Simpler: new |F|^2 = R_target in frequency domain
        # F_new = sqrt(R_target_fft) * phase(F)
        R_t_fft = rfft(R_target, fft_size)
        # R_target should give a real, non-negative FFT if it's a valid autocorrelation
        # But we're setting it artificially, so it might not be
        # Just take the real part and clip
        R_t_fft_real = np.maximum(R_t_fft.real, 0)
        new_amp = np.sqrt(R_t_fft_real)
        phase = F / (np.abs(F) + 1e-30)
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        
        if clip_neg:
            f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n
    
    return f

def run_subgradient_phase(f_init, time_limit, lr_init=0.003, n_top=50, temp_init=0.0005):
    """Subgradient optimization with momentum and adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    momentum = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Soft-max weights on peaks
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Focus on top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w = np.zeros(len(b))
            w[top_idx] = w_sel
        
        # Compute subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            # grad b[k] w.r.t. f[i] = 2*f[k-i] for valid i
            # So contribution to g[j] from peak k is w[k] * 2 * f[k-j]
            g[idx] += w[k] * 2.0 * f[k - idx]
        
        v = momentum * v - lr * g
        f_new = project_simplex(f + v)
        f = f_new
        
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.9
                    temp *= 1.1
                    no_improve = 0
                if no_improve > 30:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                    v = np.zeros(n)
                    no_improve = 0
            
            if it % 500 == 0:
                print(f"    sg iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj

def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(0, n, max(1, n // 500)):  # Subsample if n is large
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in [0.7, 0.85, 1.15, 1.3]:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                best_obj = best_j_obj
                improved = True
    return f, best_obj

"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses cutting-plane LP (iteratively adding violated constraints), IAA-based
diversification, adaptive multi-length exploration, and subgradient polish.
Key improvement over previous: the LP constraints are not fixed to top-K peaks
but are grown adaptively via a cutting-plane procedure, giving a much tighter
feasible region and better direction finding.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_direction(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix A where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, eps=1e-8):
    """Solve: max sum(g) s.t. g>=0, A@g <= M."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        g_val = None
    if g.value is None:
        # Try ECOS for smaller problems
        if n <= 2000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=20000)
            except Exception:
                return None
        else:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def lp_with_cutting_planes(f_init, time_limit, initial_k=40, max_k=400, label="cp"):
    """
    LP direction finding with cutting planes.
    Iteratively adds violated constraints to the active set.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    # Initialize active set
    if initial_k < len(b):
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Solve LP
        g = solve_lp_direction(f, n, M, active_idx, eps=1e-8)

        if g is None:
            stuck += 1
            if stuck > 8:
                break
            if len(active_idx) > initial_k:
                active_idx = active_idx[np.argsort(b[active_idx])[-initial_k:]]
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                break
            # Try expanding active set
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue

        # Cutting plane: check for violated constraints
        fg = convolve_direction(f, g)
        violated_mask = fg > M * (1 + 1e-7)
        v_count = violated_mask.sum()

        if v_count > 0 and len(active_idx) < max_k:
            v_idx = np.where(violated_mask)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(30, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                # Re-solve with updated constraints
                g = solve_lp_direction(f, n, M, active_idx, eps=1e-8)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()

        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                break
            continue

        stuck = 0
        h = g / S_g

        # Line search: scan + refine
        best_t = 0.05
        best_t_obj = M

        scan_steps = [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
        for t in scan_steps:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden-section refinement around best_t
        if best_t > 0.01:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(6):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        # Update f
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        # Periodically refresh active set
        if it % 15 == 0:
            b = fast_convolve(f)
            # Keep active set focused on current peaks
            if len(active_idx) > initial_k * 2:
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k * 2:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]

        if it % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} stuck={stuck} t={elapsed:.0f}s")

    return best_f, best_obj


def iaa_diversify(f_init, n_iters=200, peak_fraction=0.3):
    """
    IAA-based diversification: iteratively try to flatten the auto-convolution.
    Returns a diverse but valid (non-negative, normalized) sequence.
    """
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]

        # Target: reduce peak while preserving energy
        peak_val = R.max()
        peak_idx = R.argmax()
        # Create target: keep peak, reduce others toward a floor
        floor = peak_fraction * peak_val
        R_target = np.clip(R, floor, peak_val)
        # Preserve total energy (sum of R = sum(f)^2 = 1 for normalized f)
        energy = R.sum()
        if R_target.sum() > 1e-10:
            R_target *= energy / R_target.sum()

        # Update: new |F| from target R
        R_t_fft = rfft(R_target, fft_size)
        # R_target should be close to a valid autocorrelation, so FFT should be ~real non-negative
        real_part = np.maximum(R_t_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        # Preserve phase
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = f_init.copy()
            break

    return f


def run_subgradient(f_init, time_limit, lr_init=0.002, n_top=50, temp_init=0.001):
    """Subgradient method with momentum and adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99)
    momentum = 0.85

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Soft-max weights
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w = np.zeros(len(b))
            w[top_idx] = w_sel

        # Subgradient: grad F(f) = weighted sum of 2*f[k-j] contributions
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * f[k - idx]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 80 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 8:
                    lr *= 0.88
                    temp *= 1.12
                    no_improve = 0
                if no_improve > 40:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                    v = np.zeros(n)
                    no_improve = 0

            if it % 500 == 0:
                print(f"    sg it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 400)

    for sweep in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.75, 0.9, 1.1, 1.25]:
                nv = f[j] * mult
                f[j] = nv
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = nv
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break
    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining cutting-plane LP,
    IAA diversification, multi-length exploration, and subgradient polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # === Phase 1: Cutting-plane LP on parent (main optimization) ===
    print(f"\n=== Phase 1: CP-LP (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.42, 700.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = lp_with_cutting_planes(parent, time_limit=t1, initial_k=40, max_k=350, label="cp1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [cp1]: {score1:.6f}")

    # === Phase 2: Continued CP-LP with more constraints ===
    print(f"\n=== Phase 2: CP-LP deep (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.22, 300.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t2:.0f}s, initial_k=80, max_k=500")
        f2, obj2 = lp_with_cutting_planes(best_seq, time_limit=t2, initial_k=80, max_k=500, label="cp2")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [cp2]: {score2:.6f}")

    # === Phase 3: IAA diversification + LP polish ===
    print(f"\n=== Phase 3: IAA + LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3_total = min(rem * 0.18, 200.0)
    if t3_total > 30:
        n_i = len(best_seq)
        # IAA variant 1
        f_i1 = iaa_diversify(best_seq, n_iters=150, peak_fraction=0.2)
        t3a = t3_total * 0.5
        f3a, _ = lp_with_cutting_planes(f_i1, time_limit=t3a, initial_k=30, max_k=200, label="iaa1")
        s3a = float(evaluate_sequence(f3a.tolist()))
        if np.isfinite(s3a) and s3a < best_score:
            best_score = s3a
            best_seq = f3a.copy()
            print(f"  NEW BEST [iaa1]: {s3a:.6f}")

        # IAA variant 2
        rem3 = deadline - time.monotonic()
        if rem3 > 20:
            f_i2 = iaa_diversify(best_seq, n_iters=200, peak_fraction=0.4)
            t3b = min(rem3 * 0.4, 80.0)
            f3b, _ = lp_with_cutting_planes(f_i2, time_limit=t3b, initial_k=30, max_k=200, label="iaa2")
            s3b = float(evaluate_sequence(f3b.tolist()))
            if np.isfinite(s3b) and s3b < best_score:
                best_score = s3b
                best_seq = f3b.copy()
                print(f"  NEW BEST [iaa2]: {s3b:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = set()
    for factor in [0.85, 0.95, 1.05, 1.15, 1.3]:
        tl = int(n_base * factor)
        tl = max(512, min(2500, tl))
        if tl != n_base:
            target_lengths.add(tl)
    target_lengths = sorted(target_lengths, key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t4 = min(rem * 0.14, 120.0)
        if t4 < 20:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  CP-LP n={target_n}, budget={t4:.0f}s")
        f4, _ = lp_with_cutting_planes(f_interp, time_limit=t4, initial_k=30, max_k=200, label=f"n{target_n}")

        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {s4:.6f}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 120.0)

        f5, _ = run_subgradient(best_seq, time_limit=t5, lr_init=0.002,
                                n_top=max(30, n_b // 12))
        s5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg]: {s5:.6f}")

    # === Phase 6: Final LP burst on best ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 150.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}, k=100->500")
        f6, _ = lp_with_cutting_planes(best_seq, time_limit=t6, initial_k=100, max_k=500, label="final")
        s6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(s6) and s6 < best_score:
            best_score = s6
            best_seq = f6.copy()
            print(f"  NEW BEST [final]: {s6:.6f}")

    # === Phase 7: Coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 10:
        print(f"\n=== Phase 7: Coord descent (best={best_score:.6f}) ===")
        t7 = min(rem * 0.7, 60.0)
        f7, _ = coordinate_polish(best_seq, time_limit=t7)
        s7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {s7:.6f}")

    # Final
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
