                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    
    # Get dual variables for the inequality constraints
    # The dual variables correspond to T @ b <= M
    # In cvxpy, we can access them via prob.constraints[k].dual_value
    dual_vals = np.zeros(num_constr)
    for i, con in enumerate(prob.constraints):
        if isinstance(con, cp.constraints.Inequality) or 'Inequality' in type(con).__name__:
            dv = con.dual_value
            if dv is not None:
                dual_vals[i] = abs(np.asarray(dv).ravel()[0])
    
    # Normalize dual weights
    ds = dual_vals.sum()
    if ds > 1e-15:
        dual_weights = dual_vals / ds
    else:
        # Fallback: use the peak of conv(a,b)
        ab = fast_convolve(a) * 0 + fast_convolve(np.ones(n)/n)  # placeholder
        # Actually use the active set of the convolution
        conv_ab = T @ b_val
        max_val = conv_ab.max()
        active = conv_ab >= max_val - 1e-6
        dual_weights = np.zeros(num_constr)
        dual_weights[active] = 1.0 / active.sum()
    
    M_val = float(M_var.value)
    return b_val, M_val, dual_weights

def trust_region_lp_step(a, active_indices, step_bound=0.5, solver=None):
    """Take one trust-region LP step.
    Linearize the quadratic at a, solve LP for the step direction.
    Returns new a (probability vector)."""
    if solver is None:
        solver = _lp_solver_cached
    n = len(a)
    b = fast_convolve(a)
    
    # LP: minimize t
    # s.t. b[k] + 2*conv(a,δ)[k] <= t for k in active_indices
    #      sum(δ) = 0
    #      -step_bound*a[i] <= δ[i] <= step_bound*a[i]
    
    delta = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        sum(delta) == 0,
        delta >= -step_bound * a,
        delta <= step_bound * a,
    ]
    
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        js = np.arange(j_lo, j_hi + 1)
        coeffs = a[k - js]
        # conv(a, δ)[k] = sum_{j in range} a[k-j] * δ[j]
        conv_linear = coeffs @ delta[js]
        constraints.append(b[k] + 2 * conv_linear <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    delta_val = np.asarray(delta.value, dtype=np.float64)
    delta_val = np.clip(delta_val, -step_bound * a, step_bound * a)
    
    a_new = a + delta_val
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    
    return a_new

def nesterov_subgradient(a0, time_limit, solver=None, n_max_iters=10000):
    """Nesterov accelerated subgradient method with LP-dual weights."""
    if solver is None:
        solver = _lp_solver_cached
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Nesterov parameters
    t_k = 1.0
    x_prev = a.copy()
    y = a.copy()  # lookahead point
    
    no_improve = 0
    alpha = 1.0  # step size parameter
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Evaluate at y
        y_conv = fast_convolve(y)
        max_y = y_conv.max()
        
        # Find active set (near-peak indices)
        threshold = max_y - 0.01 * max_y
        active = np.where(y_conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(y_conv)])
        if len(active) > 50:
            active = np.argsort(y_conv)[::-1][:50]
        
        # Compute subgradient using active set (uniform weights as approximation)
        g = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            js = np.arange(j_lo, j_hi + 1)
            g[js] += y[k - js]
        g /= len(active)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        
        y = a + beta * (a - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        # Step size (diminishing)
        step = alpha / np.sqrt(it + 1)
        
        # Subgradient step with projection to simplex
        x_new = y - step * g
        x_new = simplex_project(x_new)
        
        x_prev = a.copy()
        a = x_new
        t_k = t_new
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if min(score, score_y) < best_score - 1e-14:
            best_score = min(score, score_y)
            best_a = a.copy() if score < score_y else y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve % 500 == 0 and no_improve > 0:
            alpha *= 0.9
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

def lbgfs_soft_max(a0, time_limit, T_start=100.0, T_end=5.0, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]
    
    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                j_lo = max(0, k - n + 1)
                j_hi = min(k, n - 1)
                js = np.arange(j_lo, j_hi + 1)
                grad_a[js] += 2.0 * w[k] * a[k - js]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': iters_per_stage * 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = softmax_from_z(z)
    return a_out

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

The trust-region LP:
- We're at point a (probability vector)
- b = conv(a, a)
- Active set A = indices where b is near max
- We want to find δ such that a_new = a + δ is a probability vector and reduces the peak

Linearization: conv(a+δ, a+δ)[k] ≈ b[k] + 2 * conv(a, δ)[k] for small δ.

conv(a, δ)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * δ[k-i]

Let me define: for a fixed k, the "convolution coefficient vector" c_k has c_k[j] = a[k-j] for j in [max(0, k-n+1), min(k, n-1)], and 0 otherwise. Then conv(a, δ)[k] = c_k · δ.

So the LP is:
  minimize t
  s.t. b[k] + 2 * c_k · δ <= t  for k in A
       sum(δ) = 0
       -r*a[i] <= δ[i] <= r*a[i]  for all i (trust region)

where r is the step bound.

After solving, we get δ*. We check if the actual improvement is close to the predicted. If not, reduce r.

This should be a clean, correct approach. Let me also add the "examine the peaks at the boundaries" option - sometimes the peak is at k=0 or k=2n-2, and we should include those.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Trust-region LP method (linearized quadratic with active set tracking)
2. Nesterov accelerated subgradient with adaptive step sizes
3. L-BFGS soft-max polishing with geometric temperature schedule
4. Peak-shaving coordinate descent
5. Multi-start with structured perturbations
Key innovation: Trust-region LP that linearizes the auto-convolution peak
and solves a small LP per iteration for the optimal local step direction.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.SCS
    return cp.SCS


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_subgradient(a, active_indices):
    """Compute subgradient of max_k (a*a)[k] at point a using active set."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        js = np.arange(j_lo, j_hi + 1)
        g[js] += a[k - js]
    g /= len(active_indices)
    return g


def trust_region_lp_step(a, active_indices, step_bound, solver):
    """Take one trust-region LP step.
    Linearize the quadratic at a, solve LP for optimal step direction.
    Returns new a (probability vector)."""
    n = len(a)
    b = fast_convolve(a)

    delta = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        cp.sum(delta) == 0,
        delta >= -step_bound * a,
        delta <= step_bound * a,
    ]

    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        js = np.arange(j_lo, j_hi + 1)
        coeffs = a[k - js]
        conv_linear = coeffs @ delta[js]
        constraints.append(b[k] + 2.0 * conv_linear <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    delta_val = np.asarray(delta.value, dtype=np.float64)
    if delta_val is None:
        return a.copy()
    delta_val = np.clip(delta_val, -step_bound * a, step_bound * a)

    a_new = a + delta_val
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s

    return a_new


def trust_region_loop(a0, time_limit, solver, n_max_iters, active_set_size):
    """Iterative trust-region LP method with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    step_bound = 0.4
    no_improve = 0
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b - 0.08 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) > active_set_size:
            active = np.argsort(b)[::-1][:active_set_size]
        elif len(active) == 0:
            active = np.array([np.argmax(b)])

        a_new = trust_region_lp_step(a, active, step_bound, solver)

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            step_bound *= 0.92
            if step_bound < 0.02:
                # Perturb to escape
                if no_improve > 100:
                    pert = rng.normal(0, 0.01, n)
                    a_pert = a + pert
                    a_pert = np.maximum(a_pert, 0)
                    s = a_pert.sum()
                    if s > 1e-15:
                        a = a_pert / s
                        step_bound = 0.2
                        no_improve = 0
                else:
                    break

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TR-LP iter={it+1} ({elapsed:.0f}s): best={best_score:.6f} step={step_bound:.4f}")

    return best_a, best_score


def nesterov_subgradient(a0, time_limit, n_max_iters):
    """Nesterov accelerated subgradient method on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    t_k = 1.0
    x_prev = a.copy()
    y = a.copy()
    alpha = 1.0
    no_improve = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        y_conv = fast_convolve(y)
        max_y = y_conv.max()

        threshold = max_y - 0.02 * max_y
        active = np.where(y_conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(y_conv)])
        if len(active) > 40:
            active = np.argsort(y_conv)[::-1][:40]

        g = compute_subgradient(y, active)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new

        y = a + beta * (a - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        step = alpha / np.sqrt(it + 1)
        x_new = y - step * g
        x_new = simplex_project(x_new)

        x_prev = a.copy()
        a = x_new
        t_k = t_new

        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if min(score, score_y) < best_score - 1e-14:
            best_score = min(score, score_y)
            best_a = a.copy() if score < score_y else y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve == 500:
            alpha *= 0.85
            no_improve = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG iter={it+1} ({elapsed:.0f}s): best={best_score:.6f} alpha={alpha:.4f}")

    return best_a, best_score


def lbgfs_soft_max(a0, time_limit, T_start, T_end, maxiter):
    """L-BFGS refinement with soft-max smoothing and geometric temperature."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                j_lo = max(0, k - n + 1)
                j_hi = min(k, n - 1)
                js = np.arange(j_lo, j_hi + 1)
                grad_a[js] += 2.0 * w[k] * a[k - js]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': iters_per_stage * 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out

                            if score < best_score - 1e-14:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lp_best_response(a_fixed, solver):
    """Given fixed a (probability vector), find b minimizing max_k conv(a,b)[k].
    Returns (b_val, M_val)."""
    n = len(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray
