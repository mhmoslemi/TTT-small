Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced multi-phase approach:
1. LogSumExp L-BFGS with very fine geometric temperature annealing
2. Exact subgradient mirror descent with Polyak step sizes
3. Enhanced peak-shaving using all near-max peaks
4. Multi-length exploration with interpolation
5. Simulated annealing restarts for escape from local minima
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def log_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def objective_and_grad_logdomain(z, T):
    """
    Compute LogSumExp objective and gradient in log-space.
    Objective: (2n/T) * log(sum_k exp(T * conv(a,a)[k]))
    where a = softmax(z).
    """
    n = len(z)
    a = softmax_fn(z)
    b = fast_autoconv(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Gradient w.r.t. a
    w = exp_Tb / exp_Tb.sum()  # weights over k
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_finemesh(a0, time_limit, T_start, T_end, n_stages=10, maxiter_per_stage=2000):
    """
    L-BFGS with fine geometric temperature mesh.
    """
    n = len(a0)
    z = log_from_a(a0)
    t0 = time.monotonic()
    best_obj = (2.0 * n) * fast_autoconv(a0).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iter_count = 0
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Scale maxiter based on remaining time
        est_time_per_iter = max(0.001, (time.monotonic() - t0) / max(iter_count, 1))
        maxiter = min(maxiter_per_stage, max(100, int(remaining / (est_time_per_iter * 5))))
        
        def func(z_in):
            return objective_and_grad_logdomain(z_in, stage_T)[0]
        
        def jac(z_in):
            return objective_and_grad_logdomain(z_in, stage_T)[1]
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': maxiter * 100, 'disp': False}
            )
            z = res.x
            iter_count += maxiter
        except Exception:
            pass
        
        a_curr = softmax_fn(z)
        curr_obj = 2.0 * n * fast_autoconv(a_curr).max()
        if curr_obj < best_obj:
            best_obj = curr_obj
    
    return softmax_fn(z)

"""
Minimizes 2*n*max(auto-conv(a))/(sum(a))^2 using a multi-phase strategy:
1. Fine-mesh LogSumExp L-BFGS optimization with 20 geometric temperature stages
2. Mirror descent (exponentiated gradient) with Polyak step sizes
3. LP-based best response with Nesterov acceleration
4. Multi-length exploration with interpolation
5. Final peak shaving and micro-optimization
Key improvements over prior: finer T-mesh, Polyak steps, perturbation restarts,
and smarter peak identification.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective value for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autoconv(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_fn(z):
    """Softmax with numerical stability."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def log_from_a(a):
    """Log-domain representation of a probability vector."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_and_grad(z, T):
    """
    LogSumExp smoothed objective and gradient in log-space.
    f(z) = (2n/T) * log(sum_k exp(T * (a*a)_k)) where a = softmax(z)
    """
    n = len(z)
    a = softmax_fn(z)
    b = fast_autoconv(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * wk * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_fine_mesh(a0, time_limit, T_start=500.0, T_end=0.5, n_stages=20):
    """
    L-BFGS optimization with fine geometric temperature mesh.
    Returns the best sequence found.
    """
    n = len(a0)
    z = log_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Estimate iteration budget
        maxiter = min(3000, max(200, int(remaining * 0.7 / (n_stages - si))))
        
        def func(z_in):
            return lse_obj_and_grad(z_in, stage_T)[0]
        
        def jac(z_in):
            return lse_obj_and_grad(z_in, stage_T)[1]
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': maxiter * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        a_curr = softmax_fn(z)
        sc = compute_score(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
    
    return best_a


def exact_subgradient(a, epsilon_frac=0.05):
    """
    Compute the exact subgradient of max_k conv(a,a)[k].
    Returns (max_val, subgradient) where subgradient is the average
    of gradients of all active (near-max) components.
    """
    n = len(a)
    b = fast_autoconv(a)
    max_b = b.max()
    
    # Find active indices (within epsilon_frac of max)
    threshold = max_b * (1 - epsilon_frac)
    active = np.where(b >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    # Compute average subgradient
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(active)
    
    return max_b, g


def mirror_descent(a0, time_limit, lr_init=0.1, polyak_lb=None):
    """
    Mirror descent (exponentiated gradient) on the simplex.
    Uses exact subgradient with adaptive/Polyak step size.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    z = log_from_a(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    # Polyak lower bound: 2n/(2n-1) for normalized sum=1
    if polyak_lb is None:
        polyak_lb = 2.0 * n / (2.0 * n - 1)
    
    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        max_b, g = exact_subgradient(a, epsilon_frac=0.1)
        score = 2.0 * n * max_b  # since sum = 1
        
        # Polyak step size
        gap = score - polyak_lb
        grad_norm_sq = np.dot(g, g)
        if grad_norm_sq < 1e-30:
            break
        polyak_lr = gap / (2.0 * grad_norm_sq)
        
        # Use min of adaptive and Polyak
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 1e-8)
        
        # Mirror descent step in log-space
        z_new = z - alpha * g
        a_new = softmax_fn(z_new)
        
        sc_new = compute_score(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1
            # Half the learning rate
            if no_improve % 50 == 0:
                lr *= 0.7
                if lr < 1e-8:
                    break
        
        # Occasionally print
        if step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD step={step} ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.6f}")
    
    return best_a

def lp_best_response(a_fixed, time_per_solve=5.0):
    """
    Solve the LP: minimize M subject to conv(a_fixed, b)[k] <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt) or (None, None) on failure.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraints: T @ b <= M for each k
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    constraints = [cp.sum(b_var) == 1]
    
    # Build the convolution matrix efficiently
    # conv(a_fixed, b)[k] = sum_j a_fixed[k-j] * b[j] for valid j
    # = sum_j T[k,j] * b[j]
    # where T[k,j] = a_fixed[k-j] if 0<=k-j<n
    
    # Use a sparse matrix or just build the dense constraints
    # For n=768, 2n-1=1535 constraints, each with at most n=768 nonzeros
    # Total nonzeros: ~n*(2n-1)/2 ≈ 590K, which is fine for the solvers
    
    # Build T as a 2D array
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints.append(T @ b_var <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    # Try solvers in order
    for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    
    return None, None


def nesterov_lp(a0, time_limit, solver_pref=None):
    """
    Nesterov-accelerated LP-based best response iteration.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    y = a.copy()  # virtual iterate
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        b_new, M_b = lp_best_response(y)
        if b_new is None:
            break
        
        # Nesterov step
        alpha = 2.0 / (it + 2)
        
        # Update virtual iterate
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        # Update actual iterate (weighted average)
        weight_new = alpha * (it + 2)
        weight_old = it + 1
        a = (weight_old * a + weight_new * b_new) / (weight_old + weight_new)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Evaluate
        sc_a = compute_score(a)
        sc_y = compute_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        it += 1
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
