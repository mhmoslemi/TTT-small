"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_weights(b, T):
    """Compute softmax weights over convolution values."""
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    return exp_scaled / exp_scaled.sum()

def grad_from_weights(a, w):
    """Compute gradient 2*n * cross_corr(w, a) / (sum(a))^2.
    Since we normalize sum(a)=1, it's just 2*n*cross_corr(w,a)."""
    n = len(a)
    fft_size = 1 << (max(3*n-1, 64)).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    return 2.0 * n * correl[:n]

def subgradient_descent(a_init, time_limit, T=0.01, eta_start=0.01, eta_end=0.001,
                        decay=0.9995, label="SGD", verbose=True):
    """Subgradient descent with soft-max smoothing and projection."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = eta_start
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
        
        # Softmax weights
        w = softmax_weights(b, T)
        grad = grad_from_weights(a, w)
        
        # Projection onto simplex
        a_new = project_simplex(a - eta * grad, 1.0)
        if a_new.sum() < 1e-10:
            a_new = best_a.copy()
        else:
            a = a_new / a_new.sum()
        
        eta *= decay
        
        if verbose and it % 5000 == 0:
            print(f"  {label} iter={it}: score={cur_score:.7f} best={best_score:.7f} eta={eta:.6f}")
    
    return best_a, best_score

def lp_direction(a, M_mult=1.0):
    """Solve LP to find a direction g that keeps a*g bounded."""
    n = len(a)
    M = a @ a  # This is conv(a,a)[0]... wait no.
    # Actually M should be max(conv(a,a))
    b_conv = fast_convolve(a)
    M = M_mult * b_conv.max()
    
    # LP: max sum(g) s.t. conv(a, g)[k] <= M for all k, g >= 0, sum(g) <= 1
    # conv(a,g)[k] = sum_i a[i]*g[k-i]
    # This is linear in g.
    
    # Build constraint matrix A_ub, b_ub
    # For each k in 0..2n-2: sum_i a[i]*g[k-i] <= M
    # g[k-i] appears with coefficient a[i], so g[j] appears with coefficient a[k-j] (when k-j is valid)
    
    # Actually, let's think of it as: for each k, the constraint is:
    # sum_{j: k-n+1 <= j <= min(k, n-1)} a[k-j] * g[j] <= M
    # So the coefficient of g[j] in constraint k is a[k-j] (if valid).
    
    # This gives a (2n-1) x n matrix A_ub.
    # But building this explicitly for large n might be slow.
    # For n=768, it's 1535 x 768, which is about 1.2M entries. Manageable.
    
    # Actually, many entries are zero. Let's build it sparsely.
    from scipy.sparse import lil_matrix, csr_matrix
    
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    A = A.tocsr()
    b_ub = np.full(2*n-1, M)
    
    # Objective: maximize sum(g), i.e., minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: 0 <= g[j] <= 1 (upper bound to prevent degenerate solutions)
    bounds = [(0, 1.0)] * n
    
    # Additional constraint: sum(g) <= 1 (to normalize)
    # Actually, let's just maximize sum(g) and normalize afterward.
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        g = res.x
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-10:
            g = g / s
        return g
    return None

def lp_refine(a_init, time_limit, n_steps=50, blend=0.1, label="LP", verbose=True):
    """Iteratively refine using LP directions."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = lp_direction(a, M_mult=1.0)
        if g is None:
            break
        
        # Blend
        a_new = (1 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc = score_from_a(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller blend
            for b in [0.05, 0.02, 0.01]:
                a_try = (1 - b) * a + b * g
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                sc_try = score_from_a(a_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, try from best
                a = best_a.copy()
                if step > 5:
                    break
    
        if verbose and step % 10 == 0:
            print(f"  {label} step={step}: best={best_score:.7f}")
    
    return best_a, best_score

def cosine_init(n, alpha=0.5):
    """Cosine-tapered initialization."""
    x = np.linspace(0, np.pi, n)
    a = 1.0 + alpha * np.cos(x)
    a = np.maximum(a, 0)
    return a / a.sum()

def triangle_init(n):
    """Triangle initialization (peak in middle)."""
    mid = n // 2
    a = 1.0 - np.abs(np.arange(n) - mid) / (mid + 1)
    a = np.maximum(a, 1e-6)
    return a / a.sum()

def mirror_descent(a_init, time_limit, T_start=0.005, T_end=0.0002,
                   eta_start=0.02, eta_end=0.0003, label="MD", verbose=True):
    """Mirror descent with temperature annealing."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stuck = 0
    
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / 50000)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        w = softmax_weights(b, T)
        grad = grad_from_weights(a, w)
        
        # Multiplicative update
        log_up = -eta * grad
        log_up = np.clip(log_up, -3.0, 3.0)
        a = a * np.exp(log_up)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s
        
        if stuck > 500:
            rng = np.random.default_rng(it * 31)
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
    
    return best_a, best_score

                old = a_try[i]

def interpolate_seq(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def project_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - s))[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(b, T):
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    s = exp_scaled.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return exp_scaled / s


def grad_from_weights(a, w):
    n = len(a)
    fft_size = 1 << (max(3 * n - 1, 64)).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    return 2.0 * n * correl[:n]


def exact_peak_subgrad(a, k_peak):
    n = len(a)
    g = np.zeros(n)
    j_start = max(0, k_peak - n + 1)
    j_end = min(n - 1, k_peak)
    for j in range(j_start, j_end + 1):
        g[j] = 2.0 * a[k_peak - j]
    return g

def lp_direction(a, M_mult=1.0):
    n = len(a)
    if n > 1200:
        return None
    b_conv = fast_convolve(a)
    M = M_mult * b_conv.max()

    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    A = A.tocsr()
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0.0, 1.0)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def lbgfs_logspace(a_init, time_limit, T=30, maxiter=5000):
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        zs = z - z.max()
        ez = np.exp(zs)
        a = ez / ez.sum()
        b = fast_convolve(a)
        scaled = b / T
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        return 2.0 * n * T * lse

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        zs = z - z.max()
        ez = np.exp(zs)
        a = ez / ez.sum()
        b = fast_convolve(a)
        w = softmax_weights(b, T)
        ga = grad_from_weights(a, w)
        mg = np.dot(a, ga)
        return a * (ga - mg)

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxfun': maxiter * 20, 'disp': False})
        z_opt = res.x
        z_opt = z_opt - z_opt.max()
        ez = np.exp(z_opt)
        return ez / ez.sum()
    except Exception:
        return a_init.copy()

            best_seq, best_score = check_update(a_lp, "LP1", best_seq, best_score)

    # Phase 3: Subgradient Descent
    print(f"\n=== Phase 3: Subgradient (best={best_score:.7f}) ===")
    for T in [0.003, 0.008, 0.015]:
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t = min(rem * 0.3, 80.0)
        a_sg, _ = subgrad_descent(best_seq, t, T=T, eta_start=0.012, label=f"SG_T{T}")
        best_seq, best_score = check_update(a_sg, f"SG_T{T}", best_seq, best_score)

    # Phase 4: Second MD round with lower temperature
    print(f"\n=== Phase 4: MD Round 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 40:
        t = min(rem * 0.4, 120.0)
        a_md2, _ = mirror_descent(best_seq, t, T_start=0.001, T_end=0.00003,
                                  eta_start=0.008, eta_end=0.00008, label="MD2")
        best_seq, best_score = check_update(a_md2, "MD2", best_seq, best_score)

    # Phase 5: LP Round 2
    print(f"\n=== Phase 5: LP Round 2 (best={best_score:.7f}) ===")
    if len(best_seq) <= 1024:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t = min(rem * 0.5, 100.0)
            a_lp2, _ = lp_refine(best_seq, t, n_steps=25, label="LP2")
            best_seq, best_score = check_update(a_lp2, "LP2", best_seq, best_score)

    # Phase 6: Pairwise peak shaving
    print(f"\n=== Phase 6: Peak Shave (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t = min(rem * 0.4, 60.0)
        a_ps, _ = pairwise_peak_shave(best_seq, t, label="PS")
        best_seq, best_score = check_update(a_ps, "PS", best_seq, best_score)

    # Phase 7: L-BFGS in log-space
    print(f"\n=== Phase 7: L-BFGS (best={best_score:.7f}) ===")
    for T in [50, 20, 8, 3]:
        rem = deadline - time.monotonic() - 120
        if rem < 15:
            break
        t = min(rem * 0.3, 50.0)
        a_lb = lbgfs_logspace(best_seq, t, T=T, maxiter=4000)
        best_seq, best_score = check_update(a_lb, f"LB_T{T}", best_seq, best_score)

    # Phase 8: Try alternative lengths
    print(f"\n=== Phase 8: Alt Lengths (best={best_score:.7f}) ===")
    n_cur = len(best_seq)
    alt_ns = []
    for f in [1.25, 1.5, 0.8, 2.0]:
        an = int(n_cur * f)
        if 128 <= an <= 3000 and an != n_cur and an not in alt_ns:
            alt_ns.append(an)

    for an in alt_ns[:3]:
        rem = deadline - time.monotonic() - 100
        if rem < 40:
            break
        t = min(rem * 0.3, 60.0)
        init_alt = interpolate_seq(best_seq, an)
        a_alt, _ = mirror_descent(init_alt, t, T_start=0.003, T_end=0.0001,
                                  eta_start=0.012, eta_end=0.0003, label=f"alt{an}")
        best_seq, best_score = check_update(a_alt, f"alt{an}", best_seq, best_score)

    # Phase 9: Coordinate descent polish
    print(f"\n=== Phase 9: Coord Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 50
    if rem > 20:
        t = min(rem * 0.5, 50.0)
        a_cd, _ = coord_descent(best_seq, t, label="CD")
        best_seq, best_score = check_update(a_cd, "CD", best_seq, best_score)

    # Phase 10: Final peak shave + low-T MD
    print(f"\n=== Phase 10: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 15
    if rem > 15:
        t = min(rem * 0.5, 30.0)
        a_fin, _ = mirror_descent(best_seq, t, T_start=0.0005, T_end=0.00002,
                                  eta_start=0.004, eta_end=0.00005, label="FIN")
        best_seq, best_score = check_update(a_fin, "FIN_MD", best_seq, best_score)

    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t = min(rem * 0.6, 15.0)
        a_ps2, _ = pairwise_peak_shave(best_seq, t, label="FIN_PS")
        best_seq, best_score = check_update(a_ps2, "FIN_PS", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.7f}")

    return [float(x) for x in best_seq]
