    # = sum_j a[j] * w[2n-2-1-k+j] = sum_j a[j] * w[2n-3-k+j]
    #
    # Hmm, this is getting confusing. Let me just use a direct FFT approach.
    
    # Simple approach: pad w to length fft_size, FFT it.
    # For each i, grad[i] = sum_j a[j] * w[i+j]
    # This is the i-th element of the cross-correlation of w with a (where a is zero-padded).
    
    # Cross-correlation (w ⋆ a)[i] = sum_j w[j+i] * a[j] (with a zero-padded to length 2n-1 or more)
    # In terms of convolution: (w ⋆ a)[i] = (w * a_rev_flipped)[i + n - 1] where a_rev_flipped[j] = a[n-1-j]
    
    # Let me just use the standard formula:
    # corr = ifft(fft(w, N) * conj(fft(a, N)))
    # This gives the circular cross-correlation. For linear, we need N >= len(w) + len(a) - 1.
    
    N = fft_size
    Wf = rfft(w, N)
    Af = rfft(a, N)
    corr = irfft(Wf * np.conj(Af), N)
    # corr[k] = sum_j w[j] * a[(j-k) mod N] ... this is getting confusing with the exact offset.
    
    # Let me just verify: the standard definition of cross-correlation is:
    # (f ⋆ g)[τ] = sum_t f[t+τ] * conj(g[t])
    # In the FFT domain: FFFT(f ⋆ g) = FFT(f) * conj(FFT(g))
    # So corr = irfft(fft(w) * conj(fft(a)))
    # corr[τ] = sum_t w[t+τ] * a[t] (for linear, with appropriate padding)
    
    # So corr[i] = sum_t w[t+i] * a[t] for t where both are valid.
    # This is exactly what we want! grad[i] = corr[i] for i = 0, ..., n-1.
    
    return corr[:n]


def nesterov_logspace(a0, time_limit, T_init=0.05, T_final=0.001, 
                       eta_init=0.01, eta_final=0.001,
                       verbose=False, verbose_interval=5000):
    """
    Nesterov-accelerated gradient descent in log-space with cosine temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = z_from_a(a)
    
    # Nesterov momentum
    z_prev = z.copy()
    momentum = 0.9
    
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_b = fast_convolve(a)
    best_score = 2.0 * n * best_b.max()
    no_improve = 0
    max_no_improve = 5000
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Cosine annealing with plateau
        if progress < 0.2:
            T = T_init
            eta = eta_init
        else:
            p = (progress - 0.2) / 0.8
            T = T_final + (T_init - T_final) * 0.5 * (1 + np.cos(np.pi * p))
            eta = eta_final + (eta_init - eta_final) * 0.5 * (1 + np.cos(np.pi * p))
        
        # Nesterov step
        # y = z + momentum * (z - z_prev)
        y = z + momentum * (z - z_prev)
        z_prev = z.copy()
        
        # Compute a from y
        a_curr = softmax_from_z(y)
        
        # Compute objective gradient
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        
        # Gradient in a-space
        grad_a = compute_autocorr_gradient(a_curr, w, n)
        
        # Transform to z-space
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Update
        z = y - eta * grad_z
        
        # Evaluate
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive restart
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            z_prev = z.copy()
            no_improve = 0
            momentum = min(0.99, momentum * 1.01)
        
        if verbose and t % verbose_interval == 0:
            print(f"      Nesterov t={t} T={T:.5f} eta={eta:.5f} score={best_score:.6f}")
    
    return best_a, best_score


def systematic_peak_reduction(a0, time_limit, n_trials=10000):
    """
    Systematic peak reduction: identify peaks, find critical pairs,
    and try structured adjustments with compensation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    rng = np.random.default_rng(999)
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find top peaks (exclude endpoints which are always small)
        b_interior = b.copy()
        b_interior[0] = -np.inf
        b_interior[-1] = -np.inf
        peak_k = int(np.argmax(b_interior))
        peak_val = b[peak_k]
        
        # Elements contributing to this peak: pairs (j, peak_k - j) for valid j
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        
        if js > je:
            continue
        
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributing pairs
        top_n = min(5, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_n]
        
        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = int(peak_k - i)
            
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue
            
            # Try reducing one element and compensating with another
            for f1 in [0.85, 0.75, 0.65]:
                if i == j:
                    # Can't reduce a single element independently
                    a_trial = a.copy()
                    reduction = a_trial[i] * (1.0 - f1)
                    a_trial[i] *= f1
                    # Redistribute to random other element
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    other_idx = np.where(mask)[0]
                    target = other_idx[rng.integers(len(other_idx))]
                    a_trial[target] += reduction
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                else:
                    # Reduce a[i], increase a[j] (or vice versa)
                    a_trial = a.copy()
                    reduction = a_trial[i] * (1.0 - f1)
                    a_trial[i] *= f1
                    a_trial[j] += reduction
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                    
                    # Also try: reduce a[i], increase a[j] by different amount
                    a_trial2 = a.copy()
                    a_trial2[i] *= f1
                    a_trial2[j] += a_trial2[i] * (1.0 - f1) / f1  # compensate
                    # Actually, let's just try: increase a[j] proportionally
                    a_trial2 = a.copy()
                    a_trial2[i] *= f1
                    a_trial2[j] /= f1  # keep product roughly the same
                    # No, that increases the product. Let's try:
                    a_trial2 = a.copy()
                    a_trial2[i] *= f1
                    # Distribute the freed mass to a[j]
                    freed = a[ci if ci < n else 0] * (1-f1)  # approximate
                    a_trial2[j] += a[i] * (1 - f1)
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 > 1e-15:
                        a_trial2 = a_trial2 / ss2
                        score2 = 2.0 * n * fast_convolve(a_trial2).max()
                        if score2 < best_score - 1e-13:
                            a = a_trial2
                            best_score = score2
                            best_a = a_trial2.copy()
                            improved = True
                            break
            
            if improved:
                break
        
        if not improved:
            # Try a more global move: reduce the entire "pair set" for this peak
            # and redistribute to off-peak positions
            if trial % 20 == 19:
                a_trial = a.copy()
                # Reduce all elements that contribute to the peak
                for ci2 in top_idx[:3]:
                    i2 = int(idx[ci2])
                    a_trial[i2] *= 0.9
                # Increase elements at "off-peak" positions
                # Positions far from the peak
                off_peak = np.random.choice(n, size=min(3, n), replace=False)
                for op in off_peak:
                    if op not in idx:
                        a_trial[op] *= 1.1
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
    
    return best_a, best_score


def coordinate_descent_linesearch(a0, time_limit, n_sweeps=100):
    """
    Coordinate descent with golden section line search in each coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    def score_of(a_trial):
        ss = a_trial.sum()
        if ss < 1e-15:
            return np.inf
        a_norm = a_trial / ss
        return 2.0 * n * fast_convolve(a_norm).max()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            # Golden section search for the best value of a[i]
            # Current value: a[i], range: [0, 2*a[i]] or [0.1*a[i], 3*a[i]]
            lo = 0.0
            hi = max(3.0 * a[i], 2.0 * a[i])
            
            # Golden ratio
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            
            best_val = a[i]
            best_local_score = best_score
            
            for _ in range(8):  # 8 iterations of golden section
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)
                
                a1 = a.copy()
                a1[i] = x1
                s1 = score_of(a1)
                
                a2 = a.copy()
                a2[i] = x2
                s2 = score_of(a2)
                
                if s1 < s2:
                    hi = x2
                    if s1 < best_local_score:
                        best_local_score = s1
                        best_val = x1
                else:
                    lo = x1
                    if s2 < best_local_score:
                        best_local_score = s2
                        best_val = x2
            
            if best_val != a[i] and best_local_score < best_score:
                a[i] = best_val
                a = a / a.sum()
                best_score = best_local_score
                best_a = a.copy()
                sweep_improved = True
        
        if not sweep_improved:
            # Try random perturbation to escape
            if sweep % 10 == 9:
                a = best_a + np.random.standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score, print_flag=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if print_flag:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def proposed_initializations(parent, n, rng):
    """Generate diverse initializations."""
    candidates = []
    
    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            for scale in [0.02, 0.05, 0.1]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append((f'pert_{scale}', pert))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('parent_interp', interp))
    
    # Uniform
    candidates.append(('uniform', np.ones(n) / n))
    
    # Raised cosine
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = np.maximum(rc, 0.01)
    rc = rc / rc.sum()
    candidates.append(('raised_cos', rc))
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x**2 / 2)
    gau = np.maximum(gau, 1e-6)
    gau = gau / gau.sum()
    candidates.append(('gaussian', gau))
    
    # "U" shaped (low middle, high edges)
    x = np.linspace(-1, 1, n)
    u_shape = 1.0 + 0.5 * x**2
    u_shape = u

shape / u_shape.sum()
    candidates.append(('u_shape', u_shape))
    
    # Hamming window
    ham = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    ham = ham / ham.sum()
    candidates.append(('hamming', ham))
    
    # Exponential
    expo = np.exp(-2.0 * np.linspace(0, 1, n))
    expo = expo / expo.sum()
    candidates.append(('expo', expo))
    
    # Random
    r = np.abs(rng.standard_normal(n)) + 0.05
    r = r / r.sum()
    candidates.append(('random', r))
    
    return candidates


def lbgfs_polish(a0, time_limit, T=50, maxiter=2000):
    """L-BFGS polish in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    fft_size = 1 << (2*n - 2).bit_length()
    t0 = time.monotonic()
    stopped = [False]
    
    def obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        return 2.0 * n * (T * (max_b / T + np.log(np.exp(b / T - max_b / T).sum())))
    
    def grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        grad_a = compute_autocorr_gradient(a, w, n)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    try:
        res = minimize(fun=obj, x0=z0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxfun': maxiter * 30, 'disp': False})
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()
    
    return a_out
