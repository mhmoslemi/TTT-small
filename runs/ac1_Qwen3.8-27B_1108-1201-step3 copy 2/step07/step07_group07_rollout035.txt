"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax parameterization
with momentum, sparse LP refinement, adaptive step size, and multi-length
exploration. Different from prior approach by using multiplicative updates
(entropy-regularized mirror descent) and sparse constraint LP.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient_sparse(a, w):
    """
    Compute gradient g[m] = sum_k w[k] * 2*a[k-m] using FFT.
    w is the weight vector over convolution positions (length 2n-1).
    Returns g of length n.
    """
    n = len(a)
    # g[m] = sum_k w[k] * 2*a[k-m] where a is zero outside [0,n-1]
    # This is 2 * (w_rev * a)[m] where w_rev[k] = w[-k] mod ... 
    # Actually, let's think of it as: g = 2 * sum_k w[k] * shift(a, k)
    # which is the cross-correlation of w and a, but with a specific convention.
    # 
    # Let's define: g[m] = sum_k w[k] * a[k-m] (with a extended by 0)
    # = (w * a_reflected)[m] where a_reflected[t] = a[-t]
    # Actually, cross-correlation: (w ⊛ a)[m] = sum_k w[k+m] * a[k]
    # That's not quite right either.
    #
    # Let me just use the direct formula via FFT:
    # g[m] = sum_k w[k] * a[k-m] for 0<=m<n, 0<=k-m<n
    #
    # Define c[m] = sum_k w[k] * a[k-m]
    # This is the cross-correlation of w and a: c = w ⊛ a (in signal processing)
    # c[m] = sum_k w[k] * a[k-m]
    #
    # In terms of convolution: c[m] = sum_k w[k] * a[k-m]
    # Let j = k-m: c[m] = sum_j w[j+m] * a[j]
    # = (w_shifted * a_rev)[m] where... this is getting complicated.
    #
    # Simplest: use the fact that this is a cross-correlation, which can be
    # computed via FFT as: ifft(conj(FFT(w)) * FFT(a)) but with proper padding.
    #
    # Actually, the cross-correlation c[m] = sum_k w[k] * a[k-m] is:
    # c = ifft(conj(fft(w, N)) * fft(a, N))[:, :n] with N >= 3n-2
    # Wait, I need to be more careful.
    #
    # Let's just use the approach from the previous code:
    # g[m] = sum_k w[k] * 2*a[k-m]
    # This equals 2 * (the m-th element of the cross-correlation of w and a)
    #
    # Cross-correlation via FFT:
    # corr[m] = sum_k w[k+m] * a[k]  (this is one convention)
    # Or: corr[m] = sum_k w[k] * a[k-m]  (another convention)
    #
    # The second one: corr[m] = sum_k w[k] * a[k-m]
    # = sum_k w[k] * a_reflected[k-m] where a_reflected[t] = a[-t]... no.
    #
    # OK let me just do it directly with the FFT approach from the previous code.
    # The previous code used: irfft(conj(A) * W, fft_size)[:n]
    # where A = fft(a), W = fft(w). Let me verify this gives the right thing.
    #
    # irfft(conj(fft(a)) * fft(w)) = ifft(conj(A) * W)
    # This is the circular cross-correlation of a and w (in a specific convention).
    #
    # For linear cross-correlation, we need padding to avoid circular aliasing.
    # With fft_size >= 3n-2, the linear cross-correlation is embedded in the
    # circular one.
    #
    # The convention: ifft(conj(fft(a, N)) * fft(w, N)) [m] = sum_k a[k] * w[k+m] (mod N)
    # For linear: = sum_k a[k] * w[k+m] (when no wrap-around)
    #
    # But we want: g[m] = sum_k w[k] * a[k-m] = sum_j w[j+m] * a[j]
    #
    # So g[m] = ifft(conj(fft(a, N)) * fft(w, N))[m] (with proper padding)
    # Let's verify: ifft(conj(A)*W)[m] = sum_k a[k] * w[(k+m) mod N]
    # For linear (no wrap): = sum_k a[k] * w[k+m]
    # We want: sum_j a[j] * w[j+m] -- same thing with j=k!
    # 
    # So g[m] = 2 * ifft(conj(fft(a, N)) * fft(w, N))[m] for m=0,...,n-1
    # with N >= 3n-2 to avoid aliasing.
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * g


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def objective_value(a):
    """Compute 2n*max_b for normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    mb = eval_max_b(a_norm)
    return 2.0 * n * mb


def softmax_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, 
                     n_top=50, seed=42, tau_anneal=True):
    """
    Softmax (multiplicative) gradient optimization with momentum.
    Optimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    Returns (best_a, best_max_b).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    
    # Work in log-space for stability
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Log parameterization: a = softmax(u)
    u = np.log(np.maximum(a, 1e-15))
    
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    it_count = 0
    tau = 0.01  # temperature for peak weighting
    
    while time.monotonic() - t0 < time_limit:
        # Compute convolution
        b = fast_convolve(a)
        b_max = b.max()
        
        # Select active peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        
        # Softmax weighting of peaks
        top_vals = b[top_idx]
        if tau_anneal:
            tau = max(0.001, 0.01 * (0.999 ** it_count))
        w_peaks = np.exp((top_vals - b_max) / (tau * max(b_max, 1e-10)))
        w_peaks /= w_peaks.sum()
        
        # Build weight vector
        w = np.zeros(2 * n - 1)
        w[top_idx] = w_peaks
        
        # Compute gradient
        g = compute_gradient_sparse(a, w)
        
        # Center the gradient (for simplex constraint)
        g_centered = g - a * g.sum()
        
        # Momentum step in log-space
        v = momentum * v + cur_lr * g_centered
        u = u + v
        
        # Softmax to get new a
        u_shifted = u - u.max()
        a_new = np.exp(u_shifted)
        a_new /= a_new.sum()
        
        # Evaluate
        obj_new = eval_max_b(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it_count += 1
        if it_count % 1000 == 0:
            print(f"    softmax_opt: iter={it_count}, max_b={best_obj:.10f}, "
                  f"obj={2.0*n*best_obj:.8f}, lr={cur_lr:.6f}")
        
        # Adaptive strategy
        if no_improve >= 5000:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            u = np.log(np.maximum(a, 1e-15))
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.1)
            no_improve = 0
        elif no_improve >= 2000:
            cur_lr *= 0.9
        elif no_improve >= 1000:
            cur_lr *= 0.95
    
    return best_a, best_obj


def sparse_lp_refine(a, max_iters=20, time_limit=30.0, n_constraints=40):
    """
    Sparse LP refinement: only use top-k convolution constraints.
    Solves max sum(g) s.t. conv(a, g)[k] <= M for top-k k, g >= 0.
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Select top constraints
        if n_constraints < len(b):
            top_idx = np.argpartition(b, -n_constraints)[-n_constraints:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        
        # Build sparse constraint matrix (only for selected constraints)
        nc = len(top_idx)
        A_ub = np.zeros((nc, n))
        for i, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_ub[i, k - j] = a[j]
        
        b_ub = np.full(nc, M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': time_limit - (time.monotonic() - t0) - 5})
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            # Line search for blending
            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend /= ss
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t
            
            if best_t > 0.0:
                new_a = (1.0 - best_t) * best_a + best_t * g_norm
                new_a = np.maximum(new_a, 0.0)
                new_a /= new_a.sum()
                obj_new = eval_max_b(new_a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = new_a.copy()
                    if iteration % 5 == 0:
                        print(f"    sparse_LP iter {iteration}: t={best_t:.2f}, max_b={best_obj:.10f}")
                else:
                    a = best_a.copy()
            else:
                break
        except Exception as e:
            print(f"    LP error: {e}")
            break
    
    return best_a, best_obj

def make_init(n, kind="uniform"):
    """Create various initializations."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gauss":
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.5) / 0.25) ** 2)
        return a / a.sum()
    elif kind == "cosine":
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "raised_cosine":
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "tukey":
        a = np.ones(n)
        r = 0.4
        taper = max(1, int(r * n // 2))
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
        return a / a.sum()
    elif kind == "bump":
        # Double bump
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.3) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.15) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss > 1e-15:
        result /= ss
    return result


def coordinate_descent_polish(a0, time_limit, multipliers=None):
    """
    Coordinate descent: try scaling one element at a time.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    if multipliers is None:
        multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 50:
                    continue
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.arange(n) != j
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-12:
                    continue
                a_tr /= ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.arange(n) != j
                a[mask] *= scale
                a /= a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Try wider multipliers
            multipliers = np.array([0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0, 5.0])
            if time.monotonic() - t0 > time_limit * 0.5:
                break
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses softmax parameterization, sparse LP, multi-length exploration,
    and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax-parameterized gradient
descent with adaptive peak weighting, sparse LP refinement (top-k constraints),
multi-length interpolation, coordinate descent polishing, and random restarts.
Key differences from prior approach: log-space parameterization for automatic
non-negativity, sparse LP for speed, flat-target error weighting, and adaptive
peak temperature annealing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad(a, w):
    """
    Compute gradient g[m] = 2 * sum_k w[k] * a[k-m] using FFT cross-correlation.
    w: weight vector over convolution positions (length 2n-1).
    Returns g of length n.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * g


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def make_init(n, kind="uniform"):

    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gauss":

        return a / a.sum()
    elif kind == "gauss_wide":
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.5) / 0.35) ** 2)
        return a / a.sum()
    elif kind == "cosine":
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "raised_cosine":
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "tukey":
        a = np.ones(n)
        r = 0.4
        taper = max(1, int(r * n // 2))
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
        return a / a.sum()
    elif kind == "bump":
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.3) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.15) ** 2)
        return a / a.sum()
    elif kind == "flat_top":
        # Very flat in the middle, tapered at edges
        a = np.ones(n)
        edge = n // 6
        t = np.linspace(0, 1, edge)
        a[:edge] = 0.5 * (1.0 + np.cos(np.pi * t))
        a[-edge:] = a[:edge][::-1]
        return a / a.sum()
    else:
        return np.ones(n) / n


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss > 1e-15:
        result /= ss
    return result


def softmax_gradient_opt(a0, time_limit, lr_init=0.008, momentum=0.92,
                         n_top=50, seed=42, use_flat_target=False):
    """
    Log-space (softmax) projected gradient with Nesterov momentum.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    Returns (best_a, best_max_b).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)

    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Log parameterization
    u = np.log(np.maximum(a, 1e-15))
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    it_count = 0
    tau = 0.01

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        # Peak selection and weighting
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        top_vals = b[top_idx]
        if use_flat_target:
            # Weight by excess above flat target
            target = 1.0 / (2 * n - 1)
            excess = np.maximum(top_vals - target, 0.0)
            if excess.max() < 1e-15:
                w_peaks = np.ones(len(top_idx)) / len(top_idx)
            else:
                w_peaks = excess
                w_peaks /= w_peaks.sum()
        else:
            # Softmax weighting with adaptive temperature
            tau = max(0.001, 0.01 * (0.9995 ** it_count))
            w_peaks = np.exp((top_vals - b_max) / (tau * max(b_max, 1e-10)))
            w_peaks /= w_peaks.sum()

        w = np.zeros(2 * n - 1)
        w[top_idx] = w_peaks

        # Gradient via FFT
        g = compute_grad(a, w)

        # Center for simplex constraint
        g_centered = g - a * g.sum()

        # Nesterov momentum in log-space
        v = momentum * v + cur_lr * g_centered
        u = u + v

        # Softmax
        u_shifted = u - u.max()
        a_new = np.exp(u_shifted)
        a_new /= a_new.sum()

        obj_new = eval_max_b(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it_count += 1

        if it_count % 5000 == 0 and time_limit > 60:
            print(f"      sg: iter={it_count}, max_b={best_obj:.10f}, "
                  f"obj={2.0*n*best_obj:.8f}, lr={cur_lr:.6f}")

        # Adaptive
        if no_improve >= 8000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            u = np.log(np.maximum(a, 1e-15))
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.4, lr_init * 0.05)
            no_improve = 0
        elif no_improve >= 3000:
            cur_lr *= 0.9
        elif no_improve >= 1500:
            cur_lr *= 0.96

    return best_a, best_obj


def projection_gradient_opt(a0, time_limit, lr_init=0.005, momentum=0.9,

def sparse_lp_refine(a, max_iters=20, time_limit=30.0, n_constraints=40):
    """
    Sparse LP: max sum(g) s.t. conv(a,g)[k] <= M for top-k positions, g >= 0.
    Blend result with current a.
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        if n_constraints < len(b):
            top_idx = np.argpartition(b, -n_constraints)[-n_constraints:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)

        nc = len(top_idx)
        A_ub = np.zeros((nc, n))
        for i, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_ub[i, k - j] = a[j]

        b_ub = np.full(nc, M)
        c = -np.ones(n)

        try:
            remaining = time_limit - (time.monotonic() - t0) - 3
            if remaining < 2:
                break
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': remaining})
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend /= ss
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0.0:
                new_a = (1.0 - best_t) * best_a + best_t * g_norm
                new_a = np.maximum(new_a, 0.0)
                new_a /= new_a.sum()
                obj_new = eval_max_b(new_a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = new_a.copy()
                else:
                    a = best_a.copy()
            else:
                break
        except Exception:
            break

    return best_a, best_obj


def full_lp_refine(a, max_iters=15, time_limit=60.0):
    """
    Full LP with all 2n-1 constraints (slower but potentially better).
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        num_constraints = 2 * n - 1

        A_ub = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_ub[k, k - j] = a[j]

        b_ub = np.full(num_constraints, M)
        c = -np.ones(n)

        try:
            remaining = time_limit - (time.monotonic() - t0) - 5
            if remaining < 5:
                break
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': remaining})
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum
            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend /= ss
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0.0:
                new_a = (1.0 - best_t) * best_a + best_t * g_norm
                new_a = np.maximum(new_a, 0.0)
                new_a /= new_a.sum()
                obj_new = eval_max_b(new_a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = new_a.copy()
                else:
                    a = best_a.copy()
            else:
                break
        except Exception:
            break

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, multipliers=None):
    """Coordinate descent: scale one element at a time."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    if multipliers is None:
        multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])

    passes = 0
    stuck_count = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 50:
                    continue
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.arange(n) != j
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-12:
                    continue
                a_tr /= ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.arange(n) != j
                a[mask] *= scale
                a /= a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            stuck_count += 1
            if stuck_count >= 2:
                break
            # Try wider multipliers
            multipliers = np.array([0.3, 0.5, 0.7, 1.4, 1.7, 2.5, 4.0])
        else:
            stuck_count = 0

    return best_a, best_obj


def combined_optimize(a_init, time_limit, n, seed=42, verbose=False):
    """
    Alternating: projection SG -> sparse LP -> softmax SG -> sparse LP -> ...
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = eval_max_b(a)

    # Phase A: Projection gradient
    t_rem = time_limit - (time.monotonic() - t0)
    t_phase = min(t_rem * 0.35, 120.0)
    if t_phase > 5:
        if verbose:
            print(f"      [A] proj_sg t={t_phase:.0f}s")
        a_sg, obj_sg = projection_gradient_opt(
            a, time_limit=t_phase, lr_init=0.005, momentum=0.9,
            n_top=max(30, n // 15), seed=seed, use_flat_target=True)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            if verbose:
                print(f"        -> max_b={best_obj:.10f}, obj={2.0*n*best_obj:.8f}")

    # Phase B: Sparse LP
    t_rem = time_limit - (time.monotonic() - t0)
    t_phase = min(t_rem * 0.2, 45.0)
    if t_phase > 3:
        if verbose:
            print(f"      [B] sparse_lp t={t_phase:.0f}s")
        a_lp, obj_lp = sparse_lp_refine(best_a, max_iters=12, time_limit=t_phase,
                                         n_constraints=35)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"        -> max_b={best_obj:.10f}, obj={2.0*n*best_obj:.8f}")

    # Phase C: Softmax gradient
    t_rem = time_limit - (time.monotonic() - t0)
    t_phase = min(t_rem * 0.3, 100.0)
    if t_phase > 5:
        if verbose:
            print(f"      [C] softmax_sg t={t_phase:.0f}s")
        a_sg2, obj_sg2 = softmax_gradient_opt(
            best_a, time_limit=t_phase, lr_init=0.004, momentum=0.93,
            n_top=max(40, n // 10), seed=seed + 111)
        if obj_sg2 < best_obj:
            best_obj = obj_sg2
            best_a = a_sg2.copy()
            if verbose:
                print(f"        -> max_b={best_obj:.10f}, obj={2.0*n*best_obj:.8f}")

    # Phase D: Final sparse LP
    t_rem = time_limit - (time.monotonic() - t0)
    t_phase = min(t_rem * 0.25, 40.0)
    if t_phase > 3:
        if verbose:
            print(f"      [D] sparse_lp2 t={t_phase:.0f}s")
        a_lp2, obj_lp2 = sparse_lp_refine(best_a, max_iters=8, time_limit=t_phase,
                                           n_constraints=30)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            if verbose:
                print(f"        -> max_b={best_obj:.10f}, obj={2.0*n*best_obj:.8f}")

    # Phase E: Fine projection gradient
    t_rem = time_limit - (time.monotonic() - t0)
    if t_rem > 5:
        if verbose:
            print(f"      [E] fine_sg t={t_rem:.0f}s")
        a_sg3, obj_sg3 = projection_gradient_opt(
            best_a, time_limit=t_rem, lr_init=0.001, momentum=0.95,
            n_top=max(60, n // 6), seed=seed + 222, use_flat_target=True)
        if obj_sg3 < best_obj:
            best_obj = obj_sg3
            best_a = a_sg3.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses flat-target weighted subgradient, sparse LP, softmax parameterization,
    multi-length exploration, coordinate descent, and random restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 60.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, max_b={eval_max_b(a_norm):.10f})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_init(1152, "uniform")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using uniform n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Quick optimization on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Optimize parent (n={n_parent}, score={best_score:.6f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t1 = min(rem * 0.3, 180.0)
        print(f"  Allocating {t1:.0f}s")
        a1, obj1 = combined_optimize(parent, time_limit=t1, n=n_parent,
                                      seed=seed, verbose=True)
        update_best(a1, "P1_parent")

    # === Phase 2: Random restart from best ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Random restarts (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        n_restarts = 3
        t_per_restart = min(rem / n_restarts, 80.0)
        for r in range(n_restarts):
            rem = deadline - time.monotonic() - 350
            if rem < 15:
                break
            t_alloc = min(t_per_restart, rem * 0.4)
            if t_alloc < 10:
                continue
            noise_scale = 0.01 * (r + 1)
            perturbed = best_seq + rng.standard_normal(n_base) * noise_scale
            perturbed = np.maximum(perturbed, 0.0)
            perturbed /= perturbed.sum()
            print(f"  Restart {r+1}, noise={noise_scale:.4f}, t={t_alloc:.0f}s")
            a_r, obj_r = combined_optimize(perturbed, time_limit=t_alloc,
                                            n=n_base, seed=seed + 1000 * (r + 1),
                                            verbose=False)
            update_best(a_r, f"P2_restart{r+1}")

    # === Phase 3: Diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    init_kinds = ["gauss", "cosine", "raised_cosine", "tukey", "bump", "flat_top"]
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_per_init = min(rem / len(init_kinds), 60.0)
        for kind in init_kinds:
            rem = deadline - time.monotonic() - 250
            if rem < 10:
                break
            t_alloc = min(t_per_init, rem * 0.4)
            if t_alloc < 8:
                continue
            print(f"  {kind} (t={t_alloc:.0f}s)")
            init_a = make_init(n_base, kind)
            a_div, obj_div = combined_optimize(init_a, time_limit=t_alloc,
                                                n=n_base, seed=seed + 5000 + hash(kind) % 9999,
                                                verbose=False)
            update_best(a_div, f"P3_{kind}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 0.65, 1.33, 0.85, 1.15, 1.75]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for tl in target_lengths[:6]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_alloc = min(rem *
