import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_gradient(a, k):
    """Gradient of conv(a,a)_k with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] = a[j]
    return grad

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_score_normalized(a):
    """Score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def subgradient_optimize(a0, time_limit, K=10, lr_init=0.01, momentum=0.9, verbose=True):
    """Subgradient method with top-K peak averaging and Nesterov."""
    n = len(a0)
    a = a0.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    lr = lr_init
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        # Find top-K peaks
        top_k = min(K, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_values = conv[peak_indices]
        
        # Compute weighted gradient
        weights = peak_values / peak_values.sum()
        g = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            g += w * conv_gradient(a, int(k))
        
        # Center the gradient (to maintain sum constraint)
        g = g - g.mean()
        
        # Nesterov step
        v = momentum * v + lr * g
        a_new = project_simplex(a - v)
        
        score = compute_score_normalized(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        lr = max(lr * 0.9999, 1e-6)
        
        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={iters}, t={elapsed:.1f}s, score={best_score:.8f}")
    
    return best_a, best_score

Key innovations over previous approach:

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_value(a, T):
    """Compute log-sum-exp of conv/T stably."""
    conv = fast_convolve(a)
    shifted = conv / T
    m = shifted.max()
    return m + np.log(np.exp(shifted - m).sum())


def lse_gradient(a, T):
    """
    Gradient of LSE(conv(a,a)/T) w.r.t. a.
    Returns gradient of the LSE part (without the 2n factor).
    """
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv / T
    m = shifted.max()
    w = np.exp(shifted - m)
    w = w / w.sum()
    # grad_j = (2/T) * sum_i w_{j+i} * a_i = (2/T) * (w cross-corr a)_j
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: ifft(conj(W) * A)
    corr = irfft(np.conj(W) * A, fft_size)
    return (2.0 / T) * corr[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n <= 0:
        return v
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

            # grad of conv_k: g[i] += a[k-i] if 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                g[i] += w_k * a[k - i]

        # Center gradient
        g = g - g.mean()

        # Multiplicative update (ensures non-negativity)
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score - 1e-16:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                lr *= 0.5
                no_improve = 0

        if iters % 5000 == 0:
            lr *= 0.9

        if verbose and iters % 10000 == 0:
            print(f"      SG iter={iters} t={time.monotonic()-t0:.1f}s score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def peak_newton(a0, time_limit, K=8, verbose=False):
    """
    Active-set Newton method:
    1. Identify active peaks (top K with values close to max)
    2. Solve the KKT system: conv_k(a) = lambda for k in active set
    3. Use Newton iteration on the nonlinear system
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    iters = 0

    while time.monotonic() - t0 < time_limit and iters < 50000:
        iters += 1
        conv = fast_convolve(a)
        max_val = conv.max()

        # Identify active peaks: those within 1% of max
        threshold = max_val * 0.995
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        active = active[:K]  # Limit to K

        # Try Newton step: we want to equalize the active peaks
        # Build the Jacobian of [conv_k for k in active] w.r.t. a
        # J[i, j] = d(conv_i)/da_j = a[i-j] if 0 <= i-j < n
        m = len(active)
        J = np.zeros((m, n))
        for r, k in enumerate(active):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            J[r, lo:hi+1] = a[k - np.arange(lo, hi + 1)]

        # Current peak values
        c_val = conv[active].copy()

        # We want to find delta such that c_val + J @ delta = lambda * ones(m)
        # And sum(delta) = 0 (maintain sum constraint)
        # Augmented system:
        # [J  1] [delta]   [lambda - c_val ... ]
        # [1' 0] [lambda]  [0              ]
        # Actually: J @ delta + lambda_vec = target
        # Let's minimize ||J @ delta - mean(J@delta)||_2^2 + penalty * (sum(delta))^2
        # Simpler: least squares with constraint sum(delta)=0

        # Use: minimize ||J @ delta - c_target||^2 subject to sum(delta) = 0
        # where c_target = mean(c_val) (we want all peaks to equal their mean)
        c_target = np.full(m, c_val.mean())

        # Augmented: [J; ones(1,n)] delta = [c_target; 0]
        J_aug = np.vstack([J, np.ones((1, n))])
        b_aug = np.concatenate([c_target, [0.0]])

        # Solve least squares
        try:
            delta = np.linalg.lstsq(J_aug, b_aug, rcond=None)[0]
        except:
            break

        # Limit step size
        delta_norm = np.linalg.norm(delta)
        if delta_norm > 1e-3:
            delta = delta * (1e-3 / delta_norm)

        # Apply step
        a_new = a + delta
        a_new = project_simplex(a_new)

        score = compute_score(a_new)
        if score < best_score - 1e-16:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for shrink in [0.1, 0.01]:
                a_try = project_simplex(a + delta * shrink)
                sc_try = compute_score(a_try)
                if sc_try < best_score - 1e-16:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                if iters % 10000 == 0:
                    # Perturb to escape
                    rng = np.random.default_rng(iters)
                    a = project_simplex(a + rng.standard_normal(n) * 0.001)

        if verbose and iters % 1000 == 0:
            print(f"      PN iter={iters} t={time.monotonic()-t0:.1f}s score={best_score:.8f} |active|={m}")

    return best_a, best_score


def multiplicative_ascent(a0, time_limit, T=100, verbose=False):
    """
    Multiplicative update method:
    a_i <- a_i * (sum of gradient contributions) / a_i
    Specifically: a_i <- a_i * exp(-lr * grad_i)
    This is like gradient flow on the simplex with natural parameterization.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 1.0 / (4.0 * n / T)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        grad = 2.0 * n * lse_gradient(a, T)
        # Center
        grad = grad - grad.mean()
        # Multiplicative update
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        if iters % 100 == 0:
            score = compute_score(a)
            if score < best_score - 1e-16:
                best_score = score
                best_a = a.copy()
            lr *= 0.999

        if verbose and iters % 5000 == 0:
            print(f"      MU iter={iters} t={time.monotonic()-t0:.1f}s score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving non-negativity and sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def suggest_step_size(a, T, g_0=None):
    """
    Estimate a good step size using the Hessian approximation.
    Uses the Rayleigh quotient of the Hessian at the current point.
    """
    n = len(a)
    # The Hessian of LSE(conv/T) is (2/T)^2 * E_w[(w cross-corr a)(w cross-corr a)^T] + (2/T) * Hessian(conv)
    # Simplified: use (2/T) * norm of gradient as a proxy for curvature
    if g_0 is not None:
        g_norm = np.linalg.norm(g_0)
        if g_norm > 1e-10:
            # Lipschitz constant estimate: L ~ 2n * 2n / T (rough)
            return min(1.0 / (2.0 * n * n / T), 1.0)
    return 1.0 / (4.0 * n * n / T)


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization routine.
    Strategy:
    Phase 1: T-annealing FISTA (50% time) - the main workhorse
    Phase 2: Peak Newton refinement (15% time)
    Phase 3: Subgradient top-K (10% time)
    Phase 4: Nearby lengths (15% time)
    Phase 5: Final polish (10% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi * 3, n_init)
        parent = parent + 0.05 * np.cos(t)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        p_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    n_parent = len(parent)

    # ============ Phase 1: T-annealing FISTA (50% of time) ============
    print(f"\n=== Phase 1: T-Annealing FISTA (n={n_parent}) ===")
    phase1_end = t_start + 0.50 * (budget_s - 30.0)
    T_schedule = [10000, 5000, 2500, 1000, 500, 250, 100, 50, 25, 10, 5, 2]
    n_phases = len(T_schedule)

    # Allocate time: give more time to lower T values
    weights = np.array([1.0, 1.0, 1.0, 1.5, 1.5, 2.0, 2.0, 2.5, 2.5, 3.0, 3.0, 3.0])
    weights = weights / weights.sum()

    for i, T in enumerate(T_schedule):
        phase1_rem = phase1_end - time.monotonic()
        if phase1_rem < 2:
            break
        # Time for this T phase
        remaining_phases = weights[i:].sum() / weights.sum()
        t_alloc = phase1_rem * (weights[i] / weights[i:].sum())
        t_alloc = max(1.0, min(t_alloc, phase1_rem * 0.6))

        print(f"  T={T}, budget={t_alloc:.0f}s, current_best={best_score:.8f}")
        a_new, sc_new = fista_with_lse(best_seq, t_alloc, T=T, verbose=False)
        best_seq, best_score = check_and_update(a_new, f"fista_T{T}", best_seq, best_score)
        print(f"    -> best={best_score:.8f}")

    # ============ Phase 2: Peak Newton (15% of time) ============
    print(f"\n=== Phase 2: Peak Newton (n={n_parent}) ===")
    phase2_end = min(t_start + 0.70 * (budget_s - 30.0), deadline)
    phase2_rem = phase2_end - time.monotonic()
    if phase2_rem > 3:
        t_alloc = min(phase2_rem * 0.8, 200.0)
        print(f"  budget={t_alloc:.0f}s, current_best={best_score:.8f}")
        a_new, sc_new = peak_newton(best_seq, t_alloc, K=8, verbose=True)
        best_seq, best_score = check_and_update(a_new, "peak_newton", best_seq, best_score)
        print(f"    -> best={best_score:.8f}")

    # ============ Phase 3: Subgradient top-K (10% of time) ============
    print(f"\n=== Phase 3: Subgradient Top-K (n={n_parent}) ===")
    phase3_end = min(t_start + 0.80 * (budget_s - 30.0), deadline)
    phase3_rem = phase3_end - time.monotonic()
    if phase3_rem > 3:
        t_alloc = min(phase3_rem * 0.7, 100.0)
        print(f"  budget={t_alloc:.0f}s, current_best={best_score:.8f}")
        a_new, sc_new = subgradient_topk(best_seq, t_alloc, K=5, lr_init=0.003, verbose=True)
        best_seq, best_score = check_and_update(a_new, "subgrad_k5", best_seq, best_score)
        print(f"    -> best={best_score:.8f}")

    # ============ Phase 4: Nearby lengths (15% of time) ============
    print(f"\n=== Phase 4: Nearby Lengths ===")
    phase4_end = min(t_start + 0.90 * (budget_s - 30.0), deadline)
    nearby_ns = [n_parent - 1, n_parent + 1, n_parent - 2, n_parent + 2,
                 n_parent - 3, n_parent + 3]
    for alt_n in nearby_ns:
        if alt_n < 128 or alt_n > 2048:
            continue
        phase4_rem = phase4_end - time.monotonic()
        if phase4_rem < 10:
            break
        t_alloc = min(phase4_rem * 0.25, 60.0)
        if t_alloc < 5:
            continue
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  n={alt_n}, budget={t_alloc:.0f}s")
        # Use FISTA with moderate T
        a_new, sc_new = fista_with_lse(init_alt, t_alloc, T=100, verbose=False)
        best_seq, best_score = check_and_update(a_new, f"nearby_n{alt_n}", best_seq, best_score)

    # ============ Phase 5: Final polish (10% of time) ============
    print(f"\n=== Phase 5: Final Polish (n={len(best_seq)}) ===")
    phase5_rem = deadline - time.monotonic()
    if phase5_rem > 5:
        # Quick FISTA with low T
        t_alloc = min(phase5_rem * 0.4, 60.0)
        print(f"  FISTA T=1, budget={t_alloc:.0f}s")
        a_new, sc_new = fista_with_lse(best_seq, t_alloc, T=1.0, verbose=False)
        best_seq, best_score = check_and_update(a_new, "final_fista", best_seq, best_score)

        # Peak Newton
        phase5_rem = deadline - time.monotonic()
        if phase5_rem > 5:
            t_alloc = min(phase5_rem * 0.5, 40.0)
            print(f"  Peak Newton, budget={t_alloc:.0f}s")
            a_new, sc_new = peak_newton(best_seq, t_alloc, K=10, verbose=False)
            best_seq, best_score = check_and_update(a_new, "final_newton", best_seq, best_score)

        # Multiplicative ascent
        phase5_rem = deadline - time.monotonic()
        if phase5_rem > 5:
            t_alloc = min(phase5_rem * 0.6, 30.0)
            print(f"  Mult Ascent, budget={t_alloc:.0f}s")
            a_new, sc_new = multiplicative_ascent(best_seq, t_alloc, T=50, verbose=False)
            best_seq, best_score = check_and_update(a_new, "final_mu", best_seq, best_score)

    # Final verification
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 1e-10:
        best_seq = best_seq / s

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={total_time:.1f}s")
    return [float(x) for x in best_seq]
