"""
Optimized LP-based search for minimum peak auto-convolution sequence.
Uses threshold-based peak selection, LP direction finding, line search,
subgradient polish, coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_active_peaks(b, M, threshold_frac=0.7, max_k=200):
    """Get indices of peaks above threshold*M."""
    threshold = threshold_frac * M
    active = np.where(b >= threshold)[0]
    if len(active) > max_k:
        # Keep only the top max_k
        top_idx = np.argpartition(b, -max_k)[-max_k:]
        active = top_idx
    return np.sort(active)

def solve_lp_direction(a, active_idx):
    """
    Solve: minimize t s.t. conv(a,g)[k] <= t for k in active,
           a[i]+g[i] >= 0, sum(g) = 0.
    Returns (g, t) or (None, None).
    """
    n = len(a)
    K = len(active_idx)
    if K == 0:
        return None, None
    
    A_mat = np.zeros((K, n))
    for row_i in range(K):
        k = active_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        a_idx = k - np.arange(lo, hi + 1)
        A_mat[row_i, lo:hi+1] = a[a_idx]
    
    g = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(g) == 0,
        g >= -a
    ]
    for row_i in range(K):
        constraints.append(A_mat[row_i] @ g <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-9)
    except Exception:
        return None, None
    
    if g.value is None:
        return None, None
    
    return g.value, t.value if t.value is not None else 0.0

def line_search(a, g, max_iters=8):
    """Find best step size t for a + t*g."""
    best_t = 0.1
    best_obj = fast_convolve(a).max()
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_trial = a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s
        obj = fast_convolve(a_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj

def subgradient_step(f, lr, v, momentum, n_top=50):
    """One subgradient step with momentum."""
    n = len(f)
    b = fast_convolve(f)
    
    # Find top peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))
    
    # Compute gradient
    g = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += f[k - idx]
    
    # Normalize gradient
    g_norm = np.linalg.norm(g)
    if g_norm > 1e-15:
        g /= g_norm
    
    v = momentum * v - lr * g
    f_new = project_simplex(f + v)
    return f_new, v

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]
    
    improved = True
    rounds = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        rounds += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    
    return f, best_obj

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_active_peaks(b, M, threshold_frac=0.7, max_k=200):
    """Get indices of peaks above threshold*M, capped at max_k."""
    threshold = threshold_frac * M
    active = np.where(b >= threshold)[0]
    if len(active) > max_k:
        top_idx = np.argpartition(b, -max_k)[-max_k:]
        active = top_idx
    else:
        active = np.sort(active)
    return active


def build_constraint_matrix(a, active_idx):
    """Build the K x n constraint matrix for conv(a, g) at active peaks."""
    n = len(a)
    K = len(active_idx)
    A_mat = np.zeros((K, n))
    for row_i in range(K):
        k = active_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        a_idx = k - np.arange(lo, hi + 1)
        A_mat[row_i, lo:hi + 1] = a[a_idx]
    return A_mat


def solve_lp_direction(a, active_idx):
    """
    Solve: minimize t s.t. A@g <= t, sum(g)=0, -a <= g.
    This finds the direction that minimizes the worst first-order increase
    at active peaks while maintaining non-negativity.
    Returns (g, t) or (None, None) on failure.
    """
    n = len(a)
    K = len(active_idx)
    if K == 0:
        return None, None

    A_mat = build_constraint_matrix(a, active_idx)

    g = cp.Variable(n)
    t = cp.Variable()

    constraints = [
        cp.sum(g) == 0,
        g >= -a
    ]
    for row_i in range(K):
        constraints.append(A_mat[row_i] @ g <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
    except Exception:
        return None, None

    if g.value is None:
        return None, None

    g_val = np.array(g.value).flatten()
    t_val = float(t.value) if t.value is not None else 0.0
    if np.any(np.isnan(g_val)):
        return None, None
    return g_val, t_val


def solve_lp_direction_v2(a, active_idx):
    """
    Alternative LP: maximize sum(g) s.t. A@g <= 0, g >= -a, sum(g)=0.
    This finds a direction that doesn't increase any active peak (to first order).

    """

    if K == 0:

    A_mat = build_constraint_matrix(a, active_idx)

    return g_val, obj


def line_search(a, direction, max_step=1.0):
    """Backtracking line search. Returns (best_t, best_a, best_obj)."""
    best_t = 0.0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
    for t in candidates:
        t_actual = t * max_step
        a_trial = a + t_actual * direction
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        obj = fast_convolve(a_trial).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_trial.copy()
            best_t = t_actual

    return best_t, best_a, best_obj

def subgradient_step(f, lr, v, momentum, n_top=50):
    """One projected subgradient step with momentum."""
    n = len(f)
    b = fast_convolve(f)

    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))

    g = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += f[k - idx]

    g_norm = np.linalg.norm(g)
    if g_norm > 1e-15:
        g /= g_norm

    v = momentum * v - lr * g
    f_new = project_simplex(f + v)
    return f_new, v

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with multipliers for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.6, 0.8, 0.95, 1.05, 1.2, 1.5]

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True

    return f, best_obj

def interpolate_sequence(f, target_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == target_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(target_n) / target_n
    return f_new

        f1, obj1 = run_lp_ils(parent, time_limit=t1, label="ils_parent")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:

            print(f"  NEW BEST [ils_parent]: score={score1:.6f}")

            tl = int(n_base * factor)
            tl = max(256, min(4000, tl))
            if tl != n_base and tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_base))

        t_per = t3_total / min(3, len(target_lengths))

                best_seq = fi.copy()
                print(f"  NEW BEST [n={target_n}]: score={fi_score:.6f}")

    rem = deadline - time.monotonic()

        f4, obj4 = run_lp_ils(best_seq, time_limit=t4, label="deep_ils")
