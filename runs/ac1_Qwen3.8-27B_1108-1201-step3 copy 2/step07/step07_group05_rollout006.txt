```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
combined with gradient-based refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def solve_lp_for_g(f, n):
    """Solve LP: min M' s.t. (f*g)[k] <= M' for all k, g>=0, sum(g)=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    num_constraints = 2 * n - 1
    A = lil_matrix((num_constraints, n + 1))
    
    for k in range(num_constraints):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A[k, k - i] = f[i]
        A[k, n] = -1.0
    
    A_ub = A.tocsr()
    b_ub = np.zeros(num_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        return g
    return f.copy()


def fictitious_play(f0, time_limit, max_iters=10000, verbose=True):
    """LP-based fictitious play for the minimax problem."""
    n = len(f0)
    f = f0.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = solve_lp_for_g(f, n)
        t = 2.0 / (it + 2.0)
        f = (1.0 - t) * f + t * g
        f = np.maximum(f, 0)
        fs = f.sum()
        if fs > 0:
            f = f / fs
        
        score = compute_score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP iter={it}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': maxiter * 40, 'disp': False})
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass
    
    result['seconds'] = time.monotonic() - t0
    return result


def coordinate_descent(a0, time_limit, n_pairs_per_iter=500):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()

        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
combined with gradient-based refinement (L-BFGS with soft-max smoothing),
peak equalization, and coordinate descent.

Key innovation: The LP-based fictitious play directly solves the best-response
problem at each iteration (minimize max(f*g) over g in simplex), which is
fundamentally different from gradient-based approaches and can escape local
minima that trap smooth optimization methods.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, coo_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def build_lp_structure(n):
    """Precompute the sparse structure of the LP constraint matrix."""
    num_constraints = 2 * n - 1
    # Precompute for each k, the list of (i, j) pairs where j = k - i
    # and both 0 <= i < n, 0 <= j < n
    k_indices = []
    i_indices = []
    j_indices = []
    for k in range(num_constraints):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            k_indices.append(k)
            i_indices.append(i)
            j_indices.append(j)
    # Add the -1 for M' in each row
    k_indices.extend(range(num_constraints))
    i_indices.extend([0] * num_constraints)
    j_indices.extend([n] * num_constraints)
    return np.array(k_indices), np.array(i_indices), np.array(j_indices)


def solve_lp_for_g(f, n, k_idx, i_idx, j_idx):
    """
    Solve LP: min M' s.t. (f*g)[k] <= M' for all k, g>=0, sum(g)=1.
    Uses precomputed structure for speed.
    """
    num_constraints = 2 * n - 1
    num_entries = len(k_idx)
    
    # Fill values
    vals = np.zeros(num_entries)
    for idx in range(num_entries):
        j = j_idx[idx]
        if j < n:
            vals[idx] = f[i_idx[idx]]
        else:
            vals[idx] = -1.0
    
    rows = k_idx
    cols = j_idx
    
    A_ub = coo_matrix((vals, (rows, cols)), shape=(num_constraints, n + 1)).tocsr()
    b_ub = np.zeros(num_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 20})
    
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        return g
    return f.copy()


def fictitious_play(f0, time_limit, max_iters=10000, verbose=True):
    """LP-based fictitious play for the minimax problem."""
    n = len(f0)
    f = f0.copy()
    f = np.maximum(f, 0)
    fs = f.sum()
    if fs < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / fs
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Precompute LP structure
    k_idx, i_idx, j_idx = build_lp_structure(n)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = solve_lp_for_g(f, n, k_idx, i_idx, j_idx)
        t = 2.0 / (it + 2.0)
        f = (1.0 - t) * f + t * g
        f = np.maximum(f, 0)
        fs = f.sum()
        if fs > 1e-15:
            f = f / fs
        
        score = compute_score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP iter={it}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': maxiter * 40, 'disp': False})
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass
    
    result['seconds'] = time.monotonic() - t0
    return result


def coordinate_descent(a0, time_limit, n_pairs_per_iter=500):
    """Coordinate descent using pairwise transfer moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        for _ in range(n_pairs_per_iter):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-14:
                continue
            
            base = a[i]
            best_t = 0.0
            best_local = best_score
            
            for frac in [0.03, 0.07, 0.15, 0.25, 0.4, 0.6]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
                
                if a[j] > 1e-15:
                    t2 = frac * a[j]
                    a_trial2 = a.copy()
                    a_trial2[j] -= t2
                    a_trial2[i] += t2
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial2 = a_trial2 / ss2
                    sc2 = compute_score(a_trial2)
                    if sc2 < best_local - 1e-14:
                        best_local = sc2
                        best_t = -t2
            
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
                stuck_count = 0
        
        if not improved:
            stuck_count += 1
        else:
            stuck_count = 0
        
        if stuck_count > 30:
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stuck_count = 0
    
    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=10):
    """Iteratively reduce the maximum peak of the auto-convolution."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))
        
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        top_contrib = np.argsort(contribs)[::-1][:8]
        
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for frac in [0.03, 0.07, 0.12, 0.2, 0.3]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve_count += 1
            # Try random perturbation on peak-contributing positions
            peak_positions = set()
            top_k = min(n_peaks, len(conv))
            peak_indices = np.argpartition(conv, -top_k)[-top_k:]
            for pi in peak_indices[:5]:
                si = max(0, int(pi) - n + 1)
                ei = min(n - 1, int(pi))
                peak_positions.update(range(si, ei + 1))
            
            if peak_positions:
                pos_list = list(peak_positions)
                for _ in range(15):
                    p = pos_list[rng.integers(0, len(pos_list))]
                    a_trial = a.copy()
                    delta = a[p] * 0.05 * rng.standard_normal()
                    a_trial[p] -= delta
                    mask = np.ones(n, dtype=bool)
                    mask[p] = False
                    oth2 = a_trial[mask]
                    o2s = oth2.sum()
                    if o2s > 1e-15:
                        a_trial[mask] = oth2 + delta * oth2 / o2s
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        
        if no_improve_count > 50 and time.monotonic() - t0 > 10:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def nesterov_accelerated(a0, time_limit, T=500, lr_init=0.03, momentum=0.92):
    """Nesterov accelerated gradient descent in log-space."""
    n = len(a0)
    z = z_from_a(a0)
    v = np.zeros(n)
    best_z = z.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        z_lookahead = z - momentum * v
        a_la = softmax_from_z(z_lookahead)
        grad = smoothed_gradient(z_lookahead, T, n)
        
        v = momentum * v + lr * grad
        z_new = z - v
        z_new = np.clip(z_new, -25, 25)
        
        a_new = softmax_from_z(z_new)
        score = compute_score(a_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_z = z_new.copy()
        
        z = z_new
        lr *= 0.9997
    
    a_out = softmax_from_z(best_z)
    return a_out, best_score
```
