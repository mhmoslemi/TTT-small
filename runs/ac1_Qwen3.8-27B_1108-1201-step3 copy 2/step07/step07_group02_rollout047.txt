Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. LP-based Nesterov dual averaging for the zero-sum game formulation
2. L-BFGS with p-norm smooth approximation (adaptive p)

5. Final fine-tuning with very low temperature smooth optimization

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return np.maximum(irfft(A * A, fft_size)[:size], 0)


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv_type(a, b) element-wise."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = j + n
        T[lo:hi, j] = a
    return T


def _get_solver():
    global _solver_cached
    if '_solver_cached' in globals():
        return _solver_cached
    for name in [cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=name)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _solver_cached = name
                return name
        except Exception:
            continue
    _solver_cached = cp.ECOS
    return _solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = _get_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    b_pos = np.maximum(b, 1e-300)
    obj = np.sum(b_pos ** p) ** (1.0 / p)
    # grad_a[i] = (p-1) * sum_k b[k]^(p-1) * a[k-i] (for valid k-i)
    # which is (p-1) * conv(b_pos^(p-1), flip(a))[i] ... let me compute directly
    b_pm1 = b_pos ** (p - 1)
    # grad[i] = (p-1) * sum_k b[k]^(p-1) * a[k-i] where 0<=k-i<n and 0<=k<2n-1
    # = (p-1) * (b_pm1 * flip(a))[i] in a correlation sense
    # Actually: db[k]/da[i] = a[k-i] if 0 <= k-i < n, else 0
    # grad[i] = (p-1) * sum_k b[k]^(p-1) * (db[k]/da[i])
    # = (p-1) * sum_k b[k]^(p-1) * a[k-i] for valid k
    # This is the cross-correlation of b_pm1 with a, computed as:
    # grad[i] = (p-1) * sum_{k=max(i,0)}^{min(i+n-1, 2n-2)} b[k]^(p-1) * a[k-i]
    # which is conv(b_pm1, flip(a))[i + n - 1] ... let me just use FFT
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    B = rfft(b_pm1, fft_size)
    A_flip = rfft(a[::-1], fft_size)
    corr = np.real(irfft(B * A_flip, fft_size))
    # corr[j] = sum_k b_pm1[k] * a_flip[j-k] = sum_k b_pm1[k] * a[n-1-j+k]
    # We want grad[i] = sum_k b_pm1[k] * a[k-i] (for valid indices)
    # Let me just compute it directly for clarity (n is at most ~2000)
    grad = np.zeros(n)
    for i in range(n):
        k_lo = max(i, 0)
        k_hi = min(i + n, 2 * n - 1)
        if k_lo < k_hi:
            ks = np.arange(k_lo, k_hi)
            grad[i] = np.dot(b_pm1[ks], a[ks - i])
    grad *= (p - 1)
    # Scale by derivative of (sum b^p)^(1/p) w.r.t. b
    # d/d b_k [(sum b^p)^(1/p)] = (1/p)(sum b^p)^(1/p - 1) * p * b_k^(p-1) = (sum b^p)^(1/p-1) * b_k^(p-1)
    scale = obj ** (1 - p) * p  # Wait let me redo
    # f = (sum b_k^p)^(1/p)
    # df/db_k = (1/p)(sum b^p)^(1/p - 1) * p * b_k^(p-1) = (sum b^p)^(1/p-1) * b_k^(p-1)
    # df/da_i = sum_k (df/db_k)(db_k/da_i) = (sum b^p)^(1/p-1) * sum_k b_k^(p-1) * a[k-i]
    # = (sum b^p)^(1/p-1) * grad_raw[i] where grad_raw[i] = sum_k b_k^(p-1) * a[k-i]
    # But I already computed grad = (p-1) * grad_raw... let me fix
    # Actually I computed grad[i] = (p-1) * sum_k b[k]^(p-1) * a[k-i]... no wait
    # Let me recompute: I set b_pm1 = b^(p-1), then grad[i] = sum b_pm1[k]*a[k-i], then grad *= (p-1)
    # So grad = (p-1) * sum_k b[k]^(p-1) * a[k-i]
    # But I want: df/da_i = (sum b^p)^(1/p - 1) * sum_k b[k]^(p-1) * a[k-i]
    # = (sum b^p)^(1/p - 1) * (grad / (p-1))
    # = obj^(1-p) * (grad/(p-1)) * ... wait
    # (sum b^p)^(1/p - 1) = (sum b^p)^((1-p)/p) = ((sum b^p)^(1/p))^(1-p) = obj^(1-p)
    # So df/da_i = obj^(1-p) * sum_k b[k]^(p-1) * a[k-i] = obj^(1-p) * grad[i] / (p-1)
    grad *= obj ** (1 - p) / (p - 1)
    return obj, grad


def pnorm_lag_obj_and_grad(a, p):
    """
    Objective: 2*n*max(conv(a,a))/(sum(a))^2
    Using p-norm approximation: 2*n*(sum (conv)^p)^(1/p) / (sum(a))^2
    We optimize over the probability simplex (sum(a)=1, a>=0) using log-parametrization.
    Returns (obj, grad_z) where z = log(a).
    """
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e15, np.zeros(n)
    a_norm = a / s
    b = fast_convolve(a_norm)
    b_pos = np.maximum(b, 1e-300)
    max_approx = np.sum(b_pos ** p) ** (1.0 / p)
    obj = 2.0 * n * max_approx
    # Gradient w.r.t. a_norm
    b_pm1 = b_pos ** (p - 1)
    grad_an = np.zeros(n)
    for i in range(n):
        k_lo = max(i, 0)
        k_hi = min(i + n, 2 * n - 1)
        if k_lo < k_hi:
            ks = np.arange(k_lo, k_hi)
            grad_an[i] = np.dot(b_pm1[ks], a_norm[ks - i])
    scale = max_approx ** (1 - p)
    grad_an *= scale * 2.0 * n
    # Chain rule through normalization: a_norm = a/s, so grad_wrt_a = grad_an * (1/s) - a_norm * (grad_an . a_norm)/s^2
    # For simplicity, we work directly in the normalized space
    # Convert to log-space gradient: grad_z = a_norm * (grad_an - sum(grad_an * a_norm))
    S = np.dot(grad_an, a_norm)
    grad_z = a_norm * (grad_an - S)
    return obj, grad_z


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lbgfs_pnorm(a0, time_limit, p_start=16, p_end=4, maxiter=5000):
    """L-BFGS with p-norm objective, decreasing p over stages."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    n_stages = 4
    p_values = np.geomspace(p_start, p_end, n_stages)
    z = z0.copy()
    iters_per_stage = max(100, maxiter // n_stages)

    for p_val in p_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        p_use = max(2.0, p_val)

        def func(z_in, _p=p_use):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_in = softmax_from_z(z_in)
            obj, _ = pnorm_lag_obj_and_grad(a_in, _p)
            return obj

        def jac(z_in, _p=p_use):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_in = softmax_from_z(z_in)
            _, gz = pnorm_lag_obj_and_grad(a_in, _p)
            return gz

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def soft_max_lbgfs(a0, time_limit, T_start=100.0, T_end=1.0, maxiter=5000):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    n_stages = 5
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = max(100, maxiter // n_stages)

    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 5:
            break

        def func(z_in, _T=T_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_in = softmax_from_z(z_in)
            b = fast_convolve(a_in)
            Tb = _T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            lse = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / _T) * lse

        def jac(z_in, _T=T_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_in = softmax_from_z(z_in)
            b = fast_convolve(a_in)
            Tb = _T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * _T * w[k] * a_in[k - idx]
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=10000):
    """Nesterov-accelerated LP best-response game."""
    if solver is None:
        solver = _get_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        t_lp_start = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        lp_times.append(time.monotonic() - t_lp_start)

        # Nesterov step size: alpha = 2/(it+2) for standard, or use sqrt-t schedule
        alpha = 2.0 / (it + 2)

        # Virtual iterate update
        y_new = (1 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            break
        y_new = y_new / sy

        # Weighted average iterate
        a_new = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
            sa = 1.0
        a_new = a_new / sa

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-50:]) if lp_times else 0
            print(f"    NL iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, avg_lp={avg_lp*1000:.0f}ms")

    return best_a, best_score


def peak_shave_aggressive(a0, time_limit, n_steps=20000):
    """Aggressive peak shaving with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the dominant peak (skip endpoints which are smaller)
        b_masked = b.copy()
        b_masked[0] = -np.inf
        b_masked[-1] = -np.inf
        peak_k = int(np.argmax(b_masked))
        peak_val = b[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Find top contributors
        topN = min(8, len(idx))
        top_contrib = np.argsort(contribs)[::-1][:topN]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            # Try reducing a[i] by various amounts, redistribute proportionally
            for fraction in [0.02, 0.05, 0.1, 0.2, 0.35]:
                delta = a[i] * fraction
                if delta < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= delta
                if a_trial[i] < 0:
                    continue
                # Redistribute to other coordinates (proportional to their values)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] += delta * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing two coordinates simultaneously
            if len(top_contrib) >= 2:
                for ci1 in top_contrib[:4]:
                    for ci2 in top_contrib[:4]:
                        if ci1 == ci2:
                            continue
                        i1 = idx[ci1]
                        i2 = idx[ci2]
                        if a[i1] < 1e-14 or a[i2] < 1e-14:
                            continue
                        f1, f2 = 0.1, 0.1
                        a_trial = a.copy()
                        a_trial[i1] *= (1 - f1)
                        a_trial[i2] *= (1 - f2)
                        delta = a[i1] * f1 + a[i2] * f2
                        mask = np.ones(n, dtype=bool)
                        mask[[i1, i2]] = False
                        others = a_trial[mask]
                        if others.sum() < 1e-15:
                            continue
                        a_trial[mask] += delta * others / others.sum()
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial /= ss
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        if not improved:
            # Try a broader perturbation: reduce all top contributors
            top3 = top_contrib[:min(4, len(top_contrib))]
            a_trial = a.copy()
            total_delta = 0
            for ci in top3:
                i = idx[ci]
                a_trial[i] *= 0.9
                total_delta += a[ci] * 0.1
            mask = np.ones(n, dtype=bool)
            for ci in top3:
                mask[idx[ci]] = False
            others = a_trial[mask]
            if others.sum() > 1e-15:
                a_trial[mask] += total_delta * others / others.sum()
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial /= ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and step > 0 and step % 100 == 0:
            # Stagnation: try random perturbation
            a_trial = a + np.random.randn(n) * 0.01
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial /= ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def coordinate_fine(a0, time_limit, n_passes=50):
    """Fine coordinate descent: try small perturbations of each coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        # Randomize order
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            # Try a few multipliers
            for mult in [0.9, 0.95, 1.05, 1.1]:
                a_trial = a.copy()
                a_trial[i] *= mult
                delta = a[i] * (mult - 1)
                if abs(delta) < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] -= delta * a[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    break  # Move to next coordinate

        if p % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc pass={p} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def mirror_descent_active(a0, time_limit, lr_init=0.1, n_steps=20000):
    """Mirror descent using active set subgradient with backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    z = z_from_a(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Active set: indices within 2% of max
        threshold = max_b - 0.02 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Mirror descent in log-space
        z_trial = z - lr * g
        a_trial = softmax_from_z(z_trial)
        score_trial = 2.0 * n * fast_convolve(a_trial).max()

        if score_trial < best_score - 1e-14:
            z = z_trial
            a = a_trial
            best_score = score_trial
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If many failures, reduce lr
            if no_improve > 50:
                lr *= 0.7
                no_improve = 0

        lr = max(lr, 1e-8)
        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.8f} lr={lr:.2e}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver)
        try_update(a_lp, "lp_nesterov")

    # === Phase 2: P-norm L-BFGS (400s) ===
    print(f"\n=== Phase 2: P-Norm L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 400.0)
    if t_alloc > 30:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s, p: 32->4")
        a_pn = lbgfs_pnorm(best_seq, t_alloc, p_start=32, p_end=4, maxiter=10000)
        try_update(a_pn, "pnorm_lbfgs")

    # === Phase 3: Soft-max L-BFGS (300s) ===
    print(f"\n=== Phase 3: Soft-Max L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.17, 300.0)
    if t_alloc > 20:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s, T: 50->0.5")
        a_sm = soft_max_lbgfs(best_seq, t_alloc, T_start=50.0, T_end=0.5, maxiter=8000)
        try_update(a_sm, "softmax_lbfgs")

    # === Phase 4: Aggressive Peak Shaving (250s) ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.14, 250.0)
    if t_alloc > 15:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave_aggressive(best_seq, t_alloc)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Mirror Descent (200s) ===
    print(f"\n=== Phase 5: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.11, 200.0)
    if t_alloc > 15:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_md, _ = mirror_descent_active(best_seq, t_alloc, lr_init=0.05)
        try_update(a_md, "mirror_desc")

    # === Phase 6: Coordinate Descent (200s) ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.11, 200.0)
    if t_alloc > 10:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_cd, _ = coordinate_fine(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")

    # === Phase 7: Final L-BFGS at very low T (100s) ===
    print(f"\n=== Phase 7: Final L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 100.0)
    if t_alloc > 10:
        a_fin = soft_max_lbgfs(best_seq, t_alloc, T_start=2.0, T_end=0.2, maxiter=5000)
        try_update(a_fin, "final_lbfgs")

    # === Phase 8: Final peak shave (50s) ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.5, 50.0)
        if t_ps > 5:
            a_fp, _ = peak_shave_aggressive(best_seq, t_ps, n_steps=5000)
            try_update(a_fp, "final_ps")

    # === Phase 9: Multi-restart with different inits (remaining time) ===
    print(f"\n=== Phase 9: Multi-Restart (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_restarts = min(4, int(rem // 60))
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_start = time.monotonic()
            print(f"  Restart {r+1}/{n_restarts}, rem={rem:.0f}s")

            # Create perturbed initialization
            noise_scale = 0.05 * (r + 1)
            a_init = best_seq + rng.normal(0, noise_scale, size=len(best_seq))
            a_init = np.maximum(a_init, 0)
            s = a_init.sum()
            if s > 1e-15:
                a_init = a_init / s

            # Quick p-norm L-BFGS
            t_alloc_r = min(rem * 0.5, 80.0)
            a_r = lbgfs_pnorm(a_init, t_alloc_r, p_start=16, p_end=4, maxiter=3000)
            try_update(a_r, f"restart_p{r+1}")

            # Quick peak shave
            rem = deadline - time.monotonic()
            if rem > 20:
                t_ps_r = min(rem * 0.4, 30.0)
                if t_ps_r > 5:
                    a_rs, _ = peak_shave_aggressive(best_seq, t_ps_r, n_steps=2000)
                    try_update(a_rs, f"restart_ps{r+1}")

    # === Phase 10: Try different lengths ===
    print(f"\n=== Phase 10: Alt Lengths (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = list(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        print(f"  Alt n={alt_n}, rem={rem:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        t_alloc_alt = min(rem * 0.5, 100.0)
        if t_alloc_alt > 20:
            a_alt = lbgfs_pnorm(init_alt, t_alloc_alt, p_start=16, p_end=4, maxiter=4000)
            try_update(a_alt, f"alt_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and len(best_seq) == alt_n:
            t_ps_alt = min(rem * 0.4, 30.0)
            if t_ps_alt > 5:
                a_ap, _ = peak_shave_aggressive(best_seq, t_ps_alt, n_steps=3000)
                try_update(a_ap, f"alt_ps_n{alt_n}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
