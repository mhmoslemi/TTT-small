Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP with multiple solvers, adaptive peak selection,
peak-shaving local steps, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgradient(b, a, w):
    """Compute weighted subgradient: v[i] = sum_k w[k] * a[k-i] for valid indices."""
    n = len(a)
    v = np.zeros(n)

    # v[i] = sum_k w[k] * a[k-i] where 0 <= k-i < n

        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        v[idx] += w[k] * a[k - idx]
    return v

        K = len(top_idx)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    try:
        prob.solve(solver=solver, max_iters=max_iters, eps=eps)
    except Exception as e:
        try:
            prob.solve(solver=cp.ECOS, max_iters=max_iters, abs_acc=eps, rel_acc=eps)
        except Exception:
            return None, M
    
    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

def line_search(f, h, b_current, M_current, max_iter=20):
    """Backtracking line search."""
    n = len(f)
    best_t = 0.05
    best_obj = M_current
    
    # Try a range of step sizes
    for t in [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:

        f_trial = np.maximum(f_trial, 0.0)

        f_trial /= s
        obj = fast_convolve(f_trial).max()

def multiplicative_descent(a, time_limit, lr_init=0.1, cooling=0.9995, n_top=100, temp_init=0.01):
    """Mirror descent with exponential tilting for peak reduction."""
    t0 = time.monotonic()
    n = len(a)
    f = a.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    temp = temp_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        
        # Softmax weighting over peaks
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros_like(b)
            w[b.argmax()] = 1.0
        
        # Focus on top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_other = w.copy()
            w = np.zeros_like(b)
            w[top_idx] = w_other[top_idx]
            ws = w.sum()
            if ws > 1e-30:
                w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            sg[idx] += w[k] * f[k - idx]
        
        # Multiplicative update
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        f = f * np.exp(-lr * sg / sg_max)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = a.copy()
            break
        f /= s
        
        # Check improvement
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 1.0)
        else:
            no_improve += 1
            if no_improve > 10:
                lr *= 0.9
                no_improve = 0
            if no_improve > 50:
                # Restart from best
                f = best_f.copy()
                temp *= 0.5
                lr = lr_init * 0.5
                no_improve = 0
        
        if it % 200 == 0:
            temp = max(temp * 1.1, 0.001)
            print(f"    MD iter={it} obj={best_obj:.10f} lr={lr:.4f} temp={temp:.5f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj

def peak_shave(f, time_limit, n_shave=10):
    """Directly shave the top peaks by adjusting elements."""
    t0 = time.monotonic()
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        
        # Find top peaks
        top_k = min(n_shave, len(b))
        peak_idx = np.argpartition(b, -top_k)[-top_k:]
        peak_idx = np.sort(peak_idx)
        
        # For each peak, find the elements that contribute most and reduce them
        improved = False
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Contribution of each element i in [lo, hi] to peak k is f[i] * f[k-i]
            contributions = f[lo:hi+1] * f[k - np.arange(lo, hi+1)]
            
            # Find the element with the highest contribution
            max_idx = lo + np.argmax(contributions)
            
            # Try reducing this element
            for factor in [0.7, 0.8, 0.9]:
                f_trial = f.copy()
                f_trial[max_idx] *= factor
                # Redistribute the mass to elements that don't contribute to this peak
                lost_mass = f[max_idx] * (1 - factor)
                # Find elements that don't contribute
                non_contrib = [i for i in range(n) if i < lo or i > hi]
                if non_contrib:
                    # Redistribute proportionally
                    target_sum = f[non_contrib].sum() + lost_mass
                    f_trial[non_contrib] = f[non_contrib] * (target_sum / f[non_contrib].sum())
                
                s = f_trial.sum()
                if s > 1e-10:
                    f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-12:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
        
        if not improved:
            # Try a random perturbation
            rng = np.random.default_rng(it)
            f_trial = f.copy()
            idx = rng.integers(0, n, size=5)
            for i in idx:
                f_trial[i] *= (0.5 + rng.random())
            s = f_trial.sum()
            if s > 1e-10:
                f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-12:
                f = f_trial
                best_obj = obj
                best_f = f.copy()
            else:
                f = best_f.copy()
                break
    
    return best_f, best_obj

"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses minimax LP formulation (all constraints), mirror descent,
and multi-length exploration with tighter solver tolerances.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lp_minimax_all(f, eps=1e-9):
    """
    Minimize t s.t. (f*g)[k] <= t for ALL k, sum(g)=1, g>=0.
    This gives the direction that minimizes worst-case convolution with f.
    """
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [g >= 0, cp.sum(g) == 1]

    # Build all convolution constraints
    # (f*g)[k] = sum_{j=lo}^{hi} f[k-j] * g[j] where lo=max(0,k-n+1), hi=min(n-1,k)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # f[k-j] for j in [lo,hi] = f[k-hi : k-lo+1] reversed = f[lo:hi+1] reversed
        f_rev = f[hi:lo - 1: -1]  # f[k-j] for j=lo..hi
        g_seg = g[lo:hi + 1]
        constraints.append(cp.sum(f_rev * g_seg) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    s = g_val.sum()
    if s < 1e-15:
        return None
    return g_val / s


def lp_topk(f, top_k, solver_name='ECOS', eps=1e-10):
    """
    Minimize t s.t. (f*g)[k] <= t for top-K peaks, sum(g)=1, g>=0.
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()

    # Use peaks above 0.9*M plus top-K
    threshold = 0.9 * M
    above = np.where(b >= threshold)[0]
    if top_k < len(b):
        top_idx_arr = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx_arr = np.arange(len(b))
    top_idx = np.unique(np.concatenate([above, top_idx_arr]))
    # Cap total constraints
    if len(top_idx) > 3 * n:
        top_idx = np.argpartition(b, -top_k)[-top_k:]

    top_idx = np.sort(top_idx)
    K = len(top_idx)

    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [g >= 0, cp.sum(g) == 1]

    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        f_rev = f[hi:lo - 1: -1]
        g_seg = g[lo:hi + 1]
        constraints.append(cp.sum(f_rev * g_seg) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        if solver_name == 'ECOS':
            prob.solve(solver=cp.ECOS, max_iters=500000, abs_acc=eps, rel_acc=eps)
        else:
            prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-8)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    s = g_val.sum()
    if s < 1e-15:
        return None
    return g_val / s


def golden_line_search(f, h, obj_current, max_evals=15):
    """Golden section line search for step size in direction h."""
    n = len(f)
    phi = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate ~0.618

    lo = 0.0
    hi = 1.5

    def eval_obj(t):
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return np.inf
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()

    # First check endpoints
    obj_lo = obj_current
    obj_hi = eval_obj(hi)

    if obj_hi >= obj_lo:
        hi = 0.5
        obj_hi = eval_obj(hi)
        if obj_hi >= obj_lo:
            hi = 0.2
            obj_hi = eval_obj(hi)

    # Golden section search
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    obj_c = eval_obj(c)
    obj_d = eval_obj(d)
    evals = 3

    while hi - lo > 0.01 and evals < max_evals:
        evals += 1
        if obj_c < obj_d:
            hi = d
            d = c
            obj_d = obj_c
            c = hi - phi * (hi - lo)
            obj_c = eval_obj(c)
        else:
            lo = c
            c = d
            obj_c = obj_d
            d = lo + phi * (hi - lo)
            obj_d = eval_obj(d)

    best_t = (lo + hi) / 2.0
    best_obj = eval_obj(best_t)

    # Also try a few fixed points for safety
    for t_try in [0.02, 0.05, 0.1, 0.2, 0.3]:
        o = eval_obj(t_try)
        if o < best_obj:
            best_obj = o
            best_t = t_try
        evals += 1

    return best_t, best_obj


def mirror_descent(f_init, time_limit, lr_init=0.12, n_top=100, temp_init=0.003):
    """
    Multiplicative weights / mirror descent for peak reduction.
    Uses exponential tilting based on peak-weighted subgradient.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    temp = temp_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Softmax weights over peaks
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros_like(b)
            w[b.argmax()] = 1.0

        # Focus on top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_masked = np.zeros_like(b)
            w_masked[top_idx] = w[top_idx]
            ws = w_masked.sum()
            if ws > 1e-30:
                w = w_masked / ws

        # Compute subgradient: sg[i] = sum_k w[k] * f[k-i]
        sg = np.zeros(n)
        active_k = np.where(w > 1e-12)[0]
        for k in active_k:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            sg[idx] += w[k] * f[k - idx]

        # Multiplicative update
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        f = f * np.exp(-lr * sg / sg_max)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = best_f.copy()
            lr *= 0.5
            continue
        f = f / s

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            # Slowly increase lr when making progress
            lr = min(lr * 1.01, 0.5)
        else:
            no_improve += 1
            if no_improve > 5:
                lr *= 0.85
                no_improve = 0
            if no_improve > 40:
                f = best_f.copy()
                temp = max(temp * 0.7, 0.0005)
                lr = max(lr * 0.5, 0.02)
                no_improve = 0

        if it % 100 == 0:
            # Anneal temperature
            temp = min(temp * 1.15, 0.5)
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it} obj={best_obj:.10f} lr={lr:.4f} "
                  f"temp={temp:.5f} t={elapsed:.0f}s")

    return best_f, best_obj


def run_lp_loop_minimax(f_init, time_limit, label="lp_minimax"):
    """LP loop using full minimax formulation (all constraints)."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = lp_minimax_all(f, eps=1e-9)
        if g is None:
            break

        S_g = g.sum()
        if S_g <= 1e-5:
            break

        h = g / S_g
        best_t, best_t_obj = golden_line_search(f, h, best_obj, max_evals=12)

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0 * n * best_obj:.6f} t={elapsed:.0f}s")

    return best_f, best_obj

def coordinate_refine(f, time_limit):
    """Simple coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f)
    best_obj = fast_convolve(f).max()
    f = f.copy()
    multipliers = [0.6, 0.8, 0.9, 1.1, 1.3, 1.6]

    improved_any = True
    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-12:
                continue
            best_j = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                nv = f[j] * mult
                if nv > 0.999:
                    continue
                f[j] = nv
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_j
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j = nv
                else:
                    f[j] = best_j
                    s = f.sum()
                    if s > 1e-10:
                        f /= s
            if best_j != f[j]:
                f[j] = best_j
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved_any = True

    return f, best_obj

            f_long = np.ones(n_long) / n_long
        print(f"  Budget: {t5:.0f}s, n={n_long}")
        f5, obj5 = run_lp_loop_topk(f_long, time_limit=t5, top_k_init=50,
                                     top_k_max=150, label="long_seq",
                                     solver_name='SCS', eps=1e-9)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [long]: {score5:.6f}")

    # === Phase 6: Final minimax LP polish ===
    print(f"\n=== Phase 6: Final minimax LP (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = max(rem - 5.0, 0.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = run_lp_loop_minimax(best_seq, time_limit=t6, label="final_mm")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [final_mm]: {score6:.6f}")

    # === Phase 7: Quick coordinate polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t7 = min(rem - 5.0, 30.0)
        if t7 > 5:
            f7, obj7 = coordinate_refine(best_seq, time_limit=t7)
            score7 = float(evaluate_sequence(f7.tolist()))
            if np.isfinite(score7) and score7 < best_score:
                best_score = score7
                best_seq = f7.copy()
                print(f"  NEW BEST [final_cd]: {score7:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
