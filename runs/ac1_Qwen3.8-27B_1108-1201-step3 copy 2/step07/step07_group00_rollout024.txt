"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with PDL P solver, basin hopping for global
exploration, coordinate descent exploiting scale invariance, and smooth
hinge-loss objective optimization.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_solve_reduced(f, n, top_k, eps=1e-8, solver='PDLP'):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns (g, M) or (None, M) on failure.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=solver, eps=eps)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
        except Exception:
            return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

def run_lp_phase(f_init, time_limit, top_k_init=50, top_k_max=250, label="lp"):
    """LP-based optimization with adaptive top_k and parabolic line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)

        g, M = lp_solve_reduced(f, n, top_k)
        if g is None:
            g, M = lp_solve_reduced(f, n, min(2 * top_k, 2 * n - 1))
            if g is None:
                break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 8:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g
        # Parabolic line search
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()
        t_candidates = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
        
        for t in t_candidates:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

    return best_f, best_obj

def smooth_objective_gradient(f, n, alpha=100.0):
    """
    Compute gradient of smooth approximation: sum_k max(0, b_k/M - 1)^2 * M
    where M = max(b). This encourages all peaks to be at or below M.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return np.zeros(n)
    
    # Soft threshold
    excess = np.maximum(b / M - 1.0, 0.0)
    # Weight by excess
    w = excess * alpha
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(b)) / len(b)
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[k] * f[k - idx]
    
    return grad

"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Fast coordinate descent with O(n) convolution updates per step (vs O(n log n))
2. LP-based direction finding with PDL P solver for improved accuracy
3. Basin hopping with diverse perturbation strategies for global exploration
4. Smooth hinge-loss objective for alternative local optimization
5. Multi-start with different initializations to avoid local minima
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(f):
    """Compute the objective score."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    b = fast_convolve(f)
    return 2.0 * n * b.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve_reduced(f, n, top_k, eps=1e-8, solver_name='PDLP'):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns (g, M) or (None, M) on failure.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=solver_name, eps=eps)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
        except Exception:
            return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def run_lp_phase(f_init, time_limit, top_k_init=50, top_k_max=250, label="lp"):
    """LP-based optimization with adaptive top_k and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 25 == 0 and top_k < top_k_max:
            top_k = min(top_k + 25, top_k_max)

        g, M = lp_solve_reduced(f, n, top_k)
        if g is None:
            g, M = lp_solve_reduced(f, n, min(2 * top_k, 2 * n - 1), solver_name='SCS')
            if g is None:
                break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 8:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g
        # Adaptive line search
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()
        t_candidates = [0.01, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7]

        for t in t_candidates:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

    return best_f, best_obj


def coordinate_descent_fast(f_init, time_limit, step_multipliers=None):
    """
    Coordinate descent with O(n) convolution update per coordinate evaluation.
    Exploits scale invariance of the objective.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        return f, np.inf

    b = fast_convolve(f)
    best_obj = 2.0 * n * b.max() / (s * s)

    if step_multipliers is None:
        step_multipliers = np.array([0.4, 0.6, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.7, 2.5])

    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        max_k = 2 * n - 2

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            a_old = f[j]
            if a_old < 1e-15:
                candidates = [1e-4, 5e-4, 2e-3, 1e-2, 5e-2]
            else:
                candidates = [a_old * m for m in step_multipliers]

            best_j_val = a_old
            best_j_score = best_obj

            lo_k = max(0, j)
            hi_k = min(max_k, j + n - 1)
            k_range = np.arange(lo_k, hi_k + 1)
            mask_noncenter = k_range != (2 * j)
            k_off_center = k_range[mask_noncenter]

            for x in candidates:
                x = max(0.0, x)
                delta = x - a_old
                if abs(delta) < 1e-16:
                    continue
                new_s = s + delta
                if new_s < 1e-10:
                    continue

                # Compute new max efficiently
                # Track the change
                new_max = b.max()
                # Check center
                k_c = 2 * j
                if 0 <= k_c <= max_k:
                    center_val = b[k_c] + x * x - a_old * a_old
                    if center_val > new_max:
                        new_max = center_val
                # Check off-center peaks (only those that increased)
                if delta > 0 and len(k_off_center) > 0:
                    off_vals = b[k_off_center] + 2.0 * delta * f[k_off_center - j]
                    off_max = off_vals.max()
                    if off_max > new_max:
                        new_max = off_max
                elif delta < 0:
                    # Max might decrease or stay; need to check if old max was at a changed location
                    # Simple: just recompute max (expensive but correct)
                    b_temp = b.copy()
                    if 0 <= k_c <= max_k:
                        b_temp[k_c] += x * x - a_old * a_old
                    if len(k_off_center) > 0:
                        b_temp[k_off_center] += 2.0 * delta * f[k_off_center - j]
                    new_max = b_temp.max()

                new_score = 2.0 * n * new_max / (new_s * new_s)

                if new_score < best_j_score - 1e-14:
                    best_j_score = new_score
                    best_j_val = x

            if best_j_val != a_old:
                delta = best_j_val - a_old
                # Update b in place
                if 0 <= k_c <= max_k:
                    b[k_c] += best_j_val * best_j_val - a_old * a_old
                if len(k_off_center) > 0:
                    b[k_off_center] += 2.0 * delta * f[k_off_center - j]

                f[j] = best_j_val
                s += delta
                if best_j_score < best_obj - 1e-14:
                    best_obj = best_j_score
                    improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 3:
                break
        else:
            no_improve_sweeps = 0

        if sweep % 10 == 0:
            print(f"    CD fast sweep={sweep} score={best_obj:.10f} t={time.monotonic()-t0:.0f}s")

    f = np.maximum(f, 0.0)
    s_final = f.sum()
    if s_final > 1e-10:
        f /= s_final
    return f, best_obj


def smooth_gradient(f, n, alpha=200.0):
    """
    Gradient of smooth approximation to max_b:
    Uses softmax-like weights on the peaks to get a smooth gradient.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return np.zeros(n)

    scaled = b / M
    excess = np.maximum(scaled - 0.95, 0.0)
    w = excess * excess * alpha
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fall back to uniform over top peaks
        top_idx = np.argpartition(b, -max(10, n // 5))[-max(10, n // 5):]
        w = np.zeros(len(b))
        w[top_idx] = 1.0 / len(top_idx)
    else:
        w /= w_sum

    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[k] * f[k - idx]

    return grad


def run_smooth_phase(f_init, time_limit, lr_init=0.003):
    """Optimize smooth approximation with momentum and restarts."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    v = np.zeros(n)
    lr = lr_init
    momentum = 0.85
    best_f = f.copy()
    best_score = compute_score(f)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad = smooth_gradient(f, n, alpha=300.0)
        v = momentum * v - lr * grad
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f_new = f.copy()
        f = f_new

        if it % 30 == 0:
            score = compute_score(f)
            if score < best_score - 1e-10:
                best_score = score
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.015)
            else:
                no_improve += 1
                if no_improve % 4 == 0:
                    lr *= 0.92
                if no_improve > 25:
                    # Restart with perturbation
                    sigma = 0.003 + 0.002 * (no_improve // 25)
                    f = best_f + rng.standard_normal(n) * sigma
                    f = np.maximum(f, 0.0)
                    s = f.sum()
                    if s > 1e-10:
                        f /= s
                    v = np.zeros(n)
                    no_improve = 0

        if it % 300 == 0:
            print(f"    Smooth iter={it} score={best_score:.8f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def basin_hopping(f_init, time_limit, rng, num_restarts=5):
    """Basin hopping with diverse perturbations and local optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    best_f = f_init.copy()
    best_score = compute_score(f_init)

    strategies = ['gauss', 'sparse', 'mix_uniform', 'block', 'gauss']
    sigma_values = [0.005, 0.01, 0.02, 0.03, 0.05]

    for restart in range(num_restarts):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        per_restart = remaining / (num_restarts - restart)

        strategy = strategies[restart % len(strategies)]
        sigma = sigma_values[restart % len(sigma_values)]

        if strategy == 'gauss':
            f_pert = f_init + rng.standard_normal(n) * sigma
        elif strategy == 'sparse':
            f_pert = f_init.copy()
            num_changes = max(5, n // 50)
            idx = rng.choice(n, size=num_changes, replace=False)
            for i in idx:
                f_pert[i] *= (1.0 + rng.uniform(-0.5, 0.5))
            f_pert = np.maximum(f_pert, 0.0)
        elif strategy == 'mix_uniform':
            mix = 0.05 + 0.05 * (restart % 3)
            f_pert = (1.0 - mix) * f_init + mix * (np.ones(n) / n)
        elif strategy == 'block':
            f_pert = f_init.copy()
            block_size = max(10, n // 20)
            start = rng.integers(0, n - block_size)
            f_pert[start:start + block_size] *= (1.0 + rng.uniform(-0.3, 0.3))
            f_pert = np.maximum(f_pert, 0.0)
        else:
            f_pert = f_init + rng.standard_normal(n) * sigma

        f_pert = np.maximum(f_pert, 0.0)
        s = f_pert.sum()
        if s > 1e-10:
            f_pert /= s

        # Try LP on perturbed point
        t_lp = min(per_restart * 0.5, 40.0)
        if t_lp > 5:
            f_opt, _ = run_lp_phase(f_pert, time_limit=t_lp, top_k_init=40,
                                     top_k_max=120, label=f"bh_{restart}")
            score = compute_score(f_opt)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_f = f_opt.copy()
                f_init = best_f.copy()
                print(f"    BH[{strategy}] r{restart}: {best_score:.8f}")

        # Try fast coordinate descent on perturbed point
        t_cd = min(per_restart * 0.4, 30.0)
        if t_cd > 3:
            f_cd, _ = coordinate_descent_fast(f_pert, time_limit=t_cd)
            score_cd = compute_score(f_cd)
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_f = f_cd.copy()
                f_init = best_f.copy()
                print(f"    BH-CD[{strategy}] r{restart}: {best_score:.8f}")

    return best_f, best_score


def try_different_length(f_init, target_n, time_limit):
    """Try optimization at a different sequence length via interpolation."""
    n_old = len(f_init)
    old_x = np.linspace(0.0, 1.0, n_old)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_interp = np.interp(new_x, old_x, f_init)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-10:
        f_interp /= s
    else:
        f_interp = np.ones(target_n) / target_n

    f_opt, _ = run_lp_phase(f_interp, time_limit=time_limit,
                            top_k_init=40, top_k_max=150,
                            label=f"len{target_n}")
    f_opt, _ = coordinate_descent_fast(f_opt, time_limit=time_limit * 0.3)
    return f_opt
