"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth max (log-sum-exp) optimization in log-space with L-BFGS
2. LP-based refinement with decreasing bounds
3. Multi-length exploration with diverse initializations
4. Adaptive restarts

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import scipy.sparse as sp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def max_conv(a):
    """Maximum of auto-convolution."""
    return float(fast_convolve(a).max())


def logsumexp_obj_grad(x, beta):
    """
    Objective and gradient in log-space.
    x: log-weights (unconstrained)
    a = softmax(x)
    f = logsumexp(beta * conv(a,a)) / beta
    """
    n = len(x)
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()
    
    b = fast_convolve(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    f = b_max + np.log(exp_shifted.sum()) / beta
    
    w = exp_shifted / exp_shifted.sum()
    
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (valid range)
    # = 2 * cross_correlate(w, a)[j]
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = irfft(np.conj(W) * A, fft_size)[:n]
    grad_a *= 2.0
    
    # Transform to x-space
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return f, grad_x


def optimize_logspace(x0, beta_list, max_iter=500, gtol=1e-10):
    """
    Optimize in log-space with increasing beta.
    Returns the best a found (normalized to sum=1).
    """
    best_a = None
    best_obj = float('inf')
    x = x0.copy().astype(np.float64)
    
    for beta in beta_list:
        res = minimize(logsumexp_obj_grad, x, args=(beta,),
                      method='L-BFGS-B',
                      options={'maxiter': max_iter, 'gtol': gtol, 'ftol': 1e-15})
        x = res.x
        
        # Convert to a
        x_shift = x - x.max()
        a = np.exp(x_shift)
        a /= a.sum()
        
        obj = max_conv(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def lp_refine(a, M_scale=1.0, max_iters=20, time_limit=60.0):
    """
    LP-based refinement.
    a: normalized sequence (sum=1)
    M_scale: scale factor for the bound (use <1 for tighter bound)
    """
    t0 = time.monotonic()
    n = len(a)
    a_cur = a.copy().astype(np.float64)
    a_cur /= a_cur.sum()
    best_a = a_cur.copy()
    best_obj = max_conv(a_cur)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = max_conv(a_cur) * M_scale
        
        # Build constraint matrix (sparse)
        # Row k: sum_j a_cur[k-j] * g[j] <= M, for j where k-j is valid
        rows = []
        cols = []
        vals = []
        
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a_cur[k - j])
        
        A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs')
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            # Try blending
            improved = False
            for t in [0.5, 0.3, 0.2, 0.1, 0.7, 0.8, 1.0, 0.15, 0.4]:
                blend = (1 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend /= s
                obj = max_conv(blend)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = blend.copy()
                    improved = True
                    break
            
            if not improved:
                break
            
            a_cur = best_a.copy()
            
        except Exception:
            break
    
    return best_a, best_obj

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp /= s
    else:
        a_interp = np.ones(new_n) / new_n
    return a_interp

    print(f"\n{'='*60}")
    print(f"Phase 1: Log-space L-BFGS on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        # Initialize x from parent
        x0 = np.log(np.maximum(parent, 1e-30))
        beta_list = [10, 50, 100, 200, 500]
        max_iter_per_beta = max(100, int(rem / len(beta_list) / 0.5))
        
        a_opt, obj_opt = optimize_logspace(x0, beta_list, max_iter=max_iter_per_beta)
        update_best(a_opt, "lbfgs_parent")
    
    # === Phase 2: LP refinement on best ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP Refinement (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_cur = len(best_seq)
        rem = deadline - time.monotonic() - 400
        if rem > 10 and n_cur <= 3000:
            t_lp = min(rem * 0.5, 120.0)
            a_lp, obj_lp = lp_refine(best_seq, M_scale=1.0, max_iters=30, time_limit=t_lp)
            update_best(a_lp, "lp_parent")
            
            # Tighter LP
            rem = deadline - time.monotonic() - 200
            if rem > 5 and best_seq is not None and len(best_seq) <= 3000:
                a_lp2, obj_lp2 = lp_refine(best_seq, M_scale=0.98, max_iters=20, time_limit=min(rem, 60.0))
                update_best(a_lp2, "lp_tight")
    
    # === Phase 3: Diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse Initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent
    
    inits = {
        "gauss": make_init_type1,
        "cosine": make_init_type2,
        "bell4": make_init_type3,
        "twopeak": make_init_type4,
        "exp_center": make_init_type5,
        "perturbed_unif": make_init_type6,
    }
    
    total_time_diverse = min(300.0, (deadline - time.monotonic() - 300.0) * 0.4)
    time_per_init = total_time_diverse / len(inits)
    
    for name, init_fn in inits.items():
        rem = deadline - time.monotonic() - 250
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.3)
        if t_alloc < 8:
            continue
        
        a_init = init_fn(n_base)
        x0 = np.log(np.maximum(a_init, 1e-30))
        beta_list = [20, 100, 300]
        max_iter = max(50, int(t_alloc / len(beta_list) / 0.3))
        
        a_div, obj_div = optimize_logspace(x0, beta_list, max_iter=max_iter)
        update_best(a_div, f"div_{name}")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length Exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [1.5, 2.0, 0.75, 1.25, 3.0, 0.5]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths and tl <= 3000:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))
        
        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            
            f_interp = interpolate_to_length(best_seq, target_n)
            x0 = np.log(np.maximum(f_interp, 1e-30))
            beta_list = [20, 100, 300]
            t_alloc = min(rem * 0.2, 60.0)
            max_iter = max(50, int(t_alloc / len(beta_list) / 0.3))
            
            a_len, obj_len = optimize_logspace(x0, beta_list, max_iter=max_iter)
            update_best(a_len, f"len_n{target_n}")
            
            # Quick LP on the result
            rem = deadline - time.monotonic() - 150
            if rem > 5 and best_seq is not None and len(best_seq) <= 3000:
                a_len_lp, _ = lp_refine(best_seq, M_scale=1.0, max_iters=10, time_limit=min(rem, 30.0))
                update_best(a_len_lp, f"len_n{target_n}_lp")
