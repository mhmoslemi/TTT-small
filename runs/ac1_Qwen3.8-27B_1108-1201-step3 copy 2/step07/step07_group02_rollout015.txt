            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 5
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': iters_per_stage * 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def fictitious_play_nesterov(a0, time_limit, solver=None, n_max_iters=50000):
    """
    Self-play fictitious play with Nesterov-style time averaging.
    At each iteration: a_{t+1} = LP_best_response(a_t), then update average.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    avg = a.copy()
    count = 0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    last_lp_time = time.monotonic()
    lp_count = 0

    for t in range(1, n_max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        a_new, M_val = solve_lp_best_response(a, solver)
        if a_new is None:
            continue
        lp_count += 1

        # Nesterov step size
        alpha = 2.0 / (t + 2)

        # Update iterate
        a = (1 - alpha) * a + alpha * a_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = a_new.copy()
            sa = 1.0
        a = a / sa

        # Update time average
        count += 1
        avg = avg + (a - avg) / count

        # Check best
        for candidate in [a, avg, a_new]:
            c = np.maximum(candidate, 0)
            sc_sum = c.sum()
            if sc_sum < 1e-15:
                continue
            c = c / sc_sum
            sc = 2.0 * n * fast_convolve(c).max()
            if sc < best_score:
                best_score = sc
                best_a = c.copy()

        if lp_count % 200 == 0:
            elapsed = time.monotonic() - t0
            lp_speed = lp_count / max(elapsed, 0.01)
            print(f"    FP n={n} lps={lp_count} ({elapsed:.0f}s, {lp_speed:.1f} lp/s): best={best_score:.6f}")

    return best_a, best_score


def symmetric_fictitious_play(a0, time_limit, solver=None, n_max_iters=50000):
    """
    Fictitious play that enforces symmetry: at each step, symmetrize the iterate.
    a[i] = (a[i] + a[n-1-i]) / 2
    This reduces the effective search space and can accelerate convergence.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    def symmetrize(x):
        return (x + x[::-1]) / 2

    a = symmetrize(a)
    avg = a.copy()
    count = 0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lp_count = 0

    for t in range(1, n_max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        a_new, M_val = solve_lp_best_response(a, solver)
        if a_new is None:
            continue
        lp_count += 1

        # Symmetrize the LP solution
        a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
            sn = 1.0
        a_new = a_new / sn

        # Nesterov step
        alpha = 2.0 / (t + 2)
        a = symmetrize((1 - alpha) * a + alpha * a_new)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = a_new.copy()
        else:
            a = a / sa

        count += 1
        avg = avg + (a - avg) / count

        for candidate in [a, avg, a_new]:
            c = symmetrize(np.maximum(candidate, 0))
            sc_sum = c.sum()
            if sc_sum < 1e-15:
                continue
            c = c / sc_sum
            sc = 2.0 * n * fast_convolve(c).max()
            if sc < best_score:
                best_score = sc
                best_a = c.copy()

        if lp_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SymFP n={n} lps={lp_count} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=10000):
    """Aggressive peak shaving with smart redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 500:
            break

        b = fast_convolve(a)
        # Find the interior peak (exclude edges which are naturally lower)
        b_mid = b.copy()
        b_mid[0] = -np.inf
        b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.92, 0.88, 0.82]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Try reducing two contributors
            if len(top_contrib) >= 2:
                done = False
                for ci in top_contrib[:5]:
                    if done:
                        break
                    i = idx[ci]
                    j = peak_k - i
                    if i == j or a[j] < 1e-14:
                        continue
                    for f1 in [0.96, 0.93]:
                        for f2 in [0.96, 0.93]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            st = a_trial.sum()
                            if st < 1e-15:
                                continue
                            a_trial = a_trial / st
                            sc = 2.0 * n * fast_convolve(a_trial).max()
                            if sc < best_score - 1e-12:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                done = True
                                break
                        if done:
                            break
            if not improved:
                # Try reducing all top 4 slightly
                a_trial = a.copy()
                for ci in top_contrib[:4]:
                    i = idx[ci]
                    a_trial[i] *= 0.97
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                if not improved:
                    stall += 1

        if step % 2000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def mirror_descent_subgrad(a0, time_limit, lr_init=0.05, n_steps=50000):
    """Mirror descent with subgradient of max-conv, using soft active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Soft active set: use all indices above 95% of max
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        z = z - lr * g
        a_new = softmax_from_z(z)

        sc_new = 2.0 * n * fast_convolve(a_new).max()
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            z = z + lr * g

        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
        lr = max(lr, 1e-7)

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def generate_initializations(n, rng):
    """Generate diverse structured initializations of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window
    x = np.linspace(0, np.pi, n)
    w = np.cos(x) ** 2
    w = np.maximum(w, 1e-10)
    inits.append(w / w.sum())

    # Hann window
    x = np.linspace(0, 1, n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x))
    w = np.maximum(w, 1e-10)
    inits.append(w / w.sum())

    # Raised cosine (flat top)
    alpha = 0.3
    x = np.linspace(0, 1, n)
    w = np.ones(n)
    w[:int(alpha * n)] = 0.5 * (1 + np.cos(np.pi * x[:int(alpha * n)] / alpha))
    w[-int(alpha * n):] = 0.5 * (1 + np.cos(np.pi * x[-int(alpha * n):] / alpha))
    w = np.maximum(w, 1e-10)
    inits.append(w / w.sum())

    # Random smoothed
    r = rng.random(n)
    # Smooth with a small kernel
    k = 5
    kernel = np.ones(k) / k
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 1e-10)
    inits.append(r_smooth / r_smooth.sum())

    # Quasi-random (van der Corput-like)
    golden = (1 + 5 ** 0.5) / 2
    vals = []
    for i in range(n):
        vals.append((i * golden) % 1.0)
    # Sort and use as density
    vals = np.sort(vals)
    # Make a density from the gaps
    gaps = np.diff(np.concatenate([[0], vals, [1]]))
    # This gives a rough density estimate
    w = gaps + 1e-10
    inits.append(w / w.sum())

    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Diverse initialization screening with quick LP
    2. Main LP-based fictitious play (longest phase)
    3. L-BFGS soft-max polishing
    4. Peak shaving
    5. Final LP rounds
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Diverse initialization screening (120s) ===
    print(f"\n=== Phase 1: Initialization Screening (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_screen = min(rem * 0.08, 120.0)
    if t_screen > 10:
        # Try the parent's n and a couple of nearby values
        candidate_ns = [n_best]
        if n_best < 1024:
            candidate_ns.append(1024)
        if n_best < 1536:
            candidate_ns.append(1536)

        for test_n in candidate_ns[:3]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            inits = generate_initializations(test_n, rng)
            # If parent exists and has this length, include it
            if parent is not None and len(parent) == test_n:
                inits[0] = parent.copy()

            best_init = None
            best_init_score = float('inf')
            per_init_time = max(5.0, t_screen / (len(candidate_ns) * len(inits)))

            for ii, init in enumerate(inits):
                rem = deadline - time.monotonic()
                if rem < 5:
                    break
                # Quick LP: 20 iterations
                t_quick = min(per_init_time, 30.0)
                a_quick, sc_quick = fictitious_play_nesterov(
                    init, t_quick, solver=solver, n_max_iters=30
                )
                if sc_quick < best_init_score:
                    best_init_score = sc_quick
                    best_init = a_quick.copy()

            if best_init is not None and best_init_score < best_score:
                best_seq = best_init.copy()
                best_score = best_init_score
                n_best = len(best_seq)
                print(f"  Phase1: n={n_best}, score={best_score:.6f}")

    # === Phase 2: Main LP fictitious play (900s) ===
    print(f"\n=== Phase 2: Main LP Fictitious Play (best={best_score:.6f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_main = min(rem * 0.55, 900.0)
    if t_main > 30:
        print(f"  Running FP: n={n_best}, budget={t_main:.0f}s")
        a_fp, sc_fp = fictitious_play_nesterov(best_seq, t_main, solver=solver, n_max_iters=100000)
        try_update(a_fp, "fp_main")

        # Also try symmetric version
        rem = deadline - time.monotonic()
        t_sym = min(rem * 0.3, 200.0)
        if t_sym > 20:
            print(f"  Running SymFP: n={n_best}, budget={t_sym:.0f}s")
            a_sym, sc_sym = symmetric_fictitious_play(best_seq, t_sym, solver=solver, n_max_iters=50000)
            try_update(a_sym, "sym_fp")

    # === Phase 3: L-BFGS polishing (300s) ===
    print(f"\n=== Phase 3: L-BFGS Polishing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_lb = min(rem * 0.25, 300.0)
    if t_lb > 20:
        print(f"  L-BFGS T: 50->2, n={len(best_seq)}, budget={t_lb:.0f}s")
        a_lb = lbgfs_soft_max(best_seq, t_lb, T_start=50.0, T_end=2.0, maxiter=6000)
        try_update(a_lb, "lbgfs_main")

        # Second pass at lower temp
        rem = deadline - time.monotonic()
        t_lb2 = min(rem * 0.15, 150.0)
        if t_lb2 > 15:
            print(f"  L-BFGS T: 2->0.5, n={len(best_seq)}, budget={t_lb2:.0f}s")
            a_lb2 = lbgfs_soft_max(best_seq, t_lb2, T_start=2.0, T_end=0.5, maxiter=5000)
            try_update(a_lb2, "lbgfs_lowT")

    # === Phase 4: Peak shaving (200s) ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_ps = min(rem * 0.15, 200.0)
    if t_ps > 10:
        print(f"  PeakShave n={len(best_seq)}, budget={t_ps:.0f}s")
        a_ps, sc_ps = peak_shave_local(best_seq, t_ps, n_steps=10000)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Mirror descent refinement (150s) ===
    print(f"\n=== Phase 5: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_md = min(rem * 0.12, 150.0)
    if t_md > 10:
        a_md, sc_md = mirror_descent_subgrad(best_seq, t_md, lr_init=0.03)
        try_update(a_md, "mirror_desc")

    # === Phase 6: Final LP rounds (remaining time) ===
    print(f"\n=== Phase 6: Final LP Rounds (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_final = min(rem * 0.7, 200.0)
        print(f"  Final LP: n={len(best_seq)}, budget={t_final:.0f}s")
        a_final, sc_final = fictitious_play_nesterov(
            best_seq, t_final, solver=solver, n_max_iters=50000
        )
        try_update(a_final, "final_lp")

        # Final peak shave
        rem = deadline - time.monotonic()
        if rem > 10:
            t_ps2 = min(rem * 0.8, 50.0)
            a_ps2, _ = peak_shave_local(best_seq, t_ps2, n_steps=5000)
            try_update(a_ps2, "final_ps")

    # === Phase 7: Try a different n if time permits ===
    print(f"\n=== Phase 7: Alternative n (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        alt_n = int(n_best * 1.25) if n_best < 1536 else int(n_best * 0.8)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            t_alt = min(rem * 0.5, 120.0)
            if t_alt > 20:
                print(f"  Alt n={alt_n}, budget={t_alt:.0f}s")
                init_alt = interpolate_sequence(best_seq, alt_n)
                a_alt, _ = fictitious_play_nesterov(init_alt, t_alt, solver=solver, n_max_iters=30000)
                try_update(a_alt, f"alt_n{alt_n}")

                # Quick polish
                rem = deadline - time.monotonic()
                if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
                    t_pol = min(rem * 0.5, 60.0)
                    a_pol = lbgfs_soft_max(best_seq, t_pol, T_start=10.0, T_end=1.0, maxiter=3000)
                    try_update(a_pol, f"alt_polish_n{alt_n}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
