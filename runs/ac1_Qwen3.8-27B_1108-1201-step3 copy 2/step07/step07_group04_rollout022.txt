"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP fictitious play with Polyak-Ruppert averaging at multiple lengths
2. L-BFGS smooth-max polishing with adaptive temperature sweep
3. Peak-focused local search with adaptive line search
4. Random perturbation + re-optimize to escape local minima
5. Multi-length exploration with interpolation

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] subject to b>=0, sum(b)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_score(a):
    """Compute the objective score for a normalized sequence (sum=1)."""
    n = len(a)
    max_conv = fast_convolve(a).max()
    return 2.0 * n * max_conv


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):
    """
    Run LP fictitious play with Polyak-Ruppert averaging.
    Uses both uniform average and exponential-weighted average, tracking best candidate.
    """
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    # Also track exponential average
    a_exp = a.copy().astype(np.float64)
    b_exp = b.copy().astype(np.float64)
    exp_rate = 0.01

    best_score = compute_score(a)
    best_a = a.copy()
    iters = 0
    M_a = None
    M_b = None
    stagnation = 0

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Polyak-Ruppert averaging: weight = 2/(t+2)
        alpha_pr = 2.0 / (iters + 2)
        a_avg = (1 - alpha_pr) * a_avg + alpha_pr * a
        b_avg = (1 - alpha_pr) * b_avg + alpha_pr * b

        # Exponential averaging
        a_exp = (1 - exp_rate) * a_exp + exp_rate * a
        b_exp = (1 - exp_rate) * b_exp + exp_rate * b

        # Evaluate multiple candidates
        candidates = [a, b, 0.5 * (a + b), a_avg, b_avg, 0.5 * (a_avg + b_avg), a_exp, b_exp]
        improved_this_iter = False
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            sc = compute_score(cand_norm)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand_norm.copy()
                improved_this_iter = True
                stagnation = 0

        if not improved_this_iter:
            stagnation += 1

        # Perturbation to escape stagnation
        if stagnation > 300:
            noise_scale = 0.002
            a = best_a + np.random.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = best_a + np.random.standard_normal(n) * noise_scale
            b = np.maximum(b, 0)
            b = b / b.sum()
            a_avg = a.copy()
            b_avg = b.copy()
            a_exp = a.copy()
            b_exp = b.copy()
            stagnation = 0

        if verbose and iters % 50 == 0:
            elapsed = monotonic() - t0
            ma_str = f"{M_a:.6f}" if M_a is not None else "N/A"
            mb_str = f"{M_b:.6f}" if M_b is not None else "N/A"
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): M_a={ma_str} M_b={mb_str} best={best_score:.6f}")

    return best_a, best_score, iters


def lbgfs_smooth(a0, time_limit, T=50, maxiter=20000):
    """L-BFGS refinement using smooth max approximation in the probability simplex."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / (T * 1.0)  # Divide by T to get actual smooth max

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        # Gradient: d/da_m [2n * sum_k w_k * c_k] where c_k = sum_i a_i a_{k-i}
        # dc_k/da_m = 2 * a_{k-m} (if valid)
        # So grad_a[m] = 2n * 2 * sum_{k valid} w_k * a_{k-m}
        # = 4n * cross-correlation of w and a
        grad_a = np.zeros(n)
        # Only loop over significant w values
        thresh = 1e-10
        significant = np.where(w > thresh)[0]
        for k in significant:
            wk = w[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * wk * a[k - idx]

        grad_a *= 2.0 * n
        # Transform from a-space to z-space (log-sum-exp parameterization)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()

    return a_out


def peak_shave(a0, time_limit):
    """Aggressive local peak shaving with multiple strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = monotonic()
    failed = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        if b[peak_k] <= best_score / (2.0 * n) + 1e-15:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        top_contrib = np.argsort(contribs)[::-1][:5]
        improved = False

        # Strategy 1: Reduce single contributor and redistribute
        for ci in top_contrib:
            if improved:
                break
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.97, 0.93, 0.88, 0.8]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        # Strategy 2: Reduce a pair symmetrically around the peak
        if not improved:
            for ci in top_contrib:
                if improved:
                    break
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        # Strategy 3: Local 3-point adjustment
        if not improved:
            i = int(idx[top_contrib[0]])
            for di in [-2, -1, 1, 2]:
                j = i + di
                if j < 0 or j >= n:
                    continue
                for fr in [1.05, 0.95, 1.1, 0.9]:
                    a_trial = a.copy()
                    a_trial[i] /= fr  # reduce i
                    a_trial[j] *= fr  # increase j
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Strategy 4: Small random perturbation
        if not improved:
            for attempt in range(3):
                scale = 0.001
                a_trial = a + np.random.standard_normal(n) * scale
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 100:
                break
        else:
            failed = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score

        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def random_restart_optimize(a0, n_restarts, time_per_restart, solver=None):
    """Try random perturbations of a0 and re-optimize briefly."""
    n = len(a0)
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = monotonic()

    for r in range(n_restarts):
        if monotonic() - t0 > n_restarts * time_per_restart:
            break
        # Perturb
        scale = 0.01 + 0.02 * (r % 3)
        a_pert = a0 + np.random.standard_normal(n) * scale
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Brief FP
        a_fp, sc_fp, _ = fictitious_play(n, time_per_restart * 0.6, initial=a_pert, solver=solver, verbose=False)
        if sc_fp < best_score:
            best_score = sc_fp
            best_a = a_fp.copy()

        # Brief L-BFGS
        a_lb = lbgfs_smooth(best_a, time_per_restart * 0.3, T=30, maxiter=2000)
        sc_lb = compute_score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_a = a_lb.copy()

    return best_a, best_score

            t_per = rem / n_restarts * 0.7

            a_rr, _ = random_restart_optimize(best_seq, n_restarts, t_per, solver=solver)
            best_seq, best_score = check_and_update(a_rr, "random_restart", best_seq, best_score)

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 5
        if rem > 10:
            t_final = min(rem * 0.5, 25.0)
            a_final = lbgfs_smooth(best_seq, t_final, T=5, maxiter=10000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs_T5", best_seq, best_score)

        rem = deadline - monotonic() - 3
        if rem > 8:
            a_ps2, _ = peak_shave(best_seq, min(rem * 0.6, 10.0))
            best_seq, best_score = check_and_update(a_ps2, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
