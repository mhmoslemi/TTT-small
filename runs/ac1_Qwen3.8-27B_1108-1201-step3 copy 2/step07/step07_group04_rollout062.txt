**Frequency-domain iteration (Grimson-type):**

The idea: minimize `max_k c[k]` where `c = a*a` and `a >= 0`, `sum(a) = 1`.

In the frequency domain: `C(f) = |A(f)|^2` (power spectrum), and `c[k] = IDFT(C)[k]`.

We want `c[k]` to be as flat as possible. A perfectly flat `c` would be `c[k] = 1/(2n-1)` for all `k`, which corresponds to `C(f) = (1/(2n-1)) * (sum of DFT of delta) = (1/(2n-1)) * n * (1,1,...,1)` ... hmm, this doesn't quite work because `c` has length `2n-1` while the DFT has length `n` (or whatever FFT size we use).

Let me think about this differently. If we use an FFT of size `N >= 2n-1`, then the circular convolution of `a` with itself (padded to size N) has DFT `|A_N(f)|^2` where `A_N` is the N-point DFT of the padded `a`.

The linear convolution is the first `2n-1` samples of the circular convolution (when `N >= 2n-1`). So if we minimize `max_k c_lin[k]`, we're approximately minimizing `max_k c_circ[k]` (if the circular wrap-around is negligible).

The "flat" target for `c_circ` is `c_circ[k] = 1/N` for all `k` (since `sum c_circ = (sum a)^2 = 1`, and there are N terms). This corresponds to `|A_N(f)|^2 = N * (1/N)^2 * N = 1` for all `f`... let me just say: flat autocorrelation corresponds to flat power spectrum.

So the algorithm:
1. Compute `A = FFT(a, N)` where `N >= 2n-1`.
2. Compute target `|A_target| = 1/sqrt(N)` (flat spectrum).
3. Blend: `|A_new| = |A| * (|A_target| / |A|)^alpha = |A|^(1-alpha) * |A_target|^alpha`.
4. `A_new = |A_new| * exp(i*angle(A))`.
5. `a_new = IFFT(A_new)[:n]` (take first n samples).
6. `a_new = max(a_new, 0)`.
7. `a_new = a_new / sum(a_new)`.
8. Repeat.

The parameter `alpha` controls the aggressiveness. `alpha=1` means fully flat (too aggressive), `alpha=0` means no change. Start with `alpha=0.5` and decrease over iterations.

This is essentially the " Iterative Amplitude and Phase Retrieval" (IAPR) or similar algorithms used in optics and signal processing.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with O(n) incremental convolution updates per coordinate
2. Frequency-domain (flat-spectrum) iteration for diversity
3. Smoothed gradient polishing with adaptive temperature
4. Multi-length and multi-initialization exploration
5. Symmetrization for final improvement
"""

import numpy as np
from scipy.fft import rfft, irfft
from time import monotonic
import math


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * max(c.max(), 0.0) / (s * s)

def coord_descent(a_init, time_budget, start_time, verbose=False, seed=42):
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    c = fast_convolve(a)
    best_score = 2.0 * n * max(c.max(), 0) / (S * S)
    best_a = a.copy()
    stuck = 0
    rng = np.random.RandomState(seed)

    for sweep in range(1000000):
        if monotonic() - start_time > time_budget:
            break
        order = rng.permutation(n)
        improved_this_sweep = False
        for j in order:
            old = a[j]
            mean_a = S / n
            if old < 1e-12:
                cands = [mean_a * 0.5, mean_a, mean_a * 1.5, mean_a * 0.25]
            else:
                cands = [0.0, old * 0.4, old * 0.7, old * 1.4, old * 2.0,
                         mean_a, mean_a * 2.0, mean_a * 0.5, old * 5.0]

            cur_s = 2.0 * n * max(c.max(), 0) / (S * S)
            best_x = old
            best_s = cur_s

            for x in cands:
                if x < 0 or not np.isfinite(x):
                    continue
                d = x - old
                if abs(d) < 1e-15:
                    continue
                ks = j
                ke = min(j + n, 2 * n - 1)
                if ke > ks:
                    c[ks:ke] += 2.0 * d * a[ks - j:ke - j]
                if 2 * j < 2 * n - 1:
                    c[2 * j] += d * d
                new_S = S + d
                if new_S > 1e-15:
                    s_val = 2.0 * n * max(c.max(), 0) / (new_S * new_S)
                else:
                    s_val = float('inf')
                if ke > ks:
                    c[ks:ke] -= 2.0 * d * a[ks - j:ke - j]
                if 2 * j < 2 * n - 1:
                    c[2 * j] -= d * d
                if s_val < best_s - 1e-15:
                    best_s = s_val
                    best_x = x

            if best_x != old:
                d = best_x - old
                ks = j
                ke = min(j + n, 2 * n - 1)
                if ke > ks:
                    c[ks:ke] += 2.0 * d * a[ks - j:ke - j]
                if 2 * j < 2 * n - 1:
                    c[2 * j] += d * d
                a[j] = best_x
                S += d
                if best_s < best_score - 1e-14:
                    best_score = best_s
                    best_a = a.copy()
                    stuck = 0
                    improved_this_sweep = True
                else:
                    stuck += 1

        if not improved_this_sweep:
            stuck += n
        if stuck > 5 * n and sweep > 20:
            # Perturb to escape local stagnation
            scale = max(best_a.mean(), 1e-6)
            a = best_a + rng.standard_normal(n) * 0.005 * scale
            a = np.maximum(a, 0)
            S = a.sum()
            if S < 1e-15:
                a = np.ones(n) / n
                S = 1.0
            c = fast_convolve(a)
            stuck = 0

        if verbose and (sweep + 1) % 500 == 0:
            print(f"    CD sweep={sweep+1} score={best_score:.8f} stuck={stuck}")

    return best_a, best_score


def freq_iteration(a_init, time_budget, start_time, verbose=False):
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return a, float('inf')
    a = a / S
    N = 1 << (2 * n - 1).bit_length()
    best_score = compute_score(a)
    best_a = a.copy()
    alpha = 0.7

    for it in range(200000):
        if monotonic() - start_time > time_budget:
            break
        A = rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        target = np.ones(N // 2 + 1)
        safe_mag = np.maximum(mag, 1e-20)
        new_mag = safe_mag * (target / safe_mag) ** alpha
        A_new = new_mag * np.exp(1j * phase)
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
        alpha = max(0.1, 0.7 * (0.9995 ** it))
        if verbose and (it + 1) % 2000 == 0:
            print(f"    Freq iter={it+1} score={best_score:.8f} alpha={alpha:.4f}")

    return best_a, best_score


def smoothed_gradient(a_init, time_budget, start_time, T_init=50.0, verbose=False):
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-30)
    S = a.sum()
    a = a / S
    best_score = compute_score(a)
    best_a = a.copy()
    T = T_init
    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.9

    for it in range(100000):
        if monotonic() - start_time > time_budget:
            break
        c = fast_convolve(a)
        shifted = (c - c.max()) / T
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()

        grad = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[k] * a[k - idx]

        grad = grad / grad.sum() if grad.sum() > 0 else np.ones(n) / n
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        lr *= 0.9995
        T *= 0.999

        if verbose and (it + 1) % 5000 == 0:
            print(f"    Smooth it={it+1} score={best_score:.8f} T={T:.2f} lr={lr:.6f}")

    return best_a, best_score


def symmetrize(a):
    return (a + a[::-1]) / 2.0


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
