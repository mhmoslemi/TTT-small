"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max (log-sum-exp) gradient descent with Adam optimizer and temperature annealing
2. LP-based damped refinement (fictitious play)
3. Coordinate-descent local search on peak-contributing elements
4. Multi-length exploration with interpolation

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute 2*n*max_conv for a sequence with sum=1."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_n = a / s
    return 2.0 * n * fast_convolve(a_n).max()


def soft_max_value_and_grad(a, T):
    """
    Compute soft-max value and gradient of (1/T)*log(sum exp(T*conv(a,a)))
    for a with sum(a)=1.
    Returns (value, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = T * (conv - max_c)
    exp_shifted = np.exp(shifted)
    S = exp_shifted.sum()
    w = exp_shifted / S
    value = max_c + np.log(S) / T

    # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i in [0,n-1])
    # = 2 * conv(w, a[::-1])[i + n - 1]
    ar = a[::-1]
    N_conv = 1 << (len(w) + n - 1 - 1).bit_length()
    Wf = rfft(w, N_conv)
    Af = rfft(ar, N_conv)
    conv_w_ar = irfft(Wf * Af, N_conv)[:n + len(w) - 1]
    grad = 2.0 * conv_w_ar[n - 1: 2 * n - 1]
    return value, grad


def project_simplex(v):
    """Project vector onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def adam_soft_max_optimize(a0, time_limit, T_start=80.0, T_end=3.0,
                           lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8,
                           restart_threshold=3000, verbose=True):
    """Adam optimization of soft-max auto-correlation peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = obj_value(a)
    best_soft = soft_max_value_and_grad(a, T_start)[0]

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    t0 = monotonic()
    iters_done = 0

    while monotonic() - t0 < time_limit:
        t += 1
        T = T_start * (T_end / T_start) ** (min(t, 10000) / 10000.0)
        T = max(T, T_end)

        val, grad = soft_max_value_and_grad(a, T)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        score = obj_value(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        iters_done += 1

        if no_improve > restart_threshold:
            # Perturb and restart
            a = best_a + np.random.standard_normal(n) * 0.003
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            lr = max(lr * 0.8, 1e-5)

        if verbose and iters_done % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    AdamSoft: iter={iters_done} ({elapsed:.0f}s) T={T:.1f} best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score, iters_done


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return cp.SCS


def solve_lp_best_response(a_fixed, solver=None):
    """LP: find b (sum=1, b>=0) minimizing max_k conv(a_fixed, b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def damped_lp_refine(a0, time_limit, solver=None, max_iters=20000):
    """Damped alternating LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = obj_value(a)
    t0 = monotonic()

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 3.0 / (it + 10))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s2 = a_new.sum()
        if s2 < 1e-15:
            break
        a_new = a_new / s2
        score = obj_value(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if (it + 1) % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"    DampedLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: try adjusting individual elements to reduce peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = obj_value(a)
    t0 = monotonic()
    failed_rounds = 0

    while monotonic() - t0 < time_limit and failed_rounds < 30:
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        target = best_score / (2.0 * n)

        if peak_val <= target + 1e-15:
            break

        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-20:
            break

        # Sort by contribution (largest first)
        order = np.argsort(contribs)[::-1]
        improved = False
        failed_rounds = 0

        for ci in order[:20]:
            if improved:
                break
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j:
                continue
            if a[i] < 1e-15 or (j != i and a[j] < 1e-15):
                continue

            # Try reducing a[i] and redistributing
            for factor in [0.97, 0.93, 0.88, 0.8]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                # Redistribute to j (or to the peak partner)
                if j != i and j < n:
                    a_trial[j] += diff
                else:
                    # Redistribute proportionally to non-peak elements
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    others = a_trial[mask]
                    os = others.sum()
                    if os > 1e-15:
                        a_trial[mask] += diff * others / os
                    else:
                        a_trial = a
                        break

                a_trial = np.maximum(a_trial, 0)
                s2 = a_trial.sum()
                if s2 < 1e-15:
                    continue
                a_trial = a_trial / s2
                score = obj_value(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try small random perturbations
            for _ in range(10):
                a_trial = a + np.random.standard_normal(n) * 0.0003
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = obj_value(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed_rounds += 1

        if verbose and (iters_done if False else failed_rounds) % 5 == 0:
            elapsed = monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate with the real evaluate_sequence and update best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def chirp_init(n, freq=1.0, eps=0.3):
    """Create a chirp-like initialization."""
    t = np.linspace(0, 1, n)
    a = 1.0 + eps * np.cos(2 * np.pi * freq * t ** 2)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining soft-max Adam, LP refinement,
    coordinate descent, and multi-length exploration.
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n0 = 768
        best_seq = np.ones(n0) / n0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform: n={n0}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization from n={n_best}, score={best_score:.8f}")
    print(f"Time budget: {budget_s:.0f}s, deadline in {deadline - monotonic():.0f}s")

    # === Phase 1: Soft-max Adam optimization (primary) ===
    print(f"\n=== Phase 1: Soft-max Adam (n={n_best}) ===")
    rem = deadline - monotonic() - 600
    t_alloc = max(60.0, min(rem * 0.55, 1000.0))
    print(f"  Budget: {t_alloc:.0f}s")

    a_opt, score_opt, iters = adam_soft_max_optimize(
        best_seq, t_alloc, T_start=60.0, T_end=2.0,
        lr=0.0015, beta1=0.9, beta2=0.999,
        restart_threshold=2000, verbose=True
    )
    best_seq, best_score = check_and_update(a_opt, f"adam_opt_it{iters}", best_seq, best_score)

    # Second Adam run with different params (finer T)
    rem = deadline - monotonic() - 500
    if rem > 60:
        t_alloc2 = max(30.0, min(rem * 0.4, 300.0))
        print(f"\n  Adam run 2: T=20->1, lr=0.0005, budget={t_alloc2:.0f}s")
        a_opt2, _, _ = adam_soft_max_optimize(
            best_seq, t_alloc2, T_start=20.0, T_end=1.0,
            lr=0.0005, beta1=0.9, beta2=0.999,
            restart_threshold=3000, verbose=True
        )
        best_seq, best_score = check_and_update(a_opt2, "adam_opt2", best_seq, best_score)

    # === Phase 2: LP refinement ===
    print(f"\n=== Phase 2: LP Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.45, 350.0)
            print(f"  DampedLP budget={t_alloc:.0f}s")
            a_lp, _ = damped_lp_refine(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_lp, "damped_lp", best_seq, best_score)

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.5, 200.0)
            print(f"  CoordDesc budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for candidate_n in [512, 768, 1024, 1280, 1536, 2048, 2560]:
            if candidate_n != n_best and 256 <= candidate_n <= 3000:
                alt_lengths.append(candidate_n)

        # Sort by proximity to n_best
        alt_lengths.sort(key=lambda x: abs(x - n_best))
        alt_lengths = alt_lengths[:3]

        for alt_n in alt_lengths:
            rem = deadline - monotonic() - 60
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 120.0)
            print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)

            # Quick Adam at new length
            a_alt, _, _ = adam_soft_max_optimize(
                init_alt, t_alloc * 0.6, T_start=40.0, T_end=2.0,
                lr=0.001, restart_threshold=1500, verbose=False
            )
            best_seq, best_score = check_and_update(a_alt, f"adam_n{alt_n}", best_seq, best_score)

            # Quick LP at new length
            rem2 = deadline - monotonic() - 40
            if rem2 > 20 and best_seq is not None:
                a_alt_lp, _ = damped_lp_refine(best_seq, min(rem2 * 0.4, 40.0))
                best_seq, best_score = check_and_update(a_alt_lp, f"lp_n{len(best_seq)}", best_seq, best_score)

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 20
        if rem > 15:
            t_alloc = min(rem * 0.4, 60.0)
            print(f"  Final Adam (T=5->1): budget={t_alloc:.0f}s")
            a_final, _, _ = adam_soft_max_optimize(
                best_seq, t_alloc, T_start=5.0, T_end=1.0,
                lr=0.0003, restart_threshold=5000, verbose=False
            )
            best_seq, best_score = check_and_update(a_final, "final_adam", best_seq, best_score)

        rem = deadline - monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.5, 30.0)
            a_cd2, _ = coordinate_descent(best_seq, t_alloc, verbose=False)
            best_seq, best_score = check_and_update(a_cd2, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
