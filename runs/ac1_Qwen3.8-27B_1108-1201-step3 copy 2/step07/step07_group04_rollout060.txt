"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive multiplicative gradient
descent with natural gradient correction, periodic LP kicks, and multi-phase
refinement. Key innovations over previous approach:
1. Multiplicative (exponential family) updates that naturally maintain non-negativity
2. Natural gradient (Fisher metric) correction for better conditioning
3. Adaptive temperature schedule for the softmax-smoothed objective
4. Periodic LP best-response kicks to escape local minima
5. Efficient gradient computation via numpy correlate (FFT-backed)
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, size)
    return irfft(A * A, size)[:2 * n - 1]


def compute_soft_gradient(a, T):
    """Compute gradient of T*log(sum(exp(conv(a,a)/T))) wrt a.
    Returns (gradient, softmax_weights, convolution)."""
    n = len(a)
    b = fast_convolve(a)
    b_scaled = b / T
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w_sum = w.sum()
    w = w / w_sum
    g = np.correlate(w, a, mode='valid')
    return g, w, b


def score_of(a):
    """Compute the score 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_best_response(a_fixed, solver=None):
    """LP: find b >= 0, sum(b)=1, minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a_fixed
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value), 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else None
    return b_val, M_val


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def multiplicative_gradient_descent(a0, time_limit, solver=None, 
                                     T_init_alpha=10.0, T_max_alpha=5000.0,
                                     eta_init=0.8, eta_min=0.05,
                                     momentum=0.85, lp_every=3000,
                                     max_iters=10000000):
    """
    Multiplicative gradient descent with natural gradient correction.
    
    a0: initial sequence (non-negative)
    time_limit: seconds to run
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_of(a)
    v_log = np.zeros(n)  # momentum in log-space
    t0 = monotonic()
    
    # Temperature parameter: T = b_max / alpha, alpha increases from T_init_alpha to T_max_alpha
    alpha = T_init_alpha
    alpha_growth = (T_max_alpha / T_init_alpha) ** (1.0 / max(1, int(time_limit * 100)))
    # Actually let's use a simpler schedule: alpha increases linearly over time
    alpha_start = T_init_alpha
    alpha_end = T_max_alpha
    total_iters_est = int(time_limit * 5000)  # estimate ~5000 iters/sec
    
    eta = eta_init
    no_improve = 0
    iter_count = 0
    last_lp_time = t0
    
    while monotonic() - t0 < time_limit and iter_count < max_iters:
        iter_count += 1
        
        # Compute gradient
        b = fast_convolve(a)
        b_max = b.max()
        T = b_max / alpha
        g, w, _ = compute_soft_gradient(a, T)
        
        # Natural gradient correction
        g_centered = g - a * np.dot(g, a)
        
        # Adaptive step: normalize by the max of |g_centered|
        g_max = np.max(np.abs(g_centered))
        if g_max < 1e-15:
            eta = eta_init
            v_log = np.zeros(n)
            # Small perturbation
            a += np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a /= a.sum()
            continue
        
        # Momentum update in log-space
        v_log = momentum * v_log - eta * g_centered / g_max
        log_update = v_log
        
        # Clip to avoid overflow
        log_update = np.clip(log_update, -5, 5)
        
        # Multiplicative update
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Check score
        new_score = score_of(a_new)
        
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        elif new_score < best_score + 0.01:
            # Accept even slightly worse (for exploration)
            a = a_new
            no_improve += 1
        else:
            # Reject, reduce step size
            a = a  # keep old
            eta = max(eta_min, eta * 0.95)
            v_log *= 0.5
            no_improve += 1
        
        # Cool temperature
        progress = min(1.0, iter_count / total_iters_est)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        # Periodic LP kick
        if solver is not None and (iter_count % lp_every == 0) and (monotonic() - last_lp_time > 5):
            last_lp_time = monotonic()
            b_lp, _ = solve_lp_best_response(best_a, solver=solver)
            if b_lp is not None:
                # Blend with current a
                blend = 0.3
                a_trial = (1 - blend) * best_a + blend * b_lp
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                trial_score = score_of(a_trial)
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    v_log = np.zeros(n)
                    no_improve = 0
                # Also check b_lp directly
                lp_score = score_of(b_lp)
                if lp_score < best_score:
                    best_score = lp_score
                    best_a = b_lp.copy()
                    a = b_lp.copy()
                    v_log = np.zeros(n)
                    no_improve = 0
        
        # Reset if stuck for too long
        if no_improve > 5000:
            a = best_a + np.random.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a /= a.sum()
            v_log = np.zeros(n)
            eta = eta_init
            no_improve = 0
            alpha = max(alpha_start, alpha * 0.5)  # reheat
        
        # Re-increase eta if making good progress
        if no_improve == 0 and eta < eta_init:
            eta = min(eta_init, eta * 1.02)
        
        if iter_count % 100000 == 0:
            elapsed = monotonic() - t0
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f} alpha={alpha:.1f} eta={eta:.3f}")
    
    return best_a, best_score


def damped_lp_refine(a0, time_limit, solver=None):
    """Damped alternating LP refinement."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a /= a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = monotonic()
    it = 0
    
    while monotonic() - t0 < time_limit:
        it += 1
        b_new, _ = solve_lp_best_response(a, solver=solver)
        if b_new is None:
            break
        
        alpha = min(1.0, 3.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        score = score_of(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        
        if it % 100 == 0:
            elapsed = monotonic() - t0
            print(f"    DLP iter={it} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T=50):
    """L-BFGS refinement in log-space with softmax smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]
    
    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        b_max = b.max()
        scaled = (b - b_max) / T
        exp_b = np.exp(scaled)
        lse = b_max + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T
    
    def jac(z):
        if stopped[0]:
            return np.zeros(n)
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        b_max = b.max()
        scaled = (b - b_max) / T
        exp_b = np.exp(scaled)
        w = exp_b / exp_b.sum()
        
        g = np.correlate(w, a, mode='valid')
        grad_objective = 2.0 * n * g
        S = np.dot(grad_objective, a)
        grad_z = a * (grad_objective - a * S)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12,
                     'maxfun': 50000, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_s = z_final - z_final.max()
            a_out = np.exp(z_s)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()
    
    return a_out


def peak_shave(a0, time_limit):
    """Targeted peak shaving."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a /= a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = monotonic()
    failed = 0
    
    while monotonic() - t0 < time_limit and failed < 100:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)
        
        if peak_val <= target + 1e-15:
            break
        
        # Find top contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.sum() < 1e-20:
            break
        
        top = np.argsort(contribs)[::-1][:8]
        improved = False
        
        # Try reducing single element
        for ci in top:
            i = int(idx[ci])
            old = a[i]
            if old < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.8, 0.7, 0.6]:
                a_trial = a.copy()
                diff = old * (1 - factor)
                a_trial[i] = old * factor
                others = a_trial.copy()
                others[i] = 0
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                others += diff * others / others_sum
                a_trial = others
                a_trial = a_trial / a_trial.sum()
                s = score_of(a_trial)
                if s < best_score - 1e-13:
                    a = a_trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Try reducing pair
        if not improved:
            for ci in top[:5]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.85]:
                    for f2 in [0.9, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = a_trial / a_trial.sum()
                        s = score_of(a_trial)
                        if s < best_score - 1e-13:
                            a = a_trial
                            best_score = s
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        # Try random perturbation
        if not improved:
            for _ in range(10):
                a_trial = a + np.random.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                s = score_of(a_trial)
                if s < best_score - 1e-13:
                    a = a_trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            failed = 0
        else:
            failed += 1
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
