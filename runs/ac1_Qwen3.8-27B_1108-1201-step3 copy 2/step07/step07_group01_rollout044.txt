"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-peak subgradient descent
with backtracking line search, momentum, spectral equalization, and targeted
coordinate descent on peak-contributing variables. Key novelty: adaptive
multi-peak gradient with line search for robust monotone convergence, combined
with frequency-domain spectral flattening and KKT-based active set refinement.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_peak_factors(a, K=10, T=0.5):
    """Compute weighted combination of top-K peak subgradients.
    Returns (combined_grad, peak_lags, peak_values, weights)."""
    n = len(a)
    b = fast_convolve(a)
    # Use top K peaks
    if K >= len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argpartition(b, -K)[-K:]
    top_vals = b[top_idx]
    top_order = np.argsort(top_vals)[::-1]
    top_idx = top_idx[top_order]
    top_vals = top_vals[top_order]

    # Softmax weights
    scaled = top_vals / T
    m = scaled.max()
    w = np.exp(scaled - m)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(len(top_vals)) / len(top_vals)
    else:
        w = w / s

    # Combined subgradient
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        grad[idx] += wi * 2.0 * a[ki - idx]

    return grad, top_idx, top_vals, w


def multi_peak_descent(a_init, time_limit, K=8, T_start=1.0, T_end=0.05,
                       eta_start=0.05, eta_end=0.001, momentum=0.9,
                       verbose=False, label=""):
    """Multi-peak subgradient descent with momentum and backtracking line search."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    vel = np.zeros(n)
    stuck = 0
    stuck_thresh = 200

    max_iters = 200000
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        grad, _, _, _ = compute_peak_factors(a, K=K, T=T)
        # Project to simplex tangent
        grad = grad - np.dot(a, grad)

        # Momentum
        vel = momentum * vel + (1 - momentum) * grad
        direction = -vel

        # Backtracking line search with multiplicative update
        best_trial_score = best_score
        best_trial_a = a.copy()
        for ls in range(20):
            step = eta * (0.5 ** ls)
            log_update = np.clip(step * direction, -4.0, 4.0)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-300)
            s = a_trial.sum()
            if s < 1e-15:
                break
            a_trial = a_trial / s
            sc = score_normalized(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial_a = a_trial.copy()
            if sc < score_normalized(a) - 1e-10:
                break

        a = best_trial_a
        cur_score = score_normalized(a)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Stuck: restart with perturbation
        if stuck > stuck_thresh:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise = rng.standard_normal(n) * 0.005
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            vel = np.zeros(n)
            stuck = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MMD iter={it} ({elapsed:.0f}s): "
                  f"T={T:.4f} eta={eta:.5f} score={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score

        # Keep DC (sum) preserved

        s = a_trial.sum()
        if s < 1e-15:
            break
        a_trial = a_trial / s

        if sc < cur_score - 1e-12:
            a = a_trial
        else:
            # Try less aggressive damping
            A = rfft(a, fft_size)
            A[hot_mask] *= 0.99
            a_trial = irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                break
            a_trial = a_trial / s
            sc = score_normalized(a_trial)
            if sc < cur_score - 1e-12:
                a = a_trial
            else:
                # No improvement, increase aggressiveness slowly
                damp_factor *= 0.99

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} Spectral iter={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def targeted_coord_descent(a_init, time_limit, top_K=8, verbose=False, label=""):
    """Coordinate descent focusing only on variables that contribute to top peaks."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    factors = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]
    no_improve = 0
    max_no_improve = 100

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved_this_pass = False

        # Find peak-contributing variables
        b = fast_convolve(a)
        top_idx = np.argpartition(b, -top_K)[-top_K:]
        active_vars = set()
        for ki in top_idx:
            ki = int(ki)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            for i in range(js, je + 1):
                if a[i] > 1e-10:
                    active_vars.add(i)

        if not active_vars:
            break

        active_list = list(active_vars)
        rng = np.random.default_rng(int(time.monotonic() * 1000))
        rng.shuffle(active_list)

        for i in active_list:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial[i] = new_val
                diff = new_val - old_val
                if abs(diff) < 1e-15:
                    continue
                # Redistribute
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_this_pass = True
                    no_improve = 0
                    break

        if not improved_this_pass:
            no_improve += 1

        if verbose and no_improve % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} TCD ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score


def lbgfs_smooth(a_init, time_limit, T=5, maxiter=5000):
    """L-BFGS on smoothed max objective in log-space."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 -= z0.mean()
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e20
        a_s = softmax_vec(z)
        b = fast_convolve(a_s)
        scaled = b / T
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        return 2.0 * n * T * lse

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        a_s = softmax_vec(z)
        b = fast_convolve(a_s)
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()
        # grad of smoothed max w.r.t. a
        grad_a = np.zeros(n)
        for wi, ki in zip(w, range(len(b))):
            if wi < 1e-15:
                continue
            ki = int(ki)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * wi * a_s[ki - idx]
        # Chain rule through softmax
        mean_g = np.dot(a_s, grad_a)
        grad_z = a_s * (grad_a - mean_g)
        return 2.0 * n * T * grad_z

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 20, 'disp': False})
        a_out = softmax_vec(res.x)
    except Exception:
        a_out = a_init.copy()

    return a_out


def softmax_vec(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def peak_equalize(a_init, time_limit, verbose=False, label=""):
    """Iteratively equalize top peaks by finding a direction that reduces all simultaneously."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        # Find top 5 peaks
        K = min(5, len(b))
        top_idx = np.argpartition(b, -K)[-K:]
        top_idx = top_idx[np.argsort(b[top_idx])[::-1]]
        top_vals = b[top_idx]

        # Only work on peaks within 5% of max
        max_val = top_vals[0]
        active_mask = top_vals > 0.95 * max_val
        active_idx = top_idx[active_mask]

        if len(active_idx) == 0:
            break
        if len(active_idx) == 1:
            # Single peak: just reduce it
            ki = int(active_idx[0])
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[ki - idx]
            # Reduce the most contributing variable
            top_ci = np.argmax(contribs)
            i_star = int(idx[top_ci])
            for f in [0.95, 0.9, 0.85, 0.8]:
                a_trial = a.copy()
                a_trial[i_star] *= f
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < cur_score - 1e-12:
                    a = a_trial
                    break
            continue

        # Multiple peaks: find direction that reduces all
        # Build constraint matrix
        m_constraints = len(active_idx)
        G = np.zeros((m_constraints, n))
        for row, ki in enumerate(active_idx):
            ki = int(ki)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            G[row, idx] = 2.0 * a[ki - idx]

        # Find direction d: minimize G@d (reduce all peaks)
        # subject to sum(d) = 0
        # Simple approach: d = -P(G^T * w) where P projects to sum=0 hyperplane
        # Use equal weights or softmax weights
        w = np.ones(m_constraints) / m_constraints
        grad_sum = G.T @ w
        grad_sum = grad_sum - np.dot(a, grad_sum)  # project to tangent
        d = -grad_sum

        # Normalize direction
        dn = np.linalg.norm(d)
        if dn < 1e-15:
            break
        d = d / dn

        # Try multiplicative update with this direction
        for step in [0.02, 0.01, 0.005, 0.002]:
            log_update = np.clip(step * d, -2.0, 2.0)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial / a_trial.sum()
            sc = score_normalized(a_trial)
            if sc < cur_score - 1e-12:
                a = a_trial
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining multi-peak subgradient with line search,
    spectral equalization, targeted coordinate descent, and L-BFGS polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        # Fallback: use a structured initialization
        n = 768
        # Use a "raised cosine" shape as initial guess
        x = np.linspace(0, 1, n)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback init: n={n}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization: n={n_best}, initial score={best_score:.8f}")

    # === Phase 1: Multi-peak subgradient with line search (primary) ===
    print("\n=== Phase 1: Multi-Peak Descent (MMD) ===")
    configs_p1 = [
        (8, 2.0, 0.1, 0.04, 0.003, 0.85, "MMD_wide"),
        (8, 1.0, 0.05, 0.03, 0.002, 0.9, "MMD_mid"),
        (10, 0.5, 0.02, 0.02, 0.001, 0.92, "MMD_tight"),
    ]
    for K, Ts, Te, es, ee, mom, lbl in configs_p1:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 150.0)
        print(f"  {lbl}: K={K}, T=[{Ts},{Te}], eta=[{es},{ee}], mom={mom}, budget={t_alloc:.0f}s")
        a_out, _ = multi_peak_descent(best_seq, t_alloc, K=K, T_start=Ts, T_end=Te,
                                       eta_start=es, eta_end=ee, momentum=mom, label=lbl)
        best_seq, best_score = check_and_update(a_out, lbl, best_seq, best_score)

    # === Phase 2: Spectral Equalization ===
    print(f"\n=== Phase 2: Spectral Equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_alloc = min(rem * 0.25, 100.0)
        print(f"  Spectral: budget={t_alloc:.0f}s")
        a_sp, _ = spectral_equalize(best_seq, t_alloc, label="spectral")
        best_seq, best_score = check_and_update(a_sp, "spectral", best_seq, best_score)

    # === Phase 3: Peak Equalization ===
    print(f"\n=== Phase 3: Peak Equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 120
    if rem > 30:
        t_alloc = min(rem * 0.3, 100.0)
        print(f"  PeakEq: budget={t_alloc:.0f}s")
        a_pe, _ = peak_equalize(best_seq, t_alloc, label="peak_eq")
        best_seq, best_score = check_and_update(a_pe, "peak_eq", best_seq, best_score)

    # === Phase 4: L-BFGS with very low temperature ===
    print(f"\n=== Phase 4: L-BFGS Low-T (best={best_score:.8f}) ===")
    for T in [10, 5, 3, 2]:
        rem = deadline - time.monotonic() - 80
        if rem < 20:
            break
        t_alloc = min(rem * 0.3, 60.0)
        print(f"  L-BFGS T={T}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_smooth(best_seq, t_alloc, T=T, maxiter=5000)
        best_seq, best_score = check_and_update(a_lb, f"lbfs_T{T}", best_seq, best_score)

    # === Phase 5: Targeted Coordinate Descent ===
    print(f"\n=== Phase 5: Targeted Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t_alloc = min(rem * 0.4, 100.0)
        print(f"  TCD: budget={t_alloc:.0f}s")
        a_cd, _ = targeted_coord_descent(best_seq, t_alloc, top_K=10, label="tcd")
        best_seq, best_score = check_and_update(a_cd, "tcd", best_seq, best_score)

    # === Phase 6: Second round MMD from current best ===
    print(f"\n=== Phase 6: MMD Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 40:
        t_alloc = min(rem * 0.4, 100.0)
        print(f"  MMD2: budget={t_alloc:.0f}s")
        a_m2, _ = multi_peak_descent(best_seq, t_alloc, K=12, T_start=0.3, T_end=0.01,
                                      eta_start=0.02, eta_end=0.0005, momentum=0.95,
                                      label="MMD2")
        best_seq, best_score = check_and_update(a_m2, "MMD2", best_seq, best_score)

    # === Phase 7: L-BFGS T=1 (final polish) ===
    print(f"\n=== Phase 7: Final L-BFGS T=1 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_alloc = min(rem * 0.5, 40.0)
        print(f"  L-BFGS T=1: budget={t_alloc:.0f}s")
        a_f1 = lbgfs_smooth(best_seq, t_alloc, T=1, maxiter=8000)
        best_seq, best_score = check_and_update(a_f1, "lbfs_T1", best_seq, best_score)

    # === Phase 8: Final targeted CD ===
    print(f"\n=== Phase 8: Final TCD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 8:
        t_alloc = min(rem * 0.6, 20.0)
        a_cd2, _ = targeted_coord_descent(best_seq, t_alloc, top_K=15, label="tcd_final")
        best_seq, best_score = check_and_update(a_cd2, "tcd_final", best_seq, best_score)

    # === Phase 9: Try a different length ===
    print(f"\n=== Phase 9: Alternative Length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 40 and best_seq is not None:
        n_cur = len(best_seq)
        # Try n*1.5
        alt_n = int(n_cur * 1.5)
        if 256 <= alt_n <= 4096:
            t_alloc = min(rem * 0.5, 50.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = multi_peak_descent(init_alt, t_alloc, K=10, T_start=0.5,
                                           T_end=0.05, eta_start=0.03, eta_end=0.002,
                                           momentum=0.9, label=f"alt_n{alt_n}")
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}",
                                                     best_seq, best_score)

    # === Final: Ultra-fine L-BFGS ===
    print(f"\n=== Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 3
    if rem > 5 and best_seq is not None:
        a_fin = lbgfs_smooth(best_seq, rem, T=0.5, maxiter=10000)
        best_seq, best_score = check_and_update(a_fin, "final_lbfgs", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
