def lp_direction(f, top_k, m_factor=1.0):
    """
    LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M*m_factor for top_k peaks.
    M is the current max of f*f.
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max() * m_factor
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g_var = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
    except Exception:
        return None, M
    
    if g_var.value is None:
        return None, M
    g_val = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Nesterov-accelerated mirror descent on smoothed objective,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_grad_fft(a, w):
    """
    Compute g[i] = sum_k w[k] * a[k-i] (valid k-i in [0, n-1]).
    Cross-correlation of w (len 2n-1) and a (len n), first n elements.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g = irfft(np.conj(W) * A, fft_size)[:n]
    return g


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, tau):
    """
    Compute gradient of F(a) = tau*log(sum(exp(b_k/tau))) where b = a*a.
    Returns (F_value, gradient, b).
    """
    b = fast_convolve(a)
    scaled = b / tau
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    F = b.max() + tau * np.log(exp_vals.sum())
    g = convolve_grad_fft(a, w)
    return F, g, b

            if it % (check_interval * 10) == 0:
                elapsed = time.monotonic() - t0
                print(f"    {label} iter={it} obj={best_obj:.10f} tau={tau:.5f} lr={lr:.4f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def nesterov_projected(f_init, time_limit, tau_init=0.005, lr_init=0.1,
                       label="nesterov_proj"):
    """
    Nesterov-accelerated projected gradient on smoothed objective.
    Projection onto simplex after each step.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f_prev = f.copy()
    tau = tau_init
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    no_improve = 0
    check_interval = max(1, 100000 // max(n, 1))
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov look-ahead
        t_k = 1 + np.sqrt(1 + 4 * (it - 1) ** 2)
        t_k = t_k / 2 * 2  # just use it for beta
        beta = (np.sqrt(1 + 4 * (it - 1) ** 2) - 1) / 2
        beta = beta / (1 + beta)  # normalize to [0,1)
        y = f + beta * (f - f_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = f.copy()
        else:
            y = y / sy
        
        # Gradient at y
        _, g, b = smoothed_grad(y, tau)
        
        # Gradient step + projection
        f_new = f - lr * g
        f_new = project_simplex(f_new)
        
        f_prev = f.copy()
        f = f_new
        
        if it % check_interval == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    lr *= 0.9
                    no_improve = 0
                if no_improve > 80:
                    break
            
            tau = max(tau * 0.999, 0.001)
            
            if it % (check_interval * 10) == 0:
                elapsed = time.monotonic() - t0
                print(f"    {label} iter={it} obj={best_obj:.10f} tau={tau:.5f} lr={lr:.4f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def lp_solve(f, top_k, m_scale=1.0, solver='SCS'):
    """
    LP: max sum(g) s.t. g>=0, (f*g)[k] <= m_scale * max(f*f) for top_k peaks.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max() * m_scale
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build constraint matrix: A[row, col] where row corresponds to peak k
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid i
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=300000, abstol=1e-8)
        except Exception:
            return None, M
    
    if g_var.value is None:
        return None, M
    g_val = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def lp_optimize(f_init, time_limit, top_k_init=30, top_k_max=200, 
                m_scale_init=1.0, label="lp"):
    """LP-based direction finding with adaptive parameters and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    m_scale = m_scale_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptively grow top_k
        if it % 25 == 0 and top_k < top_k_max:
            top_k = min(top_k + 5, top_k_max)
        
        g, M = lp_solve(f, top_k, m_scale)
        if g is None:
            # Retry with more constraints
            g, M = lp_solve(f, min(2 * top_k, 2 * n - 1), m_scale)
            if g is None:
                # Try with tighter M
                g, M = lp_solve(f, top_k, m_scale * 0.95)
                if g is None:
                    break
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 10:
                # Try tighter constraint
                m_scale *= 0.97
                top_k = min(top_k + 10, top_k_max)
                if m_scale < 0.7:
                    break
                continue
            top_k = min(top_k * 2, top_k_max)
            continue
        
        h = g / S_g
        
        # Line search: try multiple step sizes
        best_t = 0.1
        best_t_obj = fast_convolve(f).max()
        candidates = [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.35, 0.5]
        for t in candidates:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-15:
                best_t_obj = obj
                best_t = t
        
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            break
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
            # Relax m_scale slightly when making progress
            m_scale = min(m_scale * 1.02, 1.0)
        else:
            stuck += 1
            if stuck > 30:
                break
        
        if it % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} top_k={top_k} m={m_scale:.3f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent: try scaling each element by several factors."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.4, 0.6, 0.8, 1.25, 1.5, 2.0])
    
    passes = 0
    min_pass_time = 0.01
    last_pass_time = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            orig_val = f[j]
            best_val = orig_val
            best_j_obj = best_obj
            for mult in multipliers:
                f[j] = orig_val * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = orig_val
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_val = orig_val * mult
            f[j] = best_val
            if best_val != orig_val:
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    any_improved = True
        
        pass_time = time.monotonic() - last_pass_time
        last_pass_time = time.monotonic()
        if not any_improved:
            break
        if pass_time < min_pass_time and passes > 20:
            break
    
    return f, best_obj


def make_cosine_init(n, k=1):
    """Cosine-based initialization."""
    x = np.linspace(0, np.pi, n)
    a = np.abs(np.cos(k * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gaussian_mix_init(n, n_bumps=3, seed=42):
    """Gaussian mixture initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=float)
    a = np.zeros(n)
    for _ in range(n_bumps):
        mu = rng.uniform(0.15 * n, 0.85 * n)
        sigma = rng.uniform(0.03 * n, 0.12 * n)
        amp = rng.uniform(0.5, 1.5)
        a += amp * np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_window_init(n, shape='hamming'):
    """Window function initialization."""
    if shape == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    elif shape == 'hann':
        a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    elif shape == 'blackman':
        x = np.arange(n) / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif shape == 'flat_top':
        x = np.arange(n) / (n - 1)
        a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * x)
             + 0.277263158 * np.cos(4 * np.pi * x)
             - 0.083578947 * np.cos(6 * np.pi * x)
             + 0.006947368 * np.cos(8 * np.pi * x))
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f_src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(f_src)
    if n_src == n_target:
        return f_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    f_tgt = np.interp(x_tgt, x_src, f_src)
    f_tgt = np.maximum(f_tgt, 0)
    s = f_tgt.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return f_tgt / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-phase hybrid search combining mirror descent,
    Nesterov projected gradient, LP direction finding, and coordinate descent.
    Uses diverse initializations and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent available, using uniform n=1024, score={p_score:.6f}")
    
    best_seq = parent.copy()
    best_score = p_score
    
    def try_update(f_candidate, source_label):
        nonlocal best_seq, best_score
        if f_candidate is None:
            return
        s = f_candidate.sum()
        if s < 1e-10:
            return
        f_norm = f_candidate / s
        sc = float(evaluate_sequence(f_norm.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_norm.copy()
            print(f"  *** NEW BEST [{source_label}]: score={sc:.6f} n={len(f_norm)}")
    
    n_parent = len(parent)
    
    # ============ Phase 1: Diverse initializations + Mirror Descent ============
    print(f"\n{'='*60}")
    print(f"Phase 1: Diverse starts + Mirror Descent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.35, 700.0)
    
    if t_phase1 > 30:
        inits = [
            ("parent", parent),
            ("cosine1", make_cosine_init(n_parent, 1)),
            ("cosine2", make_cosine_init(n_parent, 2)),
            ("gauss3", make_gaussian_mix_init(n_parent, 3, seed=42)),
            ("gauss5", make_gaussian_mix_init(n_parent, 5, seed=123)),
            ("hamming", make_window_init(n_parent, 'hamming')),
            ("hann", make_window_init(n_parent, 'hann')),
            ("blackman", make_window_init(n_parent, 'blackman')),
        ]
        
        # Also add parent with perturbation
        rng = np.random.default_rng(seed)
        pert = parent * (1 + 0.05 * rng.standard_normal(n_parent))
        pert = np.maximum(pert, 0)
        pert /= pert.sum()
        inits.append(("parent_pert", pert))
        
        n_inits = len(inits)
        time_per_init = t_phase1 / n_inits
        
        for name, f0 in inits:
            if time.monotonic() > deadline:
                break
            t_rem = deadline - time.monotonic()
            t_alloc = min(time_per_init, t_rem * 1.5)
            if t_alloc < 5:
                break
            print(f"  [{name}] starting, budget={t_alloc:.0f}s")
            
            # Run mirror descent
            f1, obj1 = mirror_descent_nesterov(f0, time_limit=t_alloc, 
                                               tau_init=0.01, lr_init=0.05,
                                               label=f"md_{name}")
            try_update(f1, f"md_{name}")
            
            # Quick LP polish
            t_rem2 = deadline - time.monotonic()
            if t_rem2 > 15 and best_seq is not None:
                t_lp = min(20.0, t_rem2 * 0.3)
                if t_lp > 5:
                    f2, obj2 = lp_optimize(f1, time_limit=t_lp,
                                           top_k_init=20, top_k_max=60,
                                           label=f"lp_{name}")
                    try_update(f2, f"lp_{name}")
    
    # ============ Phase 2: Deep LP on current best ============
    print(f"\n{'='*60}")
    print(f"Phase 2: Deep LP refinement (best={best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.30, 500.0)
    
    if t_phase2 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t_phase2:.0f}s, n={n_b}")

        sub_budgets = [t_phase2 * 0.4, t_phase2 * 0.35, t_phase2 * 0.25]
        top_k_ranges = [(30, 80), (80, 150), (150, 250)]
        
        for sb, (tk_min, tk_max) in zip(sub_budgets, top_k_ranges):

            if sb < 10:

            f2, obj2 = lp_optimize(best_seq, time_limit=sb,
                                   top_k_init=tk_min, top_k_max=tk_max,
                                   label=f"lp_k{tk_min}-{tk_max}")
            try_update(f2, f"lp_k{tk_min}-{tk_max}")
    
    # ============ Phase 3: Nesterov Projected Gradient ============

                                       tau_init=0.005, lr_init=0.05,

    print(f"Phase 4: Multi-length exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
        tl = int(n_base * factor)
        tl = max(384, min(4000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))
    
    n_attempted = 0
    for target_n in target_lengths[:4]:

        if t_rem2 > 10:

                                      top_k_init=30, top_k_max=80,
                                      label=f"ml_lp_{target_n}")

    print(f"Phase 5: Final LP burst (best={best_score:.6f})")

    rem = deadline - time.monotonic()

                               top_k_init=100, top_k_max=300,
                               m_scale_init=0.98,
                               label="lp_final")
        try_update(f5, "lp_final")
    
    # ============ Phase 6: Coordinate descent polish ============
    print(f"\n{'='*60}")
    print(f"Phase 6: Coordinate descent (best={best_score:.6f})")
    print(f"{'='*60}")
