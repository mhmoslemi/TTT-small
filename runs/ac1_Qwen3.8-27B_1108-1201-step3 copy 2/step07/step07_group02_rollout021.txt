The point is: for non-symmetric a, the center correlation can be small, but the edge correlations (k=0, k=2n-2) are large.

So the optimization is a trade-off: we want to flatten the entire convolution profile, not just the center.

This is why smooth, slowly-varying sequences work well: they have low-frequency content, which means the convolution is concentrated in the middle (low k and high k have small values).

def simplex_project(x):
    """Project x onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    return a / s

def kaiser_window(n, beta):
    """Kaiser-Bessel window of length n with shape parameter beta."""
    if n <= 0:
        return np.array([])
    if n == 1:
        return np.array([1.0])
    from scipy.special import i0
    m = np.arange(n)
    alpha = (n - 1) / 2.0
    x = (m - alpha) / alpha
    w = i0(beta * np.sqrt(1 - x**2)) / i0(beta)
    return w


def raised_cosine_window(n, transition=0.5):
    """Raised cosine window."""
    if n <= 0:
        return np.array([])
    m = np.arange(n)
    alpha = (n - 1) / 2.0
    t = (m - alpha) / (transition * alpha) if alpha > 0 else np.zeros(n)
    t = np.clip(t, -1, 1)
    w = 0.5 * (1 + np.cos(np.pi * t))
    return w

    """Triangular (Bartlett) window."""
    if n <= 0:
        return np.array([])
    m = np.arange(n)
    alpha = (n - 1) / 2.0
    w = 1.0 - np.abs(m - alpha) / alpha
    return np.maximum(w, 0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(x):
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def normalize_to_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    return a / s
