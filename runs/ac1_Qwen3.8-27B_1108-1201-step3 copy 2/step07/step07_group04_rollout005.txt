def get_lp_solver():

def solve_lp_best_response(a_fixed, solver=None):

        solver = get_lp_solver()

    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)

                continue

                prob.solve(solver=alt)
                break

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)

    s = b_val.sum()

def lbgfs_smooth(a0, time_limit, T=50, maxiter=3000):
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = np.zeros(n)
        # Use matrix-free approach
        # grad of sum_k w[k] * conv(a,a)[k] wrt a
        # conv(a,a)[k] = sum_i a[i]*a[k-i]
        # d/da[j] = sum_k w[k] * (a[k-j] + a[j-(k)] ) ... 
        # Actually: d/da[j] of sum_k w[k] * sum_i a[i]*a[k-i]
        # = sum_k w[k] * (a[k-j] + a[j-k]) (for valid indices)
        # = sum_k w[k] * a[k-j] + sum_k w[k] * a[j-k]
        
        # More efficiently: if we define the "correlation" of w with a:
        # grad_a[j] = sum_k w[k] * (a[k-j] + a[j-k])
        # This is 2 * (w * a) * (j) where * is the appropriate cross-correlation
        
        # Let's compute it as: grad_a = 2 * conv(w, a) (with appropriate indexing)
        # Actually, let's just do it directly for clarity
        
        for j in range(n):
            # Contribution from a[j] in conv(a,a)[k]:
            # a[j] appears in conv(a,a)[k] as a[j]*a[k-j] for k=j,...,j+n-1
            # and as a[k-j]*a[j] for k=j,...,j+n-1 (same thing)
            # So d(conv(a,a)[k])/da[j] = a[k-j] (for k in [j, j+n-1] intersect [0, 2n-2])
            
            k_start = max(0, j)  # k-j >= 0 means k >= j
            k_end = min(2 * n - 2, j + n - 1)  # k-j <= n-1 means k <= j+n-1
            if k_start <= k_end:
                grad_a[j] = 2.0 * np.dot(w[k_start:k_end+1], a[k_start - j:k_end - j + 1])
        
        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:

            method='L-BFGS-B',

        else:
            a_out = a0.copy()

def coordinate_descent(a0, time_limit, max_rounds=500):
    """Coordinate descent: for each coordinate, find the optimal value given others fixed."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    round_num = 0

    while monotonic() - t0 < time_limit and round_num < max_rounds:
        round_num += 1
        improved = False
        b = fast_convolve(a)
        
        for i in range(n):
            if monotonic() - t0 > time_limit:
                break
            # Try adjusting a[i]
            # Fix all other coordinates, optimize a[i]
            # conv(a,a)[k] as a function of a[i]:
            # For each k, a[i] appears as a[i]*a[k-i] (if k-i is valid)
            # So conv(a,a)[k] = a[i]^2 (if k=2i) + 2*a[i]*a[k-i] (if k != 2i and k-i valid) + const
            # We want to minimize max_k of this
            
            # Simpler: just try a few candidate values
            old_val = a[i]
            candidates = []
            if old_val > 0:
                candidates = [old_val * 0.5, old_val * 0.7, old_val * 0.9,
                              old_val * 1.1, old_val * 1.3, old_val * 1.5]
            else:
                candidates = [0.001, 0.005, 0.01]
            
            best_cand = old_val
            best_cand_score = best_score
            
            for cv in candidates:
                a[i] = cv
                # Renormalize
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                score = 2.0 * n * fast_convolve(a_norm).max()
                if score < best_cand_score - 1e-14:
                    best_cand_score = score
                    best_cand = cv
            
            if best_cand != old_val:
                a[i] = best_cand
                a = a / a.sum()
                if best_cand_score < best_score - 1e-14:
                    best_score = best_cand_score
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = old_val
        
        if not improved and round_num > 5:
            break

    return best_a, best_score


def peak_shave(a0, time_limit):
    """Aggressive peak shaving."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        if b[peak_k] <= best_score / (2.0 * n) + 1e-15:
            break
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.sum() < 1e-20:
            break
        
        top_contrib = np.argsort(contribs)[::-1][:8]
        improved = False
        
        for ci in top_contrib:
            if improved:
                break
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute the difference
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try random perturbation
            improved = False
            for _ in range(10):
                a_trial = a + np.random.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 30:
                break
        else:
            failed = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
