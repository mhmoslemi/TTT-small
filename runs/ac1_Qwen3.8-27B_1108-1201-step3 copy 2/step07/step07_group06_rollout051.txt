import time

from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient_w(a, w, fft_size):
    """
    Compute gradient of sum_k w[k] * c_k(a) where c_k = conv(a,a)[k].
    w has length 2n-1.
    Result[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    """
    n = len(a)
    # cross_corr[j] = sum_m W[j+m] * a[m]
    W_fft = rfft(w, fft_size)
    a_fft = rfft(a, fft_size)
    cross_corr = irfft(W_fft * np.conj(a_fft), fft_size)[:n]
    return 2.0 * cross_corr

def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def get_score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def score(a):
    """Full score (handles normalization)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
