"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set Gauss-Newton 
method combined with mirror descent initialization and coordinate descent 
polishing. Key innovation: identifies the top K correlation peaks and 
solves a linearized KKT system to equalize them, providing much faster 
convergence than pure subgradient methods near the optimum.

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max()), b

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def grad_bk(a, k):
    """Gradient of b[k] w.r.t. a. Returns 2*a[k-j] for valid j."""
    n = len(a)
    g = np.zeros(n)
    # g[j] = 2*a[k-j] if 0 <= k-j < n
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        g[j] = 2.0 * a[k - j]
    return g

def gauss_newton_step(a, K=5, alpha=1.0):
    """Active-set Gauss-Newton step to equalize top K peaks."""
    n = len(a)
    b = fast_convolve(a)
    
    # Find top K peaks
    peak_indices = np.argsort(b)[::-1][:K]
    active_peaks = b[peak_indices]
    
    # Target: equalize to the minimum of the top K (or slightly below)
    M = active_peaks.min()  # We want all top K to be at most M
    # Actually, let's target the mean of top K reduced a bit
    M_target = active_peaks[0] * (1 - alpha * 0.1)  # reduce max by 10% * alpha
    
    # Build gradient matrix
    # G[i, j] = grad_b[peak_i][j] = 2*a[peak_i - j] (if valid)
    G = np.zeros((K, n))
    for i, k in enumerate(peak_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        G[i, js:je+1] = 2.0 * a[k - (js:je+1)]
    
    # Excesses
    e = b[peak_indices] - M_target  # How much we need to reduce each peak
    
    # Solve G*d = -e in least squares with sum(d) = 0
    # Use regularized pseudoinverse
    # d = G^T (G G^T + lambda*I)^{-1} * (-e)
    lambda_reg = 1e-10 * n
    GGt = G @ G.T + lambda_reg * np.eye(K)
    try:
        d = G.T @ np.linalg.solve(GGt, -e)
    except np.linalg.LinAlgError:
        d = -np.ones(n) / n * e.mean()
    
    # Project to sum(d) = 0
    d = d - np.sum(d) / n
    
    # Take step
    a_new = a + alpha * d
    a_new = normalize(a_new)
    
    return a_new, b[peak_indices[0]]

def mirror_descent(a_init, time_limit, T_start=0.01, T_end=0.001,
                   eta_start=0.02, eta_end=0.001, max_iters=50000):
    """Mirror descent with temperature annealing."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score, _ = score_from_a(a)
    t0 = time.monotonic()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
        
        # Softmax weights over peaks
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()
        
        # Gradient: 4*n * cross_corr(w, a)
        # Actually, grad of max(b) ≈ 4*n * (w convolved with reversed a)... 
        # Let me use the subgradient approach:
        # grad[i] = sum_k w[k] * 2*a[k-i] (for valid k-i)
        # This is 2 * cross_corr(w, a)
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 2.0 * n * 2.0 * correl[:n]  # 4*n * cross_corr
        
        # Project to tangent space
        grad_a = grad_a - np.dot(a, grad_a)
        
        # Multiplicative update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = normalize(a)
    
    return best_a, best_score

Now let me also think about a completely different approach that might be more effective:

**Approach: Iterative Proportional Fitting (IPF) / Sinkhorn-like**

The idea: 
1. Initialize a to be uniform
2. Compute b = conv(a, a)
3. For each position k where b[k] > target, reduce the contributing a[i] values proportionally
4. Renormalize
5. Repeat

This is like a "water-filling" where we pour water (mass) into the sequence and keep leveling the peaks.

More specifically:
- target = (sum(b)) / (2n-1) = 1/(2n-1) (the flat convolution level)
- But we can't achieve perfectly flat, so we use a higher target
- For each peak k with b[k] > target:
  - For each contributing pair (i, j) with i+j = k (in convolution indexing):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set Gauss-Newton method
combined with mirror descent warmup and coordinate descent polishing.
Key innovation: identifies top-K correlation peaks and solves a linearized
KKT system to equalize them simultaneously, providing much faster convergence
than pure subgradient methods near the optimum. Uses adaptive step sizing,
multiple K values, and a bisection-like refinement on the peak target.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max()), b


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def build_gradient_matrix(a, peak_indices):
    """Build matrix G where G[i, j] = 2*a[peak_indices[i] - j] if valid."""
    n = len(a)
    m = len(peak_indices)
    G = np.zeros((m, n))
    for i, k in enumerate(peak_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            js_arr = np.arange(js, je + 1)
            G[i, js:je + 1] = 2.0 * a[k - js_arr]
    return G


def gauss_newton_step(a, K, step_scale, target_fraction=0.0):
    """
    Active-set Gauss-Newton step.
    Identifies top K peaks, builds linear system to equalize them,
    and takes a step in the solution direction.
    
    target_fraction: if > 0, target is max * (1 - target_fraction)
                     if == 0, target is mean of top K
    """
    n = len(a)
    b = fast_convolve(a)
    
    peak_indices = np.argsort(b)[::-1][:K]
    peak_values = b[peak_indices]
    
    if target_fraction > 0:
        target = peak_values[0] * (1.0 - target_fraction)
    else:
        target = peak_values.mean()
    
    # Excess: how much each peak is above target
    e = peak_values - target
    
    # Only work with peaks that are meaningfully above target
    active_mask = e > 1e-15 * peak_values[0]
    if not np.any(active_mask):
        return a.copy()
    
    active_k = peak_indices[active_mask]
    active_e = e[active_mask]
    m = len(active_k)
    
    # Build gradient matrix
    G = build_gradient_matrix(a, active_k)
    
    # Solve G*d = -active_e in least squares with regularization
    # d = G^T (G G^T + reg*I)^{-1} (-active_e)
    GGt = G @ G.T
    trace_val = np.trace(GGt)
    if trace_val < 1e-300:
        return a.copy()
    reg = 1e-10 * trace_val / m
    try:
        d = G.T @ np.linalg.solve(GGt + reg * np.eye(m), -active_e)
    except np.linalg.LinAlgError:
        return a.copy()
    
    # Project to maintain sum(a)=1: ensure sum(d) = 0
    d = d - np.sum(d) / n
    
    # Scale step for stability
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-300:
        return a.copy()
    
    # Limit step size relative to a
    a_norm = np.linalg.norm(a)
    max_step = 0.5 * a_norm
    if d_norm * step_scale > max_step:
        d = d * (max_step / (d_norm * step_scale))
    
    a_new = a + step_scale * d
    a_new = normalize(a_new)
    
    return a_new


def mirror_descent(a_init, time_limit, T_start=0.01, T_end=0.001,
                   eta_start=0.02, eta_end=0.001, max_iters=50000):
    """Mirror descent with temperature annealing on the simplex."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score, _ = score_from_a(a)
    t0 = time.monotonic()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
        
        # Softmax weights
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()
        
        # Gradient via cross-correlation
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 4.0 * n * correl[:n]
        
        # Project to tangent space of simplex
        grad_a = grad_a - np.dot(a, grad_a)
        
        # Multiplicative (mirror) update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -4.0, 4.0)
        a = a * np.exp(log_update)
        a = normalize(a)
    
    return best_a, best_score


def coordinate_descent(a_init, time_limit):
    """Focused coordinate descent on peak-contributing elements."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score, _ = score_from_a(a)
    t0 = time.monotonic()
    
    factors = np.array([0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3])
    no_improve = 0
    max_no_improve = 150
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Identify top contributing indices for this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = idx[np.argsort(contribs)[::-1][:30]]
        
        improved = False
        rng_order = np.random.permutation(len(top_idx))
        for ii in rng_order:
            i = top_idx[ii]
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                a_trial[i] = a_trial[i] * f
                a_trial = normalize(a_trial)
                sc, _ = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1
    
    return best_a, best_score


def gn_refine(a_init, time_limit, K_list=None, alpha_list=None, verbose=True):
    """
    Main Gauss-Newton refinement loop.
    Tries different K and step scales until time runs out.
    """
    if K_list is None:
        K_list = [3, 5, 7, 10, 15]
    if alpha_list is None:
        alpha_list = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
    
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score, _ = score_from_a(a)
    t0 = time.monotonic()
    
    total_iters = 0
    for K in K_list:
        for alpha in alpha_list:
            if time.monotonic() - t0 > time_limit:
                break
            # Do up to 200 sub-iterations with this (K, alpha)
            stuck = 0
            for sub_it in range(200):
                if time.monotonic() - t0 > time_limit:
                    break
                a_new = gauss_newton_step(a, K=K, step_scale=alpha)
                sc, _ = score_from_a(a_new)
                total_iters += 1
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    stuck = 0
                else:
                    stuck += 1
                    if stuck > 3:
                        break
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  GN: {total_iters} iters in {elapsed:.1f}s, best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))

        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
