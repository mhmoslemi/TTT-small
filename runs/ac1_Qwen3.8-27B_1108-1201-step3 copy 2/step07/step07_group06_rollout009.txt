import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def soft_max_weights(b, T):
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    return exp_b / exp_b.sum()


def grad_softmax(a, w):
    """Gradient of 2n * T * logsumexp(b/T) w.r.t. a where b=conv(a,a)."""
    n = len(a)
    a_flip = a[::-1]
    conv_len = n + len(w) - 1
    fs = 1 << (conv_len - 1).bit_length()
    Af = rfft(a_flip, fs)
    W = rfft(w, fs)
    conv = irfft(Af * W, fs)[:conv_len]
    C = conv[n-1:n-1+n]
    return 4.0 * n * C


def project_simplex(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

                a_trial = project_simplex(a_trial)
                sc = compute_score(a_trial)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = a_trial[i]

            if best_local != a[i]:
                a[i] = best_local
                a = project_simplex(a)
                if best_local_sc < best_score:
                    best_score = best_local_sc
                    best_a = a.copy()
                    improved_sweep = True

        if not improved_sweep:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    return fi / ss if ss > 1e-15 else np.ones(new_n) / new_n


def get_initializations(parent, n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    if parent is not None:
        if len(parent) == n:
            p = project_simplex(parent)
            inits.append(('parent', p.copy()))
            for scale in [0.02, 0.05, 0.1]:
                pp = p + scale * rng.standard_normal(n)
                pp = project_simplex(pp)
                inits.append((f'pert_{scale}', pp))
        else:
            p = interpolate_seq(parent, n)
            inits.append(('parent_interp', p))

    # Exponential
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        e = np.exp(-alpha * np.arange(n) / n)
        inits.append((f'expo_{alpha}', e / e.sum()))

    # Reverse exponential
    for alpha in [1.0, 2.0, 3.0]:
        e = np.exp(-alpha * np.arange(n-1, -1, -1) / n)
        inits.append((f'rev_expo_{alpha}', e / e.sum()))

    # Sum of two Gaussians
    for c1, c2, sigma in [(0.3, 0.7, 0.1), (0.2, 0.8, 0.15), (0.25, 0.75, 0.08)]:
        x = np.arange(n) / n
        g = np.exp(-0.5*((x-c1)/sigma)**2) + np.exp(-0.5*((x-c2)/sigma)**2)
        inits.append((f'gau_{c1}_{c2}', g / g.sum()))

    # Linear ramp
    r = np.linspace(1, 2, n)
    inits.append(('ramp', r / r.sum()))

    return inits
