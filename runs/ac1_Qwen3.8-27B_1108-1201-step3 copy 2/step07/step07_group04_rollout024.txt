```python
def project_simplex(v, s=1):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - s)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - s) / (rho + 1.0)
        return np.maximum(v - theta, 0)
    else:
        return np.full(n, s / n)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature FISTA
with FFT-based gradients, multiple initialization strategies, length exploration,
and peak-focused local polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Project vector onto the simplex {x >= 0, sum(x) = s} in O(n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - s) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.full(n, s / n)


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def smooth_objective(a, temp):
    """Smooth approximation: (2n/temp)*log(sum exp(temp*b_k)) / sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf'), np.zeros(n)
    b = fast_convolve(a)
    sb = temp * b
    sb_max = sb.max()
    exp_b = np.exp(sb - sb_max)
    lse = sb_max + np.log(exp_b.sum())
    val = (2.0 * n / temp) * lse / (s * s)
    return val


def compute_smooth_gradient(a, temp):
    """
    Compute gradient of (2n/temp)*log(sum_k exp(temp*b_k)) w.r.t. a.
    Returns (gradient_vector, b_vector).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    b = fast_convolve(a)
    sb = temp * b
    sb_max = sb.max()
    exp_b = np.exp(sb - sb_max)
    p = exp_b / exp_b.sum()

    # grad_i = 2n * sum_k p_k * 2 * a_{k-i} (for valid k-i)
    # = 4n * cross_correlation(p, a)[i]
    # where cross_correlation(p,a)[i] = sum_m p[m+i]*a[m]
    fft_size = 1 << (2 * n - 1).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    X = irfft(np.conj(P) * A, fft_size)

    grad = np.zeros(n)
    grad[0] = X[0]
    if n > 1:
        grad[1:] = X[fft_size - 1::-1][:n - 1]

    return 4.0 * n * grad, b


def fista_optimize(a0, time_limit, verbose=False):
    """
    FISTA-like optimization with adaptive temperature schedule.
    Starts with high temperature (smooth) and gradually decreases.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t_start = monotonic()

    # Temperature schedule: start rough, get fine
    # Scale relative to typical b values
    b_scale = fast_convolve(a).max()
    if b_scale < 1e-10:
        b_scale = 1.0 / n

    temps = [8.0 / b_scale, 4.0 / b_scale, 2.0 / b_scale, 1.0 / b_scale,
             0.5 / b_scale, 0.25 / b_scale, 0.125 / b_scale, 0.0625 / b_scale,
             0.03125 / b_scale, 0.015625 / b_scale]

    # Time allocation per temperature (earlier = more time for global optimization)
    time_fractions = [0.15, 0.12, 0.10, 0.10, 0.09, 0.09, 0.08, 0.07, 0.07, 0.13]

    for idx, temp in enumerate(temps):
        if monotonic() - t_start > time_limit:
            break
        t_phase_end = t_start + time_limit * min(1.0, sum(time_fractions[:idx + 1]))

        # FISTA parameters
        a_curr = a.copy()
        a_prev = a.copy()
        t_mom = 1.0
        iter_count = 0

        # Step size: conservative based on Lipschitz estimate
        # L ~ 4n/temp (rough), so step ~ temp/(4n)
        step = temp / (2.0 * n)

        while monotonic() < t_phase_end:
            iter_count += 1

            # FISTA momentum extrapolation
            if t_mom > 1.0:
                y = a_curr + ((t_mom - 1.0) / (t_mom + 1.0)) * (a_curr - a_prev)
                y = np.maximum(y, 0)
                y_sum = y.sum()
                if y_sum < 1e-15:
                    y = a_curr.copy()
                else:
                    y = y / y_sum
            else:
                y = a_curr.copy()

            # Compute gradient
            grad, b = compute_smooth_gradient(y, temp)

            # Gradient step + projection
            a_new = y - step * grad
            a_new = project_simplex(a_new, 1.0)

            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            a_prev = a_curr.copy()
            a_curr = a_new
            t_mom = t_new

            # Track best
            score = compute_score(a_curr)
            if score < best_score:
                best_score = score
                best_a = a_curr.copy()

            # Adaptive step: if not improving, reduce step
            if iter_count % 500 == 0 and step > temp / (50.0 * n):
                step *= 0.9

        a = a_curr

        if verbose:
            elapsed = monotonic() - t_start
            print(f"    FISTA temp={temp:.6f}: score={best_score:.6f} ({elapsed:.1f}s, {iter_count} iters)")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """
    Peak-focused subgradient optimization with adaptive step and momentum.
    Focuses on reducing the top peaks of the convolution.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t_start = monotonic()

    momentum_vec = np.zeros(n)
    lr = 0.008
    beta = 0.85
    no_improve = 0
    patience = 2000

    while monotonic() - t_start < time_limit:
        b = fast_convolve(a)
        # Select top peaks with softmax weighting
        top_k = min(20, len(b))
        peak_idx = np.argpartition(b, -top_k)[-top_k:]
        peak_vals = b[peak_idx]
        temp_w = 0.003 * max(peak_vals.max(), 1e-10)
        w = np.exp((peak_vals - peak_vals.max()) / temp_w)
        w = w / w.sum()

        # Compute subgradient: weighted sum of gradients at top peaks
        grad = np.zeros(n)
        for ii in range(len(peak_idx)):
            if w[ii] < 1e-12:
                continue
            k = int(peak_idx[ii])
            # grad of b[k] w.r.t. a: 2*a_{k-j} for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                other = k - j
                if 0 <= other < n:
                    grad[j] += 2.0 * w[ii] * a[other]

        # Momentum update
        momentum_vec = beta * momentum_vec - lr * grad
        a_new = a + momentum_vec
        a_new = project_simplex(a_new, 1.0)

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new  # Still follow the trajectory
            no_improve += 1

        # Adaptive learning rate
        lr *= 0.9995
        if no_improve > patience:
            if no_improve > patience * 3:
                break
            # perturb to escape
            a = best_a + np.random.standard_normal(n) * 0.001
            a = project_simplex(a, 1.0)
            momentum_vec = np.zeros(n)
            lr = max(lr * 3, 0.004)
            no_improve = 0

        if verbose and (a - best_a).__abs__().sum() < 1e-10:
            pass

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def diverse_init(n, seed):
    """Generate diverse initial sequences."""
    rng = np.random.RandomState(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Slightly perturbed uniform
    a = np.ones(n) + rng.standard_normal(n) * 0.1
    a = np.maximum(a, 0)
    inits.append(a / a.sum())

    # Bimodal (mass at edges)
    t = np.linspace(0, 1, n)
    a = 0.3 + 0.7 * (4 * t * (1 - t)) ** 0.5  # U-shaped
    a = np.maximum(a, 0)
    inits.append(a / a.sum())

    # Triangular (mass in middle)
    a = 1.0 - np.abs(2 * t - 1)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # Exponential decay
    a = np.exp(-3 * t)
    inits.append(a / a.sum())

    # Random exponential
    a = rng.exponential(1.0, size=n)
    inits.append(a / a.sum())

    # Cosine-based
    a = 0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())

    # "Anti-peak" - suppress center
    a = np.ones(n)
    center = n // 2
    width = n // 8
    a[center - width:center + width + 1] *= 0.7
    a = np.maximum(a, 0)
    inits.append(a / a.sum())

    return inits

        a_opt, sc = fista_optimize(init, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_opt, f"fista_n{n_try}", best_seq, best_score)

    # === Phase 2: Subgradient polish on best ===
    print(f"\n=== Phase 2: Subgradient Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.2, 250.0)
            print(f"  SubGrad n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_sg, _ = subgradient_polish(best_seq, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.15, 150.0)
            print(f"  PeakShave n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave_local(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 4: FISTA with diverse initializations at best length ===
    print(f"\n=== Phase 4: Diverse Init FISTA (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - monotonic()
        if rem > 40:
            inits = diverse_init(n_best, seed + 100)
            t_per_init = min(rem * 0.6 / len(inits), 100.0)
            for i, init in enumerate(inits):
                rem = deadline - monotonic()
                if rem < 20:
                    break
                t_alloc = min(t_per_init, rem * 0.9)
                if t_alloc < 15:
                    continue
                # Blend with current best
                blend = 0.5 * init + 0.5 * interpolate_sequence(best_seq, n_best)
                blend = project_simplex(blend, 1.0)
                a_opt, _ = fista_optimize(blend, t_alloc, verbose=False)
                best_seq, best_score = check_and_update(a_opt, f"diverse_{i}_n{n_best}", best_seq, best_score)

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - monotonic()
        if rem > 40:
            alt_lengths = []
            for factor in [1.5, 2.0, 0.75, 1.25]:
                alt_n = int(n_best * factor)
                if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)
            alt_lengths = sorted(alt_lengths)[:3]

            for alt_n in alt_lengths:
                rem = deadline - monotonic()
                if rem < 30:
                    break
                t_alloc = min(rem * 0.3, 120.0)
                init_alt = interpolate_sequence(best_seq, alt_n)
                print(f"  Length explore n={alt_n}, budget={t_alloc:.0f}s")
                a_alt, _ = fista_optimize(init_alt, t_alloc, verbose=True)
                best_seq, best_score = check_and_update(a_alt, f"len_n{alt_n}", best_seq, best_score)

                # Quick polish
                rem = deadline - monotonic()
                if rem > 15:
                    a_pol, _ = subgradient_polish(best_seq, min(rem * 0.2, 40.0))
                    best_seq, best_score = check_and_update(a_pol, f"polish_n{len(best_seq)}", best_seq, best_score)

    # === Phase 6: Final intensive FISTA on best ===
    print(f"\n=== Phase 6: Final FISTA (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.5, 200.0)
            print(f"  Final FISTA n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_final, _ = fista_optimize(best_seq, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_final, "final_fista", best_seq, best_score)

    # === Phase 7: Final peak shaving ===
    print(f"\n=== Phase 7: Final Peak Shave (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic()
        if rem > 10:
            t_alloc = min(rem * 0.6, 60.0)
            a_final_ps, _ = peak_shave_local(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_final_ps, "final_ps", best_seq, best_score)

    # === Phase 8: One more subgradient pass ===
    if best_seq is not None:
        rem = deadline - monotonic()
        if rem > 10:
            t_alloc = min(rem * 0.8, 30.0)
            a_last, _ = subgradient_polish(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_last, "last_sg", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
