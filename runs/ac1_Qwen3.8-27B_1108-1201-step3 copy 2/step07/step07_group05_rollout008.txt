```python
import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def compute_score_and_conv(a):
    """Compute score and the convolution."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv
```

```python
1. Fast exact pair line search with smart pair selection
2. Peak-aware coordinate moves
3. Multi-temperature softmax L-BFGS as a global guide
4. Iterative peak equalization
The key improvement over previous approaches is the use of exact 1D
line search for pairwise moves, which converges much faster per evaluation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def pair_line_search(a, i, j):
    """
    Find optimal delta for moving mass from i to j (a[i]-=delta, a[j]+=delta)
    or from j to i. Returns (delta, new_max_conv) where delta>0 means i->j.
    """
    n = len(a)
    size = 2 * n - 1
    conv = fast_convolve(a)
    max_conv = conv.max()
    
    # Compute d[k] = change coefficient for linear term
    d = np.zeros(size)
    q = np.zeros(size)
    
    # a[j] += delta contributes:
    #   +2*a[k-j] to conv[k] for k in [j, j+n-1] (where k-j in [0,n-1])
    #   with quadratic correction at k=2j
    js, je = j, min(size, j + n)
    if js < je:
        d[js:je] += 2.0 * a[0:je-js]
    if 2*j < size:
        q[2*j] += 1.0
    
    # a[i] -= delta contributes (negative):
    #   -2*a[k-i] to conv[k] for k in [i, i+n-1] (where k-i in [0,n-1])
    #   with quadratic correction at k=2i
    is_, ie = i, min(size, i + n)
    if is_ < ie:
        d[is_:ie] -= 2.0 * a[0:ie-is_]
    if 2*i < size:
        q[2*i] -= 1.0
    
    # Minimize max(conv + d*delta + q*delta^2) over delta in appropriate range
    best_delta = 0.0
    best_val = max_conv
    
    # Forward: delta in [0, a[i]]
    if a[i] > 1e-14:
        lo, hi = 0.0, a[i]
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = (conv + d*m1 + q*m1*m1).max()
            f2 = (conv + d*m2 + q*m2*m2).max()
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2.0
        val = (conv + d*mid + q*mid*mid).max()
        if val < best_val:
            best_val = val
            best_delta = mid
    
    # Reverse: equivalent to -delta in [0, a[j]], so minimize max(conv - d*delta - q*delta^2)
    if a[j] > 1e-14:
        lo, hi = 0.0, a[j]
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = (conv - d*m1 - q*m1*m1).max()
            f2 = (conv - d*m2 - q*m2*m2).max()
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2.0
        val = (conv - d*mid - q*mid*mid).max()
        if val < best_val:
            best_val = val
            best_delta = -mid
    
    return best_delta, best_val


def exact_cd(a0, time_limit, n_pairs_per_iter=200, verbose=False):
    """Coordinate descent with exact pair line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Get current convolution to identify important coordinates
        conv = fast_convolve(a)
        # Find coordinates contributing to top peaks
        top_k_idx = np.argpartition(conv, -10)[-10:]
        important_coords = set()
        for ki in top_k_idx:
            ki = int(ki)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            important_coords.update(range(lo, hi + 1))
        
        # Generate pairs: bias towards important coordinates
        for _ in range(n_pairs_per_iter):
            r = rng.random()
            if r < 0.6 and important_coords:
                pc = list(important_coords)
                i = pc[rng.integers(0, len(pc))]
                j = rng.integers(0, n)
            else:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            
            if i == j:
                continue
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            delta, new_max = pair_line_search(a, i, j)
            
            if abs(delta) > 1e-15:
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                ss = a.sum()
                a = a / ss
                new_score = 2.0 * n * new_max
                if new_score < best_score - 1e-14:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            stall_count += 1
            if stall_count > 20:
                # Perturb to escape
                noise = rng.standard_normal(n) * 0.005
                a = best_a + noise
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_score = compute_score(a)
                stall_count = 0
        else:
            stall_count = 0
    
    return best_a, best_score
```
