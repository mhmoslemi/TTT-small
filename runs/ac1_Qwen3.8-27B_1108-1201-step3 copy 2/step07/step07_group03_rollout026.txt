    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_gradient(a, T):
    """
    Compute gradient of F_T(a) = T * logsumexp(conv(a,a) / T) w.r.t. a.
    
    Returns (gradient, weights, conv_values).
    Gradient formula: grad_m = (2/T) * sum_j w[j+m] * a[j]
    where w = softmax(conv(a,a)/T).
    Computed via FFT: IFFT(conj(FFT(w,N)) * FFT(a,N))[:n]
    """
    n = len(a)
    b = fast_convolve(a)
    # Numerically stable softmax
    b_max = b.max()
    e = np.exp((b - b_max) / T)
    w = e / e.sum()
    
    # Compute cross-correlation via FFT
    N = 1 << ((3 * n - 2) - 1).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(np.conj(W) * A)[:n].real
    grad = (2.0 / T) * corr
    return grad, w, b

def softmax_gradient_sum_top(a, T, k):
    """
    Gradient of T * logsumexp(top-k values of conv(a,a) / T).
    Uses soft selection via Lagrange multiplier.
    """
    n = len(a)
    b = fast_convolve(a)
    # Find threshold for top-k
    if k >= len(b):
        return softmax_gradient(a, T)
    b_sorted = np.sort(b)[::-1]
    threshold = b_sorted[k - 1]
    # Soft top-k weights
    b_max = b.max()
    e = np.exp((b - b_max) / T)
    # Scale by soft indicator of being in top-k
    mask = np.exp((b - threshold) / (0.5 * T))
    w = e * mask
    ws = w.sum()
    if ws < 1e-30:
        return softmax_gradient(a, T)
    w = w / ws
    
    N = 1 << ((3 * n - 2) - 1).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(np.conj(W) * A)[:n].real
    grad = (2.0 / T) * corr
    return grad, w, b

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if cond.any():
        rho_val = np.nonzero(cond)[0][-1] + 1
        theta = (cssv[rho_val - 1] - 1.0) / rho_val
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

class AdamState:
    """State for Adam optimizer."""
    def __init__(self, n):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0

def adam_update(a, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step with simplex projection."""
    state.t += 1
    state.m = beta1 * state.m + (1 - beta1) * grad
    state.v = beta2 * state.v + (1 - beta2) * (grad ** 2)
    m_hat = state.m / (1 - beta1 ** state.t)
    v_hat = state.v / (1 - beta2 ** state.t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return simplex_project(a_new)

def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    if solver is None:
        solver = _find_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def _find_solver():
    global _cached_solver
    if '_cached_solver' in globals() and _cached_solver is not None:
        return _cached_solver
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = solver_name
                return solver_name
        except Exception:
            continue
    _cached_solver = cp.ECOS
    return cp.ECOS

def optimize_softmax_adam(a_init, time_limit, T_start=0.5, T_end=0.02,
                          lr_start=0.01, lr_end=0.001, verbose=True):
    """
    Main optimizer: Adam with softmax-smoothed objective and temperature annealing.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    state = AdamState(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Temperature schedule: cosine annealing
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        
        # Learning rate schedule
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))
        
        # Compute gradient
        grad, w, b = softmax_gradient(a, T)
        
        # Add a small L2 regularization to prevent concentration
        grad += 1e-4 * a
        
        # Adam step
        a = adam_update(a, grad, state, lr)
        
        # Evaluate periodically
        if it % 20 == 0 or it <= 5:
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    iter={it} T={T:.4f} score={best_score:.6f}")
            else:
                no_improve += 1
        
        # If stuck, perturb
        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.005
            a = simplex_project(best_a + noise)
            state = AdamState(n)
            no_improve = 0
    
    return best_a, best_score

def coord_descent_polish(a0, time_limit, n_sweeps=500):
    """Coordinate descent: try scaling each coordinate to reduce the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                # Try setting a small value
                trial = a.copy()
                trial[i] = 1e-4
                trial = trial / trial.sum()
                score = 2.0 * n * fast_convolve(trial).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = trial.copy()
                    a = trial
                    improved = True
                continue
            
            for factor in [0.7, 0.8, 0.9, 1.1, 1.25, 1.5]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
        
        if not improved and sweep > 20:
            break
    
    return best_a, best_score

def peak_shave(a0, time_limit):
    """Targeted peak shaving: reduce contributions to the highest peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(3000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        top_k = min(3, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        
        improved = False
        for pk in peaks:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[pk - idx]
            
            # Find top contributors
            top_c = np.argsort(contribs)[::-1][:3]
            for ci in top_c:
                i = int(idx[ci])
                if a[i] < 1e-14:
                    continue
                for factor in [0.85, 0.75, 0.65]:
                    trial = a.copy()
                    trial[i] *= factor
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()
                    score = 2.0 * n * fast_convolve(trial).max()
                    if score < best_score - 1e-13:
                        a = trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try paired reductions
            pk = int(peaks[0])
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            if len(idx) >= 2:
                contribs = a[idx] * a[pk - idx]
                top_c = np.argsort(contribs)[::-1][:6]
                for ci1 in top_c[:3]:
                    for ci2 in top_c[:3]:
                        if ci1 == ci2:
                            continue
                        i1, i2 = int(idx[ci1]), int(idx[ci2])
                        if a[i1] < 1e-14 or a[i2] < 1e-14:
                            continue
                        for f1 in [0.9, 0.8]:
                            for f2 in [0.9, 0.8]:
                                trial = a.copy()
                                trial[i1] *= f1
                                trial[i2] *= f2
                                trial = np.maximum(trial, 0)
                                trial = trial / trial.sum()
                                score = 2.0 * n * fast_convolve(trial).max()
                                if score < best_score - 1e-13:
                                    a = trial
                                    best_score = score
                                    best_a = a.copy()
                                    improved = True
                                    break
                            if improved:
                                break
                        if improved:
                            break
                    if improved:
                        break
            if not improved:
                break
    
    return best_a, best_score

def damped_fp_refine(a0, time_limit, solver=None):
    """Damped fictitious play: alternate with LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(a, solver)
        if b_new is None:
            break
        alpha = min(1.0, 5.0 / (it + 5))
        a_new = np.maximum((1 - alpha) * a + alpha * b_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

def init_cosine(n):
    """Raised cosine initialization."""
    t = np.linspace(0, 1, n)
    a = np.cos(np.pi * (t - 0.5)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_gaussian(n, sigma_frac=0.3):
    """Gaussian bump initialization."""
    t = np.linspace(0, 1, n)
    sigma = sigma_frac
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_fourier_mix(n, rng):
    """Fourier superposition initialization."""
    n_harm = int(rng.integers(3, 12))
    a = np.full(n, 0.5)
    for j in range(n_harm):
        amp = rng.random() * 0.3
        freq = int(rng.integers(1, n // 2 + 1))
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_bimodal(n, rng, gap_frac=0.15):
    """Bimodal: two peaks with a gap in the middle."""
    center = n // 2
    gap = int(gap_frac * n)
    a = np.zeros(n)
    # Left peak
    left = np.exp(-0.5 * ((np.arange(center - gap // 2) - (center - gap) // 2) / (gap // 4 + 1)) ** 2)
    a[:center - gap // 2] = left[:len(left)]
    # Right peak  
    right = np.exp(-0.5 * ((np.arange(n - (center + gap // 2)) - gap // 4) / (gap // 4 + 1)) ** 2)
    a[center + gap // 2:] = right[:len(right)]
    a = np.maximum(a, 0)
    # Add small background
    a += 0.05
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via the actual evaluator and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

            a_fin2, _ = coord_descent_polish(best_seq, t7b, n_sweeps=200)
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_cd")
        
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t7c = min(rem * 0.5, 40.0)
            a_fin3, _ = peak_shave(best_seq, t7c)
            best_seq, best_score = check_and_update(a_fin3, best_seq, best_score, "final_ps")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
