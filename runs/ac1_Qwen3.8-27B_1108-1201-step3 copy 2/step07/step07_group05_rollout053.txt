    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return f.copy()
    if g.value is None:
        return f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.3, verbose=True):
    """LP-based fictitious play."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)

        g_sum = g.sum()

        a_new = (1 - step) * a + step * g_norm

def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def smoothed_objective(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T=300, maxiter=5000):
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        return a_out, compute_score(a_out)
    except Exception:
        return a0.copy(), compute_score(a0)


def coordinate_descent(a0, time_limit, n_pairs=200):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            
            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 0 else a[i]
            
            for frac in [0.05, 0.1, 0.2, 0.4, 0.6]:
                t = frac * base
                if t >= a[i]:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        
        if not improved:
            noise = rng.standard_normal(n) * 0.001
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
LP-based fictitious play for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Core algorithm: at each iteration, solve the best-response LP
  max sum(g) s.t. C(f) @ g <= M, g >= 0
where C(f) is the convolution matrix. Blend the solution with the current
iterate using an adaptive step size. Polish with L-BFGS and coordinate descent.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    """LP-based fictitious play with warm-starting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    stall_threshold = 500

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        # Adaptive: if stalled, reduce step and add small perturbation
        if stall_count > stall_threshold:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.0005
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def smoothed_objective(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=200, verbose=True):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]

            for frac in [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 20:
            noise = rng.standard_normal(n) * 0.002
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lp_with_decreasing_step(a0, time_limit, verbose=True):
    """LP fictitious play with 1/(1+alpha*k) step schedule."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    alpha = 0.05

    while time.monotonic() - t0 < time_limit:
        iters += 1
        step = 1.0 / (1.0 + alpha * iters)
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-dec iter={iters}, best={best_score:.8f}, step={step:.5f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def bussgang_iteration(a0, time_limit, verbose=True):
    """
    Bussgang-type iteration: flatten the power spectrum.
    1. A = FFT(a)
    2. |A|^2 is the periodogram
    3. Adjust A to flatten |A|^2
    4. a_new = IFFT(A_adjusted), project to non-negative
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0
    tau = 0.5  # smoothing parameter

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        # Target: flat power spectrum
        target = total_power / fft_size
        # Bussgang: multiply by (target/power)^tau
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.1, 10.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Bussgang iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP fictitious play (multiple step sizes) - main driver
    2. LP with decreasing step schedule
    3. Bussgang spectral flattening
    4. L-BFGS smoothing polish
    5. Coordinate descent
    6. Final LP + nearby lengths
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: LP Fictitious Play - large step (fast progress)
    print(f"\n=== Phase 1: LP-FP Large Step (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
    a_lp1, sc_lp1 = lp_fictitious_play(best_seq, t1, step=0.5, label="LP-s0.5")
    best_seq, best_score = check_and_update(a_lp1, "lp_step0.5", best_seq, best_score)

    # Phase 2: LP Fictitious Play - medium step
    print(f"\n=== Phase 2: LP-FP Medium Step (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if rem > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t2, step=0.3, label="LP-s0.3")
        best_seq, best_score = check_and_update(a_lp2, "lp_step0.3", best_seq, best_score)

    # Phase 3: LP with decreasing step (for convergence)
    print(f"\n=== Phase 3: LP Decreasing Step (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if rem > 60:
        print(f"  Budget: {t3:.0f}s")
        a_lp3, sc_lp3 = lp_with_decreasing_step(best_seq, t3, verbose=True)
        best_seq, best_score = check_and_update(a_lp3, "lp_decr", best_seq, best_score)

    # Phase 4: Bussgang spectral flattening
    print(f"\n=== Phase 4: Bussgang (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.08, 80.0)
    if rem > 40:
        print(f"  Budget: {t4:.0f}s")
        a_bg, sc_bg = bussgang_iteration(best_seq, t4, verbose=True)
        best_seq, best_score = check_and_update(a_bg, "bussgang", best_seq, best_score)

    # Phase 5: L-BFGS smoothing
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [300, 1000, 3000]:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.12, 80.0)
        if rem < 30:
            break
        print(f"  L-BFGS T={T_val}, budget={t5:.0f}s")
        a_lb, sc_lb = lbgfs_optimize(best_seq, t5, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # Phase 6: Coordinate descent
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 120.0)
    if rem > 30:
        print(f"  Budget: {t6:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t6, n_pairs=300, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # Phase 7: Final LP polish (small step)
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.20, 100.0)
    if rem > 20:
        print(f"  Budget: {t7:.0f}s")
        a_lp4, sc_lp4 = lp_fictitious_play(best_seq, t7, step=0.1, label="LP-final")
        best_seq, best_score = check_and_update(a_lp4, "lp_final", best_seq, best_score)

    # Phase 8: Try nearby lengths
    print(f"\n=== Phase 8: Nearby Lengths (best={best_score:.8f}) ===")
    for alt_n in [n_parent + 2, n_parent + 4, n_parent - 2, n_parent + 6]:
        if alt_n < 64 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t8 = min(rem * 0.15, 60.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  L-BFGS n={alt_n}, budget={t8:.0f}s")
        a_alt, sc_alt = lbgfs_optimize(init_alt, t8, T=500, maxiter=2000)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # Phase 9: Final LP on best
    print(f"\n=== Phase 9: Final LP on best (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.4, 50.0)
        a_final, sc_final = lp_fictitious_play(best_seq, t9, step=0.15, label="LP-end")
        best_seq, best_score = check_and_update(a_final, "lp_end", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
