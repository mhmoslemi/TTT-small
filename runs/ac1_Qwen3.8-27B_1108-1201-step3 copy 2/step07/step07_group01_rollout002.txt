For ternary search, we need to define the range of a_i. Since a_i >= 0 and sum(a) = 1, we have 0 <= a_i <= 1 (actually <= 1 - sum_{j≠i} a_j).

But with proportional redistribution, if we set a_i = x, the others become a_j * (1-x)/(1-a_i_old) for j ≠ i. This is valid as long as x >= 0 and the others remain non-negative (which they do if x <= 1).

    best_score = 2.0 * n * fast_convolve(a).max()

        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            # Try ternary search on a[i]
            old_val = a[i]
            # Range: [0, min(old_val * 3, 1 - (1-old_val))] = [0, ...]
            # Actually, with proportional redistribution, a[i] can be any value in [0, 1)
            lo = 0.0
            hi = min(1.0, old_val * 5 + 0.01)  # generous upper bound
            # Actually, let's be more careful. The others sum to 1 - old_val.
            # After redistribution, they sum to 1 - x where x is the new a[i].
            # They must be non-negative, which is automatically satisfied for x in [0, 1].
            # But we don't want x to be too large or too small.
            
            for _ in range(50):  # ternary search iterations
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                score1 = eval_coord(a, i, m1)
                score2 = eval_coord(a, i, m2)
                if score1 < score2:
                    hi = m2
                else:
                    lo = m1
            
            opt_val = (lo + hi) / 2
            opt_score = eval_coord(a, i, opt_val)
            if opt_score < best_score - 1e-15:
                a = set_coord(a, i, opt_val)
                best_score = opt_score
                improved = True
    
    return a, best_score

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def compute_score(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def smooth_objective(z, n, T):
    a = softmax_fn(z)
    b = fast_convolve(a)
    # T * logsumexp(b, T)
    bs = b / T
    m = bs.max()
    lse = m + np.log(np.exp(bs - m).sum())
    return float(2.0 * n * T * lse)

def smooth_gradient(z, n, T):
    a = softmax_fn(z)
    b = fast_convolve(a)
    bs = b / T
    m = bs.max()
    w = np.exp(bs - m)
    w = w / w.sum()
    
    # Cross-correlation: corr[j] = sum_k w[k] * a[k-j]
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    
    w_dot_b = float(np.dot(w, b))
    grad = 4.0 * n * a * (corr - w_dot_b)
    return grad

def adam_optimize(z0, n, T, time_budget, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8):
    z = z0.copy().astype(np.float64)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    best_score = compute_score(softmax_fn(z))
    best_z = z.copy()
    t0 = time.monotonic()
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_budget:
            break
        
        F = smooth_objective(z, n, T)
        g = smooth_gradient(z, n, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if t % 500 == 0:
            a = softmax_fn(z)
            sc = compute_score(a)
            if sc < best_score - 1e-15:
                best_score = sc
                best_z = z.copy()
    
    a = softmax_fn(z)
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_z = z.copy()
    
    return best_z, best_score

def multi_epoch_adam(a_init, n, time_budget):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    best_z = z0.copy()
    best_score = compute_score(softmax_fn(z0))
    
    T_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    lr_schedule = [0.005, 0.004, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001]
    
    t_start = time.monotonic()
    
    for idx, (T, lr) in enumerate(zip(T_schedule, lr_schedule)):
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 5:
            break
        epoch_budget = min(remaining * 0.4, 120.0)
        z, sc = adam_optimize(best_z, n, T, epoch_budget, lr=lr)
        a = softmax_fn(z)
        real_sc = compute_score(a)
        if real_sc < best_score - 1e-15:
            best_score = real_sc
            best_z = z.copy()
            print(f"  Adam epoch T={T}: score={best_score:.7f}")
    
    return softmax_fn(best_z), best_score
