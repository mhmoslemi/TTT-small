import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute the score (normalized so sum(a)=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = fast_convolve(a_norm)
    return 2.0 * n * c.max()

                    if 0 <= j1 < n:
                        val += a[j1]
                    if 0 <= j2 < n and j2 != j1:
                        val += a[j2]
                    g[i] += wm * val
        
        # Multiplicative update
        g_max = g.max()
        if g_max > 1e-15:
            a = a * np.exp(-eta * g / g_max)
            a = np.maximum(a, 1e-30)
            s = a.sum()
            a = a / s
        
        # Evaluate periodically
        if step % 20 == 0 or step <= 10:
            score = compute_objective(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adaptive: reduce eta, restart if stuck
        eta *= eta_decay
        
        if no_improve > max_no_improve:
            # Perturb from best
            a = best_a * (1.0 + 0.02 * np.random.standard_normal(n))
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
            no_improve = 0
            eta = eta_init * 0.5
        
        if verbose and step % 5000 == 0:
            print(f"      Subgrad step={step} eta={eta:.4f} score={best_score:.6f}")
    
    return best_a, best_score


def subgrad_vectorized(a0, time_limit, eta_init=0.3, eta_decay=0.9998,
                       top_k=1, verbose=False):
    """
    Vectorized multiplicative subgradient descent.
    Uses the fact that g[i] = a[k*-i] + a[i-k*] can be computed via array indexing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a / s
    
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    eta = eta_init
    step = 0
    no_improve = 0
    max_no_improve = 3000
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fast_convolve(a)
        
        if top_k == 1:
            # Vectorized subgradient for single peak
            k_star = int(np.argmax(c[1:-1])) + 1  # exclude endpoints
            # g[i] = a[k_star - i] (if valid) + a[i - k_star] (if valid and != k_star - i)
            idx1 = k_star - np.arange(n)  # k_star - i for i = 0..n-1
            idx2 = np.arange(n) - k_star  # i - k_star for i = 0..n-1
            
            g = np.zeros(n)
            valid1 = (idx1 >= 0) & (idx1 < n)
            g[valid1] += a[idx1[valid1]]
            valid2 = (idx2 >= 0) & (idx2 < n)
            # Avoid double counting when idx1[i] == idx2[i], i.e., k_star - i == i - k_star, i.e., 2*i == 2*k_star... 
            # Actually idx1[i] == idx2[i] means k_star - i == i - k_star means 2*k_star == 2*i means i == k_star
            # But that's only one index, so double counting is negligible
            g[valid2] += a[idx2[valid2]]
            # Fix the double count at i where both are valid and equal
            both_valid = valid1 & valid2
            equal_mask = both_valid & (idx1 == idx2)
            g[equal_mask] -= a[idx1[equal_mask]]  # subtract the double count
        else:
            # Multi-peak version
            c_search = c.copy()
            c_search[0] = -np.inf
            c_search[-1] = -np.inf
            top_indices = np.argsort(c_search)[::-1][:top_k]
            top_values = c[top_indices]
            weights = top_values / top_values.sum()
            
            g = np.zeros(n)
            for m, km in enumerate(top_indices):
                wm = weights[m]
                idx1 = km - np.arange(n)
                idx2 = np.arange(n) - km
                valid1 = (idx1 >= 0) & (idx1 < n)
                g[valid1] += wm * a[idx1[valid1]]
                valid2 = (idx2 >= 0) & (idx2 < n)
                g[valid2] += wm * a[idx2[valid2]]
                both_valid = valid1 & valid2
                equal_mask = both_valid & (idx1 == idx2)
                g[equal_mask] -= wm * a[idx1[equal_mask]]
        
        # Multiplicative update
        g_max = g.max()
        if g_max > 1e-15:
            a = a * np.exp(-eta * g / g_max)
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
        
        # Evaluate
        if step % 10 == 0 or step <= 5:
            score = compute_objective(a)
            if score < best_score - 1e-11:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        eta *= eta_decay
        
        if no_improve > max_no_improve:
            a = best_a * (1.0 + 0.01 * np.random.standard_normal(n))
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
            no_improve = 0
            eta = eta_init * 0.3
        
        if verbose and step % 10000 == 0:
            print(f"      VecSubgrad step={step} eta={eta:.4f} score={best_score:.6f}")
    
    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def compute_grad_conv(a, w, n, fft_size):
    """
    Compute d/d(a) of sum_k w[k]*c[k] where c = conv(a,a).
    This equals 2 * cross_correlation(a, w) (approximately).
    Actually: d(dw^T c)/da[i] = sum_k w[k] * (a[k-i] + a[i-k])
    = (w * a)[i] + (flip(w) * a)[i] where * is cross-correlation
    
    More precisely using FFT:
    grad[i] = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    The first term is the cross-correlation (w ⊛ a)[i]
    The second term is also a cross-correlation with w flipped.
    
    Actually let's just use: grad[i] = sum_k w[k] * d(c[k])/d(a[i])
    = sum_k w[k] * (a[k-i] + a[i-k]) [with bounds]
    
    This can be computed as:
    grad = ifft(conj(FFT(w)) * FFT(a)) + ifft(conj(FFT(flip(w))) * FFT(a))
    
    Or more simply: grad[i] = (w*_a)[i] + (w*_a_flipped)[i]
    where *_a means cross-correlation.
    
    Let me use a simpler approach:
    grad[i] = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    
    First sum: for fixed i, sum_k w[k]*a[k-i] where k-i is in [0,n-1]
    = sum_j w[j+i]*a[j] (substituting j = k-i)
    This is the cross-correlation of w and a at lag i.
    
    Second sum: for fixed i, sum_k w[k]*a[i-k] where i-k is in [0,n-1]
    = sum_j w[i-j]*a[j] (substituting j = i-k)
    This is the cross-correlation of a and w at lag i.
    
    So grad[i] = corr(w,a)[i] + corr(a,w)[i] = corr(w,a)[i] + corr(w,a)[-i] (by symmetry of corr)
    
    Actually for real signals, corr(a,w)[i] = corr(w,a)[-i].
    So grad[i] = corr(w,a)[i] + corr(w,a)[-i]
    
    Using FFT: corr(w,a)[i] = ifft(conj(FFT(w)) * FFT(a))[i]
    """
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation corr(w,a)[i] = ifft(conj(W) * A)[i]
    corr = irfft(np.conj(W) * A, fft_size)
    # We need corr[i] + corr[-i mod fft_size] for i = 0..n-1
    # corr[-i] = corr[fft_size - i]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = corr[i] + corr[(fft_size - i) % fft_size]
    return grad


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='exp',
                  verbose=False, verbose_interval=10000):
    """Adam optimizer in log-space with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    # Make fft_size large enough for the convolution
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 10000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        if schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress
        
        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        
        # Gradient
        grad = compute_grad_conv(a_curr, w, n, fft_size)
        
        # Transform to z-space
        S = np.dot(grad, a_curr)
        grad_z = a_curr * (grad - S)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.0001, lr * 0.5)
        
        lr *= (1.0 - 0.00005)
        
        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=50):
    """Coordinate descent with multiplicative updates on single elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-20:
                continue
            
            best_local_score = best_score
            best_factor = 1.0
            
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 2.0]:
                a_trial = a.copy()
                a_trial[i] *= factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_factor = factor
            
            if best_factor != 1.0:
                a[i] *= best_factor
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True
        
        if not sweep_improved and sweep > 5:
            break
    
    return best_a, best_score


def pair_descent(a0, time_limit, n_trials=5000):
    """
    Pair descent: identify the peak and try to reduce it by adjusting pairs
    (i, k*-i) that contribute to the peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_search = c.copy()
        c_search[0] = -np.inf
        c_search[-1] = -np.inf
        k_star = int(np.argmax(c_search))
        peak_val = c[k_star]
        
        # Find contributing pairs
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        
        # Sort by contribution (descending)
        sorted_order = np.argsort(contribs)[::-1]
        
        improved = False
        for ci in sorted_order[:8]:
            i = int(idx[ci])
            j = int(k_star - i)
            if i == j or a[i] < 1e-20 or a[j] < 1e-20:
                continue
            
            # Try reducing the larger of the pair and increasing the smaller
            for f_big in [0.7, 0.8, 0.9]:
                for f_small in [1.0, 1.1, 1.2]:
                    a_trial = a.copy()
                    if a[i] >= a[j]:
                        a_trial[i] *= f_big
                        a_trial[j] *= f_small
                    else:
                        a_trial[i] *= f_small
                        a_trial[j] *= f_big
                    a_trial = np.maximum(a_trial, 1e-30)
                    s = a_trial.sum()
                    a_trial = a_trial / s
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved and trial % 200 == 0 and trial > 100:
            # Random perturbation to escape
            a = best_a * (1.0 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
    
    return best_a, best_score


def parametric_families(n, rng):
    """Generate a collection of parametric initializations."""
    candidates = []
    
    # Linear ramp (increasing)
    a = np.arange(1, n + 1, dtype=np.float64)
    a = a / a.sum()
    candidates.append(('linear', a))
    
    # Linear ramp (decreasing)
    a = np.arange(n, 0, -1, dtype=np.float64)
    a = a / a.sum()
    candidates.append(('linear_dec', a))
    
    # Power laws
    for p in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
        a = a / a.sum()
        candidates.append((f'power_{p}', a))
    
    # Exponential
    for r in [0.999, 0.995, 0.99, 0.98, 0.95]:
        a = np.power(r, np.arange(n, dtype=np.float64))
        a = a / a.sum()
        candidates.append((f'exp_{r}', a))
    
    # Gaussian (centered)
    x = np.linspace(-3, 3, n)
    a = np.exp(-x ** 2 / 2)
    a = a / a.sum()
    candidates.append(('gaussian', a))
    
    # Gaussian (off-center, shifted right)
    x = np.linspace(-1, 5, n)
    a = np.exp(-x ** 2 / 3)
    a = a / a.sum()
    candidates.append(('gauss_offset', a))
    
    # Triangular (peak at right)
    a = np.minimum(np.arange(n, dtype=np.float64), n - np.arange(n, dtype=np.float64))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    candidates.append(('triangular', a))
    
    # Quadratic increasing
    a = np.arange(1, n + 1, dtype=np.float64) ** 2
    a = a / a.sum()
    candidates.append(('quadratic', a))
    
    # Cube root
    a = np.cbrt(np.arange(1, n + 1, dtype=np.float64))
    a = a / a.sum()
    candidates.append(('sqrt3', a))
    
    # Logarithmic
    a = np.log(np.arange(2, n + 2, dtype=np.float64))
    a = a / a.sum()
    candidates.append(('log', a))
    
    # S-curve (sigmoid)
    x = np.linspace(-5, 5, n)
    a = 1.0 / (1.0 + np.exp(-x))
    a = a / a.sum()
    candidates.append(('sigmoid', a))
    
    # Two-hump (right-skewed)
    x = np.linspace(0, 1, n)
    a = x * (1 - x) * 4  # parabola peaking at 0.5
    a2 = np.exp(-((x - 0.7) ** 2) / 0.05)
    a = 0.3 * a + 0.7 * a2
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    candidates.append(('two_hump', a))
    
    # Random
    a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum()
    candidates.append(('random', a))
    
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-30)
    f_interp = f_interp / f_interp.sum()
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):

    Multi-phase optimization combining parametric initialization,
    multiplicative subgradient descent, soft-max Adam, and local refinement
    to minimize the peak autocorrelation ratio.
    """

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    # === Phase 1: Parametric initialization (10% time) ===
    print("\n=== Phase 1: Parametric Initialization ===")

        n_default = len(parent)

    families = parametric_families(n_default, rng)

            a_check = a_init / a_init.sum()
            try:
                real_score = float(evaluate_sequence(a_check.tolist()))
                if np.isfinite(real_score) and real_score < best_score:
                    best_score = real_score
                    best_seq = a_check.copy()

    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    t_p3 = min(450.0, rem * 0.25)
    
    # Multiple Adam runs with different temperatures
    adam_configs = [
        {'T_init': 0.03, 'T_final': 0.001, 'lr': 0.008, 'schedule': 'exp'},
        {'T_init': 0.05, 'T_final': 0.002, 'lr': 0.01, 'schedule': 'hold_anneal'},
        {'T_init': 0.02, 'T_final': 0.0005, 'lr': 0.005, 'schedule': 'exp'},
    ]
    
    time_per_adam = t_p3 / len(adam_configs)
    for ai, acfg in enumerate(adam_configs):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(time_per_adam, rem * 0.9)
        if t_alloc < 20:
            continue
        
        print(f"  Adam {ai+1}/{len(adam_configs)}: T={acfg['T_init']}->{acfg['T_final']}, budget={t_alloc:.0f}s")
        
        a_res, score_res = adam_logspace(
            best_seq, t_alloc,
            T_init=acfg['T_init'],
            T_final=acfg['T_final'],
            lr=acfg['lr'],
            schedule=acfg['schedule'],
            verbose=False
        )
        best_seq, best_score = evaluate_and_track(a_res, f"adam_{ai}", best_seq, best_score)
    
    print(f"After Phase 3: best={best_score:.6f}")
    
    # === Phase 4: Local refinement (15% time) ===
    print(f"\n=== Phase 4: Local Refinement ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    t_p4 = min(270.0, rem * 0.15)
    
    # Pair descent
    t_sub = t_p4 * 0.5
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_sub = min(t_sub, rem * 0.8)
        a_res, _ = pair_descent(best_seq, t_sub, n_trials=5000)
        best_seq, best_score = evaluate_and_track(a_res, "pair_descent", best_seq, best_score)
        print(f"  After pair descent: {best_score:.6f}")
    
    # Coordinate descent
    rem = deadline - time.monotonic() - 80
    if rem > 15:
        t_sub = min(t_p4 * 0.3, rem * 0.7)
        a_res, _ = coordinate_descent(best_seq, t_sub, n_sweeps=30)
        best_seq, best_score = evaluate_and_track(a_res, "coord_desc", best_seq, best_score)
        print(f"  After coord descent: {best_score:.6f}")
    
    # Another round of pair descent
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t_sub = min(60.0, rem * 0.7)
        a_res, _ = pair_descent(best_seq, t_sub, n_trials=2000)
        best_seq, best_score = evaluate_and_track(a_res, "pair_descent_2", best_seq, best_score)
        print(f"  After pair descent 2: {best_score:.6f}")
    
    # === Phase 5: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    t_p5 = min(180.0, rem * 0.10)
    
    alt_lengths = []
    for factor in [1.25, 1.5, 0.75, 2.0, 1.333]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            if alt_n not in alt_lengths:
                alt_lengths.append(alt_n)
    
    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic() - 40
        if rem < 30:
            break
        t_sub = min(t_p5 * 0.3, rem * 0.7)
        if t_sub < 15:
            continue
        
        # Interpolate to new length
        init_alt = interpolate_sequence(best_seq, alt_n)
        
        # Quick subgrad at new length
        a_alt, _ = subgrad_vectorized(init_alt, t_sub, eta_init=0.3, eta_decay=0.9998, top_k=1)
        best_seq, best_score = evaluate_and_track(a_alt, f"alt_n{alt_n}", best_seq, best_score)
        print(f"  Alt n={alt_n}: {best_score:.6f}")
    
    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic() - 10
    
    if rem > 30:
        # Final subgrad with very small eta
        t_sub = min(rem * 0.4, 60.0)
        a_fin, _ = subgrad_vectorized(best_seq, t_sub, eta_init=0.1, eta_decay=0.9999, top_k=2)
        best_seq, best_score = evaluate_and_track(a_fin, "final_subgrad", best_seq, best_score)
    
    rem = deadline - time.monotonic() - 10
    if rem > 20:
        t_sub = min(rem * 0.5, 40.0)
        a_fin, _ = pair_descent(best_seq, t_sub, n_trials=2000)
        best_seq, best_score = evaluate_and_track(a_fin, "final_pair", best_seq, best_score)
    
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_sub = min(rem * 0.8, 20.0)
        a_fin, _ = coordinate_descent(best_seq, t_sub, n_sweeps=10)
        best_seq, best_score = evaluate_and_track(a_fin, "final_coord", best_seq, best_score)
    
    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")
    
    return [float(x) for x in best_seq]
