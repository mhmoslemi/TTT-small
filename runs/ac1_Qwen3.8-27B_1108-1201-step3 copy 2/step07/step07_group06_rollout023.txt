from scipy.fft import rfft, irfft

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Compute c[i] = sum_m a[m] * b[i+m] for i=0..n-1"""
    n = len(a)

    N = 1 << (n + len(b) - 1).bit_length()
    A = rfft(a, N)
    B = rfft(b, N)
    corr = irfft(np.conj(A) * B, N)
    return corr[:n]

    b = fast_convolve(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage optimization:
(1) Sum-of-squares gradient descent for smooth coarse optimization,
(2) Multi-peak soft-max gradient descent for fine optimization,
(3) Peak-aware coordinate descent with swap moves,
(4) Multi-length exploration and final polishing.
Key novelty: alternating between SOS and max objectives to escape local minima,
plus structured initializations (U-shaped, spectral, etc.).

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(a, w):
    """Compute c[i] = sum_m a[m] * w[i+m] for i=0..n-1.
    a has length n, w has length 2n-1.
    """
    n = len(a)
    N = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    corr = irfft(np.conj(A) * W, N)
    return corr[:n]


def sos_obj_gradient(a):
    """Gradient of sum(b[k]^2) w.r.t. a, where b = conv(a,a)."""
    b = fast_convolve(a)
    c = cross_corr(a, b)
    return 4.0 * c


def softmax_obj_gradient(a, T, n):
    """Gradient of 2*n*soft_max(b, T) w.r.t. a."""
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    w = exp_b / exp_b.sum()
    c = cross_corr(a, w)
    return 4.0 * n * c


def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def get_score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def adam_sos(a0, time_limit, lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8,
             verbose=False, verbose_interval=5000):
    """Adam optimization of sum(b^2) in log-space."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = get_score_norm(a)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:

        S = np.dot(a_curr, grad)
        grad_z = a_curr * (grad - S)

        v = beta2 * v + (1 - beta2) * (grad_z * grad_z)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

def adam_softmax(a0, time_limit, T_init=0.05, T_final=0.001, lr=0.01,
                 beta1=0.9, beta2=0.999, eps=1e-8, schedule='exp',
                 verbose=False, verbose_interval=5000):
    """Adam optimization of softmax-smoothed max in log-space."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = get_score_norm(a)
    no_improve = 0
    lr_current = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)
        grad = softmax_obj_gradient(a_curr, T, n)
        S = np.dot(a_curr, grad)
        grad_z = a_curr * (grad - S)

        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z * grad_z)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr_current * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        score = get_score_norm(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 4000:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.003
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr_current *= 0.75

        lr_current = max(0.0005, lr_current * (1.0 - 0.00005))

        if verbose and t % verbose_interval == 0:
            print(f"    SoftMax t={t} T={T:.5f} lr={lr_current:.5f} score={best_score:.6f}")

    return best_a, best_score


def alternating_adam(a0, time_limit, T_init=0.04, T_final=0.001, lr=0.01,
                     sos_lr=0.02, n_cycles=5, verbose=False):
    """Alternate between SOS and soft-max optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score_norm(a)
    t0 = time.monotonic()
    time_per_cycle = time_limit / n_cycles

    for cycle in range(n_cycles):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        half = min(time_per_cycle / 2, rem / 2)

        # SOS phase
        a_sos, score_sos = adam_sos(a, half, lr=sos_lr)
        if score_sos < best_score:
            best_score = score_sos
            best_a = a_sos.copy()
        a = a_sos

        rem = time.monotonic() and (time_limit - (time.monotonic() - t0))
        if rem < 5:
            break
        half2 = min(time_per_cycle / 2, rem / 2)

        # Soft-max phase
        T_curr = T_init * (T_final / T_init) ** (cycle / max(1, n_cycles - 1))
        a_sm, score_sm = adam_softmax(a, half2, T_init=T_curr, T_final=T_curr * 0.5,
                                       lr=lr, schedule='exp')
        if score_sm < best_score:
            best_score = score_sm
            best_a = a_sm.copy()
        a = a_sm

        if verbose:
            print(f"    Cycle {cycle+1}/{n_cycles}: best={best_score:.6f}")

    return best_a, best_score


def peak_aware_coord_descent(a0, time_limit, n_sweeps=50):
    """Coordinate descent that prioritizes elements contributing to peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top peak (interior)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))

        # Find contributing elements
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Process in order of contribution (highest first)
        order = idx[np.argsort(contribs)[::-1]]
        # Interleave with random
        other_idx = np.setdiff1d(np.arange(n), order)
        order_ext = np.concatenate([order, rng.permutation(other_idx)])

        improved = False
        for i in order_ext:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_factor = 1.0
            best_local_score = best_score

            for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = get_score_norm(a_trial)
                if score < best_local_score:
                    best_local_score = score
                    best_factor = factor

            if best_factor != 1.0:
                a[i] = a[i] * best_factor
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    improved = True

        if not improved and sweep > 5:
            # Perturb to escape
            a = best_a + np.random.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def swap_local_search(a0, time_limit, max_iter=5000):
    """Try swap moves: move mass from peak-contributing to non-peak positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))

        # Find top contributor
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_i = idx[np.argmax(contribs)]
        top_j = peak_k - top_i

        # Try reducing top_i and/or top_j
        for target in [top_i, top_j]:
            if a[target] < 1e-15:
                continue
            # Find a good target to receive mass (low contribution to all peaks)
            # Heuristic: positions far from peak_k
            distances = np.abs(np.arange(n) - peak_k)
            candidates = np.argsort(distances)[::-1][:20]
            candidates = candidates[a[candidates] < a.mean() * 2]

            for dest in candidates[:5]:
                if dest == target:
                    continue
                for eps_frac in [0.05, 0.1, 0.15, 0.2]:
                    amount = a[target] * eps_frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[target] -= amount
                    a_trial[dest] += amount
                    if a_trial[target] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = get_score_norm(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_initializations(parent, n, rng):
    """Generate diverse initializations."""
    candidates = []

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            for scale in [0.02, 0.05, 0.1]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append((f'pert_{scale}', pert))
            # Structured perturbations
            for freq in [2, 3, 5, 7, 11]:
                t = np.linspace(0, 2 * np.pi * freq, n)
                pert = p + 0.03 * np.cos(t)
                pert = np.maximum(pert, 0)
                sp = pert.sum()
                if sp > 1e-10:
                    pert = pert / sp
                    candidates.append((f'cos_{freq}', pert))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('parent_interp', interp))

    # U-shaped (high at edges)
    x = np.linspace(0, 1, n)
    u_shape = (2 * x - 1) ** 2 + 0.1
    u_shape = u_shape / u_shape.sum()
    candidates.append(('u_shape', u_shape))

    # W-shaped (bimodal)
    mid = n // 2
    quart = n // 4
    w_shape = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01) + 0.05
    w_shape = w_shape / w_shape.sum()
    candidates.append(('w_shape', w_shape))

    # Triangular
    tri = np.minimum(x, 1 - x) * 2
    tri = tri + 0.1
    tri = tri / tri.sum()
    candidates.append(('tri', tri))

    # Uniform
    uni = np.ones(n) / n
    candidates.append(('uniform', uni))

    # Gaussian
    gau = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.01
    gau = gau / gau.sum()
    candidates.append(('gauss', gau))

    # Cosine window
    cos_w = np.cos(np.pi * (x - 0.5)) ** 2 + 0.05
    cos_w = cos_w / cos_w.sum()
    candidates.append(('cos_win', cos_w))

    # Spectral: try to make |DFT| flat
    # Random phase in frequency domain
    N_fft = 1 << (2 * n - 2).bit_length()
    A = np.ones(N_fft, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, N_fft)
    A = np.abs(A) * np.exp(1j * phases)
    A = A + np.conj(A[::-1])  # Make real
    a_spec = np.real(ifft(A * A, N_fft))[:n]  # Hmm this isn't quite right
    # Better: pick random phases, take IFFT to get a real sequence
    A_real = np.zeros(N_fft, dtype=complex)
    A_real[0] = 1
    freqs = np.arange(N_fft)
    mask = (freqs > 0) & (freqs < N_fft // 2)
    A_real[mask] = np.exp(1j * rng.uniform(0, 2 * np.pi, mask.sum()))
    A_real[N_fft - freqs[mask]] = np.conj(A_real[mask])
    a_spec = np.real(irfft(A_real, N_fft)[:n])
    a_spec = np.maximum(a_spec, 0.01)
    a_spec = a_spec / a_spec.sum()
    candidates.append(('spectral', a_spec))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    rand = rand / rand.sum()
    candidates.append(('random', rand))

    return candidates


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining SOS gradient descent, multi-peak
    soft-max Adam, peak-aware coordinate descent, swap local search,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best score: {best_score:.6f}, n={n_parent}")

    # === Phase 1: Multi-start with alternating SOS + SoftMax (40% time) ===
    print(f"\n=== Phase 1: Multi-Start Alternating (n={n_parent}) ===")
    t_p1 = (deadline - time.monotonic()) * 0.40
    t_p1_start = time.monotonic()

    inits = make_initializations(parent, n_parent, rng)
    # Score them
    inits_scored = []
    for label, init_a in inits:
        s = get_score_norm(init_a)
        inits_scored.append((s, label, init_a))
    inits_scored.sort(key=lambda x: x[0])

    n_starts = min(6, len(inits_scored))
    time_per_start = t_p1 / n_starts

    for idx in range(n_starts):
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        t_alloc = min(time_per_start, rem * 0.9)
        if t_alloc < 30:
            continue

        s_init, label, init_a = inits_scored[idx]
        print(f"  Start {idx+1}/{n_starts}: {label} (init_score={s_init:.6f}), budget={t_alloc:.0f}s")

        # Alternating SOS + SoftMax
        a_res, score_res = alternating_adam(
            init_a, t_alloc, T_init=0.05, T_final=0.002,
            lr=0.01, sos_lr=0.02, n_cycles=4
        )
        best_seq, best_score = evaluate_candidate(a_res, best_seq, best_score, f"alt_{label}")

        # Quick SOS polish
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            a_sos, _ = adam_sos(a_res, min(30, rem * 0.3), lr=0.015)
            best_seq, best_score = evaluate_candidate(a_sos, best_seq, best_score, f"alt_{label}_sos")

    print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Focused refinement (25% time) ===
    print(f"\n=== Phase 2: Focused Refinement ===")
    n_best = len(best_seq)
    t_p2 = (deadline - time.monotonic()) * 0.25

    # Long alternating run
    t_sub = t_p2 * 0.4
    a_res, _ = alternating_adam(
        best_seq, t_sub, T_init=0.03, T_final=0.001,
        lr=0.008, sos_lr=0.015, n_cycles=6, verbose=True
    )
    best_seq, best_score = evaluate_candidate(a_res, best_seq, best_score, "alt_focused")
    print(f"  After alternating: {best_score:.6f}")

    # Soft-max focused
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_sub = min(rem * 0.4, 120.0)
        a_sm, _ = adam_softmax(
            best_seq, t_sub, T_init=0.02, T_final=0.001,
            lr=0.006, schedule='hold_anneal'
        )
        best_seq, best_score = evaluate_candidate(a_sm, best_seq, best_score, "softmax_focused")
        print(f"  After softmax: {best_score:.6f}")

    # SOS focused
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_sub = min(rem * 0.4, 60.0)
        a_sos, _ = adam_sos(best_seq, t_sub, lr=0.01)
        best_seq, best_score = evaluate_candidate(a_sos, best_seq, best_score, "sos_focused")
        print(f"  After SOS: {best_score:.6f}")

    # === Phase 3: Local search (20% time) ===
    print(f"\n=== Phase 3: Local Search ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 150

    if rem > 30:
        t_sub = min(rem * 0.35, 120.0)
        a_cd, _ = peak_aware_coord_descent(best_seq, t_sub, n_sweeps=30)
        best_seq, best_score = evaluate_candidate(a_cd, best_seq, best_score, "coord_desc")
        print(f"  After coord descent: {best_score:.6f}")

    rem = deadline - time.monotonic() - 120
    if rem > 20:
        t_sub = min(rem * 0.4, 80.0)
        a_swap, _ = swap_local_search(best_seq, t_sub, max_iter=3000)
        best_seq, best_score = evaluate_candidate(a_swap, best_seq, best_score, "swap")
        print(f"  After swap: {best_score:.6f}")

    # Another round of alternating after local search
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t_sub = min(rem * 0.5, 80.0)
        a_alt, _ = alternating_adam(
            best_seq, t_sub, T_init=0.015, T_final=0.0005,
            lr=0.005, sos_lr=0.01, n_cycles=3
        )
        best_seq, best_score = evaluate_candidate(a_alt, best_seq, best_score, "alt_round2")
        print(f"  After alt round 2: {best_score:.6f}")

    # === Phase 4: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 80

    if rem > 60:
        alt_lengths = []
        for factor in [1.25, 1.5, 0.75, 2.0, 1.1, 0.6]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic() - 40
            if rem < 40:
                break
            t_sub = min(rem * 0.35, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = alternating_adam(
                init_alt, t_sub, T_init=0.04, T_final=0.002,
                lr=0.008, sos_lr=0.02, n_cycles=3
            )
            best_seq, best_score = evaluate_candidate(a_alt, best_seq, best_score, f"alt_n{alt_n}")
            print(f"  Alt n={alt_n}: {best_score:.6f}")

    # === Phase 5: Final polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    rem = deadline - time.monotonic() - 10

    if rem > 20:
        t_sub = min(rem * 0.3, 50.0)
        a_fin, _ = adam_softmax(
            best_seq, t_sub, T_init=0.01, T_final=0.0003,
            lr=0.003, schedule='exp'
        )
        best_seq, best_score = evaluate_candidate(a_fin, best_seq, best_score, "final_sm")

    rem = deadline - time.monotonic() - 8
    if rem > 15:
        t_sub = min(rem * 0.4, 30.0)
        a_cd2, _ = peak_aware_coord_descent(best_seq, t_sub, n_sweeps=20)
        best_seq, best_score = evaluate_candidate(a_cd2, best_seq, best_score, "final_cd")

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_sub = min(rem * 0.5, 20.0)
        a_sos2, _ = adam_sos(best_seq, t_sub, lr=0.005)
        best_seq, best_score = evaluate_candidate(a_sos2, best_seq, best_score, "final_sos")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
