                c = fast_convolve(a_norm)
                return 2.0 * n * c.max()
                # Score with sum=1

            # Golden section search
            # Search range: [0, upper]
            upper = max(3.0 * old_t, 0.05 / n)
            if old_t < 1e-10:
                lower = 0.0
                upper = max(0.1 / n, 1e-4)
            else:
                lower = 0.0
                upper = max(2.5 * old_t, old_t + 0.01 / n)

            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_gs = lower
            b_gs = upper
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            fc = score_1d(c_gs)
            fd = score_1d(d_gs)

            for _ in range(25):
                if b_gs - a_gs < 1e-12:
                    break
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = score_1d(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = score_1d(d_gs)

            t_opt = (a_gs + b_gs) / 2.0
            a[i] = t_opt
            s_new = s - old_t + t_opt
            if s_new < 1e-12:
                a[i] = old_t
                continue
            a = a / s_new
            sc = compute_score(a)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
                improved = True
            s = a.sum()

        if verbose and pass_count % 10 == 0:
            print(f"      CD pass {pass_count}: score={best_score:.8f}")

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                # Perturb to escape
                a = best_a + np.random.standard_normal(n) * (best_a * 0.01 + 1e-6)
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
            elif no_improve >= 8:
                break
        else:
            no_improve = 0

    return best_a


def fictitious_play(a_init, time_limit=300.0, solver=None, verbose=False):
    """Run fictitious play with LP best responses."""
    n = len(a_init)
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    iters = 0
    stagnation = 0
    a_avg = a.copy()

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        a_avg = (iters * a_avg + b_new) / (iters + 1)

        # Evaluate candidates
        for cand in [b_new, a_avg, 0.5 * (a + b_new)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-12:
                continue
            cand = cand / sc
            sc_score = compute_score(cand)
            if sc_score < best_score - 1e-12:
                best_score = sc_score
                best_a = cand.copy()
                stagnation = 0
            else:
                stagnation += 1

        a = b_new
        if stagnation > 200:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            a_avg = a.copy()
            stagnation = 0

        if verbose and iters % 20 == 0:
            print(f"      FP iter {iters} ({monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters


def lbgfs_refine(a_init, time_limit=120.0, T=50.0):
    """L-BFGS refinement with log-sum-exp smoothing."""
    n = len(a_init)
    a_safe = np.maximum(a_init, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        zs = z - z.max()
        exp_z = np.exp(zs)
        a = exp_z / exp_z.sum()
        c = fast_convolve(a)
        max_c = c.max()
        scaled = (c - max_c) / T
        exp_c = np.exp(scaled)
        lse = max_c + T * np.log(exp_c.sum() / n)
        return 2.0 * n * lse

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        zs = z - z.max()
        exp_z = np.exp(zs)
        a = exp_z / exp_z.sum()
        c = fast_convolve(a)
        max_c = c.max()
        scaled = (c - max_c) / T
        exp_c = np.exp(scaled)
        w = exp_c / exp_c.sum()

        # Gradient of weighted sum of conv values wrt a
        # d/d(a[i]) of sum_k w[k] * conv[k]
        # conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/d(a[i]) = a[k-i] + (a[i-k] if valid) -- but conv is symmetric
        # Actually for auto-conv: d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[k-i] (same)
        # = 2*a[k-i] if i != k-i, else a[i] (just once for i= k-i)
        # More precisely: conv[k] = sum_j a[j] a[k-j], d/d(a[i]) = a[k-i] + a[k-i] = 2a[k-i] if k!=2i, else a[i]
        # Wait no: if we differentiate sum_j a[j]*a[k-j] wrt a[i]:
        # term j=i gives a[i]*a[k-i] -> derivative is a[k-i]
        # term j=k-i gives a[k-i]*a[i] -> derivative is a[k-i]
        # So total = 2*a[k-i] if i != k-i, or a[i] if i = k-i (i.e. k=2i)
        
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # grad_a[idx] += 2 * w[k] * a[k - idx]
            # But we need to handle the k=2i case (where the factor should be 1 not 2)
            # For simplicity, use factor 2 everywhere (slight overestimate at the one point)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': 50000, 'disp': False}
        )
        if not stopped[0] and res.x is not None:
            zf = res.x - res.x.max()
            a_out = np.exp(zf)
            a_out = a_out / a_out.sum()
        else:
            a_out = a_init
    except Exception:
        a_out = a_init

    return a_out


def peak_shave(a_init, time_limit=60.0):
    """Aggressive peak shaving: reduce the largest convolution peaks."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = monotonic()
    failed = 0

    while monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        peak_val = c[peak_k]
        threshold = best_score / (2.0 * n)
        if peak_val <= threshold * 1.001:
            break

        # Find contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total = contribs.sum()
        if total < 1e-20:
            break

        improved = False
        # Try reducing top contributors
        top = np.argsort(contribs)[::-1][:10]
        for ci in top:
            if improved:
                break
            i = int(idx[ci])
            j = int(peak_k - i)
            old_i = a[i]
            if old_i < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                a_trial = a.copy()
                a_trial[i] = old_i * factor
                diff = old_i - old_i * factor
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    break
                a_trial[mask] = others + diff * others / os
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try small random perturbation
            found = False
            for _ in range(10):
                a_trial = a + np.random.standard_normal(n) * 0.001 * a
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial = a_trial / ts
                sc = compute_score(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    found = True
                    break
            if not found:
                failed += 1
                if failed > 30:
                    break
        else:
            failed = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"    NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for auto-convolution minimization.
    Uses coordinate descent, LP fictitious play, and L-BFGS.
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform n=768: score={best_score:.8f}")

    n_main = len(best_seq)

    # === Phase 1: Coordinate Descent on parent ===
    print(f"\n=== Phase 1: Coordinate Descent (n={n_main}) ===")
    rem = deadline - monotonic()
    t1 = min(rem * 0.2, 250.0)
    print(f"  Budget: {t1:.0f}s")
    cd_result = coordinate_descend(best_seq, time_limit=t1, verbose=True)
    best_seq, best_score = check_update(cd_result, "CD1", best_seq, best_score)

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: Fictitious Play (n={n_main}) ===")
    rem = deadline - monotonic()
    t2 = min(rem * 0.25, 350.0)
    if rem > 50:
        print(f"  Budget: {t2:.0f}s")
        fp_result, fp_score, fp_iters = fictitious_play(best_seq, time_limit=t2, solver=solver, verbose=True)
        best_seq, best_score = check_update(fp_result, f"FP_{fp_iters}iters", best_seq, best_score)

    # === Phase 3: Coordinate Descent again ===
    print(f"\n=== Phase 3: CD Round 2 (n={n_main}) ===")
    rem = deadline - monotonic()
    t3 = min(rem * 0.2, 200.0)
    if rem > 30:
        print(f"  Budget: {t3:.0f}s")
        cd_result2 = coordinate_descend(best_seq, time_limit=t3, verbose=True)
        best_seq, best_score = check_update(cd_result2, "CD2", best_seq, best_score)

    # === Phase 4: L-BFGS Polish ===
    print(f"\n=== Phase 4: L-BFGS Polish (n={n_main}) ===")
    rem = deadline - monotonic()
    if rem > 20:
        for T_val in [30.0, 80.0, 150.0]:
            rem = deadline - monotonic()
            if rem < 15:
                break
            t4 = min(rem * 0.3, 80.0)
            print(f"  L-BFGS T={T_val:.0f}, budget={t4:.0f}s")
            lb_result = lbgfs_refine(best_seq, time_limit=t4, T=T_val)
            best_seq, best_score = check_update(lb_result, f"LBM_T{T_val:.0f}", best_seq, best_score)

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving ===")
    rem = deadline - monotonic()
    if rem > 15:
        t5 = min(rem * 0.2, 80.0)
        ps_result, ps_score = peak_shave(best_seq, time_limit=t5)
        best_seq, best_score = check_update(ps_result, "PeakShave", best_seq, best_score)

    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Length Exploration ===")
    rem = deadline - monotonic()
    alt_ns = []
    for f in [0.8, 1.25, 1.5, 0.67]:
        alt_n = int(n_main * f)
        if 256 <= alt_n <= 4096 and alt_n != n_main and alt_n not in alt_ns:
            alt_ns.append(alt_n)

    for alt_n in alt_ns[:3]:
        rem = deadline - monotonic()
        if rem < 40:
            break
        t6 = min(rem * 0.35, 150.0)
        print(f"  Exploring n={alt_n}, budget={t6:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Quick CD on alt length
        t_cd = t6 * 0.4
        t_fp = t6 * 0.5
        t_ps = t6 * 0.1
        cd_alt = coordinate_descend(init_alt, time_limit=t_cd)
        best_seq, best_score = check_update(cd_alt, f"CD_alt_n{alt_n}", best_seq, best_score)

        if rem > 30 and best_seq is not None and len(best_seq) != alt_n:
            # Re-interpolate to current best length for FP
            pass
        # FP on alt best
        rem = deadline - monotonic()
        if rem > 20:
            cur_n = len(best_seq)
            fp_alt, _, _ = fictitious_play(best_seq, time_limit=min(rem * 0.5, t_fp), solver=solver)
            best_seq, best_score = check_update(fp_alt, f"FP_alt_n{cur_n}", best_seq, best_score)

    # === Phase 7: Final CD + Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    rem = deadline - monotonic()
    if rem > 10:
        t7 = min(rem * 0.4, 100.0)
        cd_final = coordinate_descend(best_seq, time_limit=t7)
        best_seq, best_score = check_update(cd_final, "CD_final", best_seq, best_score)

    rem = deadline - monotonic()
    if rem > 10:
        t8 = min(rem * 0.4, 60.0)
        lb_final = lbgfs_refine(best_seq, time_limit=t8, T=50.0)
        best_seq, best_score = check_update(lb_final, "LB_final", best_seq, best_score)

    rem = deadline - monotonic()
    if rem > 8:
        t9 = min(rem * 0.5, 30.0)
        ps_final, _ = peak_shave(best_seq, time_limit=t9)
        best_seq, best_score = check_update(ps_final, "PS_final", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
