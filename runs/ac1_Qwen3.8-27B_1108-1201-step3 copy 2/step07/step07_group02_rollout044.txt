def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_ab(a, b):
    """Convolve a and b (cross-convolution)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    n = len(a)
    T = np.zeros((2 * n - 1, n))
    # Column j of T has a_j in rows j..j+n-1
    for j in range(n):
        T[j:j+n, j] = a
    return T

    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n

    return _cached_solver

def solve_lp_response(a_fixed, solver=None):
    """Solve: min M s.t. T_a @ b <= M, sum(b)=1, b>=0. Returns (b, M, duals)."""
    if solver is None:
        solver = get_solver()
    n = len(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    T = build_T(a_fixed)
    constraints = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    
    # Get dual values for the inequality constraints
    duals = None
    try:
        # The dual of T@b <= M constraints
        dual_constraint = constraints[0]
        duals = np.asarray(dual_constraint.value, dtype=np.float64)
        if duals is not None:
            duals = np.maximum(duals, 0)
    except Exception:
        duals = None
    
    return b_val, float(M.value), duals

def nesterov_lp_alternate(a0, time_limit, solver=None, verbose=True):
    """Nesterov-accelerated LP alternating."""
    if solver is None:
        solver = get_solver()
    n = len(a0)
    a = normalize(a0.copy())
    y = a.copy()  # Nesterov virtual iterate
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        b_new, M_val, duals = solve_lp_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov step
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        # Weighted average for actual iterate
        # x_{t+1} = (1/(t+2)) * sum_{s=1}^{t+1} x_s  (for Nesterov)
        # Equivalent: x_{t+1} = ((t+1)*x_t + x_{t+1}_new) / (t+2)
        a = ((it + 1) * a + b_new) / (it + 2)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if verbose and (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.6f}")
    
    return best_a, best_score

def subgradient_logspace(a0, time_limit, lr_init=0.05, soft_beta=50.0):
    """Importance-weighted subgradient descent in log-space with softmax smoothing."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Softmax weights on near-max entries
        Tb = soft_beta * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w[k] * a[k - idx]
        
        # Mirror descent update in log-space
        z = z - lr * grad
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Revert
            z = z + lr * grad
        
        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.9
            no_improve = 0
        lr = max(lr, 1e-8)
        
        # Anneal soft_beta (increase for sharper focus on peaks)
        soft_beta *= 1.001
        
        if step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f}, lr={lr:.6f}, beta={soft_beta:.1f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, n_cycles=50):
    """Coordinate descent with multi-point trial."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for cycle in range(n_cycles):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Identify peak and its contributing coordinates
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        peak_coords = list(range(js, je + 1))
        
        # Shuffle for diversity, but prioritize peak contributors
        if cycle % 2 == 0:
            order = peak_coords + [i for i in range(n) if i not in peak_coords]
        else:
            order = list(range(n))
            np.random.shuffle(order)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            # Try scaling a[i] by various factors
            base = a.copy()
            best_factor = 1.0
            best_local = best_score
            
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                t_new = a[i] * factor
                if t_new < 0 or t_new > 1:
                    continue
                a_trial = a.copy()
                a_trial[i] = t_new
                # Renormalize others proportionally
                others_sum = 1.0 - t_new
                if others_sum < 1e-15:
                    continue
                a_trial[range(n) if False else np.delete(np.arange(n), i)] *= others_sum / (1.0 - a[i]) if (1.0 - a[i]) > 1e-15 else 1.0
                # Actually simpler: just scale
                a_trial = a.copy()
                a_trial[i] = t_new
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_local:
                    best_local = sc
                    best_factor = factor
            
            if best_factor != 1.0:
                a[i] *= best_factor
                a = normalize(a)
                sc = 2.0 * n * fast_convolve(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        if cycle % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc cycle={cycle} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def peak_shave(a0, time_limit):
    """Aggressively reduce the tallest peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the tallest peak (excluding edges which are naturally small)
        b_inner = b.copy()
        if n > 10:
            b_inner[:5] = -1
            b_inner[-5:] = -1
        peak_k = int(np.argmax(b_inner))
        
        # Contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_i = idx[np.argmax(contribs)]
        
        # Try reducing top_i
        improved = False
        for frac in [0.9, 0.8, 0.7, 0.6]:
            a_trial = a.copy()
            diff = a[top_i] * (1 - frac)
            a_trial[top_i] *= frac
            # Redistribute to random other coordinates
            others = np.delete(np.arange(n), top_i)
            a_trial[others] += diff * a_trial[others] / a_trial[others].sum()
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try reducing two coordinates
            top2 = idx[np.argsort(contribs)[::-1]][:2]
            if len(top2) >= 2:
                for f1 in [0.9, 0.8]:
                    for f2 in [0.9, 0.8]:
                        a_trial = a.copy()
                        diff1 = a_trial[top2[0]] * (1 - f1)
                        diff2 = a_trial[top2[1]] * (1 - f2)
                        a_trial[top2[0]] *= f1
                        a_trial[top2[1]] *= f2
                        total_diff = diff1 + diff2
                        others = np.delete(np.arange(n), top2)
                        a_trial[others] += total_diff * a_trial[others] / a_trial[others].sum()
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            break
    
    return best_a, best_score

def lbgfs_smoothed(a0, time_limit, T_start=100.0, T_end=1.0):
    """L-BFGS with softmax-smoothed max objective in log-space."""
    from scipy.optimize import minimize
    n = len(a0)
    a_init = normalize(a0.copy())
    z = np.log(np.maximum(a_init, 1e-300))
    t0 = time.monotonic()
    
    def obj_and_grad(z_val, T):
        a = np.exp(z_val - z_val.max())
        a = a / a.sum()
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T) * log_sum_exp
        
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return obj, grad_z
    
    # Multi-stage temperature
    stages = 4
    temps = np.geomspace(T_start, T_end, stages)
    z_current = z.copy()
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        
        def fun(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            o, _ = obj_and_grad(z_val, T)
            return o
        
        def jac(z_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            _, g = obj_and_grad(z_val, T)
            return g
        
        try:
            res = minimize(fun, z_current, jac=jac, method='L-BFGS-B',
                           options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-8, 'disp': False})
            z_current = res.x
        except Exception:
            pass
    
    a_out = np.exp(z_current - z_current.max())
    a_out = a_out / a_out.sum()
    return a_out

def interpolate_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a_norm)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
