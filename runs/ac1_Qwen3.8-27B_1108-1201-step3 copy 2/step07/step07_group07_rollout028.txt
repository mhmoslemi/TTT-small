"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed subgradient descent with adaptive temperature
2. Improved LP refinement accounting for cross and self convolution terms
3. Golden-section coordinate descent

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for sequence (assumed sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def compute_subgradient(a, w):
    """
    Compute subgradient g[i] = sum_j a[j] * w[i+j mod appropriate range]
    where w is defined on the convolution indices [0, 2n-2].
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return g


def smoothed_subgradient_descent(a0, time_limit, seed=42, 
                                  lr_init=0.003, momentum=0.92,
                                  temp_init=50.0, temp_decay=0.999,
                                  n_top=None, restart_interval=2000,
                                  kick_size=0.005):
    """
    Subgradient descent using log-sum-exp smoothing of the max.
    This gives a smooth gradient that considers all peaks (weighted by exp),
    not just the top peaks.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    temp = temp_init
    no_improve = 0
    it_count = 0

    if n_top is None:
        n_top = max(10, n // 10)

    while time.monotonic() - t0 < time_limit:
        # Compute convolution
        b = fast_convolve(a)
        b_max = b.max()
        
        # Log-sum-exp weights (smoothed max)
        # w[k] = exp(temp * (b[k] - b_max)) / sum_j exp(temp * (b[j] - b_max))
        # This is more efficient than computing full log-sum-exp
        shifted = temp * (b - b_max)
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = weights / w_sum
        
        # Compute subgradient
        g = compute_subgradient(a, w)
        
        # Nesterov momentum step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        # Evaluate
        obj_new = eval_max_b(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it_count += 1
        
        # Adaptive: decay temperature (sharpen the max)
        temp *= temp_decay
        temp = max(temp, 5.0)
        
        # Adaptive restart
        if no_improve >= restart_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.4, lr_init * 0.03)
            no_improve = 0
            kick_size *= 0.6
            temp = max(temp * 2, temp_init * 0.5)
        elif no_improve >= restart_interval // 2:
            cur_lr *= 0.93

    return best_a, best_obj


def lp_refine_improved(f, max_iters=50, time_limit=60.0):
    """
    Improved LP refinement:
    - Solves max sum(g) s.t. (f*g)[k] <= M, g >= 0
    - Also computes (g*g) to check the blended result more carefully
    - Uses adaptive M based on current peak distribution
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_f = fast_convolve(a)
        M = b_f.max()

        # Build LP constraint matrix: (f*g)[k] <= M for all k
        num_constraints = 2 * n - 1
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, k - j] = a[j]

        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)

        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'maxiter': 200})
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum
            b_g = fast_convolve(g_norm)
            b_g_max = b_g.max()
            
            # Line search: find best t in (0,1] for blend
            def blend_obj(t):
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    return best_obj + 1
                blend = blend / s
                return eval_max_b(blend)
            
            # Try a grid first
            t_candidates = np.linspace(0.01, 1.0, 30)
            best_t = 0.0
            best_blend_obj = best_obj
            for t in t_candidates:
                obj = blend_obj(t)
                if obj < best_blend_obj:
                    best_blend_obj = obj
                    best_t = t
            
            # Refine with golden section if improvement found
            if best_t > 0.01 and best_t < 0.99:
                from scipy.optimize import minimize_scalar
                res_gs = minimize_scalar(blend_obj, bounds=(max(0, best_t-0.1), min(1, best_t+0.1)),
                                         method='bounded', options={'xatol': 1e-4})
                if res_gs.success and res_gs.fun < best_blend_obj:
                    best_blend_obj = res_gs.fun
                    best_t = res_gs.x
            
            if best_t > 0.0 and best_blend_obj < best_obj - 1e-14:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_obj = best_blend_obj
                best_a = a.copy()
            else:
                # Try smaller perturbations
                found = False
                for t in [0.005, 0.01, 0.02, 0.05]:
                    obj = blend_obj(t)
                    if obj < best_obj - 1e-14:
                        a = (1.0 - t) * best_a + t * g_norm
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        best_obj = obj
                        best_a = a.copy()
                        found = True
                        break
                if not found:
                    break

        except Exception:
            break

    return best_a, best_obj


def coordinate_descent_golden(a0, time_limit, n_passes_max=50):
    """
    Coordinate descent with golden section search for optimal value of each coordinate.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    improved_global = True
    pass_count = 0
    while improved_global and pass_count < n_passes_max and time.monotonic() - t0 < time_limit:
        improved_global = False
        pass_count += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-16:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section search for optimal a[j]
            def obj_j(x):
                if x < 0 or x >= 1:
                    return best_obj + 1
                scale = (1.0 - x) / denom
                if scale <= 0 or scale > 50:
                    return best_obj + 1
                a_tr = a.copy()
                a_tr[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    return best_obj + 1
                a_tr = a_tr / ss
                return eval_max_b(a_tr)

            # Search in [0, min(2*old_val, 0.9999)]
            lo = 0.0
            hi = min(3.0 * old_val, 0.9999)
            
            try:
                res = minimize_scalar(obj_j, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-6})
                if res.success and res.fun < best_obj - 1e-13:
                    a[j] = res.x
                    scale = (1.0 - res.x) / denom
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    best_obj = res.fun
                    best_a = a.copy()
                    improved_global = True
            except Exception:
                pass

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_skewed_init(n, alpha=2.0):
    """Beta-like distribution, skewed."""
    x = np.linspace(0, 1, n)
    a = x ** alpha * (1 - x) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_two_lobe_init(n, separation=0.3):
    """Two Gaussian lobes."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.3) / 0.1) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.1) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def run_optimization_cycle(a_init, time_limit, n, seed, verbose=False):
    """
    Full optimization cycle: SG -> LP -> SG -> LP -> CD
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    # Phase A: Log-sum-exp subgradient
    t_alloc = min(time_limit * 0.35, time_limit)
    if t_alloc > 5:
        a_sg, obj_sg = smoothed_subgradient_descent(
            a, time_limit=t_alloc, seed=seed,
            lr_init=0.004, momentum=0.92,
            temp_init=80.0, temp_decay=0.9995,
            n_top=max(15, n // 10), restart_interval=1500
        )
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            if verbose:
                print(f"    SG: {best_obj:.10f}")

    # Phase B: LP refinement
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        t_lp = min(rem * 0.4, 60.0)
        a_lp, obj_lp = lp_refine_improved(best_a, max_iters=40, time_limit=t_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    LP: {best_obj:.10f}")

    # Phase C: More subgradient from LP result
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_sg2, obj_sg2 = smoothed_subgradient_descent(
            best_a, time_limit=rem * 0.5, seed=seed + 7777,
            lr_init=0.002, momentum=0.94,
            temp_init=50.0, temp_decay=0.9998,
            n_top=max(20, n // 8), restart_interval=1000
        )
        if obj_sg2 < best_obj:
            best_obj = obj_sg2
            best_a = a_sg2.copy()

    # Phase D: Final LP
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        a_lp2, obj_lp2 = lp_refine_improved(best_a, max_iters=20, time_limit=rem)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed subgradient descent with adaptive temperature
2. LP-based refinement with peak-restricted constraints and golden-section blending
3. Golden-section coordinate descent

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for sequence (assumed sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def compute_subgradient(a, w):
    """Compute subgradient g[i] = sum_j a[j] * w[i+j] via FFT."""
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return g


def smoothed_subgradient_descent(a0, time_limit, seed=42,
                                 lr_init=0.004, momentum=0.92,
                                 temp_init=80.0, temp_decay=0.9995,
                                 restart_interval=1500,
                                 kick_size=0.005):
    """
    Subgradient descent using log-sum-exp smoothing of the max convolution.
    The smoothing provides a dense gradient (all peaks contribute weighted by exp).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    temp = temp_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        # Log-sum-exp weights
        shifted = temp * (b - b_max)
        shifted = np.clip(shifted, -60.0, 0.0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = weights / w_sum

        g = compute_subgradient(a, w)

        # Nesterov momentum
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        temp *= temp_decay
        temp = max(temp, 3.0)

        if no_improve >= restart_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.4, lr_init * 0.03)
            no_improve = 0
            kick_size *= 0.6
            temp = max(temp * 3.0, temp_init * 0.5)
        elif no_improve >= restart_interval // 2:
            cur_lr *= 0.93
        elif no_improve >= restart_interval // 4:
            cur_lr *= 0.97

    return best_a, best_obj


def build_lp_matrix(a, n, n_top_constraints=None):
    """Build the LP constraint matrix for (f*g)[k] <= M."""
    if n_top_constraints is None:
        n_top_constraints = 2 * n - 1
    A = np.zeros((n_top_constraints, n))
    for k in range(n_top_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, k - j] = a[j]
    return A


def lp_refine_improved(f, max_iters=40, time_limit=60.0, n_frac_constraints=1.0):
    """
    LP refinement: maximize sum(g) s.t. (f*g)[k] <= M, g >= 0.
    Uses peak-restricted constraints and golden-section blending.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)

    # Determine number of constraints (use top fraction of peaks)
    n_total_constraints = 2 * n - 1
    n_constraints = max(n, int(n_total_constraints * n_frac_constraints))

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_f = fast_convolve(a)
        M = b_f.max()

        # Select most critical constraints (peaks near max)
        # Use all if n_frac_constraints >= 1, else top peaks
        if n_frac_constraints < 1.0:
            # Sort by peak value, take top n_constraints
            peak_order = np.argsort(b_f)[::-1]
            active_k = peak_order[:n_constraints]
        else:
            active_k = np.arange(n_total_constraints)

        A = np.zeros((len(active_k), n))
        for idx, k in enumerate(active_k):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[idx, k - j] = a[j]

        b_lp = np.full(len(active_k), M)
        c = -np.ones(n)

        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs',

            if not res.success:

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:

                    blend = np.maximum(blend, 0.0)
                    s = blend.sum()
                    if s < 1e-10:
                        return best_obj + 1.0
                    blend = blend / s
                    return eval_max_b(blend)

                    if res_gs.success and res_gs.fun < best_blend_obj:

                        best_t = res_gs.x
                except Exception:
                    pass

            if best_t > 0.0 and best_blend_obj < best_obj - 1e-14:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)

                        best_obj = obj

                        found = True

def coordinate_descent_golden(a0, time_limit, n_passes_max=40):
    """
    Coordinate descent with golden-section (bounded) search per coordinate.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    improved_global = True
    pass_count = 0
    while improved_global and pass_count < n_passes_max and time.monotonic() - t0 < time_limit:
        improved_global = False
        pass_count += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-16:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            def coord_obj(x):
                if x < 0.0 or x > 0.9999:
                    return best_obj + 1.0
                scale = (1.0 - x) / denom
                if scale <= 0.0 or scale > 100.0:
                    return best_obj + 1.0
                a_tr = a.copy()
                a_tr[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    return best_obj + 1.0
                a_tr = a_tr / ss
                return eval_max_b(a_tr)

            lo = 0.0
            hi = min(3.0 * old_val, 0.9999)

            try:
                res = minimize_scalar(coord_obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-7})
                if res.success and res.fun < best_obj - 1e-13:
                    new_val = res.x
                    scale = (1.0 - new_val) / denom
                    a[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    best_obj = res.fun
                    best_a = a.copy()
                    improved_global = True
            except Exception:
                pass

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_skewed_init(n, alpha=2.0):
    x = np.linspace(0, 1, n)
    a = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_two_lobe_init(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.3) / 0.08) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.08) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_waterfall_init(n, decay=3.0):
    """Exponentially decaying weights."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-decay * x / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def run_optimization_cycle(a_init, time_limit, n, seed, verbose=False):
    """
    Full optimization cycle: SG -> LP -> SG -> LP
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    # Phase A: Log-sum-exp subgradient
    t_alloc = time_limit * 0.35
    if t_alloc > 5:
        a_sg, obj_sg = smoothed_subgradient_descent(
            a, time_limit=t_alloc, seed=seed,
            lr_init=0.004, momentum=0.92,
            temp_init=80.0, temp_decay=0.9995,
            restart_interval=1500, kick_size=0.005
        )
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            if verbose:
                print(f"    SG1: {best_obj:.10f}")

    # Phase B: LP refinement
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        t_lp = min(rem * 0.35, 50.0)
        a_lp, obj_lp = lp_refine_improved(best_a, max_iters=40, time_limit=t_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    LP1: {best_obj:.10f}")

    # Phase C: More subgradient
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        t_sg2 = rem * 0.4
        a_sg2, obj_sg2 = smoothed_subgradient_descent(
            best_a, time_limit=t_sg2, seed=seed + 7777,
            lr_init=0.002, momentum=0.94,
            temp_init=50.0, temp_decay=0.9997,
            restart_interval=1000, kick_size=0.003
        )
        if obj_sg2 < best_obj:
            best_obj = obj_sg2
            best_a = a_sg2.copy()
            if verbose:
                print(f"    SG2: {best_obj:.10f}")

    # Phase D: Final LP
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        a_lp2, obj_lp2 = lp_refine_improved(best_a, max_iters=20, time_limit=rem)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            if verbose:
                print(f"    LP2: {best_obj:.10f}")

    return best_a, best_obj


def sort_by_proximity(target):
    """Return a key function equivalent for sorting by proximity to target."""
    return abs(target)


def length_sort_key(x, center):
    return abs(x - center)


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent, optimize on parent
    2. Diverse initializations
    3. Multi-length exploration
    4. Deep optimization on best
    5. Coordinate descent polish
    6. Final micro-polish
    """
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_uniform_init(1152)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using uniform n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # === Phase 1: Deep optimization on parent ===
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Deep optimization on parent (n={n_parent})")
    print(f"{'=' * 60}")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t = min(rem * 0.5, 200.0)
        print(f"  Cycle 1: t={t:.0f}s")
        a_opt, obj_opt = run_optimization_cycle(parent, t, n_parent, 42, verbose=True)
        try_update(a_opt, "phase1")

        # Second cycle from the result
        rem = deadline - time.monotonic() - 350
        if rem > 20 and best_seq is not None:
            t2 = min(rem * 0.4, 80.0)
            print(f"  Cycle 2: t={t2:.0f}s")
            a_opt2, obj_opt2 = run_optimization_cycle(best_seq, t2, len(best_seq), 12345, verbose=True)
            try_update(a_opt2, "phase1b")

    # === Phase 2: Diverse initializations ===
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Diverse Initializations (best={best_score:.6f})")
    print(f"{'=' * 60}")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits_list = [
        ("gauss_0.15", make_gaussian_init(n_base, 0.15)),
        ("gauss_0.25", make_gaussian_init(n_base, 0.25)),
        ("cosine", make_cosine_init(n_base)),
        ("tukey_0.3", make_tukey_init(n_base, 0.3)),
        ("skewed_1.5", make_skewed_init(n_base, 1.5)),
        ("two_lobe", make_two_lobe_init(n_base)),
        ("waterfall", make_waterfall_init(n_base, 4.0)),
    ]

    total_diverse = min(200.0, (deadline - time.monotonic() - 350.0) * 0.35)
    time_per_init = total_diverse / max(1, len(inits_list))

    for idx, (name, init_a) in enumerate(inits_list):
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.2)
        if t_alloc < 8:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = run_optimization_cycle(
            init_a, t_alloc, n_base, 42 + idx * 1111, verbose=False
        )
        try_update(a_div, f"div_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Multi-length Exploration (best={best_score:.6f})")
    print(f"{'=' * 60}")
    if best_seq is not None:
        factors = [0.75, 1.0, 1.25, 1.5, 0.5, 1.33, 2.0, 0.67]
        target_lengths = []
        for factor in factors:
            tl = int(n_parent * factor)
            tl = max(256, min(5000, tl))
            if tl not in target_lengths:
                target_lengths.append(tl)
        # Sort by proximity to parent length
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:5]:
            rem = deadline - time.monotonic() - 250
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 50.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = run_optimization_cycle(
                f_interp, t_alloc, target_n, 42 + target_n, verbose=False
            )
            try_update(a_len, f"len_{target_n}")

    # === Phase 4: Deep optimization on current best ===
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Deep Optimization (best={best_score:.6f})")
    print(f"{'=' * 60}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 120
        if rem > 30:
            t = min(rem * 0.45, 120.0)
            print(f"  Deep cycle: n={n_best}, t={t:.0f}s")
            a_deep, obj_deep = run_optimization_cycle(
                best_seq, t, n_best, 99999, verbose=True
            )
            try_update(a_deep, "deep")

            # Extra LP pass
            rem = deadline - time.monotonic() - 80
            if rem > 10 and best_seq is not None:
                a_lp, obj_lp = lp_refine_improved(
                    best_seq, max_iters=30, time_limit=min(rem, 40.0)
                )
                try_update(a_lp, "deep_lp")

            # Another SG cycle
            rem = deadline - time.monotonic() - 60
            if rem > 10 and best_seq is not None:
                t_sg = min(rem, 30.0)
                a_sg, obj_sg = smoothed_subgradient_descent(
                    best_seq, time_limit=t_sg, seed=555555,
                    lr_init=0.001, momentum=0.95,
                    temp_init=100.0, temp_decay=0.9998,
                    restart_interval=800, kick_size=0.002
                )
                try_update(a_sg, "deep_sg")

    # === Phase 5: Coordinate descent ===
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Coordinate Descent (best={best_score:.6f})")
    print(f"{'=' * 60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_cd = min(rem, 50.0)
            n_cd = len(best_seq)
            print(f"  CD: n={n_cd}, t={t_cd:.0f}s")
            a_cd, obj_cd = coordinate_descent_golden(best_seq, time_limit=t_cd, n_passes_max=30)
            try_update(a_cd, "cd")

    # === Phase 6: Final micro-polish ===
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Final Micro-polish (best={best_score:.6f})")
    print(f"{'=' * 60}")
    if best_seq is not None:
        n_final = len(best_seq)
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            t_micro = min(rem * 0.4, 15.0)
            a_micro, obj_micro = smoothed_subgradient_descent(
                best_seq, time_limit=t_micro, seed=777777,
                lr_init=0.0002, momentum=0.96,
                temp_init=150.0, temp_decay=0.9999,
                restart_interval=500, kick_size=0.001
            )
            try_update(a_micro, "micro_sg")

            rem = deadline - time.monotonic() - 8
            if rem > 4:
                a_micro_lp, obj_micro_lp = lp_refine_improved(
                    best_seq, max_iters=8, time_limit=min(rem, 10.0)
                )
                try_update(a_micro_lp, "micro_lp")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
