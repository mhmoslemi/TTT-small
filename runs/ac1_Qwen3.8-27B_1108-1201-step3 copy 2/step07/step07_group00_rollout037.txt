```python
"""
Hybrid optimization combining log-sum-exp gradient descent (with continuation),
LP-based direction finding, and coordinate descent to minimize
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_direct(f, weights):
    """
    Compute grad[i] = sum_k weights[k] * f[k-i] for i=0..n-1.
    weights has length 2n-1 (the auto-convolution length).
    """
    n = len(f)
    grad = np.zeros(n)
    for k in range(len(weights)):
        wk = weights[k]
        if wk < 1e-16:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # grad[lo..hi] += wk * f[k-lo .. k-hi] reversed
        # f[k-i] for i=lo..hi is f[k-lo], f[k-lo-1], ..., f[k-hi]
        # = f[k-hi : k-lo+1][::-1]
        grad[lo:hi+1] += wk * f[k-hi : k-lo+1][::-1]
    return grad


def logsumexp_gradient_step(f, T, lr):
    """One step of gradient descent on (1/T)*log(sum exp(T*(f*f)[k]))."""
    n = len(f)
    b = fast_convolve(f)
    b_max = b.max()
    Tb = T * (b - b_max)
    # Compute softmax weights
    exp_Tb = np.exp(Tb)
    w = exp_Tb / exp_Tb.sum()
    # Compute gradient
    grad = compute_gradient_direct(f, w)
    # Gradient step
    f_new = f - lr * grad
    # Project onto simplex
    f_new = project_simplex(f_new)
    return f_new, b.max()


def run_logsumexp_phase(f_init, time_limit, T_values=None, lr_init=0.001,
                        max_iters_per_T=100):
    """
    Log-sum-exp gradient descent with continuation.
    T increases from small to large, making the problem progressively
    more like the original max problem.
    """
    if T_values is None:
        T_values = [0.05, 0.1, 0.3, 1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0]
    
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for T in T_values:
        if time.monotonic() - t0 >= time_limit:
            break
        lr = lr_init
        no_improve = 0
        for it in range(max_iters_per_T):
            if time.monotonic() - t0 >= time_limit:
                break
            f, _ = logsumexp_gradient_step(f, T, lr)
            if it % 10 == 0:
                obj = fast_convolve(f).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.1, 0.05)
                else:
                    no_improve += 1
                    if no_improve > 3:
                        lr *= 0.7
                        no_improve = 0
        print(f"    LSE T={T:.1f} obj={best_obj:.12f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def lp_solve(f, n, top_k):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for top_k peaks."""
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi+1] = f[lo:hi+1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=300000, abstol=1e-9)
        except Exception:
            return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

def coordinate_descent(f_init, time_limit):

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = project_simplex(f_trial)
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:

                    best_val = new_val

                f[j] = best_val
                f = project_simplex(f)
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    improved = True
        if not improved:
            break
    
    return f, best_obj
```

```python
Hybrid optimization: Mirror descent with log-sum-exp smoothing (continuation),

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_direct(f, weights):
    """Compute grad[i] = sum_k weights[k] * f[k-i] for i=0..n-1."""
    n = len(f)
    grad = np.zeros(n)
    for k in range(len(weights)):
        wk = weights[k]
        if wk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        grad[lo:hi + 1] += wk * f[k - hi:k - lo + 1][::-1]
    return grad

        f = project_simplex(f)

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

        if it % 15 == 0:
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0 * n * best_obj:.6f} top_k={top_k} "
                  f"t={time.monotonic() - t0:.0f}s")

    return best_f, best_obj


def coordinate_descent(f_init, time_limit):
    """Coordinate descent polish on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = project_simplex(f_trial)
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_val = new_val
            if best_val != f[j]:
                f[j] = best_val
                f = project_simplex(f)
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: Mirror descent with LSE continuation, LP polishing,
    multi-length exploration, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Mirror descent with continuation ===
    print(f"\n=== Phase 1: Mirror Descent (n={len(parent)}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_mirror_descent(
            parent, time_limit=t1,
            T_values=[1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0],
            lr_init=0.5, label="md_main"
        )
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [md]: {score1:.6f}")

    # === Phase 2: LP polishing (main) ===
    print(f"\n=== Phase 2: LP Polish (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = run_lp_phase(best_seq, time_limit=t2,
                                top_k_init=50, top_k_max=250, label="lp1")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp1]: {score2:.6f}")

    # === Phase 3: More mirror descent on best so far ===
    print(f"\n=== Phase 3: Mirror Descent refined (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        f3, obj3 = run_mirror_descent(
            best_seq, time_limit=t3,
            T_values=[30.0, 100.0, 300.0, 1000.0, 3000.0, 10000.0],
            lr_init=0.3, label="md_ref"
        )
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [md_ref]: {score3:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.85, 1.15, 1.4]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t4 = min(rem * 0.3, 250.0)
        if t4 < 30:
            break

        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  Exploring n={target_n}, budget={t4:.0f}s")
        f4a, _ = run_mirror_descent(f_interp, time_limit=t4 * 0.35,
                                     T_values=[10.0, 100.0, 1000.0],
                                     lr_init=0.4, label=f"md_n{target_n}")
        f4, obj4 = run_lp_phase(f4a, time_limit=t4 * 0.65,
                                top_k_init=40, top_k_max=200,
                                label=f"lp_n{target_n}")
        score4 = float(evaluate_sequence(f4.tolist()))

    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.25, 250.0)
    if t5 > 15:
        print(f"  Budget: {t5:.0f}s, n={len(best_seq)}")
        f5, obj5 = coordinate_descent(best_seq, time_limit=t5)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [cd]: {score5:.6f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.6, 300.0)
    if t6 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = run_lp_phase(best_seq, time_limit=t6,
                                top_k_init=100, top_k_max=350, label="lp_fin")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_fin]: {score6:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
```
