"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def score_normalized(a):
    """Compute the score for a normalized (sum=1) sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):

def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=1.0, maxiter=5000):
    """L-BFGS with soft-max smoothing and geometric temperature schedule."""
    from scipy.optimize import minimize
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = max(1, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        a_check = softmax_from_z(z)
        sc = score_normalized(a_check)
        if np.isfinite(sc):
            print(f"    L-BFGS T={stage_T:.1f}: score={sc:.6f}")

    a_out = softmax_from_z(z)
    return a_out


def lp_nesterov_iterate(a0, time_limit, solver=None, n_max_iters=10000):
    """
    Nesterov-accelerated LP best-response dynamics with adaptive step size.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()  # virtual iterate
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Pre-build the conv matrix for y (will rebuild each iteration)
    no_improve = 0
    patience = 500

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP for best response to y
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov step size
        alpha = 2.0 / (it + 2)

        # Update y
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Update a using running average
        a = ((it + 1) * a + b_new) / (it + 2)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        # Evaluate
        score_a = score_normalized(a)
        score_y = score_normalized(y)
        
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # If stuck, perturb
        if no_improve > patience:
            # Add small random perturbation in log-space
            z = z_from_a(best_a)
            noise = np.random.default_rng(it + 42).normal(0, 0.01, n)
            z = z + noise
            perturbed = softmax_from_z(z)
            sc_pert = score_normalized(perturbed)
            if sc_pert < best_score:
                best_score = sc_pert
                best_a = perturbed
            a = best_a.copy()
            y = best_a.copy()
            no_improve = 0
            patience = max(200, patience // 2)

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=5000):
    """Local peak shaving: reduce coordinates that contribute most to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the peak (excluding endpoints which are a[0]^2 and a[n-1]^2)
        b_check = b.copy()
        b_check[0] = -np.inf
        b_check[-1] = -np.inf
        peak_k = int(np.argmax(b_check))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            # Try reducing i
            for factor in [0.98, 0.95, 0.9, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
            # Try reducing both i and j (if different)
            if i != j and a[j] > 1e-14:
                for f1 in [0.97, 0.93]:
                    for f2 in [0.97, 0.93]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            # Try random perturbation
            z = z_from_a(a)
            noise = rng.normal(0, 0.005, n)
            z_pert = z + noise
            a_pert = softmax_from_z(z_pert)
            sc_pert = score_normalized(a_pert)
            if sc_pert < best_score - 1e-14:
                a = a_pert
                best_score = sc_pert
                best_a = a.copy()
            else:
                # Try reducing the top 3 contributors simultaneously
                top3 = top_contrib[:3]
                a_trial = a.copy()
                for ci in top3:
                    i = idx[ci]
                    a_trial[i] *= 0.96
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                else:
                    break

    return best_a, best_score


def coordinate_descent_active(a0, time_limit, n_passes=50):
    """Coordinate descent focusing on active coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find active coordinates (those that appear in near-max convolution positions)
        threshold = b.max() * 0.8
        active_k = np.where(b >= threshold)[0]
        active_coords = set()
        for k in active_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            active_coords.update(range(js, je + 1))
        
        active_list = sorted(active_coords)
        if not active_list:
            active_list = list(range(n))

        for i in active_list:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response dynamics
with Nesterov acceleration, adaptive L-BFGS soft-max refinement, peak shaving,
and multi-length exploration. Key innovations: adaptive LP step sizes, log-space
perturbation for escaping local minima, and aggressive low-temperature L-BFGS polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def score_normalized(a):
    """Compute the score for a normalized (sum=1) sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:

    except Exception:
        return float('inf')


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=1.0, maxiter=5000):
    """L-BFGS with soft-max smoothing and geometric temperature schedule."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = max(1, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        a_check = softmax_from_z(z)
        sc = score_normalized(a_check)
        if np.isfinite(sc):
            print(f"    L-BFGS T={stage_T:.2f}: score={sc:.6f}")

    a_out = softmax_from_z(z)
    return a_out


def lp_nesterov_iterate(a0, time_limit, solver=None, n_max_iters=10000):

    """

    a = a / s
    y = a.copy()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    no_improve = 0
    patience = 300
    perturb_scale = 0.02

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)

        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + b_new) / (it + 2)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        score_a = score_normalized(a)
        score_y = score_normalized(y)

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score - 1e-12:

            best_a = y.copy()
            no_improve = 0
        else:

            z = z + noise

            if sc_pert < best_score:
                best_score = sc_pert
                best_a = perturbed
            a = best_a.copy()
            y = best_a.copy()
            no_improve = 0
            patience = max(100, patience * 2 // 3)
            perturb_scale = max(0.005, perturb_scale * 0.7)

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

def peak_shave(a0, time_limit, n_steps=5000):
    """Local peak shaving: reduce coordinates contributing most to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_check = b.copy()
        b_check[0] = -np.inf
        b_check[-1] = -np.inf
        peak_k = int(np.argmax(b_check))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.9, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
            if i != j and a[j] > 1e-14:
                for f1 in [0.97, 0.93]:
                    for f2 in [0.97, 0.93]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            z = z_from_a(a)
            noise = rng.normal(0, 0.005, n)
            z_pert = z + noise
            a_pert = softmax_from_z(z_pert)
            sc_pert = score_normalized(a_pert)
            if sc_pert < best_score - 1e-14:
                a = a_pert
                best_score = sc_pert
                best_a = a.copy()
            else:
                top3 = top_contrib[:3]
                a_trial = a.copy()
                for ci in top3:
                    i = idx[ci]
                    a_trial[i] *= 0.96
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                else:
                    break

    return best_a, best_score


def mirror_descent_refine(a0, time_limit, lr_init=0.05, n_steps=20000):
    """Mirror descent with subgradient and adaptive learning rate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b - 0.03 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = score_normalized(a_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.85
            no_improve = 0
        lr = max(lr, 1e-7)

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP best-response with Nesterov acceleration,
    L-BFGS soft-max, mirror descent, peak shaving, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: LP Nesterov (600s) ===
    print(f"\n=== Phase 1: LP-Nesterov (best={best_score:.6f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.33, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = lp_nesterov_iterate(best_seq, t_alloc, solver=solver)
        try_update(a_lp, "lp_nesterov_p1")

    # === Phase 2: L-BFGS main (350s) ===
    print(f"\n=== Phase 2: L-BFGS Main (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 50->1, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=50.0, T_end=1.0, maxiter=6000)
        try_update(a_lb, "lbgfs_main")

    # === Phase 3: LP Round 2 (250s) ===
    print(f"\n=== Phase 3: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        a_lp2, _ = lp_nesterov_iterate(best_seq, t_alloc, solver=solver, n_max_iters=8000)
        try_update(a_lp2, "lp_round2")

    # === Phase 4: Mirror Descent (200s) ===
    print(f"\n=== Phase 4: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_md, _ = mirror_descent_refine(best_seq, t_alloc, lr_init=0.03)
        try_update(a_md, "mirror_desc")

    # === Phase 5: Peak shaving (150s) ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_ps, _ = peak_shave(best_seq, t_alloc, n_steps=8000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Low-T L-BFGS polish (200s) ===
    print(f"\n=== Phase 6: Low-T L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_lt = lbgfs_refine(best_seq, t_alloc, T_start=1.0, T_end=0.1, maxiter=5000)
        try_update(a_lt, "lbgfs_lowT")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = [int(n_best * 1.25), int(n_best * 0.8), int(n_best * 1.5), 1024, 640]
    alt_lengths = [x for x in alt_lengths if 256 <= x <= 2048 and x != n_best]
    alt_lengths = list(dict.fromkeys(alt_lengths))[:3]
    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(rem * 0.3, 100.0)
        if t_alloc < 25:
            break
        print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt = lbgfs_refine(init_alt, t_alloc, T_start=30.0, T_end=1.0, maxiter=3000)
        try_update(a_alt, f"alt_n{alt_n}")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t1 = min(rem * 0.35, 60.0)
        a_f1 = lbgfs_refine(best_seq, t1, T_start=0.5, T_end=0.05, maxiter=3000)
        try_update(a_f1, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t2 = min(rem * 0.3, 30.0)
        a_f2, _ = peak_shave(best_seq, t2, n_steps=5000)
        try_update(a_f2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 8:
        t3 = min(rem * 0.3, 25.0)
        a_f3, _ = lp_nesterov_iterate(best_seq, t3, solver=solver, n_max_iters=2000)
        try_update(a_f3, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
