```python
def alternating_projection(a, n_iters=10000):
    n = len(a)
    N = 2 * n - 1  # FFT size
    # Or use next power of 2 for efficiency
    N_fft = 1 << (N - 1).bit_length()
    
    # Target magnitude
    target_mag = (2 * n - 1) ** (-0.25)
    
    for it in range(n_iters):
        # Frequency domain: enforce flat magnitude
        A = np.fft.rfft(a, N_fft)
        phase = A / np.abs(A + 1e-300)
        A_modified = target_mag * phase
        a_new = np.fft.irfft(A_modified, N_fft)[:n]
        
        # Time domain: enforce non-negativity and sum = 1
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new = a_new / s
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-15:
            a = a_new
            break
        a = a_new
    
    return a
```

```python
def fienup_projection(a, n_iters=10000, beta=0.9):
    n = len(a)
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    target_mag = (2 * n - 1) ** (-0.25)
    
    for it in range(n_iters):
        # Frequency domain
        A = np.fft.rfft(a, N_fft)
        phase = A / (np.abs(A) + 1e-300)
        A_modified = target_mag * phase
        a_projected = np.fft.irfft(A_modified, N_fft)[:n]
        
        # Time domain projection
        a_projected = np.maximum(a_projected, 0)
        s = a_projected.sum()
        if s < 1e-300:
            break
        a_projected = a_projected / s
        
        # Mixing step (Fienup relaxation)
        a = beta * a + (1 - beta) * a_projected
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    return a
```

```python
import numpy as np
```

from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s

def alternating_projections(a0, time_limit, beta_range=(0.7, 1.3), n_restarts=5, seed=42):

    """

    fft_size = 1 << (N - 1).bit_length()

    best_score = get_score(best_a)
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Different initialization for each restart
        if restart == 0:
            a = normalize(a0.copy())
        else:
            a = best_a + rng.standard_normal(n) * (0.05 + 0.02 * restart)
            a = normalize(a)
        
        # Vary beta
        beta = rng.uniform(*beta_range)
        
        # Run alternating projections
        stall_count = 0
        local_best = a.copy()
        local_best_score = get_score(a)
        
        max_iters_per_restart = 20000
        for it in range(max_iters_per_restart):
            if time.monotonic() - t0 > time_limit:
                break

            A = rfft(a, fft_size)
            mag = np.abs(A)
            phase = A / (mag + 1e-300)

            progress = it / max_iters_per_restart

            for k in range(len(A)):

                    A_mod[k] = (mag[k] ** (1-alpha) * target_mag ** alpha) * phase[k]
            
            a_freq = irfft(A_mod, fft_size)[:n]

            a_time = np.maximum(a_freq, 0.0)

            if s < 1e-300:

                a = normalize(a)
                stall_count = 0

def subgradient_descent(a0, time_limit, initial_step=0.01, decay=0.999, seed=42):
    """
    Subgradient descent for minimizing max(autocorrelation).
    """
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = get_score(a)
    step_size = initial_step
    fft_size = 1 << (2 * n - 2).bit_length()
    
    no_improve = 0
    avg_iterate = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        # Compute autocorrelation
        c = fast_convolve(a)
        k_star = int(np.argmax(c))
        
        # Subgradient: g_m = 2 * a_{k_star - m} (if valid)
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for m in range(lo, hi + 1):
            g[m] = 2.0 * a[k_star - m]
        
        # Project to tangent space (sum g = 0)
        g = g - g.mean()
        
        # Backtracking line search
        alpha = step_size
        for _ in range(20):
            a_trial = a - alpha * g
            a_trial = normalize(a_trial)
            score_trial = get_score(a_trial)
            if score_trial < best_score or score_trial < get_score(a) * 0.999:
                a = a_trial
                break
            alpha *= 0.5
        
        # Update
        a = normalize(a)
        avg_iterate = 0.99 * avg_iterate + 0.01 * a
        
        score = get_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        step_size *= decay
        
        # Random restart if stuck
        if no_improve > 500:
            a = best_a + rng.standard_normal(n) * 0.005
            a = normalize(a)
            no_improve = 0
            step_size = initial_step * 0.5
        
        # Also check averaged iterate
        if no_improve > 100:
            avg_score = get_score(avg_iterate)
            if avg_score < best_score:
                best_score = avg_score
                best_a = normalize(avg_iterate).copy()
    
    return best_a, best_score

def peak_shave(a0, time_limit, n_trials=5000, seed=42):

    Peak-targeted local search.
    """

    rng = np.random.default_rng(seed)

        if time.monotonic() - t0 > time_limit:
            break

        # Find interior peak (not at edges)
        c_search = c.copy()
        if n > 2:
            c_search[0] = -np.inf
            c_search[-1] = -np.inf
        peak_k = int(np.argmax(c_search))
        
        if peak_k < 0 or peak_k >= 2 * n - 1:
            peak_k = int(np.argmax(c))
        
        peak_val = c[peak_k]

        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        top_count = min(8, len(idx))
        top_pos = np.argsort(contribs)[::-1][:top_count]
        
        improved = False
        for pi in top_pos:
            i = int(idx[pi])
            j = int(peak_k - i)
            if j < 0 or j >= n or i == j:

            if a[i] < 1e-15 or a[j] < 1e-15:
                continue
            
            for f1 in [0.95, 0.9, 0.85, 0.8, 0.7]:
                for f2 in [1.0, 1.05, 1.1, 1.15]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = normalize(a_trial)
                    score = get_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

                break
        
        if not improved:
            # Try single element reduction with redistribution

                i = int(idx[pi])
                if a[i] < 1e-15:
                    continue
                for factor in [0.85, 0.75, 0.65]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    os = others.sum()
                    if os < 1e-15:
                        continue
                    a_trial[mask] += diff * others / os
                    a_trial = normalize(a_trial)
                    score = get_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break

                    break

            # Perturbation
            a = best_a + rng.standard_normal(n) * 0.003
            a = normalize(a)

            best_local_score = best_score
            
            for factor in [0.7, 0.8, 0.9, 1.1, 1.2]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = normalize(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor
            
            if best_local != a[i]:
                a[i] = best_local
                a = normalize(a)
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True
        
        if not sweep_improved and sweep > 5:
            break
    
    return best_a, best_score
