```python
from scipy.optimize import linprog
import numpy as np

def lp_direction(a):
    """Solve the LP to find the best direction."""
    n = len(a)
    M = max(convolve(a, a))
    
    # Constraint: (a*b)[k] <= M for k = 0, ..., 2n-2
    # (a*b)[k] = sum_j a[k-j] * b[j] for valid j
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Build constraint matrix A (2n-1 x n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je+1] = a[k - np.arange(js, je+1)]
    
    # max 1^T b s.t. A b <= M, b >= 0
    # linprog minimizes, so we negate the objective
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return np.ones(n) / n
```

```python
LP-based iterative method for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the linear program from the state-of-the-art approach as the primary
optimization driver, combined with L-BFGS polishing and coordinate descent.
Key improvement: exploits the LP structure more efficiently with batch
evaluations, adaptive step sizes, and multi-phase refinement including
symmetry exploitation and alternative length exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Full cross-convolution of a and b (both length n)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_conv_matrix(a):
    """Build (2n-1) x n matrix where row k gives coefficients for (a*b)[k]."""
    n = len(a)
    k = np.arange(2*n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    idx_clipped = np.clip(idx, 0, n-1)
    A = np.where(valid, a[idx_clipped], 0.0)
    return A


def solve_lp_direction(a, M):
    """Solve LP: max 1^T b s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def lp_iteration(a, t=0.5, verbose=False):
    """One LP-based iteration."""
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max()
    
    b = solve_lp_direction(a, M)
    if b is None:
        return a, M, None
    
    s = b.sum()
    if s < 1e-15:
        return a, M, None
    
    g = b / s
    
    # Line search
    best_t = t
    best_obj = score_normalized((1-t)*a + t*g)
    for t_try in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

        if obj < best_obj:

            best_t = t_try
    
    a_new = (1-best_t)*a + best_t*g
    return a_new, best_obj, g
```

```python
"""
LP-guided iterative optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.

Primary method: Linear program direction finding (max sum(b) s.t. (a*b) <= M, b >= 0)
at each iteration, with adaptive step sizes, symmetry exploitation, and multi-phase
polishing (L-BFGS in log-space, coordinate descent, peak shaving). Explores
alternative sequence lengths. Key improvement over prior art: the LP gives the
provably optimal convex direction each step, leading to faster convergence than
gradient-based mirror descent.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Full cross-convolution of a and b."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_conv_matrix(a, row_starts, row_ends):
    """Build (2n-1) x n convolution constraint matrix efficiently."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = row_starts[k]
        e = row_ends[k]
        A[k, s:e + 1] = a[k - s:k - e - 1:-1]
    return A


def precompute_indices(n):
    """Precompute row structure of convolution matrix."""
    row_starts = np.maximum(0, np.arange(2 * n - 1) - n + 1)
    row_ends = np.minimum(n - 1, np.arange(2 * n - 1))
    return row_starts, row_ends


def solve_lp_direction(a, M, row_starts, row_ends):
    """Solve LP: max 1^T b s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a, row_starts, row_ends)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None

def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_soft_b_weights(b, T):
    """Softmax weights over convolution values."""
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    return exp_scaled / exp_scaled.sum()


def log_space_objective(z, T, n):
    """Smoothed objective in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return 2.0 * n * T * lse


def log_space_gradient(z, T, n):
    """Gradient of smoothed objective in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    w = compute_soft_b_weights(b, T)
    # grad_a = 4n * cross_correlate(w, a)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    grad_a = 4.0 * n * correl[:n]
    mean_grad = np.dot(a, grad_a)
    return a * (grad_a - mean_grad)


def lbgfs_log_space(a_init, time_limit, T=50, maxiter=3000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return log_space_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        return log_space_gradient(z, T, n)

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxfun': maxiter * 40,
                                'disp': False})
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a_init.copy()
    return a_out


def coordinate_descent(a_init, time_limit, verbose=True, label=""):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    factors = [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] -= diff * a_trial[mask] / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1

    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=True, label=""):
    """Targeted peak shaving."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        if b[peak_k] <= best_score / (2.0 * n) + 1e-16:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14 or i == j:
                continue
            for factor in [0.9, 0.8, 0.7, 0.5]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial[j] *= factor
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try single coordinate
            for ci in top_idx[:4]:
                i = int(idx[ci])
                if a[i] < 1e-14:
                    continue
                for factor in [0.8, 0.6, 0.4]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate using evaluate_sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-based direction finding (primary, ~60% time)
    2. L-BFGS in log-space with temperature schedule (~15% time)
    3. Coordinate descent + peak shaving (~15% time)
    4. Alternative length exploration (~10% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={p_score:.7f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: LP-based iteration (primary) ===
    print(f"\n=== Phase 1: LP-based Iteration (best={best_score:.7f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.60, 1000.0)
    print(f"  LP: n={n_best}, budget={t_alloc:.0f}s")
    a_lp, _ = lp_phase(best_seq, t_alloc, label="LP")
    best_seq, best_score = check_and_update(a_lp, "LP_main", best_seq, best_score)

    # Second LP round with lower perturbation
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc2 = min(rem * 0.3, 200.0)
            print(f"  LP round 2: budget={t_alloc2:.0f}s")
            a_lp2, _ = lp_phase(best_seq, t_alloc2, label="LP2")
            best_seq, best_score = check_and_update(a_lp2, "LP2", best_seq, best_score)

    # === Phase 2: L-BFGS polishing ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_now = len(best_seq)
        temperatures = [200, 100, 50, 25, 10, 5, 3, 2, 1]
        for T in temperatures:
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 60.0)
            print(f"  L-BFGS T={T}, n={n_now}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_log_space(best_seq, t_alloc, T=T, maxiter=2000)
            best_seq, best_score = check_and_update(a_lb, f"lbfs_T{T}", best_seq, best_score)

    # === Phase 3: Peak shaving and coordinate descent ===
    print(f"\n=== Phase 3: Peak Shaving + CD (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem * 0.4, 80.0)
            print(f"  Peak Shave: budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem * 0.4, 60.0)
            print(f"  Coord Descent: budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_cd, "CD", best_seq, best_score)

    # === Phase 4: L-BFGS again on polished result ===
    print(f"\n=== Phase 4: L-BFGS Round 2 (best={best_score:.7f}) ===")
    if best_seq is not None:
        for T in [5, 2, 1]:
            rem = deadline - time.monotonic() - 40
            if rem < 10:
                break
            t_alloc = min(rem * 0.3, 40.0)
            print(f"  L-BFGS2 T={T}, budget={t_alloc:.0f}s")
            a_lb2 = lbgfs_log_space(best_seq, t_alloc, T=T, maxiter=3000)
            best_seq, best_score = check_and_update(a_lb2, f"lbfs2_T{T}", best_seq, best_score)

    # === Phase 5: Alternative lengths ===
    print(f"\n=== Phase 5: Alternative Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_now = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5, 2.0, 0.5]:
            alt_n = int(n_now * factor)
            if 64 <= alt_n <= 4096 and alt_n != n_now and alt_n % 2 == 0:
                alt_lengths.append(alt_n)

        seen = set()
        for alt_n in alt_lengths:
            if alt_n in seen:
                continue
            seen.add(alt_n)
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick LP on new length
            a_alt, _ = lp_phase(init_alt, t_alloc * 0.6, max_iters=5000, label=f"alt_n{alt_n}")
            best_seq, best_score = check_and_update(a_alt, f"alt_lp_n{alt_n}", best_seq, best_score)

            if best_seq is not None and len(best_seq) == alt_n:
                rem = deadline - time.monotonic() - 20
                if rem > 10:
                    a_alt_lb = lbgfs_log_space(best_seq, min(rem * 0.5, 20.0), T=10)
                    best_seq, best_score = check_and_update(a_alt_lb, f"alt_lb_n{alt_n}",
                                                            best_seq, best_score)

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        # Final LP
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.4, 30.0)
            print(f"  Final LP: budget={t_alloc:.0f}s")
            a_fin, _ = lp_phase(best_seq, t_alloc, label="FinalLP")
            best_seq, best_score = check_and_update(a_fin, "final_LP", best_seq, best_score)

        # Final peak shave
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_ps_fin, _ = peak_shave(best_seq, min(rem * 0.5, 10.0))
            best_seq, best_score = check_and_update(a_ps_fin, "final_PS", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = p_score

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
