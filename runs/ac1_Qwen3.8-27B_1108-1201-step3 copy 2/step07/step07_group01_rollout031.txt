Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral flattening (Gerchberg-Saxton), adaptive peak shaving,
and LP-inspired dual updates.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def spectral_flatten(a_init, time_limit, iters=5000, verbose=True):
    """
    Gerchberg-Saxton style: alternate between
    spatial domain (nonnegativity, sum constraint) and
    frequency domain (flat magnitude).
    """
    n = len(a)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    # Target magnitude for flat spectrum
    # If auto-correlation is perfectly flat at value c, then
    # sum_k b[k] = (sum a)^2 = 1, so c = 1/(2n-1)
    # Power spectrum: |A[k]|^2 proportional to b (circular)
    # For linear, it's approximate. Let's just use a target.
    target_mag = 1.0 / np.sqrt(2 * n - 1)  # sqrt of target auto-corr value
    
    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Go to frequency domain
        A = rfft(a, 2 * n)  # Use 2n to avoid circular aliasing
        
        # Modify magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Blend: partially flatten
        # Use a "soft" flattening: new_mag = mag * (target/mag)^alpha
        alpha = 0.5  # Full flattening
        new_mag = target_mag + (1 - alpha) * (mag - target_mag)
        # Or: new_mag = mag ** (1-alpha) * target_mag ** alpha
        new_mag = np.maximum(new_mag, 1e-10)
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = irfft(A_new, 2 * n)[:n]
        
        # Project: nonnegativity and sum constraint
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        
        sc = local_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={it} ({elapsed:.0f}s): score={sc:.7f} best={best_score:.7f}")
    
    return best_a, best_score

from scipy.optimize import linprog

def lp_update(f, M):
    """
    Solve the LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    
    # Constraint matrix A: (f*g)[k] = sum_i f[i]*g[k-i] <= M
    # For lag k, the valid i range is max(0, k-n+1) <= i <= min(n-1, k)
    
    # Build A matrix (sparse would be better but for n=768 it's fine)
    # A has (2n-1) rows and n columns
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            A[k, i] = f[i]
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    return res.x
