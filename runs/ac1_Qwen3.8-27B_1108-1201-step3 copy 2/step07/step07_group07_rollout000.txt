import numpy as np

from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def build_conv_constraint_matrix(f):
    """Build matrix A such that A @ g = conv(f,g) for g of same length as f."""
    n = len(f)
    num_k = 2 * n - 1
    A = np.zeros((num_k, n))
    for k in range(num_k):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[hi:-lo-1:-1] if hi > lo else f[lo:hi+1]
    # Fix: for k, the valid j range is [max(0,k-n+1), min(n-1,k)]
    # A[k, j] = f[k-j] if k-j is in [0, n-1]
    A = np.zeros((num_k, n))
    for k in range(num_k):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    return A


def build_conv_constraint_matrix_fast(f):
    """Build A matrix efficiently using slicing."""
    n = len(f)
    num_k = 2 * n - 1
    A = np.zeros((num_k, n))
    for k in range(num_k):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For position j in [lo, hi], A[k, j] = f[k - j]
        vals = f[lo:n-1-(hi-lo): -1]  # f[hi], f[hi-1], ..., f[lo] reversed
        A[k, lo:hi+1] = f[(k - lo):(k - hi):-1] if (k - lo) > (k - hi) else f[k-hi:k-lo+1]
    # Simpler direct approach
    A = np.zeros((num_k, n))

        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        A[k_lo:k_hi+1, j] = f[j]

def lp_flattest_direction(f, method='highs'):
    """
    Solve: min t s.t. conv(f,g)[k] <= t for all k, sum(g) = 1, g >= 0.

    # Constraints: A @ g - t <= 0 for each k
    #              sum(g) = 1
    #              g >= 0, t free

    A_ub = np.hstack([A, -np.ones((num_k, 1))])
    b_ub = np.zeros(num_k)

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_step(a, lr, v_momentum, momentum, n_top):
    """One subgradient step on max(conv(a,a)) with simplex constraint."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()

    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))

    top_vals = b[top_idx]
    threshold = b_max * 0.995
    active_mask = top_vals >= threshold
    if not active_mask.any():
        active_mask[0] = True

    w = np.zeros(2 * n - 1)
    active_indices = top_idx[active_mask]
    w[active_indices] = 1.0 / len(active_indices)

    # Subgradient: g[i] = sum_j a[j] * w[i+j]
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]

    v_momentum[:] = momentum * v_momentum - lr * g
    a_new = project_simplex(a + v_momentum)
    return a_new, g

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_local_max_b(a)

            best_a = a_new.copy()

            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.05)
            no_improve = 0
        elif no_improve >= 2000:
            cur_lr *= 0.95
        elif no_improve >= 500:
            cur_lr *= 0.98

    return best_a, best_obj


def lp_line_search(f, direction, max_obj):
    """
    Line search: find best t in (0, 1] minimizing max(conv((1-t)f + t*direction, ...)).
    Returns (best_blend, best_obj, best_t).
    """
    f = f.astype(np.float64)
    direction = direction.astype(np.float64)
    direction = direction / direction.sum()

    best_obj = max_obj
    best_t = 0.0
    best_blend = f.copy()

    t_candidates = [0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]

    for t in t_candidates:
        blend = (1.0 - t) * f + t * direction
        blend = np.maximum(blend, 0.0)
        s = blend.sum()
        if s < 1e-10:
            continue
        blend = blend / s
        obj = eval_local_max_b(blend)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_t = t
            best_blend = blend.copy()

    # If coarse search found something, refine around it
    if best_t > 0.01:
        t_lo = max(0.001, best_t * 0.8)
        t_hi = min(1.0, best_t * 1.2)
        for _ in range(3):
            t_mid = (t_lo + t_hi) / 2
            # Evaluate at t_lo, t_mid, t_hi region
            for t_try in [t_lo * 0.9, t_lo, t_mid, t_hi * 0.9, t_hi]:
                if t_try <= 0 or t_try >= 1.0:
                    continue
                blend = (1.0 - t_try) * f + t_try * direction
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj = eval_local_max_b(blend)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_t = t_try
                    best_blend = blend.copy()
            # Narrow range
            best_blend_ts = [0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.5]
            best_local_t = best_t
            best_local_obj = best_obj
            for t_try in best_blend_ts:
                if abs(t_try - best_t) / max(best_t, 0.01) < 1.5:
                    blend = (1.0 - t_try) * f + t_try * direction
                    blend = np.maximum(blend, 0.0)
                    s = blend.sum()
                    if s < 1e-10:
                        continue
                    blend = blend / s
                    obj = eval_local_max_b(blend)
                    if obj < best_local_obj - 1e-14:
                        best_local_obj = obj
                        best_local_t = t_try
                        best_local_blend = blend.copy()
            t_lo = max(0.001, best_local_t * 0.9)
            t_hi = min(1.0, best_local_t * 1.1)
            best_obj = best_local_obj
            best_t = best_local_t
            best_blend = best_local_blend

    return best_blend, best_obj, best_t


def lp_optimize(f_init, time_limit, max_lp_iters=100, verbose=False):
    """
    LP-directed optimization: iteratively solve LP for flattest direction,
    then line search to blend. Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_local_max_b(a)

    if verbose:
        print(f"    LP start: n={n}, max_b={best_obj:.12f}")

    stall_count = 0
    lp_count = 0

    while time.monotonic() - t0 < time_limit:
        # Solve LP for flattest direction
        g, t_opt = lp_flattest_direction(a)

        if g is None:
            break

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Line search along this direction
        blend, blend_obj, t_used = lp_line_search(a, g, best_obj)

        if blend_obj < best_obj - 1e-14:
            best_obj = blend_obj
            best_a = blend.copy()
            a = blend.copy()
            stall_count = 0
            lp_count += 1
            if verbose and lp_count % 5 == 0:
                print(f"    LP iter {lp_count}: t={t_used:.4f}, max_b={best_obj:.12f}")
        else:
            stall_count += 1
            if stall_count >= 5:
                # Try the max-direction LP instead
                M = best_obj * 1.01
                g2, g2_sum = lp_max_direction(a, M)
                if g2 is not None and g2_sum > 1e-10:
                    g2 = g2 / g2_sum
                    blend2, blend_obj2, t_used2 = lp_line_search(a, g2, best_obj)
                    if blend_obj2 < best_obj - 1e-14:
                        best_obj = blend_obj2
                        best_a = blend2.copy()
                        a = blend2.copy()
                        stall_count = 0
                        lp_count += 1
                    else:
                        # Stuck - do some SG steps
                        rem = time_limit - (time.monotonic() - t0)
                        if rem > 5:
                            a_sg, obj_sg = sg_optimize(
                                a, time_limit=min(rem * 0.1, 30.0),
                                lr_init=0.002, momentum=0.94,
                                n_top=max(20, n // 8)
                            )
                            if obj_sg < best_obj:
                                best_obj = obj_sg
                                best_a = a_sg.copy()
                                a = a_sg.copy()
                        stall_count = 0

    return best_a, best_obj


def hybrid_optimize(f_init, time_limit, verbose=False):
    """
    Hybrid: alternate between LP (for direction) and SG (for fine-tuning).
    LP every K steps, SG in between.
    """
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_local_max_b(a)

    v = np.zeros(n)
    lr = 0.002
    momentum = 0.93
    n_top = max(20, n // 8)

    lp_interval = 50  # SG steps between LP calls
    sg_count = 0
    lp_count = 0
    stall_total = 0

    if verbose:
        print(f"  Hybrid start: n={n}, max_b={best_obj:.12f}")

    while time.monotonic() - t0 < time_limit:
        # Check if it's time for an LP step
        rem = time_limit - (time.monotonic() - t0)
        if sg_count >= lp_interval and rem > 10:
            sg_count = 0

            # Solve LP
            g, t_opt = lp_flattest_direction(a)
            if g is not None:
                g = np.maximum(g, 0.0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    blend, blend_obj, t_used = lp_line_search(a, g, best_obj)
                    if blend_obj < best_obj - 1e-14:
                        best_obj = blend_obj
                        best_a = blend.copy()
                        a = blend.copy()
                        stall_total = 0
                        lp_count += 1
                        if verbose and lp_count % 3 == 0:
                            print(f"  Hybrid LP#{lp_count}: t={t_used:.4f}, max_b={best_obj:.12f}")
                    else:
                        stall_total += 1
                        if stall_total >= 3:
                            # Perturb and reset
                            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                            v = np.zeros(n)
                            stall_total = 0
            else:
                stall_total += 1

        # SG step
        if time.monotonic() - t0 >= time_limit:
            break
        a_new, _ = subgradient_step(a, lr, v, momentum, n_top)
        obj_new = eval_local_max_b(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_total = max(0, stall_total - 1)
        else:
            stall_total += 1

        a = a_new
        sg_count += 1

        # Adaptive LR
        if stall_total > 200:
            lr *= 0.9
            if lr < 0.0001:
                lr = 0.002
                v = np.zeros(n)
                a = project_simplex(best_a + np.random.default_rng(42).standard_normal(n) * 0.003)
                stall_total = 0
        elif stall_total > 50:
            lr *= 0.97

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    """Gaussian window initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    """Tukey window initialization."""
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_powerlaw_init(n, alpha=0.5):
    """Power-law tapering."""
    x = np.linspace(0, 1, n)
    a = (1 - np.abs(x - 0.5) * 2) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp
