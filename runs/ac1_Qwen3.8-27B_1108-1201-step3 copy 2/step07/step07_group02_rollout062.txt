import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Global cache for LP solver
_lp_solver = None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))

def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def lbgfs_softmax_full(a0, time_limit, T_start=100.0, T_end=2.0, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing on the full sequence."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = max(500, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp
        
        def jac(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters_per_stage, 'ftol': 1e-14, 
                                   'gtol': 1e-8, 'maxfun': iters_per_stage * 100, 'disp': False})
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def stochastic_local_search(a0, time_limit, n_steps=5000, rng=np.random.default_rng(42)):
    """Stochastic local search with simulated annealing acceptance."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    T_anneal = 0.001  # initial temperature for SA
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        T_anneal *= 0.999
        
        # Random perturbation
        a_trial = a.copy()
        n_perturb = rng.integers(1, 5)
        for _ in range(n_perturb):
            i = rng.integers(0, n)
            if a_trial[i] < 1e-14:
                continue
            factor = rng.uniform(0.8, 1.2)
            a_trial[i] *= factor
        
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        
        sc = compute_score(a_trial)
        
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        elif T_anneal > 1e-8 and rng.random() < np.exp(-(sc - best_score) / T_anneal):
            a = a_trial
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def peak_shave_global(a0, time_limit, n_steps=10000):

            top_contrib = np.argsort(contribs)[::-1][:6]

            for ci in top_contrib:
                i = idx[ci]
                j = k_star - i
                if a[i] < 1e-14:
                    continue
                for factor in [0.99, 0.97, 0.94, 0.90]:
                    new_val = a[i] * factor
                    diff = a[i] - new_val
                    if diff < 1e-16:
                        continue
                    # Redistribute to positions that are "cold" (low contribution to all peaks)
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue

                    # Weight redistribution: put more mass where it contributes less
                    # Approximate: use 1/(local convolution contribution)
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try reducing multiple top contributors simultaneously
            k_star = int(np.argmax(b))
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k_star - idx]

def stochastic_refine(a0, time_limit, rng, n_steps=5000):
    """Stochastic local search with structured perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Structured perturbation: pick a random block and smooth/reshape it
        block_size = rng.integers(2, min(20, n // 4))
        start = rng.integers(0, n - block_size)
        end = start + block_size

        a_trial = a.copy()
        block = a_trial[start:end].copy()
        block_sum = block.sum()
        if block_sum < 1e-15:
            continue

        # Random reshape of the block
        mode = rng.integers(0, 3)
        if mode == 0:
            # Flatten
            a_trial[start:end] = block_sum / block_size
        elif mode == 1:
            # Gaussian bump
            x = np.linspace(-1, 1, block_size)
            w = np.exp(-2 * x * x)
            a_trial[start:end] = block_sum * w / w.sum()
        else:
            # Random
            w = rng.dirichlet(np.ones(block_size) * 0.5)
            a_trial[start:end] = block_sum * w

        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s

        sc = compute_score(a_trial)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial

    return best_a, best_score

        try:
            a_lb2 = lbgfs_softmax(best_a, t3, T_start=3.0, T_end=0.3, maxiter=4000)
            try_update(a_lb2, "lbgfs_lowT")
        except Exception as e:
            print(f"      Error: {e}")

    # Step 4: Peak shaving
    rem = time_limit - (time.monotonic() - t0)
    if rem > 20:
        t4 = rem * 0.30
        print(f"    [4/5] Peak shaving (n={n}, budget={t4:.0f}s)")
        try:
            a_ps, _ = peak_shave_global(best_a, t4, n_steps=5000)
            try_update(a_ps, "peak_shave")
        except Exception as e:
            print(f"      Error: {e}")

    # Step 5: Stochastic + coordinate
    rem = time_limit - (time.monotonic() - t0)
    if rem > 15:
        t5a = rem * 0.5
        t5b = rem * 0.5
        print(f"    [5/5] Stochastic+Coord (n={n}, budget={t5a:.0f}s + {t5b:.0f}s)")
        try:
            a_st, _ = stochastic_refine(best_a, t5a, rng, n_steps=3000)
            try_update(a_st, "stochastic")
        except Exception as e:
            print(f"      Error: {e}")
        rem = time_limit - (time.monotonic() - t0)
        if rem > 10:
            try:
                a_cd, _ = coordinate_refine(best_a, rem * 0.8, n_passes=20)
                try_update(a_cd, "coord")
            except Exception as e:
                print(f"      Error: {e}")

    return best_a, best_score
