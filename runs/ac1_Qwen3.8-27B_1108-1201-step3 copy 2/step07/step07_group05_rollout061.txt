d(a*a)[0]/da[0] = 2*a0, d(a*a)[1]/da[0] = 2*a1, d(a*a)[2]/da[0] = 2*a2, d(a*a)[3]/da[0] = 0, d(a*a)[4]/da[0] = 0

So grad[0] = 2*(w[0]*a0 + w[1]*a1 + w[2]*a2) = 2*sum_j a[j]*w[0+j] ✓

grad[1] = 2*(w[1]*a0 + w[2]*a1 + w[3]*a2) = 2*sum_j a[j]*w[1+j] ✓

grad[2] = 2*(w[2]*a0 + w[3]*a1 + w[4]*a2) = 2*sum_j a[j]*w[2+j] ✓

So grad_a[i] = 2 * sum_j a[j] * w[i+j]

This is the "sliding dot product" of a with windows of w. I can compute it as:
- Create a matrix W where W[i,j] = w[i+j] - this is a Hankel matrix
- grad_a = 2 * W @ a

But for large n, constructing the matrix is expensive. Using FFT:
- grad_a[i] = 2 * sum_j a[j] * w[i+j]
- Let b = a (length n), and we want to compute for each i: sum_j b[j] * w[i+j]
- This is the cross-correlation of b with w, evaluated at lags 0, 1, ..., n-1
- cross_correlation(b, w)[lag] = sum_j b[j] * w[j+lag]
- So grad_a[i] = 2 * cross_correlation(b, w)[i]

Using FFT for cross-correlation:
cross_correlation(b, w) = ifft(conj(fft(b)) * fft(w))

Let me use this:

```python
def compute_soft_max_gradient(a, beta, n):
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    w = np.exp(shifted)
    w = w / w.sum()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # cross_correlation(a, w)[i] = sum_j a[j] * w[j+i]
    # = ifft(conj(A) * W) ... let me verify
    # ifft(conj(A) * W)[k] = sum_j a[j] * w[(k-j) mod fft_size]
    # We want sum_j a[j] * w[j+i]
    # = sum_j a[j] * w[(n-1-j) + (j+i)] ... hmm
    # Let me use: cross_corr[k] = sum_j a[j] * w[k+j]
    # = ifft(conj(fft(a_rev)) * fft(w))[n-1+k] where a_rev = a[::-1]
    
    a_rev = a[::-1]
    Ar = rfft(a_rev, fft_size)
    conv_res = irfft(Ar * W, fft_size)
    # conv_res[n-1+i] = sum_j a_rev[j] * w[n-1+i-j] = sum_j a[n-1-j] * w[n-1+i-j]
    # Let j' = n-1-j: = sum_{j'} a[j'] * w[i+j']
    # Yes! That's what we want.
    grad_a = 2.0 * conv_res[n-1 : 2*n-1]
    
    return (2.0 * n) * grad_a
```

Now let me write the full algorithm. I'll structure it as:

1. **Phase 0**: Get parent, initialize
2. **Phase 1**: L-BFGS with annealed beta (start moderate, go high)
3. **Phase 2**: Proper coordinate descent with golden section search
4. **Phase 3**: Subgradient/active-set method
5. **Phase 4**: Final high-precision L-BFGS
6. **Phase 5**: Try different lengths

For the coordinate descent with golden section:
- For each pair (i, j), we want to find the optimal t that minimizes:
  f(t) = score((a - t*e_i + t*e_j) / sum(...))
  Since the score is scale-invariant, we can just work with the direction.
  Actually, we need to be careful: if we transfer t from i to j, the sum changes unless we renormalize.
  
  Better: fix sum(a) = 1. Then transferring t from i to j gives:
  a' = a - t*e_i + t*e_j (sum is still 1)
  We need a'[i] = a[i] - t >= 0, so t <= a[i]
  And a'[j] = a[j] + t >= 0 (always true if t > 0 and a[j] >= 0)
  
  So we search for t in [0, a[i]] (or [-a[j], a[i]] if we allow negative t).
  
  The function f(t) = 2n * max_k (conv(a - t*e_i + t*e_j))_k
  is piecewise linear in t (since conv is quadratic in a, but the max of quadratics is piecewise... actually conv is quadratic in a, so conv_k is quadratic in t, and max of quadratics is not piecewise linear but piecewise quadratic).
  
  For golden section search, we just need the function to be unimodal (or at least have a single minimum). It might not be perfectly unimodal, but in practice it should work well enough.

Let me now write the complete solution:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Annealed log-sum-exp smoothed L-BFGS optimization
2. Proper 1D coordinate descent with golden section search along coordinate pairs
3. Active-peak subgradient descent
4. Multi-start from diverse initializations
Key improvements over previous: proper scalar optimization in CD, 
better smoothing schedule, active-set focused subgradient steps.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def lse_smooth_obj_and_grad(a, beta, n):
    """
    Returns (objective, gradient) for beta-smoothed max of conv(a,a).
    objective = 2n * (1/beta) * log(sum(exp(beta * conv_k)))
    gradient = 2n * sum_k w_k * d(conv_k)/da
    where w_k = softmax(beta * conv_k)
    """
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = 2.0 * n * (max_c + np.log(exp_shifted.sum()) / beta)
    
    # Gradient: grad_a[i] = 2n * 2 * sum_j a[j] * w[i+j]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a_rev = a[::-1]
    Ar = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv_res = irfft(Ar * W, fft_size)
    grad_a = 2.0 * conv_res[n-1 : 2*n-1]
    
    return obj, (2.0 * n) * grad_a

def lbgfs_optimize_lse(a0, time_limit, beta_start, beta_end, maxiter=10000):
    """
    L-BFGS with annealed LSE smoothing.
    Starts with beta_start (smoother) and anneals to beta_end (sharper).
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Anneal beta over the optimization
    n_anneal_steps = 5
    betas = np.geomspace(beta_start, beta_end, n_anneal_steps)
    
    for beta in betas:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        time_per_beta = remaining / (n_anneal_steps - 1) if n_anneal_steps > 1 else remaining
        
        def func(x):
            obj, _ = lse_smooth_obj_and_grad(x, beta, n)
            return obj
        
        def jac(x):
            _, g = lse_smooth_obj_and_grad(x, beta, n)
            return g
        
        try:
            res = minimize(
                fun=func, x0=a, jac=jac,
                method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': max(100, int(maxiter / n_anneal_steps)),
                         'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 100000, 'disp': False}
            )
            a_new = np.maximum(res.x, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
    
    return best_a, best_score

def coordinate_descent_golden(a0, time_limit, verbose=False):
    """
    Coordinate descent with proper 1D golden section search.
    For each random pair (i,j), find optimal transfer t via golden section.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    no_improve_count = 0
    
    # Golden section ratio
    gr = (np.sqrt(5) - 1) / 2  # ~0.618
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Try multiple random pairs per iteration
        n_pairs = min(n, 50)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Search for optimal t in [-a[j], a[i]]
            # f(t) = score(a + t*(e_j - e_i))
            lo = -a[j] * 0.999
            hi = a[i] * 0.999
            
            if hi - lo < 1e-15:
                continue
            
            def f_of_t(t):
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                if a_trial[i] < 0 or a_trial[j] < 0:
                    return float('inf')
                return compute_score(a_trial)
            
            # Quick check: is the current point (t=0) already good?
            f0 = best_score
            
            # Coarse scan with 5 points
            scan_points = np.linspace(lo, hi, 5)
            best_scan_t = 0
            best_scan_f = f0
            for sp in scan_points:
                f_sp = f_of_t(sp)
                if f_sp < best_scan_f:
                    best_scan_f = f_sp
                    best_scan_t = sp
            
            if best_scan_f < f0 - 1e-14:
                # Refine with golden section around the best scan point
                idx = np.argmin([f_of_t(sp) for sp in scan_points])
                # Actually let's just do golden section on [lo, hi]
                a_gs = lo
                b_gs = hi
                c_gs = b_gs - gr * (b_gs - a_gs)
                d_gs = a_gs + gr * (b_gs - a_gs)
                fc = f_of_t(c_gs)
                fd = f_of_t(d_gs)
                
                for _ in range(30):
                    if b_gs - a_gs < 1e-12 * max(1.0, abs(lo)):
                        break
                    if fc < fd:
                        b_gs = d_gs
                        d_gs = c_gs
                        fd = fc
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        fc = f_of_t(c_gs)
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        fc = fd
                        d_gs = a_gs + gr * (b_gs - a_gs)
                        fd = f_of_t(d_gs)
                
                t_opt = (a_gs + b_gs) / 2
                f_opt = f_of_t(t_opt)
                
                if f_opt < best_score - 1e-14:
                    a[i] -= t_opt
                    a[j] += t_opt
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss > 1e-15:
                        a = a / ss
                    best_score = f_opt
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            no_improve_count += 1
            if no_improve_count > 20:
                # Perturb to escape local minimum
                noise_scale = 0.005 / np.sqrt(n)
                a = best_a + rng.standard_normal(n) * noise_scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve_count = 0
        else:
            no_improve_count = 0
        
        if verbose and iters % 100 == 0:
            print(f"  CD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score

def active_peak_subgradient(a0, time_limit, n_peaks=20, lr_init=0.1, verbose=False):
    """
    Subgradient descent focusing on active (top) peaks.
    At each step, compute subgradient from top peaks and take a step.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find active peaks (within tolerance of max)
        threshold = max_val * 0.999
        active_mask = conv >= threshold
        
        # Compute subgradient: grad[i] = 2 * sum_{k active} a[k-i]
        # (with bounds: 0 <= k-i < n)
        grad = np.zeros(n)
        active_indices = np.where(active_mask)[0]
        
        for k in active_indices:
            # grad[i] += 2 * a[k-i] for valid i
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                grad[i] += 2.0 * a[k - i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-30:
            break
        
        # Projected subgradient step: minimize grad^T * (a_new - a) over simplex
        # Frank-Wolfe: step towards e_j where j = argmin(grad)
        # Or: regularized step: a_new = a - lr * grad, project to simplex
        a_trial = a - lr * grad / grad_norm
        # Project to simplex (a >= 0, sum(a) = 1)
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            break
        a_trial = a_trial / ss
        
        sc = compute_score(a_trial)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            lr *= 1.01  # slightly increase if making progress
        else:
            lr *= 0.95  # decrease if not
        
        # Adaptive lr
        lr = np.clip(lr, 0.001, 0.5)
        
        if verbose:
            print(f"  APSG: score={best_score:.8f}, lr={lr:.4f}")
    
    return best_a, best_score

def multi_pair_cd(a0, time_limit, verbose=False):
    """
    More aggressive coordinate descent: try all pairs in a smart order.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    # Precompute: which pairs are most "connected" (same residue class mod something)
    # For now, just do random pairs but with more thorough search
    
    while time.monotonic() - t0 < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] < 1e-15:
            continue
        
        # Golden section search for optimal transfer
        lo = -a[j] * 0.9999
        hi = a[i] * 0.9999
        
        if hi - lo < 1e-14:
            continue
        
        gr = (np.sqrt(5) - 1) / 2
        a_gs = lo
        b_gs = hi
        c_gs = b_gs - gr * (b_gs - a_gs)
        d_gs = a_gs + gr * (b_gs - a_gs)
        
        def f_of_t(t):
            ai = a[i] - t
            aj = a[j] + t
            if ai < -1e-15 or aj < -1e-15:
                return float('inf')
            a_trial = a.copy()
            a_trial[i] = max(0, ai)
            a_trial[j] = max(0, aj)
            ss = a_trial.sum()
            if ss < 1e-15:
                return float('inf')
            return compute_score(a_trial / ss)
        
        fc = f_of_t(c_gs)
        fd = f_of_t(d_gs)
        
        for _ in range(25):
            if b_gs - a_gs < 1e-13:
                break
            if fc < fd:
                b_gs = d_gs
                d_gs = c_gs
                fd = fc
                c_gs = b_gs - gr * (b_gs - a_gs)
                fc = f_of_t(c_gs)
            else:
                a_gs = c_gs
                c_gs = d_gs
                fc = fd
                d_gs = a_gs + gr * (b_gs - a_gs)
                fd = f_of_t(d_gs)
        
        t_opt = (a_gs + b_gs) / 2
        f_opt = f_of_t(t_opt)
        
        if f_opt < best_score - 1e-15:
            a[i] -= t_opt
            a[j] += t_opt
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            best_score = f_opt
            best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Annealed log-sum-exp (LSE) smoothed L-BFGS in log-space (softmax parameterization)
2. Proper 1D golden section coordinate descent along coordinate pairs
3. Active-peak focused subgradient steps
4. Multi-length exploration with interpolation
5. High-precision final polish with very high beta
Key improvements: verified gradient computation, log-space parameterization
handles non-negativity/normalization automatically, proper scalar optimization.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)

def lse_obj_and_grad_a(a, beta, n):
    """
    Compute LSE-smoothed objective and gradient w.r.t. a.
    F(a) = 2n * (1/beta) * log(sum_k exp(beta * conv_k))
    grad_a[i] = 4n * sum_m a[m] * w[i+m]
    where w = softmax(beta * conv)
    """
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = 2.0 * n * (max_c + np.log(exp_shifted.sum()) / beta)

    # Gradient: grad_a[i] = 4n * cross_correlation(a, w)[i]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    Ar = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv_res = irfft(Ar * W, fft_size)
    grad_a = 4.0 * n * conv_res[n - 1:2 * n - 1]

    return obj, grad_a

def lse_obj_and_grad_z(z, beta, n):
    """
    LSE-smoothed objective and gradient in log-space (z = log(a) + const).
    grad_z = a * (grad_a - sum(a * grad_a))
    """
    a = softmax_from_z(z)
    obj, grad_a = lse_obj_and_grad_a(a, beta, n)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lbgfs_logspace(a0, time_limit, beta_start, beta_end, n_anneal=5, maxiter_per_beta=3000):
    """
    L-BFGS in log-space with annealed LSE smoothing.
    """
    n = len(a0)
    z = z_from_a(a0)
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()

    betas = np.geomspace(beta_start, beta_end, n_anneal)

    for beta in betas:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        def func(z_x):
            obj, _ = lse_obj_and_grad_z(z_x, beta, n)
            return obj

        def jac(z_x):
            _, g = lse_obj_and_grad_z(z_x, beta, n)
            return g

        try:
            result = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter_per_beta * 50,
                         'disp': False}
            )
            a_new = softmax_from_z(result.x)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss > 1e-15:
                a_new = a_new / ss
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                z = result.x
        except Exception:
            pass

    return best_a, best_score

def golden_section_cd(a0, time_limit, n_pairs_per_batch=30, batch_iterations=20, verbose=False):
    """
    Coordinate descent with golden section 1D search on random pairs.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    iter_count = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        improved = False

        for _ in range(n_pairs_per_batch):
            if time.monotonic() - t0 > time_limit:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            lo = -a[j] * 0.9999
            hi = a[i] * 0.9999
            if hi - lo < 1e-14:
                continue

            # Quick rejection: check a few points before doing full search
            test_t = (hi - lo) * 0.5
            a_test = a.copy()
            a_test[i] = max(0.0, a[i] - test_t)
            a_test[j] = a[j] + test_t
            ss_t = a_test.sum()
            if ss_t < 1e-15:
                continue
            f_test = compute_score(a_test / ss_t)
            if f_test > best_score * 1.01 and test_t > 0:
                # Likely not helpful, skip
                continue

            # Golden section search
            a_gs = lo
            b_gs = hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def f_t(t_val):
                ai = a[i] - t_val
                aj = a[j] + t_val
                if ai < -1e-12 or aj < -1e-12:
                    return float('inf')
                a_trial = a.copy()
                a_trial[i] = max(0.0, ai)
                a_trial[j] = max(0.0, aj)
                ss = a_trial.sum()
                if ss < 1e-12:
                    return float('inf')
                return compute_score(a_trial / ss)

            fc = f_t(c_gs)
            fd = f_t(d_gs)

            for _ in range(35):
                if b_gs - a_gs < 1e-12:
                    break
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = f_t(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = f_t(d_gs)

            t_opt = (a_gs + b_gs) / 2.0
            f_opt = f_t(t_opt)

            if f_opt < best_score - 1e-14:
                a[i] = max(0.0, a[i] - t_opt)
                a[j] = max(0.0, a[j] + t_opt)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                best_score = f_opt
                best_a = a.copy()

            if stagnation > batch_iterations:
                noise_scale = 0.003 / np.sqrt(n)

                a = np.maximum(a, 0)
                a = a / a.sum()
                stagnation = 0
        else:

                c_gs = d_gs
                fc = fd
                d_gs = a_gs + gr * (b_gs - a_gs)
                fd = f_t(d_gs)

        t_opt = (a_gs + b_gs) / 2.0
        f_opt = f_t(t_opt)

        if f_opt < best_score - 1e-14:
            a[i] = max(0.0, a[i] - t_opt)
            a[j] = max(0.0, a[j] + t_opt)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            best_score = f_opt
            best_a = a.copy()

    return best_a, best_score

def active_peak_refinement(a0, time_limit, verbose=False):
    """
    Focused refinement: identify top peaks, then do targeted coordinate moves
    on the indices that contribute most to those peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        # Find indices contributing to the top peaks
        top_k = min(15, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]

        # Collect important indices
        important = set()
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                if a[i] > 1e-10:
                    important.add(i)

        if not important:
            break

        imp_list = list(important)
        improved = False

        # Try moving mass from high-contribution indices to random others
        for _ in range(min(50, len(imp_list))):
            if time.monotonic() - t0 > time_limit:
                break

            src = imp_list[rng.integers(0, len(imp_list))]
            dst = rng.integers(0, n)
            if src == dst or a[src] < 1e-14:
                continue

            # Small golden section search
            lo = -a[dst] * 0.3
            hi = a[src] * 0.3

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_gs = lo
            b_gs = hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def f_t(t_val):
                ai = a[src] - t_val
                aj = a[dst] + t_val
                if ai < -1e-12 or aj < -1e-12:
                    return float('inf')
                a_trial = a.copy()
                a_trial[src] = max(0.0, ai)
                a_trial[dst] = max(0.0, aj)
                ss = a_trial.sum()
                if ss < 1e-12:
                    return float('inf')
                return compute_score(a_trial / ss)

            fc = f_t(c_gs)
            fd = f_t(d_gs)
            for _ in range(20):
                if b_gs - a_gs < 1e-13:
                    break
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = f_t(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = f_t(d_gs)

            t_opt = (a_gs + b_gs) / 2.0
            f_opt = f_t(t_opt)

            if f_opt < best_score - 1e-14:
                a[src] = max(0.0, a[src] - t_opt)
                a[dst] = max(0.0, a[dst] + t_opt)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                best_score = f_opt
                best_a = a.copy()
                improved = True

        if verbose:
            print(f"    APR: score={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS log-space with low-to-mid beta annealing (find basin)
    Phase 2: Golden section coordinate descent (local refinement)
    Phase 3: L-BFGS with mid-to-high beta (sharpen)
    Phase 4: Golden section CD again
    Phase 5: Active peak refinement
    Phase 6: Try nearby lengths
    Phase 7: Final high-beta L-BFGS + CD polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)
    print(f"Starting optimization, n={n_parent}, initial score={best_score:.8f}")

    # === Phase 1: L-BFGS log-space, beta 30 -> 200 ===
    print(f"\n=== Phase 1: L-BFGS log-space beta 30->200 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem * 0.35, 250.0)
        print(f"  budget={t_alloc:.0f}s")
        a_new, sc_new = lbgfs_logspace(best_seq, t_alloc,
                                       beta_start=30, beta_end=200,
                                       n_anneal=4, maxiter_per_beta=4000)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  Improved to {best_score:.8f}")
        else:
            print(f"  No improvement (score={sc_new:.8f})")

    # === Phase 2: Golden section CD ===
    print(f"\n=== Phase 2: Golden Section CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        t_alloc = min(rem * 0.4, 200.0)
        print(f"  budget={t_alloc:.0f}s")
        a_new, sc_new = golden_section_cd(best_seq, t_alloc,
                                          n_pairs_per_batch=30, verbose=True)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  Improved to {best_score:.8f}")

    # === Phase 3: L-BFGS log-space, beta 500 -> 5000 ===
    print(f"\n=== Phase 3: L-BFGS log-space beta 500->5000 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 20:
        t_alloc = min(rem * 0.45, 200.0)
        print(f"  budget={t_alloc:.0f}s")
        a_new, sc_new = lbgfs_logspace(best_seq, t_alloc,
                                       beta_start=500, beta_end=5000,
                                       n_anneal=4, maxiter_per_beta=4000)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  Improved to {best_score:.8f}")

    # === Phase 4: Fast pair CD ===
    print(f"\n=== Phase 4: Fast Pair CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 15:
        t_alloc = min(rem * 0.5, 150.0)
        print(f"  budget={t_alloc:.0f}s")
        a_new, sc_new = fast_pair_cd(best_seq, t_alloc)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  Improved to {best_score:.8f}")

    # === Phase 5: Active peak refinement ===
    print(f"\n=== Phase 5: Active Peak Refinement (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 15:
        t_alloc = min(rem * 0.5, 120.0)
        print(f"  budget={t_alloc:.0f}s")
        a_new, sc_new = active_peak_refinement(best_seq, t_alloc)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  Improved to {best_score:.8f}")

    # === Phase 6: Nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.8f}) ===")
    nearby_ns = [n_parent + 2, n_parent - 2, n_parent + 4, n_parent - 4,
                 n_parent + 1, n_parent - 1]
    for alt_n in nearby_ns:
        if alt_n < 64 or alt_n > 2048:
            continue
        rem = deadline - time.monotonic() - 60
        if rem < 25:
            break
        t_alloc_lbfgs = min(rem * 0.35, 50.0)
        t_alloc_cd = min(rem * 0.25, 30.0)
        if t_alloc_lbfgs < 10:
            continue
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  n={alt_n}: L-BFGS budget={t_alloc_lbfgs:.0f}s")
        a_new, sc_new = lbgfs_logspace(init_alt, t_alloc_lbfgs,
                                       beta_start=200, beta_end=5000,
                                       n_anneal=3, maxiter_per_beta=3000)
        rem2 = deadline - time.monotonic() - 40
        if rem2 > 10:
            a_new2, sc_new2 = fast_pair_cd(a_new, t_alloc_cd)
            if sc_new2 < sc_new:
                a_new, sc_new = a_new2, sc_new
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  n={alt_n}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={alt_n}: score={sc_new:.8f} (no improvement)")

    # === Phase 7: Final high-beta L-BFGS ===
    print(f"\n=== Phase 7: Final High-Beta L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_alloc = min(rem * 0.5, 60.0)
        print(f"  budget={t_alloc:.0f}s")
        a_new, sc_new = lbgfs_logspace(best_seq, t_alloc,
                                       beta_start=5000, beta_end=50000,
                                       n_anneal=3, maxiter_per_beta=5000)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  Improved to {best_score:.8f}")

    # === Phase 8: Final CD polish ===
    rem = deadline - time.monotonic() - 15
    if rem > 5:
        t_alloc = min(rem * 0.8, 40.0)
        print(f"\n=== Phase 8: Final CD Polish (budget={t_alloc:.0f}s) ===")
        a_new, sc_new = fast_pair_cd(best_seq, t_alloc)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  Final improvement to {best_score:.8f}")

    # Final verification with evaluate_sequence
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: evaluate_sequence gives {fs:.8f}")
        print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
```
