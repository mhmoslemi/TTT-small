Our problem: `min_a max_k conv(a,a)_k` (with a >= 0, sum a = 1).

If we define the "game" as: Player 1 chooses a, payoff is `max_k conv(a,a)_k`. Player 2 doesn't exist. It's a single-player optimization.

The LP approach: given a fixed `a`, find `b` that minimizes `max_k conv(a,b)_k`. This is the "best response" in a related bilinear game.

The connection: if `a*` minimizes `max_k conv(a,a)_k`, then `a*` is also the best response to itself (i.e., `b*(a*) = a*`). Why? Because if there were a `b` with `max_k conv(a*,b)_k < max_k conv(a*,a*)_k`, then by convexity, we could improve `a*` by moving toward `b`.

Wait, that's not quite right because `conv(a,a)` is not linear in `a` (it's quadratic). But `conv(a,b)` is bilinear, so `conv(a, ta+(1-t)b) = t*conv(a,b) + (1-t)*conv(a,a)`. So moving from `a` toward `b` linearly interpolates the "cross-term."

If `b` minimizes `max_k conv(a,b)_k` and `max_k conv(a,b)_k < max_k conv(a,a)_k`, then for small t>0:
`max_k conv(a, ta+(1-t)b)_k = max_k [t*conv(a,b)_k + (1-t)*conv(a,a)_k]`
≤ `t*max_k conv(a,b)_k + (1-t)*max_k conv(a,a)_k`
< `max_k conv(a,a)_k`

But this is the cross-convolution, not the auto-convolution. We need `conv(ta+(1-t)b, ta+(1-t)b)`.

`conv(ta+(1-t)b, ta+(1-t)b)_k = t²*conv(a,a)_k + 2t(1-t)*conv(a,b)_k + (1-t)²*conv(b,b)_k`

This is more complex. The LP best response doesn't directly give us a descent direction for the auto-convolution.

However, for small t:
`conv(a+t(a-b), a+t(a-b))_k ≈ conv(a,a)_k + 2t*conv(a, a-b)_k = conv(a,a)_k + 2t*(conv(a,a)_k - conv(a,b)_k)`

So the first-order change in `conv(a,a)_k` when moving from `a` toward `b` is `2t*(conv(a,a)_k - conv(a,b)_k)`.

If `b` minimizes `max_k conv(a,b)_k`, and if the active set of `max_k conv(a,a)_k` is the same as `max_k conv(a,b)_k`, then moving toward `b` reduces the peak.

This gives us a principled descent direction! The LP best response gives us a direction that (at first order) reduces the peak.

So the algorithm is:
1. Given `a`, solve the LP to find `b` that minimizes `max_k conv(a,b)_k`.
2. Update: `a_new = (1-t)*a + t*b` for small t.
3. Repeat.

This is essentially a "Bregman-like" method where the "mirror" is the LN (which gives the LP as the mirror map's inverse).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a refined multi-phase strategy:
1. Multi-start initialization (parent, structured, spectral)
2. L-BFGS in log-space with geometric temperature annealing (coarse to fine)
3. LP best-response alternating optimization with Nesterov acceleration
4. Subgradient descent with exact max (unsmoothed) for final precision
5. Targeted coordinate descent on peak-contributing coordinates
Key improvements over prior approach: tighter temperature schedule, subgradient

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

_lp_solver_cached = None


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def lse_value_and_grad_a(a, T):
    """Compute LSE-smoothed objective and its gradient w.r.t. a (sum(a)=1)."""
    n = len(a)
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse = max_Tc + np.log(exp_Tc.sum())
    w = exp_Tc / exp_Tc.sum()
    value = (2.0 * n / T) * lse
    # grad[i] = 4n * sum_{m=0}^{n-1} w[i+m] * a[m]
    idx = np.arange(n)
    W_mat = w[idx[:, None] + idx[None, :]]
    grad = 4.0 * n * (W_mat @ a)
    return value, grad


def lse_z_value(z, T):
    """Objective in log-space z (a = softmax(z))."""
    a = softmax_from_z(z)
    val, _ = lse_value_and_grad_a(a, T)
    return val


def lse_z_grad(z, T):
    """Gradient of LSE objective w.r.t. z."""
    n = len(z)
    a = softmax_from_z(z)
    _, grad_a = lse_value_and_grad_a(a, T)
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lbgfs_optimize(a0, time_limit, T_start=100.0, T_end=1.0, n_stages=6, maxiter_per_stage=2000):
    """L-BFGS optimization in log-space with geometric temperature schedule."""
    n = len(a0)
    z = z_from_a(a0.copy())
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = maxiter_per_stage // n_stages

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def obj_func(zz):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            aa = softmax_from_z(zz)
            val, _ = lse_value_and_grad_a(aa, stage_T)
            return val

        def grad_func(zz):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zz)
            return lse_z_grad(zz, stage_T)

        try:
            res = minimize(
                fun=obj_func,
                x0=z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if int(stage_T) == stage_T:
            print(f"    L-BFGS T={int(stage_T)} done, elapsed={time.monotonic()-t0:.0f}s")

    a_out = softmax_from_z(z)
    return a_out


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_nesterov_refine(a0, time_limit, solver=None, n_max_iters=5000):
    """LP-based Nesterov dual averaging refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        score_a = compute_score(a)
        score_y = compute_score(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 100 == 0:
            print(f"    LP-Nesterov iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=100000):
    """Subgradient descent using exact max (no smoothing)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr_init = 0.05
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(a)
        max_c = c.max()

        # Find active set (near-max)
        threshold = max_c - 0.01 * max_c
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Compute subgradient: g[i] = (4n/|active|) * sum_{k in active} a[k-i] (valid)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 4.0 * n * a[k - idx]
        g /= len(active)

        # Gradient step with projection
        a_new = simplex_project(a - lr * g)
        score_new = compute_score(a_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
        lr = max(lr, 1e-8)

        if step % 10000 == 0:
            print(f"    SubGrad step={step} ({time.monotonic()-t0:.0f}s): score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, n_passes=200):
    """Coordinate descent focusing on coordinates that contribute to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(a)
        # Find top peaks and their contributing coordinates
        peak_indices = np.argsort(c)[::-1][:10]
        contrib_set = set()
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                if a[j] > 0.01 * a.max():
                    contrib_set.add(j)
        if len(contrib_set) == 0:
            contrib_set = set(range(n))
        # Limit to top 50 contributors for speed
        contrib_list = sorted(contrib_set, key=lambda j: -a[j])[:50]

        for i in contrib_list:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            base = a.copy()
            base[i] = 0
            base_sum = base.sum()
            if base_sum < 1e-15:
                continue
            best_t = a[i]
            best_local = best_score
            for factor in [0.75, 0.85, 0.92, 1.0, 1.08, 1.15, 1.25]:
                trial_t = a[i] * factor
                if trial_t < 0:
                    continue
                a_trial = base * (1.0 - trial_t) + trial_t * np.eye(n, dtype=np.float64)[i]
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_t = trial_t
            if best_t != a[i]:
                a = base * (1.0 - best_t) + best_t * np.eye(n, dtype=np.float64)[i]
                a = a / a.sum()
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if p % 20 == 0:
            print(f"    CoordDesc pass={p} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_shave_precise(a0, time_limit, n_steps=50000):
    """Precise peak shaving: reduce the top contributing coordinate to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(a)
        # Find the interior peak (avoid endpoints)
        c_inner = c.copy()
        if len(c_inner) > 2:
            c_inner[0] = -np.inf
            c_inner[-1] = -np.inf
        if c_inner.max() < 0:
            peak_k = int(np.argmax(c))
        else:
            peak_k = int(np.argmax(c_inner))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_ci = np.argsort(contribs)[::-1][:3]

        improved = False
        for ci in top_ci:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / os
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing two coordinates
            if len(top_ci) >= 2:
                i1 = idx[top_ci[0]]
                i2 = idx[top_ci[1]]
                if i1 != i2 and a[i2] > 1e-14:
                    for f in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i1] *= f
                        a_trial[i2] *= f
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            break
                        a_trial = a_trial / st
                        score = compute_score(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
            if not improved:
                # Try micro-reduction of all top contributors
                a_trial = a.copy()
                for ci in top_ci:
                    a_trial[idx[ci]] *= 0.98
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    score = compute_score(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                if not improved:
                    break

    return best_a, best_score


def cosine_init(n):
    """Create a cosine-windowed initial sequence."""
    j = np.arange(n)
    a = np.cos(np.pi * j / n) ** 2
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def spectral_init(n, rng):
    """Spectral initialization: try to flatten the periodogram."""
    N = 1 << (2 * n - 1).bit_length()
    for attempt in range(50):
        phases = rng.uniform(-np.pi, np.pi, N // 2 + 1)
        A = np.ones(N // 2 + 1) * np.exp(1j * phases)
        A[-1] = A[0]  # Ensure real result (conjugate symmetry)
        a_full = np.fft.irfft(A, N)
        a = np.maximum(a_full[:n], 0)
        s = a.sum()
        if s < 0.1 * n / N:
            continue
        a = a / s
        sc = compute_score(a)
        if sc < 1.8:
            return a
    # Fallback
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def fista_optimize(a0, time_limit, T=50.0, n_steps=50000):
    """FISTA (accelerated proximal gradient) on the simplex with LSE smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    y = a.copy()
    bk = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Estimate Lipschitz constant
    c0 = fast_convolve(a)
    L = 8.0 * n * c0.max() + 1.0  # Rough upper bound

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        # Compute gradient at y
        _, grad_y = lse_value_and_grad_a(y, T)
        # Gradient step
        a_new = simplex_project(y - grad_y / L)
        # FISTA update
        bk_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * bk * bk))
        y = a_new + ((bk - 1.0) / bk_new) * (a_new - a)
        y = simplex_project(y)
        a = a_new
        bk = bk_new

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Adapt L
        if step % 1000 == 0:
            L *= 0.95
            print(f"    FISTA step={step} ({time.monotonic()-t0:.0f}s): score={best_score:.6f} L={L:.2f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing the auto-convolution peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Multi-start coarse optimization (300s) ===
    print(f"\n=== Phase 1: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)

    if t_alloc > 30:
        # Try current best with L-BFGS
        print(f"  L-BFGS coarse on parent (n={n_best}), budget={t_alloc/2:.0f}s")
        a_c1 = lbgfs_optimize(best_seq, t_alloc * 0.5, T_start=200.0, T_end=20.0, n_stages=3, maxiter_per_stage=2000)
        try_update(a_c1, "coarse_parent")

        # Try cosine init at same n
        rem = deadline - time.monotonic()
        if rem > 20:
            a_cos = cosine_init(n_best)
            t_alloc2 = min(rem * 0.3, 100.0)
            if t_alloc2 > 10:
                print(f"  L-BFGS coarse on cosine init, budget={t_alloc2:.0f}s")
                a_c2 = lbgfs_optimize(a_cos, t_alloc2, T_start=100.0, T_end=10.0, n_stages=3, maxiter_per_stage=1000)
                try_update(a_c2, "coarse_cosine")

        # Try different n
        rem = deadline - time.monotonic()
        if rem > 30:
            n_alt = 1024
            alt_init = interpolate_sequence(best_seq, n_alt)
            t_alloc3 = min(rem * 0.4, 100.0)
            if t_alloc3 > 10:
                print(f"  L-BFGS coarse on n={n_alt}, budget={t_alloc3:.0f}s")
                a_c3 = lbgfs_optimize(alt_init, t_alloc3, T_start=100.0, T_end=10.0, n_stages=3, maxiter_per_stage=1000)
                try_update(a_c3, f"coarse_n{n_alt}")

    # === Phase 2: Intensive L-BFGS (500s) ===
    print(f"\n=== Phase 2: Intensive L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        n_cur = len(best_seq)
        print(f"  L-BFGS T: 50->2, n={n_cur}, budget={t_alloc:.0f}s")
        a_i1 = lbgfs_optimize(best_seq, t_alloc, T_start=50.0, T_end=2.0, n_stages=5, maxiter_per_stage=4000)
        try_update(a_i1, "lbgfs_intensive")

    # === Phase 3: LP Nesterov refinement (300s) ===
    print(f"\n=== Phase 3: LP Nesterov (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 30:
        n_cur = len(best_seq)
        print(f"  LP-Nesterov n={n_cur}, budget={t_alloc:.0f}s")
        a_lp, _ = lp_nesterov_refine(best_seq, t_alloc, solver=solver, n_max_iters=3000)
        try_update(a_lp, "lp_nesterov")

    # === Phase 4: Second L-BFGS at low T (250s) ===
    print(f"\n=== Phase 4: Low-T L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        n_cur = len(best_seq)
        print(f"  L-BFGS T: 5->0.5, n={n_cur}, budget={t_alloc:.0f}s")
        a_l1 = lbgfs_optimize(best_seq, t_alloc, T_start=5.0, T_end=0.5, n_stages=4, maxiter_per_stage=3000)
        try_update(a_l1, "lbgfs_lowT")

    # === Phase 5: FISTA optimization (150s) ===
    print(f"\n=== Phase 5: FISTA (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 150.0)
    if t_alloc > 15:
        n_cur = len(best_seq)
        print(f"  FISTA T=30, n={n_cur}, budget={t_alloc:.0f}s")
        a_f, _ = fista_optimize(best_seq, t_alloc, T=30.0)
        try_update(a_f, "fista")

    # === Phase 6: Subgradient polish with exact max (200s) ===
    print(f"\n=== Phase 6: Subgradient Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 10:
        n_cur = len(best_seq)
        print(f"  SubGrad n={n_cur}, budget={t_alloc:.0f}s")
        a_sg, _ = subgradient_polish(best_seq, t_alloc)
        try_update(a_sg, "subgrad")

    # === Phase 7: Targeted coordinate descent (150s) ===
    print(f"\n=== Phase 7: Coord Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        n_cur = len(best_seq)
        print(f"  CoordDesc n={n_cur}, budget={t_alloc:.0f}s")
        a_cd, _ = targeted_coord_descent(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")

    # === Phase 8: Peak shaving (150s) ===
    print(f"\n=== Phase 8: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        n_cur = len(best_seq)
        print(f"  PeakShave n={n_cur}, budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave_precise(best_seq, t_alloc)
        try_update(a_ps, "peak_shave")

    # === Phase 9: Final L-BFGS micro-polish (100s) ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_f1 = min(rem * 0.5, 60.0)
        if t_f1 > 10:
            a_f1 = lbgfs_optimize(best_seq, t_f1, T_start=2.0, T_end=0.3, n_stages=3, maxiter_per_stage=2000)
            try_update(a_f1, "final_lbgfs")

        rem = deadline - time.monotonic()
        if rem > 10:
            t_f2 = min(rem * 0.5, 40.0)
            if t_f2 > 5:
                a_f2, _ = peak_shave_precise(best_seq, t_f2, n_steps=20000)
                try_update(a_f2, "final_shave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
