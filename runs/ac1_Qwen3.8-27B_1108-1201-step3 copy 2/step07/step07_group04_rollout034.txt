def gradient_pressure(a, c=None, temp=0.1):
    """Compute the gradient of the soft-max smoothed objective.

    n = len(a)
    if c is None:

    c_max = c.max()
    scaled = (c - c_max) / (temp * c_max)
    w = np.exp(scaled)
    w = w / w.sum()

    pressure = np.correlate(w, a, mode='valid')
    return 2.0 * pressure

def project_onto_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_val = np.nonzero(cond)[0][-1]
        theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

    pressure[j] = 2 * sum_k w[k] * a[k-j] where w = softmax(c/(temp*max(c)))."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(len(a))
    scaled = (c - c_max) / (temp * c_max)
    w = np.exp(scaled)
    w = w / w.sum()
    # pressure[j] = sum_k w[k] * a[k-j] = correlate(w, a, 'valid')[j]
    pressure = np.correlate(w, a, mode='valid')
    return 2.0 * pressure


def project_onto_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_val = np.nonzero(cond)[0][-1]
        theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def accelerated_reweight(a0, time_limit, temp_init=0.3, temp_min=0.005,
                         lr_init=0.05, momentum_init=0.9, n_phases=4):
    """Accelerated iterative reweighting with Nesterov momentum and adaptive temperature."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = monotonic()

    for phase in range(n_phases):
        phase_budget = (time_limit - (monotonic() - t0)) / (n_phases - phase)
        temp = temp_init * (temp_min / temp_init) ** (phase / max(1, n_phases - 1))
        momentum = momentum_init * (0.7 ** phase)
        lr = lr_init / (1 + 0.5 * phase)
        velocity = np.zeros(n)
        no_improve = 0
        it = 0

        while (monotonic() - t0) < time_limit and (monotonic() - t0) <= (monotonic() - t0 - phase_budget + (time_limit - (monotonic() - t0)) + phase_budget):
            if monotonic() - t0 > time_limit:
                break
            it += 1
            # Bound per-phase time
            phase_elapsed = monotonic() - t0
            if phase_elapsed > time_limit:
                break

            c = fast_convolve(a)
            pressure = gradient_pressure(a, c, temp)

            # Nesterov step
            velocity = momentum * velocity - lr * pressure
            a_new = a + velocity
            a_new = project_onto_simplex(a_new)

            sc = compute_score(a_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 30:
                    # Backtrack: halve learning rate and perturb
                    lr = max(0.001, lr * 0.7)
                    momentum = min(0.95, momentum * 0.95)
                    # Structured perturbation using Fourier modes
                    k = no_improve // 10 + 1
                    freq = k * 2 * np.pi * np.arange(n) / n
                    pert = np.cos(freq) * 0.003 + np.sin(2 * freq) * 0.002
                    a = best_a + pert
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    velocity = np.zeros(n)
                    no_improve = 0

            # Early exit if converged
            c = fast_convolve(a)
            c_max = c.max()
            c_mean = c.mean()
            if c_max / c_mean < 1.001:
                break

        # Update for next phase
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if it % 100 == 0:
            elapsed = monotonic() - t0
            print(f"      Reweight phase={phase}, iter={it} ({elapsed:.0f}s): score={best_score:.6f}, T={temp:.4f}")

    return best_a, best_score


def fourier_perturb(a, strength=0.01, seed=None):
    """Apply structured perturbation in frequency domain."""
    if seed is not None:
        rng = np.random.RandomState(seed)
    else:
        rng = np.random
    n = len(a)
    A = rfft(a)
    # Add random phase perturbation to non-DC components
    phases = rng.uniform(-strength, strength, len(A))
    phases[0] = 0  # Keep DC component
    A_pert = A * np.exp(1j * phases)
    a_pert = irfft(A_pert, n)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def active_set_optimize(a0, time_limit, max_iters=500):
    """Optimize using active-set peak targeting with Newton-like steps."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = monotonic()

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Identify active peaks (top few)
        n_active = min(20, max(5, n // 20))
        active_idx = np.argsort(c)[::-1][:n_active]

        # Build gradient at active peaks
        # grad_j = sum_{k in active} w_k * 2*a[k-j]
        # Use inverse-c variance weighting
        active_vals = c[active_idx]
        weights = np.zeros(len(c))
        # Inverse of value: peaks get less weight (we want to reduce them)
        # Actually, we want to REDUCE the peaks, so weight proportional to value
        w_norm = active_vals / active_vals.sum()

        # Compute pressure only from active peaks
        pressure = np.zeros(n)
        for i, k in enumerate(active_idx):
            if i >= len(w_norm):
                break
            # pressure += w[k] * 2*a[k-j]
            # a[k-j] for j=0..n-1: valid when 0<=k-j<n, i.e., k-n+1<=j<=k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                pressure[js:je+1] += w_norm[i] * 2.0 * a[je - (je - js):je+1][::-1]

        # Wait, let me redo this. pressure[j] = sum_k w[k] * 2*a[k-j]
        # For a specific k: a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        # a[k-j] gives the sequence a[k], a[k-1], ..., a[k-(je-js)]
        # So pressure[j] += w[k] * 2*a[k-j] for j in [k-n+1, k]
        # Let me just use correlate for efficiency

        # Actually, let's use the full correlate approach but only with active weights
        w_full = np.zeros(len(c))
        w_full[active_idx] = w_norm
        pressure = 2.0 * np.correlate(w_full, a, mode='valid')

        # Newton-like step: a_new = a - step * pressure, projected
        p_norm = np.sqrt(np.sum(pressure ** 2))
        if p_norm < 1e-15:
            break
        step = 0.1 / p_norm

        for factor in [1.0, 0.5, 0.25, 0.1]:
            a_trial = a - step * factor * pressure
            a_trial = project_onto_simplex(a_trial)
            sc = compute_score(a_trial)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                break
        else:
            # No improvement, try perturbation
            if (it + 1) % 10 == 0:
                a = fourier_perturb(best_a, strength=0.02, seed=it)
                a = a / a.sum()

        if (it + 1) % 50 == 0:
            elapsed = monotonic() - t0
            print(f"    ActiveSet iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=50, maxiter=3000):
    """L-BFGS refinement with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if stopped[0]:
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        c = fast_convolve(a)
        c_max = c.max()
        scaled_c = (c - c_max) / T
        exp_c = np.exp(scaled_c)
        lse = c_max + T * np.log(exp_c.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if stopped[0]:
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        c = fast_convolve(a)
        c_max = c.max()
        scaled_c = (c - c_max) / T
        exp_c = np.exp(scaled_c)
        w = exp_c / exp_c.sum()

        # Gradient: 4n * correlate(w, a, 'valid')
        grad_a = 2.0 * np.correlate(w, a, mode='valid')
        grad_a *= 4.0 * n / (2.0 * n)  # = 2n * 2*correlate = 4n*correlate... let me be careful

        # f = 2n * lse/T
        # dlse/da[j] = sum_k w[k] * dc[k]/da[j] = sum_k w[k] * 2*a[k-j] = 2*correlate(w,a)[j]
        # df/da[j] = 2n/T * 2*correlate(w,a)[j] = 4n/T * correlate(w,a)[j]
        grad_a = (4.0 * n / T) * np.correlate(w, a, mode='valid')

        # Transform to z-space: a = softmax(z) (unnormalized exp)
        # da[j]/dz[i] = a[j]*(delta[j,i] - a[i])... wait
        # Actually a[j] = exp(z[j]) / sum(exp(z))
        # da[j]/dz[i] = a[j] * (delta[j,i] - a[i])
        # grad_z[i] = sum_j grad_a[j] * da[j]/dz[i] = sum_j grad_a[j] * a[j] * (delta[j,i] - a[i])
        # = a[i] * sum_j grad_a[j]*a[j] - a[i]^2 * ... wait
        # = a[i] * (grad_a dot a) - a[i] * sum_j grad_a[j]*a[j]
        # Hmm let me redo: grad_z[i] = sum_j grad_a[j] * a[j]*(delta[j,i] - a[i])
        # = grad_a[i]*a[i] - a[i]*sum_j grad_a[j]*a[j]
        # = a[i] * (grad_a[i] - sum_j grad_a[j]*a[j])
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return grad_z

    try:
        if monotonic() - t0 > time_limit:
            return a0
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                     'maxfun': maxiter * 20}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out


def lp_refine(a0, time_limit, max_iters=200):
    """LP-based refinement using cvxpy. Solves best-response LP iteratively."""
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = monotonic()

    # Try to find a working solver
    solver = None
    for s_name in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s_name
                break
        except Exception:
            continue
    if solver is None:
        return best_a, best_score

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break

        # Solve LP: minimize M subject to conv(a,x) <= M, sum(x)=1, x>=0
        x_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a
        constraints = [T @ x_var <= M_var, cp.sum(x_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        x_val = np.asarray(x_var.value, dtype=np.float64)
        x_val = np.maximum(x_val, 0)
        s_val = x_val.sum()
        if s_val < 1e-15:
            break
        x_val = x_val / s_val

        # Damped update
        alpha = min(1.0, 2.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * x_val
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if (it + 1) % 30 == 0:
            elapsed = monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def make_initialization(n, method='uniform', seed=42):
    """Create various initializations."""
    rng = np.random.RandomState(seed)
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'triangular':
        a = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float) + 0.1
    elif method == 'gaussian':
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.5) ** 2) + 0.01
    elif method == 'hann':
        a = np.hanning(n) + 0.01
    elif method == 'blackman':
        a = np.blackman(n) + 0.01
    elif method == 'raised_cosine':
        x = np.linspace(0, 2 * np.pi, n)
        a = (1 + np.cos(x)) / 2 + 0.01
    elif method == 'random':
        a = rng.exponential(1.0, n)
        a = a.max() - a + 0.1  # Inverted: peaks at random positions
    elif method == 'spectral_flat':
        # Try to have flat power spectrum
        A = np.ones(next_fast_len(2 * n - 1)) + 0.1 * rng.randn(next_fast_len(2 * n - 1))
        A = A + 1j * 0.1 * rng.randn(next_fast_len(2 * n - 1))
        A[0] = np.abs(A[0])
        a = irfft(A)[:n]
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a


def peak_shave_fine(a0, time_limit):
    """Fine-grained peak shaving with pairwise adjustments."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = monotonic()
    failed = 0

    while monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        c_peak = c[peak_k]

        if c_peak * 2 * n <= best_score * (1 + 1e-10):
            break

        # Find top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-20:
            break

        top_n = min(8, len(contribs))
        top_contrib = np.argsort(contribs)[::-1][:top_n]
        improved = False

        # Try reducing individual contributors
        for ci in top_contrib:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-12:
                continue
            for factor in [0.98, 0.95, 0.9, 0.8]:
                new_val = old_val * factor
                diff = old_val - new_val
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to maintain sum
                a_trial[1:] += diff * a_trial[1:] / a_trial[1:].sum() if a_trial[1:].sum() > 1e-15 else 0
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try pairwise reduction
        if not improved:
            for ci in top_contrib[:4]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or j < 0 or j >= n or a[j] < 1e-12:
                    continue
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            failed += 1
            if failed > 30:
                break
            # Small perturbation
            a = a + np.random.standard_normal(n) * 0.0005
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            failed = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Diverse initialization with quick screening
    2. Accelerated reweighting with Nesterov + adaptive temperature
    3. LP-based best response refinement
    4. L-BFGS polish with multiple temperatures
    5. Fine peak shaving
    6. Final multi-restart exploration
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Diverse initialization (60s) ===
    print(f"\n=== Phase 1: Diverse Initialization ===")
    t0 = monotonic()
    phase1_budget = min(60.0, (deadline - monotonic()) * 0.1)

    init_methods = ['uniform', 'hann', 'blackman', 'raised_cosine',
                    'triangular', 'gaussian', 'spectral_flat']
    candidates = [best_seq.copy()]

    for method in init_methods:
        if monotonic() - t0 > phase1_budget:
            break
        a_init = make_initialization(n_best, method=method, seed=seed)
        # Quick reweighting for 3 seconds
        a_quick, _ = accelerated_reweight(a_init, 3.0, n_phases=2)
        candidates.append(a_quick)

    # Also try different lengths
    for n_try in [512, 1024]:
        if monotonic() - t0 > phase1_budget:
            break
        if n_try == n_best:
            continue
        a_interp = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
        a_interp = np.maximum(a_interp, 0)
        a_interp = a_interp / a_interp.sum()
        a_quick, _ = accelerated_reweight(a_interp, 3.0, n_phases=2)
        candidates.append(a_quick)

    for i, cand in enumerate(candidates):
        check_and_update(cand, f"init_{i}")

    # === Phase 2: Main accelerated reweighting (400s) ===
    print(f"\n=== Phase 2: Accelerated Reweighting (best={best_score:.6f}) ===")
    t0 = monotonic()
    phase2_budget = min(400.0, (deadline - monotonic()) * 0.35)

    a_rw, _ = accelerated_reweight(best_seq, phase2_budget,
                                   temp_init=0.3, temp_min=0.003,
                                   lr_init=0.05, momentum_init=0.9, n_phases=5)
    check_and_update(a_rw, "reweight_main")

    # === Phase 3: LP refinement (250s) ===
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.6f}) ===")
    t0 = monotonic()
    phase3_budget = min(250.0, (deadline - monotonic()) * 0.25)

    if phase3_budget > 30:
        a_lp, _ = lp_refine(best_seq, phase3_budget, max_iters=300)
        check_and_update(a_lp, "lp_refine")

        # L-BFGS polish after LP
        if (deadline - monotonic()) > 60:
            a_lb = lbgfs_polish(best_seq, min(60.0, (deadline - monotonic()) * 0.3), T=50)
            check_and_update(a_lb, "lbgfs_after_lp")

    # === Phase 4: More reweighting with different params (200s) ===
    print(f"\n=== Phase 4: Reweighting v2 (best={best_score:.6f}) ===")
    t0 = monotonic()
    phase4_budget = min(200.0, (deadline - monotonic()) * 0.2)

    if phase4_budget > 30:
        # Lower temperature, more momentum
        a_rw2, _ = accelerated_reweight(best_seq, phase4_budget,
                                        temp_init=0.1, temp_min=0.002,
                                        lr_init=0.03, momentum_init=0.92, n_phases=4)
        check_and_update(a_rw2, "reweight_v2")

    # === Phase 5: L-BFGS with multiple temperatures (150s) ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.6f}) ===")
    t0 = monotonic()
    phase5_budget = min(150.0, (deadline - monotonic()) * 0.2)

    if phase5_budget > 20:
        for T in [30, 100, 300]:
            if (deadline - monotonic()) < 15:
                break
            t_alloc = min(40.0, (deadline - monotonic()) * 0.4)
            a_lb = lbgfs_polish(best_seq, t_alloc, T=T)
            check_and_update(a_lb, f"lbgfs_T{T}")

    # === Phase 6: Fine peak shaving (100s) ===
    print(f"\n=== Phase 6: Peak Shaving (best={best_score:.6f}) ===")
    t0 = monotonic()
    phase6_budget = min(100.0, (deadline - monotonic()) * 0.15)

    if phase6_budget > 10:
        a_ps, _ = peak_shave_fine(best_seq, phase6_budget)
        check_and_update(a_ps, "peak_shave")

    # === Phase 7: Multi-restart exploration (remaining time) ===
    print(f"\n=== Phase 7: Multi-Restart (best={best_score:.6f}) ===")
    n_restarts = 5
    restart_budget = max(10.0, (deadline - monotonic()) * 0.25 / n_restarts)

    for r in range(n_restarts):
        if (deadline - monotonic()) < 15:
            break
        # Structured perturbation of best
        a_start = fourier_perturb(best_seq, strength=0.05, seed=seed + 100 + r)
        # Quick optimization
        a_r, _ = accelerated_reweight(a_start, restart_budget,
                                      temp_init=0.2, temp_min=0.01,
                                      lr_init=0.04, momentum_init=0.85, n_phases=3)
        check_and_update(a_r, f"restart_{r}")

        # Also try L-BFGS on the perturbed start
        if (deadline - monotonic()) > 20:
            a_r2 = lbgfs_polish(a_start, min(20.0, (deadline - monotonic()) * 0.2), T=80)
            check_and_update(a_r2, f"restart_lbfgs_{r}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic()
        if rem > 10:
            a_final = lbgfs_polish(best_seq, min(rem * 0.4, 30.0), T=20)
            check_and_update(a_final, "final_lbgfs")

        rem = deadline - monotonic()
        if rem > 10:
            a_ps_final, _ = peak_shave_fine(best_seq, min(rem * 0.5, 20.0))
            check_and_update(a_ps_final, "final_ps")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
