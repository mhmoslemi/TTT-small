"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding, exponential subgradient, peak chasing,
coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    # Scale to sum=1 first
    v_scaled = v / v.sum() if v.sum() > 1e-30 else np.ones(n) / n
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v_scaled - theta, 0.0)


def compute_subgradient(f, weights):
    """Compute subgradient of weighted sum of auto-conv peaks."""
    n = len(f)
    b_len = 2 * n - 1
    g = np.zeros(n)
    for k in range(b_len):
        if weights[k] < 1e-14:
            continue
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # g_j gets contribution from w * f_{k-j} for j in [lo, hi]
        # f_{k-j} for j in [lo, hi] means indices k-hi to k-lo
        f_idx_lo = k - hi
        f_idx_hi = k - lo
        g[lo:hi+1] += w * f[f_idx_lo:f_idx_hi+1]
    return g


def run_lp_phase(f_init, time_limit, top_k_init=30, top_k_max=200, 
                 label="lp", solver=cp.SCS):
    """LP-based optimization with adaptive top_k and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 40 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        b = fast_convolve(f)
        M = b.max()
        if M < 1e-15:
            break

        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        K = len(top_idx)

        # Build constraint matrix
        A = np.zeros((K, n))
        for row_i in range(K):
            k = top_idx[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        g = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
        try:
            prob.solve(solver=solver, max_iters=200000, eps=1e-8)
        except Exception:
            # Try with ECOS
            try:
                prob.solve(solver=cp.ECOS, max_iters=200000)
            except Exception:
                break

        if g.value is None or np.any(np.isnan(g.value)):
            top_k = min(top_k * 2, top_k_max)
            stuck += 1
            if stuck > 15:
                break
            continue

        g_val = np.maximum(g.value, 0.0)
        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 10:
                break
            top_k = min(top_k + 20, top_k_max)
            continue

        # Line search
        h = g_val / S_g
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()

        for t in [0.02, 0.04, 0.06, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                # Try a perturbation
                f = f + np.random.default_rng(it).standard_normal(n) * 0.002
                f = np.maximum(f, 0)
                f /= f.sum()
                stuck = 0

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def run_exponential_method(f_init, time_limit, T_init=0.0002, label="exp"):
    """Exponential (mirror) descent method."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    T = T_init
    lr = 0.01
    it = 0
    stuck = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        
        # Soft weights
        b_scaled = b / T
        b_scaled -= b_scaled.max()
        w = np.exp(b_scaled)
        w /= w.sum()
        
        # Periodically focus more on top peaks
        if it % 100 == 0 and T < 0.5:
            T *= 1.1
        
        # Compute subgradient
        g = compute_subgradient(f, w)
        
        # Multiplicative update (mirror descent)
        f_new = f * np.exp(-lr * g / max(g.max(), 1e-15))
        f_new = np.maximum(f_new, 1e-15)
        f_new /= f_new.sum()
        
        f = f_new
        obj = fast_convolve(f).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            stuck = 0
            lr = min(lr * 1.02, 0.05)
        else:
            stuck += 1
            if stuck > 30:
                lr *= 0.9
                stuck = 0
            if stuck > 100:
                f = best_f + rng.standard_normal(n) * 0.003
                f = np.maximum(f, 0)
                f /= f.sum()
                stuck = 0
                lr = 0.01

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} T={T:.6f} t={elapsed:.0f}s")

def run_subgradient_momentum(f_init, time_limit, lr_init=0.003, n_top=50, label="sg"):

One more idea: **Iterative Peak Equalization (IPE)**

At each iteration:
1. Compute b = conv(f, f)
2. Find the peak value M = max(b)
3. For each peak position k where b[k] > M*(1-eps):

4. Renormalize

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based direction finding, iterative peak equalization,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    v_scaled = v * (target_sum / s)
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * (target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v_scaled - theta, 0.0)


def compute_subgradient(f, weights):
    """Compute subgradient of weighted sum of auto-conv peaks."""
    n = len(f)
    b_len = 2 * n - 1
    g = np.zeros(n)
    for k in range(b_len):
        w_k = weights[k]
        if w_k < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        f_idx_lo = k - hi
        f_idx_hi = k - lo
        g[lo:hi + 1] += w_k * f[f_idx_lo:f_idx_hi + 1]
    return g

                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.015)
            else:
                no_improve += 1
                if no_improve > 6:
                    lr *= 0.93
                    no_improve = 0

            if it % 400 == 0:
                T = min(T * 1.08, 0.3)
                print(f"    {label} iter={it} obj={best_obj:.10f} "
                      f"score={2.0 * n * best_obj:.6f} lr={lr:.5f} "
                      f"t={time.monotonic() - t0:.0f}s")

            if no_improve > 20:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent with randomized ordering and multiple multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.2, 1.4, 1.7, 2.2])
    rng = np.random.default_rng(99)
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        order = np.arange(n)
        rng.shuffle(order)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            old_val = f[j]

            if old_val < 1e-14:
                for trial_val in [0.005, 0.02, 0.05]:
                    f_trial = f.copy()
                    f_trial[j] = trial_val / n
                    s = f_trial.sum()
                    f_trial /= s
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-12:
                        f = f_trial
                        best_obj = obj
                        break
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val > 0.5:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    f = f_trial
                    break

    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-phase hybrid approach combining LP, peak
    equalization, exponential descent, subgradient, and coordinate descent.
    Explores multiple lengths and uses the parent as a warm start.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        print("No parent available, using default init.")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n=1024 uniform, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    def update_best(candidate, source):
        nonlocal best_seq, best_score
        if candidate is None:
            return
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"  *** NEW BEST [{source}]: score={score:.6f} (n={len(candidate)})")

    # === Phase 1: LP main optimization on parent ===
    print(f"\n=== Phase 1: LP main (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        f1, _ = run_lp_phase(best_seq, time_limit=t1, top_k_init=40,
                             top_k_max=150, label="lp1")
        update_best(f1, "lp1")

    # === Phase 2: Peak equalization ===
    print(f"\n=== Phase 2: Peak equalization (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, _ = run_peak_equalization(best_seq, time_limit=t2)
        update_best(f2, "ipe")

    # === Phase 3: LP deep (more peaks) ===
    print(f"\n=== Phase 3: LP deep (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 400.0)
    if t3 > 60:
        n_b = len(best_seq)
        print(f"  Budget: {t3:.0f}s, n={n_b}")
        f3, _ = run_lp_phase(best_seq, time_limit=t3, top_k_init=80,
                             top_k_max=250, label="lp3")
        update_best(f3, "lp3")

    # === Phase 4: Exponential descent ===
    print(f"\n=== Phase 4: Exponential descent (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 180.0)
    if t4 > 30:
        print(f"  Budget: {t4:.0f}s")
        f4, _ = run_exponential_descent(best_seq, time_limit=t4)
        update_best(f4, "exp")

    # === Phase 5: Subgradient momentum ===
    print(f"\n=== Phase 5: Subgradient (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.1, 150.0)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}")
        f5, _ = run_subgradient_momentum(best_seq, time_limit=t5,
                                         lr_init=0.002,
                                         n_top=max(40, n_b // 8))
        update_best(f5, "sg")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.6f}) ===")
    n_base = len(best_seq)
    candidate_lengths = []
    for factor in [0.8, 0.9, 1.1, 1.25, 1.5]:
        cl = int(n_base * factor)
        cl = max(512, min(2500, cl))
        if cl != n_base and cl not in candidate_lengths:
            candidate_lengths.append(cl)
    candidate_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in candidate_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t6 = min(rem * 0.15, 180.0)
        if t6 < 30:
            break

        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  LP n={target_n}, budget={t6:.0f}s")
        f6, _ = run_lp_phase(f_interp, time_limit=t6, top_k_init=50,
                             top_k_max=150, label=f"lp_n{target_n}")
        update_best(f6, f"lp_n{target_n}")

    # === Phase 7: Final LP burst on best found ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.2, 200.0)
    if t7 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t7:.0f}s, n={n_b}")
        f7, _ = run_lp_phase(best_seq, time_limit=t7, top_k_init=100,
                             top_k_max=300, label="lp_final")
        update_best(f7, "lp_final")

    # === Phase 8: Coordinate descent polish ===
    print(f"\n=== Phase 8: Coordinate descent (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.3, 150.0)
    if t8 > 15:
        print(f"  Budget: {t8:.0f}s")
        f8, _ = coordinate_polish(best_seq, time_limit=t8)
        update_best(f8, "cd")

    # === Final verification ===
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 55}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Target: 1.5027 | Gap: {best_score - 1.5027:.6f}")
    print(f"{'=' * 55}")

    return [float(x) for x in best_seq]
