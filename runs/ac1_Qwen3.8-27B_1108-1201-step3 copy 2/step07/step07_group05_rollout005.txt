import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def p_norm_objective(a, p):
    n = len(a)
    conv = fast_convolve(a)
    S = np.sum(conv ** p)
    return 2.0 * n * S ** (1.0/p)

def p_norm_gradient(a, p):
    n = len(a)
    conv = fast_convolve(a)
    w = conv ** (p - 1)
    S = np.sum(conv ** p)
    g = correlate(w, a, mode='valid')
    coef = 4.0 * n / S ** (1.0 - 1.0/p)
    return coef * g

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Find t such that sum(max(v - t, 0)) = 1
    # Using bisection
    lo = v.min() - 1
    hi = v.max()

        s = np.sum(np.maximum(v - mid, 0))
        if s > 1:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.maximum(v - t, 0)
