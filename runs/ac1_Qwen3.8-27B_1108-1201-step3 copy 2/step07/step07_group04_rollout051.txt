"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Active-set-based local refinement with coordinate descent
3. L-BFGS smoothing with adaptive temperature

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play_improved(n, time_limit, initial=None, solver=None, verbose=True):
    """Improved fictitious play with adaptive damping and restarts."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    a_avg = a.copy().astype(np.float64)
    iters = 0
    stagnation = 0
    last_improve = monotonic()

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            break
        a_new, M_a = solve_lp_best_response(b_new, solver)
        if a_new is None:
            break

        # Adaptive damping: faster early, slower later
        alpha = 2.0 / (iters + 10)
        a_damped = (1 - alpha) * a + alpha * a_new
        a_damped = np.maximum(a_damped, 0)
        s = a_damped.sum()
        if s < 1e-15:
            break
        a_damped = a_damped / s

        # Update running average (exponential with decay)
        decay = 0.95 if iters > 200 else 0.99
        a_avg = decay * a_avg + (1 - decay) * a_new

        # Evaluate candidates
        candidates = [a_damped, a_new, a_avg, 0.5*a_new + 0.5*a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand_n = cand / cs
            sc = 2.0 * n * fast_convolve(cand_n).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand_n.copy()
                stagnation = 0
                last_improve = monotonic()
            else:
                stagnation += 1

        a = a_damped

        # Restart if stagnating too long
        if stagnation > 300:
            a = best_a + np.random.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            a_avg = a.copy()
            stagnation = 0

        if verbose and iters % 50 == 0:
            elapsed = monotonic() - t0
            print(f"    FP n={n} iter={iters} ({elapsed:.0f}s): M={M_a:.6f} best={best_score:.6f}")

    return best_a, best_score


def lbgfs_smooth_v2(a0, time_limit, T=50, maxiter=3000):
    """L-BFGS with soft-max smoothing of the max of convolution."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    n_fft = 1 << (2 * n - 1 + 63).bit_length()

    def compute_grad(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        A = rfft(a, n_fft)
        b = irfft(A * A, n_fft)[:2*n-1]
        max_b = b.max()
        scaled = (b - max_b) / T
        exp_b = np.exp(scaled)
        w = exp_b / exp_b.sum()
        lse = max_b + T * np.log(exp_b.sum() / len(b))
        obj = 2.0 * n * lse

        # Gradient wrt a: 2 * sum_k w[k] * a[k-i] (convolved weights)
        # grad_a[i] = 2 * sum_k w[k] * (T_conv)(k,i) where T_conv[k,i] = a[k-i]
        # = 2 * convolve(w, a)[i] (with appropriate alignment)
        # Actually: d/d a[i] of conv(a,a)[k] = 2*a[k-i] (if valid)
        # So grad_a[i] = 2 * sum_k w[k] * 2*a[k-i] = 4 * (w_reversed conv a)[i]... 
        # Let's just compute: grad_a[i] = sum_k w[k] * d conv[k]/d a[i]
        # d conv[k]/d a[i] = 2*a[k-i] (if 0<=k-i<n and k-i != i) or a[k-i] (if k-i==i)
        # Simplified: grad_a = 2 * (w * a circular-ish convolution)
        
        # More carefully: grad_a[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n else 0) * (1 if k-i != i else 0.5) * 2
        # Approximate: grad_a[i] ≈ 2 * sum_k w[k] * a[k-i] (ignoring the 0.5 for k=2i)
        
        # Use FFT: this is like a "cross-correlation" of w and a
        # sum_k w[k] * a[k-i] = (w * a_flipped)[i] where * is cross-correlation
        # = convolve(w, a reversed) shifted... 
        # Let's just do it directly for correctness
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        
        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return obj, grad_z

    try:
        res = minimize(
            fun=lambda z: compute_grad(z)[0],
            x0=z0,
            jac=lambda z: compute_grad(z)[1],
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        z_final = res.x
    except Exception:
        z_final = z0

    z_shifted = z_final - z_final.max()
    a_out = np.exp(z_shifted)
    a_out = a_out / a_out.sum()
    return a_out


def coordinate_descent_polish(a0, time_limit, n_sweeps=50):
    """Coordinate descent: optimize each a[i] via 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    for sweep in range(n_sweeps):
        if monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            def obj_1d(v):
                a_trial = a.copy()
                a_trial[i] = max(0.0, v)
                ts = a_trial.sum()
                if ts < 1e-15:
                    return 1e10
                a_n = a_trial / ts
                return 2.0 * n * fast_convolve(a_n).max()

            # Golden section search
            # Search range: [0, 3*a[i]] (allowing significant change)
            lo = 0.0
            hi = max(3.0 * a[i], best_score / (2.0 * n) * 2.0)
            gr = (np.sqrt(5) - 1) / 2
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = obj_1d(c)
            fd = obj_1d(d)
            for _ in range(30):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_1d(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_1d(d)

            v_opt = (lo + hi) / 2
            a_trial = a.copy()
            a_trial[i] = max(0.0, v_opt)
            ts = a_trial.sum()
            if ts < 1e-15:
                continue
            a_trial = a_trial / ts
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True

        if not improved and sweep > 5:
            break

    return best_a, best_score


def pair_swap_polish(a0, time_limit, n_iterations=5000):
    """Polish via random pair swaps with 1D line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    rng = np.random.RandomState(42)

    for it in range(n_iterations):
        if monotonic() - t0 > time_limit:
            break

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if a[i] < 1e-12 and a[j] < 1e-12:
            continue

        def obj_delta(delta):
            v_i = a[i] - delta
            v_j = a[j] + delta
            if v_i < -1e-15 or v_j < -1e-15:
                return 1e10
            a_trial = a.copy()
            a_trial[i] = max(0, v_i)
            a_trial[j] = max(0, v_j)
            ts = a_trial.sum()
            if ts < 1e-15:
                return 1e10
            a_n = a_trial / ts
            return 2.0 * n * fast_convolve(a_n).max()

        # Determine search range
        if a[i] >= a[j]:
            lo_d = -a[j] * 0.5
            hi_d = a[i] * 0.8
        else:
            lo_d = -a[i] * 0.8
            hi_d = a[j] * 0.5

        try:
            result = minimize_scalar(obj_delta, bounds=(lo_d, hi_d), method='bounded',
                                     options={'xatol': 1e-10})
            delta_opt = result.x
            a_trial = a.copy()
            a_trial[i] = max(0, a[i] - delta_opt)
            a_trial[j] = max(0, a[j] + delta_opt)
            ts = a_trial.sum()
            if ts < 1e-15:
                continue
            a_trial = a_trial / ts
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
        except Exception:
            continue

    return best_a, best_score


def active_set_polish(a0, time_limit):
    """Polish using active set: identify peak lags, do targeted reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    stale = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_val = b.max()
        # Find active set: lags within 1% of peak
        threshold = peak_val * 0.995
        active = np.where(b >= threshold)[0]

        if len(active) == 0:
            break

        improved = False
        for k in active:
            if monotonic() - t0 > time_limit:
                break
            # Find top contributors to b[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            total = contribs.sum()
            if total < 1e-20:
                continue

            # Try reducing top contributor
            top_idx = idx[np.argmax(contribs)]
            for factor in [0.99, 0.98, 0.97, 0.95]:
                diff = a[top_idx] * (1 - factor)
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[top_idx] *= factor
                # Redistribute to others proportionally
                mask = np.ones(n, dtype=bool)
                mask[top_idx] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stale += 1
            if stale > 20:
                # Perturb
                a = best_a + np.random.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                stale = 0
        else:
            stale = 0

    return best_a, best_score


def parametric_init(n, family='cosine', alpha=0.3):
    """Generate a parametric initial sequence."""
    t = np.arange(n) / n
    if family == 'cosine':
        a = 1.0 + alpha * np.cos(2 * np.pi * t)
    elif family == 'raised_cos':
        a = (1.0 + alpha * np.cos(2 * np.pi * t)) ** 2
    elif family == 'exp_cos':
        a = np.exp(alpha * np.cos(2 * np.pi * t))
    elif family == 'sinc':
        a = np.sinc(t - 0.5) ** 2
        a = np.maximum(a, 0.01)
    elif family == 'triangle':
        a = 1.0 - 2.0 * np.abs(t - 0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP Fictitious Play (main driver)
    2. Active-set local polish
    3. L-BFGS smoothing
    4. Coordinate/pair-swap descent
    5. Multi-length exploration
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: Try parametric initializations ===
    print(f"\n=== Phase 1: Parametric Inits (best={best_score:.6f}) ===")
    param_families = ['cosine', 'raised_cos', 'exp_cos', 'triangle', 'sinc']
    for fam in param_families:
        rem = deadline - monotonic()
        if rem < 100:
            break
        a_param = parametric_init(n_parent, fam)
        best_seq, best_score = check_and_update(a_param, f"param_{fam}", best_seq, best_score)

    # === Phase 2: Main FP at parent length ===
    print(f"\n=== Phase 2: FP Main n={n_parent} (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 500
    t_fp = min(rem * 0.6, 600.0)
    if t_fp > 30:
        print(f"  FP budget={t_fp:.0f}s")
        a_fp, score_fp = fictitious_play_improved(n_parent, t_fp, initial=best_seq, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"FP_{n_parent}", best_seq, best_score)

    # === Phase 3: Active set polish ===
    print(f"\n=== Phase 3: Active Set Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 300
    if rem > 30:
        t_ps = min(rem * 0.4, 120.0)
        print(f"  Active set budget={t_ps:.0f}s")
        a_as, _ = active_set_polish(best_seq, t_ps)
        best_seq, best_score = check_and_update(a_as, "active_set", best_seq, best_score)

    # === Phase 4: L-BFGS with multiple temps ===
    print(f"\n=== Phase 4: L-BFGS (best={best_score:.6f}) ===")
    for T in [20, 50, 100, 200]:
        rem = deadline - monotonic() - 200
        if rem < 20:
            break
        t_lb = min(rem * 0.3, 40.0)
        print(f"  L-BFGS T={T} budget={t_lb:.0f}s")
        a_lb = lbgfs_smooth_v2(best_seq, t_lb, T=T)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 5: Pair swap polish ===
    print(f"\n=== Phase 5: Pair Swap (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 150
    if rem > 20:
        t_ps2 = min(rem * 0.5, 100.0)
        print(f"  Pair swap budget={t_ps2:.0f}s")
        a_ps, _ = pair_swap_polish(best_seq, t_ps2)
        best_seq, best_score = check_and_update(a_ps, "pair_swap", best_seq, best_score)

    # === Phase 6: Second FP round ===
    print(f"\n=== Phase 6: FP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 200
    if rem > 60:
        t_fp2 = min(rem * 0.5, 200.0)
        print(f"  FP2 budget={t_fp2:.0f}s")
        a_fp2, _ = fictitious_play_improved(n_parent, t_fp2, initial=best_seq, solver=solver, verbose=False)
        best_seq, best_score = check_and_update(a_fp2, "FP2", best_seq, best_score)

        # Quick polish after FP2
        rem = deadline - monotonic() - 100
        if rem > 20:
            a_as2, _ = active_set_polish(best_seq, min(rem * 0.5, 60.0))
            best_seq, best_score = check_and_update(a_as2, "active_set_2", best_seq, best_score)

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    alt_lengths = []
    for factor in [0.75, 1.0, 1.25, 1.5]:
        alt_n = int(n_parent * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_parent and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:2]:
        rem = deadline - monotonic() - 80
        if rem < 50:
            break
        t_alt = min(rem * 0.5, 120.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = fictitious_play_improved(alt_n, t_alt, initial=init_alt, solver=solver, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"FP_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - monotonic() - 20
    if rem > 10:
        t_final = min(rem * 0.4, 30.0)
        a_final = lbgfs_smooth_v2(best_seq, t_final, T=80)
        best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

    rem = deadline - monotonic() - 10
    if rem > 8:
        a_ps_f, _ = active_set_polish(best_seq, min(rem * 0.6, 20.0))
        best_seq, best_score = check_and_update(a_ps_f, "final_active", best_seq, best_score)

    rem = deadline - monotonic() - 5
    if rem > 5:
        a_pf, _ = pair_swap_polish(best_seq, min(rem * 0.7, 15.0))
        best_seq, best_score = check_and_update(a_pf, "final_swap", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
