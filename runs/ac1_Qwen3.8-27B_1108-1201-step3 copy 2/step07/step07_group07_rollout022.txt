    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def max_conv(a):
    return float(fast_convolve(a).max())

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(a, M, time_budget=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0"""

    peak_indices = top_idx[active]

    g = np.zeros(n)
    for k in peak_indices:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        m = k - np.arange(i_lo, i_hi + 1)
        g[i_lo:i_hi + 1] += a[m]
    g *= 2.0 / len(peak_indices)

    return g, b_max


def subgradient_loop(a_init, time_limit, lr_init=0.004, momentum=0.92,
                     n_top=30, seed=42):
    """Projected subgradient with Nesterov momentum on the simplex."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = max_conv(a)
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        g, b_max = compute_subgradient(a, n_top=n_top)

        # Nesterov: step from lookahead point
        a_lookahead = project_simplex(a + momentum * v)
        g_ls, _ = compute_subgradient(a_lookahead, n_top=n_top)
        v = momentum * v - lr * g_ls
        a_new = project_simplex(a + v)

        obj = max_conv(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        # Adaptive learning rate
        if no_improve > 1000:
            lr *= 0.5
            v *= 0.5
            no_improve = 0
            # Small perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
        elif no_improve > 500:
            lr *= 0.8
        elif no_improve > 200:
            lr *= 0.95

    return best_a, best_obj


def lp_iteration(a, time_limit, M_reduction=0.002, max_iters=500):
    """Iterative LP-based optimization with line search blending."""
    t0 = time.monotonic()
    best_a = a.copy().astype(np.float64)
    best_a = best_a / best_a.sum()
    best_obj = max_conv(best_a)
    n = len(best_a)
    it = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = max_conv(best_a)

        # Try slightly reduced M for more aggressive direction
        for delta in [M_reduction, 0.0, 0.001]:
            M_try = M * (1.0 - delta) if delta > 0 else M
            if M_try < 1e-15:
                M_try = M
            g = lp_solve(best_a, M_try, time_limit=20.0)
            if g is not None and g.sum() > 1e-10:
                break
        else:
            break

        a_new, obj_new = line_search_blend(best_a, g, best_obj)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            if it % 20 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP it {it}: obj={best_obj:.10f} (t={elapsed:.0f}s)")
        else:
            # Try direct replacement
            g_norm = np.maximum(g, 0.0)
            gs = g_norm.sum()
            if gs > 1e-10:
                g_norm = g_norm / gs
                obj_direct = max_conv(g_norm)
                if obj_direct < best_obj - 1e-15:
                    best_obj = obj_direct
                    best_a = g_norm.copy()
                else:
                    break
            else:
                break

    return best_a, best_obj


def smooth_lsqp(a_init, time_limit, T_init=5.0, T_max=50.0):
    """
    Smooth optimization using log-sum-exp approximation with L-BFGS-B.
    Parameterize a = softmax(theta) to handle positivity and normalization.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()

    # Initialize theta from a
    a_clipped = np.maximum(a, 1e-30)
    theta = np.log(a_clipped / a_clipped.sum())

    best_a = a.copy()
    best_obj = max_conv(a)
    T = T_init

    def objective_and_grad(theta):
        a_s = np.exp(theta - theta.max())
        a_s = a_s / a_s.sum()
        b = fast_convolve(a_s)
        # Log-sum-exp for smooth max
        b_max_val = b.max()
        exp_b = np.exp(T * (b - b_max_val))
        lse = b_max_val + np.log(exp_b.sum()) / T
        # Gradient w.r.t. a_s
        weights = exp_b / exp_b.sum() / T
        # grad_a[i] = sum_k weights[k] * d(conv(a,a))_k / da[i]
        # d(conv(a,a))_k / da[i] = 2*a[k-i] (with boundary handling)
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if weights[k] < 1e-10:
                continue
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            m = k - np.arange(i_lo, i_hi + 1)
            grad_a[i_lo:i_hi + 1] += 2.0 * weights[k] * a_s[m]
        # Chain rule through softmax
        grad_theta = a_s * (grad_a - grad_a.dot(a_s))
        return lse, grad_theta

    # Run L-BFGS-B with increasing T
    for phase in range(4):
        if time.monotonic() - t0 > time_limit:
            break
        T = min(T_max, T_init * (2.0 ** phase))
        t_phase = min(time_limit - (time.monotonic() - t0), time_limit * 0.25)

        if t_phase < 5:
            break

        try:
            res = minimize(objective_and_grad, theta, method='L-BFGS-B',
                          options={'maxiter': 500, 'maxfun': 2000,
                                   'gtol': 1e-6, 'iprint': 0})
            if res.x is not None:
                a_new = np.exp(res.x - res.x.max())
                a_new = a_new / a_new.sum()
                obj_new = max_conv(a_new)
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    print(f"    Smooth phase {phase} (T={T:.1f}): obj={best_obj:.10f}")
                    theta = res.x
        except Exception:
            pass

    return best_a, best_obj


def coordinate_descent(a_init, time_limit, n_multipliers=15):
    """Coordinate descent: scale one coordinate, rescale rest to maintain sum=1."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = max_conv(a)

    multipliers = np.logspace(-1.5, 1.5, n_multipliers)

    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0.01 or scale > 100.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = max_conv(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def make_gaussian(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def full_optimize(a_init, time_budget, seed=42, verbose=False):
    """
    Full optimization pipeline: LP -> Smooth -> Subgradient -> LP -> CD.
    Returns (best_a, best_obj).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = max_conv(a)

    # Phase A: LP-based (largest allocation)
    t_lp = min(time_budget * 0.45, time_budget)
    if t_lp > 10:
        a_lp, obj_lp = lp_iteration(a, t_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    [LP] obj={best_obj:.10f} (t={time.monotonic()-t0:.0f}s)")

    # Phase B: Smooth optimization
    t_smooth = min(time_budget * 0.25, time_budget - (time.monotonic() - t0))
    if t_smooth > 10:
        a_sm, obj_sm = smooth_lsqp(best_a, t_smooth)
        if obj_sm < best_obj:
            best_obj = obj_sm
            best_a = a_sm.copy()
            if verbose:
                print(f"    [Smooth] obj={best_obj:.10f} (t={time.monotonic()-t0:.0f}s)")

    # Phase C: Subgradient
    t_sg = min(time_budget * 0.2, time_budget - (time.monotonic() - t0))
    if t_sg > 10:
        a_sg, obj_sg = subgradient_loop(best_a, t_sg, seed=seed,
                                        n_top=max(20, n // 25))
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            if verbose:
                print(f"    [SG] obj={best_obj:.10f} (t={time.monotonic()-t0:.0f}s)")

    # Phase D: More LP
    rem = time_budget - (time.monotonic() - t0)
    if rem > 15:
        a_lp2, obj_lp2 = lp_iteration(best_a, min(rem, time_budget * 0.3))
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            if verbose:
                print(f"    [LP2] obj={best_obj:.10f} (t={time.monotonic()-t0:.0f}s)")

    # Phase E: Coordinate descent polish
    rem = time_budget - (time.monotonic() - t0)
    if rem > 10:
        a_cd, obj_cd = coordinate_descent(best_a, min(rem, time_budget * 0.2))
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            if verbose:
                print(f"    [CD] obj={best_obj:.10f} (t={time.monotonic()-t0:.0f}s)")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LP-based peak reduction, smooth log-sum-exp, subgradient,
    coordinate descent, and multi-start multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_uniform(1152)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Full optimization on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Full Optimization on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 500.0)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj_opt = full_optimize(parent, t_alloc, seed=seed, verbose=True)
        update_best(a_opt, "phase1_parent")

    # === Phase 2: Multi-start with diverse inits ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Multi-Start Diverse Inits (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = n_parent

    inits = {
        "gauss_0.15": make_gaussian(n_base, 0.15),
        "gauss_0.25": make_gaussian(n_base, 0.25),
        "gauss_0.35": make_gaussian(n_base, 0.35),
        "cosine": make_cosine(n_base),
        "tukey_0.3": make_tukey(n_base, 0.3),
        "tukey_0.6": make_tukey(n_base, 0.6),
        "tukey_0.8": make_tukey(n_base, 0.8),
    }

    total_time = min(350.0, (deadline - time.monotonic() - 400.0) * 0.5)
    time_per_init = total_time / len(inits)

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 350
        if rem < 20:
            break
        t_alloc = min(time_per_init, rem * 0.4)
        if t_alloc < 15:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = full_optimize(init_a, t_alloc, seed=seed + len(name) * 31,
                                       verbose=False)
        update_best(a_div, f"div_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-Length Exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 0.625, 1.33, 0.875]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths and tl != n_parent:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 300
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 80.0)
            if t_alloc < 15:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = full_optimize(f_interp, t_alloc,
                                           seed=seed + target_n, verbose=False)
            update_best(a_len, f"len_{target_n}")

    # === Phase 4: Deep LP optimization on current best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep LP on Best (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_deep = min(rem, 250.0)
            print(f"  Deep LP: t={t_deep:.0f}s, n={len(best_seq)}")
            a_deep, obj_deep = lp_iteration(best_seq, t_deep)
            update_best(a_deep, "deep_lp")

            # Quick subgradient after LP
            rem2 = deadline - time.monotonic() - 150
            if rem2 > 15 and best_seq is not None:
                a_sg, obj_sg = subgradient_loop(best_seq, min(rem2, 60.0),
                                                seed=seed + 7777,
                                                n_top=max(30, len(best_seq) // 20))
                update_best(a_sg, "deep_sg")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: CD Polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_cd = min(rem, 80.0)
            print(f"  CD: t={t_cd:.0f}s")
            a_cd, obj_cd = coordinate_descent(best_seq, t_cd)
            update_best(a_cd, "cd_polish")

    # === Phase 6: Final micro-optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Micro (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            # One more LP pass with tight time
            t_micro = min(rem * 0.5, 40.0)
            a_m, obj_m = lp_iteration(best_seq, t_micro, max_iters=20)
            update_best(a_m, "micro_lp")

            rem = deadline - time.monotonic() - 10
            if rem > 10:
                a_m2, obj_m2 = coordinate_descent(best_seq, min(rem, 20.0),
                                                  n_multipliers=10)
                update_best(a_m2, "micro_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))
    except Exception:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
