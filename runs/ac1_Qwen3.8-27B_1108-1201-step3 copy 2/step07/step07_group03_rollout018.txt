```python
def compute_softmax_gradient(a, w, N):
    """Compute sg_j = 2 * sum_k w_k * a_{k-j} via FFT."""
    # This is 2 * (w cross-corr a)
    # Using: ifft(fft(w) * conj(fft(a)))
    W = rfft(w, N)
    A = rfft(a, N)
    h = irfft(W * np.conj(A), N)[:len(a)]
    return 2.0 * h
```
