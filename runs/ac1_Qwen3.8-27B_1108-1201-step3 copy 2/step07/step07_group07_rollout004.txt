"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax-weighted subgradient
descent with adaptive temperature, LP-based refinement with quadratic line search,
momentum, and multi-length/multi-init exploration.

Key improvements over previous approach:
- Softmax subgradient (smooth, adaptive) instead of hard-threshold top-peaks
- FFT-based O(n log n) gradient computation via cross-correlation
- Quadratic line search in LP blending for better step selection
- Momentum with adaptive decay
- More aggressive LP refinement with reduced constraint sets
- Better time allocation and parallel exploration of lengths
"""

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    return ifft(A * A).real[:size]


def compute_cross_correlation(a, w):
    """
    Compute cross_correlate(a, w)[i] = sum_j a[j] * w[j+i] for i in [0, n-1].
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    cross = ifft(conj(A) * W).real[:n]
    return cross


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_subgradient(a0, time_limit, T_init=10.0, T_max=200.0,
                        T_growth=1.003, lr_init=0.008, momentum=0.9,
                        restart_every=800, kick_size=0.01, seed=42,
                        use_momentum=True):
    """
    Softmax-weighted subgradient descent with momentum.
    Minimizes max(conv(a,a)) s.t. a >= 0, sum(a) = 1.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    T = T_init
    no_improve = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        # Compute convolution
        b = fast_convolve(a)
        b_max = b.max()

        # Softmax weights (shift for numerical stability)
        b_shifted = b - b_max
        exp_vals = np.exp(T * b_shifted)
        w_sum = exp_vals.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = exp_vals / w_sum

        # Compute subgradient: grad_i = 2 * sum_k w[k] * a[k-i] = 2 * cross_corr(a, w)[i]
        grad = 2.0 * compute_cross_correlation(a, w)

        # Momentum step
        if use_momentum:
            v = momentum * v - lr * grad
            a_new = a + v
        else:
            a_new = a - lr * grad

        # Project to simplex
        a_new = project_simplex(a_new)

        # Evaluate
        obj = fast_convolve(a_new).max()

            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        it_count += 1

        T = min(T * T_growth, T_max)

            # Restart with perturbation from best

            a = project_simplex(a)
            v = np.zeros(n)
            lr = max(lr * 0.4, lr_init * 0.05)

            no_improve = 0
            kick_size *= 0.6

            lr *= 0.97

    return best_a, best_obj


def lp_refine(f, max_iters=50, time_limit=60.0, verbose=False):
    """
    LP-based refinement: solve max sum(g) s.t. (f*g)_k <= M, g >= 0.
    Use quadratic line search for blending.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_f = fast_convolve(a)
        M = b_f.max()

        # Build LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0
        # (f*g)_k = sum_j f[j] * g[k-j]
        num_constraints = 2 * n - 1
        # For efficiency, only include constraints where f has support
        # Build constraint matrix: A[k, j] = f[k-j] if 0<=k-j<n, else 0
        # (f*g)_k = sum_j A[k,j] * g[j]
        A_mat = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_mat[k, j] = a[k - j]

        b_vec = np.full(num_constraints, M)
        c_vec = -np.ones(n)

        try:
            res = linprog(c_vec, A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            # Quadratic line search for blending parameter t
            # Minimize max(conv((1-t)*a + t*g_norm, (1-t)*a + t*g_norm)) over t in [0, 1]
            best_t = 0.0
            best_blend_obj = best_obj

            c_pt = hi - gr * d
            b_pt = lo + gr * d

                ss = blend.sum()
                if ss < 1e-15:

                    b_pt = c_pt
                    fb = fc
                    d = hi - lo
                    c_pt = hi - gr * d
                    fc = blend_obj(c_pt)

                    lo = c_pt
                    c_pt = b_pt
                    fc = fb
                    d = hi - lo
                    b_pt = lo + gr * d
                    fb = blend_obj(b_pt)

            best_t = (lo + hi) / 2.0
            if fb < fc:
                best_t = b_pt
                best_blend_obj = fb
            else:
                best_t = c_pt
                best_blend_obj = fc

            if best_blend_obj < best_obj - 1e-14:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.4f}, max_b={best_obj:.10f}")
                else:
                    a = best_a.copy()
            else:

                found = False
                for t_try in [0.01, 0.02, 0.05]:
                    blend = (1.0 - t_try) * best_a + t_try * g_norm
                    blend = np.maximum(blend, 0.0)
                    blend = blend / blend.sum()
                    obj_try = fast_convolve(blend).max()
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = blend.copy()
                        a = blend.copy()
                        found = True
                        if verbose:
                            print(f"    LP iter {iteration}: small t={t_try}, max_b={best_obj:.10f}")
                        break
                if not found:
                    break

        except Exception:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None, verbose=False):
    """
    Coordinate descent on the simplex.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0])

    improved_global = True
    max_passes = 10
    pass_count = 0
    while improved_global and pass_count < max_passes and time.monotonic() - t0 < time_limit:
        improved_global = False
        pass_count += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 1.0 - 1e-9:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = fast_convolve(a_tr).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.25):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        window = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[:taper] = window
        a[-taper:] = window[::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_optimal_shape_init(n):
    """Initialize with a shape that approximates the optimal flat-autocorrelation sequence."""
    # The optimal shape for flat autocorrelation is related to the
    # square root of a triangular function (since conv of sqrt-triangle with itself
    # gives a triangle, which is "flatter" in some sense)
    # Actually, let's use a raised cosine to the power of 0.5
    x = np.linspace(0, 1, n)
    base = 0.5 * (1.0 + np.cos(np.pi * (2.0 * x - 1.0)))
    a = np.sqrt(base + 1e-10)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def combined_optimize(a_init, time_budget, n, seed, verbose=False):
    """
    Combined optimization: alternate between softmax subgradient and LP.
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # Determine time allocation
    n_phases = max(2, int(time_budget / 30))
    phase_time = time_budget / n_phases

    for phase in range(n_phases):
        rem = time_budget - (time.monotonic() - t0)
        if rem < 5:
            break

        # Subgradient phase
        t_sg = min(phase_time * 0.6, rem * 0.5)
        if t_sg > 3:
            a_sg, obj_sg = softmax_subgradient(
                best_a, time_limit=t_sg,
                T_init=8.0 + phase * 2.0,
                T_max=150.0 + phase * 20.0,
                lr_init=0.006 / (1.0 + 0.1 * phase),
                momentum=0.88 + 0.01 * min(phase, 5),
                restart_every=600,
                seed=seed + phase * 1000
            )
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if verbose:
                    print(f"    Phase {phase} SG: max_b={best_obj:.10f}")

        # LP phase
        rem = time_budget - (time.monotonic() - t0)
        t_lp = min(phase_time * 0.4, rem * 0.5)
        if t_lp > 3:
            a_lp, obj_lp = lp_refine(best_a, max_iters=20, time_limit=t_lp, verbose=verbose)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"    Phase {phase} LP: max_b={best_obj:.10f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search for optimal sequence minimizing 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 20.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian_init(1152, 0.3)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using Gaussian n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Intensive optimization on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Intensive optimization on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_total = min(rem, 500.0)
        print(f"  Time budget: {t_total:.0f}s")
        a_opt, obj_opt = combined_optimize(parent, t_total, n_parent, seed, verbose=True)
        update_best(a_opt, "phase1_parent")

    # Coordinate descent polish on best
    rem = deadline - time.monotonic() - 500
    if rem > 10 and best_seq is not None:
        t_cd = min(rem * 0.3, 80.0)
        print(f"  CD polish: t={t_cd:.0f}s")
        a_cd, obj_cd = coordinate_descent(best_seq, time_limit=t_cd)
        update_best(a_cd, "phase1_cd")

    # === Phase 2: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Multi-length exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.75, 0.85, 1.1, 1.25, 1.4, 0.6, 1.6, 2.0]:
            tl = int(n_parent * factor)
            tl = max(200, min(5000, tl))
            if tl not in target_lengths and tl != n_parent:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:6]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 80.0)
            if t_alloc < 15:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = combined_optimize(f_interp, t_alloc, target_n, seed + target_n)
            update_best(a_len, f"len_{target_n}")

    # === Phase 3: Diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {
        "gauss_0.15": make_gaussian_init(n_base, 0.15),
        "gauss_0.35": make_gaussian_init(n_base, 0.35),
        "cosine": make_cosine_init(n_base),
        "tukey_0.3": make_tukey_init(n_base, 0.3),
        "tukey_0.7": make_tukey_init(n_base, 0.7),
        "opt_shape": make_optimal_shape_init(n_base),
        "uniform": make_uniform_init(n_base),
    }

    total_diverse_time = min(200.0, (deadline - time.monotonic() - 150.0) * 0.3)
    time_per_init = total_diverse_time / len(inits)

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 150
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.25)
        if t_alloc < 10:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = combined_optimize(init_a, t_alloc, n_base, seed + hash(name) % 100000)
        update_best(a_div, f"div_{name}")

    # === Phase 4: Deep optimization on global best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep optimization (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_deep = min(rem * 0.6, 200.0)
            print(f"  Deep: n={n_best}, t={t_deep:.0f}s")
            a_deep, obj_deep = combined_optimize(best_seq, t_deep, n_best, seed + 99999, verbose=True)
            update_best(a_deep, "deep")

            # Extra LP
            rem = deadline - time.monotonic() - 40
            if rem > 10 and best_seq is not None:
                a_lp, obj_lp = lp_refine(best_seq, max_iters=30, time_limit=min(rem, 60.0), verbose=True)
                update_best(a_lp, "deep_lp")

            # CD
            rem = deadline - time.monotonic() - 30
            if rem > 10 and best_seq is not None:
                a_cd2, obj_cd2 = coordinate_descent(best_seq, time_limit=min(rem, 40.0))
                update_best(a_cd2, "deep_cd")

    # === Phase 5: Final micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Final micro-polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_final = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            # Fine subgradient
            t_micro = min(rem * 0.4, 30.0)
            if t_micro > 3:
                a_micro, obj_micro = softmax_subgradient(
                    best_seq, time_limit=t_micro,
                    T_init=100.0, T_max=300.0, T_growth=1.005,
                    lr_init=0.0005, momentum=0.95,
                    restart_every=300, kick_size=0.003,
                    seed=seed + 777777
                )
                update_best(a_micro, "micro_sg")

            # Tiny LP
            rem = deadline - time.monotonic() - 5
            if rem > 5 and best_seq is not None:
                a_micro_lp, obj_micro_lp = lp_refine(best_seq, max_iters=10, time_limit=min(rem, 20.0))
                update_best(a_micro_lp, "micro_lp")

            # Tiny CD
            rem = deadline - time.monotonic() - 2
            if rem > 3 and best_seq is not None:
                a_micro_cd, obj_micro_cd = coordinate_descent(
                    best_seq, time_limit=rem,
                    multipliers=np.array([0.8, 0.9, 0.95, 1.0, 1.05, 1.1, 1.2])
                )
                update_best(a_micro_cd, "micro_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
