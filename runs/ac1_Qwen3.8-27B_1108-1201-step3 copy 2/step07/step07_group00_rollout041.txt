"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP direction finding, exact coordinate descent, and parametric
initialization with careful time management.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, top_k, solver=cp.SCS, eps=1e-8):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top_k peaks, g >= 0.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build constraint matrix (K x n): A[k,j] = f[k-j] for valid k-j
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For j in [lo, hi], k-j is in [0, n-1]
        # A[i, j] = f[k-j]
        j_range = np.arange(lo, hi + 1)
        A[i, j_range] = f[k - j_range]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=solver, max_iters=1000000, eps=eps)
    except Exception:
        # Try alternative solver
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000000, abstol=eps, reltol=eps)
        except Exception:
            return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

def coordinate_descent_exact(f_init, time_limit, n_sweeps_max=50):
    """
    Coordinate descent with exact 1D minimization at each step.
    For each coordinate j, find x >= 0 that minimizes the score.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max() / (f.sum()**2)
    
    sweep = 0
    while sweep < n_sweeps_max and time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            if f[j] < 1e-15:
                # Try setting to small positive value
                for x in [1e-6, 1e-5, 1e-4, f.sum()/n * 0.1]:
                    f_try = f.copy()
                    f_try[j] = x
                    s = f_try.sum()
                    if s < 1e-10:
                        continue
                    score = 2.0 * n * fast_convolve(f_try).max() / (s*s)
                    if score < best_score - 1e-12:
                        f = f_try
                        best_score = score
                        best_f = f.copy()
                        improved = True
                        break
                continue
            
            # Explore around current value
            current = f[j]
            candidates = [current * mult for mult in 
                         [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]]
            candidates.append(0.0)
            
            best_x = current
            best_local = best_score
            
            for x in candidates:
                f_try = f.copy()
                f_try[j] = x
                s = f_try.sum()
                if s < 1e-10:
                    continue
                score = 2.0 * n * fast_convolve(f_try).max() / (s*s)
                if score < best_local - 1e-13:
                    best_local = score
                    best_x = x
            
            if best_x != current:
                f[j] = best_x
                if best_local < best_score - 1e-12:
                    best_score = best_local
                    best_f = f.copy()
                    improved = True
        
        if not improved and sweep > 3:
            break
        
        if sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} score={best_score:.6f} t={elapsed:.0f}s")

    return best_f, best_score

**Alternative: Use the closed-form solution of the LP dual**

The LP is: max 1^T g s.t. A g ≤ M, g ≥ 0.
The dual is: min M * 1^T y s.t. A^T y ≥ 1, y ≥ 0.

If the LP is feasible and bounded, the dual optimum equals the primal optimum. The dual has K variables (one per constraint) and n constraints. For K << n, the dual might be easier to solve.

But cvxpy should handle this automatically.

**Alternative: Use a "proximal" version of the LP**

Instead of max sum(g) s.t. constraints, use:
max sum(g) - (alpha/2) ||g - f||^2 s.t. constraints

This is a QP, which is still convex and can be solved efficiently. The regularization term helps with stability.

**Alternative: Use Preconditioned Gradient Descent on the max**

Define a smoothed version of max: `f_smooth(a) = (1/temp) * log(sum exp(temp * b[k]))` (log-sum-exp). This is convex (max of convex functions is convex, and log-sum-exp is a smooth upper bound on max).

Wait, but b[k] is quadratic in a, so f_smooth is not convex. However, for small temp and near the maximum, it behaves like the max.

The gradient of f_smooth w.r.t. a[j] is:
`sum_k w_k * db[k]/da[j]` where `w_k = exp(temp*b[k]) / sum_m exp(temp*b[m])`.

This gives a "soft" subgradient that weights all peaks proportionally to their values.

I could use this with a decreasing temperature (simulated annealing-like).

def smooth_gradient_step(f, lr, temp, n_top=None):
    """One step of gradient descent on the smooth max."""
    n = len(f)
    b = fast_convolve(f)
    
    # Compute weights (softmax of b)
    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))
    
    b_top = b[top_idx]
    b_max = b_top.max()
    w = np.exp((b_top - b_max) / temp)
    w /= w.sum()
    
    # Compute gradient: sum_k w_k * db[k]/da[j] ≈ sum_k w_k * 2*a[k-j]
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        if w[i] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        vals = k - j_range
        # gradient contribution: w[i] * 2 * f[vals]
        # But we need to add to grad[j_range]
        grad[j_range] += w[i] * 2.0 * f[vals]
    
    # Projected gradient step (onto simplex)
    f_new = project_simplex(f - lr * grad, target_sum=f.sum())
    return f_new

def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP direction finding, smooth gradient descent (log-sum-exp surrogate),
exact coordinate descent, and perturbation-based restarts with adaptive time management.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top_k peaks, g >= 0.
    Returns (g, M) or (None, M) on failure.
    """
    n = len(f)

    if M < 1e-15:
        return None, M

        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))

    K = len(top_idx)

    # Build constraint matrix: A[i, j] = f[k_i - j] for valid indices
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        A[i, j_range] = f[k - j_range]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    solvers_to_try = []
    if solver_name == 'SCS':
        solvers_to_try = [cp.SCS, cp.CLARABEL, cp.ECOS]
    else:
        solvers_to_try = [cp.CLARABEL, cp.SCS, cp.ECOS]

    for solver in solvers_to_try:
        try:
            if solver == cp.SCS:
                prob.solve(solver=solver, max_iters=1000000, eps=1e-8)
            elif solver == cp.CLARABEL:
                prob.solve(solver=solver)
            else:
                prob.solve(solver=solver, max_iters=1000000)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def run_lp(f_init, time_limit, top_k_init=50, top_k_max=250, label="lp"):
    """LP-based optimization with adaptive top_k and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f_sum = f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 60 == 0 and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)

        g, M = lp_direction(f, top_k)
        if g is None:
            stuck += 3
            if stuck > 20:
                break
            top_k = min(top_k + 30, top_k_max)
            continue

        S_g = g.sum()
        if S_g <= f_sum * (1.0 + 3e-4):
            stuck += 1
            if stuck > 15:
                break
            continue

        h = (g / S_g) * f_sum

        # Line search for best step
        best_t = 0.05
        best_obj_t = best_obj
        for t in [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.25]:
            f_try = (1.0 - t) * f + t * h
            f_try = np.maximum(f_try, 0.0)
            s = f_try.sum()
            if s < 1e-10:
                continue
            obj = fast_convolve(f_try).max()
            if obj < best_obj_t:
                best_obj_t = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = (f / s) * f_sum

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 25:
                break

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj / (f_sum ** 2)
            print(f"    {label} it={it} score={score:.6f} tk={top_k} "
                  f"stuck={stuck} t={elapsed:.0f}s")

    return best_f, best_obj


def smooth_gradient_phase(f_init, time_limit, lr_init=0.0005, temp_init=0.0005,
                          n_top=60, label="sg"):
    """Smooth gradient descent using log-sum-exp surrogate of max."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f_sum = f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    temp = temp_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(f)

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        b_top = b[top_idx]
        b_max = b_top.max()
        w = np.exp((b_top - b_max) / temp)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= w_sum

        # Gradient: grad[j] = sum_k w_k * 2 * f[k-j]
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j_range = np.arange(lo, hi + 1)
            vals = k - j_range
            grad[j_range] += w[i] * 2.0 * f[vals]

        f = project_simplex(f - lr * grad, target_sum=f_sum)

        if it % 100 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.85
                    no_improve = 0

            temp = min(temp * 1.05, 0.5)
            if it % 500 == 0:
                score = 2.0 * n * best_obj / (f_sum ** 2)
                print(f"    {label} it={it} score={score:.6f} lr={lr:.6f} "
                      f"temp={temp:.4f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 30:
                lr = min(lr * 2.0, 0.01)
                temp = max(temp * 0.3, 0.0001)
                no_improve = 0

    return best_f, best_obj


def coordinate_descent(f_init, time_limit, n_sweeps_max=30, label="cd"):
    """Coordinate descent with multi-candidate evaluation per coordinate."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max() / (f.sum() ** 2)
    f_sum = f.sum()

    sweep = 0
    while sweep < n_sweeps_max and time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            current = f[j]
            if current < 1e-15:
                candidates = [1e-8, 1e-6, 1e-4, f_sum / n * 0.05, f_sum / n * 0.2]
            else:
                candidates = [current * m for m in
                              [0.4, 0.7, 0.85, 1.15, 1.3, 1.6, 2.5]]
                candidates.append(0.0)

            best_x = current
            best_local = best_score

            for x in candidates:
                f_try = f.copy()
                f_try[j] = x
                s = f_try.sum()
                if s < 1e-10:
                    continue
                obj = fast_convolve(f_try).max()
                score = 2.0 * n * obj / (s * s)
                if score < best_local - 1e-13:
                    best_local = score
                    best_x = x

            if best_x != current:
                f[j] = best_x
                if best_local < best_score - 1e-12:
                    best_score = best_local
                    best_f = f.copy()
                    improved = True

        if not improved and sweep > 2:
            break

        if sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} sweep={sweep} score={best_score:.6f} "
                  f"t={elapsed:.0f}s")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: LP direction finding + smooth gradient + coordinate descent,
    with perturbation restarts and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(seed)

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: LP main (most budget) ===
    print(f"\n=== Phase 1: LP main (n={len(parent)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp(parent, time_limit=t1, top_k_init=50,
                          top_k_max=200, label="lp1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp1]: {score1:.6f}")
        else:
            print(f"  lp1: {score1:.6f}")

    # === Phase 2: Smooth gradient ===
    print(f"\n=== Phase 2: Smooth gradient (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.2, 250.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = smooth_gradient_phase(best_seq, time_limit=t2,
                                         lr_init=0.0003, temp_init=0.0003,
                                         n_top=80, label="sg")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [sg]: {score2:.6f}")
        else:
            print(f"  sg: {score2:.6f}")

    # === Phase 3: LP deep (higher K) ===
    print(f"\n=== Phase 3: LP deep (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 350.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        f3, obj3 = run_lp(best_seq, time_limit=t3, top_k_init=100,
                          top_k_max=300, label="lp3")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp3]: {score3:.6f}")
        else:
            print(f"  lp3: {score3:.6f}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: CD (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.2, 200.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        f4, score4 = coordinate_descent(best_seq, time_limit=t4,
                                        n_sweeps_max=20, label="cd")
        score4c = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4c) and score4c < best_score:
            best_score = score4c
            best_seq = f4.copy()
            print(f"  NEW BEST [cd]: {score4c:.6f}")
        else:
            print(f"  cd: {score4c:.6f}")

    # === Phase 5: Perturbation + re-optimize ===
    print(f"\n=== Phase 5: Perturb+Reopt (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5_total = min(rem * 0.2, 200.0)
    if t5_total > 40:
        n_perturb = min(3, int(t5_total // 60))
        for pi in range(n_perturb):
            if time.monotonic() > deadline - 30:
                break
            t5 = min((deadline - time.monotonic()) * 0.4, 80.0)
            if t5 < 20:
                break
            # Perturb
            noise_scale = 0.0005 * (pi + 1)
            f_pert = best_seq + rng.standard_normal(len(best_seq)) * noise_scale
            f_pert = np.maximum(f_pert, 0.0)
            s = f_pert.sum()
            if s > 1e-10:
                f_pert = (f_pert / s) * best_seq.sum()
            else:
                continue

            print(f"  Perturb {pi+1} (scale={noise_scale:.5f}), budget={t5:.0f}s")
            f5, obj5 = run_lp(f_pert, time_limit=t5, top_k_init=80,
                              top_k_max=250, label=f"pert{pi}")
            score5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(score5) and score5 < best_score:
                best_score = score5
                best_seq = f5.copy()
                print(f"  NEW BEST [pert{pi}]: {score5:.6f}")
            else:
                print(f"  pert{pi}: {score5:.6f}")

    # === Phase 6: Final LP + CD polish ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.5, 120.0)
    if t6 > 20:
        print(f"  Budget: {t6:.0f}s")
        # Quick LP
        t6a = t6 * 0.6
        f6, obj6 = run_lp(best_seq, time_limit=t6a, top_k_init=150,
                          top_k_max=400, label="lp_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {score6:.6f}")

        # Quick CD
        t6b = t6 * 0.4
        if t6b > 10:
            f6b, score6b = coordinate_descent(best_seq, time_limit=t6b,
                                              n_sweeps_max=5, label="cd_final")
            score6b_c = float(evaluate_sequence(f6b.tolist()))
            if np.isfinite(score6b_c) and score6b_c < best_score:
                best_score = score6b_c
                best_seq = f6b.copy()
                print(f"  NEW BEST [cd_final]: {score6b_c:.6f}")

    # === Phase 7: Multi-length if time remains ===
    print(f"\n=== Phase 7: Multi-length (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    for factor in [1.2, 0.85, 1.4]:
        if time.monotonic() > deadline - 30:
            break
        target_n = int(n_base * factor)
        target_n = max(512, min(2500, target_n))
        if target_n == n_base:
            continue
        t7 = min(rem * 0.2, 100.0)
        if t7 < 30:
            break
        old_x = np.linspace(0, 1, n_base)
        new_x = np.linspace(0, 1, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n
        print(f"  n={target_n}, budget={t7:.0f}s")
        f7, obj7 = run_lp(f_interp, time_limit=t7, top_k_init=50,
                          top_k_max=150, label=f"n{target_n}")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [n{target_n}]: {score7:.6f}")
        else:
            print(f"  n{target_n}: {score7:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
