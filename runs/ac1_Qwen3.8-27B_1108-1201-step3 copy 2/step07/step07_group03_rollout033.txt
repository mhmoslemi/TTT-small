def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

_lp_solver = None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    T_b = T @ b_val
    return b_val, float(T_b.max())

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses with running average tracking."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:

    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        if b_new is not None:
            b = b_new

        if a_new is not None:

def coord_descent_fast(a0, time_limit, max_sweeps=500, verbose=False):

    Fast coordinate descent using O(n) formula per coordinate per step.
    For coordinate i, changing a[i] -> a[i]+t and rescaling gives:
    conv(a',a')[k] = (c[k] + 2t*a[k-i] + t^2*delta_{k,2i}) / (1+t)^2

    c = fast_convolve(a)

    rng = np.random.default_rng(2024)

        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            # Precompute shifted array
            a_shifted = np.zeros(2 * n - 1)

            hi = min(2 * n - 1, i + n)
            idx = np.arange(lo, hi)
            a_shifted[idx] = a[idx - i]

            delta = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                delta[2 * i] = 1.0

            # Grid search for best t
            t_low = -0.999 * a[i]
            t_high = max(3.0 * a[i], 0.5 / n)
            t_vals = np.linspace(t_low, t_high, 120)

def pairwise_coord_descent(a0, time_limit, verbose=False):
    """Try adjusting pairs of coordinates that contribute to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

        a = np.ones(n) / n

    c = fast_convolve(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        # Find top peak
        peak_k = int(np.argmax(c))

        # Find contributing coordinates
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.max() < 1e-15:
            break

        # Try top pairs of contributors
        top_n = min(5, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_n]
        improved = False

        for ci in range(len(top_idx)):
            for cj in range(ci + 1, len(top_idx)):
                i = int(idx[top_idx[ci]])
                j = int(idx[top_idx[cj]])
                if a[i] < 1e-15 or a[j] < 1e-15:
                    continue

                # Precompute for 2D search
                a_sh_i = np.zeros(2 * n - 1)
                a_sh_j = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    ki = k - i
                    kj = k - j
                    if 0 <= ki < n:
                        a_sh_i[k] = a[ki]
                    if 0 <= kj < n:
                        a_sh_j[k] = a[kj]

                # Cross term: a[k - i - j]
                a_cross = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    kk = k - i - j
                    if 0 <= kk < n:
                        a_cross[k] = a[kk]

                delta_2i = np.zeros(2 * n - 1)
                delta_2j = np.zeros(2 * n - 1)
                delta_sum = np.zeros(2 * n - 1)
                if 2 * i < 2 * n - 1:
                    delta_2i[2 * i] = 1.0
                if 2 * j < 2 * n - 1:
                    delta_2j[2 * j] = 1.0
                if i + j < 2 * n - 1:
                    delta_sum[i + j] = 1.0

                # Try a small 2D grid
                t1_range = np.linspace(-0.5 * a[i], 0.5 * a[i], 10)
                t2_range = np.linspace(-0.5 * a[j], 0.5 * a[j], 10)

                for t1 in t1_range:
                    for t2 in t2_range:
                        new_ai = a[i] + t1
                        new_aj = a[j] + t2
                        if new_ai < 0 or new_aj < 0:
                            continue
                        denom = (1.0 + t1 + t2) ** 2
                        if denom < 1e-15:
                            continue
                        vals = (c + t1 * (a_sh_i + a_sh_j) + t2 * (a_sh_j + a_sh_i)
                                + t1 * t1 * delta_2i + t2 * t2 * delta_2j
                                + 2 * t1 * t2 * delta_sum)
                        vals = vals / denom
                        val = 2.0 * n * vals.max()
                        if val < best_score - 1e-14:
                            a[i] = new_ai
                            a[j] = new_aj
                            a = a / a.sum()
                            c = fast_convolve(a)
                            best_score = val
                            best_a = a.copy()
                            improved = True

                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score

def subgradient_optimize(a0, time_limit, top_k=20, lr_init=0.004, verbose=False):
    """Nesterov-accelerated projected subgradient on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    v = np.zeros(n)
    y = a.copy()
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        # Compute subgradient
        b = fast_convolve(y)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peaks)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_arr = np.arange(i_start, i_end + 1)
                sg[j_arr] += 2.0 * w[idx] * y[k - j_arr]

        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        y_new = y - lr * v
        # Project onto simplex
        y_proj = simplex_project(y_new)

        score = 2.0 * n * fast_convolve(y_proj).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_proj.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_proj

        if no_improve > 100:
            lr *= 0.9
            no_improve = 0

        if no_improve > 400:
            y = simplex_project(best_a + rng.standard_normal(n) * 0.02)
            v = np.zeros(n)
            lr = max(lr_init * 0.7, lr * 1.5)
            no_improve = 0

        lr *= 0.9998

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Random Fourier superposition initialization."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, 30), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    t_fp_main = 400.0
    rem = deadline - time.monotonic() - 800
    t_fp_main = min(t_fp_main, max(60, rem))
    print(f"  FP main n={n_main}, budget={t_fp_main:.0f}s")

    a_fp, score_fp, iters = fictitious_play(
        n_main, t_fp_main, initial=best_seq, solver=solver, verbose=True
    )
    best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_main_it{iters}")

    # Second FP at different length
    rem = deadline - time.monotonic() - 700
    if rem > 60:
        n_alt = int(n_main * 1.25) if n_main < 900 else int(n_main * 0.8)
        n_alt = max(384, min(1536, n_alt))
        if n_alt != n_main:
            t_fp_alt = min(rem * 0.5, 200.0)
            print(f"  FP alt n={n_alt}, budget={t_fp_alt:.0f}s")
            initial_alt = interpolate_seq(best_seq, n_alt)
            a_fp2, score_fp2, it2 = fictitious_play(
                n_alt, t_fp_alt, initial=initial_alt, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp2, best_seq, best_score, f"fp_alt_n{n_alt}")

    # Third FP at another length
    rem = deadline - time.monotonic() - 600
    if rem > 60 and best_seq is not None:
        n_alt2 = int(len(best_seq) * 1.5) if len(best_seq) < 800 else int(len(best_seq) * 0.7)
        n_alt2 = max(384, min(1536, n_alt2))
        if n_alt2 != len(best_seq):
            t_fp_alt2 = min(rem * 0.4, 150.0)
            if t_fp_alt2 > 30:
                print(f"  FP alt2 n={n_alt2}, budget={t_fp_alt2:.0f}s")
                initial_alt2 = interpolate_seq(best_seq, n_alt2)
                a_fp3, score_fp3, it3 = fictitious_play(
                    n_alt2, t_fp_alt2, initial=initial_alt2, solver=solver, verbose=False
                )
                best_seq, best_score = check_and_update(a_fp3, best_seq, best_score, f"fp_alt2_n{n_alt2}")

    # === Phase 3: Fast Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_cd = min(rem * 0.7, 350.0)
            print(f"  CD n={n_best}, budget={t_cd:.0f}s")
            a_cd, s_cd = coord_descent_fast(best_seq, t_cd, max_sweeps=1000, verbose=True)
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd_main")

        # Pairwise coordinate descent
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_pw = min(rem * 0.5, 80.0)
            print(f"  PairwiseCD budget={t_pw:.0f}s")
            a_pw, s_pw = pairwise_coord_descent(best_seq, t_pw)
            best_seq, best_score = check_and_update(a_pw, best_seq, best_score, "pairwise_cd")

    # === Phase 4: Subgradient + CD alternation ===
    print(f"\n=== Phase 4: Subgrad + CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        for round_num in range(3):
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_sg = min(rem * 0.4, 100.0)
            t_cd4 = min(rem * 0.3, 80.0)

            print(f"  Round {round_num+1}: Subgrad({t_sg:.0f}s) + CD({t_cd4:.0f}s)")
            a_sg, s_sg = subgradient_optimize(best_seq, t_sg, top_k=25, lr_init=0.003)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, f"sg_r{round_num}")

            a_cd4, s_cd4 = coord_descent_fast(best_seq, t_cd4, max_sweeps=200)
            best_seq, best_score = check_and_update(a_cd4, best_seq, best_score, f"cd_r{round_num}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # More coordinate descent sweeps
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_fin = min(rem * 0.6, 80.0)
            print(f"  Final CD, budget={t_fin:.0f}s")
            a_fin, s_fin = coord_descent_fast(best_seq, t_fin, max_sweeps=500)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_fin2 = min(rem * 0.5, 40.0)
            a_fin2, s_fin2 = pairwise_coord_descent(best_seq, t_fin2)
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_pw")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_fin3, s_fin3 = coord_descent_fast(best_seq, rem * 0.8, max_sweeps=200)
            best_seq, best_score = check_and_update(a_fin3, best_seq, best_score, "final_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
