```python
from scipy.optimize import linprog

def solve_lp(a, M):
    """Solve: max sum(b') s.t. conv(a, b')_k <= M for all k, b' >= 0"""
    n = len(a)
    c = -np.ones(n)  # minimize -sum(b')
    
    # Build constraint matrix A_ub such that A_ub @ b' <= M * ones
    n_constraints = 2*n - 1
    A_ub = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            # b'[j] appears in constraint k with coefficient a[k-j]
            A_ub[k, j] = a[k - j]
    
    b_ub = np.full(n_constraints, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    return res.x
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using corrected FFT-based gradient
(descent direction via conj(FFT(w))*FFT(a)), LP-guided fictitious play for
global exploration, Nesterov-accelerated Adam in log-space with temperature
annealing, and multi-phase local refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import scipy.sparse as sp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_correct(a, w, n, fft_size):
    """
    Correct gradient of 2n * soft_max(conv(a,a), T) w.r.t. a.
    Uses conj(FFT(w)) * FFT(a) for the cross-correlation.
    """
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross_corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 4.0 * n * cross_corr


def compute_grad_original(a, w, n, fft_size):
    """Original (buggy) gradient for comparison."""
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def get_score_fast(a):
    """Compute the objective score for a sequence (assuming sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_fictitious_lp(a, M):
    """
    Solve LP: max sum(b') s.t. conv(a, b')_k <= M for all k, b' >= 0.
    Returns the optimal b'.
    """
    n = len(a)
    n_con = 2 * n - 1
    c = -np.ones(n)  # minimize -sum(b')
    
    # Build constraint matrix
    # A[k, l] = a[k-l] if max(0,k-n+1) <= l <= min(n-1,k)
    # Using sparse for efficiency
    rows = []
    cols = []
    vals = []
    for k in range(n_con):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for l in range(lo, hi + 1):
            rows.append(k)
            cols.append(l)
            vals.append(a[k - l])
    
    A_ub = sp.csr_matrix((vals, (rows, cols)), shape=(n_con, n))
    b_ub = np.full(n_con, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    
    if res.status == 0:
        return res.x
    else:
        # Fallback: return a
        return a.copy()


def lp_fictitious_play(a0, time_limit, n_iter=100, t=0.1):
    """
    LP-based fictitious play iteration.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * len(a) * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        b_prime = solve_fictitious_lp(a, M)
        
        # Rescale to match sum
        sp_sum = b_prime.sum()
        if sp_sum < 1e-10:
            break
        b_rescaled = b_prime * (s / sp_sum)
        
        # Update
        a_new = (1 - t) * a + t * b_rescaled
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * len(a_new) * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='exp',
                  use_correct_grad=True, verbose=False, verbose_interval=5000):
    """
    Adam optimizer in log-space with temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Temperature schedule
        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress
        
        a_curr = softmax_from_z(z)
        
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        
        if use_correct_grad:
            grad_a = compute_grad_correct(a_curr, w, n, fft_size)
        else:
            grad_a = compute_grad_original(a_curr, w, n, fft_size)
        
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.001, lr * 0.5)
        
        lr = max(0.0005, lr * (1.0 - 0.0002))
        
        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")
    
    return best_a, best_score
