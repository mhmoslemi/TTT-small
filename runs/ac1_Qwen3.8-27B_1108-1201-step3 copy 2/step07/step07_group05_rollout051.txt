7. Repeat.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using spectral shaping + soft subgradient + L-BFGS.
Key innovations over previous approach:
1. Spectral flattening via power-law compression of |FFT(a)|
2. Direct soft subgradient on the convolution peaks (not log-space smoothing)
3. Mirror descent step for better handling of non-negativity
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a)) for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_subgradient(a, w):
    """
    Compute 2 * sum_k w[k] * R_k * a (the gradient of sum_k w[k]*c[k]).
    w has length 2n-1, a has length n. Returns vector of length n.
    """
    n = len(a)
    w_len = 2 * n - 1
    a_rev = a[::-1]
    size_conv = n + w_len - 1  # 3n - 2
    fft_size = 1 << (size_conv - 1).bit_length()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(A_rev * W, fft_size)[:size_conv]
    g = 2.0 * conv_result[n - 1: 2 * n - 1]
    return g


def spectral_shaping(a_init, time_limit, alphas=None, n_iters_per_alpha=500):
    """
    Iteratively flatten the spectrum of a.
    At each step: modify |FFT(a)| by raising to power alpha, keep phase, IFFT, project.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    if alphas is None:
        alphas = [0.7, 0.8, 0.9, 0.5, 0.6]
    
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for alpha in alphas:
        for _ in range(n_iters_per_alpha):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute FFT of a
            A = rfft(a, fft_size)
            mag = np.abs(A)
            phase = np.angle(A)
            
            # Compress magnitude
            mag_new = mag ** alpha
            # Keep DC component (to preserve sum)
            A_new = mag_new * np.exp(1j * phase)
            A_new[0] = A[0]  # Keep DC
            
            a_new = irfft(A_new, fft_size)[:n]
            
            # Project
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            a = a_new
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_score


def soft_subgradient_descent(a_init, time_limit, T_init=0.001, T_end=0.0001, 
                             alpha_init=0.01, alpha_decay=0.99999):
    """
    Soft subgradient descent on the auto-convolution peaks.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    T = T_init
    alpha = alpha_init
    fft_size = 1 << (2 * n - 2).bit_length()
    total_time = time_limit
    iters = 0
    stuck_count = 0
    patience = 5000
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Compute auto-convolution
        A = rfft(a, fft_size)
        c = irfft(A * A, fft_size)[:2*n-1]
        
        # Compute weights (soft-max over peaks)
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        T_cur = T_init * (T_end / T_init) ** progress
        
        shifted = c / T_cur
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(2*n-1)
            w[c.argmax()] = 1.0
        else:
            w = w / sw
        
        # Compute subgradient
        g = compute_subgradient(a, w)
        
        # Normalize gradient for stable step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Take step
        a_new = a - alpha * g
        
        # Project: a ≥ 0, sum = 1
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        a = a_new
        alpha *= alpha_decay
        
        # If stuck, try a smaller step or restart
        if stuck_count > patience:
            # Slightly perturb
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            stuck_count = 0
            alpha = alpha_init * 0.5
    
    return best_a, best_score


def mirror_descent(a_init, time_limit, T_init=0.001, T_end=0.00005,

    Mirror descent (exponential weights) for the subgradient problem.
    a_new[i] ∝ a[i] * exp(-lr * g[i])
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Work in log-space for mirror descent
    z = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    lr = lr_init
    total_time = time_limit
    iters = 0
    stuck_count = 0
    patience = 10000
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        # Compute auto-convolution
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        T_cur = T_init * (T_end / T_init) ** progress
        
        conv = fast_convolve(a)
        shifted = conv / T_cur
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(2*n-1)
            w[conv.argmax()] = 1.0
        else:
            w = w / sw
        
        g = compute_subgradient(a, w)
        
        # Mirror descent update in log-space
        z = z - lr * g
        z = np.clip(z, -30, 30)
        
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        lr *= lr_decay
        
        if stuck_count > patience:
            rng = np.random.default_rng(iters)
            a_pert = best_a + rng.standard_normal(n) * 0.0005
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            z = np.log(np.maximum(a_pert, 1e-30))
            stuck_count = 0
            lr = lr_init * 0.3
    
    return best_a, best_score


def lbgfs_smoothed(a_init, time_limit, T=500, maxiter=5000):

        S = np.dot(grad_a, a_local)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
