"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature L-BFGS
with multi-stage soft-max annealing, peak-targeted local search, and
population-based diversity.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_soft_max_weights(b, T):
    """Compute softmax weights for the soft-max approximation of max."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / T)
    return exp_b / exp_b.sum()


def soft_max_score(a, T):
    """Compute soft-max approximation of 2n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    b_shifted = b - b.max()
    lse = b.max() + T * np.log(np.exp(b_shifted / T).sum())
    return 2.0 * n * lse, b


def soft_max_gradient(a, T):
    """Compute gradient of soft-max score w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / T)
    w = exp_b / exp_b.sum()
    
    # grad[i] = 4n * sum_j a[j] * w[i+j] for i = 0..n-1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j = np.arange(js, je + 1)
        grad[j] += 4.0 * n * wk * a[k - j]
    return grad, w


def lbgfs_adaptive(a0, T, time_limit, maxiter=5000):
    """L-BFGS optimization with given temperature."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]
    
    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        score, _ = soft_max_score(a, T)
        return score
    
    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        grad_a, _ = soft_max_gradient(a, T)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0
    
    return a_out

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

            v = np.zeros(n)

def subgradient_refine(a0, time_limit, lr_init=0.0015, momentum=0.9):
    """Subgradient method with peak-focused weighting and adaptive step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    lr = lr_init
    t0 = monotonic()
    no_improve = 0

    while monotonic() - t0 < time_limit:
        if no_improve > 300:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Soft weights concentrated on peaks
        top_k = max(15, n // 15)
        peaks = np.argpartition(b, -top_k)[-top_k:]
        pv = b[peaks]
        temp = 0.0008 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * sg
        a_candidate = a + v

        # Project onto simplex {x >= 0, sum(x) = 1}
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        a_new = a_new / a_new.sum()
        max_conv = fast_convolve(a_new).max()
        score = 2.0 * n * max_conv

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9985
        if no_improve > 150:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 1.5, 0.0015)
            no_improve = 0

    return best_a, best_score


def population_search(a0, time_limit, pop_size=10):
    """Population-based search with crossover and mutation."""
    n = len(a0)
    t0 = monotonic()

    population = [a0.copy()]
    best_score = 2.0 * n * fast_convolve(a0).max()
    best_a = a0.copy()

    # Initialize diverse population
    for _ in range(pop_size - 1):
        r = np.random.random()
        if r < 0.33:
            variant = a0 + np.random.standard_normal(n) * 0.008
        elif r < 0.66:
            # Mirror blend
            variant = 0.5 * a0 + 0.5 * a0[::-1]
            variant = variant + np.random.standard_normal(n) * 0.005
        else:
            # Scale perturbation
            variant = a0 * (1.0 + np.random.standard_normal(n) * 0.02)
        variant = np.maximum(variant, 0)
        vs = variant.sum()
        if vs > 1e-15:
            variant = variant / vs
        else:
            variant = a0.copy()
        population.append(variant)

    generation = 0
    while monotonic() - t0 < time_limit:
        generation += 1
        new_population = []

        for i in range(pop_size):
            if monotonic() - t0 > time_limit:
                break

            parent = population[i % len(population)]
            r = np.random.random()

            if r < 0.4:
                # Mutation
                scale = 0.003 + 0.003 * (generation % 10) / 10.0
                child = parent + np.random.standard_normal(n) * scale
            elif r < 0.75 and len(population) > 1:
                # Crossover
                j = np.random.randint(0, len(population))
                if j == i % len(population):
                    j = (j + 1) % len(population)
                other = population[j]
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * parent + (1.0 - alpha) * other
            else:
                # Elitism with tiny mutation
                child = parent + np.random.standard_normal(n) * 0.0005

            child = np.maximum(child, 0)
            cs = child.sum()
            if cs < 1e-15:
                continue
            child = child / cs

            cs_score = 2.0 * n * fast_convolve(child).max()
            new_population.append((cs_score, child))

            if cs_score < best_score:
                best_score = cs_score
                best_a = child.copy()

        # Select top
        new_population.sort(key=lambda x: x[0])
        population = [child for _, child in new_population[:pop_size]]

        # Ensure best is in population
        population[0] = best_a.copy()

        if generation % 5 == 0:
            print(f"    PopSearch gen={generation}: best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def peak_equalize(a0, time_limit):
    """Iterative peak equalization: reduce top peaks towards the second-highest."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0

    while monotonic() - t0 < time_limit and no_improve < 100:
        b = fast_convolve(a)
        sorted_b = np.sort(b)[::-1]
        target = sorted_b[1] if len(sorted_b) > 1 else sorted_b[0]
        peak_k = int(np.argmax(b))

        if b[peak_k] <= target + 1e-15:
            break

        # Find contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        # Try reducing each top contributor
        sorted_ci = np.argsort(contribs)[::-1]
        improved = False

        for ci in sorted_ci[:6]:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-12:
                continue

            # Reduce by small amount, redistribute to all non-contributing positions
            for frac in [0.01, 0.02, 0.03, 0.05]:
                reduce_amt = old_val * frac
                if reduce_amt < 1e-15:
                    continue

                mask = np.ones(n, dtype=bool)
                mask[i] = False
                # Also mask other contributors to this peak
                for j in idx:
                    if contribs[j - js] > 0.1 * max(contribs.max(), 1e-20):
                        mask[j] = False

                a_trial = a.copy()
                a_trial[i] = old_val - reduce_amt
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + reduce_amt * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()

                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Multi-temperature L-BFGS (main workhorse, ~700s)
    2. Peak equalization (~200s)
    3. Subgradient refinement (~200s)
    4. Population search (~200s)
    5. Second L-BFGS round (~200s)
    6. Multi-length exploration (~100s)
    7. Final polish (~50s)
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.8f}")

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")

    # === Phase 1: Multi-temperature L-BFGS ===
    print(f"\n=== Phase 1: Multi-T L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - monotonic()
    t_p1 = min(700.0, rem * 0.45)

    if t_p1 > 30:
        a_p1, _ = multi_temperature_lbgfs(
            best_seq, t_p1,
            temperatures=[200, 100, 60, 40, 25, 15, 10, 5, 3, 2, 1, 0.5, 0.3]
        )
        check_and_update(a_p1, "multiT_lbgfs")

    # === Phase 2: Peak equalization ===
    print(f"\n=== Phase 2: Peak Equalize (best={best_score:.8f}) ===")
    rem = deadline - monotonic()
    t_p2 = min(200.0, rem * 0.15)

    if t_p2 > 15:
        a_p2, _ = peak_equalize(best_seq, t_p2)
        check_and_update(a_p2, "peak_eq")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.8f}) ===")
    rem = deadline - monotonic()
    t_p3 = min(200.0, rem * 0.15)

    if t_p3 > 15:
        a_p3, _ = subgradient_refine(best_seq, t_p3)
        check_and_update(a_p3, "subgrad")

    # === Phase 4: Population search ===
    print(f"\n=== Phase 4: Population Search (best={best_score:.8f}) ===")
    rem = deadline - monotonic()
    t_p4 = min(200.0, rem * 0.15)

    if t_p4 > 20:
        a_p4, _ = population_search(best_seq, t_p4, pop_size=8)
        check_and_update(a_p4, "pop_search")

    # === Phase 5: Second L-BFGS round ===
    print(f"\n=== Phase 5: L-BFGS Round 2 (best={best_score:.8f}) ===")
    rem = deadline - monotonic()
    t_p5 = min(250.0, rem * 0.2)

    if t_p5 > 20:
        a_p5, _ = multi_temperature_lbgfs(
            best_seq, t_p5,
            temperatures=[50, 30, 20, 10, 5, 3, 2, 1, 0.5, 0.3]
        )
        check_and_update(a_p5, "lbgfs_r2")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    rem = deadline - monotonic()
    n_best = len(best_seq)

    alt_lengths = []
    for factor in [0.85, 1.15, 1.25, 0.75]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:2]:
        rem = deadline - monotonic()
        if rem < 40:
            break
        t_alloc = min(100.0, rem * 0.35)
        init_alt = interpolate_sequence(best_seq, alt_n)

        a_alt, _ = multi_temperature_lbgfs(
            init_alt, t_alloc,
            temperatures=[50, 30, 10, 5, 3, 2, 1, 0.5]
        )
        check_and_update(a_alt, f"alt_n{alt_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - monotonic()

    if rem > 20:
        t_ps = min(60.0, rem * 0.4)
        a_final, _ = peak_shave_v2(best_seq, t_ps)
        check_and_update(a_final, "final_ps")

    rem = deadline - monotonic()
    if rem > 15:
        t_lbgfs = min(40.0, rem * 0.5)
        a_final2 = lbgfs_at_temperature(best_seq, T=1.0, time_limit=t_lbgfs)
        check_and_update(a_final2, "final_lbgfs")

    rem = deadline - monotonic()
    if rem > 10:
        t_eq = min(30.0, rem * 0.6)
        a_final3, _ = peak_equalize(best_seq, t_eq)
        check_and_update(a_final3, "final_eq")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
