"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Gradient descent in log-space with adaptive softmax temperature (mirror descent)
2. LP-based fictitious play for direction finding

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring faster ones."""
    global _lp_solver_init
    if '_lp_solver_init' in globals():
        return _lp_solver_init
    for solver_name in [cp.PDLP, cp.MOSEK, cp.GURBI if hasattr(cp, 'GURBI') else cp.ECOS, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_init = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_init = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_conv_grad(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] wrt a.
    Returns grad[i] = sum_k w[k] * a[k-i] for valid indices.
    """
    n = len(a)
    corr = np.correlate(w, a, mode='full')
    return corr[n - 1: 2 * n - 1]


def softmax_temperature(a, T):
    """Compute softmax weights for the convolution with temperature T."""
    b = fast_convolve(a)
    shifted = (b - b.max()) / T
    exp_b = np.exp(shifted)
    w = exp_b / exp_b.sum()
    return w, b


def grad_descent_logspace(a0, time_limit, T_init=50.0, T_final=5.0, lr_init=0.8, lr_decay=0.9995, verbose=True):
    """Gradient descent in log-space with adaptive softmax temperature."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).astype(np.float64)
    s = a.sum()
    a = a / s
    z = np.log(a)
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = monotonic()
    it = 0
    no_improve = 0
    patience = 2000
    
    total_iters = 50000
    
    while monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive temperature
        frac = min(1.0, it / total_iters)
        T = T_init * (T_final / T_init) ** frac
        
        # Compute gradient
        w, b = softmax_temperature(a, T)
        g = compute_conv_grad(a, w)
        
        # Gradient in z-space: grad_z = a * (g - dot(a, g))
        dot_ag = np.dot(a, g)
        grad_z = a * (g - dot_ag)
        
        # Adaptive learning rate
        lr = lr_init * (lr_decay ** it)
        lr = max(lr, 0.01)
        
        # Step
        z = z - lr * grad_z
        # Keep z bounded
        z = np.clip(z, np.log(1e-15), np.log(10.0))
        
        # Convert back to a
        z_shifted = z - z.max()
        a_new = np.exp(z_shifted)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Restart if stuck
        if no_improve > patience:
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
            z = z + np.random.standard_normal(n) * 0.05
            a = np.exp(z - z.max())
            a = a / a.sum()
            no_improve = 0
            lr_init = max(lr_init * 0.5, 0.05)
        
        if verbose and it % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"    GD-log iter={it} ({elapsed:.0f}s): T={T:.1f} score={best_score:.6f} lr={lr:.4f}")

def peak_shave_quadratic(a0, time_limit, max_steps=5000):
    """Peak shaving with quadratic line search on contributing pairs."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0
    
    for step in range(max_steps):
        if monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        if peak_val <= best_score / (2.0 * n) + 1e-15:
            break
        
        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.sum() < 1e-20:
            break
        
        improved = False
        
        # Strategy 1: Reduce top contributor with optimal line search
        top_k_contrib = min(10, len(idx))
        top_ci = np.argsort(contribs)[::-1][:top_k_contrib]

            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            
            # Try a range of reductions

            best_trial_score = best_score
            for factor in [0.98, 0.96, 0.94, 0.92, 0.90, 0.85, 0.80]:

                diff = old_val - new_val

                if others_sum < 1e-15:

                improved = True
                break
        
        # Strategy 2: Reduce symmetric pair
        if not improved:
            for ci in top_ci[:5]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or j >= n or a[j] < 1e-14:
                    continue
                for f in [0.95, 0.9, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        # Strategy 3: Gaussian perturbation focused on peak contributors
        if not improved:
            for attempt in range(3):
                pert = np.random.standard_normal(n) * 0.001
                # Focus perturbation on contributing indices
                pert[idx] *= 3.0
                a_trial = a + pert
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            failed += 1
            if failed > 100:
                # Try larger random restart
                a = best_a + np.random.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                failed = 0
        else:
            failed = 0
        
        if step % 500 == 0 and step > 0:
            elapsed = monotonic() - t0
            print(f"    PeakShave step={step} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=50, maxiter=10000):
    """L-BFGS refinement in log-space with softmax smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]
    
    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse
    
    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()
        
        g = compute_conv_grad(a, w)
        dot_ag = np.dot(a, g)
        grad_z = 2.0 * n * a * (g - dot_ag)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = np.maximum(a_out, 0)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0
    
    return a_out


def multi_start_mirror_descent(a0, time_limit, n_starts=3, verbose=True):
    """Run mirror descent from multiple perturbed starting points."""
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    t0 = monotonic()
    time_per_start = time_limit / n_starts
    
    for start_i in range(n_starts):
        if monotonic() - t0 >= time_limit:
            break
        
        # Create perturbed starting point
        if start_i == 0:
            a_start = a0.copy()
        elif start_i == 1:
            a_start = a0 + np.random.standard_normal(n) * 0.01
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            # Frequency-modulated perturbation
            t = np.linspace(0, 2 * np.pi, n)
            freq = np.random.uniform(1, 5)
            a_start = a0 * (1 + 0.05 * np.cos(freq * t))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        
        rem_time = time_limit - (monotonic() - t0)
        if rem_time < 10:
            break
        
        a_res, score_res = grad_descent_logspace(
            a_start, rem_time,
            T_init=80.0, T_final=3.0,
            lr_init=1.0, lr_decay=0.9997,
            verbose=False
        )
        
        if verbose:
            print(f"    MultiStart {start_i}: score={score_res:.6f}")
        
        if score_res < best_score:
            best_score = score_res
            best_a = a_res.copy()
    
    return best_a, best_score


def adaptive_peak_reduction(a0, time_limit, n_rounds=5):
    """Iteratively reduce peaks by solving small sub-problems."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    for round_i in range(n_rounds):
        if monotonic() - t0 > time_limit:
            break
        
        rem = time_limit - (monotonic() - t0)
        if rem < 5:
            break
        
        # Find top peaks
        b = fast_convolve(a)
        n_peaks = min(5, len(b) // 3)
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        # For each peak, try to reduce it by adjusting contributing elements
        improved_any = False
        for pk in peak_indices:
            pk = int(pk)
            if b[pk] <= best_score / (2.0 * n) + 1e-12:
                continue
            
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            
            if len(idx) < 2:
                continue
            
            # Try reducing the top 3 contributors
            contribs = a[idx] * a[pk - idx]
            top_ci = np.argsort(contribs)[::-1][:3]
            
            for ci in top_ci:
                i = int(idx[ci])
                old_val = a[i]
                if old_val < 1e-12:
                    continue
                
                # Binary search for optimal reduction
                lo, hi = 0.0, 0.5
                best_f = None
                best_s = best_score
                
                for _ in range(8):
                    mid = (lo + hi) / 2
                    new_val = old_val * (1 - mid)
                    if new_val < 0:
                        hi = mid
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    diff = old_val - new_val
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    os = others.sum()
                    if os < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / os
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_s:
                        best_s = sc
                        best_f = mid
                        lo = mid
                    else:
                        hi = mid
                
                if best_f is not None and best_s < best_score - 1e-13:
                    a_trial = a.copy()
                    a_trial[i] = old_val * (1 - best_f)
                    diff = old_val * best_f
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    os = others.sum()
                    a_trial[mask] = others + diff * others / os
                    a_trial = a_trial / a_trial.sum()
                    a = a_trial
                    best_score = best_s
                    best_a = a.copy()
                    improved_any = True
                    b = fast_convolve(a)
        
        if not improved_any:
            break
    
    return best_a, best_score


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
