"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play as the
primary engine, combined with Adam log-space gradient descent for refinement,
peak-targeted local search for polishing, and multi-length exploration.
The LP approach directly solves the optimal response at each iteration,
providing more structured updates than pure gradient methods.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes auto-convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(f, n):
    """Build dense matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0. Shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_for_b(f, M, n, time_limit_lp=60.0):
    """Solve LP: maximize sum(b) subject to A*b <= M, b >= 0."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_lp, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
        return None
    except Exception:
        return None


def lp_fictitious_play(f0, time_limit, lr_init=0.3, lr_decay=0.998,
                       time_limit_lp=30.0, verbose=False):
    """LP-based fictitious play: iteratively solve LP and blend updates."""
    n = len(f0)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    f = f * np.sqrt(2.0 * n) / s

    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max() / (f.sum() ** 2)

    t0 = time.monotonic()
    lr = lr_init
    it = 0
    no_improve = 0
    max_no_improve = 150

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        M = fast_convolve(f).max()

        b = solve_lp_for_b(f, M, n, time_limit_lp=time_limit_lp)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b * np.sqrt(2.0 * n) / b_sum

        f_new = (1.0 - lr) * f + lr * b

        score = 2.0 * n * fast_convolve(f_new).max() / (f_new.sum() ** 2)

        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        lr = max(0.01, lr * lr_decay)

        if verbose and it % 20 == 0:
            print(f"    LP-FP it={it} lr={lr:.4f} score={best_score:.8f} "
                  f"elapsed={elapsed:.0f}s")

        if no_improve > max_no_improve:
            f = best_f.copy()
            lr = max(0.05, lr * 0.5)
            no_improve = 0

    return best_f, best_score


def adam_logspace(a0, time_limit, T_init=0.03, T_final=0.001, lr=0.008,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='exp'):
    """Adam optimizer in log-space with temperature annealing for soft-max smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 4000

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'hold_anneal':
            if progress < 0.2:
                T = T_init
            else:
                p2 = (progress - 0.2) / 0.8
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        z_shifted = z - np.max(z)
        a_curr = np.exp(z_shifted)
        a_curr = a_curr / a_curr.sum()

        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr

        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        z_shifted = z - np.max(z)
        a_new = np.exp(z_shifted)
        a_new = a_new / a_new.sum()
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = np.log(np.maximum(best_a, 1e-300)) + np.random.standard_normal(n) * 0.003
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.0001, lr * 0.6)

        lr = max(0.0001, lr * 0.99995)

    return best_a, best_score


def peak_shave(a0, time_limit, n_trials=5000):
    """Peak-targeted local search: identify autocorrelation peak and reduce contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_search = b.copy()
        if len(b_search) > 2:
            b_search[0] = -np.inf
            b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:6]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            j = int(peak_k - i)
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue
            for f1 in [0.95, 0.9, 0.85, 0.8, 0.7]:
                for f2 in [1.0, 1.02, 1.05, 1.1]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for ci in top_idx[:4]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in [0.9, 0.8, 0.7, 0.6]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and trial > 300 and trial % 150 == 0:
            noise_scale = 0.01 + 0.02 * (trial / n_trials)
            a = best_a + np.random.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=200):
    """Coordinate descent with multiplicative updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_local = a[i]
            best_local_score = best_score

            for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor

            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence function."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    a = a / s
    try:
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return np.inf


def diverse_inits(parent, n, rng):

    candidates = []

    if parent is not None:
        if len(parent) == n:

            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            for scale in [0.02, 0.05, 0.1]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                ps = pert.sum()
                if ps > 1e-10:
                    candidates.append((f'pert_{scale}', pert / ps))
            for freq in [2, 3, 5]:
                t = np.linspace(0, 2 * np.pi * freq, n)
                pert = p + 0.02 * np.cos(t)
                pert = np.maximum(pert, 0)
                ps = pert.sum()
                if ps > 1e-10:
                    candidates.append((f'cos_{freq}', pert / ps))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('interp', interp))

    candidates.append(('uniform', np.ones(n) / n))

    tri = np.linspace(0, 1, n) + np.linspace(1, 0, n)
    tri = np.maximum(tri, 0.01)
    candidates.append(('triangular', tri / tri.sum()))

    x = np.linspace(-3, 3, n)
    gau = np.exp(-x ** 2 / 4)
    gau = np.maximum(gau, 1e-6)
    candidates.append(('gaussian', gau / gau.sum()))

    rand = np.abs(rng.standard_normal(n)) + 0.01
    candidates.append(('random', rand / rand.sum()))

    expo = np.exp(-np.linspace(0, 3, n))
    candidates.append(('expo', expo / expo.sum()))

    return candidates

        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default uniform: n={n_def}, score={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting best score: {best_score:.8f}")

    # === Phase 1: LP Fictitious Play at parent length (45% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Fictitious Play (n={n_parent})")
    print(f"{'='*60}")

    t_p1 = min(810.0, (deadline - time.monotonic()) * 0.45)
    p1_start = time.monotonic()

    # Try multiple learning rate schedules
    lr_configs = [
        (0.5, 0.997),
        (0.3, 0.998),
        (0.2, 0.999),
        (0.1, 0.9995),
    ]

    n_configs = min(4, len(lr_configs))
    time_per_config = t_p1 / n_configs

    for ci in range(n_configs):
        rem = deadline - time.monotonic() - 250
        if rem < 45:
            break
        t_sub = min(time_per_config, rem * 0.9)
        if t_sub < 30:
            continue

        lr_i, lr_d = lr_configs[ci]
        print(f"\n  LP-FP config {ci+1}/{n_configs}: lr_init={lr_i}, lr_decay={lr_d}, "
              f"budget={t_sub:.0f}s")

        a_res, score_res = lp_fictitious_play(
            best_seq, t_sub, lr_init=lr_i, lr_decay=lr_d,
            time_limit_lp=20.0, verbose=True
        )
        s = evaluate_safe(a_res)
        if s < best_score:
            best_score = s
            best_seq = a_res.copy()
            print(f"  *** NEW BEST (LP-FP config {ci+1}): {best_score:.8f} ***")

    print(f"\n  After Phase 1: best={best_score:.8f}, "
          f"elapsed={time.monotonic() - p1_start:.0f}s")

    # === Phase 2: Adam refinement (20% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Adam Refinement (n={len(best_seq)})")
    print(f"{'='*60}")

    n_best = len(best_seq)
    t_p2 = min(360.0, (deadline - time.monotonic()) * 0.20)
    p2_start = time.monotonic()

    # Multiple Adam runs with different hyperparameters
    adam_configs = [
        (0.03, 0.001, 0.008, 'exp'),
        (0.02, 0.0005, 0.005, 'hold_anneal'),
        (0.05, 0.002, 0.01, 'linear'),
    ]

    rem = deadline - time.monotonic() - 200
    if rem > 30:
        for ai, (Ti, Tf, lr_a, sched) in enumerate(adam_configs):
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_sub = min(t_p2 / len(adam_configs), rem * 0.8)
            if t_sub < 20:
                continue

            print(f"  Adam config {ai+1}: T=[{Ti},{Tf}], lr={lr_a}, sched={sched}, "
                  f"budget={t_sub:.0f}s")
            a_res, _ = adam_logspace(
                best_seq, t_sub, T_init=Ti, T_final=Tf,
                lr=lr_a, schedule=sched
            )
            s = evaluate_safe(a_res)
            if s < best_score:
                best_score = s
                best_seq = a_res.copy()
                print(f"  *** NEW BEST (Adam {ai+1}): {best_score:.8f} ***")

    print(f"  After Phase 2: best={best_score:.8f}, "
          f"elapsed={time.monotonic() - p2_start:.0f}s")

    # === Phase 3: Peak shaving + coordinate descent (15% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Local Polishing (n={len(best_seq)})")
    print(f"{'='*60}")

    n_best = len(best_seq)
    t_p3 = min(270.0, (deadline - time.monotonic()) * 0.15)
    p3_start = time.monotonic()

    rem = deadline - time.monotonic() - 120
    if rem > 20:
        t_sub = min(t_p3 * 0.5, rem * 0.7)
        print(f"  Peak shaving: budget={t_sub:.0f}s")
        a_res, _ = peak_shave(best_seq, t_sub, n_trials=8000)
        s = evaluate_safe(a_res)
        if s < best_score:
            best_score = s
            best_seq = a_res.copy()
            print(f"  *** NEW BEST (peak shave): {best_score:.8f} ***")

    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_sub = min(t_p3 * 0.3, rem * 0.7)
        print(f"  Coordinate descent: budget={t_sub:.0f}s")
        a_res, _ = coordinate_descent(best_seq, t_sub, n_sweeps=150)
        s = evaluate_safe(a_res)
        if s < best_score:
            best_score = s
            best_seq = a_res.copy()
            print(f"  *** NEW BEST (coord descent): {best_score:.8f} ***")

    # Final peak shave after coord descent
    rem = deadline - time.monotonic() - 80
    if rem > 10:
        t_sub = min(rem * 0.8, 60.0)
        a_res, _ = peak_shave(best_seq, t_sub, n_trials=3000)
        s = evaluate_safe(a_res)
        if s < best_score:
            best_score = s
            best_seq = a_res.copy()
            print(f"  *** NEW BEST (peak shave 2): {best_score:.8f} ***")

    print(f"  After Phase 3: best={best_score:.8f}, "
          f"elapsed={time.monotonic() - p3_start:.0f}s")

    # === Phase 4: Another LP-FP round with smaller lr (10% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: LP-FP Polish (n={len(best_seq)})")
    print(f"{'='*60}")

    n_best = len(best_seq)
    t_p4 = min(180.0, (deadline - time.monotonic()) * 0.10)
    p4_start = time.monotonic()

    rem = deadline - time.monotonic() - 60
    if rem > 30:
        t_sub = min(t_p4, rem * 0.9)
        print(f"  LP-FP polish: budget={t_sub:.0f}s")
        a_res, _ = lp_fictitious_play(
            best_seq, t_sub, lr_init=0.08, lr_decay=0.9995,
            time_limit_lp=15.0, verbose=False
        )
        s = evaluate_safe(a_res)
        if s < best_score:
            best_score = s
            best_seq = a_res.copy()
            print(f"  *** NEW BEST (LP-FP polish): {best_score:.8f} ***")

    # === Phase 5: Multi-length exploration (10% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-Length Exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    t_p5 = min(180.0, (deadline - time.monotonic()) * 0.10)
    p5_start = time.monotonic()

    alt_lengths = []
    for factor in [1.25, 1.5, 0.75, 2.0, 1.1, 0.9]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            if alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic() - 40
        if rem < 30:
            break
        t_sub = min(t_p5 / max(1, len(alt_lengths[:3])), rem * 0.8)
        if t_sub < 15:
            continue

        print(f"  Trying n={alt_n}: budget={t_sub:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick LP-FP at new length
        a_alt, _ = lp_fictitious_play(
            init_alt, t_sub * 0.6, lr_init=0.2, lr_decay=0.998,
            time_limit_lp=10.0, verbose=False
        )
        # Quick Adam polish
        s_interim = evaluate_safe(a_alt)
        rem2 = deadline - time.monotonic() - 30
        if rem2 > 10 and s_interim < best_score + 0.01:
            a_alt2, _ = adam_logspace(a_alt, min(rem2 * 0.5, 30.0),
                                       T_init=0.02, T_final=0.001, lr=0.005)
            a_alt = a_alt2

        s = evaluate_safe(a_alt)
        if s < best_score:
            best_score = s
            best_seq = a_alt.copy()
            print(f"  *** NEW BEST (n={alt_n}): {best_score:.8f} ***")

    print(f"  After Phase 5: best={best_score:.8f}, "
          f"elapsed={time.monotonic() - p5_start:.0f}s")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Polish")
    print(f"{'='*60}")

    rem = deadline - time.monotonic() - 5

    if rem > 15:
        t_sub = min(rem * 0.4, 60.0)
        a_fin, _ = adam_logspace(
            best_seq, t_sub, T_init=0.01, T_final=0.0003,
            lr=0.003, schedule='exp'
        )
        s = evaluate_safe(a_fin)
        if s < best_score:
            best_score = s
            best_seq = a_fin.copy()
            print(f"  *** NEW BEST (final Adam): {best_score:.8f} ***")

    rem = deadline - time.monotonic() - 3
    if rem > 8:
        a_ps_fin, _ = peak_shave(best_seq, rem, n_trials=2000)
        s = evaluate_safe(a_ps_fin)
        if s < best_score:
            best_score = s
            best_seq = a_ps_fin.copy()
            print(f"  *** NEW BEST (final peak shave): {best_score:.8f} ***")

    # Final verification
    if best_seq is not None:
        a_final = np.maximum(best_seq, 0.0)
        s_final = a_final.sum()
        a_final = a_final / s_final
        final_score = evaluate_safe(a_final)
        if np.isfinite(final_score):
            best_score = final_score
            best_seq = a_final
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        print(f"{'='*60}")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.8f}")

    return [float(x) for x in best_seq]
