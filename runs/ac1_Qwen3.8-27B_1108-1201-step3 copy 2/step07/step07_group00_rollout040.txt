"""
LP + Mirror Descent + Multi-length optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key improvements:
1. Full constraint LP (all 2n-1 peaks) for accurate direction finding
2. Mirror descent (exponential tilting) with adaptive step size
3. Golden-section line search on the convex 1D subproblem
4. Multi-length exploration with structured initializations
5. Coordinate descent polish
6. Multiple solver fallbacks for robustness

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(f, n):
    """Build constraint matrix A where A[k,i] = f[k-i] for valid indices."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        A[k, i_lo:i_hi + 1] = f[k - i_lo:k - i_hi - 1:-1]
    return A


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def try_solve_lp(prob, f, n):
    """Try multiple solvers for the LP problem."""
    solvers_opts = [
        (cp.CLARABEL, {"tol_gap_abs": 1e-8, "tol_gap_rel": 1e-8, "tol_feas": 1e-8, "max_iter": 100000}),
        (cp.SCS, {"max_iters": 200000, "eps": 1e-8, "alpha": 1.5}),
        (cp.MOSEK, {"msk.PARAM.PMI_TOL_REL_INFEAS": 1e-8}),
        (cp.ECOS, {"abstol": 1e-8, "reltol": 1e-8, "max_iters": 100000}),
        (cp.SCS, {"max_iters": 300000, "eps": 1e-9, "alpha": 2.0}),
    ]
    for solver, opts in solvers_opts:
        try:
            prob.solve(solver=solver, **opts)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                return True
        except Exception:
            continue
    return False


def lp_direction(f, n, all_constraints=True, top_k=None):
    """
    Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M.
    Returns g (direction) or None on failure.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None

    if all_constraints or top_k is None or top_k >= len(b):
        idx = np.arange(len(b))
    else:
        idx = np.argpartition(b, -top_k)[-top_k:]
        idx = np.sort(idx)

    # Build constraint matrix for selected peaks
    K = len(idx)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = idx[row_i]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        A[row_i, i_lo:i_hi + 1] = f[k - i_lo:k - i_hi - 1:-1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    if not try_solve_lp(prob, f, n):
        return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or g_val.sum() < 1e-15:
        return None
    return g_val


def line_search(f, h, t_candidates=None):
    """Find best t in [0,1] minimizing F((1-t)f + t*h)."""
    if t_candidates is None:
        t_candidates = [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]

    best_t = 0.0
    best_obj = fast_convolve(f).max()

    for t in t_candidates:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Refine around best_t with golden section
    if best_t > 0:
        lo = max(0, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(20):
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            f1 = np.maximum((1 - t1) * f + t1 * h, 0)
            f2 = np.maximum((1 - t2) * f + t2 * h, 0)
            s1, s2 = f1.sum(), f2.sum()
            if s1 < 1e-10:
                o1 = np.inf
            else:
                o1 = fast_convolve(f1 / s1).max()
            if s2 < 1e-10:
                o2 = np.inf
            else:
                o2 = fast_convolve(f2 / s2).max()
            if o1 < o2:
                hi = t2
            else:
                lo = t1
        best_t = (lo + hi) / 2
        f_trial = np.maximum((1 - best_t) * f + best_t * h, 0)
        s = f_trial.sum()
        if s > 1e-10:
            obj = fast_convolve(f_trial / s).max()
            if obj < best_obj:
                best_obj = obj

    return best_t, best_obj

    """LP-based optimization loop."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 80 == 0 and not use_all and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)

        if use_all:
            g = lp_direction(f, n, all_constraints=True)
        else:
            g = lp_direction(f, n, all_constraints=False, top_k=top_k)

        if g is None:
            if use_all:
                g = lp_direction(f, n, all_constraints=False, top_k=min(100, 2 * n - 1))
            if g is None:
                stuck += 5
                if stuck > 20:
                    break
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                break
            if not use_all:
                top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g
        best_t, best_t_obj = line_search(f, h)

        if best_t <= 0:
            stuck += 1
            if stuck > 10:
                break
            continue

        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()
            f = f_new.copy()
            stuck = 0
        else:
            f = f_new.copy()
            stuck += 1
            if stuck > 30:

            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0 * n * best_obj:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def mirror_descent_step(f, eta, n_peaks=None, temperature=3.0):
    """One mirror descent step with exponential tilting."""
    n = len(f)
    b = fast_convolve(f)
    b_max = b.max()
    if b_max < 1e-15:
        return f.copy()

    # Compute weights on peaks
    if n_peaks is not None and n_peaks < len(b):
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        w_full = np.zeros(len(b))
        w_sel = b[top_idx].copy()
        scaled = w_sel / b_max * temperature
        w_sel = np.exp(scaled - scaled.max())
        w_sel /= w_sel.sum()
        w_full[top_idx] = w_sel
    else:
        scaled = b / b_max * temperature
        w_full = np.exp(scaled - scaled.max())
        w_full /= w_full.sum()

    # Compute subgradient
    grad = np.zeros(n)
    nonzero = np.where(w_full > 1e-10)[0]
    for k in nonzero:
        w = w_full[k]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        grad[i_lo:i_hi + 1] += w * f[k - i_lo:k - i_hi - 1:-1]

    # Exponential update
    f_new = f * np.exp(-eta * grad)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-30:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def run_mirror_descent(f_init, time_limit, label="md"):
    """Mirror descent optimization with adaptive step size."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    eta = 0.05
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    n_peaks = max(30, n // 5)

    while time.monotonic() - t0 < time_limit:
        t_batch = min(0.5, time_limit - (time.monotonic() - t0))
        if t_batch <= 0:
            break

        batch_size = max(1, min(20, int(t_batch * 2)))
        for _ in range(batch_size):
            f_new = mirror_descent_step(f, eta, n_peaks, temperature=3.0)
            obj_new = fast_convolve(f_new).max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new.copy()
                no_improve = 0
                eta = min(eta * 1.02, 0.5)
            else:
                no_improve += 1
                if no_improve > 5:
                    eta *= 0.9
                    no_improve = 0
                f = f_new.copy()

        it += 1
        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0 * n * best_obj:.6f} eta={eta:.4f} t={elapsed:.0f}s")

        if no_improve > 30:
            # Perturbation restart
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            eta = 0.05
            no_improve = 0

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements with scaling."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0])
    improved_any = True

    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.999:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved_any = True

    return f, best_obj


def interpolate_sequence(f_src, target_n):
    """Interpolate a sequence to a different length."""
    n_src = len(f_src)
    if n_src == target_n:
        return f_src.copy()
    old_x = np.linspace(0.0, 1.0, n_src)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_new = np.interp(new_x, old_x, f_src)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(target_n) / target_n
    return f_new


def generate_structured(n, kind="sine", rng=None):
    """Generate structured initial sequences."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    if kind == "sine":
        f = np.abs(np.sin(np.pi * x)) + 0.1
    elif kind == "cosine":
        f = np.abs(np.cos(np.pi * x)) + 0.1
    elif kind == "bilateral":
        f = np.exp(-2 * (x - 0.5) ** 2) + 0.05
    elif kind == "bilateral_inv":
        f = 1.0 - np.exp(-2 * (x - 0.5) ** 2) + 0.05
    elif kind == "random":
        f = rng.random(n) + 0.01
    elif kind == "zigzag":
        f = np.abs(np.sin(3 * np.pi * x) * np.sin(np.pi * x)) + 0.05
    else:
        f = np.ones(n) / n
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    return f
