Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set projected gradient
descent with BB step size, direct peak targeting, adaptive exploration,
and multi-phase optimization. Key improvements: exact subgradient (no soft-max
bias), smarter active set management, and targeted peak reduction.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    s = v.sum()
    if s < 1e-300:
        return np.ones(n) / n
    # If already on simplex
    if v.min() >= 0 and abs(s - 1.0) < 1e-10:
        return v.copy()
    # Standard projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    ws = w.sum()
    if ws < 1e-300:
        return np.ones(n) / n
    return w / ws


def score_normalized(a):
    """Compute score assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def active_set_grad(a, peaks, n):
    """Compute exact gradient of sum_{k in peaks} (a*a)[k] w.r.t. a."""
    grad = np.zeros(n)
    for k in peaks:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            ki = k - idx
            grad[i_lo:i_hi + 1] += 2.0 * a[ki]
    return grad


def optimize_active_set(a0, time_limit, n_peaks=5, lr_scale=1.0, 
                        verbose=False, verbose_interval=5000, seed=42):
    """
    Active-set projected gradient descent with BB step size.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # BB step size state
    s_prev = None
    y_prev = None
    alpha = 1.0 / np.max(active_set_grad(a, [np.argmax(fast_convolve(a))], n)) if n > 0 else 1.0
    alpha = min(alpha, 1.0)
    
    no_improve = 0
    max_no_improve = 5000
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Compute convolution and find active set
        conv = fast_convolve(a)
        c_max = conv.max()
        current_score = 2.0 * n * c_max
        
        # Find top peaks
        sorted_idx = np.argsort(conv)[::-1]
        # Adaptive peak threshold
        threshold = c_max - 0.001 * c_max * (1.0 + progress)
        active = sorted_idx[conv > threshold][:n_peaks]
        if len(active) == 0:
            active = sorted_idx[:1]
        
        # Compute gradient
        grad = active_set_grad(a, active, n)
        
        # Normalization: project gradient to tangent space
        grad_tangent = grad - np.mean(grad)
        
        # BB step size
        if s_prev is not None and y_prev is not None:
            sy = np.dot(s_prev, y_prev)
            yy = np.dot(y_prev, y_prev)
            ss = np.dot(s_prev, s_prev)
            if yy > 1e-300:
                alpha_bb1 = sy / yy
                alpha_bb2 = ss / sy if sy > 1e-300 else alpha
                alpha = 0.5 * (alpha_bb1 + alpha_bb2)
                alpha = np.clip(alpha, 1e-8, 1.0)
        
        # Damping based on progress
        damping = 1.0 - 0.5 * progress
        step = alpha * lr_scale * damping
        
        # Gradient step (descent)
        a_new = a - step * grad_tangent
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()
        
        # Update BB state
        s_curr = a_new - a
        y_curr = (active_set_grad(a_new, active, n) - grad) - (np.mean(active_set_grad(a_new, active, n)) - np.mean(grad))
        
        # Accept if improved
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update state
        s_prev = s_curr
        y_prev = y_curr
        a = a_new
        
        # Periodic restart
        if no_improve > max_no_improve:
            # Perturb from best
            a = best_a + rng.standard_normal(n) * 0.01 * (1.0 - progress)
            a = project_simplex(a)
            no_improve = 0
            s_prev = None
            y_prev = None
            # Reduce step size
            lr_scale *= 0.7
        
        # Adaptive n_peaks
        if iter_count % 5000 == 0:
            n_peaks = 3 + int(7 * progress)  # Start with 3, end with 10
        
        if verbose and iter_count % verbose_interval == 0:
            print(f"    iter={iter_count} score={best_score:.6f} cur={score_new:.6f} "
                  f"peaks={len(active)} alpha={alpha:.6f} no_imp={no_improve}")
    
    return best_a, best_score

def spectral_flatten(a0, time_limit, n_steps=5000, strength=0.1):
    """
    Iteratively flatten the power spectrum to reduce time-domain peaks.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(77)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-300:
            break
        
        # Soft flattening: reduce high-power bins
        # Target: P_target = mean(P)
        P_mean = P.mean()
        # Apply log-domain attenuation
        log_P = np.log(P + 1e-300)
        log_P_shifted = log_P - log_P.mean()
        A_new = A * np.exp(-strength * log_P_shifted)
        
        # Reconstruct and project
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn
        
        sc = score_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reject, add small noise
            a = a + rng.standard_normal(n) * 0.002
            a = project_simplex(a)
    
    return best_a, best_score


def coordinate_fine(a0, time_limit, n_sweeps=50):
    """Fine coordinate descent with binary search for optimal value."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        order = rng.permutation(n)
        any_improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-18:
                continue
            
            # Binary search for optimal a[i]
            lo = 0.0
            hi = a[i] * 2.0
            mid = a[i]
            
            for _ in range(8):  # 8 iterations of binary search
                mid = (lo + hi) / 2.0
                a_t = a.copy()
                a_t[i] = mid
                st = a_t.sum()
                if st < 1e-15:
                    break
                a_t = a_t / st
                sc = score_normalized(a_t)
                base_sc = score_normalized(a)
                if sc < base_sc:
                    lo = mid
                else:
                    hi = mid
            
            # Use the best found
            a_t = a.copy()
            a_t[i] = lo
            st = a_t.sum()
            if st < 1e-15:
                continue
            a_t = a_t / st
            sc = score_normalized(a_t)
            if sc < best_score - 1e-15:
                a = a_t
                best_score = sc
                best_a = a_t.copy()
                any_improved = True
        
        if not any_improved:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)

    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact active-set subgradient
descent with Barzilai-Borwein step sizes, direct peak targeting, spectral
flattening regularization, coordinate refinement, and multi-scale exploration.
Key improvements: exact (non-approximated) gradients, adaptive active sets,
and targeted peak reduction strategies.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project vector onto the probability simplex (sum=1, components >= 0)."""
    n = len(v)
    if n == 0:
        return v
    s = v.sum()
    if s < 1e-300:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    ws = w.sum()
    if ws < 1e-300:
        return np.ones(n) / n
    return w / ws


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def active_set_grad(a, peaks, n):
    """Compute exact gradient of sum_{k in peaks} (a*a)[k] w.r.t. a."""
    grad = np.zeros(n)
    for k in peaks:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            ki = k - idx
            grad[i_lo:i_hi + 1] += 2.0 * a[ki]
    return grad


def optimize_active_set(a0, time_limit, n_peaks_init=5, n_peaks_max=12,
                        lr_scale=1.0, verbose=False, verbose_interval=10000,
                        seed=42, restart_every=8000):
    """
    Active-set projected gradient descent with BB step size and adaptive
    active set management.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # BB step size state
    s_prev = None
    g_prev = None
    alpha = 0.5 / n
    no_improve = 0
    iter_count = 0
    lr_current = lr_scale
    n_peaks = n_peaks_init

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Adaptive peak count
        n_peaks = n_peaks_init + int((n_peaks_max - n_peaks_init) * progress)

        # Compute convolution and find active set
        conv = fast_convolve(a)
        c_max = conv.max()

        # Find top peaks with adaptive threshold
        sorted_idx = np.argsort(conv)[::-1]
        threshold = c_max * (1.0 - 0.005 * (1.0 + 2.0 * progress))
        active = sorted_idx[conv >= threshold][:n_peaks]
        if len(active) == 0:
            active = sorted_idx[:1]

        # Compute gradient at active set
        grad = active_set_grad(a, active, n)
        grad_tangent = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_tangent)

        if grad_norm < 1e-15:
            # At a critical point, perturb
            a = best_a + rng.standard_normal(n) * 0.005
            a = project_simplex(a)
            s_prev = None
            g_prev = None
            continue

        # BB step size update
        if s_prev is not None and g_prev is not None:
            y = grad_tangent - g_prev
            sy = np.dot(s_prev, y)
            yy = np.dot(y, y)
            ss = np.dot(s_prev, s_prev)
            if yy > 1e-300:

                a2 = ss / sy if sy > 1e-300 else a1

        damping = 1.0 - 0.6 * progress
        step = alpha * lr_current * damping

        # Gradient descent step
        a_new = a - step * grad_tangent
        a_new = project_simplex(a_new)

        # Evaluate
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()

        # Update BB state

        g_curr = active_set_grad(a_new, active, n)
        g_curr_tangent = g_curr - np.mean(g_curr)

            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        g_prev = g_curr_tangent
        a = a_new

            no_improve = 0
            s_prev = None
            g_prev = None
            lr_current *= 0.75

def peak_directed_search(a0, time_limit, n_trials=50000):
    """
    Directly target peaks: for the worst peak, identify top contributing pairs
    and adjust them with optimal redistribution.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        c_max = conv.max()
        peak_k = int(np.argmax(conv))

        # Elements contributing to this peak
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        idx = np.arange(i_lo, i_hi + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Top contributors
        top_k = np.argsort(contribs)[::-1][:5]

        improved = False
        for ti in top_k:
            i = int(idx[ti])
            j = int(peak_k - i)
            if i == j or i >= n or j >= n:
                continue
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue

            # Try reducing the pair while redistributing
            for f1 in [0.95, 0.90, 0.85, 0.75]:
                for f2 in [1.0, 1.05, 1.10, 1.20]:
                    a_t = a.copy()
                    a_t[i] *= f1
                    a_t[j] *= f2
                    a_t = np.maximum(a_t, 0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t /= st
                    sc = score_normalized(a_t)
                    if sc < best_score - 1e-14:
                        a = a_t
                        best_score = sc
                        best_a = a_t.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Single element reduction with smart redistribution
            if trial % 3 == 0 and len(top_k) > 0:
                i = int(idx[top_k[0]])
                if a[i] > 1e-10:
                    for f in [0.90, 0.80, 0.70]:
                        a_t = a.copy()
                        diff = a_t[i] * (1.0 - f)
                        a_t[i] *= f
                        # Redistribute to elements that DON'T contribute to peak
                        mask = np.ones(n, dtype=bool)
                        for ki in idx:
                            kii = int(ki)
                            if kii == i or int(peak_k - ki) == i:
                                mask[kii] = False
                        if mask.sum() > 1:
                            a_t[mask] += diff * a_t[mask] / a_t[mask].sum()
                        a_t = np.maximum(a_t, 0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t /= st
                        sc = score_normalized(a_t)
                        if sc < best_score - 1e-14:
                            a = a_t
                            best_score = sc
                            best_a = a_t.copy()
                            improved = True
                            break

        # Periodic perturbation to escape local minima
        if not improved and trial % 500 == 0 and trial > 0:
            pert_scale = 0.005 * (1.0 - trial / float(n_trials))
            a = best_a + rng.standard_normal(n) * pert_scale
            a = project_simplex(a)

    return best_a, best_score


def spectral_flatten(a0, time_limit, n_steps=20000, strength=0.05):
    """
    Iteratively flatten the power spectrum to reduce time-domain peaks.
    Uses log-domain attenuation of high-power frequency bins.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(7777)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-300:
            break

        P_mean = P.mean()
        if P_mean < 1e-300:
            break

        # Log-domain flattening
        log_P = np.log(P + 1e-300)
        log_P_shifted = log_P - log_P.mean()
        A_new = A * np.exp(-strength * log_P_shifted)

        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a = best_a + rng.standard_normal(n) * 0.003
            a = project_simplex(a)
            no_improve += 1
            continue
        a_new = a_new / sn

        sc = score_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Small perturbation
            a = best_a + rng.standard_normal(n) * 0.002
            a = project_simplex(a)

        # Restart if stuck
        if no_improve > 2000:
            a = best_a + rng.standard_normal(n) * 0.01
            a = project_simplex(a)
            no_improve = 0
            strength *= 0.7

    return best_a, best_score


def coordinate_fine(a0, time_limit, n_sweeps=100):
    """Fine coordinate descent with ternary search for optimal value per element."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(5555)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        any_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-18:
                continue

            # Ternary search for optimal a[i] in [0, 2*a[i]]
            lo = 0.0
            hi = a[i] * 2.5
            base_sc = score_normalized(a)

            for _ in range(6):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                a_t1 = a.copy()
                a_t1[i] = m1
                st1 = a_t1.sum()
                if st1 < 1e-15:
                    lo = m1
                    continue
                a_t1 /= st1
                sc1 = score_normalized(a_t1)

                a_t2 = a.copy()
                a_t2[i] = m2
                st2 = a_t2.sum()
                if st2 < 1e-15:
                    hi = m2
                    continue
                a_t2 /= st2
                sc2 = score_normalized(a_t2)

                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1

            # Evaluate at the best point
            a_t = a.copy()
            a_t[i] = (lo + hi) / 2.0
            st = a_t.sum()
            if st < 1e-15:
                continue
            a_t /= st
            sc = score_normalized(a_t)
            if sc < best_score - 1e-15:
                a = a_t
                best_score = sc
                best_a = a_t.copy()
                any_improved = True

        if not any_improved:
            break

    return best_a, best_score


def multi_peak_balance(a0, time_limit, n_iters=20000):
    """
    Balance multiple peaks simultaneously using weighted gradient.
    Weights are proportional to how much each peak exceeds the target.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(4444)
    alpha = 1.0 / n
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        c_max = conv.max()
        current_score = 2.0 * n * c_max

        # Find peaks above 95% of max
        threshold = c_max * 0.95
        peak_mask = conv >= threshold
        peak_indices = np.where(peak_mask)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # Weighted gradient: weight = (c[k] - threshold) / sum of excesses
        excesses = conv[peak_indices] - threshold
        total_excess = excesses.sum()
        if total_excess < 1e-300:
            weights = np.ones(len(peak_indices)) / len(peak_indices)
        else:
            weights = excesses / total_excess

        # Compute weighted gradient
        grad = np.zeros(n)
        for idx, w in zip(peak_indices, weights):
            k = int(idx)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                ii = np.arange(i_lo, i_hi + 1)
                ki = k - ii
                grad[i_lo:i_hi + 1] += w * 2.0 * a[ki]

        grad_tangent = grad - np.mean(grad)
        gn = np.linalg.norm(grad_tangent)
        if gn < 1e-15:
            break

        # Adaptive step
        step = alpha * 1.0 / gn
        a_new = a - step * grad_tangent
        a_new = project_simplex(a_new)

        sc = score_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            alpha = min(alpha * 1.05, 1.0)
        else:
            no_improve += 1
            alpha *= 0.95
            if no_improve > 1000:
                a = best_a + rng.standard_normal(n) * 0.005
                a = project_simplex(a)
                no_improve = 0
                alpha = 0.5 / n

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining active-set subgradient descent with
    BB step sizes, direct peak targeting, spectral flattening, multi-peak
    balancing, coordinate refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default uniform: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best: {best_score:.6f} (n={n_parent})")

    # === Phase 1: Active-set gradient descent (40% time) ===
    print(f"\n=== Phase 1: Active-Set Gradient Descent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t_p1 = rem * 0.40

    configs = [
        (5, 0.8, 42),
        (3, 1.2, 43),
        (8, 0.5, 44),
        (5, 0.6, 45),
    ]

    time_per_config = t_p1 / len(configs)
    for npk, lr, s in configs:
        rem = deadline - time.monotonic() - 120
        if rem < 30:
            break
        t_alloc = min(time_per_config, rem * 0.9)
        if t_alloc < 15:
            continue

        print(f"  Config: peaks={npk}, lr={lr}, budget={t_alloc:.0f}s")
        a_res, _ = optimize_active_set(
            best_seq, t_alloc, n_peaks_init=npk, n_peaks_max=npk + 5,
            lr_scale=lr, verbose=True, verbose_interval=20000, seed=s
        )
        best_seq, best_score = evaluate_and_track(a_res, f"p1_pk{npk}_lr{lr}", best_seq, best_score)

    print(f"  After Phase 1: {best_score:.6f}")

    # === Phase 2: Multi-peak balancing (20% time) ===
    print(f"\n=== Phase 2: Multi-Peak Balancing ===")
    rem = deadline - time.monotonic()
    t_p2 = rem * 0.20

    a_res, _ = multi_peak_balance(best_seq, t_p2, n_iters=50000)
    best_seq, best_score = evaluate_and_track(a_res, "p2_balance", best_seq, best_score)
    print(f"  After Phase 2: {best_score:.6f}")

    # === Phase 3: Direct peak targeting (15% time) ===
    print(f"\n=== Phase 3: Peak-Directed Search ===")
    rem = deadline - time.monotonic()
    t_p3 = rem * 0.15

    a_res, _ = peak_directed_search(best_seq, t_p3, n_trials=50000)
    best_seq, best_score = evaluate_and_track(a_res, "p3_peak", best_seq, best_score)
    print(f"  After Phase 3: {best_score:.6f}")

    # === Phase 4: Spectral flattening (10% time) ===
    print(f"\n=== Phase 4: Spectral Flattening ===")
    rem = deadline - time.monotonic()
    t_p4 = rem * 0.10

    a_res, _ = spectral_flatten(best_seq, t_p4, n_steps=30000, strength=0.03)
    best_seq, best_score = evaluate_and_track(a_res, "p4_spec", best_seq, best_score)

    rem = deadline - time.monotonic() - 30
    if rem > 10:
        a_res, _ = spectral_flatten(best_seq, rem * 0.5, n_steps=20000, strength=0.08)
        best_seq, best_score = evaluate_and_track(a_res, "p4_spec2", best_seq, best_score)

    print(f"  After Phase 4: {best_score:.6f}")

    # === Phase 5: Coordinate refinement (10% time) ===
    print(f"\n=== Phase 5: Coordinate Refinement ===")
    rem = deadline - time.monotonic()
    t_p5 = rem * 0.10

    a_res, _ = coordinate_fine(best_seq, t_p5, n_sweeps=50)
    best_seq, best_score = evaluate_and_track(a_res, "p5_coord", best_seq, best_score)
    print(f"  After Phase 5: {best_score:.6f}")

    # === Phase 6: Final refinement round (5% time) ===
    print(f"\n=== Phase 6: Final Refinement ===")
    rem = deadline - time.monotonic()
    t_p6 = rem * 0.5

    if rem > 20:
        a_res, _ = optimize_active_set(
            best_seq, t_p6, n_peaks_init=8, n_peaks_max=15,
            lr_scale=0.3, verbose=False, seed=seed + 200
        )
        best_seq, best_score = evaluate_and_track(a_res, "p6_grad", best_seq, best_score)

    rem = deadline - time.monotonic() - 15
    if rem > 10:
        a_res, _ = peak_directed_search(best_seq, rem * 0.6, n_trials=10000)
        best_seq, best_score = evaluate_and_track(a_res, "p6_peak", best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 5:
        a_res, _ = multi_peak_balance(best_seq, rem, n_iters=10000)
        best_seq, best_score = evaluate_and_track(a_res, "p6_bal", best_seq, best_score)

    print(f"  After Phase 6: {best_score:.6f}")

    # === Phase 7: Multi-length exploration (remaining) ===
    print(f"\n=== Phase 7: Multi-Length Exploration ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)

    if rem > 60:
        alt_lengths = []
        for factor in [1.25, 0.8, 1.5, 0.7, 1.1, 1.33]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic() - 10
            if rem < 25:
                break
            t_alloc = min(rem * 0.35, 50.0)
            if t_alloc < 10:
                continue
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = optimize_active_set(
                init_alt, t_alloc, n_peaks_init=5, n_peaks_max=10,
                lr_scale=0.6, verbose=False, seed=seed + alt_n
            )
            best_seq, best_score = evaluate_and_track(a_alt, f"p7_n{alt_n}", best_seq, best_score)
            print(f"  Alt n={alt_n}: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.6f}")

    return [float(x) for x in best_seq]
