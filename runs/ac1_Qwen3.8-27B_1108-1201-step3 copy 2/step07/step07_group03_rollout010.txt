import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def softmax(z):
    """Stable softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

def compute_grad_smoothed(a, tau):
    """
    Compute gradient of tau * log(sum_k exp(conv(a,a)[k] / tau)) w.r.t. a.
    Returns gradient vector of same shape as a.

    b = fast_convolve(a)
    # Numerical stability: subtract max
    b_max = b.max()
    exp_b = np.exp((b - b_max) / tau)
    w = exp_b / exp_b.sum()
    # grad_i = sum_k w[k] * (1/tau) * 2 * a[k-i] * 1(0 <= k-i < n)
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = np.arange(i_start, i_end + 1)
            grad[j] += w[k] * (2.0 / tau) * a[k - j]
    return grad

def objective_and_grad_z(z, tau):
    """
    Objective and gradient w.r.t. z (softmax parameterization).
    a = softmax(z)
    F(a) = tau * log(sum_k exp(conv(a,a)[k] / tau))
    dF/dz_j = a_j * (g_j - <a, g>) where g = grad_a F

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent (multiplicative weights) with temperature annealing
2. LP-based fictitious play for global moves
3. Exact coordinate descent with golden section line search
4. L-BFGS with softmax parameterization for final polishing
Key improvements over previous: exact line search, adaptive step sizes,

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score(a):
    """Compute the objective value (assumes sum(a) = 1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def softmax_fn(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

    """Evaluate using the provided evaluate_sequence function."""
    a = np.maximum(a, 0.0)

    if s < 0.01:

    return float(evaluate_sequence(a_norm.tolist()))

def mirror_descent_step(a, tau, eta):
    """
    One step of mirror descent (multiplicative weights) on the simplex.
    Minimizes tau * log(sum_k exp(conv(a,a)[k]/tau)) approximately.
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / tau)
    w = exp_b / exp_b.sum()
    
    # Compute gradient: grad_a[i] = (2/tau) * sum_m w[i+m] * a[m]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad_a = (2.0 / tau) * conv_result[n-1:n-1+n]
    
    # Multiplicative update: a[i] <- a[i] * exp(-eta * grad_a[i])
    a_new = a * np.exp(-eta * grad_a)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    return a_new

    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s

    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Anneal temperature
        progress = min(1.0, it / max(100, int(time_limit / 0.01)))
        tau = tau_initial * (tau_final / tau_initial) ** progress
        eta = eta_initial * (0.999 ** it)
        
        a_new = mirror_descent_step(a, tau, eta)
        
        s_new = a_new.sum()
        sc_new = score(a_new)

            best_a = a_new.copy()
            no_improve = 0

            no_improve += 1

                if sc < best_sc:
                    best_sc = sc
                    best_t = t
            
            # Golden section refinement
            if best_t != a_i_old:
                # Refine around best_t
                span = 0.15 / len(grid) * 2
                gr = best_t - span
                gl = best_t + span
                for _ in range(15):
                    if time.monotonic() - t0 > time_limit:
                        break
                    c1 = gr + phi * (gl - gr)
                    c2 = gl - phi * (gl - gr)
                    
                    def eval_at_t(t):
                        if t < 0 or t >= 1.0:
                            return 1e10
                        scale = (1.0 - t) / (1.0 - a_i_old) if a_i_old < 1.0 else 0.0
                        a_trial = a.copy()
                        a_trial[i] = t
                        others = np.arange(n) != i
                        a_trial[others] *= scale
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            return 1e10
                        a_trial = a_trial / st
                        return score(a_trial)
                    
                    sc1 = eval_at_t(c1)
                    sc2 = eval_at_t(c2)
                    if sc1 < sc2:
                        gl = c2
                    else:
                        gr = c1
                
                mid = (gr + gl) / 2
                if mid < 0 or mid >= 1.0:
                    mid = max(0.0, min(1.0 - 1e-10, mid))
                scale = (1.0 - mid) / (1.0 - a_i_old) if a_i_old < 1.0 else 0.0
                a[i] = mid
                a[np.arange(n) != i] *= scale
                a = np.maximum(a, 0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    sc = score(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
        
        if not improved and sweep > 3:
            break
    
    return best_a, best_score

def lbfgs_softmax_optimize(a0, time_limit, tau=0.002, maxiter=500):
    """
    L-BFGS optimization with softmax parameterization.
    Minimize tau * log(sum_k exp(conv(a,a)[k]/tau)) where a = softmax(z).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize z from a (inverse softmax)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()  # Center for numerical stability
    
    def obj_grad(z):
        a = softmax_fn(z)
        b = fast_convolve(a)
        b_max = b.max()
        exp_b = np.exp((b - b_max) / tau)
        w = exp_b / exp_b.sum()
        obj = tau * (np.log(exp_b.sum()) + b_max / tau)
        
        # Gradient w.r.t. a
        a_rev = a[::-1]
        conv_res = np.convolve(w, a_rev, mode='full')
        grad_a = (2.0 / tau) * conv_res[n-1:n-1+n]
        
        # Gradient w.r.t. z: grad_z[j] = a[j] * (grad_a[j] - <a, grad_a>)
        weighted_avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - weighted_avg)
        
        return obj, grad_z
    
    t0 = time.monotonic()
    result = minimize(obj_grad, z, method='L-BFGS-B', 
                      options={'maxiter': maxiter, 'maxfun': 2000, 'ftol': 1e-12, 'gtol': 1e-10})
    
    a_opt = softmax_fn(result.x)
    a_opt = a_opt / a_opt.sum()
    sc = score(a_opt)
    
    return a_opt, sc

def get_lp_solver():
    import cvxpy as cp
    global _lp_solver_mine
    if '_lp_solver_mine' in globals() and _lp_solver_mine is not None:
        return _lp_solver_mine
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_mine = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_mine = cp.ECOS
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(a0, time_limit, solver=None, verbose=False):
    """Alternating LP best responses with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    b = a.copy()
    a_avg = a.copy()
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        # Nesterov-type averaging
        t_k = (1 + np.sqrt(1 + 4 * (iters - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k
        a_avg_new = beta_k * a_avg + (1 - beta_k) * a
        a_avg = a_avg_new
        
        candidates = [a, b, 0.5 * (a + b), a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            sc_val = score(cand)
            if sc_val < best_score:
                best_score = sc_val
                best_a = cand.copy()
        
        if verbose and iters % 10 == 0:
            print(f"      FP it={iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score, iters

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_random_smooth(n, rng):
    """Random smooth initialization."""
    a = rng.random(n)
    # Smooth with a Gaussian kernel
    kernel_size = max(3, n // 50)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4) ** 2)
    kernel = kernel / kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    return a
