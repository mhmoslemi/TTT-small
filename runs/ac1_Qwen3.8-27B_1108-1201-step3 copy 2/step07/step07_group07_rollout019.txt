"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based successive convex
approximation (SCA) with Nesterov-style acceleration, combined with FFT
subgradient polish, LP refinement, and multi-length exploration.

Key innovation: Successive convex approximation via LP linearization of
the auto-convolution peak constraint, which provides a principled step
size and direction selection without manual learning rate tuning.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local_max_b(a):
    """Return max of auto-convolution for sequence (assumes sum(a)=1)."""
    b = fast_convolve(a)
    return float(b.max())


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_sca_step(a, n_active=50, use_all=False):
    """
    One step of successive convex approximation via LP.
    Solves: min t s.t. 2*sum_i a_{k-i}*g_i - t <= b_k, sum(g)=1, g>=0
    for active peaks k.
    Returns (g, t, b) or (None, None, b) if LP fails.
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    
    if use_all:
        active_idx = np.arange(len(b))
    else:
        threshold = b_max * 0.9
        active_idx = np.where(b >= threshold)[0]
        if len(active_idx) > n_active:
            order = np.argsort(b[active_idx])[::-1][:n_active]
            active_idx = active_idx[order]
    
    num_con = len(active_idx)
    
    # Build LP
    A_ub = np.zeros((num_con, n + 1))
    b_ub = np.zeros(num_con)
    
    for row, k in enumerate(active_idx):
        # 2*sum_i a_{k-i}*g_i - t <= b_k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A_ub[row, i] = 2.0 * a[k - i]
        A_ub[row, n] = -1.0
        b_ub[row] = b[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0.0)
            s = g.sum()
            if s > 1e-15:
                g = g / s
            return g, res.x[n], b
        else:
            return None, None, b
    except Exception:
        return None, None, b

    s = a.sum()

        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_obj = eval_local_max_b(a)
    no_improve = 0
    kick_size = 0.05
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        g, t_lp, b = lp_sca_step(a, n_active=n_active)
        if g is None:
            break
        
        # Line search with damping
        best_t = 0.0
        best_blend_obj = best_obj
        cur_obj = eval_local_max_b(a)
        
        for tau in damping_candidates:
            blend = (1.0 - tau) * a + tau * g
            blend = np.maximum(blend, 0.0)
            ss = blend.sum()
            if ss < 1e-15:
                continue
            blend = blend / ss
            obj = eval_local_max_b(blend)
            if obj < best_blend_obj:
                best_blend_obj = obj
                best_t = tau
        
        if best_t > 0 and best_blend_obj < best_obj - 1e-14:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_obj = best_blend_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:

                a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
                best_obj = eval_local_max_b(a)
                if best_obj < eval_local_max_b(best_a):
                    best_a = a.copy()
                no_improve = 0
                kick_size *= 0.7

def subgradient_optimize_fft(a0, time_limit, lr_init=0.003, momentum=0.92,
                             n_top=None, seed=42, restart_interval=2000):
    """FFT-accelerated projected subgradient with momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_local_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    kick_size = 0.005
    
    if n_top is None:
        n_top = max(10, n // 10)
    
    # Precompute FFT size
    fft_size = 1 << (3 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        A = rfft(a, fft_size)
        b_full = irfft(A * A, fft_size)
        b = b_full[:2 * n - 1]
        b_max = b.max()
        
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        
        threshold = b_max * 0.999
        top_vals = b[top_idx]
        active_mask = top_vals >= threshold
        if not active_mask.any():
            active_mask[0] = True
        
        w = np.zeros(2 * n - 1)
        active_indices = top_idx[active_mask]
        w[active_indices] = 1.0 / len(active_indices)
        
        W = rfft(w, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]
        
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        obj_new = eval_local_max_b(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve >= restart_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.3, lr_init * 0.02)
            no_improve = 0
            kick_size *= 0.5
            if kick_size < 0.0001:
                break
        elif no_improve >= restart_interval // 3:
            cur_lr *= 0.95
    
    return best_a, best_obj


def lp_refine(f, max_iters=30, time_limit=60.0):

    a = a / a.sum()

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        num_constraints = 2 * n - 1
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, k - j] = a[j]
        
        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend = blend / ss
                obj = eval_local_max_b(blend)
                if obj < best_blend_obj:
                    best_blend_obj = obj
                    best_t = t
            
            if best_t > 0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_local_max_b(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                else:
                    a = best_a.copy()
            else:
                break
        except Exception:
            break
    
    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_triangular_init(n):
    a = np.ones(n)
    half = n // 2
    a[:half] = np.linspace(0.1, 1.0, half)
    a[half:] = np.linspace(1.0, 0.1, n - half)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_antitriangular_init(n):
    """Anti-triangular: high at edges, low in center."""
    a = np.ones(n)
    half = n // 2
    a[:half] = np.linspace(1.0, 0.3, half)
    a[half:] = np.linspace(0.3, 1.0, n - half)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_bump_init(n, center=0.5, width=0.2):
    """Gaussian bump at a specific location."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on the simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_local_max_b(a)
    
    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue
                
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_local_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val
