import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def solve_dual_lp(f, M, max_time=5.0):
    """
    Solve the dual LP to get optimal peak weights.
    minimize: M * sum(y)
    subject to: sum_j f[j] * y[i+j] >= 1 for i=0..n-1
                y >= 0
    """
    n = len(f)
    try:
        import cvxpy as cp
        
        y = cp.Variable(2*n - 1)
        constraints = []
        
        # Build constraint matrix: A^T where A[k,i] = f[k-i] if valid
        # Constraint i: sum_{k=i}^{i+n-1} f[k-i] * y[k] >= 1
        # i.e., sum_{j=0}^{n-1} f[j] * y[i+j] >= 1
        
        # Build as sparse matrix
        row_indices = []
        col_indices = []
        values = []
        
        for i in range(n):
            for j in range(n):
                k = i + j
                if 0 <= k < 2*n - 1:
                    row_indices.append(i)
                    col_indices.append(k)
                    values.append(f[j])
        
        from scipy.sparse import csc_matrix
        A_mat = csc_matrix((values, (row_indices, col_indices)), shape=(n, 2*n-1))
        
        constraints.append(A_mat @ y >= 1)
        constraints.append(y >= 0)
        
        problem = cp.Problem(cp.Minimize(M * cp.sum(y)), constraints)
        problem.solve(solver=cp.CBC, verbose=False)
