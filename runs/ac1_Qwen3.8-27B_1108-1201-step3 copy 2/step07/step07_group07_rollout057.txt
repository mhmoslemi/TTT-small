def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Projection onto probability simplex."""
    n = len(v)
    if n <= 1:
        return np.array([max(v[0], 0.0)]) if n == 1 else np.array([0.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, weight_mode='exponential', s=30.0, top_k=1):

    b = fast_convolve(a)
    b_max = b.max()
    
    if weight_mode == 'uniform':

    elif weight_mode == 'exponential':

        b_shifted = b - b_max
        w = np.exp(s * b_shifted)
        w /= w.sum()
    elif weight_mode == 'top_k':

def golden_section_search(f, a, b, tol=1e-10, max_iter=100):
    """Golden section search for minimum of unimodal function on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP refinement with
golden-section line search (including extrapolation), exponential smooth-max
subgradient, coordinate descent polish, and multi-length exploration.

Key improvements over previous approach:
- Golden-section line search for LP blending (explores t in [0, 2.5])
- Exponential peak weighting in subgradient for sharper directional focus
- Adaptive LP: detects stagnation and switches to extrapolation mode
- Spectral analysis to identify dominant frequency components
- More aggressive multi-length exploration with better time allocation
- Second-order polish via L-BFGS on smoothed objective
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.array([max(v[0], 0.0)]) if n == 1 else np.array([0.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_exp(a, s=40.0):
    """
    Compute subgradient of max(conv(a,a)) with exponential peak weighting.
    Returns (gradient_vector, max_b, b_array).
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = float(b.max())

    b_shifted = b - b_max
    w = np.exp(s * b_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[b == b_max] = 1.0
        w_sum = w.sum()
    w /= w_sum

    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid indices
    # = 2 * cross_correlation(w, a)[i]
    # cross_corr(i) = sum_k w[k] * a[k-i]
    # = (w_rev * a)[len(w)-1-i] where * is linear convolution
    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a, fft_size)
    w_rev = w[::-1]
    Wr = rfft(w_rev, fft_size)
    conv = irfft(A * Wr, fft_size)
    # cross_corr[i] = conv[2n-2-i] for i=0..n-1
    cross = conv[n-1:2*n-1][::-1]

    grad = 2.0 * cross
    return grad, b_max, b


def compute_subgradient_topk(a, k=3):
    """Compute subgradient with uniform weight on top-k peaks."""
    n = len(a)
    b = fast_convolve(a)
    b_max = float(b.max())
    k = min(k, len(b))
    idx = np.argpartition(b, -k)[-k:]
    w = np.zeros_like(b)
    w[idx] = 1.0 / k

    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a, fft_size)
    w_rev = w[::-1]
    Wr = rfft(w_rev, fft_size)
    conv = irfft(A * Wr, fft_size)
    cross = conv[n-1:2*n-1][::-1]

    grad = 2.0 * cross
    return grad, b_max, b


def smooth_max_obj(a, s=50.0):
    """Smooth approximation of max(conv(a,a)) using log-sum-exp."""
    b = fast_convolve(a)
    b_max = float(b.max())
    log_sum = np.log(np.sum(np.exp(s * (b - b_max)))) / s + b_max
    return log_sum


def smooth_max_grad(a, s=50.0):
    """Gradient of smooth max objective."""
    n = len(a)
    b = fast_convolve(a)
    b_max = float(b.max())
    weights = np.exp(s * (b - b_max))
    weights /= weights.sum()

    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a, fft_size)
    w_rev = weights[::-1]
    Wr = rfft(w_rev, fft_size)
    conv = irfft(A * Wr, fft_size)
    cross = conv[n-1:2*n-1][::-1]

    grad = 2.0 * cross
    obj = np.log(np.sum(weights)) / s + b_max
    return obj, grad


def lp_refine_aggressive(f, time_limit=120.0, max_iters=100, verbose=False):

                    return best_obj + 1.0

                return float(fast_convolve(blend).max())

    v = np.zeros(n)

    cur_lr = lr_init
    no_improve = 0
    kick_size = 0.005

    while time.monotonic() - t0 < time_limit:
        _, b_max, b = compute_subgradient_exp(a, s=s_exp)
        # Recompute gradient properly
        grad, _, _ = compute_subgradient_exp(a, s=s_exp)

        # Nesterov step
        v = momentum * v - cur_lr * grad
        a_new = project_simplex(a + v)

        obj_new = float(fast_convolve(a_new).max())

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive strategy

            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.03)
            no_improve = 0
            kick_size *= 0.6
            if kick_size < 1e-5:
                break
        elif no_improve >= 1000:
            cur_lr *= 0.93
        elif no_improve >= 500:
