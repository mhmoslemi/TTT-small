def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)

    """Compute 2*n * (sum conv^p)^(1/p) / (sum(a))^2 as a smooth proxy for the true objective."""

    # d/d a_i of (sum conv^p)^(1/p) = (1/p) * (sum conv^p)^(1/p - 1) * p * conv^{p-1} * d conv / d a_i
    # d conv[k] / d a[i] = a[k-i] (for valid k-i)
    # So d/d a_i of (sum conv^p)^(1/p) = (sum conv^p)^(1/p-1) * sum_k conv[k]^{p-1} * a[k-i]
    # = (sum conv^p)^(1/p-1) * cross_corr(conv^{p-1}, a)[i]

    conv_pw = conv ** (p - 1)
    W = rfft(conv_pw, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]

    base = (np.sum(conv ** p)) ** (1.0 / p - 1.0)
    grad_a = 2.0 * n * base * corr

    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def adam_logspace_pmean(a0, time_limit, p_init=8.0, p_final=64.0, T_schedule='exp',
                        lr=0.008, beta1=0.9, beta2=0.999, eps=1e-8,
                        restart_every=2500, verbose=False, verbose_interval=3000):
    """
    Adam optimizer in log-space with power-mean smoothing and adaptive p.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Adaptive p: start small (smoother), increase (sharper)
        if T_schedule == 'exp':
            p = p_init * (p_final / p_init) ** progress
        elif T_schedule == 'linear':
            p = p_init + (p_final - p_init) * progress
        else:
            p = p_init + (p_final - p_init) * progress * progress

        a_curr = softmax_fn(z)

        # Compute gradient
        try:
            grad_z = grad_power_mean(z, p, n, fft_size)
        except Exception:
            break

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate with true objective periodically
        if t % 50 == 0 or t == 1:
            a_new = softmax_fn(z)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 0.01:
                score = float(evaluate_sequence(a_new.tolist()))
                if np.isfinite(score) and score < best_score - 1e-12:
                    best_score = score
                    best_a = a_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        # Restart
        if no_improve > restart_every // 50:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.02
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.0005, lr * 0.8)

        if verbose and t % verbose_interval == 0:
            a_check = softmax_fn(z)
            s = a_check.sum()
            if s > 0.01:
                print(f"      Adam t={t} p={p:.1f} lr={lr:.6f} best={best_score:.6f}")

    return best_a, best_score


def adam_softmax_logsumexp(a0, time_limit, T_init=0.1, T_final=0.005, lr=0.006,
                           beta1=0.9, beta2=0.999, eps=1e-8,
                           restart_every=2000, verbose=False):
    """
    Adam with log-sum-exp smoothing (equivalent to soft-max with temperature).
    Different from power-mean approach - uses exp smoothing directly.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_fn(z)
        conv = fast_convolve(a_curr)
        max_b = conv.max()
        exp_b = np.exp((conv - max_b) / T)
        w = exp_b / exp_b.sum()

        # Gradient of 2n * (T * (max_b/T + log(sum exp((conv-max_b)/T)))) w.r.t. a
        # = 2n * sum_k w[k] * d conv[k] / d a[i]
        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr

        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 50 == 0 or t == 1:
            a_new = softmax_fn(z)
            s = a_new.sum()
            if s > 0.01:
                score = float(evaluate_sequence(a_new.tolist()))
                if np.isfinite(score) and score < best_score - 1e-12:
                    best_score = score
                    best_a = a_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        if no_improve > restart_every // 50:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.015
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.0003, lr * 0.75)

        if verbose and t % 3000 == 0:
            print(f"      LSE t={t} T={T:.5f} lr={lr:.6f} best={best_score:.6f}")

    return best_a, best_score


def peak_shave_v2(a0, time_limit, n_trials=10000):
    """
    Enhanced peak shaving: identifies ALL peaks above target and tries to reduce them
    by adjusting the top contributing pairs simultaneously.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    target = best_score / (2.0 * n)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        # Find all peaks significantly above target
        peaks = np.where(conv > target * 1.005)[0]
        if len(peaks) == 0:
            # Try to find any improvement
            peak_k = int(np.argmax(conv))
            peaks = np.array([peak_k])

        improved = False
        for peak_k in peaks[:5]:
            peak_val = conv[peak_k]
            js = max(0, int(peak_k) - n + 1)
            je = min(n - 1, int(peak_k))
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[int(peak_k) - idx]
            top_idx = np.argsort(contribs)[::-1][:8]

            for ci in top_idx:
                i = int(idx[ci])
                j = int(peak_k) - i
                if j < 0 or j >= n or i == j:
                    continue
                if a[i] < 1e-15 or a[j] < 1e-15:
                    continue

                for f1 in [0.92, 0.88, 0.84, 0.78]:
                    for f2 in [1.0, 1.03, 1.06, 1.10]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = float(evaluate_sequence(a_trial.tolist()))
                        if np.isfinite(score) and score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            target = best_score / (2.0 * n)
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and trial % 200 == 199:
            # Periodic perturbation
            a = best_a + np.random.standard_normal(n) * 0.008
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-10:
                a = a / ss

    return best_a, best_score


def coordinate_descent_v2(a0, time_limit, n_sweeps=100):
    """
    Coordinate descent with golden-section-like search for optimal value of each coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_best = best_score

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-18:
                continue

            # Try several factors
            best_val = a[i]
            best_loc_score = best_score

            for factor in [0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.2]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = float(evaluate_sequence(a_trial.tolist()))
                if np.isfinite(score) and score < best_loc_score:
                    best_loc_score = score
                    best_val = a[i] * factor

            if best_val != a[i]:
                a[i] = best_val
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                    if best_loc_score < best_score:
                        best_score = best_loc_score
                        best_a = a.copy()

        if best_score < sweep_best - 1e-14:
            sweep_best = best_score
        else:
            # No improvement in this sweep
            if sweep > 10:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def chebyshev_init(n, order=3):
    """Initialize using Chebyshev polynomial absolute value (creates oscillating structure)."""
    x = np.linspace(-1, 1, n)
    for _ in range(order):
        x = 2 * x * x - 1
    a = np.abs(x) + 0.01
    a = a / a.sum()
    return a


def polynomial_init(n, degree=4):
    """Initialize using a polynomial shape."""
    x = np.linspace(0, 1, n)
    a = 1.0
    for d in range(1, degree + 1):
        a = a * x + (1.0 - x) * 0.5
    # Mix with a concave shape
    concave = 1.0 - x * x
    a = 0.5 * a + 0.5 * concave
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def bimodal_optim_init(n):
    """Initialize with two peaks - optimal for reducing center convolution."""
    x = np.linspace(0, 1, n)
    # Two Gaussians at 0.25 and 0.75
    a = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def sawtooth_init(n):
    """Sawtooth pattern - creates destructive interference in convolution."""
    x = np.linspace(0, 1, n)
    a = np.abs(np.sin(np.pi * x * 3)) + 0.1
    a = a / a.sum()
    return a


def diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    candidates = []

    if parent is not None and len(parent) == n:
        p = np.maximum(parent, 0)
        p = p / p.sum()
        candidates.append(('parent', p.copy()))
        for scale in [0.02, 0.05, 0.1, 0.2]:
            pert = p + rng.standard_normal(n) * scale
            pert = np.maximum(pert, 0)
            sp = pert.sum()
            if sp > 1e-10:
                pert = pert / sp
                candidates.append((f'pert_{scale}', pert))

    candidates.append(('uniform', np.ones(n) / n))
    candidates.append(('cheb3', chebyshev_init(n, 3)))
    candidates.append(('cheb5', chebyshev_init(n, 5)))
    candidates.append(('poly4', polynomial_init(n, 4)))
    candidates.append(('bimodal', bimodal_optim_init(n)))
    candidates.append(('sawtooth', sawtooth_init(n)))
    candidates.append(('gauss', np.exp(-((np.linspace(-2, 2, n)) ** 2) / 2) / np.exp(-((np.linspace(-2, 2, n)) ** 2) / 2).sum()))

    # Triangular (concave)
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(float)
    tri = tri / tri.sum()
    candidates.append(('triangular', tri))

    return candidates


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining power-mean Adam, log-sum-exp Adam,
    peak shaving, and coordinate descent across multiple sequence lengths.

    parent = None

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best score: {best_score:.6f}")

    # === Phase 1: Multi-length exploration (35% time) ===
    print(f"\n=== Phase 1: Multi-Length Exploration ===")
    t_p1 = min(630.0, (deadline - time.monotonic()) * 0.35)
    t_p1_start = time.monotonic()

    lengths_to_try = [n_parent]
    for factor in [1.25, 1.5, 2.0]:
        alt_n = int(n_parent * factor)
        if 512 <= alt_n <= 3072 and alt_n not in lengths_to_try:
            lengths_to_try.append(alt_n)
    lengths_to_try = lengths_to_try[:4]

    for alt_n in lengths_to_try:
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break

        t_alloc = min(rem * 0.3, 150.0)
        print(f"  Length n={alt_n}, budget={t_alloc:.0f}s")

        if alt_n == n_parent and parent is not None:
            init_a = parent
        else:
            init_a = interpolate_sequence(best_seq, alt_n)

        inits = diverse_inits(alt_n, init_a, rng)
        # Score and sort
        inits_scored = []
        for label, ia in inits:
            s = 2.0 * alt_n * fast_convolve(ia).max()
            inits_scored.append((s, label, ia))
        inits_scored.sort(key=lambda x: x[0])

        # Run power-mean Adam on top 2-3
        n_starts = min(3, len(inits_scored))
        t_per_start = t_alloc / n_starts

        for idx in range(n_starts):
            rem = deadline - time.monotonic() - 350
            if rem < 30:
                break
            t_sub = min(t_per_start, rem * 0.8)
            if t_sub < 20:
                continue

            s_init, label, ia = inits_scored[idx]
            print(f"    Start: {label} (init={s_init:.6f}), t={t_sub:.0f}s")

            a_res, _ = adam_logspace_pmean(
                ia, t_sub, p_init=6.0, p_final=48.0,
                T_schedule='exp', lr=0.007, restart_every=2000, verbose=False
            )
            s = a_res.sum()
            if s > 0.01:
                sc = float(evaluate_sequence(a_res.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_res.copy()
                    print(f"    NEW BEST: {sc:.6f} (n={alt_n}, {label})")

    # === Phase 2: Focused refinement on best (30% time) ===
    print(f"\n=== Phase 2: Focused Refinement (best={best_score:.6f}) ===")

    # === Phase 3: L-BFGS polish (10% time) ===
    print(f"\n=== Phase 3: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100

    if rem > 30:
        n_best = len(best_seq)
        t_sub = min(rem * 0.6, 80.0)
        print(f"  L-BFGS, t={t_sub:.0f}s")

        z0 = z_from_a(best_seq)
        fft_size = 1 << (2 * n_best - 2).bit_length()
        t_start = time.monotonic()
        stopped = [False]
        T_lbfgs = 30.0  # Temperature for the LSE smoothing in L-BFGS

        def lbfgs_obj(z):
            if time.monotonic() - t_start > t_sub:
                stopped[0] = True
                return 1e15
            a = softmax_fn(z)
            conv = fast_convolve(a)
            max_b = conv.max()
            exp_b = np.exp((conv - max_b) / T_lbfgs)
            return 2.0 * n_best * T_lbfgs * (max_b / T_lbfgs + np.log(exp_b.sum() / len(conv)))

        def lbfgs_grad(z):
            if time.monotonic() - t_start > t_sub:
                stopped[0] = True
                return np.zeros_like(z)
            a = softmax_fn(z)
            conv = fast_convolve(a)
            max_b = conv.max()
            exp_b = np.exp((conv - max_b) / T_lbfgs)
            w = exp_b / exp_b.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            corr = irfft(W * np.conj(A), fft_size)[:n_best]
            grad_a = 2.0 * n_best * corr
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=lbfgs_obj, x0=z0, jac=lbfgs_grad,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxfun': 100000, 'disp': False}
            )
            a_lb = softmax_fn(res.x)
            s = a_lb.sum()
            if s > 0.01:
                sc = float(evaluate_sequence(a_lb.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_lb.copy()
                    print(f"  L-BFGS new best: {sc:.6f}")
        except Exception as e:
            print(f"  L-BFGS failed: {e}")

    # === Phase 4: Intensive local polishing (20% time) ===
    print(f"\n=== Phase 4: Local Polishing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 60

    if rem > 30:
        t_sub = min(rem * 0.4, 80.0)
        print(f"  Peak shave v2, t={t_sub:.0f}s")
        a_ps, _ = peak_shave_v2(best_seq, t_sub, n_trials=15000)
        s = a_ps.sum()
        if s > 0.01:
            sc = float(evaluate_sequence(a_ps.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_ps.copy()
                print(f"  Peak shave new best: {sc:.6f}")

    rem = deadline - time.monotonic() - 40
    if rem > 20:
        t_sub = min(rem * 0.5, 60.0)
        print(f"  Coordinate descent, t={t_sub:.0f}s")
        a_cd, _ = coordinate_descent_v2(best_seq, t_sub, n_sweeps=50)
        s = a_cd.sum()
        if s > 0.01:
            sc = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_cd.copy()
                print(f"  Coord descent new best: {sc:.6f}")

    # Another peak shave after coord descent
    rem = deadline - time.monotonic() - 20
    if rem > 15:
        t_sub = min(rem * 0.6, 40.0)
        a_ps2, _ = peak_shave_v2(best_seq, t_sub, n_trials=8000)
        s = a_ps2.sum()
        if s > 0.01:
            sc = float(evaluate_sequence(a_ps2.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_ps2.copy()
                print(f"  Peak shave 2 new best: {sc:.6f}")

    # === Phase 5: Final Adam with very low temperature ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 5

    if rem > 15:
        t_sub = min(rem * 0.5, 30.0)
        a_fin, _ = adam_logspace_pmean(
            best_seq, t_sub, p_init=16.0, p_final=128.0,
            T_schedule='linear', lr=0.002, restart_every=1000, verbose=False
        )
        s = a_fin.sum()
        if s > 0.01:
            sc = float(evaluate_sequence(a_fin.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_fin.copy()
                print(f"  Final Adam new best: {sc:.6f}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
