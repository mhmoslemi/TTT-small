Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play combined
with Adam log-space gradient descent and peak-targeted surgery. The LP approach
finds the "best response" sequence that maximizes mass while keeping the
convolution with the current sequence bounded, providing a different optimization
dynamic than pure gradient descent.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score_normalized(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * fast_convolve(a).max() / (s * s)


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    diff = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (diff >= 0) & (diff < n)
    # Use np.take with where-safe indexing
    safe_diff = np.clip(diff, 0, n - 1)
    A = np.where(mask, a[safe_diff], 0.0)
    return A


def solve_lp_fictitious(a, M):
    """

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    return None

        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'hold_anneal':

                T = T_init
            else:

                T = T_init * (T_final / T_init) ** p2

        grad_z = a_curr * (grad_a - S)

        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 3000:
            z = np.log(np.maximum(best_a, 1e-300)) + np.random.standard_normal(n) * 0.005
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play as the
primary optimization driver (exploiting the duality between the LP relaxation
and soft-max gradient), combined with Adam log-space gradient descent for
polishing and peak-targeted surgery for final refinement. The LP approach
provides a different optimization dynamic than pure gradient descent by
computing exact best responses under the l-infinity convolution constraint.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score_normalized(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 for normalized a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * fast_convolve(a).max() / (s * s)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    diff = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (diff >= 0) & (diff < n)
    safe_idx = np.where(mask, diff, 0)
    A = np.where(mask, a[safe_idx], 0.0)
    return A


def solve_lp_best_response(a, M):
    """
    Solve LP: max sum(b) s.t. (a*b)(k) <= M for all k, b >= 0.
    Returns optimal b vector or None.
    """
    n = len(a)
    A_ub = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None


def solve_lp_restricted(a, M, top_k=30):
    """
    Restricted LP: only enforce top-k peak constraints.
    Faster but may need violated constraint checking.
    """
    n = len(a)
    conv = fast_convolve(a)
    # Find the k indices with highest (a*b) potential
    # Use current peak positions as proxy
    peak_indices = np.argsort(conv)[::-1][:top_k]
    
    A_ub = np.zeros((len(peak_indices), n))
    for row, k in enumerate(peak_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[row, j] = a[k - j]
    
    b_ub = np.full(len(peak_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='exp'):
    """Adam optimizer in log-space with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-300))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    lr_current = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        elif schedule == 'cosine':
            T = T_final + 0.5 * (T_init - T_final) * (1 + np.cos(np.pi * progress))
        else:
            T = T_init

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(np.clip(b / T - max_b / T, -50, 50))
        w = exp_b / exp_b.sum()

        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr

        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr_current * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            z = np.log(np.maximum(best_a, 1e-300)) + np.random.standard_normal(n) * 0.008
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr_current = max(0.001, lr_current * 0.6)

        lr_current *= 0.99995

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=100, maxiter=3000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    stopped = [False]

    def soft_max_obj(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            stopped[0] = True
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(np.clip(b / T - max_b / T, -50, 50))
        return 2.0 * n * (T * (max_b / T + np.log(exp_b.sum())))

    def soft_max_grad(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(np.clip(b / T - max_b / T, -50, 50))
        w = exp_b / exp_b.sum()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=soft_max_obj, x0=z0, jac=soft_max_grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()
    return a_out


def peak_shave(a0, time_limit, n_trials=20000):
    """Peak-targeted local search with pair adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue
            for f1 in [0.98, 0.95, 0.9, 0.85, 0.8]:
                for f2 in [1.0, 1.01, 1.03, 1.05]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for ci in top_idx[:4]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in [0.92, 0.85, 0.75]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and trial > 200 and trial % 100 == 0:
            a = best_a + np.random.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def coordinate_descent(a0, time_limit):
    """Coordinate descent with multiplicative updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        sweep_improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            best_local = a[i]
            best_local_score = best_score
            for factor in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor
            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True
        if not sweep_improved:
            break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def lp_fictitious_play(a_init, time_limit, verbose=True):
    """
    Main LP-based fictitious play loop.
    At each step: solve LP for best response, update sequence.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    M = best_score / (2.0 * n)
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0

    # Pre-allocate
    diff = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (diff >= 0) & (diff < n)
    safe_idx = np.where(mask, diff, 0)

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Build constraint matrix
        A_ub = np.where(mask, a[safe_idx], 0.0)

        # Solve LP
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            res = type('R', (), {'success': False})()

        if not res.success:
            # Perturb and retry
            a = a + np.random.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            no_improve += 1
            if no_improve > 200:
                break
            continue

        b_star = res.x
        b_sum = b_star.sum()
        if b_sum < 1e-15:
            break
        g = b_star / b_sum

        # Adaptive step size: try a few values
        current_score = 2.0 * n * fast_convolve(a).max()
        best_trial_score = current_score
        best_trial_a = a.copy()

        # Anneal step size over iterations
        progress = min(1.0, iter_count / max(1, (time_limit / 0.1)))
        base_t = 0.3 * (1.0 - 0.5 * progress) + 0.05

        for t in [base_t * 0.5, base_t, base_t * 1.5, base_t * 2.0]:
            t = min(t, 0.9)
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_trial_score:
                best_trial_score = score
                best_trial_a = a_trial.copy()

        if best_trial_score < current_score - 1e-12:
            a = best_trial_a
            M = fast_convolve(a).max()
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                # Stuck: perturb
                a = best_a + np.random.standard_normal(n) * 0.015
                a = np.maximum(a, 0)
                a = a / a.sum()
                M = fast_convolve(a).max()
                no_improve = 0

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: t={elapsed:.0f}s, M={M:.6f}, "
                  f"score={get_score_normalized(a):.6f}, best={best_score:.6f}")

    return best_a, best_score


def diverse_initializations(parent, n, rng):
    """Generate diverse starting points."""
    candidates = []

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            for scale in [0.02, 0.05, 0.1]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                sp = pert.sum()
                if sp > 1e-10:
                    candidates.append((f'pert_{scale}', pert / sp))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('parent_interp', interp))

    candidates.append(('uniform', np.ones(n) / n))

    # Asymmetric ramp (important: asymmetry helps!)
    ramp = np.linspace(0.1, 1.0, n)
    candidates.append(('ramp_up', ramp / ramp.sum()))

    ramp2 = np.linspace(1.0, 0.1, n)
    candidates.append(('ramp_down', ramp2 / ramp2.sum()))

    # Bimodal
    mid = n // 2
    bi = np.ones(n) * 0.3
    bi[:n // 5] = 1.0
    bi[-n // 5:] = 1.0
    candidates.append(('bimodal', bi / bi.sum()))

    # Cosine-weighted
    x = np.linspace(0, np.pi, n)
    cos_w = np.abs(np.cos(x)) + 0.1
    candidates.append(('cosine', cos_w / cos_w.sum()))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.05
    candidates.append(('random', rand / rand.sum()))

    return candidates


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: LP Fictitious Play (40% time) - novel direction-finding
    Phase 2: Adam Gradient Descent with annealing (25% time)
    Phase 3: L-BFGS + Peak Surgery (20% time)
    Phase 4: Multi-length exploration + final polish (15% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n = len(best_seq) if best_seq is not None else 768

    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default uniform n={n}: score={best_score:.6f}")

    print(f"Starting best: {best_score:.6f} (n={n})")
    total_budget = float(budget_s) - 15.0

    # === Phase 1: LP Fictitious Play (40% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Fictitious Play (n={n}, budget={0.40*total_budget:.0f}s)")
    print(f"{'='*60}")
    t_p1 = 0.40 * (deadline - time.monotonic())
    t_p1_end = time.monotonic() + t_p1

    # Run LP from parent
    a_res, score_res = lp_fictitious_play(best_seq, t_p1 * 0.6, verbose=True)
    best_seq, best_score = evaluate_and_track(a_res, "lp_parent", best_seq, best_score)
    print(f"  After LP from parent: best={best_score:.6f}")

    # Run LP from a diverse initialization if time remains
    rem_p1 = t_p1_end - time.monotonic()
    if rem_p1 > 60:
        inits = diverse_initializations(parent, n, rng)
        # Pick the best non-parent init
        for label, init_a in inits:
            if label == 'parent':
                continue
            s_init = 2.0 * n * fast_convolve(init_a).max()
            if s_init < 1.8:  # Only try promising inits
                t_sub = min(rem_p1 * 0.4, 120.0)
                a_res2, _ = lp_fictitious_play(init_a, t_sub, verbose=False)
                best_seq, best_score = evaluate_and_track(a_res2, f"lp_{label}", best_seq, best_score)
                break

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Adam Gradient Descent (25% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Adam Gradient Descent (budget={0.25*total_budget:.0f}s)")
    print(f"{'='*60}")
    t_p2 = 0.25 * (deadline - time.monotonic())
    t_p2_end = time.monotonic() + t_p2

    # Multiple Adam runs with different schedules
    adam_configs = [
        ('hold_anneal', 0.04, 0.002, 0.008),
        ('exp', 0.03, 0.001, 0.006),
        ('cosine', 0.05, 0.001, 0.007),
        ('hold_anneal', 0.02, 0.0005, 0.005),
    ]

    n_configs = len(adam_configs)
    for idx, (sched, T_i, T_f, lr_val) in enumerate(adam_configs):
        rem = t_p2_end - time.monotonic()
        if rem < 30:
            break
        t_sub = min(rem / (n_configs - idx), rem * 0.7)
        if t_sub < 20:
            continue
        print(f"  Adam config {idx+1}/{n_configs}: sched={sched}, T=[{T_i},{T_f}], lr={lr_val}")
        a_res, score_res = adam_logspace(
            best_seq, t_sub, T_init=T_i, T_final=T_f,
            lr=lr_val, schedule=sched
        )
        best_seq, best_score = evaluate_and_track(a_res, f"adam_{sched}_{idx}", best_seq, best_score)

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: L-BFGS + Peak Surgery (20% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS + Peak Surgery (budget={0.20*total_budget:.0f}s)")
    print(f"{'='*60}")
    t_p3 = 0.20 * (deadline - time.monotonic())
    t_p3_end = time.monotonic() + t_p3

    # L-BFGS at multiple temperatures
    for T in [300, 100, 30]:
        rem = t_p3_end - time.monotonic()
        if rem < 20:
            break
        t_sub = min(rem * 0.25, 60.0)
        a_lb = lbgfs_polish(best_seq, t_sub, T=T, maxiter=2000)
        best_seq, best_score = evaluate_and_track(a_lb, f"lbgfs_T{T}", best_seq, best_score)
        print(f"  L-BFGS T={T}: {best_score:.6f}")

    # Peak shaving
    rem = t_p3_end - time.monotonic()
    if rem > 30:
        t_sub = min(rem * 0.6, 300.0)
        a_ps, _ = peak_shave(best_seq, t_sub, n_trials=30000)
        best_seq, best_score = evaluate_and_track(a_ps, "peak_shave", best_seq, best_score)
        print(f"  Peak shave: {best_score:.6f}")

    # Coordinate descent
    rem = t_p3_end - time.monotonic()
    if rem > 20:
        t_sub = min(rem * 0.7, 120.0)
        a_cd, _ = coordinate_descent(best_seq, t_sub)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_descent", best_seq, best_score)
        print(f"  Coord descent: {best_score:.6f}")

    # Another round of peak shaving
    rem = t_p3_end - time.monotonic()
    if rem > 15:
        t_sub = min(rem * 0.8, 100.0)
        a_ps2, _ = peak_shave(best_seq, t_sub, n_trials=20000)
        best_seq, best_score = evaluate_and_track(a_ps2, "peak_shave_2", best_seq, best_score)

    print(f"  Phase 3 done: best={best_score:.6f}")

    # === Phase 4: Multi-length + Final Polish (15% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length + Final Polish (budget={0.15*total_budget:.0f}s)")
    print(f"{'='*60}")
    t_p4 = 0.15 * (deadline - time.monotonic())
    t_p4_end = time.monotonic() + t_p4

    # Try nearby lengths
    n_best = len(best_seq)
    alt_lengths = []
    for factor in [1.25, 1.5, 0.75, 2.0, 1.1, 0.8]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            if alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:3]:
        rem = t_p4_end - time.monotonic()
        if rem < 40:
            break
        t_sub = min(rem * 0.3, 100.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = adam_logspace(init_alt, t_sub, T_init=0.04, T_final=0.002,
                                 lr=0.007, schedule='hold_anneal')
        best_seq, best_score = evaluate_and_track(a_alt, f"alt_n{alt_n}", best_seq, best_score)
        print(f"  Alt n={alt_n}: {best_score:.6f}")

    # Final intensive Adam
    rem = t_p4_end - time.monotonic()
    if rem > 30:
        t_sub = min(rem * 0.4, 60.0)
        a_fin, _ = adam_logspace(best_seq, t_sub, T_init=0.01, T_final=0.0003,
                                 lr=0.003, schedule='exp')
        best_seq, best_score = evaluate_and_track(a_fin, "final_adam", best_seq, best_score)

    # Final peak shave
    rem = t_p4_end - time.monotonic()
    if rem > 15:
        t_sub = min(rem * 0.5, 40.0)
        a_ps_fin, _ = peak_shave(best_seq, t_sub, n_trials=10000)
        best_seq, best_score = evaluate_and_track(a_ps_fin, "final_ps", best_seq, best_score)

    # === Final verification ===
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
        print(f"{'='*60}")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.6f}")

    return [float(x) for x in best_seq]
