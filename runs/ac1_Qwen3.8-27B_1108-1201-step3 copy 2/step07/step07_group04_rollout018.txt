"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play for initialization
2. FFT-accelerated FISTA with multi-temperature annealing (O(n log n) per step)
3. Direct subgradient with peak focusing and momentum
4. Targeted peak shaving
5. Multi-length exploration
Key innovation: FFT cross-correlation for gradient, enabling 50x+ more
iterations than O(n^2) subgradient methods.
"""

import numpy as np
from scipy.fft import rfft, irfft
from time import monotonic
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(w, a):
    """
    grad[i] = 2 * sum_k w[k] * a[k-i] * (0<=k-i<n)
    via FFT cross-correlation. O(n log n).
    """
    n = len(a)
    m = len(w)
    N = 1 << (m + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T: T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """LP: given a, find b>=0, sum(b)=1, min max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None):
    """Fictitious play for initialization."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n
    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0
    while monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        for cand in [a, b, 0.5 * (a + b)]:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
    return best_a, best_score, iters


def fista_smooth(a0, time_limit, temps=None, verbose=False):
    """
    FISTA on T*logsumexp(conv(a,a)/T) on simplex.
    FFT-based gradient: O(n log n) per iteration.
    """
    n = len(a0)
    a = np.maximum(a0, 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    if temps is None:
        temps = [100, 50, 20, 10, 5, 2, 1, 0.5, 0.25, 0.1]

    N_fft = 1 << (3 * n - 1).bit_length()
    total_time_per_T = time_limit / len(temps)

    for Ti, T in enumerate(temps):
        t_start_T = monotonic()
        a_curr = a.copy()
        y = a_curr.copy()
        t_fista = 1.0
        lr = max(T * 0.5 / n, 0.01)

        while monotonic() - t_start_T < total_time_per_T:
            if monotonic() - t0 > time_limit:
                return best_a, best_score

            # Compute b = conv(y, y)
            fy = np.maximum(y, 0)
            fy = fy / fy.sum() if fy.sum() > 1e-15 else np.ones(n) / n
            b = fast_convolve(fy)
            b = np.maximum(b, 0)

            # Softmax weights
            bs = b / T
            bs -= bs.max()
            w = np.exp(bs)
            ws = w.sum()
            w = w / ws if ws > 1e-30 else np.ones(2 * n - 1) / (2 * n - 1)

            # Gradient via FFT
            W = rfft(w, N_fft)
            Ay = rfft(fy, N_fft)
            grad = 2.0 * irfft(W * np.conj(Ay), N_fft)[:n]

            # Gradient step + projection
            a_new = project_simplex(fy - lr * grad)

            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
            y = a_new + ((t_fista - 1.0) / t_new) * (a_new - a_curr)
            y = project_simplex(y)

            a_curr = a_new
            t_fista = t_new

            # Periodic evaluation of true objective
            sc = 2.0 * n * fast_convolve(a_curr).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_curr.copy()

            # Adapt learning rate
            lr *= 0.99999

    return best_a, best_score


def subgradient_direct(a0, time_limit, top_k=50):
    """
    Direct subgradient on max_k (a*a)[k] with momentum.
    Uses top-k peaks for gradient approximation.
    """
    n = len(a0)
    a = np.maximum(a0, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = 0.005
    momentum = 0.9
    t0 = monotonic()
    no_improve = 0
    it = 0

    while monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws if ws > 1e-30 else np.ones(len(peaks)) / len(peaks)

        # Gradient: sum_k w[k] * 2*a[k-i] for valid i
        N_fft = 1 << (3 * n - 1).bit_length()
        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = w
        W = rfft(w_full, N_fft)
        A = rfft(a, N_fft)
        grad = 2.0 * irfft(W * np.conj(A), N_fft)[:n]

        v = momentum * v - lr * grad
        a_trial = project_simplex(a + v)

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 500:
            a = best_a + np.random.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.004)
            no_improve = 0

        if it % 2000 == 0:
            print(f"    SubGrad it={it} t={monotonic()-t0:.0f}s score={best_score:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit):
    """Targeted peak shaving: reduce the largest autocorrelation peak."""
    n = len(a0)
    a = np.maximum(a0, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    stalled = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)

        if peak_val <= target + 1e-14:
            break

        # Find dominant contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_c = contribs.sum()
        if total_c < 1e-20:
            break

        improved = False
        # Try reducing top contributors
        top_idx = np.argsort(contribs)[::-1][:8]
        for ci in top_idx:
            i = int(idx[ci])
            old = a[i]
            if old < 1e-14:
                continue
            for f in [0.92, 0.85, 0.75]:
                a_trial = a.copy()
                diff = old * (1 - f)
                a_trial[i] = old * f
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pairwise
            for ci in top_idx[:4]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8]:
                    for f2 in [0.9, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = a_trial / a_trial.sum()
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Small random perturbation
            a_trial = a + np.random.standard_normal(n) * 0.0003
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-13:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True

        if not improved:
            stalled += 1
            if stalled > 100:
                break
        else:
            stalled = 0

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new


def check_update(a, label, best_seq, best_score):

def make_window(n, shape='parabolic'):
    """Create a non-negative window as initialization."""
    x = np.linspace(0, 1, n)
    if shape == 'parabolic':
        w = 1.0 - (2 * x - 1) ** 2
    elif shape == 'gaussian':
        w = np.exp(-2.0 * (2 * x - 1) ** 2)
    elif shape == 'hann':
        w = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif shape == 'raised_cosine':
        w = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        w = np.ones(n) / n
    w = np.maximum(w, 0)
    s = w.sum()
    return w / s if s > 1e-15 else np.ones(n) / n
