Let me verify: `correlate(w, f)[j] = sum_n w[n+j] * f[n]` (for real)
`= sum_n w[j+n] * f[n]` for n such that both indices are valid
`= sum_{n=0}^{n-1} w[j+n] * f[n]` (since f has length n and w has length 2n-1)

Yes! That's exactly what I want.

For the second term: `f[j] * sum_{i=0}^{min(n-1, 2n-2-j)} w[j+i]`
`= f[j] * (cumsum_w[min(j+n, 2n-1)] - cumsum_w[j])`

where `cumsum_w[k] = sum_{i=0}^{k-1} w[i]` (with cumsum_w[0] = 0).

So: `second_term = f * (cumsum_w[j+n] - cumsum_w[j])` (element-wise, with appropriate clamping)

Great, so the full gradient computation is:
1. `b = fast_convolve(f)` - O(n log n)
2. `w = softmax(b / temp)` - O(n)
3. `first_term = correlate(w, f, mode='full')[:n]` - O(n log n) via FFT
4. `csw = np.cumsum(w)` - O(n)
5. `second_term = f * (csw[np.minimum(j+n, 2n-1)] - csw[j])` - O(n)
6. `grad = first_term + second_term` - O(n)

Total: O(n log n) per iteration!

```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient(f, temp):
    """Compute gradient of smooth max autocorrelation."""
    n = len(f)
    b = fast_convolve(f)
    
    # Softmax weights
    b_scaled = b / temp
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    # First term: cross-correlation of w and f, first n elements
    first_term = correlate(w, f, mode='full')[:n]
    
    # Second term: f[j] * sum_{i=j}^{j+n-1} w[i]
    csw = np.zeros(len(w) + 1)
    csw[1:] = np.cumsum(w)
    j_idx = np.arange(n)
    lo = j_idx
    hi = np.minimum(j_idx + n, len(w))
    second_term = f * (csw[hi] - csw[lo])
    
    return first_term + second_term


def mirror_descent_solve(f_init, time_limit, temp_init=0.01, temp_final=0.001,
                         lr_init=0.01, momentum=0.9, label="md"):
    """Accelerated mirror descent for smooth max autocorrelation."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    v = np.zeros(n)
    temp = temp_init
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    
    total_time = time_limit
    # Anneal temperature over first 80% of time
    # Increase learning rate decay
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        grad = compute_gradient(f, temp)
        
        # Momentum update
        v = momentum * v - lr * grad
        f_new = f + v
        f_new = project_simplex(f_new)
        
        # Evaluate
        if it % 10 == 0:
            obj = fast_convolve(f_new).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        f = f_new
        
        # Annealing
        progress = (time.monotonic() - t0) / time_limit
        if progress < 0.8:
            temp = temp_init * (temp_final / temp_init) ** (progress / 0.8)
            lr = lr_init * (1.0 / (1.0 + 0.1 * it))
        else:
            temp = temp_final
            lr = lr_init * 0.1
        
        if no_improve > 100:
            # Restart with noise
            f = project_simplex(best_f + np.random.standard_normal(n) * 0.001)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} temp={temp:.6f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    v = np.asarray(v, dtype=np.float64)
    # Project onto {x >= 0, sum(x) = target_sum}
    # First project onto {x >= 0, sum(x) = 1}, then scale
    v_shifted = v - v.min()
    s = v_shifted.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    v_norm = v_shifted / s
    # Now project v_norm onto standard simplex
    u = np.sort(v_norm)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        result = np.ones(n) / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        result = np.maximum(v_norm - theta, 0.0)
    return result * target_sum
```

```python
def peak_targeting(f_init, time_limit, n_peaks=5, label="peak"):
    """Specifically target reduction of the worst peaks."""

    f = f_init.copy() / f_init.sum()

    best_obj = fast_convolve(f).max()

        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        top_idx = np.sort(top_idx)

        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                pressure[j] += f[k - j]

        f_new = f - lr * pressure
        f_new = project_simplex(f_new)
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
        else:
            lr *= 0.5
            f_new = f - lr * pressure
            f_new = project_simplex(f_new)
            f = f_new
    
    return best_f, best_obj
```

"""
Combined optimization: accelerated gradient descent + LP refinement + coordinate descent.
Uses FFT-based O(n log n) gradient computation for the smooth max autocorrelation objective.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    v = np.asarray(v, dtype=np.float64).copy()
    v = v - v.min()
    s = v.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    v = v / s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    return result * target_sum


def compute_gradient_smoothed(f, temp):
    n = len(f)
    b = fast_convolve(f)
    b_scaled = b / temp
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    first_term = correlate(w, f, mode='full')[:n]
    csw = np.zeros(len(w) + 1)
    csw[1:] = np.cumsum(w)
    j_idx = np.arange(n)
    lo = j_idx
    hi = np.minimum(j_idx + n, len(w))
    second_term = f * (csw[hi] - csw[lo])
    return first_term + second_term


def gradient_descent_phase(f_init, time_limit, label="gd"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 0:
        f /= s
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(999)
    
    # Adaptive temperature
    b0 = fast_convolve(f)
    temp_scale = b0.max()
    temp = temp_scale * 0.05
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive temperature: decrease over time
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        temp = temp_scale * (0.05 * (0.01 ** progress))
        temp = max(temp, temp_scale * 0.001)
        
        lr = 0.01 / (1.0 + 0.01 * it)
        
        grad = compute_gradient_smoothed(f, temp)
        v = 0.9 * v - lr * grad
        f_new = f + v
        f_new = project_simplex(f_new)
        
        if it % 5 == 0:
            obj = fast_convolve(f_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        f = f_new
        
        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.01
        
        if it % 5000 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def lp_solve(f, top_k, eps=1e-9):
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.CLARABEL, max_iters=500000, eps_abs=eps, eps_rel=eps)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
        except Exception:
            return None, M
    
    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def lp_phase(f_init, time_limit, top_k_init=40, top_k_max=300, label="lp"):
    t0 = time.monotonic()

    f = f_init.copy()
    s = f.sum()
    if s > 0:
        f /= s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)
        
        g, M = lp_solve(f, top_k)
        if g is None:

            if g is None:

        if S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k * 2, top_k_max)

        h = g / S_g
        best_t = 0.05

def coordinate_polish(f_init, time_limit, label="cd"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 0:
        f /= s
    best_obj = fast_convolve(f).max()
    
    # Order coordinates by "importance" (contribution to worst peak)
    b = fast_convolve(f)
    worst_k = b.argmax()
    lo = max(0, worst_k - n + 1)
    hi = min(n - 1, worst_k)
    importance = np.zeros(n)
    for j in range(lo, hi + 1):
        importance[j] = f[worst_k - j]
    order = np.argsort(-importance)
    
    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5]:
                new_val = f[j] * mult
                if new_val > 0.999 or new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s_t = f_trial.sum()
                if s_t < 1e-10:
                    continue
                f_trial /= s_t
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 0:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    
    return f, best_obj
