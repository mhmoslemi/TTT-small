"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-domain L-BFGS-B with log-sum-exp smoothed objective (quasi-Newton, superlinear convergence)
2. Adaptive temperature annealing for smooth→exact transition
3. LP-based refinement with sparse active-set constraints
4. Mirror descent (multiplicative) in log-domain for fast convergence
5. Coordinate descent polish
6. Multi-length exploration with diverse initializations

Key innovations over subgradient approach:
- Quasi-Newton (L-BFGS-B) converges superlinearly vs. 1/sqrt(T) for subgradient
- Log-domain parametrization handles non-negativity naturally
- Smoothed objective enables efficient gradient computation
- Mirror descent in log-domain as fast pre-optimizer before BFGS
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad(a, w):
    """
    Compute grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for j=0..n-1.
    w is the weighting on convolution lags (length 2n-1).
    """
    n = len(a)
    grad = np.zeros(n)
    active = np.nonzero(w)[0]
    for k in active:
        wk = w[k]
        # Part 1: a[k-j] for valid j
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += wk * a[k - hi:k - lo + 1][::-1]
        # Part 2: a[j-k] for valid j (only when k < n)
        if k < n:
            lo2 = k
            hi2 = n - 1
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] += wk * a[0:n - k]
    return grad


def softmax_stable(b):
    """Numerically stable softmax."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    return exp_b / exp_b.sum()


def smoothed_objective(b, T, n):
    """
    Compute F_T = 2n * T * log(sum_k exp(conv[k]/T)) and its gradient w.r.t. b.
    b is the log-domain parameter (length n).
    """
    a = softmax_stable(b)
    conv = fast_convolve(a)
    
    # Log-sum-exp for numerical stability
    max_conv = conv.max()
    log_sum_exp = max_conv / T + np.log(np.sum(np.exp((conv - max_conv) / T)))
    F = 2.0 * n * T * log_sum_exp
    
    # Weights for active lags
    w = np.exp((conv - max_conv) / T)
    w = w / w.sum()
    
    # Gradient of F w.r.t. a: 2n * sum_k w[k] * dconv_k/da[j]
    G = 2.0 * n * compute_grad(a, w)
    
    # Chain rule through softmax: dF/db[j] = a[j] * (G[j] - sum_i G[i]*a[i])
    H = np.dot(G, a)
    grad_b = a * (G - H)
    
    return F, grad_b


def mirror_descent_log(b0, time_limit, lr_init=0.01, n_top=20, T=0.01):
    """
    Mirror descent in log-domain (equivalent to multiplicative updates on a).
    Fast pre-optimizer before BFGS.
    """
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    n = len(b0)
    b = b0.copy().astype(np.float64)
    a = softmax_stable(b)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    lr = lr_init
    no_improve = 0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        b_max = conv.max()
        obj = 2.0 * n * b_max
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active peaks
        if n_top < len(conv):
            top_idx = np.argpartition(conv, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(conv))
        threshold = b_max * 0.999
        active_mask = conv[top_idx] >= threshold
        if not active_mask.any():
            active_mask[0] = True
        active_indices = top_idx[active_mask]
        
        w = np.zeros(2 * n - 1)
        w[active_indices] = 1.0 / len(active_indices)
        
        # Subgradient in a-domain
        G = compute_grad(a, w)
        
        # Mirror descent step in log-domain: b_new = b - lr * G (since d(log a)/da = 1/a, 
        # and the Bregman divergence for entropy gives multiplicative updates)
        # Actually, mirror descent with negative entropy: a_new = a * exp(-lr * G / a)
        # In log-domain: b_new = b - lr * G / a ... which is just b - lr * G since a = exp(b)/sum
        # Wait, for mirror descent with KL divergence: a_new[j] = a[j] * exp(-lr * G[j]) / Z
        # In log-domain: b_new[j] = b[j] - lr * G[j] (then re-normalize via softmax)
        b = b - lr * G
        a = softmax_stable(b)
        
        it += 1
        # Adaptive learning rate
        if no_improve > 500:
            lr *= 0.8
            no_improve = 0
        elif no_improve > 100:
            lr *= 0.95
        if it % 1000 == 0 and no_improve > 2000:
            # Restart with perturbation
            b = np.log(best_a + 1e-15) + rng.standard_normal(n) * 0.01
            a = softmax_stable(b)
            lr = lr_init * 0.5
            no_improve = 0
    
    return b, best_a, best_obj


def bfgs_optimize(b0, time_limit, T_init=0.05, T_min=0.001, T_decay=0.85, n_top=None):
    """
    L-BFGS-B optimization in log-domain with adaptive temperature.
    """
    t0 = time.monotonic()
    n = len(b0)
    b = b0.copy().astype(np.float64)
    a = softmax_stable(b)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    T = T_init
    it_between_restarts = 0
    
    while time.monotonic() - t0 < time_limit:
        # Define objective and gradient for scipy
        def obj_grad(b_val):
            a_val = softmax_stable(b_val)
            conv = fast_convolve(a_val)
            # Use log-sum-exp
            max_c = conv.max()
            if max_c < 1e-300:
                return 2.0 * n * max_c, np.zeros(n)
            lse = max_c / T + np.log(np.sum(np.exp((conv - max_c) / T)))
            F = 2.0 * n * T * lse
            w = np.exp((conv - max_c) / T)
            w = w / w.sum()
            G = 2.0 * n * compute_grad(a_val, w)
            H = np.dot(G, a_val)
            grad_b = a_val * (G - H)
            return F, grad_b
        
        # Run L-BFGS-B for a limited time
        deadline_phase = min(t0 + time_limit, time.monotonic() + 30.0)
        
        def callback(xk):
            return time.monotonic() > deadline_phase
        
        res = minimize(obj_grad, b, method='L-BFGS-B',
                       options={'maxiter': 500, 'maxfun': 200, 'callback': callback,
                                'gtol': 1e-8, 'ftol': 1e-14})
        
        if res.x is not None:
            a_new = softmax_stable(res.x)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
            
            b = res.x if res.x is not None else b
            it_between_restarts += 1
        
        # Check time
        if time.monotonic() >= t0 + time_limit - 5:
            break
        
        # Decrease temperature
        T = max(T_min, T * T_decay)
        
        # If no improvement for a while, add perturbation
        if it_between_restarts > 3:
            b = b + np.random.default_rng(it_between_restarts).standard_normal(n) * 0.02
            it_between_restarts = 0
    
    return b, best_a, best_obj


def lp_refine(f, max_iters=20, time_limit=60.0, verbose=False):
    """
    LP-based refinement: solve max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Uses sparse constraint matrix for efficiency.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n * fast_convolve(a / n).max()
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Only include constraints near the active set (within 10% of max)
        threshold = M * 0.95
        active_lags = np.nonzero(conv >= threshold)[0]
        
        # Also always include a few random lags for diversity
        if len(active_lags) < 5:
            extra = np.random.choice(2 * n - 1, size=min(5, 2 * n - 1), replace=False)
            active_lags = np.unique(np.concatenate([active_lags, extra]))
        
        # Build constraint matrix (sparse)
        num_constraints = len(active_lags)
        A = np.zeros((num_constraints, n))
        for i, k in enumerate(active_lags):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[i, k - j] = a[j]
        
        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            # Line search for blending
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 1e-15)
                blend = blend / blend.sum()
                obj_blend = 2.0 * n * fast_convolve(blend).max()
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t
            
            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                obj = 2.0 * n * fast_convolve(a).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.2f}, max_b={obj:.10f}")
                else:
                    a = best_a.copy()
            else:
                break
                
        except Exception:
            break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on the simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue
                
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                a_tr = a_tr / a_tr.sum()
                val = 2.0 * n * fast_convolve(a_tr).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    
    return best_a, best_obj


def make_asymmetric_init(n, center=0.45, sigma=0.2):
    """Asymmetric Gaussian initialization (shifted left for asymmetry)."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_gaussian_init(n, sigma=0.3):
    """Gaussian window initialization (symmetric)."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_linear_init(n, slope=0.5):
    """Linear ramp initialization (asymmetric)."""
    x = np.linspace(0, 1, n)
    a = 1.0 + slope * (2 * x - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_init(n, curvature=1.0):
    """Quadratic (parabolic) initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + curvature * (x - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-12)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def optimize_sequence(a_init, time_budget, seed=42, verbose=False):

        update_best(a_lp, "lp")
    
    # Stage 5: Coordinate descent
    rem = time_budget - (time.monotonic() - t0)
    if rem > 10:
        t_cd = min(rem * 0.5, 30.0)
        a_cd, obj_cd = coordinate_descent(best_a, time_limit=t_cd)
        update_best(a_cd, "cd")
    
    # Stage 6: Final LP + BFGS round
    rem = time_budget - (time.monotonic() - t0)
    if rem > 10:
        t_final = min(rem * 0.5, 30.0)
        a_final, obj_final = lp_refine(best_a, max_iters=10, time_limit=t_final)
        update_best(a_final, "final_lp")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = make_asymmetric_init(1152, center=0.45, sigma=0.2)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using asymmetric init n=1152, score={best_score:.6f}")
    
    n_parent = len(parent)
    
    # === Phase 1: Optimize parent ===
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Optimize parent (n={n_parent})")
    print(f"{'=' * 60}")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem * 0.6, 200.0)
        print(f"  Time allocated: {t_alloc:.0f}s")
        a_opt, obj_opt = optimize_sequence(parent, time_budget=t_alloc, seed=seed, verbose=True)
        update_best(a_opt, "phase1_parent")
    
    # === Phase 2: Diverse initializations ===
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Diverse Initializations (best={best_score:.6f})")
    print(f"{'=' * 60}")
    n_base = len(best_seq) if best_seq is not None else n_parent
    
    inits = {
        "asym_gauss_0.4": make_asymmetric_init(n_base, 0.4, 0.18),
        "asym_gauss_0.55": make_asymmetric_init(n_base, 0.55, 0.22),
        "asym_gauss_0.3": make_asymmetric_init(n_base, 0.3, 0.15),
        "sym_gauss_0.2": make_gaussian_init(n_base, 0.2),
        "sym_gauss_0.35": make_gaussian_init(n_base, 0.35),
        "cosine": make_cosine_init(n_base),
        "linear_0.5": make_linear_init(n_base, 0.5),
        "linear_1.0": make_linear_init(n_base, 1.0),
        "quadratic": make_quadratic_init(n_base, 1.5),
        "uniform": np.ones(n_base) / n_base,
    }
    
    total_time_diverse = min(350.0, (deadline - time.monotonic() - 300.0) * 0.45)
    time_per_init = total_time_diverse / len(inits)
    
    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 200
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.3)
        if t_alloc < 8:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = optimize_sequence(init_a, time_budget=t_alloc,
                                            seed=seed + hash(name) % 100000, verbose=False)
        update_best(a_div, f"div_{name}")

    print(f"Phase 3: Multi-length Exploration (best={best_score:.6f})")
    print(f"{'=' * 60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
            tl = int(n_parent * factor)
            tl = max(128, min(4096, tl))
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))
        
        for target_n in target_lengths[:5]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.25, 60.0)
            if t_alloc < 8:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = optimize_sequence(f_interp, time_budget=t_alloc,
                                                seed=seed + target_n, verbose=False)
            update_best(a_len, f"len_n{target_n}")

    print(f"\n{'=' * 60}")
    print(f"Phase 4: Deep Optimization (best={best_score:.6f})")
    print(f"{'=' * 60}")

        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_deep = min(rem * 0.6, 180.0)
            n_best = len(best_seq)
            print(f"  Deep: n={n_best}, t={t_deep:.0f}s")
            a_deep, obj_deep = optimize_sequence(best_seq, time_budget=t_deep,
                                                  seed=seed + 99999, verbose=True)

    print(f"\n{'=' * 60}")
    print(f"Phase 5: Final Polish (best={best_score:.6f})")

            if t_lp > 5:

                update_best(a_lp, "polish_lp")
            
            # Quick CD
            rem = deadline - time.monotonic() - 15
            if rem > 5:
                a_cd, obj_cd = coordinate_descent(best_seq, time_limit=min(rem, 20.0),
                                                   multipliers=np.array([0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]))
                update_best(a_cd, "polish_cd")
            
            # Final BFGS with very small T
            rem = deadline - time.monotonic() - 10
            if rem > 8:
                b_init = np.log(best_seq + 1e-15)
                b_f, a_f, obj_f = bfgs_optimize(b_init, time_limit=min(rem, 20.0),
                                                  T_init=0.005, T_min=0.001)
                update_best(a_f, "polish_bfgs")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'=' * 60}")
    
    return [float(x) for x in best_seq]
