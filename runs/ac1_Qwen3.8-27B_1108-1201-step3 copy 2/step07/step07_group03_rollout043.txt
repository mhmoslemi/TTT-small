Minimize 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative a.

- Start from the supplied parent if present, plus symmetrized, smoothed, and

  of one coordinate can be evaluated very cheaply without a full FFT.
- Escape local minima with multiplicative perturbations and with smooth
  log-sum-exp L-BFGS-B optimization in softmax coordinates.
- Optional LP-based fictitious play / alternating best responses to generate
  another basin.
- Limited two-coordinate peak-pair search for small joint moves.
- Final exact (np.convolve) verification among saved good candidates.
"""

import numpy as np
import time
import scipy.optimize as sopt
import scipy.sparse as sp
from scipy.fft import rfft, irfft


MULTS_LARGE = np.array(
    [0.0, 0.05, 0.1, 0.2, 0.3, 0.5, 0.75, 0.9, 1.0,
     1.1, 1.25, 1.5, 2.0, 3.0, 5.0, 8.0, 12.0], dtype=np.float64
)
MULTS_SMALL = np.array(
    [0.0, 0.1, 0.25, 0.5, 0.75, 1.0,
     1.25, 1.6, 2.2, 3.2, 5.0, 8.0], dtype=np.float64
)
ABS_FACTORS = np.array([0.1, 0.25, 0.5, 1.0, 2.0], dtype=np.float64)
PAIR_MULTS = np.array(
    [0.0, 0.5, 0.75, 0.9, 1.0, 1.1, 1.3, 1.7], dtype=np.float64
)


def fft_conv(a):
    n = int(len(a))
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = rfft(a, fsize)
    return irfft(A * A, fsize)[:size]


def normalize_sequence(a):
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return None
    return a / s


def fast_score(a):
    if a is None:
        return np.inf
    an = normalize_sequence(a)
    if an is None:
        return np.inf
    b = fft_conv(an)
    peak = float(b.max())
    sc = 2.0 * float(len(an)) * peak
    if not np.isfinite(sc):
        return np.inf
    return sc


def exact_score(a):
    if a is None:
        return np.inf
    an = normalize_sequence(a)
    if an is None:
        return np.inf
    b = np.convolve(an, an)
    peak = float(b.max())
    sc = 2.0 * float(len(an)) * peak
    if not np.isfinite(sc):
        return np.inf
    return sc


def symmetrize(a):
    an = normalize_sequence(a)
    if an is None:
        return None
    return normalize_sequence(0.5 * (an + an[::-1]))


def smooth_filter(a, passes=2):
    an = normalize_sequence(a)
    if an is None:
        return None
    k = 5
    kernel = np.ones(k, dtype=np.float64) / float(k)
    for _ in range(int(passes)):
        an = np.convolve(an, kernel, mode="same")
        an = np.maximum(an, 0.0)
    return normalize_sequence(an)


def interpolate_seq(a, new_n):
    an = normalize_sequence(a)
    if an is None:
        return None
    old_n = int(len(an))
    new_n = int(new_n)
    if old_n == new_n or new_n < 2:
        return an
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    y = np.interp(x_new, x_old, an)
    return normalize_sequence(y)


def build_shifts(a):
    n = int(len(a))
    L = 2 * n - 1
    C = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = a
    return C


def make_xs(old, mean, max_a, limit):
    if limit > 28:
        mults = MULTS_LARGE
    else:
        mults = MULTS_SMALL

    old = float(old)
    mean = float(mean)
    max_a = float(max_a)
    base = max(old, mean, 1e-300)

    xs = base * mults
    if old > 0.0:
        xs = np.concatenate((xs, old * mults))
    if max_a > 2.0 * mean:
        xs = np.concatenate((xs, max_a * ABS_FACTORS))

    xs = np.concatenate((xs, np.array([0.0, old], dtype=np.float64)))
    xs = np.unique(np.round(xs, 14))
    xs = xs[xs >= 0.0]

    if xs.size == 0:
        xs = np.array([old], dtype=np.float64)

    if xs.size > limit:
        idx = np.unique(np.linspace(0, int(xs.size) - 1, int(limit)).astype(int))
        xs = xs[idx]
        xs = np.unique(np.concatenate(
            (xs, np.array([0.0, old, float(xs.max())], dtype=np.float64))
        ))
    return xs


def eval_col_candidates(b, S, Ccol, two, xs, old):
    xs = np.asarray(xs, dtype=np.float64)
    valid = xs >= 0.0
    if not valid.any():
        empty = np.array([], dtype=np.float64)
        return empty, empty
    xs_v = xs[valid]
    D = xs_v - float(old)
    Snew = float(S) + D
    ok = Snew > 1e-14
    if not ok.any():
        empty = np.array([], dtype=np.float64)
        return empty, empty

    xs_v = xs_v[ok]
    D = D[ok]
    Snew = Snew[ok]

    M = b[None, :] + (2.0 * D)[:, None] * Ccol[None, :]
    M[:, two] += D[:, None] * D[:, None]
    row_max = np.max(M, axis=1)
    peaks = row_max / (Snew * Snew)
    return xs_v, peaks


def optimize_col(b, S, C, i, a, mean, max_a, refine, k_limit):
    old = float(a[i])
    Ccol = C[:, i]
    two = 2 * i

    xs = make_xs(old, mean, max_a, k_limit)
    X, peaks = eval_col_candidates(b, S, Ccol, two, xs, old)
    if X.size == 0:
        return old, float(b.max()) / (float(S) * float(S))

    idx = int(np.argmin(peaks))
    best_x = float(X[idx])
    best_peak = float(peaks[idx])

    if refine:
        lref = 10 if int(len(b)) > 2000 else 12
        parts = []

        if idx > 0 and idx + 1 < int(X.size):
            parts.append(np.linspace(float(X[idx - 1]), float(X[idx + 1]), lref))
        elif idx > 0:
            parts.append(np.linspace(0.0, float(X[idx]), lref))
        else:
            parts.append(
                np.linspace(float(X[idx]), float(X[idx]) + max(float(mean), 1e-12), lref)
            )

        if best_x > 0.0:
            parts.append(np.linspace(max(0.0, best_x * 0.85), best_x * 1.15, lref))
        else:
            parts.append(np.linspace(0.0, max(float(mean) * 0.5, 1e-12), lref))

        R = np.concatenate(parts)
        R = np.unique(np.concatenate((R, np.array([best_x, 0.0], dtype=np.float64))))
        Rx, Rpeaks = eval_col_candidates(b, S, Ccol, two, R, old)
        if Rx.size > 0:
            ridx = int(np.argmin(Rpeaks))
            if float(Rpeaks[ridx]) < best_peak:
                best_x = float(Rx[ridx])
                best_peak = float(Rpeaks[ridx])

    return best_x, best_peak


def exact_cd(a_init, time_deadline, max_passes=300, refine=False, perturb=True, sigma=0.03):
    a = normalize_sequence(a_init)
    if a is None:
        return None, np.inf

    n = int(len(a))
    if n < 2:
        return a, fast_score(a)

    b = fft_conv(a)
    C = build_shifts(a)
    S = 1.0
    peak = float(b.max())
    best = a.copy()
    best_score = float(2.0 * n * peak)

    col_idx = np.arange(n, dtype=np.int64)
    rng = np.random.default_rng(987)

    if n > 1000:
        k_limit = 18
    elif n > 700:
        k_limit = 26
    else:
        k_limit = 32

    pass_no = 0
    stagnation = 0
    max_stagnation = 8 if perturb else 0

    while pass_no < int(max_passes) and time.monotonic() < float(time_deadline):
        if abs(S - 1.0) > 1e-8 or pass_no % 4 == 0:
            s = float(a.sum())
            if s > 1e-14:
                a = a / s
                b = fft_conv(a)
                C = build_shifts(a)
                S = 1.0
                peak = float(b.max())
                if peak < best_score / (2.0 * n) - 1e-15:
                    best = a.copy()
                    best_score = float(2.0 * n * peak)

        perm = rng.permutation(n) if perturb else np.arange(n)
        improved = 0

        for i in perm:
            if time.monotonic() >= float(time_deadline):
                break

            mean = S / float(n)
            max_a = float(a.max())
            best_x, best_peak = optimize_col(
                b, S, C, int(i), a, mean, max_a, refine, k_limit
            )

            if best_x >= 0.0 and best_peak < peak - 1e-15:
                old = float(a[i])
                d = float(best_x) - old
                if abs(d) > 1e-18:
                    i = int(i)
                    b += (2.0 * d) * C[:, i]
                    two = 2 * i
                    if two < int(len(b)):
                        b[two] += d * d

                    a[i] = float(best_x)
                    S += d
                    C[i + col_idx, col_idx] += d

                    peak = float(best_peak)
                    improved += 1

        if improved > 0:
            stagnation = 0
        else:
            stagnation += 1
            if perturb and stagnation <= max_stagnation:
                a = a * np.exp(rng.standard_normal(n) * float(sigma))
                a = np.maximum(a, 0.0)
                s = float(a.sum())
                if s > 1e-14:
                    a = a / s
                    b = fft_conv(a)
                    C = build_shifts(a)
                    S = 1.0
                    peak = float(b.max())
                stagnation = 0
            else:
                break

        if peak < best_score / (2.0 * n) - 1e-15:
            s = float(a.sum())
            if s > 1e-14:
                best = a / s
                best_score = float(2.0 * n * peak)

        pass_no += 1

    best = normalize_sequence(best)
    if best is None:
        return normalize_sequence(a_init), np.inf
    return best, fast_score(best)


def eval_pair_candidates(b, S, Ci, Cj, two_i, two_j, cross, xs_i, xs_j, old_i, old_j):
    Xi_full, Xj_full = np.meshgrid(xs_i, xs_j, indexing="ij")
    Xi = Xi_full.ravel()
    Xj = Xj_full.ravel()

    Dp = Xi - float(old_i)
    Dq = Xj - float(old_j)
    Snew = float(S) + Dp + Dq
    ok = Snew > 1e-14
    if not ok.any():
        empty = np.array([], dtype=np.float64)
        return empty, empty, empty

    Xi = Xi[ok]
    Xj = Xj[ok]
    Dp = Dp[ok]
    Dq = Dq[ok]
    Snew = Snew[ok]

    M = (
        b[None, :]
        + (2.0 * Dp)[:, None] * Ci[None, :]
        + (2.0 * Dq)[:, None] * Cj[None, :]
    )
    M[:, two_i] += Dp * Dp
    M[:, two_j] += Dq * Dq
    if cross is not None:
        M[:, cross] += 2.0 * Dp * Dq

    row_max = np.max(M, axis=1)
    peaks = row_max / (Snew * Snew)
    return Xi, Xj, peaks


def pair_cd(a_init, time_deadline, max_rounds=180, max_pairs=45):
    a = normalize_sequence(a_init)
    if a is None:
        return None, np.inf

    n = int(len(a))
    if n < 2:
        return a, fast_score(a)

    b = fft_conv(a)
    C = build_shifts(a)
    S = 1.0
    peak = float(b.max())
    best = a.copy()
    best_score = float(2.0 * n * peak)

    col_idx = np.arange(n, dtype=np.int64)
    rng = np.random.default_rng(4242)

    for r in range(int(max_rounds)):
        if time.monotonic() >= float(time_deadline):
            break

        top_p = min(12, int(len(b)))
        pidx = np.argpartition(b, -top_p)[-top_p:]
        order = np.argsort(b[pidx])[::-1]
        pidx = pidx[order]

        pairs = []
        seen = set()

        for k in pidx:
            k = int(k)
            start = max(0, k - n + 1)
            end = min(n - 1, k)
            cands = []
            for ii in range(start, end + 1):
                jj = k - ii
                if ii == jj:
                    continue
                prod = float(a[ii] * a[jj])
                cands.append((prod, int(ii), int(jj)))

            cands.sort(reverse=True)
            for _, ii, jj in cands[:5]:
                if ii > jj:
                    ii, jj = jj, ii
                key = (int(ii), int(jj))
                if key not in seen:
                    seen.add(key)
                    pairs.append(key)
                    if len(pairs) >= 220:
                        break
            if len(pairs) >= 220:
                break

        if not pairs:
            break

        pairs = pairs[:int(max_pairs)]
        improved = 0
        mean = S / float(n)

        for ii, jj in pairs:
            if time.monotonic() >= float(time_deadline):
                break

            ii = int(ii)
            jj = int(jj)
            old_i = float(a[ii])
            old_j = float(a[jj])

            base_i = max(old_i, mean, 1e-300)
            base_j = max(old_j, mean, 1e-300)

            xs_i = base_i * PAIR_MULTS
            if old_i > 0.0:
                xs_i = np.concatenate((xs_i, old_i * PAIR_MULTS))
            xs_i = np.unique(xs_i[xs_i >= 0.0])
            if xs_i.size == 0:
                xs_i = np.array([old_i], dtype=np.float64)
            if xs_i.size > 9:
                xs_i = xs_i[:9]

            xs_j = base_j * PAIR_MULTS
            if old_j > 0.0:
                xs_j = np.concatenate((xs_j, old_j * PAIR_MULTS))
            xs_j = np.unique(xs_j[xs_j >= 0.0])
            if xs_j.size == 0:
                xs_j = np.array([old_j], dtype=np.float64)
            if xs_j.size > 9:
                xs_j = xs_j[:9]

            Xi, Xj, peaks = eval_pair_candidates(
                b, S, C[:, ii], C[:, jj], 2 * ii, 2 * jj,
                ii + jj, xs_i, xs_j, old_i, old_j
            )
            if Xi.size == 0:
                continue

            idx = int(np.argmin(peaks))
            new_peak = float(peaks[idx])
            if new_peak < peak - 1e-14:
                new_i = float(Xi[idx])
                new_j = float(Xj[idx])
                dp = float(new_i) - old_i
                dq = float(new_j) - old_j

                Ci = C[:, ii]
                Cj = C[:, jj]

                b += (2.0 * dp) * Ci
                b += (2.0 * dq) * Cj
                b[2 * ii] += dp * dp
                b[2 * jj] += dq * dq
                b[ii + jj] += 2.0 * dp * dq

                a[ii] = float(new_i)
                a[jj] = float(new_j)
                S += dp + dq

                C[ii + col_idx, col_idx] += dp
                C[jj + col_idx, col_idx] += dq

                peak = new_peak
                improved += 1

        if improved == 0:
            break

        if peak < best_score / (2.0 * n) - 1e-14:
            best = a / float(a.sum())
            best_score = float(2.0 * n * peak)

        if (r + 1) % 3 == 0:
            s = float(a.sum())
            if s > 1e-14:
                a = a / s
                b = fft_conv(a)
                C = build_shifts(a)
                S = 1.0
                peak = float(b.max())
                if peak < best_score / (2.0 * n) - 1e-14:
                    best = a.copy()
                    best_score = float(2.0 * n * peak)

    best = normalize_sequence(best)
    if best is None:
        return normalize_sequence(a_init), np.inf
    return best, fast_score(best)


def lp_best_response(a):
    an = normalize_sequence(a)
    if an is None:
        return None

    n = int(len(an))
    if n > 1024:
        return None

    L = 2 * n - 1
    A_ub = np.zeros((L, n + 1), dtype=np.float64)
    A_ub[:, n] = -1.0
    for j in range(n):
        A_ub[j:j + n, j] = an

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    b_ub = np.zeros(L, dtype=np.float64)
    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = sopt.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs"
        )
        if res is not None and res.x is not None and int(res.status) in (0, 1):
            x = np.asarray(res.x[:n], dtype=np.float64)
            x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
            x = np.maximum(x, 0.0)
            s = float(x.sum())
            if s > 1e-12:
                return x / s
    except Exception:
        pass

    return lp_best_response_cvx(an)


def lp_best_response_cvx(an):
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = int(len(an))
    if n > 1024:
        return None

    L = 2 * n - 1
    cols = np.repeat(np.arange(n), n)
    rows = cols + np.tile(np.arange(n), n)
    data = np.tile(an, n)
    T = sp.csr_matrix((data, (rows, cols)), shape=(L, n))

    bvar = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [T @ bvar <= M, cp.sum(bvar) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)

    solvers = []
    for attr in ("PDLP", "ECOS", "CLARABEL", "SCS", "GLPK", "GUROBI", "MOSEK", "CBC", "SCIP", "XPRESS"):
        obj = getattr(cp, attr, None)
        if obj is not None:
            solvers.append(obj)

    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ("optimal", "optimal_inaccurate") and bvar.value is not None:
                x = np.asarray(bvar.value, dtype=np.float64)
                x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
                x = np.maximum(x, 0.0)
                s = float(x.sum())
                if s > 1e-12:
                    return x / s
                break
        except Exception:
            continue
    return None


def fictitious_play(a_init, time_deadline, max_iter=250):
    a = normalize_sequence(a_init)
    if a is None:
        return None, np.inf

    n = int(len(a))
    if n > 1024:
        a = interpolate_seq(a, 1024)

    a = a.copy()
    avg = a.copy()
    best = a.copy()
    best_score = fast_score(best)

    for it in range(1, int(max_iter) + 1):
        if time.monotonic() >= float(time_deadline):
            break

        b_new = lp_best_response(a)
        if b_new is None:
            break

        a_new = lp_best_response(b_new)
        if a_new is None:
            a_new = a

        avg += (a_new - avg) / float(it + 1)

        candidates = (a_new, b_new, avg, 0.5 * (a_new + b_new))
        for cand in candidates:
            if time.monotonic() >= float(time_deadline):
                break
            cn = normalize_sequence(cand)
            if cn is not None:
                sc = fast_score(cn)
                if sc < best_score:
                    best_score = sc
                    best = cn.copy()

    best = normalize_sequence(best)
    if best is None:
        return None, np.inf
    return best, fast_score(best)


def safe_softmax(w):
    w = np.asarray(w, dtype=np.float64)
    w = np.nan_to_num(w, nan=0.0, posinf=50.0, neginf=-50.0)
    w = np.clip(w, -50.0, 50.0)
    w = w - float(np.max(w))
    e = np.exp(w)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(w, dtype=np.float64) / float(len(w))
    return e / s


def correlation_fft(p, a):
    n = int(len(a))
    required = max(1, 3 * n - 2)
    size = 1 << (required - 1).bit_length()
    P = rfft(p, size)
    A = rfft(a, size)
    corr = irfft(P * np.conj(A), size)
    return corr[:n]


def smooth_obj(w, gamma):
    a = safe_softmax(w)
    b = fft_conv(a)
    mb = float(b.max())
    if mb <= 0.0:
        return 0.0
    z = np.exp(float(gamma) * (b - mb))
    ps = float(z.sum())
    if ps <= 0.0:
        return float(2.0 * len(a) * mb)
    val = mb + float(np.log(ps)) / float(gamma)
    return float(2.0 * len(a) * val)


def smooth_grad(w, gamma):
    a = safe_softmax(w)
    b = fft_conv(a)
    mb = float(b.max())
    if mb <= 0.0:
        return np.zeros_like(a, dtype=np.float64)
    z = np.exp(float(gamma) * (b - mb))
    ps = float(z.sum())
    if ps <= 0.0:
        return np.zeros_like(a, dtype=np.float64)
    p = z / ps

    c = correlation_fft(p, a)
    g = 2.0 * a * (c - np.dot(a, c))
    return float(2.0 * len(a)) * g


def smooth_optimize(a0, gamma, maxiter=60):
    a = normalize_sequence(a0)
    if a is None:
        return None, np.inf

    if len(a) > 2500:
        return a, fast_score(a)

    w = np.log(np.maximum(a, 1e-12))
    w = w - float(np.mean(w))
    bounds = [(-50.0, 50.0)] * int(len(a))

    try:
        res = sopt.minimize(
            smooth_obj,
            w,
            args=(float(gamma),),
            jac=smooth_grad,
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": int(maxiter),
                "maxfun": int(max(120, 3 * int(maxiter))),
                "ftol": 1e-10,
                "gtol": 1e-7,
                "disp": False
            }
        )
        if res is not None and getattr(res, "x", None) is not None:
            a = safe_softmax(res.x)
        else:
            a = safe_softmax(w)
    except Exception:
        a = a

    return a, fast_score(a)


def smooth_restarts(a_init, time_deadline, max_runs=8):
    a0 = normalize_sequence(a_init)
    if a0 is None:
        return None, np.inf

    best = a0.copy()
    best_score = fast_score(best)

    rng = np.random.default_rng(555)
    gammas = (4000.0, 12000.0, 35000.0)
    n = int(len(a0))
    maxiter = 35 if n > 1024 else 60

    runs = 0
    while runs < int(max_runs) and time.monotonic() < float(time_deadline):
        if time.monotonic() + 20.0 > float(time_deadline):
            break

        sigma = 0.02 * (float(runs) + 1.0)
        a_try = a0 * np.exp(rng.standard_normal(n) * sigma)
        if runs % 3 == 1:
            a_try = symmetrize(a_try)
        a_try = normalize_sequence(a_try)
        if a_try is None:
            runs += 1
            continue

        gamma = float(gammas[runs % len(gammas)])
        a_out, sc = smooth_optimize(a_try, gamma, maxiter=maxiter)
        if a_out is not None and np.isfinite(sc) and sc < best_score:
            best = a_out.copy()
            best_score = sc

        a_sym = symmetrize(a_out)
        if a_sym is not None:
            sc_sym = fast_score(a_sym)
            if np.isfinite(sc_sym) and sc_sym < best_score:
                best = a_sym.copy()
                best_score = sc_sym

        runs += 1

    best = normalize_sequence(best)
    if best is None:
        return a0, fast_score(a0)
    return best, fast_score(best)


def init_fourier(n, rng):
    n = int(n)
    if n < 32:
        n = 32
    m = int(rng.integers(3, 15))
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for _ in range(m):
        freq = int(rng.integers(1, max(2, n // 2)))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.05, 0.45))
        a += amp * np.cos(2.0 * np.pi * float(freq) * t / float(n) + phase)
    return normalize_sequence(a)


def candidate_add(scored, name, a, idx):
    if a is None:
        return idx
    an = normalize_sequence(a)
    if an is None:
        return idx
    sc = fast_score(an)
    if np.isfinite(sc):
        scored.append((float(sc), int(idx), str(name), an))
    return idx + 1
