The dual variables y_k tell us: at the optimum, the weights y_k indicate which convolution positions are "most important" (most binding). We can use this to guide a gradient step: instead of using a uniform subgradient over the active set, we can weight by the dual variables.

This is essentially what the LP does implicitly, but we can make it explicit in a gradient-like method.

1. Solve LP to get primal b* and dual y*
2. Use y* as weights for a "weighted subgradient": g[i] = sum_k y[k] * 2*a[k-i] / sum(y)
3. Take a step in this direction
4. Repeat

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_for_fixed_a(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a[j]
    return T


def _lp_solver_cache():

def get_lp_solver():
    """Find a working LP solver."""
    if hasattr(get_lp_solver, '_cached'):
        return get_lp_solver._cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                get_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    get_lp_solver._cached = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_for_fixed_a(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_fn(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def log_of_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def subgradient_max_conv(a, active_frac=0.05):
    """Compute subgradient of max_k (a*a)[k] at point a. Returns (g, max_b, active)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - active_frac)
    active = np.where(b >= threshold)[0]

    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(active)
    return g, max_b, active


def log_space_grad(a, g_a):
    """Convert gradient in a-space to gradient in log-space (softmax parametrization)."""
    S = np.dot(g_a, a)
    return a * (g_a - S)

def lp_nesterov_alternate(a0, time_limit, solver=None, n_max_iters=20000):
    """
    LP-based alternating with Nesterov dual averaging.
    Alternates between solving LP for best response.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()  # virtual iterate
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    n_solves = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP for best response to y
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        n_solves += 1

        # Nesterov step size
        alpha = 2.0 / (it + 2.0)

        # Update virtual iterate
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Update actual iterate (weighted average)
        weight_new = alpha * (it + 2.0)
        a = ((it + 1.0) * a + weight_new * b_new) / (it + 1.0 + weight_new)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if n_solves % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov solves={n_solves} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def adam_log_space(a0, time_limit, lr_init=0.1, n_steps=100000,
                   beta1=0.9, beta2=0.999, eps=1e-8, adaptive_frac=0.05):
    """
    Adam optimizer in log-space with exact max subgradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = log_of_a(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient
        g_a, max_b, active = subgradient_max_conv(a, active_frac=adaptive_frac)
        score = 2.0 * n * max_b

        # Convert to log-space gradient
        g_z = log_space_grad(a, g_a)

        # Adam update
        step_num = step + 1
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * g_z ** 2
        m_hat = m / (1.0 - beta1 ** step_num)
        v_hat = v / (1.0 - beta2 ** step_num)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back
        a_new = softmax_fn(z)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Keep old z (undo)
            z = z + lr * m_hat / (np.sqrt(v_hat) + eps)

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.85
            no_improve = 0
            # Occasionally try a bigger jump
            if int(lr * 10000) % 7 == 0:
                lr *= 1.5
        lr = max(lr, 1e-8)

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam {elapsed:.0f}s: score={best_score:.6f} lr={lr:.2e} step={step}")

    return best_a, best_score


def p_norm_continuation(a0, time_limit, p_values=None):
    """
    Optimize p-norm of auto-convolution with increasing p.
    p=2 is smooth, p=inf is the actual objective.
    """
    if p_values is None:
        p_values = [2, 4, 8, 16, 32, 64, 128]

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for p in p_values:
        if time.monotonic() - t0 > time_limit:
            break
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 5:
            break
        # Allocate time proportional to p (later p values get more time)
        frac = time_left / len(p_values[p_values.index(p):])
        t_alloc = min(frac * 0.8, 200.0)

        if t_alloc < 5:
            continue

        # For p=2, use gradient descent (smooth)
        # For p>2, use Adam
        if p <= 4:
            a, sc = _gd_p_norm(a, t_alloc, p, n)
        else:
            a, sc = _adam_p_norm(a, t_alloc, p, n)

        # Check actual score
        actual = 2.0 * n * fast_convolve(a).max()
        if actual < best_score:
            best_score = actual
            best_a = a.copy()
            print(f"    p-norm p={p}: score={best_score:.6f}")

    return best_a, best_score


def _gd_p_norm(a0, time_limit, p, n):
    """Gradient descent on p-norm objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 1.0 / n

    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Gradient of (sum b^p)^(1/p)
        b_p_sum = np.sum(b ** p)
        prefactor = b_p_sum ** (1.0 / p - 1.0)
        g = np.zeros(n)
        for k in range(len(b)):
            w = b[k] ** (p - 1)
            if w < 1e-30:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w * a[k - idx]
        g *= 2.0 * n * prefactor

        # Projected gradient step on simplex
        a_new = a - lr * g
        # Project onto simplex
        a_new = _project_simplex(a_new)
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.95
        lr = max(lr, 1e-8)

    return best_a, best_score


def _adam_p_norm(a0, time_limit, p, n):

    a = a0.copy()
    z = log_of_a(a)
    best_a = a.copy()

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * bkm1 * a[k - idx]
        g *= 2.0 * n * prefactor

        g_z = log_space_grad(a, g)
        sn = step + 1
        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * g_z ** 2
        m_hat = m / (1 - beta1 ** sn)

def _project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def peak_shave_optimal(a0, time_limit, n_steps=20000):
    """
    Optimal peak shaving: for the peak convolution position,
    find the optimal redistribution of mass to minimize the peak.
    Uses the dual information from the LP to guide direction.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the interior peak (avoid boundaries)
        b_mid = b.copy()
        b_mid[0] = -np.inf
        b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            # Try reducing a[i] and distributing to others
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88, 0.82]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try reducing both i and j simultaneously
        if not improved and len(top_idx) >= 2:
            for ci in top_idx[:4]:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.98, 0.95, 0.9]:
                    for f2 in [0.98, 0.95, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Try reducing top 3 simultaneously
        if not improved:
            top3 = top_idx[:3]
            for factors in [[0.98, 0.98, 0.98], [0.96, 0.96, 0.96],
                           [0.95, 0.95, 0.95], [0.97, 0.97, 0.94]]:
                a_trial = a.copy()
                for t, f in zip(top3, factors):
                    a_trial[idx[t]] *= f
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            break

def coordinate_descent_refine(a0, time_limit, n_passes=50):
    """Cyclic coordinate descent with golden section on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

    best_a = a.copy()

    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improvement = 0
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            base = a.copy()
            base[i] = 0
            base_sum = base.sum()
            if base_sum < 1e-15:
                continue
            # Try several multipliers
            best_t = a[i]
            best_local = best_score
            for factor in [0.8, 0.9, 1.1, 1.25, 1.5]:
                trial_t = a[i] * factor
                if trial_t < 0 or trial_t > 0.5:

                a_trial = base * (1.0 - trial_t) + trial_t * np.eye(n)[i]
                if a_trial.sum() < 1e-15:

                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_local:
                    best_local = sc
                    best_t = trial_t
            if best_t != a[i]:
                a = base * (1.0 - best_t) + best_t * np.eye(n)[i]
                a = a / a.sum()
                sc = 2.0 * n * fast_convolve(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improvement += 1

        if p % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc pass={p} ({elapsed:.0f}s): score={best_score:.6f} imp={improvement}")
        if improvement == 0 and p > 5:
            break

    return best_a, best_score

            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_fn(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = max(100, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 80, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_fn(z)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def rand_restart_refine(parent, n, rng, time_limit):
    """Create a random perturbation of parent and refine with Adam."""
    a = parent.copy()
    # Random perturbation in log-space
    z = log_of_a(a)
    noise_scale = 0.01
    z += noise_scale * rng.standard_normal(n)
    a = softmax_fn(z)
    # Quick Adam refinement
    a_opt, sc = adam_log_space(a, min(time_limit, 30.0), lr_init=0.05, n_steps=5000)
    return a_opt, 2.0 * n * fast_convolve(a_opt).max()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines LP alternating, Adam, L-BFGS, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: LP Nesterov Alternating (600s) ===
    print(f"\n=== Phase 1: LP Nesterov (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.38, 600.0)
    if t_alloc > 30:
        print(f"  LP-Nesterov n={n_best}, budget={t_alloc:.0f}s")
        a_lp, _ = lp_nesterov_alternate(best_seq, t_alloc, solver=solver)
        try_update(a_lp, "lp_main")

    # === Phase 2: Adam in log-space (350s) ===
    print(f"\n=== Phase 2: Adam Log-Space (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Adam n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_adam, _ = adam_log_space(best_seq, t_alloc, lr_init=0.08,
                                   n_steps=200000, adaptive_frac=0.03)
        try_update(a_adam, "adam_main")

    # === Phase 3: L-BFGS low temperature (250s) ===
    print(f"\n=== Phase 3: L-BFGS Low-T (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 20->0.5, n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_smooth(best_seq, t_alloc, T_start=20.0, T_end=0.5, maxiter=6000)
        try_update(a_lb, "lbgfs_main")

    # === Phase 4: Peak shaving (200s) ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 15:
        print(f"  PeakShave n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave_optimal(best_seq, t_alloc, n_steps=50000)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Second LP round (150s) ===
    print(f"\n=== Phase 5: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  LP-Nesterov R2 n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lp2, _ = lp_nesterov_alternate(best_seq, t_alloc, solver=solver)
        try_update(a_lp2, "lp_r2")

    # === Phase 6: Random restarts (100s) ===
    print(f"\n=== Phase 6: Random Restarts (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_restarts = min(5, int(rem // 25))
        for ri in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            a_rr, sc = rand_restart_refine(best_seq, len(best_seq), rng, min(rem * 0.3, 20.0))
            try_update(a_rr, f"restart_{ri}")

    # === Phase 7: Coordinate descent (80s) ===
    print(f"\n=== Phase 7: Coord Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_passes=30)
        try_update(a_cd, "coord")

    # === Phase 8: Try different lengths (100s) ===
    print(f"\n=== Phase 8: Alt Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = sorted(set([int(n_best * 0.7), int(n_best * 1.2), int(n_best * 1.5)]))
    alt_lengths = [x for x in alt_lengths if 256 <= x <= 3000 and x != n_best]
    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t_alloc = min(rem * 0.4, 60.0)
        if t_alloc < 10:
            break
        print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt = lbgfs_smooth(init_alt, t_alloc, T_start=20.0, T_end=1.0, maxiter=3000)
        try_update(a_alt, f"alt_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 15 and len(best_seq) == alt_n:
            t_lp = min(rem * 0.4, 30.0)
            if t_lp > 10:
                a_al, _ = lp_nesterov_alternate(best_seq, t_lp, solver=solver)
                try_update(a_al, f"lp_alt_n{alt_n}")

    # === Phase 9: Final polish (remaining time) ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_final = min(rem * 0.5, 40.0)
        a_final = lbgfs_smooth(best_seq, t_final, T_start=5.0, T_end=0.3, maxiter=3000)
        try_update(a_final, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.5, 30.0)
        a_ps_final, _ = peak_shave_optimal(best_seq, t_ps, n_steps=20000)
        try_update(a_ps_final, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_adam = min(rem * 0.5, 20.0)
        a_adam_f, _ = adam_log_space(best_seq, t_adam, lr_init=0.02, n_steps=50000,
                                     adaptive_frac=0.02)
        try_update(a_adam_f, "final_adam")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
