import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:

def parametric_poly(n, coeffs):
    """
    Generate sequence: a_i ∝ exp(Σ_{m=0}^{M} c_m * (2i/(n-1) - 1)^m)
    coeffs: list of M+1 coefficients.
    """
    x = 2.0 * np.arange(n) / max(n - 1, 1) - 1.0
    log_a = np.zeros(n)
    for m, c in enumerate(coeffs):
        log_a += c * (x ** m)
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def parametric_fourier(n, coeffs):
    """
    Generate sequence: a_i ∝ exp(c_0 + Σ_{m=1}^{M}(c_m * cos(2πmi/n) + d_m * sin(2πmi/n)))
    coeffs: [c0, c1, d1, c2, d2, ...]
    """
    t = np.arange(n, dtype=np.float64)
    log_a = np.full(n, coeffs[0])
    idx = 1
    m = 1
    while idx + 1 < len(coeffs):
        cm = coeffs[idx]
        dm = coeffs[idx + 1]
        log_a += cm * np.cos(2 * np.pi * m * t / n) + dm * np.sin(2 * np.pi * m * t / n)
        idx += 2
        m += 1
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def score_seq(a):
    """Compute the evaluation score for a sequence (normalized sum=1)."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a = a / s
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def hankel_qp_step(a, temperature, damping=0.5):
    """
    One step of symmetric QP iteration:
    1. Compute peaks b = conv(a,a)
    2. Compute softmax weights w_k ∝ exp(b[k]/T)
    3. Compute gradient: sg_j = 2 * sum_k w_k * a[k-j] (if 0<=k-j<n)
    4. Update: a_new = a - lr * sg, project to simplex
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    b_min = b.min()
    if b_max - b_min < 1e-15:
        return a, b_max
    
    # Softmax weights (shifted for numerical stability)
    shifted = (b - b_max) / temperature
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[int(np.argmax(b))] = 1.0
    else:
        w = w / w_sum
    
    # Compute gradient: sg_j = 2 * sum_k w_k * a[k-j] for valid k-j
    sg = np.zeros(n)
    for j in range(n):
        # k-j must be in [0, n-1], so k in [j, j+n-1]
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        if k_start <= k_end:
            # a[k-j] for k in [k_start, k_end] means indices 0 to (k_end-j)
            a_slice = a[0:k_end - k_start + 1]
            w_slice = w[k_start:k_end + 1]
            sg[j] = 2.0 * np.dot(w_slice, a_slice)
    
    return sg, b_max, w

def subgradient_optimize(a0, time_limit, top_k=30, lr_init=0.003, verbose=False):
    """
    Subgradient method with adaptive step size and momentum.
    Uses softmax-weighted peaks for a smoother gradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_seq(a)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    lr = lr_init
    it = 0
    no_improve = 0
    temperature = 0.01 * best_score / (2.0 * n)  # initial temperature
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        b_max = b.max()
        
        # Softmax-averaged subgradient
        top_k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -top_k_actual)[-top_k_actual:]
        pv = b[peaks]
        temp = max(temperature, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = np.arange(i_start, i_end + 1)
                sg[j] += 2.0 * w[idx] * a[k - j]
        
        # Momentum
        momentum = beta * momentum + sg
        norm_mom = np.linalg.norm(momentum)
        if norm_mom > 1e-15:
            momentum = momentum / norm_mom
        
        # Step
        y = a - lr * momentum
        a_new = simplex_project(y)
        
        score = score_seq(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adapt
        if no_improve > 100:
            lr *= 0.9
            no_improve = 0
        if no_improve > 400:
            # Restart with perturbation
            noise = np.random.default_rng(it * 31 + 7).standard_normal(n) * 0.02
            a = simplex_project(best_a + noise)
            momentum = np.zeros(n)
            lr = lr_init * 1.5
            no_improve = 0
        
        lr *= 0.9999
        temperature *= 0.999
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score

def coord_descent_polish(a0, time_limit, n_sweeps=300):
    """Full coordinate descent polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_seq(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            
            for factor in [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss < 1e-15:
                    a[i] = old_val
                    continue
                a = a / ss
                sc = score_seq(a)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
            
            # Try smaller adjustments if no improvement
            if not improved and sweep > 3:
                for factor in [0.97, 0.98, 1.02, 1.03]:
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss < 1e-15:
                        a[i] = old_val
                        continue
                    a = a / ss
                    sc = score_seq(a)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()
        
        if not improved and sweep > 10:
            break
    
    return best_a, best_score

def peak_targeted_shave(a0, time_limit):

    n = len(a0)

        a = np.ones(n) / n

        a = a / s

    best_score = score_seq(a)
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        top_k = min(5, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        peak_k = int(peaks[0])
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.max() < 1e-15:
            break
        
        top_contrib = np.argsort(contribs)[::-1][:5]
        improved = False
        
        for ci in top_contrib:

                continue

                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = score_seq(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            
            if not improved and i != j and a[j] >= 1e-14:
                for f1 in [0.85, 0.75, 0.65]:
                    for f2 in [0.85, 0.75, 0.65]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        sc = score_seq(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            # Try adding mass to a low-contribution coordinate
            b = fast_convolve(a)
            # Find coordinate that minimally affects peaks
            peak_set = set(peaks.tolist())
            min_contrib = float('inf')
            best_j = -1
            for j_try in range(0, n, max(1, n // 200)):
                if a[j_try] > 0.9 / n:
                    continue
                max_increase = 0.0
                for k_try in peaks:
                    if 0 <= k_try - j_try < n:
                        max_increase = max(max_increase, a[k_try - j_try])
                    if 0 <= j_try - k_try < n:
                        max_increase = max(max_increase, a[j_try - k_try])
                if max_increase < min_contrib:
                    min_contrib = max_increase
                    best_j = j_try
            
            if best_j >= 0:
                a_trial = a.copy()
                a_trial[best_j] += 0.02 / n
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_seq(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def peak_balance_iteration(a, n_steps=100, step_size=0.01):
    """
    Iterative peak balancing: identify which coordinates contribute
    most to the max peak and try reducing them while adding to
    low-impact coordinates.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = score_seq(a)
    
    for step in range(n_steps):
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        
        # Find coordinate to reduce (max contributor with high value)
        if contribs.max() < 1e-15:
            break
        top_i = int(idx[np.argmax(contribs)])
        
        # Find coordinate to add to (minimize max increase)
        # Score each candidate j by max_k a[k-j] over peak region
        peaks = np.argpartition(b, -max(5, len(b)//50))[-max(5, len(b)//50):]
        peak_max_vals = b[peaks].max()
        
        best_j = -1
        best_cost = float('inf')
        # Sample candidates
        candidates = list(range(0, n, max(1, n // 100)))
        for j_cand in candidates:
            if j_cand == top_i:
                continue
            cost = 0.0
            for k_cand in peaks:
                if 0 <= k_cand - j_cand < n:
                    cost = max(cost, a[k_cand - j_cand])
                if 0 <= j_cand - k_cand < n:
                    cost = max(cost, a[j_cand - k_cand])
            if cost < best_cost:
                best_cost = cost
                best_j = j_cand
        
        if best_j < 0:
            break
        
        # Transfer mass
        delta = step_size * a[top_i]
        if delta < 1e-15:
            break
        a_trial = a.copy()
        a_trial[top_i] -= delta
        a_trial[best_j] += delta
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            break
        a_trial = a_trial / st
        sc = score_seq(a_trial)
        if sc < best_score - 1e-14:
            a = a_trial
            best_score = sc
            best_a = a.copy()
    
    return best_a

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
