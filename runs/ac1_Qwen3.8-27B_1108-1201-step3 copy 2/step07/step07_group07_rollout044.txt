    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def lp_refine_intensive(a_init, time_limit, verbose=True):
    """
    Intensive LP-based iterative refinement.
    Repeatedly solves max sum(g) s.t. conv(a,g)<=M, g>=0, and blends.
    """
    t_start = time.monotonic()
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    rows, cols = build_conv_coo(n)

    best_a = a.copy()
    best_M = max_conv_val(a)
    no_improve = 0
    max_no_improve = 8

    for it in range(500):
        if time.monotonic() - t_start > time_limit:
            break

        # Solve LP with current M
        M = max_conv_val(a)
        if M < best_M:
            best_M = M
            best_a = a.copy()

        g = solve_lp_conv(a, M, rows, cols)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break

        g_norm = g / g_sum

        # Check if g is proportional to a (convergence)
        ratio = g_norm / np.maximum(a, 1e-15)
        if np.std(ratio[np.isfinite(ratio)]) < 1e-6:
            no_improve += 2
        else:
            no_improve += 0.5

        # Line search for best blend
        best_t = 0.0
        best_obj = best_M
        t_candidates = [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]

        for t in t_candidates:
            a_try = (1.0 - t) * best_a + t * g_norm
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize(a_try)
            obj = max_conv_val(a_try)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * best_a + best_t * g_norm
            a = np.maximum(a, 0.0)
            a = normalize(a)
            cur_M = max_conv_val(a)
            if cur_M < best_M - 1e-14:
                best_M = cur_M
                best_a = a.copy()
                no_improve = 0
            else:
                a = best_a.copy()
                no_improve += 1
            if verbose and it % 5 == 0:
                print(f"    LP iter {it}: M={best_M:.12f}, t={best_t:.3f}")
        else:
            # Try with slightly reduced M to force movement
            M_red = M * 0.998
            g2 = solve_lp_conv(a, M_red, rows, cols)
            if g2 is not None and g2.sum() > 1e-12:
                g2_norm = g2 / g2.sum()
                for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    a_try = (1.0 - t) * best_a + t * g2_norm
                    a_try = np.maximum(a_try, 0.0)
                    a_try = normalize(a_try)
                    obj = max_conv_val(a_try)
                    if obj < best_M - 1e-14:
                        best_M = obj
                        best_a = a_try.copy()
                        a = a_try.copy()
                        no_improve = 0
                        break
            if no_improve >= max_no_improve:
                # Perturb and restart
                rng = np.random.default_rng(it)
                a = best_a + rng.standard_normal(n) * 0.001 * best_M * n
                a = np.maximum(a, 0.0)
                a = normalize(a)
                no_improve = 0
                if verbose:
                    print(f"    LP iter {it}: restart with perturbation, M={max_conv_val(a):.12f}")

    return best_a, best_M


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.92, seed=42):
    """FFT-accelerated projected subgradient with momentum."""
    t_start = time.monotonic()
    n = len(a0)
    a = normalize(a0.copy().astype(np.float64))
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = max_conv_val(a)
    cur_lr = lr_init
    no_improve = 0
    it = 0

    fft_size = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - t_start < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        # Active peaks
        threshold = b_max * 0.995
        active_idx = np.where(b >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(b)])

        w = np.zeros(2 * n - 1)
        w[active_idx] = 1.0 / len(active_idx)

        # Subgradient via FFT
        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]

        # Nesterov momentum
        v = momentum * v - cur_lr * g
        a_new = a + v
        # Project to simplex
        a_new = _project_simplex(a_new)

        obj_new = max_conv_val(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        if no_improve >= 500:
            cur_lr *= 0.5
            v = np.zeros(n)
            a = _project_simplex(best_a + np.random.default_rng(it).standard_normal(n) * 0.002)
            no_improve = 0

    return best_a, best_obj


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent: adjust one element at a time on the simplex."""
    t_start = time.monotonic()
    n = len(a0)
    a = normalize(a0.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = max_conv_val(a)

    multipliers = np.array([0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.3, 1.5, 2.0])

    improved = True
    while improved and time.monotonic() - t_start < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t_start > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 100.0:
                    continue
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                a_tr = normalize(a_tr)
                val = max_conv_val(a_tr)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = normalize(a)
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def make_gaussian_init(n, sigma=0.25):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.4):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: intensive LP refinement, diverse initializations,
    multi-length exploration, subgradient polish, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 1152
        parent = np.ones(n_default) / n_default
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, uniform n=1152, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Intensive LP on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Intensive LP on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        t_lp = min(rem * 0.7, 500.0)
        print(f"  LP time={t_lp:.0f}s")
        a_lp, obj_lp = lp_refine_intensive(parent, time_limit=t_lp, verbose=True)
        update_best(a_lp, "phase1_lp")

        # Follow up with subgradient
        rem = deadline - time.monotonic() - 350
        if rem > 30 and best_seq is not None:
            t_sg = min(rem * 0.5, 100.0)
            print(f"  SG time={t_sg:.0f}s")
            a_sg, obj_sg = subgradient_optimize(best_seq, time_limit=t_sg,
                                                lr_init=0.002, momentum=0.93, seed=seed)
            update_best(a_sg, "phase1_sg")

            # Another LP pass
            rem = deadline - time.monotonic() - 300
            if rem > 20 and best_seq is not None:
                t_lp2 = min(rem * 0.6, 100.0)
                print(f"  LP2 time={t_lp2:.0f}s")
                a_lp2, obj_lp2 = lp_refine_intensive(best_seq, time_limit=t_lp2, verbose=False)
                update_best(a_lp2, "phase1_lp2")

    # === Phase 2: Diverse initializations at same length ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse Inits (best={best_score:.8f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {
        "gauss_0.15": make_gaussian_init(n_base, 0.15),
        "gauss_0.30": make_gaussian_init(n_base, 0.30),
        "cosine": make_cosine_init(n_base),
        "tukey_0.3": make_tukey_init(n_base, 0.3),
        "tukey_0.6": make_tukey_init(n_base, 0.6),
    }

    total_time_diverse = min(200.0, (deadline - time.monotonic() - 200.0) * 0.4)
    time_per_init = total_time_diverse / len(inits)

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 180
        if rem < 20:
            break
        t_alloc = min(time_per_init, rem * 0.3)
        if t_alloc < 10:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = lp_refine_intensive(init_a, time_limit=t_alloc, verbose=False)
        update_best(a_div, f"div_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.75, 1.0, 1.25, 0.6, 1.5, 0.8, 1.2]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths and tl != n_parent:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 120
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.25, 50.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n} (t={t_alloc:.0f}s)")
            a_len, obj_len = lp_refine_intensive(f_interp, time_limit=t_alloc, verbose=False)
            update_best(a_len, f"len_{target_n}")

    # === Phase 4: Deep optimization on best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep Opt (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            # LP
            t_lp_deep = min(rem * 0.5, 120.0)
            print(f"  Deep LP (t={t_lp_deep:.0f}s)")
            a_deep, obj_deep = lp_refine_intensive(best_seq, time_limit=t_lp_deep, verbose=True)
            update_best(a_deep, "deep_lp")

            # Subgradient
            rem = deadline - time.monotonic() - 50
            if rem > 15 and best_seq is not None:
                t_sg_deep = min(rem * 0.6, 60.0)
                print(f"  Deep SG (t={t_sg_deep:.0f}s)")
                a_deep_sg, _ = subgradient_optimize(best_seq, time_limit=t_sg_deep,
                                                    lr_init=0.001, momentum=0.94, seed=seed+777)
                update_best(a_deep_sg, "deep_sg")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: CD Polish (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd = min(rem, 45.0)
            print(f"  CD (t={t_cd:.0f}s)")
            a_cd, obj_cd = coordinate_descent_polish(best_seq, time_limit=t_cd)
            update_best(a_cd, "cd")

    # === Phase 6: Final micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Micro (best={best_score:.8f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_final = min(rem * 0.7, 25.0)
            a_fin, _ = lp_refine_intensive(best_seq, time_limit=t_final, verbose=False)
            update_best(a_fin, "final_lp")

            rem = deadline - time.monotonic() - 5
            if rem > 5:
                a_fin2, _ = coordinate_descent_polish(best_seq, time_limit=rem)
                update_best(a_fin2, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
