def fast_convolve(a):

    a_pad = np.zeros(L)
    a_pad[:n] = a
    h_pad = np.zeros(L)
    h_pad[:len(active_mask)] = active_mask.astype(np.float64)
    
    A = rfft(a_pad)
    H = rfft(h_pad)
    # Cross-correlation: (a ⋆ h)[j] = sum_m a[m] * h[m+j]
    corr = irfft(np.conj(A) * H, L)[:n]
    return 2.0 * corr


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s  # Should already sum to 1, but normalize for safety


def local_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()
