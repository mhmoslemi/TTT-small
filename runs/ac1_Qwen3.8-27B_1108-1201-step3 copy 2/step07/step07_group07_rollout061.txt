import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_and_subgrad(a, w):
    """
    Compute auto-convolution of a and cross-correlation of w with a.
    w has length 2n-1. Returns (b, g) where b is conv (length 2n-1), g is length n.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    b = irfft(A * A, fft_size)[:2 * n - 1]
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return b, g


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence."""
    b = fast_conv(a)
    return float(b.max())


def exponentiated_gradient(a0, time_limit, lr_init=0.01, n_top=50, seed=42,
                           use_top_k=True, top_frac=0.1):
    """
    Exponentiated gradient (multiplicative weights) for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_obj = eval_max_b(a)
    lr = lr_init
    no_improve = 0
    it = 0

    if n_top is None:
        n_top = max(10, int(n * top_frac))

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        b_max = b.max()

        if use_top_k and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            threshold = b_max * 0.995
            active_mask = np.zeros(len(b), dtype=bool)
            active_mask[top_idx[b[top_idx] >= threshold]] = True
            if not active_mask.any():
                active_mask[np.argmax(b)] = True
        else:
            threshold = b_max * 0.99
            active_mask = b >= threshold
            if not active_mask.any():
                active_mask[np.argmax(b)] = True

        w = 2.0 * active_mask.astype(np.float64) / active_mask.sum()
        _, g = conv_and_subgrad(a, w)

        # Exponentiated gradient update
        a = a * np.exp(-lr * g)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        obj = eval_max_b(a)
        it += 1

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve == 500:
            lr *= 0.8
        elif no_improve == 1500:
            lr *= 0.7
        elif no_improve == 3000:
            # Restart with perturbation
            kick = 0.01
            a = np.maximum(best_a + rng.standard_normal(n) * kick, 1e-10)
            a = a / a.sum()
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def subgradient_momentum(a0, time_limit, lr_init=0.003, momentum=0.9,
                         n_top=None, seed=42):

    t0 = time.monotonic()

    a = a0.copy().astype(np.float64)

    best_a = a.copy()

    lr = lr_init
    no_improve = 0

    if n_top is None:
        n_top = max(15, n // 10)

        b = fast_conv(a)
        b_max = b.max()

        if n_top < len(b):

            threshold = b_max * 0.998
            active_mask = np.zeros(len(b), dtype=bool)
            active_mask[top_idx[b[top_idx] >= threshold]] = True
            if not active_mask.any():
                active_mask[np.argmax(b)] = True
        else:
            active_mask = b >= b_max * 0.998
            if not active_mask.any():
                active_mask[np.argmax(b)] = True

        w = 2.0 * active_mask.astype(np.float64) / active_mask.sum()
        _, g = conv_and_subgrad(a, w)

        v = momentum * v - lr * g
        a_new = project_simplex(a + v)

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve == 1000:
            lr *= 0.85
        elif no_improve == 3000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = max(lr * 0.5, lr_init * 0.03)
            no_improve = 0

    return best_a, best_obj


def make_conv_matrix(a):
    """Construct the (2n-1) x n convolution constraint matrix (Toeplitz)."""
    n = len(a)
    idx = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[idx[mask]]
    return A


def lp_refine(f, max_iters=50, time_limit=120.0):
    """
    LP-based refinement: iterate max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Blend with current f to reduce peak.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)
    iteration = 0

    while iteration < max_iters and time.monotonic() - t0 < time_limit:
        M = eval_max_b(a)

        # Build constraint matrix
        A_mat = make_conv_matrix(a)
        b_lp = np.full(2 * n - 1, M)
        c = -np.ones(n)

        try:
            res = linprog(c, A_ub=A_mat, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, time_limit - (time.monotonic() - t0))})
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-12:
                break

            g_norm = g / g_sum

            # Line search for blending parameter t
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t <= 0.0:
                for t in [0.01, 0.02, 0.03, 0.05]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    blend = blend / blend.sum()
                    obj_blend = eval_max_b(blend)
                    if obj_blend < best_obj - 1e-14:
                        best_t = t
                        best_blend_obj = obj_blend
                        break

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_max_b(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                else:
                    a = best_a.copy()
            else:
                break

        except Exception:
            break

        iteration += 1

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_multipliers=15):
    """
    Coordinate descent: for each coordinate, try scaling it and rescaling others.
    Uses FFT for fast convolution evaluation.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 1.8, 2.0, 3.0, 5.0])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 50.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                idx = np.ones(n, dtype=bool)
                idx[j] = False
                a_tr[idx] *= scale
                a_tr = a_tr / a_tr.sum()
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                idx = np.ones(n, dtype=bool)
                idx[j] = False
                a[idx] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def make_init_flat_conv(n):
    """
    Initialize using the 'ideal flat auto-convolution' recurrence.
    a[0]=1, and for k>=1: 2*a[0]*a[k] + sum_{j=1}^{k-1} 2*a[j]*a[k-j] = 1
    So a[k] = (1 - sum_{j=1}^{k-1} 2*a[j]*a[k-j]) / (2*a[0])
    Truncate to n, ensure non-negativity, normalize.
    """
    a_full = np.zeros(2 * n)
    a_full[0] = 1.0
    for k in range(1, 2 * n):
        s = 0.0
        for j in range(1, k):
            if j < 2 * n and (k - j) < 2 * n:
                s += 2.0 * a_full[j] * a_full[k - j]
        a_full[k] = (1.0 - s) / (2.0 * a_full[0])
        if a_full[k] < 0:
            a_full[k] = 0.0

    a = a_full[:n].copy()
    # Make symmetric
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_init_gaussian(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_init_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_init_tukey(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_init_uniform(n):
    return np.ones(n) / n


def make_init_edges(n, beta=0.5):
    """U-shaped initialization (heavier at edges)."""
    x = np.arange(n, dtype=np.float64)
    a = (1.0 + x / n) ** (-beta) * (1.0 + (n - 1 - x) / n) ** (-beta)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linearly interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def run_optimizer(a_init, time_budget, n, seed, method="combined"):
    """
    Run combined optimization: exponentiated gradient + LP + subgradient.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    # Phase A: Exponentiated gradient
    t_remaining = time_budget - (time.monotonic() - t0)
    t_eg = min(t_remaining * 0.3, 80.0)
    if t_eg > 10:
        a_eg, obj_eg = exponentiated_gradient(a, t_eg, lr_init=0.008,
                                              n_top=max(20, n // 8), seed=seed)
        if obj_eg < best_obj:
            best_obj = obj_eg
            best_a = a_eg.copy()

    # Phase B: LP refinement
    t_remaining = time_budget - (time.monotonic() - t0)
    t_lp = min(t_remaining * 0.4, 120.0)
    if t_lp > 15:
        a_lp, obj_lp = lp_refine(best_a, max_iters=50, time_limit=t_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase C: Subgradient momentum
    t_remaining = time_budget - (time.monotonic() - t0)
    t_sg = min(t_remaining * 0.3, 60.0)
    if t_sg > 10:
        a_sg, obj_sg = subgradient_momentum(best_a, t_sg, lr_init=0.002,
                                            momentum=0.93, n_top=max(20, n // 8),
                                            seed=seed + 111)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()

    # Phase D: Final LP
    t_remaining = time_budget - (time.monotonic() - t0)
    if t_remaining > 10:
        a_lp2, obj_lp2 = lp_refine(best_a, max_iters=30, time_limit=t_remaining)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()

    return best_a, best_obj
