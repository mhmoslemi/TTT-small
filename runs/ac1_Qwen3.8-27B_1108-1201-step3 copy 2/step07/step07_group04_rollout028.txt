"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature soft-gradient
descent with momentum, spectral refinement, and peak shaving.
Key innovation: analytically-derived gradient of the soft-max smoothed objective,
computed efficiently via FFT-based cross-correlation, combined with adaptive
temperature control targeting a desired effective peak count.
"""

import numpy as np
from time import monotonic
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(x):
    """Project x onto the probability simplex {y >= 0, sum(y) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = (x.sum() - 1.0) / n
    return np.maximum(x - theta, 0.0)


def compute_gradient_parts(a, T):
    """
    Compute the gradient of f(a) = 2n * soft_max(T, conv(a,a)) for sum(a)=1.
    Returns (grad, g_val) where grad[i] = 4n*(h[i] - g_val*a[i]).
    h[i] = sum_j a[j]*W[i+j], W = softmax(T*(c - c.max())).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    # Numerically stable softmax
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    S = exp_shifted.sum()
    W = exp_shifted / S
    g_val = c_max + (1.0 / T) * np.log(S)
    
    # Compute h[i] = sum_j a[j] * W[i+j] via FFT cross-correlation
    # h[i] = (a * W_rev)[len(W)-1-i]
    W_rev = W[::-1]
    L = 1 << (n + len(W) - 1 - 1).bit_length()
    A = rfft(a, L)
    Wr = rfft(W_rev, L)
    conv_full = irfft(A * Wr, L)
    h = conv_full[len(W) - 1::-1][:n]
    
    grad = 4.0 * n * (h - g_val * a)
    return grad, g_val, c, c_max


def gradient_descent_adaptive(a0, time_limit, T_init=10.0, T_max=500.0,
                               lr_init=0.01, beta1=0.9, beta2=0.999,
                               eps=1e-8, target_neff=10.0, verbose=True,
                               max_iters=500000, restart_interval=2000):
    """
    Adaptive-temperature gradient descent on the simplex.
    T adapts to maintain target effective number of peaks in the softmax.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    T = T_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = monotonic()
    no_improve = 0
    lr = lr_init
    
    while t < max_iters:
        if monotonic() - t0 > time_limit:
            break
        
        grad, g_val, c, c_max = compute_gradient_parts(a, T)
        
        # Adapt temperature based on effective number of peaks
        shifted = T * (c - c_max)
        shifted -= shifted.max()
        exp_s = np.exp(shifted)
        W = exp_s / exp_s.sum()
        neff = 1.0 / np.sum(W * W)
        
        # Adjust T to target neff
        if neff < target_neff * 0.5:
            T = max(T_init, T * 0.9)
        elif neff > target_neff * 2.0:
            T = min(T_max, T * 1.1)
        
        t += 1
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Decay learning rate
        lr = lr_init * 0.999 ** (t // 1000)
        
        # Restart if stuck
        if no_improve > restart_interval:
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_init
            no_improve = 0
            T = max(T_init, T * 0.7)
        
        if verbose and (t + 1) % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    GD iter={t}, T={T:.1f}, neff={neff:.1f}, "
                  f"score={best_score:.6f}, lr={lr:.6f} ({elapsed:.0f}s)")
    
    return best_a, best_score

def spectral_refine(a0, time_limit, n_iter=50):
    """
    Spectral shaping: adjust a in frequency domain to flatten autocorrelation.
    Uses GS-style iterations in the amplitude domain.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    L = 1 << (2 * n - 1 - 1).bit_length()
    
    for it in range(n_iter):
        if monotonic() - t0 > time_limit:
            break
        
        # Current spectrum
        A = rfft(a, L)
        power = np.abs(A) ** 2
        
        # Target: flatten the power spectrum (but keep DC)
        # Blend current and flat
        flat = np.ones(L // 2 + 1)
        flat[0] = power[0]  # Keep DC
        blend = 0.3
        target_power = blend * flat + (1 - blend) * power
        
        # New amplitude = sqrt(target_power), keep phase
        new_amp = np.sqrt(target_power)
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        
        # Inverse FFT, take first n, clip, normalize
        a_new = irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                improved = True
        
        if not improved:
            # Try small random perturbation
            a_trial = a + np.random.standard_normal(n) * 0.0003
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                improved = True
        
        if not improved:
            stuck += 1
            if stuck > 100:
                break
        else:
            stuck = 0
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_score(a):
    """Evaluate using the actual evaluate_sequence function."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(evaluate_sequence(a.tolist()))
