```python
"""
Minimize  2*n*max_k (a*a)[k] / (sum a)^2  over non-negative sequences a.

The quantity is homogeneous of degree 0, so we work with sum(a)=1 and minimise
  h(a) = 2*n*max_k (a conv a)_k .

Strategy (deliberately different from the LP-only SOTA):
  1.  Fast first-order "fictitious play" with LP best-responses (PDLP/ECOS)
      at a few sequence lengths to land in the good, near-global valley.
  2.  A smooth-continuation L-BFGS polish: minimise a log-sum-exp surrogate
      of max_k (a*a)_k with a *decreasing* temperature in a softmax
      re-parameterisation (a = softmax(z)).  An O(n log n) analytic gradient
      (FFT auto-convolution + a reverse-convolution trick) makes this cheap.
  3.  Cheap "bigger-n" exploitation: upsample the best profile and re-polish
      with L-BFGS; a finer discretisation of the same good profile lowers the
      value.  This is the main cheap win and is fast because L-BFGS is cheap.
  4.  A projected-subgradient / peak-shave finisher on the global best to
      close the last fraction of a percent.

All helpers are top level (no nested functions / no lambdas).
"""

import time
import numpy as np
from scipy.fft import rfft, irfft

# ----------------------------------------------------------------------------
# globals used to pass state to top-level objective / gradient (no closures)

_G = {"T": 1.0, "n": 0, "a": None, "c": None, "prev_z": None, "prev_J": None, "prev_gz": None}
_SOLVER = {"name": None}


# ----------------------------------------------------------------------------
# convolution helpers (FFT based for speed, exact enough)
# ----------------------------------------------------------------------------
def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def convolve_full(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    nx, ny = len(x), len(y)
    out_len = nx + ny - 1
    if nx + ny < 32:
        return np.convolve(x, y, mode="full")
    size = _next_pow2(out_len)
    X = rfft(x, size)
    Y = rfft(y, size)
    return irfft(X * Y, size)[:out_len]


def autoconv(a):
    return convolve_full(a, a)


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if s <= 1e-12:
        return float("inf")
    n = len(a)
    return 2.0 * n * float(np.max(autoconv(a))) / (s * s)


def interp_upsample(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    s = float(f.sum())
    if s <= 1e-12:
        return np.ones(new_n) / new_n
    f = f / s
    if len(f) == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, len(f))
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    g = np.maximum(g, 0.0)
    gs = g.sum()
    if gs <= 1e-12:
        return np.ones(new_n) / new_n
    return g / gs


def simplex_project(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1.0)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.ones(n) / n
    ws = w.sum()
    if ws <= 1e-12:
        return np.ones(n) / n
    return w / ws

# LP best response  (used inside fictitious play)

def get_lp_solver():
    if not _HAS_CP:
        return None
    if _SOLVER["name"] is not None:
        return _SOLVER["name"]
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CVXOPT, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _SOLVER["name"] = solver_name
                return solver_name
        except Exception:
            continue
    _SOLVER["name"] = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    d = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    valid = (d >= 0) & (d <= n - 1)
    idx = np.clip(d, 0, n - 1)
    T = np.where(valid, a[idx], 0.0)
    return T


def solve_lp_best_response(a_fixed, solver):
    a_fixed = np.asarray(a_fixed, dtype=np.float64)
    n = len(a_fixed)
    s = a_fixed.sum()
    if s <= 1e-12:
        return None
    a = a_fixed / s
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    for sol in ([solver] if solver is not None else []) + [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            prob.solve(solver=sol)
            if prob.status in ("optimal", "optimal_inaccurate"):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return None
    b = np.asarray(b_var.value, dtype=np.float64)
    b = np.maximum(b, 0.0)
    bs = b.sum()
    if bs <= 1e-12:
        return None
    return b / bs


def fictitious_play(a0, time_limit, solver):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    b = a.copy()
    a_avg = a.copy()
    b_avg = b.copy()
    best = a.copy()
    best_s = local_score(a)
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        a_avg = (it * a_avg + a) / (it + 1)
        b_avg = (it * b_avg + b) / (it + 1)
        cands = (a, b, 0.5 * (a + b), a_avg, 0.5 * (a_avg + b_avg))
        for c in cands:
            c = np.maximum(c, 0.0)
            cs = c.sum()
            if cs <= 1e-12:
                continue
            cn = c / cs
            sc = local_score(cn)
            if sc < best_s - 1e-13:
                best_s = sc
                best = cn.copy()
        if it % 40 == 0:
            print("    FP n=%d it=%d t=%ds best=%.6f" % (n, it, int(time.monotonic() - t0), best_s))
        if it > 2500:
            break
    return best, best_s


# ----------------------------------------------------------------------------
# smooth-continuation L-BFGS polish (log-sum-exp surrogate, softmax params)
# ----------------------------------------------------------------------------
def _lse_core(z):
    n = _G["n"]
    z = np.asarray(z, dtype=np.float64)
    if _G["prev_z"] is not None and z.shape == _G["prev_z"].shape and np.array_equal(z, _G["prev_z"]):
        return _G["prev_J"], _G["prev_gz"]
    zs = z - z.max()
    e = np.exp(zs)
    a = e / e.sum()
    c = autoconv(a)
    T = _G["T"]
    T = max(1e-12, T)
    cs = (c - np.max(c)) / T
    w = np.exp(cs)
    w = w / w.sum()
    J = np.max(c) + T * np.log(float(np.sum(np.exp(cs))))
    # gradient of LSE wrt a : 2 * (conv(a_rev, w))[n-1 : 2n-1]
    g = 2.0 * convolve_full(a[::-1], w)[n - 1:2 * n - 1]
    ag = float(np.dot(a, g))
    gz = a * (g - ag)
    _G["prev_z"] = z.copy()
    _G["prev_J"] = float(J)
    _G["prev_gz"] = gz
    return float(J), gz


def lse_objfun(z):
    J, _ = _lse_core(z)
    return J


def lse_jacfun(z):
    _, gz = _lse_core(z)
    return gz


def polish_lbgfs(a0, time_limit, T_fracs):
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a0.sum()
    a = a0 / s
    best = a.copy()
    best_s = local_score(a)
    n = len(a)
    t0 = time.monotonic()
    for frac in T_fracs:
        if time.monotonic() - t0 > time_limit:
            break
        c = autoconv(a)
        cmax = float(np.max(c))
        if cmax <= 1e-15:
            break
        T = max(1e-5 * cmax, frac * cmax)
        _G["T"] = T
        _G["n"] = n
        _G["prev_z"] = None
        z0 = np.log(np.maximum(a, 1e-30))
        z0 = z0 - z0.mean()
        sub = 1.0
        while True:
            if time.monotonic() - t0 > time_limit:
                break
            try:
                res = minimize(lse_objfun, z0, jac=lse_jacfun, method="L-BFGS-B",
                               options={"maxiter": 600, "gtol": 1e-9, "ftol": 1e-14,
                                        "maxls": 40, "maxfun": 600, "disp": False})
                z0 = res.x
            except Exception:
                break
            zs = z0 - z0.max()
            e = np.exp(zs)
            a = e / e.sum()
            sc = local_score(a)
            if sc < best_s - 1e-14:
                best_s = sc
                best = a.copy()
            # stop inner loop once temperature continuation gives no gain
            break
        if int(time.monotonic() - t0) % 10 < 1:
            print("    LBGFS frac=%.4f T=%.3g t=%ds best=%.6f" % (frac, T, int(time.monotonic() - t0), best_s))
    return best, best_s


# ----------------------------------------------------------------------------
# projected sub-gradient / peak-shave finisher
# ----------------------------------------------------------------------------
def _topk_grad(a, T_frac, top_k):
    n = len(a)
    c = autoconv(a)
    cmax = float(np.max(c))
    T = max(1e-12, T_frac * cmax)
    cs = (c - cmax) / T
    w = np.exp(cs)
    w = w / w.sum()
    # focus on the top_k peaks (harder)
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        wm = np.zeros_like(w)
        wm[idx] = w[idx]
        ws = wm.sum()
        if ws > 1e-15:
            wm = wm / ws
        w = wm
    g = 2.0 * convolve_full(a[::-1], w)[n - 1:2 * n - 1]
    return g, cmax


def polish_subgrad(a0, time_limit, T_frac=0.01, top_k=None):
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best = a.copy()
    best_s = local_score(a)
    t0 = time.monotonic()
    lr_scale = 0.02
    stall = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        g, cmax = _topk_grad(a, T_frac, top_k)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        d = -g / gn
        improved = False
        for base in (lr_scale, lr_scale * 0.5, lr_scale * 0.25, lr_scale * 2.0, lr_scale * 5.0):
            for eta in (base, base * 0.4, base * 1.6):
                trial = a + eta * d
                # project onto the simplex (keeps sum=1 & non-neg)
                trial = simplex_project(trial)
                sc = local_score(trial)
                if sc < best_s - 1e-14:
                    best_s = sc
                    best = trial.copy()
                    a = trial
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall > 400:
                a = best + np.random.standard_normal(len(best)) * 0.0015
                a = simplex_project(a)
                lr_scale = max(0.002, lr_scale * 0.7)
                stall = 0
        else:
            lr_scale = min(0.05, lr_scale * 1.02)
            if it % 500 == 0:
                print("    SUBGRAD t=%ds best=%.6f lr=%.4f" % (int(time.monotonic() - t0), best_s, lr_scale))
    return best, best_s


# ----------------------------------------------------------------------------
# helper: symmetrise only if it strictly helps
# ----------------------------------------------------------------------------
def maybe_symmetrise(a):
    b = 0.5 * (a + a[::-1])
    if local_score(b) < local_score(a) - 1e-13:
        return b
    return a

# orchestration
# ----------------------------------------------------------------------------
def _get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            p = np.clip(p, 0.0, 1000.0)
            if p.sum() > 1e-10:
                return p / p.sum()
    except Exception:
        return None
    return None


def propose_candidate(seed=0, budget_s=1800.0):
    np.random.seed(int(seed))
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    T_fracs = [0.12, 0.07, 0.04, 0.025, 0.016, 0.010, 0.006, 0.004, 0.0025, 0.0016, 0.0010]

    parent = _get_parent()
    if parent is None:
        parent = np.ones(768) / 768.0
        parent += 0.01 * np.cos(3.0 * np.linspace(0, 2 * np.pi, 768))
        parent = np.maximum(parent, 0.0)
        parent = parent / parent.sum()

    n_parent = len(parent)

    best = parent.copy()
    best_s = float("inf")
    try:
        best_s = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_s = local_score(best)
    print("parent n=%d score=%.6f" % (n_parent, best_s))

    solver = get_lp_solver()
    print("LP solver:", solver)

    def consider(a, tag):
        nonlocal best, best_s
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.size == 0 or a.sum() <= 1e-10:
            return
        a = a / a.sum()
        sc = local_score(a)
        if sc + 1e-12 < best_s:
            try:
                real = float(evaluate_sequence(a.tolist()))
            except Exception:
                real = sc
            if np.isfinite(real) and real + 1e-12 < best_s:
                best_s = real
                best = a.copy()
                print("NEW BEST [%s] n=%d score=%.6f (t=%ds)" % (tag, a.size, best_s, int(time.monotonic() - (deadline - float(budget_s) + 35))))

    # ---- Phase 1: fictitious play at a few lengths --------------------------
    fp_lengths = sorted({n_parent, 1024, 1536})
    fp_lengths = [n for n in fp_lengths if 256 <= n <= 1536]
    if not fp_lengths:
        fp_lengths = [768]
    # give ~ equal but bounded time to each
    per = max(60.0, min(220.0, (deadline - time.monotonic()) / max(1, len(fp_lengths)) - 60.0))
    for n_try in fp_lengths:
        if time.monotonic() >= deadline:
            break
        rem = deadline - time.monotonic() - 60.0
        t_alloc = max(0.0, min(per, rem))
        if t_alloc < 25.0:
            continue
        a_init = interp_upsample(parent, n_try)
        print("PHASE1 FP n=%d budget=%.0fs" % (n_try, t_alloc))
        a_fp, _ = fictitious_play(a_init, t_alloc, solver)
        consider(a_fp, "FP%d" % n_try)
        # quick polish so we capture gains even under time pressure
        if time.monotonic() < deadline:
            rp = max(5.0, min(35.0, deadline - time.monotonic() - 40.0))
            a_pol, _ = polish_lbgfs(a_fp, rp, T_fracs[:5])
            consider(a_pol, "FP%d_pol" % n_try)

    # ---- Phase 2: heavy polish on the current best --------------------------
    if time.monotonic() < deadline:
        rem = deadline - time.monotonic() - 40.0
        print("PHASE2 polish best n=%d (score=%.6f)" % (len(best), best_s))
        a_p, _ = polish_lbgfs(best, rem * 0.7, T_fracs)
        consider(a_p, "polish")
        a_p = maybe_symmetrise(a_p)
        consider(a_p, "polish_sym")
        a_p, _ = polish_subgrad(a_p, rem * 0.25, T_frac=0.004, top_k=max(8, len(a_p) // 128))
        consider(a_p, "subgrad")

    # ---- Phase 3: bigger-n exploitation (upsample + cheap L-BFGS) ----------
    if time.monotonic() < deadline:
        base_n = len(best)
        targets = set()
        for mult in (2, 3, 4):
            targets.add(max(512, base_n * mult))
        for extra in (2048, 4096, 6144):
            targets.add(extra)
        targets = sorted({int(t) for t in targets if base_n < int(t) <= 8192})
        per_up = max(8.0, min(30.0, (deadline - time.monotonic() - 30.0) / max(1, len(targets))))
        for N in targets:
            if time.monotonic() >= deadline:
                break
            rem = deadline - time.monotonic() - 25.0
            if rem < 6.0:
                break
            t_alloc = min(per_up, rem)
            a_up = interp_upsample(best, N)
            a_up, _ = polish_lbgfs(a_up, t_alloc, T_fracs[-6:])
            consider(a_up, "up%d" % N)

    # ---- Phase 4: final close on the global best ---------------------------
    if time.monotonic() < deadline:
        rem = deadline - time.monotonic() - 6.0
        a_f, _ = polish_subgrad(best, rem * 0.5, T_frac=0.0025, top_k=max(6, len(best) // 128))
        consider(a_f, "final_subgrad")
        a_f, _ = polish_lbgfs(a_f, rem * 0.4, [0.004, 0.0025, 0.0016, 0.0010])
        consider(a_f, "final_lbgfs")
        a_f = maybe_symmetrise(a_f)
        consider(a_f, "final_sym")

    # ---- final safety evaluation -------------------------------------------
    try:
        final_s = float(evaluate_sequence(best.tolist()))
    except Exception:
        final_s = local_score(best)
    if not np.isfinite(final_s):
        best = np.ones(512) / 512.0
        final_s = float(evaluate_sequence(best.tolist()))
    print("FINAL n=%d score=%.6f" % (len(best), final_s))
    return [float(x) for x in best]
```
