"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def cross_corr_wa(w, a):
    """Compute cross-correlation of w (len 2n-1) and a (len n): result[j] = sum_k w[k]*a[k-j]"""
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    return correl[:n]

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def lse_func_and_grad(z, T, n):
    """Compute LSE-smoothed objective and its gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    lse = m + np.log(exp_scaled.sum())
    obj = 2.0 * n * T * lse
    # Gradient
    cc = cross_corr_wa(w, a)
    grad_a = 4.0 * n * cc
    mean_g = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_g)
    return obj, grad_z

def lbgfs_logspace(a_init, time_limit, T, maxiter=5000, verbose=False):
    """L-BFGS optimization in log-space with LSE smoothing at temperature T."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        f, _ = lse_func_and_grad(z, T, n)
        return f
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        _, g = lse_func_and_grad(z, T, n)
        return g
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 50, 'disp': False})
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a_init.copy()
    return a_out

                            eta_decay=0.9999, T=0.01, verbose=False):
    """Momentum-based mirror descent with LSE-smoothed gradient."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_from_a(a)
    eta = eta_start
    t0 = time.monotonic()
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        obj, grad_z = lse_func_and_grad(z, T, n)
        v = beta * v - eta * grad_z
        z = z + v
        z = z - z.max()  # numerical stability
        a = softmax_from_z(z)
        
        sc = score_from_a(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        eta *= eta_decay
        if verbose and it % 10000 == 0:
            print(f"    MMD iter={it}: score={sc:.7f} best={best_score:.7f} eta={eta:.6f}")
    
    return best_a, best_score

def exact_subgradient_momentum(a_init, time_limit, eta_start=0.01, beta=0.8,

    n = len(a_init)

def perturb_and_optimize(a_init, rng, n_perturbs, time_per_perturb, T=5.0):
    """Generate perturbed versions and briefly optimize each."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = score_from_a(a_init)
    
    for i in range(n_perturbs):
        # Different perturbation strategies
        strategy = i % 4
        if strategy == 0:
            # Gaussian noise
            noise = rng.standard_normal(n) * 0.05
            a_pert = a_init + noise
        elif strategy == 1:
            # Random coordinate boost
            idx = rng.integers(0, n)
            a_pert = a_init.copy()
            a_pert[idx] *= rng.uniform(0.5, 2.0)
        elif strategy == 2:
            # Block perturbation
            block_size = rng.integers(10, 50)
            start = rng.integers(0, n - block_size)
            a_pert = a_init.copy()
            a_pert[start:start+block_size] *= rng.uniform(0.5, 1.5)
        else:
            # Mirror flip
            a_pert = a_init[::-1].copy()
            if rng.random() < 0.5:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def coordinate_descent_polish(a_init, time_limit):
    """Quick coordinate descent polish."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    factors = [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f
                diff = a_trial.sum() - 1.0
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            break
    
    return best_a, best_score
