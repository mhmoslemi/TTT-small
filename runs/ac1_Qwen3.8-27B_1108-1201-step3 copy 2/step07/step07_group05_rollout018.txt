    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)

def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)

def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z

def solve_best_response_lp(f, M, floor=1e-10):
    """
    Solve the LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    # Build constraint matrix A: (2n-1) x n
    # A[k, i] = f[i] if k-i in [0, n-1], else 0
    # i.e., i in [max(0, k-n+1), min(n-1, k)]
    
    # Use sparse representation for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: minimize -sum(g) subject to A*g <= M*ones, g >= 0
    c = -np.ones(n)
    b = M * np.ones(2 * n - 1)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.status == 0:
        g = res.x
        # Add small floor to avoid exact zeros
        g = g + floor
        g = g / g.sum()
        return g
    else:
        # Fallback: return uniform
        return np.ones(n) / n

def fictitious_play(f_init, n_iters, time_limit, t_init=0.5, t_end=0.05, 
                    lp_floor=1e-10, verbose=True):
    """
    LP-based fictitious play to minimize max(f*f) subject to sum(f)=1, f>=0.
    """
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s  # normalize to sum=1
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current M
        conv = fast_convolve(f)
        M = conv.max()
        current_score = 2.0 * n * M  # since sum(f) = 1
        
        # Solve LP for best response
        g = solve_best_response_lp(f, M, floor=lp_floor)
        
        # Blend
        progress = it / max(1, n_iters)
        t = t_init * (t_end / t_init) ** progress
        
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        new_score = compute_score(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        
        f = f_new
        
        if verbose and (it % 50 == 0 or it == n_iters - 1):
            elapsed = time.monotonic() - t0
            print(f"    FP iter={it}, M={M:.8f}, score={current_score:.8f}, "
                  f"best={best_score:.8f}, t={t:.4f}, time={elapsed:.1f}s")
    
    return best_f

def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass
    
    result['seconds'] = time.monotonic() - t0
    return result

def coordinate_descent(a0, time_limit, n_pairs_per_iter=200):
    """Coordinate descent using pairwise moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        pairs = []
        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((i, j))
        
        for (i, j) in pairs:
            if a[i] < 1e-14:
                continue
            base = min(a[i], max(a[j], 1e-10))
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                t_val = frac * base
                if t_val >= a[i] - 1e-15:
                    t_val = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial
                    best_a = a_trial.copy()
                    improved = True
                if a[j] > 1e-15:
                    t2 = frac * a[j]
                    a_trial2 = a.copy()
                    a_trial2[j] -= t2
                    a_trial2[i] += t2
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial2 = a_trial2 / ss2
                    sc2 = compute_score(a_trial2)
                    if sc2 < best_score - 1e-14:
                        best_score = sc2
                        a = a_trial2
                        best_a = a_trial2.copy()
                        improved = True
        
        if not improved:
            noise = rng.standard_normal(n) * 0.001
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = compute_score(a)
            best_a = a.copy()
    
    return best_a, best_score

def peak_equalization(a0, time_limit, n_peaks=10):
    """Iteratively try to equalize the top peaks of the auto-correlation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        top_k = min(n_peaks, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        max_peak = conv.max()
        min_peak = conv[peak_indices].min()
        
        if max_peak - min_peak < 1e-10:
            break
        
        # Try to reduce the maximum peak by modifying contributing elements
        max_idx = int(np.argmax(conv))
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]
        
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            for _ in range(10):
                p = rng.integers(0, n)
                a_trial = a.copy()
                delta = a[p] * 0.1 * rng.standard_normal()
                a_trial[p] -= delta
                mask = np.ones(n, dtype=bool)
                mask[p] = False
                oth2 = a_trial[mask]
                o2s = oth2.sum()
                if o2s > 1e-15:
                    a_trial[mask] = oth2 + delta * oth2 / o2s
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, label, best_seq, best_score):

        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play as the
primary optimization engine, combined with L-BFGS local refinement, coordinate
descent, and peak equalization. The LP-based fictitious play computes exact best
responses via linear programming to guide the sequence toward flat auto-convolution.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def build_conv_matrix(f):
    """Build the convolution constraint matrix for the LP."""
    n = len(f)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            A[k, i_start:i_end + 1] = f[i_start:i_end + 1]
    return A.tocsr()


def solve_best_response_lp(f, M, floor=1e-12):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns normalized optimal g.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = M * np.ones(2 * n - 1)

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.status == 0 and res.x is not None:
            g = res.x.astype(np.float64)
            g_sum = g.sum()
            if g_sum > 1e-15:
                g = g / g_sum
                # Add small floor to preserve support
                g = g + floor
                g = g / g.sum()
                return g
    except Exception:
        pass

    # Fallback: return uniform
    return np.ones(n) / n


def fictitious_play(f_init, n_iters, time_limit, t_init=0.5, t_end=0.05,
                    lp_floor=1e-12, verbose=True, tol=1e-6):
    """
    LP-based fictitious play to minimize max(f*f) subject to sum(f)=1, f>=0.
    """
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s

    t0 = time.monotonic()
    best_f = f.copy()
    best_score = compute_score(f)
    lp_time = 0.0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()
        current_score = 2.0 * n * M

        t_start_lp = time.monotonic()
        g = solve_best_response_lp(f, M, floor=lp_floor)
        lp_time += time.monotonic() - t_start_lp

        # Check convergence: is g close to f?
        diff = np.abs(g - f).sum()
        if diff < tol and it > 20:
            if verbose:
                print(f"    FP converged at iter={it}, diff={diff:.2e}")
            break

        # Adaptive step size
        progress = it / max(1, n_iters)
        t = t_init * (t_end / t_init) ** progress

        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / f_new_sum

        new_score = compute_score(f_new)
        if new_score < best_score - 1e-15:
            best_score = new_score
            best_f = f_new.copy()

        f = f_new

        if verbose and (it % 50 == 0 or it == n_iters - 1):
            elapsed = time.monotonic() - t0
            lp_avg = lp_time / max(1, it + 1) * 1000
            print(f"    FP iter={it}, score={current_score:.8f}, "
                  f"best={best_score:.8f}, t={t:.4f}, "
                  f"lp_avg={lp_avg:.1f}ms, time={elapsed:.1f}s")

    return best_f


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass

    result['seconds'] = time.monotonic() - t0
    return result


def coordinate_descent(a0, time_limit, n_pairs_per_iter=200):
    """Coordinate descent using pairwise moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stale = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        pairs = []
        for _ in range(n_pairs_per_iter):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((i, j))

        for (i, j) in pairs:
            if a[i] < 1e-14:
                continue
            best_local = best_score
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                t_val = frac * min(a[i], max(a[j], 1e-10))
                if t_val >= a[i] - 1e-15:
                    t_val = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    a = a_trial
                    best_score = sc
                    best_a = a_trial.copy()
                    improved = True
                if a[j] > 1e-15:
                    t2 = frac * a[j]
                    a_trial2 = a.copy()
                    a_trial2[j] -= t2
                    a_trial2[i] += t2
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial2 = a_trial2 / ss2
                    sc2 = compute_score(a_trial2)
                    if sc2 < best_local - 1e-14:
                        best_local = sc2
                        a = a_trial2
                        best_score = sc2
                        best_a = a_trial2.copy()
                        improved = True

        if improved:
            stale = 0
        else:
            stale += 1
            if stale > 5:
                noise = rng.standard_normal(n) * 0.002
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                stale = 0

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=10):
    """Iteratively equalize top peaks of the auto-correlation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        top_k = min(n_peaks, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        max_peak = conv.max()
        min_peak = conv[peak_indices].min()

        if max_peak - min_peak < 1e-10:
            break

        max_idx = int(np.argmax(conv))
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            for _ in range(15):
                p = int(rng.integers(0, n))
                a_trial = a.copy()
                delta = a[p] * 0.1 * rng.standard_normal()
                a_trial[p] -= delta
                mask = np.ones(n, dtype=bool)
                mask[p] = False
                oth2 = a_trial[mask]
                o2s = oth2.sum()
                if o2s > 1e-15:
                    a_trial[mask] = oth2 + delta * oth2 / o2s
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")

def nesterov_gradient(a0, time_limit, T=500, lr_init=0.03, momentum=0.9):
    """Nesterov accelerated gradient descent in log-space."""
    n = len(a0)
    z = z_from_a(a0)
    v = np.zeros(n)
    best_z = z.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        z_lookahead = z - momentum * v
        a_la = softmax_from_z(z_lookahead)
        grad = smoothed_gradient(z_lookahead, T, n)
        v = momentum * v + lr * grad
        z_new = z - v
        z_new = np.clip(z_new, -25, 25)
        a_new = softmax_from_z(z_new)
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_z = z_new.copy()
        z = z_new
        lr *= 0.9997

    a_out = softmax_from_z(best_z)
    return a_out, best_score
