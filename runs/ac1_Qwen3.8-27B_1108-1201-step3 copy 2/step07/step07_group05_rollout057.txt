"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. LP-based Fictitious Play (fundamentally different from gradient methods)
2. Mirror descent with adaptive temperature and iterate averaging
3. Peak-targeted coordinate descent focusing on contributing positions
4. Spectral clipping with non-negativity projection

import numpy as np
import time
import scipy.sparse
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def build_conv_matrix_sparse(f):
    """Build sparse (2n-1) x n matrix A such that A @ g = conv(f, g)."""
    n = len(f)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(size, j + n)
        for k in range(k_start, k_end):
            v = f[k - j]
            if v > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    if len(rows) == 0:
        return scipy.sparse.csr_matrix((size, n))
    A = scipy.sparse.coo_matrix((vals, (rows, cols)), shape=(size, n))
    return A.tocsr()


def solve_best_response_lp(f, M):
    """
    Solve: min t s.t. conv(f,g)_k <= t for all k, sum(g)=1, g>=0
    Returns optimal g and t.
    """
    n = len(f)
    size = 2 * n - 1
    A = build_conv_matrix_sparse(f)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    # Minimize t => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality: A@g - t <= 0 => [A, -1] @ [g, t] <= 0
    A_dense = A.toarray()
    A_ub = np.hstack([A_dense, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x[:n], 0)
            g = g / g.sum()
            t_val = res.x[n]
            return g, t_val
    except Exception:
        pass
    
    # Fallback: uniform
    return np.ones(n) / n, M


def fictitious_play(f0, time_limit, max_iters=5000, verbose=False):
    """
    LP-based fictitious play: iteratively find best response g (minimizes max peak
    of conv(f,g)), then mix with current iterate.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    if s < 1e-15:
        return f0.copy(), compute_score(f0)
    f = f / s
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    iters = 0
    
    # Running average (Hedge algorithm style)
    avg_f = f.copy()
    avg_count = 0
    avg_best_score = best_score
    
    mix_schedule = "adaptive"
    base_mix = 0.3
    min_mix = 0.02
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv_f = fast_convolve(f)
        M = conv_f.max()
        
        # Solve best response LP
        g, t_star = solve_best_response_lp(f, M)
        
        # Mixing rate
        if mix_schedule == "adaptive":
            mix = max(min_mix, base_mix / (1.0 + 0.01 * iters))
        else:
            mix = 0.1
        
        # Update
        f_new = (1.0 - mix) * f + mix * g
        f_new = np.maximum(f_new, 0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = f
        else:
            f_new = f_new / f_new_sum
        
        score = compute_score(f_new)
        f = f_new
        
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        # Update average
        avg_f = avg_f * (avg_count / (avg_count + 1)) + f / (avg_count + 1)
        avg_count += 1
        
        if avg_count % 20 == 0:
            avg_score = compute_score(avg_f)
            if avg_score < avg_best_score:
                avg_best_score = avg_score
                if avg_score < best_score:
                    best_score = avg_score
                    best_f = avg_f.copy()
        
        if verbose and iters % 100 == 0:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    FP iter={iters}, best={best_score:.8f}, t*={t_star:.6f}, M={M:.6f}, rem={rem:.0f}s")
        
        # Convergence check
        if iters > 100 and abs(score - best_score) < 1e-12 and iters % 500 == 0:
            # Try to escape with perturbation
            rng = np.random.default_rng(iters)
            f_pert = f + rng.standard_normal(n) * 0.005
            f_pert = np.maximum(f_pert, 0)
            f_pert = f_pert / f_pert.sum()
            f = f_pert
    
    return best_f, best_score


def mirror_descent_averaged(f0, time_limit, T_init=500, T_min=50,
                            lr_init=0.1, decay=0.9995, verbose=False):
    """
    Mirror descent (softmax) with adaptive temperature and iterate averaging.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    iters = 0
    
    # Iterate averaging
    avg_f = f.copy()
    avg_weight_sum = 0.0
    current_weight = 1.0
    
    T = T_init
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Compute subgradient using softmax-weighted peaks
        conv = fast_convolve(f)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        
        # Subgradient: d/d a_i [max_k (a*a)_k] approx = 2 * sum_k w_k * a_{k-i}
        # This is cross-correlation of w with a (reversed)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(f, fft_size)
        grad = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        
        # Scale by 2n (the actual objective has factor 2n)
        # But since we're minimizing 2n*max/sum^2 and sum=1, gradient of 2n*max is 2n*grad
        # The 2n factor doesn't affect the direction, just the step size
        grad = grad / grad.max()  # Normalize for stability
        
        # Mirror descent update in log-space
        # a_new = f * exp(-lr * grad) / sum
        logits = -lr * grad
        logits_max = logits.max()
        exp_logits = np.exp(logits - logits_max)
        f_new = f * exp_logits
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = f
        else:
            f_new = f_new / f_new_sum
        
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        score = compute_score(f_new)
        f = f_new
        
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        # Update iterate average (importance-weighted)
        avg_f = (avg_weight_sum * avg_f + current_weight * f) / (avg_weight_sum + current_weight)
        avg_weight_sum += current_weight
        current_weight *= 0.999
        
        if iters % 30 == 0:
            avg_score = compute_score(avg_f)
            if avg_score < best_score:
                best_score = avg_score
                best_f = avg_f.copy()
        
        # Adaptive parameters
        T *= 0.9998
        T = max(T, T_min)
        lr *= decay
        
        if verbose and iters % 200 == 0:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    MD iter={iters}, best={best_score:.8f}, T={T:.1f}, lr={lr:.6f}, rem={rem:.0f}s")
    
    return best_f, best_score


def peak_targeted_cd(f0, time_limit, n_trials_per_iter=200, verbose=False):
    """
    Coordinate descent that focuses moves on positions contributing to top peaks.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(f)
        
        # Find top peaks and their contributing positions
        peak_idx = int(np.argmax(conv))
        n_peaks = min(5, 2 * n - 1)
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        
        # For each top peak, find the positions that contribute most
        # conv[peak]_k = sum_i f[i] * f[k-i]
        # Position i contributes f[i] * f[peak-i] to peak at lag 'peak'
        
        # Gather important positions
        important = np.zeros(n)
        for pk in top_peaks:
            pk = int(pk)
            i_start = max(0, pk - n + 1)
            i_end = min(n - 1, pk)
            for i in range(i_start, i_end + 1):
                j = pk - i
                if 0 <= j < n:
                    important[i] += f[i] * f[j]
        
        # Also include the positions of the current max
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n - 1, peak_idx)
        for i in range(i_start, i_end + 1):
            j = peak_idx - i
            if 0 <= j < n:
                important[i] = max(important[i], f[i] * f[j])
        
        # Normalize to get sampling weights
        important = np.maximum(important, 1e-15)
        w = important / important.sum()
        
        # Generate moves biased towards important positions
        improved = False
        for _ in range(n_trials_per_iter):
            # Sample two positions
            i = rng.choice(n, p=w)
            j = rng.integers(0, n)
            if i == j:
                continue
            if f[i] < 1e-14:
                continue
            
            # Try transferring mass
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                t = frac * f[i]
                f_trial = f.copy()
                f_trial[i] -= t
                f_trial[j] += t
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                sc = compute_score(f_trial)
                if sc < best_score - 1e-14:
                    f = f_trial
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                    break
            
            # Also try: scale one position and redistribute
            if not improved and rng.random() < 0.3:
                factor = 1.0 + rng.standard_normal() * 0.15
                f_trial = f.copy()
                old_val = f_trial[i]
                new_val = max(0.0, old_val * factor)
                diff = old_val - new_val
                f_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = f_trial[mask]
                osum = others.sum()
                if osum > 1e-15:
                    f_trial[mask] = others + diff * others / osum
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                sc = compute_score(f_trial)
                if sc < best_score - 1e-14:
                    f = f_trial
                    best_score = sc
                    best_f = f.copy()
                    improved = True
        
        if not improved and iters % 50 == 0:
            # Perturb to escape
            noise = rng.standard_normal(n) * 0.003
            f = best_f + noise
            f = np.maximum(f, 0)
            f = f / f.sum()
    
    return best_f, best_score


def spectral_clip_iteration(f0, time_limit, T_init=None, verbose=False):
    """
    Spectral clipping: clip the power spectrum, inverse FFT, project to non-negative.
    Iteratively reduces peaks in autocorrelation.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    iters = 0
    
    if T_init is None:
        conv0 = fast_convolve(f)
        T_init = conv0.max() * 0.8
    
    T = T_init
    T_min = conv0.min() * 1.01 if conv0.min() > 0 else 1e-6
    
    F = fft(f)
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Current power spectrum
        power = np.abs(F) ** 2
        power_max = power.max()
        
        # Clip: reduce peaks in power spectrum
        power_clipped = np.minimum(power, T)
        
        # Reconstruct with original phase
        phase = np.angle(F)
        F_new = np.sqrt(power_clipped) * np.exp(1j * phase)
        
        # Inverse FFT
        f_new = np.real(ifft(F_new))
        
        # Project to non-negative
        f_new = np.maximum(f_new, 0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            break
        f_new = f_new / f_new_sum
        
        score = compute_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        
        f = f_new
        F = fft(f)
        
        # Adapt threshold
        T *= 0.999
        T = max(T, T_min)
        
        if verbose and iters % 100 == 0:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    SC iter={iters}, best={best_score:.8f}, T={T:.6f}, rem={rem:.0f}s")
    
    return best_f, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def lbgfs_smoothed(f0, time_limit, T, maxiter=5000, verbose=False):
    """L-BFGS optimization with softmax-smoothed max."""
    n = len(f0)
    z0 = z_from_a(f0)
    t0 = time.monotonic()
    best_a = f0.copy()
    best_score = compute_score(f0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * log_sum_exp)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 50,
                                'disp': False})
        a_out = softmax_from_z(res.x)
        sc = compute_score(a_out)
        if sc < best_score:
            best_score = sc
            best_a = a_out
    except Exception:
        pass
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):

    # Fallback
    if best_seq is None:

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
