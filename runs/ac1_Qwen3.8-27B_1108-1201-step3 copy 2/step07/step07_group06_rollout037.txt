    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lbgfs_polish(a0, time_limit, T=100, maxiter=3000):
    """L-BFGS polishing with soft-max smoothing."""
    n = len(a0)

    fft_size = 1 << (3 * n - 3).bit_length()

    def soft_max_obj(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            stopped[0] = True
            return 1e15
        a = softmax_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        b_norm = b / T - max_b / T
        exp_b = np.exp(b_norm)
        return 2.0 * n * T * (max_b / T + np.log(exp_b.sum()))
    
    def soft_max_grad(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            stopped[0] = True
            return np.zeros_like(z)

        b = fast_convolve(a)

        b_norm = b / T - max_b / T

            fun=soft_max_obj, x0=z0, jac=soft_max_grad,

            break
        
        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)
        
        if peak_val <= target * 1.001:
            continue
        
        # Find contributing pairs for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]
        
        # Also check secondary peaks
        b_secondary = b.copy()
        b_secondary[peak_k] = -np.inf
        b_secondary[0] = -np.inf
        b_secondary[-1] = -np.inf
        
        improved = False
        
        # Try reducing top contributing pairs
        for ci in top_idx:
            if improved:
                break
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue
            
            for f1 in [0.92, 0.85, 0.78, 0.7]:
                if improved:
                    break
                for f2 in [1.0, 1.03, 1.06, 1.1]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = score_of_a(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
        
        if not improved:
            # Try single-element reduction with redistribution
            for ci in top_idx[:5]:
                if improved:
                    break
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in [0.85, 0.75, 0.65]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    score = score_of_a(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
        
        if not improved and trial > 200 and trial % 100 == 0:
            # Frequency-domain perturbation to escape
            freq = np.fft.rfft(a)
            freq[1:-1] *= (1.0 + 1j * np.random.standard_normal(len(freq[1:-1])) * 0.08)
            a_new = np.fft.irfft(freq, n)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                if score_of_a(a_new) < best_score + 0.01:
                    a = a_new
    
    return best_a, best_score


def coordinate_descent_v2(a0, time_limit, n_sweeps=150):
    """Coordinate descent with adaptive step sizes."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of_a(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            # Adaptive factors based on current value
            base_factors = [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]
            
            best_local = a[i]
            best_local_score = best_score
            
            for factor in base_factors:
                a_trial = a.copy()
                a_trial[i] *= factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = score_of_a(a_trial)
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor
            
            if best_local < a[i] - 1e-15 or best_local > a[i] + 1e-15:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True
        
        if not sweep_improved:
            break
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    # Frequency-domain interpolation for better quality
    N_old = old_n
    N_new = new_n
    # Zero-pad in frequency domain
    freq = np.fft.rfft(f_n)
    N_freq_old = len(freq)
    N_freq_new = N_new // 2 + 1
    freq_new = np.zeros(N_freq_new, dtype=complex)
    # Map old frequencies to new
    for i in range(min(N_freq_old, N_freq_new)):
        freq_new[i] = freq[i]
    f_interp = np.fft.irfft(freq_new, N_new)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_inits(parent, n, rng):
    """Generate diverse initializations."""
    candidates = []
    
    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            for scale in [0.02, 0.05, 0.1]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append((f'pert_{scale}', pert))
            # Frequency-domain perturbation
            freq = np.fft.rfft(p)
            phase = rng.standard_normal(len(freq)) * 0.3
            freq[1:-1] *= (1 + 1j * phase)
            a_flip = np.fft.irfft(freq, n)
            a_flip = np.maximum(a_flip, 0)
            if a_flip.sum() > 1e-10:
                a_flip = a_flip / a_flip.sum()
                candidates.append(('freq_pert', a_flip))
        else:
            interp = interpolate_seq(parent, n)
            candidates.append(('parent_interp', interp))
    
    # Structured initializations
    candidates.append(('uniform', np.ones(n) / n))
    
    # Chirp-inspired (quadratic frequency sweep)
    i = np.arange(n)
    chirp = 1.0 + 0.3 * np.cos(2 * np.pi * i**2 / (4 * n))
    chirp = np.maximum(chirp, 0.01)
    chirp = chirp / chirp.sum()
    candidates.append(('chirp', chirp))
    
    # Comb pattern
    for m in [2, 3]:
        comb = 1.0 + 0.4 * np.cos(2 * np.pi * m * i / n)
        comb = np.maximum(comb, 0.01)
        comb = comb / comb.sum()
        candidates.append((f'comb_{m}', comb))
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x**2 / 3)
    gau = np.maximum(gau, 1e-6)
    gau = gau / gau.sum()
    candidates.append(('gaussian', gau))
    
    # Bimodal
    mid = n // 2
    bi = np.zeros(n)
    bi[max(0, mid - n//5):mid + n//5] = 1.0
    bi[max(0, mid - n//3):mid + n//3] = 0.4
    bi = np.maximum(bi, 0.005)
    bi = bi / bi.sum()
    candidates.append(('bimodal', bi))
    
    # Exponential
    expo = np.exp(-2.0 * np.linspace(0, 1, n))
    expo = expo / expo.sum()
    candidates.append(('expo', expo))
    
    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.05
    rand = rand / rand.sum()
    candidates.append(('random', rand))
    
    # Symmetric random
    half_n = n // 2
    half = np.abs(rng.standard_normal(half_n)) + 0.05
    if n % 2 == 0:
        sym = np.concatenate([half, half[::-1]])
    else:
        sym = np.concatenate([half, [half.mean()], half[::-1]])
    sym = sym / sym.sum()
    candidates.append(('sym_random', sym))
    
    return candidates


def evaluate_track(a, label, best_seq, best_score, print_flag=True):
    """Evaluate and track best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if print_flag:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
