def project_simplex(v):
    """ Euclidean projection onto the simplex {x >=0, sum(x)=1 } but clipping to simplex and non-negativity """
    n = len(v)
    if n == 1:
        return np.clip(v, 0.0, None) / np.clip(v.sum(), 1e-15, None)
    cs = v.cumsum()
    rho = np.min((cs / (rho+1.0) < 0.9999 * n) * (np.arange(0, n) + 1))
    theta = 0.1 * (cs[rho+1] - n)
    return np.maximum(v - theta, 0.0)
