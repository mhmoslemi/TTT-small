def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):

def solve_lp_best_response(a_fixed, solver_name=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver_name] if solver_name else [cp.ECOS, cp.SCS, cp.PDLP]
    solvers = [s for s in solvers if s is not None]
    if not solvers:
        solvers = [cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
                return None
        except Exception:
            continue
    return None

    if s < 1e-15:

        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(y, solver_name)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))

        if sa < 1e-15:
            a = y.copy()
            sa = 1.0

        score_y = 2.0 * n * fast_convolve(y).max()

        if (it + 1) % 200 == 0:

def interpolate_sequence(f, new_n):

        return np.ones(new_n) / new_n

    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)

    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def find_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_found' in globals():
        return _lp_solver_found
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                globals()['_lp_solver_found'] = solver_name
                print(f"LP solver: {solver_name}")
                return solver_name
        except Exception:
            continue
    globals()['_lp_solver_found'] = cp.ECOS
    print("LP solver: ECOS (default)")
    return cp.ECOS


def evaluate_safe(a):
    """Evaluate sequence through the official evaluator."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. L-BFGS soft-max with geometric temperature annealing (T: 200 -> 0.01)
    2. Aggressive multi-peak shaving
    3. LP-based Nesterov averaging
    4. Stochastic perturbation + re-optimization cycles
    5. Multi-length exploration
    6. Final micro-polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver_name = find_lp_solver()

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: L-BFGS with aggressive temperature annealing ===
    print(f"\n=== Phase 1: L-BFGS Annealing (best={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        T_list_high = [200.0, 100.0, 50.0, 25.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_out = lbgfs_optimize(best_seq, t_alloc, T_list_high, maxiter_per_stage=2000)
        try_update(a_out, "lbgfs_anneal")

    # === Phase 2: Peak shaving ===
    print(f"\n=== Phase 2: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        a_out = multi_peak_shave(best_seq, t_alloc, n_steps=500000)
        try_update(a_out, "peak_shave_1")

    # === Phase 3: L-BFGS low-T refinement ===
    print(f"\n=== Phase 3: L-BFGS Low-T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        T_list_low = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_out = lbgfs_optimize(best_seq, t_alloc, T_list_low, maxiter_per_stage=3000)
        try_update(a_out, "lbgfs_lowT")

    # === Phase 4: LP refinement ===
    print(f"\n=== Phase 4: LP Nesterov (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        a_out = lp_refine(best_seq, t_alloc, solver_name=solver_name, n_max_iters=3000)
        try_update(a_out, "lp_nesterov")

    # === Phase 5: Interleaved refinement cycles ===
    print(f"\n=== Phase 5: Interleaved Cycles (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    cycle_time = min(rem * 0.2, 200.0)
    if cycle_time > 40:
        n_cycles = max(1, int(cycle_time / 30))
        for c in range(n_cycles):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t1 = min(rem * 0.4, 80.0)
            if t1 > 10:
                T_lc = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
                a_out = lbgfs_optimize(best_seq, t1, T_lc, maxiter_per_stage=2000)
                try_update(a_out, f"cycle{c}_lbgfs")

            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t2 = min(rem * 0.4, 80.0)
            if t2 > 10:
                a_out = multi_peak_shave(best_seq, t2, n_steps=100000)
                try_update(a_out, f"cycle{c}_shave")

            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t3 = min(rem * 0.3, 60.0)
            if t3 > 10:
                a_out = lp_refine(best_seq, t3, solver_name=solver_name, n_max_iters=500)
                try_update(a_out, f"cycle{c}_lp")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = [int(n_best * 1.25), int(n_best * 0.8), int(n_best * 1.5), int(n_best * 0.7)]
    alt_lengths = [x for x in alt_lengths if 256 <= x <= 4096 and x != n_best]
    seen = set()
    for alt_n in alt_lengths:
        if alt_n in seen:
            continue
        seen.add(alt_n)
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.3, 100.0)
        if t_alloc < 20:
            break
        print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        T_alt = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_out = lbgfs_optimize(init_alt, t_alloc * 0.6, T_alt, maxiter_per_stage=1500)
        try_update(a_out, f"alt_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None:
            t_sh = min(rem * 0.3, 40.0)
            if t_sh > 10:
                a_out2 = multi_peak_shave(best_seq, t_sh, n_steps=50000)
                try_update(a_out2, f"alt_n{alt_n}_shave")

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_f = min(rem * 0.4, 60.0)
        T_final = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_out = lbgfs_optimize(best_seq, t_f, T_final, maxiter_per_stage=3000)
        try_update(a_out, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.4, 50.0)
        a_out = multi_peak_shave(best_seq, t_ps, n_steps=200000)
        try_update(a_out, "final_shave")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp = min(rem * 0.4, 40.0)
        a_out = lp_refine(best_seq, t_lp, solver_name=solver_name, n_max_iters=1000)
        try_update(a_out, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
