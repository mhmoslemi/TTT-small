def backtrack_line_search(a, grad, c_max, n, max_iter=30):
    """Backtracking line search for multiplicative update.

    reduces the objective.

    """

    grad_proj = grad - np.dot(a, grad)
    
    # Initial step size

    current_obj = c_max

        log_update = np.clip(log_update, -10.0, 10.0)

        if new_max < current_obj - 1e-10:

        elif new_max < current_obj:

        else:

                return a, eta
    
    return a, eta

            eta = min(eta * 2.0, 0.05)
            stuck = 0
        elif progress_streak > 5:
            T = max(T * 0.95, T_end)
        
        # Choose gradient type based on temperature
        if T > 0.02:
            grad, _ = smooth_gradient_fft(a, T)
        else:
            grad, _ = multi_peak_subgradient(a, top_k=top_k, alpha=alpha)
        
        # Adaptive step size: increase when making consistent progress
        if progress_streak > 10:
            eta = min(eta * 1.05, 0.05)
        elif stuck > 30:
            eta = max(eta * 0.7, 0.0001)
        
        # Multiplicative update
        a_new = multiplicative_update(a, grad, eta)
        
        # Check improvement
        c_new = fast_convolve(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            a_half = multiplicative_update(a, grad, eta * 0.3)
            c_half = fast_convolve(a_half)
            if c_half.max() < cur_max:
                a = a_half
                eta *= 0.5
            else:
                eta *= 0.5
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): T={T:.4f} eta={eta:.5f} "
                  f"score={cur_score:.7f} best={best_score:.7f}")
    
    return best_a, best_score


def pairwise_local_search(a_init, time_limit, verbose=True, label=""):
    """Pairwise exchange local search targeting peak-contributing pairs."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    deltas = [0.01, 0.02, 0.05, 0.005]
    no_improve = 0
    max_no_improve = 200
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        
        # Identify contributing pairs for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx_range = np.arange(js, je + 1)
        
        # Sort contributions in descending order
        contribs = np.array([a[i] * a[peak_k - i] for i in idx_range])
        top_pairs = idx_range[np.argsort(contribs)[::-1][:20]]
        
        # Also pick some random positions
        random_positions = rng.choice(n, size=min(20, n), replace=False)
        
        # Try moves: reduce a peak-contributor, increase a low contributor
        candidates = np.unique(np.concatenate([top_pairs, random_positions]))
        
        improved_this_round = False
        for i in candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-10:
                continue
            
            for delta in deltas:
                # Move mass from i to a random other position
                # Pick a target: a position with low contribution to all peaks
                a_trial = a.copy()
                a_trial[i] -= a[i] * delta
                if a_trial[i] < 0:
                    continue
                # Redistribute to others proportionally
                others_sum = 1.0 - a_trial[i]
                if others_sum < 1e-10:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] = a[mask] * (others_sum / (1.0 - a[i]))
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_this_round = True
                    break
        
        if not improved_this_round:
            no_improve += 1
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    {label} PL ({elapsed:.0f}s): best={best_score:.7f} no_improve={no_improve}")
    
    return best_a, best_score


def spectral_perturbation(a, strength=0.1):
    """Apply a random perturbation in the frequency domain to escape local minima."""
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = fft(a, fft_size)
    
    # Perturb the magnitude (not phase, to maintain real-valued a)
    mag = np.abs(A)
    phase = np.angle(A)
    
    rng = np.random.default_rng(np.random.randint(0, 2**31))
    noise = rng.standard_normal(n) * strength
    # Keep DC component fixed
    # For real signal, A[n-k] = conj(A[k])
    half = fft_size // 2
    new_mag = mag.copy()
    new_mag[1:half] = np.maximum(mag[1:half] + noise[:half], 0)
    new_mag[half+1:] = new_mag[1:half][::-1]
    if fft_size % 2 == 0:
        new_mag[half] = mag[half] + noise[half] * 0.5
        new_mag[half] = max(0, new_mag[half])
    
    new_A = new_mag * np.exp(1j * phase)
    # Enforce Hermitian symmetry
    new_A[half+1:] = np.conj(new_A[1:half][::-1])
    if fft_size % 2 == 0:
        new_A[half] = np.real(new_A[half])
    new_A[0] = np.real(new_A[0])
    
    a_new = np.real(ifft(new_A, fft_size)[:n])
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def coordinate_fine_tune(a_init, time_limit, verbose=True, label=""):
    """Fine coordinate descent: adjust individual elements with optimal redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    factors = [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]
    no_improve_rounds = 0
    
    while time.monotonic() - t0 < time_limit and no_improve_rounds < 5:
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-10:
                continue
            for f in factors:
                a_trial = a.copy()
                old = a_trial[i]
                new = old * f
                a_trial[i] = new
                # Renormalize
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            no_improve_rounds += 1
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    {label} CF ({elapsed:.0f}s): best={best_score:.7f} rounds_no_improve={no_improve_rounds}")
    
    return best_a, best_score


def lbfgs_refine(a_init, time_limit, T=None, maxiter=5000):
    """L-BFGS refinement in log-space with smooth objective."""
    n = len(a_init)
    if T is None:
        T = 0.02
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        c = fast_convolve(a_tmp)
        scaled = c / T
        m = scaled.max()
        return T * (m + np.log(np.exp(scaled - m).sum()))
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        grad_a, _ = smooth_gradient_fft(a_tmp, T)
        # Transform gradient from a-space to z-space
        mean_grad = np.dot(a_tmp, grad_a)
        return a_tmp * (grad_a - mean_grad)
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxfun': maxiter * 20, 'disp': False})
        z_opt = res.x
        a_out = np.exp(z_opt - z_opt.max())
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a_init.copy()
    
    return a_out


def generate_initializations(n, parent=None):
    """Generate diverse initializations."""
    inits = []
    labels = []
    
    if parent is not None and len(parent) == n:
        inits.append(parent / parent.sum())
        labels.append("parent")
    
    # Uniform
    inits.append(np.ones(n) / n)
    labels.append("uniform")
    
    # Raised cosine (Hanning)
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())
    labels.append("hann")
    
    # Gaussian
    center = n / 2
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / (n / 4)) ** 2)
    inits.append(gauss / gauss.sum())
    labels.append("gauss")
    
    #Triangular
    tri = 1 - np.abs(np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)
    tri = np.maximum(tri, 1e-10)
    inits.append(tri / tri.sum())
    labels.append("triangular")
    
    # Random
    rng = np.random.default_rng(42)
    rand = rng.random(n)
    inits.append(rand / rand.sum())
    labels.append("random")
    
    # Parabolic (concentrated in middle)
    para = 1 - ((np.arange(n) - (n-1)/2) / ((n-1)/2)) ** 2
    para = np.maximum(para, 1e-10)
    inits.append(para / para.sum())
    labels.append("parabolic")
    
    return inits, labels


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):

    Multi-phase optimization with diverse strategies:
    1. Multiple initializations + short smoothed GD to find good basins
    2. Extended smoothed GD with adaptive step size and multi-peak subgradient
    3. Spectral perturbation + re-optimization for escaping local minima
    4. L-BFGS refinement at multiple temperature levels
    5. Pairwise local search for fine-tuning
    6. Coordinate fine-tuning
    7. Alternative lengths exploration

    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    parent = None

            parent = p / p.sum()

        pass

    print(f"\n=== Phase 1: Diverse Initialization (n={n_best}) ===")

        for T_s, T_e, eta_i, tk, al in [(0.08, 0.003, 0.004, 3, 1.0),

            a_opt, _ = optimize_smoothed_gradient(
                best_seq, t_sub, T_start=T_s, T_end=T_e,
                eta_init=eta_i, top_k=tk, alpha=al,
                verbose=True, label=f"SGD_T{T_s}_pk{tk}"
            )
            best_seq, best_score = check_and_update(a_opt, f"SGD_T{T_s}_pk{tk}", best_seq, best_score)
    
    # === Phase 3: Spectral perturbation + re-optimization ===
    print(f"\n=== Phase 3: Spectral Perturbation (best={best_score:.7f}) ===")

    rem = deadline - time.monotonic()
    if rem > 40:
        t_sub = min(rem * 0.3, 120.0)
        a_opt, _ = optimize_smoothed_gradient(
            best_seq, t_sub, T_start=0.02, T_end=0.002,

        # Quick L-BFGS
        rem = deadline - time.monotonic()
        if rem > 15:
            a_lb = lbfgs_refine(best_seq, min(rem * 0.5, 40.0), T=0.01, maxiter=3000)
            best_seq, best_score = check_and_update(a_lb, "lbfgs_r2", best_seq, best_score)
    
    # === Phase 8: Alternative lengths ===
    print(f"\n=== Phase 8: Alternative Lengths (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        for factor in [1.5, 2.0, 0.75]:

            alt_n = int(n_cur * factor)

            # Interpolate
            init_alt = interpolate_sequence(best_seq, alt_n)

            if t_sub < 15:

            )

            rem = deadline - time.monotonic()
    
    # === Phase 9: Final deep refinement ===
    print(f"\n=== Phase 9: Final Deep Refinement (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Very low temperature L-BFGS
        if rem > 15:
            a_fin = lbfgs_refine(best_seq, min(rem * 0.4, 30.0), T=0.005, maxiter=5000)
            best_seq, best_score = check_and_update(a_fin, "final_lbfgs", best_seq, best_score)

        rem = deadline - time.monotonic()

            best_seq, best_score = check_and_update(a_cf, "final_cf", best_seq, best_score)

            )
            a_fin, _ = a_fin
            best_seq, best_score = check_and_update(a_fin, "final_gd", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
