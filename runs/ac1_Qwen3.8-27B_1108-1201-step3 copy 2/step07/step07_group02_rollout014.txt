def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):

    return T

def solve_lp_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b_value, M_value, dual_y) where dual_y are the dual variables.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    y_var = cp.Variable(2 * n - 1, nonneg=True)  # dual variables
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    
    # Extract dual variables
    dual_y = None
    try:
        # The dual values are associated with the constraint T @ b_var <= M_var
        # In cvxpy, we can access them via the constraint's dual value
        dual_y = np.abs(np.asarray(constraints[0].dual_value, dtype=np.float64))
        dual_y = np.maximum(dual_y, 0)
        sy = dual_y.sum()
        if sy > 1e-10:
            dual_y = dual_y / sy
        else:
            dual_y = None
    except Exception:
        dual_y = None
    
    return b_val, float(M_var.value), dual_y


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_subgradient(a, active_indices):
    """Compute subgradient of max_k (a*a)[k] at point a."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active_indices)
    return g


def compute_weighted_subgradient(a, weights):
    """Compute weighted subgradient using LP dual weights."""
    n = len(a)
    g = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += weights[k] * a[k - idx]
    total_w = weights.sum()
    if total_w > 1e-15:
        g /= total_w
    return g


def lbgfs_soft_max(a0, time_limit, T_start=100.0, T_end=5.0, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing and adaptive temperature."""
    from scipy.optimize import minimize
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 5
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': iters_per_stage * 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=10000):
    """
    Extended Nesterov dual averaging LP refinement with adaptive step sizes
    and periodic evaluation.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, dual_y = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)

        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        if (it + 1) % 50 == 0:
            score_a = 2.0 * n * fast_convolve(a).max()
            score_y = 2.0 * n * fast_convolve(y).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

    # Final evaluation
    score_a = 2.0 * n * fast_convolve(a).max()
    score_y = 2.0 * n * fast_convolve(y).max()
    if score_a < best_score:
        best_score = score_a
        best_a = a.copy()
    if score_y < best_score:
        best_score = score_y
        best_a = y.copy()

    return best_a, best_score


def subgradient_refine(a0, time_limit, solver=None, n_steps=5000):
    """
    Refine using exact subgradient from LP dual variables.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    z = z_from_a(a)
    lr = 0.1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Get LP dual for weighted subgradient
        _, _, dual_y = solve_lp_best_response(a, solver)
        
        if dual_y is not None:
            g = compute_weighted_subgradient(a, dual_y)
        else:
            # Fall back to uniform active set
            threshold = max_b - 0.01 * max_b
            active = np.where(b >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(b)])
            g = compute_subgradient(a, active)

        # Mirror descent step
        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
        else:
            lr *= 0.9

        lr = max(lr, 1e-8)

        if step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubgradRefine {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def multi_start_lp(a0, time_limit, solver=None, n_starts=3, n_iters=3000):
    """
    Multi-start LP Nesterov with perturbations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for start in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        t_alloc = remaining / (n_starts - start)

        if start == 0:
            a_start = a.copy()
        else:
            # Perturb the starting point
            noise = rng.random(n) * 0.1
            a_start = a + noise
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_res, score_res = nesterov_lp_refine(a_start, t_alloc, solver=solver,
                                              n_max_iters=n_iters)
        if score_res < best_score:
            best_score = score_res
            best_a = a_res.copy()
            print(f"    MultiStart {start}: improved to {best_score:.6f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=5000):
    """Aggressive local peak shaving."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_mid = b.copy()
        b_mid[0] = -np.inf
        b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.93, 0.88, 0.8]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            if len(top_contrib) >= 2:
                for ci in top_contrib[:3]:
                    i = idx[ci]
                    j = peak_k - i
                    if i == j or a[j] < 1e-14:
                        continue
                    for f1 in [0.95, 0.9]:
                        for f2 in [0.95, 0.9]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
            if not improved:
                top3 = top_contrib[:3]
                a_trial = a.copy()
                for ci in top3:
                    i = idx[ci]
                    a_trial[i] *= 0.95
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                if not improved:
                    break

    return best_a, best_score
