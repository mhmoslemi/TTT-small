import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve_auto(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve_auto(a)
    return 2.0 * n * b.max()

def cross_convolve(f, g):
    """Linear cross-convolution of f and g (both length n) via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def lp_find_direction(f, M, solver='ECOS'):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k. Returns g."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid i
    constraints = []
    for k in range(2*n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        # (f*g)[k] = sum_{i=i_lo}^{i_hi} f[i] * g[k-i]
        terms = []
        for i in range(i_lo, i_hi + 1):
            if f[i] > 1e-15:
                terms.append(f[i] * g[k - i])
        if terms:
            constraints.append(cp.sum(terms) <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve(solver='SCS')
        except:
            return None
    
    if g.value is not None:
        g_val = np.maximum(g.value, 0)
        s = g_val.sum()
        if s > 1e-10:

    return None

        best_trial_score = current_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()

        if best_t > 0:

            f = np.maximum(f, 0)

                M = new_score / (2.0 * len(f))  # Update M to new peak

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: iteratively solve LP to find direction that keeps cross-convolution
bounded, move toward that direction. Combined with local polishing and multi-scale
search.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp

# Helper functions (all top-level, no closures)

def fast_conv_auto(a):
    n = len(a)
    size = 2*n - 1
    fs = 1 << (size-1).bit_length()
    A = rfft(a, fs)
    return irfft(A*A, fs)[:size]

def score_norm(a):
    """Score for sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_conv_auto(a).max()

def build_sparse_conv(f):
    """Build sparse (2n-1) x n matrix A where A@g = f*g."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_lo = max(0, k-n+1)
        j_hi = min(n-1, k)
        for j in range(j_lo, j_hi+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp_direction(f, M, timeout=30):
    """Solve LP: max sum(g) s.t. g>=0, A@g <= M*1. Returns normalized g or None."""
    n = len(f)
    A = build_sparse_conv(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, timeout=timeout)
    except:
        try:
            prob.solve(solver=cp.SCS, timeout=timeout)
        except:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0)
    s = g_val.sum()
    if s < 1e-10:
        return None
    return g_val / s

def lp_iterate(f, time_budget, verbose=True, label=""):
    """Iterative LP-based optimization."""
    f = np.maximum(f.copy(), 0).astype(np.float64)

    no_imp = 0
    it = 0

        it += 1
        g = solve_lp_direction(f, M, timeout=15)
        if g is None:
            M *= 1.005
            continue
        
        cur_sc = score_norm(f)
        best_trial = cur_sc
        best_t = 0
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:

            trial = np.maximum(trial, 0)

                best_t = t

def interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    out = np.interp(new_x, old_x, f_norm)
    out = np.maximum(out, 0)
    s = out.sum()
    if s > 1e-10:
        out = out / s
    return out
