"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-n exploration with diverse initializations
2. Aggressive L-BFGS with soft-max annealing to very low temperature
3. Dual LP alternating (exploiting a-b symmetry)
4. Peak redistribution heuristic
5. Final micro-polish
"""
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    global _lp_solver_cached
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

    return 2.0 * len(a) * b.max() / (s * s)

def lbgfs_anneal(a0, time_limit, T_stages=None, maxiter_total=10000):
    """L-BFGS with multi-stage temperature annealing."""
    n = len(a0)
    if T_stages is None:
        T_stages = [100.0, 30.0, 10.0, 3.0, 1.0, 0.3, 0.1]
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    n_stages = len(T_stages)
    iters_per_stage = maxiter_total // n_stages
    z = z0.copy()
    
    for stage_idx, T_val in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z, T=T_val):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T) * log_sum_exp
        
        def jac(z, T=T_val):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 30, 'disp': False}
            )
            z = res.x
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS stage T={T_val}: done ({elapsed:.0f}s)")
        except Exception as e:
            print(f"    L-BFGS stage T={T_val}: error {e}")
    
    a_out = softmax_from_z(z)
    return a_out

def peak_redistribute(a0, time_limit, n_steps=5000):
    """Move mass from peak contributors to safe positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the peak (excluding endpoints which are just a[0]^2 and a[n-1]^2)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        
        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Top contributor
        top_i = idx[np.argmax(contribs)]
        top_j = peak_k - top_i
        
        if a[top_i] < 1e-14:
            break
        
        # Try to move mass from top_i to a "safe" position
        # Safe position: one that doesn't contribute to the peak
        # and whose current value is low
        safe_mask = np.ones(n, dtype=bool)
        safe_mask[top_i] = False
        # Positions that would contribute to the peak if increased:
        for k_check in range(max(0, peak_k - 10), min(2*n-1, peak_k + 11)):
            if k_check == peak_k:
                continue
            j_check = k_check - top_i  # would j_check contribute with top_i?
            # Actually, we want positions that don't create NEW peaks
            # Simple heuristic: positions far from top_i in the convolution sense
        # Simpler: just try reducing top_i and redistributing uniformly to low positions
        for factor in [0.99, 0.98, 0.95, 0.9, 0.8]:
            new_val = a[top_i] * factor
            diff = a[top_i] - new_val
            if diff < 1e-16:
                continue
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            # Redistribute to positions with lowest current value
            others = a[mask]
            # Give more to positions that are "far" from peak contributors
            a_trial = a.copy()
            a_trial[top_i] = new_val
            # Redistribute proportional to current values (keeps distribution similar)
            a_trial[mask] = others + diff * others / others.sum()
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = compute_score(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                break
        else:
            # Try reducing both top_i and top_j
            if top_i != top_j and a[top_j] > 1e-14:
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = a.copy()
                        a_trial[top_i] *= f1
                        a_trial[top_j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        mask = np.ones(n, dtype=bool)
                        mask[top_i] = False
                        mask[top_j] = False
                        others = a_trial[mask]
                        excess = a[top_i] - a_trial[top_i] + a[top_j] - a_trial[top_j]
                        if others.sum() > 1e-15:
                            a_trial[mask] = others + excess * others / others.sum()
                        a_trial = a_trial / a_trial.sum()
                        score = compute_score(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            break
                    else:
                        continue
                    break
            else:
                break
    
    return best_a, best_score

def nesterov_dual_lp(a0, time_limit, solver=None, n_iters=3000):
    """Nesterov-averaged dual LP with symmetry."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    y = a.copy()  # Nesterov virtual iterate
    avg = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        a_new, M = solve_lp_best_response(y, solver)
        if a_new is None:
            break
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        avg = ((it + 1) * avg + a_new) / (it + 2)
        avg = np.maximum(avg, 0)
        avg = avg / avg.sum()
        
        sc_y = compute_score(y)
        sc_avg = compute_score(avg)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg.copy()
        
        if (it + 1) % 200 == 0:
            print(f"    NesterovDual iter={it+1}: score={best_score:.6f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    return exp_z / s


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_score(a):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max() / (s * s)


def evaluate_safe(a):
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _lbgfs_obj(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def _lbgfs_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_anneal(a0, time_limit, T_stages=None, maxiter_total=10000):
    n = len(a0)
    if T_stages is None:
        T_stages = [100.0, 30.0, 10.0, 3.0, 1.0, 0.3, 0.1]
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    n_stages = len(T_stages)
    iters_per_stage = max(maxiter_total // n_stages, 100)
    z = z0.copy()

    for stage_idx, T_val in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        func = partial(_lbgfs_obj, T=T_val, n=n)
        jac = partial(_lbgfs_grad, T=T_val, n=n)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 30, 'disp': False}
            )
            z = res.x
            elapsed = time.monotonic() - t0
            a_chk = softmax_from_z(z)
            sc = compute_score(a_chk)
            print(f"    L-BFGS T={T_val}: score={sc:.6f} ({elapsed:.0f}s)")
        except Exception as e:
            print(f"    L-BFGS T={T_val}: error {e}")
    a_out = softmax_from_z(z)
    return a_out


def lbgfs_perturbed(a0, time_limit, T_stages=None, n_restarts=3, rng=None):
    """L-BFGS with perturbation restarts to escape local minima."""
    if T_stages is None:
        T_stages = [5.0, 1.0, 0.3, 0.1]
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    time_per_restart = time_limit / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        if r == 0:
            start = a0.copy()
        else:
            # Perturb: add small random noise in log-space
            z = z_from_a(best_a)
            scale = 0.05 * (r + 1)
            z = z + scale * rng.standard_normal(n)
            start = softmax_from_z(z)
        remaining = time_limit - (time.monotonic() - t0)
        alloc = min(time_per_restart - r * 0, remaining)
        if alloc < 10:
            break
        a_out = lbgfs_anneal(start, alloc, T_stages=T_stages, maxiter_total=4000)
        sc = compute_score(a_out)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_out.copy()
            print(f"    PerturbRestart r={r}: NEW score={sc:.6f}")
        else:
            print(f"    PerturbRestart r={r}: score={sc:.6f} (no improvement)")
    return best_a, best_score


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_dual_lp(a0, time_limit, solver=None, n_iters=3000):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    y = a.copy()
    avg = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        a_new, M = solve_lp_best_response(y, solver)
        if a_new is None:
            break
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a_new.copy()
            sy = 1.0
        y = y / sy
        avg = ((it + 1) * avg + a_new) / (it + 2)
        avg = np.maximum(avg, 0)
        avg = avg / avg.sum()

        sc_y = compute_score(y)
        sc_avg = compute_score(avg)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg.copy()
        if (it + 1) % 300 == 0:
            print(f"    NesterovDual iter={it+1}: score={best_score:.6f}")
    return best_a, best_score


def peak_redistribute(a0, time_limit, n_steps=5000, rng=None):
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 200:
            break
        b = fast_convolve(a)
        b_search = b.copy()
        if n > 2:
            b_search[0] = -np.inf
            b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx_sorted = idx[np.argsort(contribs)[::-1][:8]]

        improved = False
        for top_i in top_idx_sorted:
            if a[top_i] < 1e-14:
                continue
            top_j = peak_k - top_i
            # Try single-coordinate reduction
            for factor in [0.99, 0.97, 0.93, 0.85]:
                new_val = a[top_i] * factor
                diff = a[top_i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[top_i] = False
                others = a[mask]
                a_trial = a.copy()
                a_trial[top_i] = new_val
                if others.sum() > 1e-15:
                    a_trial[mask] = others + diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Try dual reduction
            if len(top_idx_sorted) >= 2:
                i1 = top_idx_sorted[0]
                i2 = top_idx_sorted[1]
                j1 = peak_k - i1
                j2 = peak_k - i2
                candidates = set()
                if i1 != j1 and a[j1] > 1e-14:
                    candidates.add((i1, j1))
                if i2 != j2 and a[j2] > 1e-14:
                    candidates.add((i2, j2))
                for (ci, cj) in list(candidates)[:4]:
                    for f1 in [0.95, 0.9]:
                        for f2 in [0.95, 0.9]:
                            a_trial = a.copy()
                            a_trial[ci] *= f1
                            a_trial[cj] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            mask = np.ones(n, dtype=bool)
                            mask[ci] = False
                            mask[cj] = False
                            others = a_trial[mask]
                            excess = a[ci] - a_trial[ci] + a[cj] - a_trial[cj]
                            if others.sum() > 1e-15:
                                a_trial[mask] = others + excess * others / others.sum()
                            a_trial = a_trial / a_trial.sum()
                            score = compute_score(a_trial)
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                no_improve = 0
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
        if not improved:
            no_improve += 1
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search:
    1. Load parent, explore alternative n
    2. Medium-T L-BFGS annealing
    3. Aggressive low-T L-BFGS with perturbation restarts
    4. Nesterov dual LP (symmetry)
    5. Peak redistribution
    6. Ultra-low-T final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={ps:.6f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        best_seq = np.ones(768) / 768
        best_score = compute_score(best_seq)
        print(f"Default init: score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Explore alternative n (250s) ===
    print(f"\n=== Phase 1: Alt-n exploration (best={best_score:.6f}) ===")
    n_cur = len(best_seq)
    alt_ns = [n_cur + 256, n_cur + 512, n_cur + 768, n_cur + 1024]
    alt_ns = [x for x in alt_ns if 512 <= x <= 4000 and x != n_cur]
    # Also try some different structured n
    for tn in [1024, 1280, 1536, 2048, 2560]:
        if tn not in alt_ns and tn != n_cur and 512 <= tn <= 4000:
            alt_ns.append(tn)
    alt_ns = sorted(set(alt_ns))[:4]

    for alt_n in alt_ns:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.25, 70.0)
        if t_alloc < 20:
            break
        print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt = lbgfs_anneal(init_alt, t_alloc, T_stages=[50.0, 15.0, 5.0, 1.5], maxiter_total=3000)
        try_update(a_alt, f"alt_n{alt_n}")

    # === Phase 2: Medium-T L-BFGS on best (300s) ===
    print(f"\n=== Phase 2: Medium-T L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 300.0)
    if t_alloc > 30:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_m = lbgfs_anneal(best_seq, t_alloc, T_stages=[30.0, 10.0, 3.0, 1.0], maxiter_total=6000)
        try_update(a_m, "med_lbgfs")

    # === Phase 3: Low-T with perturbation restarts (500s) ===
    print(f"\n=== Phase 3: Low-T Perturbed L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_p, _ = lbgfs_perturbed(best_seq, t_alloc,
                                  T_stages=[3.0, 1.0, 0.3, 0.1],
                                  n_restarts=3, rng=rng)
        try_update(a_p, "perturbed_lbgfs")

    # === Phase 4: Nesterov Dual LP (250s) ===
    print(f"\n=== Phase 4: Nesterov Dual LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.14, 250.0)
    if t_alloc > 20:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_nd, _ = nesterov_dual_lp(best_seq, t_alloc, solver=solver)
        try_update(a_nd, "nesterov_dual")

    # === Phase 5: Peak redistribution (150s) ===
    print(f"\n=== Phase 5: Peak Redistribution (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 150.0)
    if t_alloc > 10:
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_pr, _ = peak_redistribute(best_seq, t_alloc, n_steps=5000, rng=rng)
        try_update(a_pr, "peak_redist")

    # === Phase 6: Ultra-low-T final polish (remaining) ===
    print(f"\n=== Phase 6: Ultra-Low-T Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_final = min(rem * 0.65, 200.0)
        if t_final > 15:
            print(f"  Final L-BFGS, budget={t_final:.0f}s")
            a_f = lbgfs_anneal(best_seq, t_final,
                                T_stages=[1.0, 0.3, 0.1, 0.05, 0.02],
                                maxiter_total=6000)
            try_update(a_f, "ultra_lowT")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_pr = min(rem * 0.6, 100.0)
        if t_pr > 10:
            a_pr2, _ = peak_redistribute(best_seq, t_pr, n_steps=3000, rng=rng)
            try_update(a_pr2, "final_pr")

    # One more L-BFGS if time
    rem = deadline - time.monotonic()
    if rem > 15:
        a_f2 = lbgfs_anneal(best_seq, rem * 0.8,
                             T_stages=[0.5, 0.1, 0.05],
                             maxiter_total=3000)
        try_update(a_f2, "last_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            print(f"Evaluated: {fs:.6f}")

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
