import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)

def subgradient_at(y, topk=1, T_peaks=50.0):
    """Compute weighted subgradient of max autocorrelation."""
    n = len(y)
    c = fast_autocorr(y)
    if topk == 1:
        k_star = int(c.argmax())
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            g[j] = 2.0 * y[k_star - j]
        if k_star % 2 == 0:
            jc = k_star // 2
            if 0 <= jc < n:
                g[jc] = y[jc]
        return g
    else:
        k_use = min(topk, len(c))
        top_idx = np.argsort(c)[::-1][:k_use]
        top_vals = c[top_idx]
        # Softmax weights over top peaks
        Tc = T_peaks * top_vals
        Tc -= Tc.max()
        w = np.exp(Tc)
        w = w / w.sum()
        g = np.zeros(n)
        for idx in range(k_use):
            k = top_idx[idx]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            wm = w[idx]
            for j in range(js, je + 1):
                g[j] += wm * 2.0 * y[k - j]
            if k % 2 == 0:
                jc = k // 2
                if 0 <= jc < n:
                    g[jc] -= wm * y[jc]  # correct the double count at center
        return g

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def multiplicative_weights_game(a0, time_limit, topk=10, T_peaks=100.0, 
                                eta_schedule='nesterov', seed=42):
    """
    Multiplicative weights for the "autocorrelation game".
    Uses running average for the output.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    # Running average
    avg = np.zeros(n)
    count = 0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    t = 0
    while time.monotonic() - t0 < time_limit - 1:
        t += 1
        
        # Compute distribution over peaks (approximate LP best response)
        c = fast_autocorr(a)
        k_use = min(topk, len(c))
        top_idx = np.argsort(c)[::-1][:k_use]
        top_vals = c[top_idx]
        
        # Softmax weights
        Tc = T_peaks * top_vals
        Tc -= Tc.max()
        w_peaks = np.exp(Tc)
        w_peaks /= w_peaks.sum()
        
        # Compute "best response" direction: b_j = sum_k w_k * a_{k-j}
        b = np.zeros(n)
        for idx in range(k_use):
            k = top_idx[idx]
            wk = w_peaks[idx]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                b[j] += wk * a[k - j]
        b_sum = b.sum()
        if b_sum > 1e-15:
            b = b / b_sum
        else:
            b = a.copy()
        
        # Update: multiplicative weights
        # a_new_j ∝ a_j * (b_j / a_j)^eta  ... this is the standard MW update
        # Or equivalently: a_new = a * (b/a)^eta normalized
        # Which is: a_new_j ∝ a_j^{1-eta} * b_j^eta
        
        if eta_schedule == 'nesterov':
            eta = 2.0 / (t + 2.0)
        else:
            eta = 1.0 / np.sqrt(t)
        
        # a_new ∝ a * (b/a)^eta = a^{1-eta} * b^eta
        a_safe = np.maximum(a, 1e-30)
        b_safe = np.maximum(b, 1e-30)
        log_a_new = (1 - eta) * np.log(a_safe) + eta * np.log(b_safe)
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        a = a_new
        
        # Update running average
        avg += a
        count += 1
        
        # Check improvement periodically
        if t % 10 == 0:
            score_curr = 2.0 * n * fast_autocorr(a).max()
            if score_curr < best_score:
                best_score = score_curr
                best_a = a.copy()
            
            avg_curr = avg / count
            score_avg = 2.0 * n * fast_autocorr(avg_curr).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_curr.copy()
    
    return best_a, best_score
