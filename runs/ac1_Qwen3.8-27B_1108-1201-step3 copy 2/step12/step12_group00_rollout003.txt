    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = b_var.value
    if b_val is None:
        return None
    b_val = np.maximum(np.asarray(b_val, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, m_decay=0.999):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    M_current = fast_autocorr(y).max()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent_multi_peak(a, time_limit, top_k=10, verbose=True):
    """Aggressive coordinate descent targeting top-K autocorrelation peaks."""
    n = len(a)
    rng = np.random.default_rng(12345)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, fast_autocorr(a).max()
    norm = np.sqrt(2.0 * n)
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get top K peaks
        peak_indices = np.argsort(b)[::-1][:top_k]
        current_max = b.max()
        
        if current_max < best_score:
            best_score = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial * norm / a_trial.sum()
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall += 100
            if stall > 1000:
                break
        
        # For each top peak, try to reduce it
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Source: the pair with max contribution
            src_offset = np.argmax(contrib)
            src = idx[src_offset]
            dst_partner = k - src
            
            # Build "danger" map: which positions contribute to top peaks
            danger = np.zeros(n)
            for k2 in peak_indices:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                danger[idx2] += b[k2]
                danger[k2 - idx2] += b[k2]
            
            # Target: position with lowest danger and some mass to spare
            safe_mask = danger == 0
            if safe_mask.any():
                candidates = np.where(safe_mask)[0]
            else:
                candidates = np.where(danger < danger.min() * 2)[0]
            
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Pick the candidate with lowest current value among safe ones
            dst = candidates[np.argmin(a[candidates] + danger[candidates] * 1e-6)]
            
            if dst == src:
                continue
            
            # Try to move mass from src to dst
            # Use line search
            best_frac = 0
            best_trial_score = fast_autocorr(a).max()
            
            for frac in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_trial_score:
                    best_trial_score = score
                    best_frac = frac
            
            if best_frac > 0:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = a * norm / a.sum()
                if best_trial_score < best_score:
                    best_score = best_trial_score
                    best_a = a.copy()
                    stall = 0
        
        iteration += 1
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iteration} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
    
    return best_a, best_score


def lse_gradient_descent(a, time_limit, verbose=True):
    """Gradient descent on log-sum-exp surrogate for the max autocorrelation."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Anneal temperature
    T_start = 100.0
    T_end = 1.0
    total_expected = 5000
    iter_count = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / total_expected)
        T = T_start * (T_end / T_start) ** progress
        step_size = 1.0 / (T * np.sqrt(it + 1) * np.log(n + 1))
        
        b = fast_autocorr(a)
        b_shifted = b - b.max()
        w = np.exp(b_shifted / T)
        w /= w.sum()
        
        # Gradient of weighted sum of autocorrelations
        # w[k] * (a*a)[k] = w[k] * sum_i a[i] a[k-i]
        # d/d a[j] = w[k] * (a[k-j] + a[j-k]) for valid indices
        # Total gradient: sum_k w[k] * (a[k-j] + a[j-k])
        # = sum_k w[k] a[k-j] + sum_k w[k] a[j-k]
        # First term: (w * a)[j] (cross-correlation of w with a)
        # Second term: (w * a_rev)[j] where a_rev is reversed
        
        # Use FFT for efficiency
        fft_size = 1 << (3 * n).bit_length()
        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)
        
        # Cross-correlation w * a = sum_k w[k] a[k-j]
        # This is convolution of w with reversed a
        A_rev = np.fft.rfft(a[::-1], fft_size)
        cross1 = np.fft.irfft(W * A_rev, fft_size)[:2*n-1]
        
        # w * a_rev[j] = sum_k w[k] a[j-k] = sum_k w[k] a[k-j] (by symmetry of convolution)
        # Actually w * a[k-j] and w * a[j-k] are the same (reversal of convolution)
        cross2 = np.fft.irfft(W * A, fft_size)[n-1:n-1+n]  # This gives (w*a)[j]
        
        # Actually let me be more careful
        # grad[j] = sum_k w[k] * (a[k-j] if 0<=k-j<n else 0 + a[j-k] if 0<=j-k<n else 0)
        # = sum_k w[k] a[k-j] + sum_k w[k] a[j-k]
        # First sum: let m = k-j, = sum_m w[m+j] a[m] for valid m = (a conv w_rev)[j]... 
        # This is getting complicated. Let me just do it directly for small n or use a simpler approach.
        
        # Simpler: grad[j] = 2 * sum_k w[k] * a[|k-j|] (approximately, for the symmetric part)
        # Actually for the full convolution, grad[j] = sum_k w[k] * (a[k-j] + a[j-k])
        # where a[x] = 0 if x out of range.
        
        # Let's compute it as: grad = 2 * ifft(fft(w) * fft(a)) appropriately
        # (w * a) in the convolution sense gives us one half
        # The full gradient is the cross-correlation
        
        # Let me just use the direct formula with FFT:
        # grad[j] = sum_k w[k] a[k-j] + sum_k w[k] a[j-k]
        # = (w ⋆ a)[j] + (w ⋆ a)[j] where ⋆ is cross-correlation
        # = 2 * (w ⋆ a)[j]
        # where (w ⋆ a)[j] = sum_k w[k] conj(a[k-j]) = sum_k w[k] a[k-j] (real case)
        
        # Cross-correlation via FFT: (w ⋆ a) = ifft(conj(fft(w)) * fft(a))
        # But we need to be careful about the definition.
        
        # Let's just compute it directly for correctness:
        # For n up to ~4000, O(n^2) might be too slow. Let's use FFT.
        
        # (w * a)[m] = sum_k w[k] a[m-k] (convolution)
        # (w ⋆ a)[j] = sum_k w[k] a[k-j] = (w * a)[j] with appropriate indexing
        # Actually (w ⋆ a)[j] = (w * a_rev)[-j + len - 1] ... this is getting confusing.
        
        # Let me just use the property that if we define the full convolution c = conv(w, a)
        # then c[m] = sum_k w[k] a[m-k]
        # And the cross-correlation is r[j] = sum_k w[k] a[k-j] = c[-j] (if we extended w and a)
        # In the finite case, r[j] = c[n_w + n_a - 1 - j] where n_w = len(w), n_a = len(a)
        
        # w has length 2n-1, a has length n
        # Full convolution has length (2n-1) + n - 1 = 3n - 2
        # r[j] = conv(w,a)[(2n-1)+n-1-j] = conv(w,a)[3n-2-j]
        
        # So grad[j] = 2 * conv(w,a)[3n-2-j] ... but we only need j in [0,n-1]
        # So we need conv(w,a)[3n-2-(n-1)] to conv(w,a)[3n-2-0] = conv(w,a)[2n-1] to conv(w,a)[3n-2]
        
        w_full = w  # length 2n-1
        conv_wa = fast_convolution(w_full, a)  # length 3n-2
        
        # grad[j] = 2 * conv_wa[3n-2-j] for j=0,...,n-1
        # = 2 * conv_wa[2n+1 : 3n-1][::-1]  (indexing from 3n-2-(n-1)=2n-1 to 3n-2)
        idx_start = 3 * n - 2 - (n - 1)  # = 2n - 1
        idx_end = 3 * n - 2 + 1  # = 3n-1
        grad = 2.0 * conv_wa[idx_start:idx_end][::-1]
        
        # Gradient step with projection
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        iter_count += 1
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSEGD iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.3f}")
    
    return best_a, best_score


def fast_convolution(a, b):
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.15 * np.cos(3.0 * x) + 0.07 * np.cos(7.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
