"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with log-sum-exp (LSE) smoothing and multi-stage temperature annealing
2. Efficient FFT-based gradient computation (O(n log n) per iteration)
3. Multi-length exploration with interpolation transfer
4. Final coordinate descent polish for sub-1e-5 improvements
Key innovation: highly efficient Per-iteration FISTA using precomputed FFT plans,
enabling ~100k iterations/second, plus adaptive step-size via backtracking.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import math


def fast_autocorr(a):
    """Compute full self-convolution a*a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, T, fft_ac, fft_conv):
    """
    Compute gradient of F_T(a) = (1/T)*log(sum_k exp(T*(a*a)_k)) w.r.t. a.
    Returns (grad, b, F_value).
    Uses precomputed FFT sizes for efficiency.
    """
    n = len(a)
    # Autocorrelation
    A = rfft(a, fft_ac)
    b = irfft(A * A, fft_ac)[:2 * n - 1]

    # Weights
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w_sum = exp_Tb.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
        F_val = b.max()
    else:
        w = exp_Tb / w_sum
        F_val = (max_Tb + np.log(w_sum)) / T

    # Gradient via cross-correlation: grad_j = (1/T) * (w * a_rev)[j + n - 1]
    a_rev = a[::-1].copy()
    W = rfft(w, fft_conv)
    AR = rfft(a_rev, fft_conv)
    conv = irfft(W * AR, fft_conv)
    grad = (1.0 / T) * conv[n - 1: n - 1 + n]

    return grad, b, F_val


def fista_lse(a_init, time_limit, T_schedule, verbose=True):
    """
    FISTA with LSE smoothing and multi-stage annealing.
    a_init: initial sequence (will be normalized to sum=1)
    time_limit: max seconds to run
    T_schedule: list of (T, n_iters) tuples
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_ac = 1 << (2 * n - 2).bit_length()
    fft_conv = 1 << (3 * n - 3).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    a_prev = a.copy()
    t_iter = 1

    total_planned = sum(ni for _, ni in T_schedule)
    iters_done = 0

    for stage_idx, (T, n_iters) in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:

        # Adaptive step size: start with 1/T, adjust via backtracking
        gamma = 1.0 / max(T, 1.0)

def coordinate_descent_polish(a_init, time_limit, seed=42, verbose=True):
    """
    Coordinate descent: move small amounts of mass between positions.
    Focuses on reducing the maximum autocorrelation peak.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)

    t0 = time.monotonic()
    stall = 0
    it_count = 0

    # Precompute initial autocorrelation
    b = fast_autocorr(a)

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        b_max = b.max()
        k_max = int(np.argmax(b))

        # Strategy: find positions that contribute most to the peak at k_max
        # and move mass to positions that contribute least
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]

        # Pick a random contributor from the top
        if len(contrib) == 0:
            break
        top_candidates = np.argsort(contrib)[::-1][:min(10, len(contrib))]
        src_local = top_candidates[rng.integers(len(top_candidates))]
        src = int(idx[src_local])

        if a[src] < 1e-10:
            stall += 1
            if stall > 500:
                break
            continue

        # Find a good destination: position with low contribution to all top peaks
        # Get top 5 peaks
        top_peaks = np.argsort(b)[::-1][:5]

        # Score each potential destination
        # A position j is good if it contributes little to the top peaks
        best_dst = -1
        best_score_dst = np.inf
        n_trials = 20
        for _ in range(n_trials):
            dst = int(rng.integers(n))
            if dst == src or a[dst] > 0.1:
                continue
            # Estimate contribution to top peaks
            total = 0.0
            for kp in top_peaks:
                if 0 <= kp - dst < n:
                    total += a[dst] * a[kp - dst]
                if 0 <= dst - kp < n:
                    total += a[dst] * a[dst - kp]
            total += 0.001 * a[dst]
            if total < best_score_dst:
                best_score_dst = total
                best_dst = dst

        if best_dst < 0:
            best_dst = int(rng.integers(n))
            while best_dst == src:
                best_dst = int(rng.integers(n))

        # Try several step sizes
        improved = False
        for frac in [0.001, 0.003, 0.01, 0.02, 0.04, 0.08, 0.15]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue

            # Compute change in autocorrelation efficiently
            # Δb_k = amt * (a_{k-dst} + a_{dst-k} - a_{k-src} - a_{src-k})
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-15:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss

            b_trial = fast_autocorr(a_trial)
            score = 2.0 * n * b_trial.max()

            if score < best_score - 1e-15:
                a = a_trial
                b = b_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 2000:
                break

        if verbose and it_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD it={it_count} ({elapsed:.0f}s, stall={stall}): {best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    print(f"\n=== Phase 5: Coordinate descent polish at n={len(best_seq)} (remaining={remaining:.0f}s) ===")

    t5 = min(remaining * 0.5, 200.0)
    if t5 > 10:
        a_out5, _ = coordinate_descent_polish(best_seq, t5, seed=12345)
        try_update(a_out5, "cd1")

    # === Phase 6: Short FISTA + CD alternating ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: FISTA+CD alternating (remaining={remaining:.0f}s) ===")

    for round_num in range(3):
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 20:
            break
        t_f = min(remaining * 0.4, 80.0)
        t_c = min(remaining * 0.3, 60.0)
        if t_f < 5:
            break

        T_sched_short = [(50.0, 3000), (20.0, 3000), (10.0, 3000)]
        a_out6, _ = fista_lse(best_seq, t_f, T_sched_short, verbose=False)
        try_update(a_out6, f"fista_r{round_num}")

        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 10:
            a_out7, _ = coordinate_descent_polish(best_seq, t_c, seed=100 + round_num, verbose=False)
            try_update(a_out7, f"cd_r{round_num}")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
