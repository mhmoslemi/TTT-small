Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed optimization (log-sum-exp) with L-BFGS on softmax parameterization
2. Fast subgradient FISTA on the simplex
3. Coordinate descent with exact 1D line search

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar


def fast_conv(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def logsumexp_fn(x, beta):
    """Compute (1/beta)*log(sum(exp(beta*x))) stably."""
    mx = x.max()
    if mx < -700/beta:
        return mx  # all exp terms underflow
    return mx + np.log(np.exp(beta * (x - mx)).sum()) / beta

    c = fast_conv(a)  # c = a*a, length 2n-1
    
    # Objective
    obj = logsumexp_fn(c, beta)

    max_c = c.max()
    exp_bc = np.exp(beta * (c - max_c))
    p = exp_bc / exp_bc.sum()
    
    # Gradient of obj w.r.t. a: g_a = sum_k p_k * dc_k/da
    # dc_k/da_i = 2*a[k-i] if |k-i| <= n-1, else 0
    # This is: g_a = 2 * (p * convolve(p, flip(a)))... let me think.
    # Actually: g_a[i] = sum_k p_k * 2*a[k-i] (for valid k-i)
    # = 2 * sum_k p_k * a[k-i]
    # Let d = k - i, so k = i + d. Then g_a[i] = 2 * sum_d p[i+d] * a[d] (for valid i+d)
    # This is 2 * (p * a)[i] evaluated appropriately... 
    # Actually it's: g_a = 2 * cross_correlation(p, a) = 2 * convolve(p, flip(a))
    # Let me verify: convolve(p, flip(a))[i] = sum_j p[j] * flip(a)[i-j] = sum_j p[j] * a[n-1-(i-j)]
    # Hmm, that's not quite right. Let me just compute it directly.
    
    # g_a[i] = 2 * sum_{k: |k-i| <= n-1} p[k] * a[k-i]
    # Let d = k - i (ranges from -(n-1) to n-1)
    # g_a[i] = 2 * sum_{d=-(n-1)}^{n-1} p[i+d] * a[d] (where a[d] = 0 if d < 0 or d >= n)
    
    # This is a cross-correlation of p and a:
    # g_a = 2 * np.correlate(p, a, mode='full')... 
    # Actually, let me use the FFT approach for speed.
    # g_a[i] = 2 * sum_d p[i+d] * a[d]
    # Let's define q[d] = a[d] for d in [0,n-1], 0 otherwise.
    # Then g_a[i] = 2 * sum_d p[i+d] * q[d]
    # This is the cross-correlation (p ⊛ a)[i] = sum_d p[i+d] * a[d]... 
    # No, cross-correlation is (p ⊛ a)[i] = sum_d p[d] * a[d-i] or sum_d p[d+i]*a[d].
    # Let me just use: g_a = 2 * full cross-correlation of p and a.
    
    # The cross-correlation (p ⋆ a)[i] = sum_d p[d] * a[d+i] (with appropriate bounds)
    # numpy.correlate(p, a, 'full')[i + n - 1] = sum_j p[j] * a[j - (i + n - 1) + (n-1)]... 
    # This is confusing. Let me just use FFT.
    
    # g_a[i] = 2 * sum_d p[i+d] * a[d] where a[d] is 0 outside [0,n-1]
    # This equals 2 * (p reversed, convolved with a)[i + n - 1]...

    # g_a[i] = 2 * sum_{d=0}^{n-1} a[d] * p[i+d]
    # Let r = i + d, so d = r - i. g_a[i] = 2 * sum_{r=i}^{i+n-1} a[r-i] * p[r]
    # This is the cross-correlation of a and p evaluated at lag -i... 
    
    # OK let me just do it with a different approach.
    # g_a[i] = 2 * (sum_d a[d] * p[i+d])
    # = 2 * (convolve(flip(a), p))[i + n - 1]  (if we define things correctly)
    
    # Actually, the simplest way: note that the gradient of sum_k p_k * (a*a)[k] w.r.t. a
    # is: sum_k p_k * grad_k(a) where grad_k(a) = 2 * flip_shift(a, k)
    # flip_shift(a, k)[i] = a[k-i] if 0 <= k-i <= n-1, else 0
    # = shift(flip(a), k-n+1) or something like that.
    
    # Let me verify: flip(a) = a[::-1], so flip(a)[j] = a[n-1-j].
    # flip_shift(a, k)[i] = a[k-i]. 
    # If I define flip(a)[j] = a[n-1-j], then a[k-i] = flip(a)[n-1-k+i].
    # So flip_shift(a, k) = shift(flip(a), n-1-k).
    
    # So g_a = 2 * sum_k p_k * shift(flip(a), n-1-k)

    # convolve(p, flip(a))[m] = sum_j p[j] * flip(a)[m-j] = sum_j p[j] * a[n-1-(m-j)]
    # We want: sum_k p[k] * a[n-1-k+i] for each i
    # = convolve(p, flip(a))[i + ... ]
    
    # Let me just compute it directly using FFT for efficiency.
    # g_a[i] = 2 * sum_{d=0}^{n-1} a[d] * p[i+d]
    # This is the cross-correlation of a and p.
    # Using FFT: cross_corr(a, p) = ifft(conj(fft(a)) * fft(p))
    # But we need the aperiodic version.

    # g_a[i] = 2 * sum_{d=max(0, -i)}^{min(n-1, 2n-1-i)} a[d] * p[i+d]
    # where p has length 2n-1 (indices 0 to 2n-2)
    
    # For efficiency with n=1600, I'll use the FFT approach:

    # Let me verify: np.correlate(p, a, 'full')[m] = sum_j p[j] * conj(a[j - (m - (len(a)-1))])

    # Hmm, that gives sum_j p[j] * a[j-i], but I want sum_d a[d] * p[i+d] = sum_j p[j] * a[j-i] (with j = i+d, d = j-i).
    # So yes! g_a[i] = 2 * np.correlate(p, a, 'full')[n-1+i]
    
    # But np.correlate is O(n^2) for large inputs. For n=1600, that's 2.5M operations, which is fine.
    # But let me use FFT for speed.
    
    # Using FFT: the cross-correlation can be computed as:
    # corr = ifft(conj(fft(p, L)) * fft(a, L)) where L >= len(p) + len(a) - 1
    # This gives the linear cross-correlation.
    
    L = 1 << (len(p) + n - 1).bit_length()
    P = np.fft.fft(p, L)
    A = np.fft.fft(a, L)
    corr = np.fft.ifft(np.conj(P) * A).real
    # corr[m] = sum_j p[j] * a[j - m] (circular, but L is large enough for linear)
    # We want g_a[i] = 2 * sum_j p[j] * a[j-i] = 2 * corr[i] (for i >= 0)
    # Wait, corr[m] = sum_j p[j+m] * a[j] (by the definition of the VIP)... 
    # Let me just verify numerically later.
    
    # Actually, let me use a simpler and more reliable method:
    # g_a[i] = 2 * sum_{d=0}^{n-1} a[d] * p[i+d]
    # = 2 * (a * flip(p))[i + n - 1]  (full convolution)
    # Let me verify: (a * flip(p))[m] = sum_j a[j] * flip(p)[m-j] = sum_j a[j] * p[len(p)-1-(m-j)]
    # With len(p) = 2n-1: = sum_j a[j] * p[2n-2-m+j]
    # At m = i + n - 1: = sum_j a[j] * p[n-1-i+j]
    # Hmm, that's sum_j a[j] * p[n-1-i+j], but I want sum_d a[d] * p[i+d].
    
    # Let me just use the direct formula with FFT:
    # g_a[i] = 2 * sum_d a[d] * p[i+d]
    # Define b[d] = a[d] for d in [0,n-1]
    # Then g_a[i] = 2 * (cross_correlation of b and p at lag i)
    # = 2 * sum_d b[d] * conj(p[d-i])... no.
    
    # I'll just use the computation: g_a = 2 * np.convolve(a, p[::-1], mode='full')[n-1:n-1+n]
    # Let me verify: np.convolve(a, p[::-1])[m] = sum_j a[j] * p[::-1][m-j] = sum_j a[j] * p[len(p)-1-(m-j)]
    # = sum_j a[j] * p[2n-2-m+j] (since len(p)=2n-1)
    # At m = n-1-i: sum_j a[j] * p[2n-2-(n-1-i)+j] = sum_j a[j] * p[n-1+i+j]
    # I want: sum_d a[d] * p[i+d]. Setting d=j: sum_j a[j] * p[i+j].
    # But I got sum_j a[j] * p[n-1+i+j]. Off by n-1.
    
    # So: g_a[i] = 2 * np.convolve(a, p[::-1], mode='full')[n-1-i]  for appropriate i?
    # At m = n-1-i: sum_j a[j] * p[n-1+i+j]. That gives p[i+j+n-1], but I want p[i+j].
    
    # I think the issue is that p has length 2n-1 and I'm getting confused with indices.
    
    # Let me just brute-force it for correctness first, then optimize.
    # For n=1600, a direct O(n*(2n)) computation is O(n^2) = 2.5M, which is fine.
    
    # g_a[i] = 2 * sum_{d=0}^{n-1} a[d] * p[i+d]  (where p[i+d] = 0 if i+d >= 2n-1)
    
    # Using FFT: 
    # Let's define the linear cross-correlation: (a ⋆ p)[i] = sum_d a[d] * p[d+i]
    # This equals ifft(fft(a) * conj(fft(p)))... let me check.
    # Actually: sum_d a[d] * p[d+i] = sum_d a[d] * conj(conj(p))[d+i]
    # The VIP: sum_d a[d] * b[d+i] = ifft(fft(a) * fft(b))[i] (circular, with zero padding)
    # Wait no. The circular cross-correlation is: (a ⋆ b)[i] = sum_d a[(d+i) mod L] * conj(b[d])
    # For the linear case with zero padding: sum_d a[d] * conj(b[d-i])
    
    # I'm going in circles (pun intended). Let me just use the direct approach.
    # For n=1600, the O(n^2) computation takes about 2.5M multiply-adds, which is < 1ms in numpy.
    
    # Actually, the cleanest way:
    # g_a[i] = 2 * sum_d a[d] * p[i+d]
    # = 2 * (convolve(a, p))[i] ... no.
    # convolve(a, p)[k] = sum_j a[j] * p[k-j]
    # I want sum_d a[d] * p[i+d] = sum_d a[d] * p[(i+d)]
    # Let k = i+d, then d = k-i: sum_k a[k-i] * p[k] = convolve(flip_shift(a, i), p)... 
    
    # OK I'll just compute it as: 
    # Create a matrix M where M[i,d] = a[d]*p[i+d] and sum over d.
    # This is a Toeplitz-like structure.
    
    # Most efficient: use the FFT.
    # g_a[i] = sum_d a[d] * p[i+d]
    # This is the cross-correlation of sequences a (length n) and p (length 2n-1).
    # Using the theorem: cross_corr(a, p)[i] = ifft(fft(a, N) * conj(fft(p, N)))[i]
    # where N >= n + (2n-1) - 1 = 3n-2.
    
    # Let me verify with a small example later. For now, let me just use the direct formula
    # since n=1600 is manageable.
    
    # DIRECT COMPUTATION (O(n^2) but fine for n=1600):
    # g_a[i] = 2 * sum_{d=0}^{n-1} a[d] * p[i+d]
    # We can write this as: g_a = 2 * (a^T ⊗ I) @ p_vec... no.
    
    # Actually the simplest vectorized form:
    # For each i, g_a[i] = 2 * dot(a, p[i:i+n])
    # This is a "sliding dot product" which can be done with FFT.
    
    # Using the relation: dot(a, p[i:i+n]) = (a * flip(p))[i + n - 1] ... let me just verify.
    # (a * flip(p))[m] = sum_{j=0}^{n-1} a[j] * flip(p)[m-j] = sum_j a[j] * p[(2n-2)-(m-j)] = sum_j a[j] * p[2n-2-m+j]
    # At m = n-1+i: sum_j a[j] * p[n-1+i+j]
    # I want: sum_j a[j] * p[i+j]. So I need m such that 2n-2-m = i, i.e., m = 2n-2-i.
    # (a * flip(p))[2n-2-i] = sum_j a[j] * p[i+j]. ✓
    
    # So: g_a[i] = 2 * convolve(a, flip(p))[2n-2-i]
    # In numpy: full_conv = np.convolve(a, p[::-1]); g_a = 2 * full_conv[2n-2-n+1 : 2n-2 : -1]... 
    # full_conv has length n + (2n-1) - 1 = 3n-2.
    # full_conv[m] for m = 2n-2-i, where i ranges from 0 to n-1.
    # So m ranges from 2n-2 down to n-1.
    # g_a = 2 * full_conv[n-1 : 2n-1][::-1]
    
    # Let me verify with a tiny example.
    # a = [1, 2], p = [0.5, 0.3, 0.2] (n=2, len(p)=3=2n-1)
    # g_a[0] = 2 * (a[0]*p[0] + a[1]*p[1]) = 2*(0.5 + 0.6) = 2.2
    # g_a[1] = 2 * (a[0]*p[1] + a[1]*p[2]) = 2*(0.3 + 0.4) = 1.4
    # full_conv = np.convolve([1,2], [0.2, 0.3, 0.5]) = [0.2, 0.5, 1.0, 1.0, 1.0]
    # Wait: [1,2] * [0.2, 0.3, 0.5]:
    # [0] = 1*0.2 = 0.2
    # [1] = 1*0.3 + 2*0.2 = 0.7
    # [2] = 1*0.5 + 2*0.3 = 1.1
    # [3] = 2*0.5 = 1.0
    # full_conv = [0.2, 0.7, 1.1, 1.0]
    # n-1 = 1, 2n-1 = 3
    # full_conv[1:3] = [0.7, 1.1]
    # Reversed: [1.1, 0.7]
    # g_a = 2 * [1.1, 0.7] = [2.2, 1.4] ✓
    
    full_conv_ap = np.convolve(a, p[::-1])
    g_a = 2.0 * full_conv_ap[n-1:2*n-1][::-1]
    
    # Now, gradient w.r.t. u: da_i/du_i = a_i*(1-a_i) (for softmax)
    # Actually, for softmax: a_i = exp(u_i)/sum(exp(u_j))
    # da_i/du_j = a_i*(delta_ij - a_j)
    # dG/du_j = sum_i (dG/da_i) * (da_i/du_j) = sum_i g_a[i] * a_i*(delta_ij - a_j)
    # = g_a[j]*a_j*(1-a_j) - a_j*sum_i g_a[i]*a_i
    # = a_j * (g_a[j] - sum_i g_a[i]*a_i)
    
    grad_a_dot_a = np.dot(g_a, a)
    grad_u = a * (g_a - grad_a_dot_a)
    
    return obj, grad_u


def smoothed_objective_only(u, beta, n):
    a = softmax_fn(u)
    c = fast_conv(a)
    return logsumexp_fn(c, beta)


def fista_subgradient(a0, time_limit, max_iters=50000):
    """
    FISTA (accelerated proximal gradient) on the simplex.
    Minimizes max_k (a*a)[k] using subgradients.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    
    # FISTA parameters
    t = 1.0
    a_prev = a.copy()
    step_size = 0.5 / n  # Initial step (conservative)
    step_max = 2.0 / n  # Maximum step
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        if it == 0:
            y = a.copy()
        else:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            momentum = (t - 1.0) / t_new
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            y_s = y.sum()
            if y_s < 1e-15:
                y = a.copy()
                y_s = 1.0
            y = y / y_s
            t = t_new
        
        a_prev = a.copy()
        
        # Compute subgradient at y
        c = fast_conv(y)
        k_star = c.argmax()
        # Subgradient: 2 * flip_shift(y, k_star)
        # grad[i] = 2 * y[k_star - i] if 0 <= k_star - i <= n-1, else 0
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        grad = np.zeros(n)
        if lo <= hi:
            # grad[lo:hi+1] = 2 * y[k_star - lo : k_star - hi : -1]
            idx = np.arange(k_star - hi, k_star - lo + 1)
            grad[lo:hi+1] = 2.0 * y[idx]
        
        # Projected gradient step on simplex
        a_new = y - step_size * grad
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        # Adaptive step size (backtracking)
        score_new = fast_conv(a_new).max()
        score_y = c.max()
        if score_new > score_y * 1.01 + 1e-15:
            step_size = max(step_size * 0.5, 1e-10)
            a_new = y - step_size * grad
            a_new = project_simplex(a_new)
        else:
            step_size = min(step_size * 1.1, step_max)
        
        a = a_new
        score = fast_conv(a).max()
        if 2.0 * n * score < best_score:
            best_score = 2.0 * n * score
            best_a = a.copy()
        
        if it % 10000 == 0:
            print(f"      FISTA it={it}: best={best_score:.6f}")
    
    return best_a, best_score


def project_simplex(v, sz=1.0):
    """Project v onto the probability simplex {x >= 0, sum(x) = sz}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - sz))[0][-1]
    theta = (cssv[rho] - sz) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def coordinate_descent(a0, time_limit, max_rounds=200):
    """
    Coordinate descent: for each coordinate, find the best value.
    The objective max_k (a*a)[k] as a function of a[i] is the max of quadratics.
    We find the minimum of this max-of-quadratics by bisection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Fix all coordinates except i
            # (a*a)[k] as function of a[i]:
            # (a*a)[k] = sum_j a[j]*a[k-j]
            # Terms involving a[i]: a[i]*a[k-i] (from j=i) + a[k-i]*a[i] (from k-j=i, j=k-i)
            # = 2*a[i]*a[k-i] if i != k-i, or a[i]^2 if i = k-i
            # Plus terms not involving a[i]: sum_{j != i, k-j != i} a[j]*a[k-j]
            
            # Let base[k] = (a*a)[k] with a[i] set to 0
            # Then (a*a)[k] with a[i] = t:
            # If k-i is valid (0 <= k-i <= n-1) and k-i != i:
            #   (a*a)[k] = base[k] + 2*t*a[k-i] (the old a[i] is 0 in base)
            # Wait, base[k] already has a[i]=0, so:
            # (a*a)[k] = base[k] + t*a[k-i] + a[k-i]*t = base[k] + 2*t*a[k-i] (if k-i valid and k-i != i)
            # If k-i = i (k = 2i): (a*a)[k] = base[k] + t^2 + 2*t*0 = base[k] + t^2... 
            # Hmm, this is getting complicated. Let me just compute directly.
            
            # Actually, let me use a simpler approach:
            # For coordinate i, the new value t (with other coords fixed and renormalized)
            # changes the objective in a complex way. Instead, let me just try a few
            # candidate values and pick the best.
            
            old_val = a[i]
            if old_val < 1e-15:
                # Try adding a small amount
                for t in [0.001/n, 0.01/n, 0.05/n]:
                    a_trial = a.copy()
                    a_trial[i] = t
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_conv(a_trial).max()
                    if 2*n*sc < best_score:
                        a = a_trial
                        best_score = 2*n*sc
                        best_a = a.copy()
                        improved = True
                continue
            
            # Try reducing / increasing a[i]
            for frac in [0.3, 0.5, 0.7, 1.4, 1.7, 2.0]:
                t = old_val * frac
                a_trial = a.copy()
                a_trial[i] = t
                if t > a_trial.sum() - 1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s < 1e-15:
                    continue
                a_trial = a_trial / a_trial_s
                sc = fast_conv(a_trial).max()
                if 2*n*sc < best_score:
                    a = a_trial
                    best_score = 2*n*sc
                    best_a = a.copy()
                    improved = True
        
        if not improved and rnd > 5:
            break
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses smoothed L-B
