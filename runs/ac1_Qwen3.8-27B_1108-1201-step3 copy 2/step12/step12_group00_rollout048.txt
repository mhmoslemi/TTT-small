def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize(a, target_sum):
    """Normalize array to have the given sum."""
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s

def spectral_init(n, seed=42):
    """
    Generate initial sequence with flat power spectrum in frequency domain.
    Try multiple approaches and return the best.
    """
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    best_a = None
    best_score = float('inf')
    
    # Approach 1: Random phase in frequency domain (flat magnitude, random phase)
    for trial in range(20):
        fft_size = 1 << (2 * n - 1).bit_length()
        # Create flat spectrum with random phase
        freq_mag = np.ones(fft_size // 2 + 1)
        freq_phase = np.exp(1j * rng.uniform(0, 2 * np.pi, fft_size // 2 + 1))
        freq_phase[0] = 1.0  # DC component real
        freq_phase[-1] = 1.0  # Nyquist real
        spectrum = freq_mag * freq_phase
        a = irfft(spectrum, fft_size)[:n]
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-10:
            a = normalize(a, norm)
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    # Approach 2: Raised cosine window
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = normalize(a, norm)
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    # Approach 3: Blackman-Harris window
    x = np.linspace(0, 1, n)
    a = 0.36358 - 0.48917 * np.cos(2*np.pi*x) + 0.13659 * np.cos(4*np.pi*x) - 0.01064 * np.cos(6*np.pi*x)
    a = np.maximum(a, 0)
    a = normalize(a, norm)
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    # Approach 4: Gaussian-like
    x = np.linspace(0, 1, n)
    a = np.exp(-2 * (x - 0.5)**2)
    a = normalize(a, norm)
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    # Approach 5: Multi-lobe
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.3*np.cos(2*np.pi*x) + 0.1*np.cos(4*np.pi*x) + 0.05*np.cos(6*np.pi*x)
    a = np.maximum(a, 0)
    a = normalize(a, norm)
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    # Approach 6: Fejer kernel (sinc^2)
    x = np.arange(n)
    a = 1.0 - x / n
    a = np.maximum(a, 0)
    a = normalize(a, norm)
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    # Approach 7: Optimized cosine sum
    for n_terms in [2, 3, 4, 5]:
        coeffs = rng.dirichlet(np.ones(n_terms))

        x = np.linspace(0, 1, n)

        for j in range(n_terms):

            score = fast_autocorr(a).max()
            if score < best_score:

                best_a = a.copy()

def lp_best_response(a, M, solver):
    """Solve LP: maximize sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint: (a*b)_k <= M for all k
    # (a*b)_k = sum_j a_j * b_{k-j} for valid indices
    # This is a linear function of b for fixed a
    # (a*b)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_{k-j} * b_j
    
    # Build the matrix T where T[k,j] = a_{k-j} if 0<=k-j<n else 0
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = j + n
        T[start:end, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated Frank-Wolfe with LP best response."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(y)
        M_curr = b.max()
        if M_curr < 1e-12:
            break
        
        b_br = lp_best_response(y, M_curr, solver)
        if b_br is None:
            continue
        
        g = normalize(b_br, norm)
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = normalize(avg_a / n_avg, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def subgradient_nesterov(a0, time_limit, verbose=True):
    """Subgradient method with Nesterov acceleration (fast, no LP needed)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(y)
        k_star = int(np.argmax(b))
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                val += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                val += y[j2]
            grad[i] = 2.0 * val
        
        # Frank-Wolfe vertex: put all mass on argmin(grad)
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)
        
        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)
        
        y = y_new
        a = a_new
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = normalize(avg_a / n_avg, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent: move mass from peak-contributing positions to non-contributing ones."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_success = 0
    n_fail = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks
        top_k = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_k:
            # Positions contributing to peak k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            # Source: position with highest contribution
            src = idx[np.argmax(contrib)]
            
            # Find destination: position that contributes least to any top peak
            # Build peak influence mask
            influence = np.zeros(n)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                if len(idx2) > 0:
                    influence[idx2] += a[idx2] * a[k2 - idx2]
            
            # Candidates: positions with low influence
            candidates = np.where(influence < total_contrib * 0.01)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            if len(candidates) == 0:
                continue
                
            # Pick destination randomly from low-influence candidates
            # Prefer positions with low current value (to spread mass)
            cand_vals = a[candidates]
            # Weight towards lower values
            weights = 1.0 / (cand_vals + 1e-10)
            weights /= weights.sum()
            dst = candidates[rng.choice(len(candidates), p=weights)]
            
            # Try different transfer fractions
            found = False
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = normalize(a_trial, norm)
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if found:
                n_success += 1
                break
        
        if not improved:
            n_fail += 1
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.05, 0.3)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = normalize(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_success += 1
        
        if verbose and (n_success + n_fail) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.8f} (s={n_success}, f={n_fail})")
        
        if n_fail > 2000:
            break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def normalize(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def spectral_init(n, seed=42):

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm_val)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    consecutive_fail = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        M_curr = b.max()
        if M_curr < 1e-12:
            break

        b_br = lp_best_response(y, M_curr, solver)
        if b_br is None:
            consecutive_fail += 1
            if consecutive_fail > 50:
                break
            continue

        consecutive_fail = 0
        g = normalize(b_br, norm_val)
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm_val)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm_val)

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = normalize(avg_a / n_avg, norm_val)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:

                best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:

            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f} obj={2*n*best_score/norm_val**2:.8f}")

    return best_a, best_score


def proximal_gradient_polish(a0, time_limit, verbose=True):
    """Proximal gradient with weighted-peak surrogate and simplex projection."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm_val)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    conv_size = 2 * n - 1

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        # Weighted peak: use log-sum-exp with adaptive temperature
        T = max(0.1, 5.0 / (1.0 + t * 0.01))
        b_max_val = b.max()
        b_shifted = (b - b_max_val) / T
        b_shifted = np.clip(b_shifted, -500, 0)
        w = np.exp(b_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(conv_size) / conv_size
        else:
            w = w / w_sum

        # Gradient of weighted autocorrelation: grad_i = 2 * sum_k w_k * (a_{k-i} + a_{i-k})
        fft_grad_size = 1 << (3 * n - 2).bit_length()
        A_flip = rfft(a[::-1], fft_grad_size)
        W = rfft(w.astype(np.float64), fft_grad_size)
        conv1 = irfft(A_flip * W, fft_grad_size)
        c1 = conv1[n - 1:2 * n - 1]

        A_fwd = rfft(a, fft_grad_size)
        conv2 = irfft(A_fwd * W, fft_grad_size)
        c2 = conv2[0:n]

        grad = 2.0 * (c1 + c2)

        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = 1.0 / (grad_norm + 1e-10)
        step = min(step, 0.1)

        a_raw = a - step * grad

        # Project onto {a >= 0, sum(a) = norm_val}
        # Water-filling: a_proj_i = max(0, a_raw_i - tau)
        sorted_raw = np.sort(a_raw)[::-1]
        cumsum = np.cumsum(sorted_raw)
        rho = np.where(sorted_raw > 0)[0]
        if len(rho) == 0:
            break
        condition = (cumsum[rho] - norm_val) / (rho + 1)
        valid = condition < sorted_raw[rho]
        if valid.size == 0:
            break
        tau = condition[valid][-1]
        a_new = np.maximum(a_raw - tau, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm_val / s

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ProxGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
