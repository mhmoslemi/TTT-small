    return np.maximum(v - theta, 0.0)

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, x)) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def compute_subgrad_fft(a, u, fft_size, A_cache):
    """Compute g_j = 2 * sum_m a[m] * u[j+m] using FFT.
    a: length n, u: length 2n-1. Returns g of length n."""
    n = len(a)
    a_rev = a[::-1]
    # We need conv = convolve(a_rev, u), then g = conv[n-1:2n-1]
    # Use cached FFT of a_rev
    U = rfft(u, fft_size)
    conv = irfft(A_cache * U, fft_size)
    return 2.0 * conv[n-1 : 2*n-1]

def compute_v_from_u(a, u, fft_size, A_cache):
    """Compute v_j = sum_m a[m] * u[j+m] for j=0..n-1 (without the factor 2)."""
    n = len(a)
    g = compute_subgrad_fft(a, u, fft_size, A_cache)
    return g / 2.0

def solve_dual_mirror(a, fft_size, A_cache, n_iters=20, eta=0.1):
    """Solve dual game using mirror descent. Returns u (probability on 2n-1)."""
    n = len(a)
    u = np.ones(2*n-1) / (2*n-1)
    q = np.ones(n) / n
    
    for _ in range(n_iters):
        # Column player: compute v = f(u, *) and update q
        v = compute_v_from_u(a, u, fft_size, A_cache)
        # q ∝ q * exp(-eta * v)
        log_q = np.log(np.maximum(q, 1e-300)) - eta * v
        log_q -= log_q.max()
        q = np.exp(log_q)
        q_sum = q.sum()
        if q_sum < 1e-300:
            q = np.ones(n) / n
        else:
            q /= q_sum
        
        # Row player: compute w = f(*, q) and update u
        # w_k = sum_j q[j] * a[k-j] for valid k-j (0<=k-j<n)
        # = convolve(a, q)[k] for k = 0..2n-2
        Q = rfft(q, fft_size)
        w = np.real(irfft(A_cache * Q, fft_size))[:2*n-1]
        # u ∝ u * exp(eta * w)
        log_u = np.log(np.maximum(u, 1e-300)) + eta * w
        log_u -= log_u.max()
        u = np.exp(log_u)
        u_sum = u.sum()
        if u_sum < 1e-300:
            u = np.ones(2*n-1) / (2*n-1)
        else:
            u /= u_sum
    
    return u
