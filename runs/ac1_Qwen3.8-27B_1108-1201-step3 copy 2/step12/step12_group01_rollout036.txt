"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP Nesterov with constraint reduction via HiGHS (main workhorse)
2. Top-k LogSumExp gradient descent with Adam (gradient refinement)
3. Direct peak shaving (coordinate descent on peaks)
4. Multi-length exploration and final polish
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp


def fast_autocorr(a):
    """Full convolution a*a using FFT, returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_conv_rows(a, rows):
    """Build specified rows of convolution matrix T where T[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    K = len(rows)
    T = np.zeros((K, n))
    for idx in range(K):
        k = rows[idx]
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        if js <= je:
            j_arr = np.arange(js, je + 1)
            T[idx, js:je + 1] = a[k - j_arr]
    return T


def solve_lp_reduced(a, rows, use_highs=True):
    """Solve reduced LP: min M s.t. conv(a,b)[k] <= M for k in rows, sum(b)=1, b>=0.
    Returns (b_opt, M_opt) or (None, None)."""
    n = len(a)
    K = len(rows)
    if K == 0:
        return np.ones(n) / n, 0.0
    T = build_conv_rows(a, rows)

    if use_highs:
        c = np.zeros(n + 1)
        c[n] = 1.0
        # A_ub: T @ b - M <= 0  ->  [T | -1] @ [b; M] <= 0
        A_ub = np.hstack([T, -np.ones((K, 1))])
        b_ub = np.zeros(K)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
            if result.success and result.x is not None:
                b_opt = np.maximum(result.x[:n], 0.0)
                s = b_opt.sum()
                if s > 1e-15:
                    b_opt = b_opt / s
                    return b_opt, float(result.x[n])
        except Exception:
            pass

    # Fallback: CVXPY with ECOS
    try:
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=cp.ECOS)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
                return b_val, float(M_var.value)
    except Exception:
        pass

    return None, None


def solve_lp_full(a, use_highs=True):
    """Solve the full LP with all 2n-1 constraints."""
    n = len(a)
    rows = np.arange(2 * n - 1)
    return solve_lp_reduced(a, rows, use_highs)

def compute_topk_grad(a, T_temp, topk):
    """Compute top-k LSE objective and gradient w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argpartition(c, -k_use)[-k_use:]
    top_vals = c[top_idx]

    Tc = T_temp * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T_temp
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            j_arr = np.arange(js, je + 1)
            grad_a[js:je + 1] += 2.0 * wK * a[K - j_arr]

    return lse_val, grad_a


def topk_lse_grad_z(z, T_temp, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_grad(a, T_temp, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_topk_lse(a0, time_limit, T_start=150.0, T_end=0.5, n_stages=25, topk=10):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.75 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 60.0) * 5))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Multi-scale peak shaving: move mass from peak-contributing positions to safe ones."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:15]

        improved = False
        for k in top_peaks[:25]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_arr = np.arange(js, je + 1)
            if len(j_arr) == 0:
                continue
            contrib = a[j_arr] * a[k - j_arr]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue

            src_pos = js + np.argmax(contrib)

            # Danger zone
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:25]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src_pos]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src_pos)[0]

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(2, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08, 0.12]:
                amt = a[src_pos] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src_pos] -= amt
                a_trial[dst] += amt
                if a_trial[src_pos] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            n_pert = rng.integers(1, 5)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = (1.0 + np.cos(np.pi * x)) * np.sin(np.pi * x)
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.8 + 0.15 * np.cos(3 * x) + 0.05 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (x - 0.4) ** 2) + 0.6 * np.exp(-3.0 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.sqrt(np.sin(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining fast LP Nesterov with constraint reduction,
    top-k LSE gradient descent, and direct peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1680.0)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Fast LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s) ===")
    t1 = min(remaining * 0.62, 1050.0)
    if t1 > 60:
        a_out, _ = nesterov_fast(best_seq, t1, K=80, use_highs=True, verbose=True)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Second LP Nesterov pass with more constraints ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        t2 = min(remaining * 0.18, 300.0)
        if t2 > 30:
            print(f"=== Phase 2: LP Nesterov K=150 ({t2:.0f}s) ===")
            a_out, _ = nesterov_fast(best_seq, t2, K=150, use_highs=True, verbose=True)
            try_update(a_out, f"lp2_n{n_base}")

    # === Phase 3: Top-k LSE gradient refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t3 = min(remaining * 0.14, 220.0)
        if t3 > 20:
            print(f"=== Phase 3: Top-k LSE ({t3:.0f}s) ===")
            a_out, _ = adam_topk_lse(best_seq, t3, T_start=200.0, T_end=0.3, n_stages=25, topk=8)
            try_update(a_out, f"lse_n{n_base}")

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t4 = min(remaining * 0.12, 180.0)
        if t4 > 10:
            print(f"=== Phase 4: Peak shaving ({t4:.0f}s) ===")
            a_out, _ = peak_shaving(best_seq, t4, seed=42)
            try_update(a_out, f"peak_{n_base}")

    # === Phase 5: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 40:
        for delta_n in [100, -100, 200]:
            n_try = n_base + delta_n
            if n_try < 200:
                continue
            t_try = min(remaining * 0.12, 120.0)
            if t_try > 20:
                print(f"=== Phase 5: LP Nesterov n={n_try} ({t_try:.0f}s) ===")
                init_try = interpolate_sequence(best_seq, n_try)
                a_out, _ = nesterov_fast(init_try, t_try, K=80, use_highs=True, verbose=False)
                try_update(a_out, f"lp_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global_start)
            if remaining < 30:
                break

    # === Phase 6: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t6 = min(remaining * 0.25, 100.0)
        if t6 > 10:
            print(f"=== Phase 6: Final LSE polish ({t6:.0f}s) ===")
            a_out, _ = adam_topk_lse(best_seq, t6, T_start=5.0, T_end=0.1, n_stages=10, topk=6)
            try_update(a_out, "final_lse")

    # === Phase 7: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t7 = min(remaining * 0.5, 60.0)
        if t7 > 5:
            print(f"=== Phase 7: Final peak shave ({t7:.0f}s) ===")
            a_out, _ = peak_shaving(best_seq, t7, seed=999)
            try_update(a_out, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
