def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_topk_grad(a, T, topk):
    """Compute top-k LSE objective and gradient w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def get_lp_solver():

def solve_lp_best_response(a_fixed, solver, alt_solvers):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build constraint matrix implicitly using cp operations
    # conv(a_fixed, b)[k] = sum_j a_fixed[j] * b[k-j] for valid k,j
    # We express this as a linear constraint
    constraints = [cp.sum(b_var) == 1]
    # Build the convolution constraint efficiently
    # For each lag k, the constraint is sum_j a_fixed[j]*b[k-j] <= M
    # Use a banded matrix approach
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints.append(T @ b_var <= M_var)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver] + [s for s in alt_solvers if s != solver]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def adam_topk_lse(a0, time_limit, T_start=100.0, T_end=0.2, n_stages=30, topk=12, adaptive_k=True):
    """
    Adam-optimized top-k LSE with multi-stage annealing and adaptive top-k.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 8e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.75 ** si)
        # Adaptive top-k: more peaks early, fewer late
        if adaptive_k:
            k_cur = max(4, int(topk * (1.0 - 0.5 * si / max(1, n_stages - 1))))
        else:
            k_cur = topk

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 60.0) * 4))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, k_cur)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 50 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Multi-scale peak shaving with adaptive step sizes.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_peaks[:25]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:25]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.0005, 0.002, 0.005, 0.01, 0.02, 0.04, 0.07]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            n_pert = rng.integers(1, 5)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov at base length (most important, give it most time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.32, 550.0)
    if t1 > 60:
        a_out, sc = nesterov_lp_optimize(best_seq, t1, solver, alt_solvers, max_iters=500000)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: LSE refinement at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"=== Phase 2: LSE Adam n={n_base} ({remaining:.0f}s left) ===")
        t2 = min(remaining * 0.10, 120.0)
        if t2 > 15:
            a_out, _ = adam_topk_lse(best_seq, t2, T_start=80.0, T_end=0.2,
                                      n_stages=25, topk=14, adaptive_k=True)
            try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: LP Nesterov at medium length ===
    n_med = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 80:
        print(f"=== Phase 3: LP Nesterov n={n_med} ({remaining:.0f}s left) ===")
        t3 = min(remaining * 0.25, 400.0)
        if t3 > 60:
            init_med = interpolate_sequence(best_seq, n_med)
            a_out, sc = nesterov_lp_optimize(init_med, t3, solver, alt_solvers, max_iters=400000)
            try_update(a_out, f"lp_n{n_med}")

    # === Phase 4: LSE at medium length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"=== Phase 4: LSE Adam n={n_med} ({remaining:.0f}s left) ===")
        t4 = min(remaining * 0.08, 100.0)
        if t4 > 15:
            a_out, _ = adam_topk_lse(best_seq, t4, T_start=60.0, T_end=0.3,
                                      n_stages=20, topk=12, adaptive_k=True)
            try_update(a_out, f"lse_n{n_med}")

    # === Phase 5: LP Nesterov at long length ===
    n_long = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 80:
        print(f"=== Phase 5: LP Nesterov n={n_long} ({remaining:.0f}s left) ===")
        t5 = min(remaining * 0.22, 350.0)
        if t5 > 60:
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, sc = nesterov_lp_optimize(init_long, t5, solver, alt_solvers, max_iters=300000)
            try_update(a_out, f"lp_n{n_long}")

    # === Phase 6: LSE at long length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t6 = min(remaining * 0.08, 80.0)
        if t6 > 10:
            print(f"=== Phase 6: LSE n={n_long} ({remaining:.0f}s left) ===")
            a_out, _ = adam_topk_lse(best_seq, t6, T_start=40.0, T_end=0.5,
                                      n_stages=15, topk=10, adaptive_k=True)
            try_update(a_out, f"lse_n{n_long}")

    # === Phase 7: Multi-start at base length (diversity) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 40:
        print(f"=== Phase 7: Multi-start ({remaining:.0f}s left) ===")
        t7_total = min(remaining * 0.12, 130.0)
        if t7_total > 30:
            n_starts = 3
            t7_each = t7_total / n_starts
            for si in range(n_starts):
                if time.monotonic() - t_global_start > time_budget - 60:
                    break
                pert = perturbed_init(best_seq, rng, scale=0.04 * (si + 1))
                a_out, _ = adam_topk_lse(pert, t7_each, T_start=50.0, T_end=1.0,
                                          n_stages=12, topk=10, adaptive_k=True)
                try_update(a_out, f"multi_start_{si}")

    # === Phase 8: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t8 = min(remaining * 0.12, 100.0)
        if t8 > 8:
            print(f"=== Phase 8: Peak shaving ({remaining:.0f}s left) ===")
            a_out, _ = peak_shaving(best_seq, t8, seed=456)
            try_update(a_out, "peak_shave_1")

    # === Phase 9: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t9 = min(remaining * 0.25, 120.0)
        if t9 > 8:
            print(f"=== Phase 9: Final LSE polish ({remaining:.0f}s left) ===")
            a_out, _ = adam_topk_lse(best_seq, t9, T_start=5.0, T_end=0.05,
                                      n_stages=15, topk=8, adaptive_k=False)
            try_update(a_out, "final_lse")

    # === Phase 10: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 8:
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 5:
            print(f"=== Phase 10: Final peak shave ({remaining:.0f}s left) ===")
            a_out, _ = peak_shaving(best_seq, t10, seed=789)
            try_update(a_out, "peak_shave_2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
