def _fast_autocorr(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def _build_conv_matrix(a):
    """Build the convolution matrix T where (T @ b)_k = (a*b)_k."""
    n = len(a)
    # T is (2n-1) x n
    # T[k, i] = a[i] if 0 <= k-i < n, else 0
    # i.e., T[k, i] = a[k-i] if 0 <= k-i < n
    # Wait, let me re-derive.
    # (a*b)_k = sum_i a[i] * b[k-i] for valid indices
    # = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * b[k-i]
    # = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} a[i] * b[k-i]
    # Let j = k-i, so i = k-j:
    # = sum_{j: max(0,n-1-k) <= j <= min(n-1,k)} a[k-j] * b[j]
    # So T[k, j] = a[k-j] if max(0, n-1-k) <= j <= min(n-1, k), else 0
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, n - 1 - k)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            T[k, j] = a[k - j]
    return T


def _build_conv_matrix_fast(a):

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        # T[k, j] = a[k-j] for k in [j, j+n-1]
        T[j:j+n, j] = a

def _lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix_fast(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _nesterov_lp(a0, time_limit, solver, verbose=True):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improvement = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            last_improvement = n_iters

            best_a = a.copy()
            last_improvement = n_iters

        # Averaging

        n_avg += 1

            if s_avg > 1e-15:

def _lse_gradient_descent(a0, time_limit, T_start=20.0, T_end=200.0, n_stages=8):
    """Optimize using log-sum-exp smoothed objective with Nesterov."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        n_steps = max(50, min(500, int(remaining * 3)))
        step_base = norm / (T * n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            b = _fast_autocorr(a)
            # LSE gradient
            max_b = b.max()
            exp Tb = np.exp(T * (b - max_b))
            w = exp_Tb / exp_Tb.sum()

            # grad_i = 2 * sum_k w_k * a[k-i] (for valid k-i)
            # This is 2 * (reversed_a convolved with w)
            fft_size = 1 << (2 * n).bit_length()
            A_rev = rfft(a[::-1], fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad = 2.0 * conv[n - 1:n - 1 + n]

            # Nesterov
            step = step_base / (1.0 + 0.5 * it / n_steps)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            score = _fast_autocorr(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()

        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_score


def _coordinate_descent(a0, time_limit, seed=123):
    """Subgradient-guided coordinate descent with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        k_star = int(np.argmax(b))

        # Compute subgradient: for each i, how much does a[i] contribute to peak k*?
        # (a*a)_{k*} = sum_i a[i]*a[k*-i]
        # d/d a[i] (a*a)_{k*} = a[k*-i] + a[i-k*] (for valid indices)
        contrib = np.zeros(n)
        for i in range(n):
            c = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                c += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                c += a[j2]
            contrib[i] = c

        # Also consider other peaks
        top_peaks = np.argsort(b)[::-1][:5]
        for k2 in top_peaks:
            if k2 == k_star:
                continue
            for i in range(n):
                j1 = k2 - i
                if 0 <= j1 < n:
                    contrib[i] += 0.5 * a[j1]
                j2 = i - k2
                if 0 <= j2 < n:
                    contrib[i] += 0.5 * a[j2]

        # Source: highest contribution among positions with significant mass
        src_candidates = np.where(a > norm * 0.005)[0]
        if len(src_candidates) == 0:
            src_candidates = np.where(a > 0)[0]
        if len(src_candidates) == 0:
            break
        src = src_candidates[np.argmax(contrib[src_candidates])]

        # Destination: lowest contribution among non-critical positions
        dst_candidates = np.where(a < norm * 0.1)[0]
        if len(dst_candidates) == 0:
            dst_candidates = np.arange(n)
        dst = dst_candidates[np.argmin(contrib[dst_candidates])]

        if src == dst:
            # Random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst:
                continue

        # Line search: try different amounts
        improved = False
        for frac in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
            amt = min(a[src] * frac, a[src] * 0.5)
            if amt < 1e-18:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = _fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved:
            stall += 1
            # Random perturbation
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            if i1 != i2 and a[i1] > norm * 1e-6:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i1] * frac
                a_trial = a.copy()
                a_trial[i1] -= amt
                a_trial[i2] += amt
                if a_trial[i1] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 200:
                break
        else:
            stall = 0

    return best_a, best_score


def _interpolate(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

This is very cheap to compute! We just need to find the minimum of y (excluding the first element, or including 2*y[0]).

More generally, for arbitrary k*:
h[i] = y[k*-i] (if 0 <= k*-i < n) + y[i-k*] (if 0 <= i-k* < n)

For i in [max(0, k*-n+1), min(n-1, k*)]: both terms might be valid.
For i < k*-n+1 or i > k*: only one or zero terms are valid.

This is still O(n) to compute, which is fine.

So the Frank-Wolfe step is:
1. Compute b = autocorr(a), find k* = argmax(b). [O(n log n)]
2. Compute h[i] = y[k*-i] + y[i-k*] for all i. [O(n)]
3. Find i* = argmin h[i]. [O(n)]
4. v = e_{i*}. [O(1)]
5. Nesterov update. [O(n)]

Total per iteration: O(n log n) (dominated by the FFT).

This is very efficient! We can do many more iterations than the LP-based method (which is O(n^2) per iteration due to the LP solve).

But the question is: does the Frank-Wolfe method converge to a better solution than the LP best response? The LP best response gives the exact best response in the game, while Frank-Wolfe gives a vertex of the simplex. In general, the LP best response should give a better (more "spread out") candidate, leading to faster convergence.

However, the Frank-Wolfe method is much faster per iteration, so we can do many more of them. The question is whether the total progress is better.

In practice, I think the LP method is better for the main optimization, and Frank-Wolfe is useful as a fast "warm-up" or "polish" step.

Wait, that's the circular autocorrelation of the zero-padded sequence, which equals the linear autocorrelation (since L >= 2n-1 prevents wrap-around).

So: c = IDFT(|FFT(a_padded)|^2)[:2n-1]

The peak of c is what we want to minimize. In the frequency domain, |A|^2 is the power spectrum. We want the IDFT of the power spectrum to have a small peak.

This is equivalent to: the power spectrum |A|^2 should be as "flat" as possible (in the sense that its IDFT has small peak).

A flat power spectrum corresponds to a "white noise"-like time-domain signal. But our signal must be non-negative, which constrains the possible spectra.

One approach: **iterative spectral flattening**.
1. Start with a (non-negative).
2. Compute |A|^2 = |FFT(a)|^2.
3. Modify |A|^2 to be flatter (e.g., reduce the peaks in the power spectrum).
4. Invert: a_new = IDFT(sqrt(|A_modified|^2) * phase(A)).
5. Project: a_new = max(0, a_new), then normalize.
6. Repeat.

But step 4 is tricky because we need to maintain the phase of A, and the result might not be real.

Alternative: use the "projection onto the intersection of convex sets" (POCS) approach:
- Set 1: {a: a >= 0, sum(a) = 1} (the simplex)
- Set 2: {a: max_k (a*a)_k <= M} (the sublevel set of the max autocorrelation)

We want to find a point in the intersection (or near it). POCS alternates projections onto each set.

Projection onto Set 1 (simplex): well-known algorithm (sort + threshold).
Projection onto Set 2: this is the LP! min ||a - a0||_2 s.t. (a*a)_k <= M for all k, a >= 0, sum(a) = 1.

But this projection is non-convex (the constraint (a*a)_k <= M is non-convex in a). So POCS doesn't directly apply.

However, we can use the LP best response as a "quasi-projection" onto Set 2: given a, find b in the simplex that minimizes max_k (a*b)_k (or equivalently, maximizes sum(b) subject to (a*b)_k <= M). This is the "best response" in the game sense.
