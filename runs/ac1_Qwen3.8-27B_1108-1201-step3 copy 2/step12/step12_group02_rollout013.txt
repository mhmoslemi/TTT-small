Let me just define it in terms of convolution:
`sum_j q[i+j]*p[j]`. Let `j' = i+j`, so `j = j'-i`:
`= sum_{j'} q[j']*p[j'-i]`
This is `(q * p_reversed_shifted)`... 

OK, I think the cleanest way is:
Define `p_flip[j] = p[len(p)-1-j]` (i.e., reverse p).
Then `sum_j q[i+j]*p[j] = sum_j q[i+j]*p_flip[len(p)-1-j]`.
Let `j'' = len(p)-1-j`: `= sum_{j''} q[i+len(p)-1-j'']*p_flip[j'']`.
Let `k = i + len(p) - 1`: `= sum_{j''} q[k-j'']*p_flip[j''] = (q * p_flip)[k]` (convolution).

So `grad[i]/2 = (q * p_flip)[i + len(p) - 1]` where `*` denotes linear convolution.

With q = u (length 2n-1) and p = a (length n):
`grad[i]/2 = (u * a_flip)[i + n - 1]`

So I compute the convolution of u (length 2n-1) with a_flip (length n), take elements at positions n-1, n, ..., 2n-2, and multiply by 2.

The convolution has length (2n-1) + n - 1 = 3n-2. We need elements at indices n-1 through 2n-2, which is n elements. 

Let me just code this up cleanly.

Now, for the overall algorithm, here's my refined plan:

**Phase 1: Initialize**
- Use parent sequence or diverse initializations
- Normalize to sum = 1

**Phase 2: LogSumExp optimization (main workhorse)**
- Use T from high to low (annealing)
- At each T, run L-BFGS with FFT-based gradients
- The gradient uses the full softmax weighting (not just top-k)
- This naturally handles multiple competing peaks

**Phase 3: Subgradient mirror descent with soft weighting**
- Use exponential weights over all peaks (temperature-controlled)
- Multiplicative updates for non-negativity
- Adaptive learning rate

**Phase 4: Local search (pair swaps)**
- Randomly pick pairs (i,j) and try moving mass from i to j
- Use FFT to quickly evaluate the impact
- Accept if it improves the score

**Phase 5: Final polish**
- Very fine coordinate descent
- Try small perturbations

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. LogSumExp-smoothed optimization with L-BFGS and FFT-based gradients
2. Softmax-weighted subgradient mirror descent (all peaks, not just top-k)
3. Targeted pair-swap local search
4. Multi-restart with diverse initializations and symmetry exploitation
Key improvements over previous: full-peak softmax weighting, adaptive
temperature annealing, and pair-swap local search for fine-tuning.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full autocorrelation a*a (linear convolution) using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1 + 1).bit_length()  # next power of 2 >= 2n
    A = rfft(a, size)
    return irfft(A * A, size)[:2 * n - 1]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def compute_soft_gradient(a, b, T):
    """
    Compute the gradient of the LogSumExp approximation of max(b).
    b = conv(a,a), length 2n-1.
    Returns gradient wrt a (length n).
    
    Uses the identity: grad_a = 2 * (w * a_flip)[n-1:2n-1]
    where w = softmax(T*b) / sum(softmax(T*b))
    """
    n = len(a)
    Tb = T * (b - b.max())
    w = np.exp(Tb)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(b)) / len(b)
    else:
        w = w / w_sum
    
    a_flip = a[::-1]
    conv_size = 1 << ((2 * n - 1) + n - 1 + 1).bit_length()
    W = rfft(w, conv_size)
    AF = rfft(a_flip, conv_size)
    conv = irfft(W * AF, conv_size)
    
    # grad[i] = 2 * conv[i + n - 1] for i = 0..n-1
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, w


def lse_objective_and_grad(a, T):
    """
    Compute the LSE-smoothed objective and its gradient.
    Objective: (1/T) * log(sum(exp(T*b_k)))
    """
    n = len(a)
    b = fast_autocorr(a)
    Tmax = T * b.max()
    exp_b = np.exp(T * (b - b.max()))
    obj = (Tmax + np.log(exp_b.sum())) / T
    
    grad_b = exp_b / exp_b.sum()  # softmax(T*b)
    
    a_flip = a[::-1]
    conv_size = 1 << ((2 * n - 1) + n - 1 + 1).bit_length()
    W = rfft(grad_b, conv_size)
    AF = rfft(a_flip, conv_size)
    conv = irfft(W * AF, conv_size)
    grad_a = 2.0 * conv[n - 1:2 * n - 1]
    
    return obj, grad_a


def lse_optimize(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=15, seed=42):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    Optimizes in the log-domain (z-space) where a = softmax(z)/sum(softmax(z)).
    Actually, we optimize in a-space directly since a >= 0 is a convex constraint.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_score(a)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Adaptive iteration count based on remaining time and stage
        iters = max(10, min(60, int(20 + 40 * (si + 1) / n_stages)))
        
        def func(a_flat):
            a_clip = np.maximum(a_flat, 0)
            s = a_clip.sum()
            if s < 1e-15:
                return 1e10
            a_clip = a_clip / s
            obj, _ = lse_objective_and_grad(a_clip, stage_T)
            return obj
        
        def jac(a_flat):
            a_clip = np.maximum(a_flat, 0)
            s = a_clip.sum()
            if s < 1e-15:
                return np.zeros(n)
            a_clip = a_clip / s
            _, grad = lse_objective_and_grad(a_clip, stage_T)
            # Project gradient to sum=0 subspace (for the normalization)
            grad = grad - grad.mean()
            return grad
        
        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': iters, 'ftol': 1e-14,
                                    'gtol': 1e-8, 'maxfun': iters * 50,
                                    'disp': False})
            a_new = np.maximum(res.x, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc = fast_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
        except Exception as e:
            pass
        
        if (si + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def softmax_mirror_descent(a0, time_limit, T=5.0, eta0=0.1, decay=0.5, seed=42):
    """
    Mirror descent with full-softmax subgradient (all peaks weighted).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    last_best = best_score
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        
        # Full softmax weights
        Tb = T * (b - b.max())
        w = np.exp(Tb)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(b)) / len(b)
        else:
            w = w / w_sum
        
        # Compute gradient
        a_flip = a[::-1]
        conv_size = 1 << ((2 * n - 1) + n - 1 + 1).bit_length()
        W = rfft(w, conv_size)
        AF = rfft(a_flip, conv_size)
        conv = irfft(W * AF, conv_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # Adaptive learning rate
        eta = eta0 / (1.0 + t / 200.0) ** decay
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / s_new
        
        a = a_new
        
        # Check score
        if t % 20 == 0 or t < 100:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        
        # Adaptive temperature: increase T when stalled to focus more on max peak
        if stall > 100:
            T = min(T * 1.3, 50.0)
            stall = 0
        elif stall > 300:
            # Add noise to escape
            noise = 0.02 * rng.standard_normal(n)
            a = np.maximum(a + noise, 1e-30)
            a = a / a.sum()
            T = max(1.0, T * 0.8)
            stall = 0
        
        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.2f}")
    
    return best_a, best_score


def pair_swap_local_search(a0, time_limit, seed=42):
    """
    Local search: randomly swap mass between pairs of positions.
    Accept if it reduces the max autocorrelation peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, fast_score(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    # Precompute initial autocorrelation
    b = fast_autocorr(a)
    b_max = b.max()
    
    while time.monotonic() - t0 < time_limit:
        # Pick a source (high value) and destination (low value)
        # Weight source selection by a[i]^2 (high values contribute more to peaks)
        src = int(rng.choice(n, p=a / a.sum()))
        
        # Pick destination from low-value positions
        # Use temperature-based sampling to occasionally pick non-obvious destinations
        dst_weights = np.ones(n) / (a + 0.01)
        dst_weights[src] = 0
        dst = int(rng.choice(n, p=dst_weights / dst_weights.sum()))
        
        # Try moving a fraction of mass from src to dst
        found = False
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = a_trial / a_trial.sum()
            
            b_trial = fast_autocorr(a_trial)
            new_max = b_trial.max()
            
            if new_max < b_max - 1e-14:
                a = a_trial
                b = b_trial
                b_max = new_max
                sc = 2.0 * n * b_max / (a.sum() ** 2)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                found = True
                stall = 0
                break
            else:
                # Revert
                a_trial[src] += amt
                a_trial[dst] -= amt
        
        if not found:
            stall += 1
            if stall > 500:
                # Larger perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    if amt < a[i]:
                        a[i] -= amt
                        a[j] += amt
                        a = a / a.sum()
                        b = fast_autocorr(a)
                        b_max = b.max()
                        sc = 2.0 * n * b_max / (a.sum() ** 2)
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                stall = 0
            if stall > 3000:
                break
    
    return best_a, best_score


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))
    
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        # Sinc-squared
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        # Multi-peak Gaussian
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.1
        for _ in range(5):
            center = rng.uniform(0.1, 0.9)
            width = rng.uniform(0.02, 0.15)
            a += np.exp(-(x - center) ** 2 / (2 * width ** 2))
    elif variant == 8:
        # Periodic with random phases
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0
        for k in [2, 3, 5, 7, 11]:
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.2 * np.cos(k * x + phase)
        a = np.maximum(a, 0)
    elif variant == 9:
        # V-shaped
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        # Random smooth
        a = np.ones(n) * 0.1
        x = np.linspace(0, 1, n)
        for _ in range(8):
            center = rng.uniform(0, 1)
            width = rng.uniform(0.05, 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)
    else:
        # Thinned uniform (alternating high/low)
        a = np.ones(n)
        a[::2] = 0.5
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Fine-grained coordinate descent: adjust individual elements
    to reduce the maximum peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

        return a0, fast_score(a0)
    a = a / s
    
    best_a = a.copy()

    t0 = time.monotonic()
    
    b = fast_autocorr(a)
    b_max = b.max()
    peak_k = int(b.argmax())
    
    while time.monotonic() - t0 < time_limit:
        # Focus on the current peak
        b = fast_autocorr(a)
        b_max = b.max()
        peak_k = int(b.argmax())
        
        # The peak at position peak_k is: c[peak_k] = sum_i a[i]*a[peak_k-i]
        # To reduce it, we should decrease products a[i]*a[peak_k-i]
        # Find the largest contributor
        js = np.arange(max(0, peak_k - n + 1), min(n - 1, peak_k) + 1)
        if len(js) == 0:
            break
        contrib = a[js] * a[peak_k - js]
        max_idx = int(js[np.argmax(contrib)])
        
        # Try reducing a[max_idx] and adding to a far position
        # Pick a position far from the peak's influence
        far_positions = np.abs(np.arange(n) - max_idx) > n // 4

            far_positions = np.ones(n, dtype=bool)
            far_positions[max_idx] = False
        
        candidates = np.where(far_positions)[0]
        # Choose the one with lowest a value
        dst = int(candidates[np.argmin(a[candidates])])
        
        improved = False

                    best_a = a.copy()

                break
        
        if not improved:

            i = int(rng.integers(n))

                a[j] += amt
                a = a / a.sum()
                b = fast_autocorr(a)
                if b.max() < best_score / (2.0 * n):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Full-peak softmax-weighted mirror descent (all peaks, adaptive temperature)

Key improvement: full-peak softmax weighting (not just top-k), careful annealing,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, size)
    return irfft(A * A, size)[:2 * n - 1]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def compute_autocorr_gradient(a, weights):
    """
    Compute gradient of sum_k weights[k]*c[k] wrt a.
    c = conv(a,a), so dc[k]/da[i] = 2*a[k-i] (when 0<=k-i<n).
    grad[i] = 2 * sum_{k: 0<=k-i<n} weights[k]*a[k-i]
    This equals 2 * (weights * a_flipped)[n-1 : 2n-1]
    """
    n = len(a)
    a_flip = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    size = 1 << (conv_len).bit_length()
    W = rfft(weights, size)
    AF = rfft(a_flip, size)
    conv = irfft(W * AF, size)
    return 2.0 * conv[n - 1:2 * n - 1]


class LSEProblem:
    """Encapsulates LSE-smoothed objective and gradient for scipy minimize."""

    def __init__(self, T):
        self.T = T

    def func(self, a_flat):
        a = np.maximum(a_flat, 0.0)
        s = a.sum()
        if s < 1e-15:
            return 1e10
        a = a / s
        b = fast_autocorr(a)
        T = self.T
        Tbm = T * b.max()
        exp_sum = np.exp(T * (b - b.max())).sum()
        return (Tbm + np.log(exp_sum)) / T

    def jac(self, a_flat):
        a = np.maximum(a_flat, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.zeros(len(a_flat))
        a = a / s
        b = fast_autocorr(a)
        T = self.T
        exp_b = np.exp(T * (b - b.max()))
        w = exp_b / exp_b.sum()
        grad = compute_autocorr_gradient(a, w)
        grad = grad - grad.mean()
        return grad

def lse_optimize(a0, time_limit, T_start=300.0, T_end=0.5, n_stages=15, seed=42):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_score(a)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        stage_T = temps[si]
        iters = max(15, min(80, int(25 + 55 * (si + 1) / n_stages)))

        prob = LSEProblem(stage_T)

        try:
            res = minimize(fun=prob.func, x0=a, jac=prob.jac,
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': iters, 'ftol': 1e-14,
                                    'gtol': 1e-9, 'maxfun': iters * 100,
                                    'disp': False})
            a_new = np.maximum(res.x, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc = fast_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

        if (si + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def softmax_mirror_descent(a0, time_limit, T_init=10.0, eta0=0.08, decay=0.5, seed=42):

            w = np.ones(len(b)) / len(b)
        else:
            w = w / w_sum

        # Compute gradient
        grad = compute_autocorr_gradient(a, w)

        # Adaptive learning rate
        eta = eta0 / (1.0 + t / 200.0) ** decay

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)

        else:

def pair_swap_local_search(a0, time_limit, seed=42):
    """
    Local search: move mass between positions to reduce max peak.
    Uses FFT for fast evaluation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.asarray(a0), fast_score(np.asarray(a0))
    a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    b = fast_autocorr(a)
    b_max = b.max()

    while time.monotonic() - t0 < time_limit:
        # Weighted source selection (high a values contribute more to peaks)
        src = int(rng.choice(n, p=a / a.sum()))

        # Destination: prefer low-value positions far from source
        dst_w = 1.0 / (a + 0.005)
        dst_w[src] = 0.0
        dst_w /= dst_w.sum()
        dst = int(rng.choice(n, p=dst_w))

        found = False
        for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.12]:
            amt = a[src] * frac
            if amt < 1e-15 or amt > a[src]:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = a_trial / a_trial.sum()

            b_trial = fast_autocorr(a_trial)
            new_max = b_trial.max()

            if new_max < b_max - 1e-14:
                a = a_trial
                b_bmax = new_max
                b_max = new_max
                sc = 2.0 * n * b_max
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                found = True
                stall = 0
                break

        if not found:
            stall += 1
            if stall > 400:
                # Bigger kick
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.02, 0.15)
                    amt = min(a[i] * frac, a[i] * 0.9)

                stall = 0

                break

def targeted_peak_reduce(a0, time_limit, seed=42):
    """
    Iteratively identify the dominant peak and reduce it by redistributing
    mass from its contributing elements to distant positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.asarray(a0), fast_score(np.asarray(a0))
    a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        peak_k = int(b.argmax())

        # Find contributing indices for this peak
        js = np.arange(max(0, peak_k - n + 1), min(n - 1, peak_k) + 1)
        if len(js) == 0:
            break
        contrib = a[js] * a[peak_k - js]

        # Find the dominant contributor
        max_j = int(js[np.argmax(contrib)])

        # Find a distant, low-value position to move mass to
        dist = np.abs(np.arange(n) - max_j)
        far_mask = dist > n // 3
        if not far_mask.any():
            far_mask = np.ones(n, dtype=bool)
            far_mask[max_j] = False
        candidates = np.where(far_mask)[0]
        dst = int(candidates[np.argmin(a[candidates])])

        improved = False
        for frac in [0.0005, 0.001, 0.003, 0.007, 0.015, 0.03]:
            amt = a[max_j] * frac
            if amt < 1e-16 or amt > a[max_j]:
                continue
            a_trial = a.copy()
            a_trial[max_j] -= amt
            a_trial[dst] += amt
            a_trial = a_trial / a_trial.sum()
            b_trial = fast_autocorr(a_trial)
            if b_trial.max() < b_max - 1e-15:
                a = a_trial
                b_max = b_trial.max()
                sc = 2.0 * n * b_max
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Try random perturbation

                i = int(rng.integers(n))
                j = int(rng.integers(n))

                    a[j] += amt

                    sc = 2.0 * n * new_max

                        best_a = a.copy()

        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_arr = np.clip(a_arr, 0.0, 1000.0)
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function with multi-phase optimization:
    Phase 1: LogSumExp L-BFGS on parent (main workhorse)
    Phase 2: Softmax mirror descent (global exploration)
    Phase 3: Targeted peak reduction (fine local search)
    Phase 4: Pair swap local search
    Phase 5: Second LSE round with different parameters
    Phase 6: Length augmentation and re-optimization
    Phase 7: Final combined polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1760.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent or create default
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = fast_score(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_base = len(parent)
        else:
            raise ValueError("Invalid parent data")
    except Exception:
        print("No valid parent available, using U-shaped default")
        best_seq = diverse_init(n_base, rng, variant=4)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Helper to update best (inlined to avoid closure)
    # (We'll just repeat the pattern inline)

    # === Phase 1: LogSumExp optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LogSumExp L-BFGS (n={n_base}) ===")
    t1 = min(remaining * 0.28, 450.0)
    if t1 > 30 and best_seq is not None:
        a_out, sc_out = lse_optimize(best_seq, t1, T_start=500.0, T_end=0.3, n_stages=16, seed=1)
        sc_check = fast_score(a_out)
        if sc_check < best_score:
            best_score = sc_check
            best_seq = a_out.copy()
            print(f"  *** Phase 1 improved: {sc_check:.6f}")

    # === Phase 2: Softmax mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Softmax Mirror Descent (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 30 and best_seq is not None:
        a_out, _ = softmax_mirror_descent(best_seq, t2, T_init=8.0, eta0=0.06, decay=0.55, seed=7)
        sc_check = fast_score(a_out)
        if sc_check < best_score:
            best_score = sc_check
            best_seq = a_out.copy()
            print(f"  *** Phase 2 improved: {sc_check:.6f}")

    # === Phase 3: Targeted peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Targeted Peak Reduction ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20 and best_seq is not None:
        a_out, _ = targeted_peak_reduce(best_seq, t3, seed=33)
        sc_check = fast_score(a_out)
        if sc_check < best_score:
            best_score = sc_check
            best_seq = a_out.copy()
            print(f"  *** Phase 3 improved: {sc_check:.6f}")

    # === Phase 4: Pair swap local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Pair Swap Local Search ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20 and best_seq is not None:
        a_out, _ = pair_swap_local_search(best_seq, t4, seed=55)
        sc_check = fast_score(a_out)
        if sc_check < best_score:
            best_score = sc_check
            best_seq = a_out.copy()
            print(f"  *** Phase 4 improved: {sc_check:.6f}")

    # === Phase 5: Second LSE round (different regime) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE Round 2 ===")

            best_score = sc_check
            best_seq = a_out.copy()
            print(f"  *** Phase 5 improved: {sc_check:.6f}")

    # === Phase 6: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = int(round(len(best_seq) * 1.25))
    print(f"\n=== Phase 6: Longer length (n={n_long}) ===")
    t6 = min(remaining * 0.15, 220.0)
    if t6 > 40 and best_seq is not None:
        init_long = interpolate_to_length(best_seq, n_long)
        a_out, _ = lse_optimize(init_long, t6, T_start=200.0, T_end=0.3, n_stages=12, seed=29)
        sc_check = fast_score(a_out)

                    best_score = sc_check2

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Polish (remaining={remaining:.0f}s) ===")
    if best_seq is not None:
        # Short LSE at very low temperature
        t7a = min(remaining * 0.30, 120.0)
        if t7a > 15:
            a_out, _ = lse_optimize(best_seq, t7a, T_start=30.0, T_end=0.05, n_stages=8, seed=41)
            sc_check = fast_score(a_out)
            if sc_check < best_score:
                best_score = sc_check
                best_seq = a_out.copy()
                print(f"  *** Final LSE: {sc_check:.6f}")

        # Targeted peak reduction
        remaining2 = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining2 * 0.30, 100.0)
        if t7b > 15 and best_seq is not None:
            a_out, _ = targeted_peak_reduce(best_seq, t7b, seed=53)
            sc_check = fast_score(a_out)
            if sc_check < best_score:
                best_score = sc_check
                best_seq = a_out.copy()

        # Final pair swap
        remaining3 = time_budget - (time.monotonic() - t_global)
        t7c = min(remaining3 * 0.35, 100.0)

            if sc_check < best_score:
                best_score = sc_check
                best_seq = a_out.copy()
                print(f"  *** Final pair swap: {sc_check:.6f}")

    # Final evaluation with official function
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
