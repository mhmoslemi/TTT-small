def build_conv_matrix_rows(a_fixed, row_indices):
    """Build rows of convolution matrix for given row indices.
    Returns T where (T @ b)[i] = (a_fixed * b)[row_indices[i]]."""
    n = len(a_fixed)

        hi = min(n - 1, k)
        T[i, lo:hi+1] = a_fixed[lo:hi+1][::-1]
    return T

    n = len(a_fixed)

    # Actually compute conv of a_fixed with b_prev to find top K
    conv_pb = np.convolve(a_fixed, b_prev)
    top_k = np.argsort(conv_pb)[-K:]
    
    T = build_conv_matrix_rows(a_fixed, top_k)

    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:

                break

                continue

        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

    b_val = b_val / s
    return b_val, float(M_var.value)

        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    # Gradient w.r.t. a: sum_k w[k] * d(conv(a,a)[k])/da
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            grad_a[lo1:hi1+1] += 2.0 * w[k] * a[k - hi1:k - lo1 + 1][::-1]
    
    # Project to tangent space of simplex
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start=50.0, T_min=0.1, n_stages=30, iters_per_stage=50):
    """LogSumExp L-BFGS with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    T = T_start
    stagnant = 0
    prev_score = best_score
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func_cb(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj
        
        def jac_cb(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad
        
        try:
            res = minimize(
                fun=func_cb, x0=z, jac=jac_cb, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            
            # Adaptive temperature
            if prev_score is not None and sc > best_score * 1.001:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 1)
            prev_score = sc
            
            if stagnant >= 4:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)
                
        except Exception:
            T = max(T_min, T * 0.8)
    
    return best_a, best_score


def symmetric_constraint(a):
    """Enforce palindromic symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a[i]
        a_sym[n - 1 - i] = a[i]
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def welch_window(n):
    """Welch window: raised cosine."""
    x = np.arange(n)
    return 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))


def flat_top_window(n):
    """Flat-top window approximation."""
    x = np.arange(n)
    a1, a2, a3, a4, a5 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
    w = (a1 - a2 * np.cos(2 * np.pi * x / n) + a3 * np.cos(4 * np.pi * x / n) 
         - a4 * np.cos(6 * np.pi * x / n) + a5 * np.cos(8 * np.pi * x / n))
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    return w


def gaussian_window(n, sigma_frac=0.3):
    """Gaussian window."""
    x = np.linspace(-1, 1, n)
    w = np.exp(-x**2 / (2 * sigma_frac**2))
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    return w


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


_lp_solver_global = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver_global
    if _lp_solver_global is not None:
        return _lp_solver_global
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_global = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_global = cp.ECOS
    return _lp_solver_global


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: hybrid optimization with multi-length exploration."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    global_deadline = t_start + max(1.0, float(budget_s) - 20.0)
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s")
    
    # Get starting sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = welch_window(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default Welch init: n={n_base}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")
    
    def remaining_time():
        return max(0, global_deadline - time.monotonic())
    
    # === Phase 1: Intensive hybrid Nesterov at base length (40% time) ===
    print(f"\n=== Phase 1: Hybrid Nesterov LP at n={n_base} ({best_score:.8f}) ===")
    t1 = 0.40 * (budget_s - 20)
    a_out, sc_out = nesterov_lp_topk(best_seq, t1, K=150, lp_every=3, frac=0.99, solver=solver)
    try_update(a_out, "hybrid_nesterov_base")
    
    # Subgradient polish
    remaining = remaining_time()
    if remaining > 10:
        a_sg, _ = nesterov_subgradient(best_seq, remaining * 0.5, frac=0.99)
        try_update(a_sg, "subgrad_polish")
    print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: LSE polishing (15% time) ===
    print(f"\n=== Phase 2: LSE Polishing ({best_score:.8f}) ===")
    remaining = remaining_time()
    t2 = min(0.15 * (budget_s - 20), remaining * 0.4)
    if t2 > 15:
        a_lse, _ = lse_lbfgs(best_seq, t2, T_start=30.0, T_min=0.05, n_stages=25, iters_per_stage=40)
        try_update(a_lse, "lse_polish_1")
        
        remaining = remaining_time()
        if remaining > 10:
            a_lse2, _ = lse_lbfgs(best_seq, remaining * 0.4, T_start=5.0, T_min=0.01, n_stages=15, iters_per_stage=30)
            try_update(a_lse2, "lse_polish_2")
    print(f"  Phase 2 best: {best_score:.8f}")
    
    # === Phase 3: Multi-length exploration (30% time) ===
    print(f"\n=== Phase 3: Multi-length exploration ({best_score:.8f}) ===")
    remaining = remaining_time()
    t3_total = min(0.30 * (budget_s - 20), remaining * 0.6)
    
    if t3_total > 30:
        # Try longer sequences
        lengths = [n_base + 300, n_base + 600, n_base + 1000, n_base + 1500]
        # Also try shorter
        if n_base > 800:
            lengths.append(n_base - 200)
        
        per_length = t3_total / len(lengths)
        
        for n_try in lengths:
            remaining = remaining_time()
            if remaining < 15 or per_length < 10:
                break
            
            t_alloc = min(per_length, remaining * 0.9)
            print(f"  Trying n={n_try}, budget={t_alloc:.0f}s")
            
            # Interpolate from current best
            init_try = interpolate_sequence(best_seq, n_try)
            
            # Hybrid Nesterov
            a_try, _ = nesterov_lp_topk(init_try, t_alloc * 0.6, K=100, lp_every=3, frac=0.99, solver=solver)
            try_update(a_try, f"len_{n_try}_nesterov")
            
            remaining = remaining_time()
            if remaining > 10:
                # Quick LSE polish
                a_try2, _ = lse_lbfgs(a_try, remaining * 0.3, T_start=20.0, T_min=0.1, n_stages=10, iters_per_stage=25)
                try_update(a_try2, f"len_{n_try}_lse")
    
    print(f"  Phase 3 best: {best_score:.8f}")
    
    # === Phase 4: Symmetric optimization (10% time) ===
    print(f"\n=== Phase 4: Symmetric optimization ({best_score:.8f}) ===")
    remaining = remaining_time()
    t4 = min(0.10 * (budget_s - 20), remaining * 0.4)
    if t4 > 15:
        a_sym = symmetric_constraint(best_seq)
        # Run subgradient on symmetric sequence
        a_sg_sym, _ = nesterov_subgradient(a_sym, t4 * 0.5, frac=0.99)
        a_sg_sym = symmetric_constraint(a_sg_sym)
        try_update(a_sg_sym, "sym_subgrad")
        
        remaining = remaining_time()
        if remaining > 10:
            a_sym_lse, _ = lse_lbfgs(a_sg_sym, remaining * 0.4, T_start=10.0, T_min=0.05, n_stages=12, iters_per_stage=25)
            a_sym_lse = symmetric_constraint(a_sym_lse)
            try_update(a_sym_lse, "sym_lse")
    
    print(f"  Phase 4 best: {best_score:.8f}")
    
    # === Phase 5: Alternative initializations (best remaining time) ===

    remaining = remaining_time()
    if remaining > 20:
        n_try = len(best_seq)
        inits = [

            ("welch", welch_window(n_try)),
            ("gauss", gaussian_window(n_try, 0.25)),
        ]
        
        per_init = remaining * 0.6 / len(inits)
        for name, init in inits:
            remaining = remaining_time()
            if remaining < 10:

            t_alloc = min(per_init, remaining * 0.8)
            print(f"  Trying {name} init, budget={t_alloc:.0f}s")
            
            a_alt, _ = nesterov_lp_topk(init, t_alloc * 0.7, K=100, lp_every=2, frac=0.99, solver=solver)
            try_update(a_alt, f"alt_{name}")
            
            remaining = remaining_time()
            if remaining > 8:
                a_alt2, _ = lse_lbfgs(a_alt, remaining * 0.4, T_start=15.0, T_min=0.1, n_stages=8, iters_per_stage=20)

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final polish ({best_score:.8f}) ===")
    remaining = remaining_time()
    if remaining > 10:
        # Final Nesterov
        a_final, _ = nesterov_lp_topk(best_seq, remaining * 0.5, K=200, lp_every=2, frac=0.995, solver=solver)

        remaining = remaining_time()
        if remaining > 8:
            # Final LSE with very low temperature
            a_final2, _ = lse_lbfgs(best_seq, remaining * 0.6, T_start=3.0, T_min=0.01, n_stages=12, iters_per_stage=30)
            try_update(a_final2, "final_lse")
        
        remaining = remaining_time()
        if remaining > 5:
            # Final subgradient
            a_final3, _ = nesterov_subgradient(best_seq, remaining * 0.7, frac=0.999)
            try_update(a_final3, "final_subgrad")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
