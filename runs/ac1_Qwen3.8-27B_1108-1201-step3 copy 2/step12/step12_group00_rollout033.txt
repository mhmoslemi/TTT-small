from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute linear autocorrelation (convolution of a with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Compute the objective value for a non-negative sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    max_b = fast_autocorr(a).max()
    return float(2.0 * n * max_b / (s * s))


def normalize_to_sum(a, target=1.0):
    """Normalize a to have the given sum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a * target / s


def subgradient_of_max_conv(a):
    """Compute subgradient of max_k (a*a)[k] with respect to a."""
    conv = fast_autocorr(a)
    max_val = conv.max()
    # Find argmax positions
    tol = max_val * 1e-10 + 1e-15
    argmax_positions = np.where(conv >= max_val - tol)[0]
    
    n = len(a)
    grad = np.zeros(n)
    for k in argmax_positions:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    return grad, max_val


def subgradient_of_max_conv_fast(a):
    """Faster subgradient computation using FFT."""
    conv = fast_autocorr(a)
    max_val = conv.max()
    tol = max_val * 1e-10 + 1e-15
    argmax_mask = (conv >= max_val - tol).astype(np.float64)
    
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1], fft_size)
    W = rfft(argmax_mask, fft_size)
    conv_out = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv_out[n - 1:n - 1 + n]
    return grad, max_val, argmax_mask.sum()


def build_convolution_matrix(a):
    """Build the dense convolution matrix M where (M@b)[k] = (a*b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    M = np.zeros((num_rows, n))
    for k in range(num_rows):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            M[k, idx] = a[k - idx]
    return M


def lp_best_response(a, M_val, solver=None):
    """Solve LP: max sum(b) s.t. conv_matrix(a) @ b <= M_val, b >= 0."""
    if not HAS_CVXPY:
        return None, 0.0
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_mat = build_convolution_matrix(a)
    constraints = [M_mat @ b_var <= M_val]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        if solver is not None:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return None, 0.0
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, 0.0
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    dual_val = prob.value if prob.value is not None else 0.0
    return b_val, float(dual_val)

def katyla_momentum(theta_prev, s_curr, s_prev, x_param=0.5):
    """Update Katyla momentum parameter."""
    norm_curr = np.linalg.norm(s_curr)
    norm_prev = np.linalg.norm(s_prev)
    if norm_curr < 1e-14 or norm_prev < 1e-14:
        return 0.9 * theta_prev
    cos_sim = np.dot(s_curr, s_prev) / (norm_curr * norm_prev)
    cos_sim = np.clip(cos_sim, -1.0, 1.0)
    theta_new = theta_prev - x_param * cos_sim
    theta_new = max(0.0, θ_new) if False else max(0.0, theta_new)
    return min(theta_new, 0.9)

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g

        if s_y < 1e-15:

        else:

        # Away step: move away from the largest component
        max_idx = int(np.argmax(y))

            diff = g - y_new
            away_dir = y - np.zeros(n)

def lse_gradient_descent(a0, time_limit, T_start=20.0, T_end=0.1, n_stages=8, verbose=True):
    """
    Gradient descent on the LogSumExp smoothing of max convolution.
    With Nesterov acceleration.
    """
    n = len(a0)
    a = normalize_to_sum(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    temperatures = np.geomspace(T_start, T_end, n_stages)
    
    y = a.copy()
    x = a.copy()
    s_prev = np.zeros(n)
    theta = 0.0
    global_t = 0
    
    for si, T in enumerate(temperatures):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Lipschitz constant estimate
        L = 4.0 * T * n + 4.0
        step_base = 1.0 / L
        
        stage_time = time_limit / (len(temperatures) - si)
        n_steps = max(50, min(5000, int(stage_time * 200)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            global_t += 1
            step = step_base / (1.0 + 0.05 * it)
            
            # Compute LSE gradient
            conv = fast_autocorr(x)
            max_conv = conv.max()
            exp_conv = np.exp(T * (conv - max_conv))
            w = exp_conv / exp_conv.sum()
            
            # Gradient: 2 * sum_k w[k] * a[k-i]
            fft_size = 1 << (3 * n - 2).bit_length()
            A_rev = rfft(a[::-1], fft_size)
            W = rfft(w, fft_size)
            conv_grad = irfft(A_rev * W, fft_size)
            grad = 2.0 * conv_grad[n - 1:n - 1 + n]
            
            # Gradient step with projection
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_a
            
            # Nesterov/Katyla
            s_curr = a_new - a
            if global_t > 1:
                theta = katyla_momentum(theta, s_curr, s_prev, 0.4)
            s_prev = s_curr.copy()
            
            x_new = a_new + theta * s_curr
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = a_new.copy()
            else:
                x_new = x_new / s_x
            
            a = a_new
            x = x_new
            
            val = fast_autocorr(a).max()

        if verbose:

    return best_a, best_val


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """
    Peak-aware coordinate descent: find the element contributing most to
    the peak convolution value and reduce it, redistributing to elements
    that contribute least.
    """
    n = len(a0)
    a = normalize_to_sum(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        max_val = conv.max()
        
        if max_val < best_val - 1e-13:
            best_val = max_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find peak lags
        top_k = np.argsort(conv)[::-1][:5]
        
        improved = False
        for k in top_k:
            # Find which pair (i, j) with i+j=k contributes most
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            pairs = idx * (k - idx)
            src = idx[np.argmax(pairs)]
            
            # Find destination: position that contributes least to all top peaks
            contrib_to_peaks = np.zeros(n)
            for k2 in top_k:
                j_lo = max(0, k2 - n + 1)
                j_hi = min(k2, n - 1)
                if j_lo <= j_hi:
                    jdx = np.arange(j_lo, j_hi + 1)
                    contrib_to_peaks[jdx] += a[jdx] * a[k2 - jdx]
            
            # Pick position with lowest contribution and lowest value
            combined = contrib_to_peaks + 0.1 * a
            dst_candidates = np.argsort(combined)[:max(5, n // 10)]
            dst = dst_candidates[rng.integers(len(dst_candidates))]
            
            if dst == src:
                continue
            
            # Try transferring
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = normalize_to_sum(a_trial, 1.0)
                new_conv = fast_autocorr(a_trial)
                new_max = new_conv.max()
                if new_max < max_val - 1e-14:
                    a = a_trial
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-6:
                amt = a[i] * 0.02
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = normalize_to_sum(a_trial, 1.0)
                new_max = fast_autocorr(a_trial).max()
                if new_max < max_val:
                    a = a_trial
                    improved = True
        
        if not improved:
            stall += 1
            if stall > 2000:
                break
        
        if verbose and (stall == 0 or stall % 100 == 0):
            elapsed = time.monotonic() - t0
            print(f"    CD polish ({elapsed:.0f}s): val={best_val:.10f}, stall={stall}")
    
    return best_a, best_val


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, seed=0):
    """Generate structured initial sequences."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2.0 * np.pi * x) + 0.15 * np.cos(4.0 * np.pi * x) + 0.05 * np.cos(8.0 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.2) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.5) ** 2) + np.exp(-6.0 * (x - 0.8) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    else:
        # Bimodal: high at edges
        a = np.exp(-20.0 * (x - 0.1) ** 2) + np.exp(-20.0 * (x - 0.9) ** 2) + 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def load_parent_sequence():
    """Try to load the parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Returns the best sequence found within the time budget.
    Uses a multi-phase approach with LP Frank-Wolfe, subgradient methods,
    LSE smoothing, and coordinate descent polishing.
    """
    total_start = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    rng = np.random.default_rng(seed)
    solver = find_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    # Load parent
    parent = load_parent_sequence()
    if parent is not None:
        best_seq = parent.copy()
        best_score = evaluate_local(parent)
        print(f"Parent loaded: n={len(parent)}, score={best_score:.8f}")
    else:
        n_init = 1600
        best_seq = structured_init(n_init, seed)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_local(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - total_start
            print(f"  *** NEW BEST [{label}]: score={sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    n_base = len(best_seq)
    
    # Phase 1: LP Frank-Wolfe at base length
    remaining = time_budget - (time.monotonic() - total_start)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LP FW n={n_base} (budget={t1:.0f}s) ===")
        a_init = normalize_to_sum(best_seq, 1.0)
        a_out, _ = lp_frank_wolfe(a_init, t1, solver, verbose=True)
        try_update(a_out, f"lpfw_n{n_base}")
    
    # Phase 2: Subgradient FW at base length (faster, complementary)
    remaining = time_budget - (time.monotonic() - total_start)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 10:
        print(f"\n=== Phase 2: SubGrad FW n={len(best_seq)} (budget={t2:.0f}s) ===")
        a_init = normalize_to_sum(best_seq, 1.0)
        a_out, _ = subgrad_frank_wolfe(a_init, t2, verbose=True)
        try_update(a_out, f"subfw_n{len(best_seq)}")
    
    # Phase 3: LP FW at longer length n2
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - total_start)
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP FW n={n2} (budget={t3:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_out, _ = lp_frank_wolfe(init_long, t3, solver, verbose=True)
        try_update(a_out, f"lpfw_n{n2}")
    
    # Phase 4: Subgradient FW at longer length
    remaining = time_budget - (time.monotonic() - total_start)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        print(f"\n=== Phase 4: SubGrad FW n={n_cur} (budget={t4:.0f}s) ===")
        a_init = normalize_to_sum(best_seq, 1.0)
        a_out, _ = subgrad_frank_wolfe(a_init, t4, verbose=True)
        try_update(a_out, f"subfw_n{n_cur}")
    
    # Phase 5: LSE gradient descent
    remaining = time_budget - (time.monotonic() - total_start)
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 15:
        print(f"\n=== Phase 5: LSE GD n={len(best_seq)} (budget={t5:.0f}s) ===")
        a_init = normalize_to_sum(best_seq, 1.0)
        a_out, _ = lse_gradient_descent(a_init, t5, T_start=15.0, T_end=0.05, n_stages=10, verbose=True)
        try_update(a_out, f"lse_n{len(best_seq)}")
    
    # Phase 6: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - total_start)
    t6 = min(remaining * 0.30, 200.0)
    if t6 > 5:
        print(f"\n=== Phase 6: CD Polish n={len(best_seq)} (budget={t6:.0f}s) ===")
        a_init = normalize_to_sum(best_seq, 1.0)
        a_out, _ = coordinate_descent_polish(a_init, t6, verbose=True)
        try_update(a_out, "cd_polish")
    
    # Phase 7: Try another length if time remains
    remaining = time_budget - (time.monotonic() - total_start)
    n3 = len(best_seq) + 400
    t7 = min(remaining * 0.5, 200.0)
    if t7 > 30 and remaining > 60:
        print(f"\n=== Phase 7: LP FW n={n3} (budget={t7:.0f}s) ===")
        init_3 = interpolate_sequence(best_seq, n3)
        a_out, _ = lp_frank_wolfe(init_3, t7, solver, verbose=True)
        try_update(a_out, f"lpfw_n{n3}")
    
    # Phase 8: Final coordinate descent
    remaining = time_budget - (time.monotonic() - total_start)
    t8 = min(remaining * 0.8, 100.0)
    if t8 > 5:
        print(f"\n=== Phase 8: Final CD n={len(best_seq)} (budget={t8:.0f}s) ===")
        a_init = normalize_to_sum(best_seq, 1.0)
        a_out, _ = coordinate_descent_polish(a_init, t8, verbose=True)
        try_update(a_out, "cd_final")
    
    # Final validation
    if best_seq is not None:
        final_score = evaluate_local(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-total_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    # Return as list of floats
    result = [float(x) for x in best_seq]
    # Safety: ensure no negatives
    result = [max(0.0, x) for x in result]
    return result
