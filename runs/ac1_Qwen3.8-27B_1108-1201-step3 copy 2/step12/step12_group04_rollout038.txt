def fast_autocorr(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score(a):
    """Compute 2n*max(a*a)/(sum a)^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)

def grad_l2_deviation(a):
    """
    Gradient of sum_k ((a*a)_k - mean)^2 w.r.t. a.
    Using FFT for O(n log n) computation.
    """
    n = len(a)
    c = fast_autocorr(a)
    c_mean = c.mean()  # = (sum a)^2 / (2n-1)
    d = c - c_mean  # length 2n-1
    
    a_rev = a[::-1]
    # Linear convolution of d (length 2n-1) and a_rev (length n)
    conv_len = (2*n-1) + n - 1  # = 3n-1
    fft_size = 1 << (conv_len - 1).bit_length()
    D = rfft(d, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(D * AR, fft_size)[:conv_len]
    
    # grad_j = 2 * conv[j + n - 1] for j = 0,...,n-1
    return 2.0 * conv[n-1:2*n-1]

def grad_lse(a, T):
    """
    Gradient of (1/T) * log(sum_k exp(T*(a*a)_k)) w.r.t. a.
    """
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    w = exp_Tc / exp_Tc.sum()  # softmax weights
    
    # grad_j = 2 * sum_k w[k] * a_{k-j} = 2 * (w_star_a)_j
    # Same computation as before but with w instead of d
    a_rev = a[::-1]
    conv_len = (2*n-1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)[:conv_len]
    
    return 2.0 * conv[n-1:2*n-1]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    i_star = idx[-1]
    theta = cssv[i_star] / (i_star + 1)
    return np.maximum(v - theta, 0)

def momentum_gradient_descent(a0, time_limit, method='l2', T=100.0, lr_init=0.01, 
                                momentum=0.9, max_iters=50000):
    """
    Momentum gradient descent on the simplex.
    method: 'l2' for L2 deviation, 'lse' for log-sum-exp
    """
    n = len(a0)
    a = project_simplex(np.maximum(np.asarray(a0, float), 0))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)  # momentum velocity
    best_a = a.copy()
    best_score = score(a)
    
    t0 = time.monotonic()
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        if method == 'l2':
            g = grad_l2_deviation(a)
        else:
            g = grad_lse(a, T)
        
        # Normalize gradient
        g_norm = np.sqrt(np.sum(g**2))
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Momentum update (in the "unconstrained" space)
        v = momentum * v - lr * g
        a_new = a + v
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        a_new = a_new / a_new.sum()
        
        # Adaptive learning rate: reduce if score increases
        new_score = score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            # Maybe increase lr slightly

            v *= 0.5
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MGD {method} it={it} ({elapsed:.0f}s): score={best_score:.8f}")

def lse_l_bfgs(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=10):
    """
    L-BFGS optimization of the log-sum-exp objective with adaptive temperature.

    a = np.maximum(np.asarray(a0, float), 0)
    s = a.sum()
    a = a / s

    z = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        def func(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            c = fast_autocorr(a_in)
            Tc = T * c
            return (Tc.max() + np.log(np.exp(Tc - Tc.max()).sum())) / T

            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            g_a = grad_lse(a_in, T)

            return a_in * (g_a - S)

            z = res.x
            a = np.exp(z - z.max())
            a = a / a.sum()
            sc = score(a)

                best_score = sc
                best_a = a.copy()
        except Exception:
            pass
        
        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si} T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    d = c - c_mean  # length 2n-1

    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1  # 3n-1
    fft_size = 1 << (conv_len - 1).bit_length()
    D = rfft(d, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(D * AR, fft_size)[:conv_len]

    return 2.0 * conv[n - 1:2 * n - 1]


def grad_lse(a, T):
    """
    Gradient of (1/T)*log(sum_k exp(T*(a*a)_k)) w.r.t. a.
    O(n log n) via FFT.
    """
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    w = exp_Tc / exp_Tc.sum()

    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)[:conv_len]

    return 2.0 * conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    i_star = idx[-1]
    theta = cssv[i_star] / (i_star + 1.0)
    return np.maximum(v - theta, 0.0)


def l2_momentum_descent(a0, time_limit, n_stages=5):
    """
    Multi-stage momentum gradient descent minimizing L2 deviation of convolution from mean.
    Each stage uses a different momentum and learning rate schedule.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Multi-stage: start with larger steps, then refine
    stage_configs = [
        {'lr': 0.05, 'mom': 0.9, 'iters': 3000},
        {'lr': 0.02, 'mom': 0.9, 'iters': 5000},
        {'lr': 0.01, 'mom': 0.95, 'iters': 8000},
        {'lr': 0.005, 'mom': 0.95, 'iters': 10000},
        {'lr': 0.002, 'mom': 0.97, 'iters': 15000},
    ]

    v = np.zeros(n)
    total_iters = sum(cfg['iters'] for cfg in stage_configs[:n_stages])

    for si in range(min(n_stages, len(stage_configs))):
        if time.monotonic() - t0 > time_limit:
            break
        cfg = stage_configs[si]
        lr = cfg['lr']
        mom = cfg['mom']
        max_iter = cfg['iters']

        # Dynamic time allocation: more iterations if we have time
        remaining = time_limit - (time.monotonic() - t0)
        est_time_per_iter = 0.005  # rough estimate for n~1600
        max_iter = min(max_iter, int(remaining / max(est_time_per_iter, 0.001)))
        max_iter = max(max_iter, 100)

        for it in range(max_iter):
            if time.monotonic() - t0 > time_limit:
                break

            g = grad_l2_deviation(a)
            g_norm = np.sqrt(np.sum(g * g))
            if g_norm > 1e-15:
                g = g / g_norm

            v = mom * v - lr * g
            a_new = a + v
            a_new = project_simplex(a_new)

            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                # Line search: try smaller step
                v = 0.9 * v
                a_new = a + 0.5 * v
                a_new = project_simplex(a_new)
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                else:
                    v = 0.5 * v
                    lr *= 0.9

            if it % 2000 == 0 and si > 0:
                elapsed = time.monotonic() - t0
                print(f"      L2MD stage={si} it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lse_lbfgs_optimize(a0, time_limit, T_start=300.0, T_end=0.5, n_stages=12):
    """
    Multi-stage L-BFGS on log-sum-exp of convolution in log-parametrization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    z = np.log(np.maximum(a, 1e-30))

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        max_iter = max(20, min(80, int(30 + 50 * si / max(1, n_stages - 1))))

        def func(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            c = fast_autocorr(a_in)
            Tc = T * c
            return (Tc.max() + np.log(np.exp(Tc - Tc.max()).sum())) / T

        def jac(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            g_a = grad_lse(a_in, T)
            S = np.dot(g_a, a_in)
            return a_in * (g_a - S)

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': max_iter * 200,
                                    'disp': False})
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=3000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Pre-build matrix if n is not too large
    if n <= 2500:
        T_cache = {}

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        T = build_conv_matrix(y)
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        prob = cp.Problem(cp.Minimize(M_var), [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1
        ])
        try:
            prob.solve(solver=solver)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_star = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
        bs = b_star.sum()
        b_star = b_star / bs if bs > 1e-15 else np.ones(n) / n

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_star
        y_new = project_simplex(y_new)

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Check average iterate periodically
        if t % 200 == 0:
            avg = project_simplex((a + y) / 2.0)
            sc_avg = compute_score(avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg.copy()
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={float(M_var.value):.10f}")

    return best_a, best_score


def coord_descent_peak_shave(a0, time_limit, seed=42):
    """Peak-aware coordinate descent for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        top_peaks = np.argsort(c)[::-1][:6]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            top_contrib = np.argsort(contrib)[::-1][:4]
            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Find target with minimal peak contribution
                peak_positions = set()
                for k2 in top_peaks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for j2 in range(js2, je2 + 1):
                        peak_positions.add(j2)

                candidates = [j for j in range(n) if j not in peak_positions and j != src]
                if not candidates:
                    candidates = [j for j in range(n) if j != src]

                # Score and pick best candidate
                best_dst = None
                best_score_dst = float('inf')
                for j in candidates[:30]:
                    total = 0.0
                    for k2 in top_peaks:
                        if max(0, k2 - n + 1) <= j <= min(n - 1, k2):
                            partner = k2 - j
                            if 0 <= partner < n:
                                total += a[j] * a[partner]
                    total += 0.005 * a[j]
                    if total < best_score_dst:
                        best_score_dst = total
                        best_dst = j

                if best_dst is None:
                    continue
                dst = best_dst

                for frac in [0.005, 0.002, 0.01, 0.02, 0.001]:
                    amt = src_val * frac
                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[dst] += amt
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    sc = compute_score(a_try)
                    if sc < best_score - 1e-14:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbations
            for _ in range(8):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                sc = compute_score(a_try)
                if sc < best_score:
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                    break
            stall += 1
            if stall > 800:
                break

        if stall > 0 and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2 + 0.1 * np.sin(2 * np.pi * x) ** 2
    elif choice == 3:
        a = 1.0 / (1.0 + 8 * (x - 0.3) ** 2) + 0.5 / (1.0 + 8 * (x - 0.7) ** 2)
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    elif choice == 5:
        a = (1 - x) * (0.5 + 0.5 * np.sin(8 * np.pi * x))
    elif choice == 6:
        # Two-gaussian
        a = np.exp(-6 * (x - 0.35) ** 2) + 0.7 * np.exp(-6 * (x - 0.65) ** 2)
    else:
        a = np.ones(n)
        a += 0.2 * rng.random(n)

    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def interpolate_sequence(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    return a_interp / s if s > 1e-15 else np.ones(new_n) / new_n


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate():
    """
    Main search: multi-phase optimization with multi-length exploration.
    Phase 1: L2-deviation momentum descent (fast, smooth)
    Phase 2: LogSumExp L-BFGS (peak-targeted)
    Phase 3: Nesterov LP (game-theoretic)
    Phase 4: Coordinate descent (final polish)
    Multi-length: try a few different n values.
    """
    t_global = time.monotonic()
    time_budget = 1740.0  # leave 60s safety margin

    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # --- Get parent sequence ---
    parent = None
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent.")

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # --- Try multiple initializations and pick best ---
    rng = np.random.default_rng(12345)
    inits = [best_seq.copy()]
    for i in range(6):
        inits.append(structured_init(n_base, 100 + i * 7))

    # Evaluate all inits
    init_scores = []
    for i, init_a in enumerate(inits):
        sc = compute_score(init_a)
        init_scores.append((sc, i))
        print(f"  Init {i}: score={sc:.8f}")
    init_scores.sort()
    print(f"  Best init: {init_scores[0][0]:.8f}")

    # --- Optimize the best few initializations ---
    n_good_inits = min(3, len(inits))
    time_per_init = time_budget * 0.6 / n_good_inits

    for idx in range(n_good_inits):
        _, init_idx = init_scores[idx]
        a_init = inits[init_idx]
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        tl = min(time_per_init, remaining * 0.45)

        print(f"\n  --- Optimizing init {init_idx} (score={compute_score(a_init):.8f}, time={tl:.0f}s) ---")

        # Phase 1: L2 momentum descent
        t1 = tl * 0.4
        if t1 > 10:
            a_out, _ = l2_momentum_descent(a_init, t1, n_stages=5)
            try_update(a_out, f"L2MD_i{init_idx}")
            a_init = best_seq if best_seq is not None else a_init

        # Phase 2: LSE L-BFGS
        remaining = time_budget - (time.monotonic() - t_global)
        t2 = min(tl * 0.3, remaining * 0.25)
        if t2 > 10:
            a_out, _ = lse_lbfgs_optimize(a_init, t2, T_start=200.0, T_end=0.5, n_stages=10)
            try_update(a_out, f"LSE_i{init_idx}")

        # Phase 3: Nesterov LP
        remaining = time_budget - (time.monotonic() - t_global)
        t3 = min(tl * 0.2, remaining * 0.2)
        if t3 > 20 and n_base <= 2000:
            a_out, _ = nesterov_lp_optimize(a_init, t3, solver, max_iters=2000)
            try_update(a_out, f"NLP_i{init_idx}")

    # --- Multi-length exploration ---
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Multi-length exploration (remaining={remaining:.0f}s) ===")

    n_configs = [n_base - 200, n_base + 200, n_base + 400]
    n_configs = [x for x in n_configs if 800 <= x <= 3000]
    n_configs = sorted(set(n_configs), key=lambda x: abs(x - n_base))

    for nl in n_configs:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        tl = min(remaining * 0.25, 200.0)
        if tl < 20:
            continue

        print(f"  Trying n={nl} (time={tl:.0f}s)")
        a_interp = interpolate_sequence(best_seq, nl)

        # Quick L2 descent
        t1 = tl * 0.5
        a_out, _ = l2_momentum_descent(a_interp, t1, n_stages=4)
        try_update(a_out, f"L2MD_n{nl}")

        # Quick LSE
        remaining = time_budget - (time.monotonic() - t_global)
        t2 = min(tl * 0.3, remaining * 0.15)
        if t2 > 10:
            a_out2, _ = lse_lbfgs_optimize(a_out, t2, T_start=100.0, T_end=1.0, n_stages=8)
            try_update(a_out2, f"LSE_n{nl}")

    # --- Final refinement on best sequence ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Final refinement at n={n_best} (remaining={remaining:.0f}s) ===")

    # Coordinate descent
    t_cd = min(remaining * 0.35, 200.0)
    if t_cd > 10:
        a_out, _ = coord_descent_peak_shave(best_seq, t_cd, seed=999)
        try_update(a_out, "CD_final")

    # More LSE with very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse = min(remaining * 0.3, 150.0)
    if t_lse > 10:
        a_out, _ = lse_lbfgs_optimize(best_seq, t_lse, T_start=50.0, T_end=0.1, n_stages=10)
        try_update(a_out, "LSE_refine")

    # Another L2 pass
    remaining = time_budget - (time.monotonic() - t_global)
    t_l2 = min(remaining * 0.25, 100.0)
    if t_l2 > 10:
        a_out, _ = l2_momentum_descent(best_seq, t_l2, n_stages=3)
        try_update(a_out, "L2MD_refine")

    # Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    t_cd2 = min(remaining * 0.3, 80.0)
    if t_cd2 > 5:
        a_out, _ = coord_descent_peak_shave(best_seq, t_cd2, seed=777)
        try_update(a_out, "CD_final2")

    # --- Final verification ---
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
