Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Nesterov-accelerated mirror descent on LSE-smoothed autocorrelation peak
2. Coordinate descent with analytic candidate evaluation for final polish
3. Multi-length exploration (n=1600, 2000, 2400, 3000)
4. Multiplicative (exponential) updates staying in positive orthant naturally
Key: much faster per-iteration than LP-based approaches, enabling thousands of
steps in the time budget for fine-grained improvement.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation using FFT. Input length n, output length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_weights(a, T):
    """Compute softmax weights for LSE smoothing of autocorrelation."""
    c = fast_autocorr(a)
    Tc = T * c
    Tc_shifted = Tc - Tc.max()
    w = np.exp(Tc_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    return w, c


def compute_lse_grad(a, w):
    """
    Compute gradient of LSE(T*conv(a,a)) w.r.t. a.
    w: softmax weights of length 2n-1
    grad[j] = 2 * sum_k w[k] * a[k-j]  (a[m]=0 outside [0,n-1])
    Using: grad[j] = 2 * sum_m a[m] * w[j+m]  (m=0..n-1, j+m in [0,2n-2])
    This is np.correlate(w, a, 'valid') * 2
    """
    v = np.correlate(w, a, mode='valid')
    return 2.0 * v


def nesterov_mirror_descent(a0, time_limit, T_start=200.0, T_end=5.0,
                            n_anneal_stages=8, verbose=True):
    """
    Nesterov-accelerated mirror descent (KL-based) on LSE-smoothed objective.
    Minimizes max_k conv(a,a)_k subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()  # Nesterov extrapolation point
    iter_count = 0
    no_improve = 0

    temps = np.geomspace(T_start, T_end, n_anneal_stages)

    for stage, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        stage_start = time.monotonic()
        stage_budget = time_limit - (time.monotonic() - t0)

        while time.monotonic() - t0 < time_limit:
            # Compute LSE weights and gradient at y
            w, c_y = compute_lse_weights(y, T)
            grad = compute_lse_grad(y, w)

            # Mirror descent step: a_new proportional to y * exp(-eta * grad)
            # Use approximation: a_new = y / (1 + eta * grad) for stability
            # Step size: eta = c / sqrt(t)
            eta = 0.5 / (1.0 + 0.1 * iter_count ** 0.5)

            # Cap the exponent to avoid overflow
            eta_grad = eta * grad
            eta_grad = np.clip(eta_grad, -5.0, 5.0)

            a_new = y * np.exp(-eta_grad)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
                s_new = 1.0
            else:
                a_new = a_new / s_new

            iter_count += 1

            # Nesterov momentum update
            momentum = (iter_count - 1.0) / (iter_count + 2.0)
            y_new = a_new + momentum * (a_new - a)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / s_y

            a = a_new
            y = y_new

            # Track best
            max_val = fast_autocorr(a).max()
            if max_val < best_max:
                best_max = max_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Check y as well
            max_val_y = fast_autocorr(y).max()
            if max_val_y < best_max:
                best_max = max_val_y
                best_a = y.copy()
                no_improve = 0

            # Periodic check of Cesaro average
            if iter_count % 500 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"      NM iter={iter_count} T={T:.1f} ({elapsed:.0f}s): best_max={best_max:.10f}")

        if verbose and stage < n_anneal_stages - 1:
            elapsed = time.monotonic() - t0
            print(f"    NM stage {stage+1}/{n_anneal_stages} (T={T:.1f}): best={best_max:.10f} ({elapsed:.0f}s)")

    return best_a, best_max


def coordinate_descent(a0, time_limit, n_sweeps_max=50000, verbose=True):
    """
    Coordinate descent: for each coordinate, try several candidate values
    and pick the one minimizing max autocorrelation (with sum=1 constraint).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    sweep = 0
    stall = 0

    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and sweep < n_sweeps_max:
        c = fast_autocorr(a)  # current autocorrelation
        current_max = c.max()
        improvement = 0.0

        # Process coordinates in random order to avoid bias
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_x = a[j]
            if old_x < 1e-12:
                continue

            # Compute c_base = c - contribution of a[j]
            # Contribution of a[j] to conv[k]:
            #   if 0 <= k-j < n and k != 2j: 2*old_x*a[k-j]
            #   if k == 2j: old_x^2
            #   else: 0

            c_base = c.copy()
            # Subtract linear contributions
            m_start = 0
            m_end = n  # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
            k_start = max(0, j)
            k_end = min(2 * n - 2, j + n - 1)

            # For k in [k_start, k_end], k-j is in [0, n-1]
            # Contribution is 2*old_x*a[k-j] except when k=2j where it's old_x^2
            k_valid = np.arange(k_start, k_end + 1)
            m_valid = k_valid - j  # these are in [0, n-1]

            # Subtract 2*old_x*a[m_valid] from c_base
            c_base[k_valid] -= 2.0 * old_x * a[m_valid]
            # Add back old_x^2 at k=2j (since we subtracted 2*old_x*old_x = 2*old_x^2 but should subtract only old_x^2)
            if 0 <= 2 * j <= 2 * n - 2:
                c_base[2 * j] += old_x * old_x

            # Now c_base[k] = conv(a_without_j, a_without_j)[k]
            # For candidate x, new conv[k] = c_base[k] + delta_k(x)
            # delta_k(x) = 2*x*a[k-j] for k in valid range, k != 2j
            #             x^2 for k = 2j
            #             0 otherwise

            # Candidate x values
            candidates = [0.0, old_x * 0.1, old_x * 0.3, old_x * 0.6, old_x, old_x * 1.4]
            # Also add the x that makes the binding constraint tight
            # Find the k where c_base[k] + 2*old_x*a[k-j] is max (the binding constraint)
            if len(k_valid) > 0:
                vals = c_base[k_valid] + 2.0 * old_x * a[m_valid]
                k_bind = k_valid[np.argmax(vals)]
                a_bind = a[m_valid[np.argmax(vals)]]
                c_bind = c_base[k_bind]
                if a_bind > 1e-15:
                    x_tight = max(0.0, (current_max - c_bind) / (2.0 * a_bind))
                    candidates.append(x_tight)
                    candidates.append(x_tight * 0.5)

            # Evaluate candidates
            best_obj = 2.0 * n * current_max  # current objective (sum=1)
            best_x = old_x

            for x in candidates:
                x = max(0.0, x)
                # New max conv
                new_vals = c_base[k_valid].copy() + 2.0 * x * a[m_valid]
                if 0 <= 2 * j <= 2 * n - 2:
                    new_vals[2 * j - k_start] = c_base[2 * j] + x * x
                # For k outside k_valid (and k != 2j): value is c_base[k]
                # We need max over ALL k
                # max = max(max(new_vals), max(c_base outside k_valid, excluding k=2j if not in k_valid))
                max_in_range = new_vals.max()
                # Outside k_valid:
                if k_start > 0:
                    outside1 = c_base[:k_start].max() if k_start > 0 else 0.0
                else:
                    outside1 = 0.0
                if k_end < 2 * n - 2:
                    outside2 = c_base[k_end + 1:].max()
                else:
                    outside2 = 0.0
                # Handle k=2j if not in k_valid range
                if 0 <= 2 * j <= 2 * n - 2:
                    if 2 * j < k_start or 2 * j > k_end:
                        outside2 = max(outside2, c_base[2 * j])  # x^2 term already in new_vals
                        # Actually outside, it's just c_base[2j], but we need to add x^2
                        # No wait - if 2j is outside k_valid, then delta_{2j} = x^2 still applies
                        # So the value at k=2j is c_base[2j] + x^2
                        outside2 = max(outside2, c_base[2 * j] + x * x)
                new_max = max(max_in_range, outside1, outside2)

                new_s = s - old_x + x
                if new_s < 0.01:
                    continue
                new_obj = 2.0 * n * new_max / (new_s * new_s)

                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_x = x

            if best_x != old_x:
                a[j] = best_x
                s = s - old_x + best_x
                improvement += abs(best_x - old_x)

        # Renormalize to sum=1
        if s > 1e-15:
            a = a / s
            s = 1.0

        sweep += 1
        c_new = fast_autocorr(a)
        new_max = c_new.max()

        if new_max < best_max - 1e-15:
            best_max = new_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and sweep % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} ({elapsed:.0f}s): best_max={best_max:.10f}")

        if stall > 50 and improvement < 1e-10:
            # Try random perturbation
            i = rng.integers(n)
            j2 = rng.integers(n)
            if i != j2 and a[i] > 1e-6:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j2] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    stall = 0

        if stall > 200:
            break

    return best_a, best_max


def lse_subgradient_nesterov(a0, time_limit, T_start=100.0, T_end=2.0,
                             n_stages=6, verbose=True):
    """
    Nesterov-accelerated subgradient method on LSE-smoothed objective.
    Uses Euclidean projection onto {a >= 0, sum(a) = 1}.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    iter_count = 0

    temps = np.geomspace(T_start, T_end, n_stages)

    for stage, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        while time.monotonic() - t0 < time_limit:
            w, c_y = compute_lse_weights(y, T)
            grad = compute_lse_grad(y, w)

            # Center the gradient (projection onto sum=0 hyperplane)
            grad_centered = grad - grad.mean()

            # Step size for subgradient method: O(1/sqrt(t))
            eta = 1.0 / (1.0 + 0.5 * np.sqrt(iter_count + 1))

            a_new = y - eta * grad_centered
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            iter_count += 1

            # Nesterov
            momentum = (iter_count - 1.0) / (iter_count + 2.0)
            y_new = a_new + momentum * (a_new - a)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / s_y

            a = a_new
            y = y_new

            # Track best
            max_val = fast_autocorr(a).max()
            if max_val < best_max:
                best_max = max_val
                best_a = a.copy()

            if iter_count % 2000 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"      SG iter={iter_count} T={T:.1f} ({elapsed:.0f}s): best={best_max:.10f}")

    return best_a, best_max


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, seed=42):
    """Generate a structured initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    # Multi-frequency cosine mixture
    a = 1.0
    for freq, amp in [(2, 0.3), (5, 0.15), (11, 0.08), (23, 0.04), (47, 0.02)]:
        a = a + amp * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Returns the best sequence found.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    print(f"Total time budget: {time_budget:.0f}s")

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
        print("No parent available, using structured init")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return sc

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, seed)
        best_score = try_update(best_seq, "init")
        print(f"Initial: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Nesterov Mirror Descent on base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: NM Mirror Descent n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, _ = nesterov_mirror_descent(best_seq, t1, T_start=150.0, T_end=3.0,
                                           n_anneal_stages=6, verbose=True)
        try_update(a_out, f"nm_n{n_base}")

    # === Phase 2: Subgradient Nesterov on base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SubGrad Nesterov n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_out, _ = lse_subgradient_nesterov(best_seq, t2, T_start=80.0, T_end=2.0,
                                            n_stages=5, verbose=True)
        try_update(a_out, f"sg_n{len(best_seq)}")

    # === Phase 3: Coordinate Descent on best so far ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord Descent n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_out, _ = coordinate_descent(best_seq, t3, n_sweeps_max=20000, verbose=True)
        try_update(a_out, f"cd_n{len(best_seq)}")

    # === Phase 4: Try longer length ===
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: NM Mirror Descent n={n_long} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_mirror_descent(init_long, t4, T_start=100.0, T_end=3.0,
                                           n_anneal_stages=5, verbose=True)
        try_update(a_out, f"nm_n{n_long}")

    # === Phase 5: Coordinate Descent on longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord Descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 20:
        a_out, _ = coordinate_descent(best_seq, t5, n_sweeps_max=15000, verbose=True)
        try_update(a_out, f"cd_n{n_best}_2")

    # === Phase 6: Second NM pass with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: NM Mirror Descent (fine) n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 20:
        a_out, _ = nesterov_mirror_descent(best_seq, t6, T_start=50.0, T_end=1.0,
                                           n_anneal_stages=5, verbose=True)
        try_update(a_out, f"nm_fine_n{n_best}")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Coord Descent n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.30, 200.0)
    if t7 > 10:
        a_out, _ = coordinate_descent(best_seq, t7, n_sweeps_max=10000, verbose=True)
        try_update(a_out, f"cd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
