"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Nesterov best-response with pre-built convolution matrix
2. Running average tracking for guaranteed convergence
3. Subgradient refinement with adaptive step size
4. Peak-shaving coordinate descent
5. Multi-length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_T(a):
    """Build (2n-1) x n convolution matrix where T@b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, n + j):
            T[k, j] = a[k - j]
    return T


def get_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(T, solver, warm_start=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = T.shape[1]
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP Nesterov best-response with running average tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    
    T = build_T(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average (theoretically guaranteed to converge)
    running_avg = np.zeros(n)
    n_avg = 0
    
    # Exponential moving average (faster adapting)
    ema = a.copy()
    
    iter_count = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP best response
        b_star, M_val = solve_lp_best_response(T, solver)
        if b_star is None:
            continue
        iter_count += 1
        
        # Nesterov momentum
        if t == 1:
            a_new = b_star.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y + beta * (y - a)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y.copy()
                a_s = 1.0
            a_new = a_new / a_s
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_s = y_new.sum()
        if y_s < 1e-15:
            y_new = y.copy()
            y_s = 1.0
        y_new = y_new / y_s
        
        y = y_new
        a = a_new
        
        # Update running average
        running_avg += a_new
        n_avg += 1
        
        # Update EMA (adaptive rate)
        ema_rate = 0.02 + 0.03 * (1 - min(1.0, t / 500))
        ema = (1 - ema_rate) * ema + ema_rate * a_new
        
        # Check scores
        for label, candidate in [("a", a_new), ("y", y_new), ("avg", running_avg / n_avg), ("ema", ema)]:
            csum = candidate.sum()
            if csum < 1e-15:
                continue
            c = candidate / csum
            score = 2.0 * n * fast_autocorr(c).max()
            if score < best_score:
                best_score = score
                best_a = c.copy()
        
        # Rebuild T periodically (when a changes significantly)
        if t % 50 == 0:
            T = build_T(a)
        
        if t % 200 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M_LP={M_val:.8f}")
    
    return best_a, best_score, iter_count


def subgradient_refine(a0, time_limit, lr_start=0.01, lr_end=0.0001):
    """Subgradient method with clipping on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    n_iters = 20000
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Learning rate: polynomial decay
        progress = it / n_iters
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute subgradient of max_k conv(a,a)_k
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Use soft-max over top peaks for smoother gradient
        top_k = min(20, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1][:top_k]
        weights = np.exp((b[sorted_idx] - max_b) * 50)
        weights = weights / weights.sum()
        
        # Gradient: for each peak k, grad_k = 2 * shift_k(a)
        # (conv(a,a))[k] = sum_j a[j]*a[k-j]
        # d/d a[i] (conv(a,a))[k] = a[k-i] + a[i-k] (for symmetric conv)
        # Actually for conv(a,a)[k] = sum_j a[j]*a[k-j]:
        # d/da[i] = a[k-i] (from first factor) + a[i-k+...] 
        # Let's be careful: conv(a,a)[k] = sum_j a[j]*a[k-j]
        # d/da[i] conv(a,a)[k] = a[k-i] + a[i-k] if both are valid
        # For k-i valid: 0 <= k-i < n => k-n+1 <= i <= k
        # For i-k valid: 0 <= i-k < n => k <= i < k+n
        # Combined: max(0, k-n+1) <= i < min(n, k+n) = n
        # And the two terms: a[k-i] when k-i is in [0,n), and a[i-k] when i-k is in [0,n)
        
        grad = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            # Term 1: a[k-i] for i where 0<=k-i<n => max(0,k-n+1) <= i <= min(n-1,k)
            i_start1 = max(0, k - n + 1)
            i_end1 = min(n - 1, k)
            grad[i_start1:i_end1+1] += w * a[k - i_end1:k - i_start1 + 1][::-1]
            
            # Term 2: a[i-k] for i where 0<=i-k<n => max(0,k) <= i <= min(n-1, k+n-1) = n-1
            i_start2 = max(0, k)
            i_end2 = n - 1
            grad[i_start2:i_end2+1] += w * a[i_start2-k:i_end2-k+1]
        
        # Since conv(a,a) is symmetric, grad should be approximately symmetric
        # Apply gradient step with projection onto simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            a_new = a.copy()
        else:

def peak_shaving(a0, time_limit, seed=42):
    """Coordinate descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        peak_k = b.argmax()
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Try moving mass from the largest contributor
        top_c = np.argsort(contrib)[::-1][:3]
        improved = False
        
        for ci in top_c:
            src = idx[ci]
            src2 = peak_k - src
            
            # Try moving to a position that's far from the peak
            # Score positions by distance from peak
            dists = np.abs(np.arange(n) - peak_k)
            # Add penalty for positions close to other peaks
            other_peaks = np.argsort(b)[::-1][:5]
            for op in other_peaks:
                dists = np.minimum(dists, np.abs(np.arange(n) - op))
            
            # Prefer positions with large distance and small current value
            scores = dists / (a + 0.01)
            # Exclude source positions
            scores[src] = -np.inf
            if src2 != src:
                scores[src2] = -np.inf
            
            top_dst = np.argsort(scores)[::-1][:5]
            
            for dst in top_dst:
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    if src2 != src:
                        a_trial[src2] -= amt * 0.3
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation every 5 stalls
            if stall % 5 == 0:
                n_pert = rng.integers(1, 4)
                srcs = rng.choice(n, size=n_pert, replace=False)
                dsts = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for sr, ds in zip(srcs, dsts):
                    amt = a_trial[sr] * 0.01
                    a_trial[sr] -= amt
                    a_trial[ds] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s > 1e-15:
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break
    
    return best_a, best_score


def chirp_init(n, gamma=1.0):
    """Generate a chirp-like sequence with flat spectral magnitude."""
    x = np.arange(n)
    phase = gamma * np.pi * x * x / n
    a = np.abs(np.cos(phase)) + 0.1
    return a / a.sum()


def step_init(n, transition=0.2):
    """Step-like initialization for anti-symmetric structure."""
    # First half high, second half low (with smooth transition)
    x = np.linspace(0, 1, n)
    a = np.where(x < 1 - transition, 1.0, np.maximum(0, 1 - (x - (1 - transition)) / transition))
    return a / a.sum()


def random_init(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    return a / a.sum()


def interpolate(a, new_n):
    """Interpolate to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def solve_lp_with_highs(a):
    """Solve LP: min M s.t. conv(a,b)_k <= M, sum(b)=1, b>=0 using HiGHS."""
    n = len(a)
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: T @ b - M <= 0 for each k
    # T[k, j] = a[k-j] if valid
    # Build A_ub of shape (2n-1, n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        for k in range(j, n + j):
            A_ub[k, j] = a[k - j]
    A_ub[:, -1] = -1.0  # -M
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Bounds: b_i >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(res.x[-1])

    """Solve dual LP: max sum(y) s.t. sum_k y_k * a[k-i] <= 1 for all i, y >= 0."""
    n = len(a)
    m = 2 * n - 1  # number of y variables

    # Objective: maximize sum(y) = minimize -sum(y)
    c = -np.ones(m)
    
    # Constraints: for each i in 0..n-1: sum_k y_k * a[k-i] <= 1
    # A[i, k] = a[k-i] if 0 <= k-i < n

        for k in range(max(0, i), min(m, i + n)):
            A[i, k] = a[k - i]

    if res.status != 0:

    return res.x, float(-res.fun)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient LP best-response Nesterov with HiGHS solver
2. Running average and EMA tracking for convergence
3. Subgradient refinement with soft-max gradient
4. Adaptive peak-shaving coordinate descent
5. Multi-length exploration with structured initialization
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_A_ub(a):
    """Build A_ub matrix for the LP: rows are lags k, cols are [b_0..b_{n-1}, -M]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n + 1))
    for j in range(n):
        k_start = j
        k_end = n + j - 1
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    A[:, -1] = -1.0
    return A


def solve_lp_highs(a):
    """Solve LP: min M s.t. conv(a,b)_k <= M, sum(b)=1, b>=0 using HiGHS."""
    n = len(a)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = build_A_ub(a)
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    b_val = np.maximum(res.x[:n], 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val / s, float(res.x[-1])


def solve_lp_cvxpy(a, solver):
    """Fallback LP solve using cvxpy."""
    n = len(a)
    A_ub = build_A_ub(a)
    m = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A_ub @ np.vstack([b_var, M_var]) <= 0, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val / s, float(M_var.value)


def get_lp_solver():
    """Find a working LP solver for fallback."""
    for sn in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(4, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            p.solve(solver=sn)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return sn
        except Exception:
            continue
    return cp.SCS


def nesterov_lp_main(a0, time_limit, solver, verbose=True):
    """LP Nesterov best-response with multiple tracking strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    running_avg = np.zeros(n)
    n_avg = 0
    ema = a.copy()
    ema_rate = 0.03

    iter_count = 0
    failed_count = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_highs(a)
        if b_star is None:
            b_star, M_val = solve_lp_cvxpy(a, solver)
        if b_star is None:
            failed_count += 1
            if failed_count > 100:
                break
            continue
        iter_count += 1
        failed_count = 0

        # Nesterov momentum update
        if t == 1:
            a_new = b_star.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y + beta * (y - a)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y.copy()
                s = 1.0
            a_new = a_new / s

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = a.copy()
            ys = 1.0
        y_new = y_new / ys

        y = y_new
        a = a_new

        running_avg += a_new
        n_avg += 1
        ema_rate = 0.02 + 0.05 * max(0, 1 - t / 300)
        ema = (1 - ema_rate) * ema + ema_rate * a_new

        # Evaluate candidates
        candidates = [a_new, y_new]
        if n_avg >= 5:
            candidates.append(running_avg / n_avg)
        candidates.append(ema)

        for cand in candidates:
            cs = cand.sum()
            if cs < 1e-15:
                continue
            c = cand / cs
            sc = 2.0 * n * fast_autocorr(c).max()
            if sc < best_score:
                best_score = sc
                best_a = c.copy()

        if t % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

        if t > 100:
            recent_best = 2.0 * n * fast_autocorr(a).max()
            if recent_best > best_score * 1.01 and t > 5000:
                # Stalled badly, break
                if verbose:
                    print(f"    Stalled at t={t}, breaking")
                break

    return best_a, best_score, iter_count


def subgradient_refine(a0, time_limit, lr_start=0.005, lr_end=0.0001):
    """Subgradient method with soft-max gradient on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    n_iters = 30000

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / n_iters
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()

        # Soft-max over top peaks
        n_peaks = min(30, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1][:n_peaks]
        w = np.exp((b[sorted_idx] - max_b) * 40)
        w = w / w.sum()

        # Compute gradient
        grad = np.zeros(n)
        for wi, k in zip(w, sorted_idx):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            idx = np.arange(i_lo, i_hi + 1)
            # d/da[i] of conv(a,a)[k] = a[k-i] + a[i-k] (each if in range)
            grad[i_lo:i_hi + 1] += wi * a[k - i_hi:k - i_lo + 1][::-1]
            i_lo2 = max(0, k)
            i_hi2 = n - 1
            if i_hi2 >= i_lo2:
                grad[i_lo2:i_hi2 + 1] += wi * a[i_lo2 - k:i_hi2 - k + 1]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        sc = 2.0 * n * fast_autocorr(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if it % 5000 == 0 and it > 0:
            print(f"    SubGrad it={it}: best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_final(a0, time_limit, seed=42):
    """Adaptive coordinate descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        max_b = b.max()
        peak_k = b.argmax()

        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        idx = np.arange(i_lo, i_hi + 1)
        contrib = a[idx] * a[peak_k - idx]
        top_c = np.argsort(contrib)[::-1][:5]

        improved = False
        for ci in top_c:
            src = idx[ci]
            src2 = peak_k - src
            if a[src] < 1e-10:
                continue
            # Find good target: far from peak and other peaks
            other_peaks = np.argsort(b)[::-1][:8]
            dists = np.abs(np.arange(n) - peak_k).astype(float)
            for op in other_peaks[1:]:
                dists = np.minimum(dists, np.abs(np.arange(n) - op).astype(float))
            scores = dists / (a + 0.005)
            scores[src] = -np.inf
            if src2 != src:
                scores[src2] = -np.inf
            top_dst = np.argsort(scores)[::-1][:8]

            for dst in top_dst:
                for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                    amt = a[src] * frac
                    a_tr = a.copy()
                    a_tr[src] -= amt
                    if src2 != src:
                        a_tr[src2] -= amt * 0.3
                    a_tr[dst] += amt
                    a_tr = np.maximum(a_tr, 0)
                    s = a_tr.sum()
                    if s < 1e-15:
                        continue
                    a_tr = a_tr / s
                    sc = 2.0 * n * fast_autocorr(a_tr).max()
                    if sc < best_score - 1e-15:
                        a = a_tr
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 8 == 0:
                np_pert = rng.integers(1, 5)
                srcs = rng.choice(n, size=np_pert, replace=False)
                dsts = rng.choice(n, size=np_pert, replace=False)
                a_tr = a.copy()
                for sr, ds in zip(srcs, dsts):
                    amt = a_tr[sr] * rng.uniform(0.005, 0.03)
                    a_tr[sr] -= amt
                    a_tr[ds] += amt
                a_tr = np.maximum(a_tr, 0)
                s = a_tr.sum()
                if s > 1e-15:
                    a_tr = a_tr / s
                    sc = 2.0 * n * fast_autocorr(a_tr).max()
                    if sc < best_score:
                        a = a_tr
                        best_score = sc
                        best_a = a.copy()
                        stall = 0

    return best_a, best_score


def chirp_init(n, gamma=1.0):
    """Chirp-like initialization."""
    x = np.arange(n)
    phase = gamma * np.pi * x * x / n
    a = np.abs(np.cos(phase)) + 0.05
    return a / a.sum()


def step_init(n, transition=0.15):
    """Step-like initialization emphasizing anti-symmetric structure."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    # Smooth transition in the middle
    mid = n // 2
    w = max(1, int(n * transition))
    for i in range(max(0, mid - w), min(n, mid + w)):
        t = (i - (mid - w)) / (2 * w)
        a[i] = 1.0 - 0.5 * np.cos(t * np.pi)
    return a / a.sum()


def random_init(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.01
    return a / a.sum()


def spectral_init(n, seed=0):
    """Initialize with approximately flat DFT magnitude."""
    rng = np.random.default_rng(seed)
    m = n // 2 + 1
    phase = rng.uniform(0, 2 * np.pi, m)
    X = np.zeros(n, dtype=complex)
    X[:m] = np.exp(1j * phase[:m])
    # Hermitian symmetry
    for k in range(1, n - n // 2):
        X[k] = np.conj(X[n - k])
    a = np.abs(irfft(X, n))
    a = a + 0.01
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
