def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def coordinate_descent_polish(a0, time_limit, n_trials=50000):
    """
    Local polish via pairwise mass transfer.
    Tries to move small amounts of mass from position i to position j
    to reduce the max convolution.
    """
    n = len(a0)

    s = a.sum()
    if s < 1e-15:

    b = fast_autocorr(a)
    peak_idx = np.argmax(b)

    trials = 0
    stall = 0
    while trials < n_trials and time.monotonic() - t0 < time_limit:
        # Find top peaks
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:4]

        # For each top peak, try to reduce it
        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            # Find the biggest contributor
            tc = np.argmax(contrib)
            src = int(idx[tc])
            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Try moving to random positions far from the peak
            for attempt in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                # Don't move to a position that also contributes to this peak
                if js <= dst <= je:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.03]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        break
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:

                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

            stall += 1
            if stall > 50:
                break

        if trials % 5000 == 0:

        a_out3 = softmax_fn(z_out3)
        try_update(a_out3, "lse_refine")

        # Adam at very low T
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_adam2 = min(60.0, remaining * 0.2)
            z_out4, _ = adam_refine(z_out3, T=0.1, n_steps=300, lr=0.001,
                                     time_limit=t_adam2, tag="adam_refine")
            a_out4 = softmax_fn(z_out4)
            try_update(a_out4, "adam_refine")

    # === Phase 4: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-length exploration (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        for n_try in [2000, 1200, 2500]:
            if time.monotonic() - t_global > time_budget - 30:
                break
            a_try = make_ramp(n_try, 1, 1.0)
            sc = 2.0 * n_try * fast_autocorr(a_try).max()
            print(f"  n={n_try} ramp score: {sc:.8f}")
            if sc < best_score:
                z_try = np.log(np.maximum(a_try, 1e-30))
                t_try = min(60.0, (time_budget - (time.monotonic() - t_global)) * 0.15)
                if t_try > 10:
                    z_out_t, _ = multi_stage_optimize(z_try, T_start=30.0, T_end=0.5,
                                                       n_stages=10, time_limit=t_try, tag=f"n{n_try}")
                    a_out_t = softmax_fn(z_out_t)
                    try_update(a_out_t, f"n{n_try}_lse")
                else:
                    try_update(a_try, f"n{n_try}_ramp")
            else:
                try_update(a_try, f"n{n_try}_ramp")

    # === Phase 5: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Final polish (remaining={remaining:.0f}s) ===")
    if remaining > 20 and best_a is not None:
        t_cd2 = min(60.0, remaining * 0.4)
        a_cd2 = coordinate_descent_polish(best_a, t_cd2)
        try_update(a_cd2, "final_cd")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            z_fin = np.log(np.maximum(best_a, 1e-30))
            t_fin = min(30.0, remaining * 0.4)
            z_fin_out, _ = multi_stage_optimize(z_fin, T_start=5.0, T_end=0.005,
                                                 n_stages=6, time_limit=t_fin, tag="final_lse")
            a_fin = softmax_fn(z_fin_out)
            try_update(a_fin, "final_lse")

    # === Phase 6: Verify with evaluate_sequence ===
    if best_a is not None:
        final_score = evaluate_safe(best_a)
        print(f"\nFinal evaluate_sequence score: {final_score:.8f}")
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"FINAL: score={best_score:.8f}, n={len(best_a)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    # Ensure output is a valid list of floats
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    return result
