import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def score_normalized(a):
    """Compute objective for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    c = fast_autocorr(a)
    return 2.0 * n * c.max()


def compute_topk_lse(a, T, topk):
    """
    Compute top-k LSE smoothed objective and gradient w.r.t. a.
    Returns (lse_value, gradient).
    """
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space (log-softmax parametrization)."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_lse(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.ECOS


def build_conv_matrix_a(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, top_constraints=None):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    Optionally only enforce top_constraints most important inequalities.
    Returns (b_star, M_value).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_a(a_fixed)

    all_constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    if top_constraints is not None and top_constraints < (2 * n - 1):
        # Use all constraints but this is a hint for future optimization
        constraints = all_constraints
    else:
        constraints = all_constraints

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000):
    """
    Nesterov-accelerated LP best-response with:
    - Exponential moving average for best tracking
    - Adaptive restart based on gap
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Exponential moving averages with different rates
    ema_fast = a.copy()
    ema_slow = a.copy()
    BETA_FAST = 0.05
    BETA_SLOW = 0.01

    # Circular buffer for windowed average
    WINDOW = 100
    buf = np.zeros((WINDOW, n))
    buf_idx = 0
    buf_filled = 0

    stall_count = 0
    last_improvement_score = best_score

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Update EMAs
        ema_fast = (1 - BETA_FAST) * ema_fast + BETA_FAST * a
        ema_fast = np.maximum(ema_fast, 0)
        ema_fast = ema_fast / ema_fast.sum()
        ema_slow = (1 - BETA_SLOW) * ema_slow + BETA_SLOW * a
        ema_slow = np.maximum(ema_slow, 0)
        ema_slow = ema_slow / ema_slow.sum()

        # Circular buffer
        buf[buf_idx] = a
        buf_idx = (buf_idx + 1) % WINDOW
        buf_filled = min(buf_filled + 1, WINDOW)

        # Check scores periodically (every 10 iters to save time)
        if t % 10 == 0:
            candidates = [a, y, ema_fast, ema_slow]
            # Add windowed average if enough samples
            if buf_filled >= 50:
                candidates.append(buf[:buf_filled].mean(axis=0))

            for cand in candidates:
                cand = np.maximum(cand, 0)
                cs = cand.sum()
                if cs < 1e-15:
                    continue
                cand = cand / cs
                sc = score_normalized(cand)
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    stall_count = 0
                else:
                    stall_count += 1

        # Adaptive restart
        if stall_count > 1500 and t > 500:
            a = best_a.copy()
            y = best_a.copy()
            ema_fast = best_a.copy()
            ema_slow = best_a.copy()
            buf[:] = best_a[np.newaxis, :]
            buf_filled = WINDOW
            buf_idx = 0
            stall_count = 0
            if t % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      NLP restart at t={t} ({elapsed:.0f}s), best={best_score:.6f}, LP_lb={M_val:.8f}")

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_lb={M_val:.8f}, gap={best_score-2*n*M_val:.6f}")

    return best_a, best_score


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=30, topk=40):
    """
    Adam-optimized top-k LSE with cosine learning rate schedule.
    Uses increasing top-k over stages for wider peak coverage.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    # Top-k schedule: start small, grow to cover more peaks
    topk_schedule = np.concatenate([
        np.full(n_stages // 3, topk // 3),
        np.full(n_stages // 3, topk // 2),
        np.full(n_stages - 2 * (n_stages // 3), topk)
    ])

    best_a = a_init.copy()
    best_score = score_normalized(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        cur_topk = int(topk_schedule[si])
        # Cosine learning rate within the overall budget
        progress = si / max(1, n_stages - 1)
        base_lr = 3e-3 * min(1.0, 60.0 / max(1, n - 100))
        lr = base_lr * 0.5 * (1 + np.cos(np.pi * progress))
        lr = max(lr, 1e-5)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 100.0) * 4))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, cur_topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = score_normalized(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 50 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        a_stage = softmax_fn(z)
        score_stage = score_normalized(a_stage)
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Coordinate descent that:
    1. Identifies the active support (non-negligible components)
    2. Optimizes coordinates on the support
    3. Periodically tries to zero out small components
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0

    # Determine initial support threshold
    threshold = 1e-4

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        cmax = c.max()
        # Find peaks near max
        peak_thresh = cmax * 0.999
        peaks = np.where(c >= peak_thresh)[0]
        if len(peaks) == 0:
            peaks = np.argsort(c)[::-1][:5]

        improved = False

        # For each peak, find the contributing pair and try to reduce it
        for k in peaks[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_ci = np.argmax(contrib)
            src = idx[max_ci]
            partner = k - src

            if a[src] < threshold * a.mean():
                continue

            # Find destination: position that contributes least to any peak
            # Score each position by how much it contributes to peaks
            dest_scores = np.zeros(n)
            for pk in peaks[:10]:
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                for j in range(js2, je2 + 1):
                    pkj = pk - j
                    if 0 <= pkj < n:
                        dest_scores[j] += a[pkj]

            # Prefer low-score destinations
            excluded = set([src, partner])
            candidates = [j for j in range(n) if j not in excluded and a[j] < a.mean() * 2]
            if not candidates:
                candidates = [j for j in range(n) if j not in excluded]
            if not candidates:
                continue
            # Sort by dest_score (prefer low)
            candidates.sort(key=lambda j: dest_scores[j])
            dst_pool = candidates[:max(5, len(candidates) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            # Try moving mass
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                st = a_trial.sum()
                a_trial = a_trial / st
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Try zeroing small components
            small_idx = np.where(a < threshold * a.mean() * 0.5)[0]
            small_idx = small_idx[small_idx != np.argmax(a)]
            if len(small_idx) > 0:
                # Zero out a few at a time
                n_zero = min(3, len(small_idx))
                zero_idx = rng.choice(small_idx, size=n_zero, replace=False)
                a_trial = a.copy()
                mass_to_redistribute = a_trial[zero_idx].sum()
                a_trial[zero_idx] = 0
                # Redistribute to random large positions
                large_idx = np.where(a_trial > threshold * a.mean())[0]
                if len(large_idx) > 0:
                    targets = rng.choice(large_idx, size=min(n_zero, len(large_idx)), replace=False)
                    for i, ti in enumerate(targets):
                        a_trial[ti] += mass_to_redistribute / len(targets)
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-10:
                        a_trial = a_trial / st
                        sc = score_normalized(a_trial)
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0

            if not improved:
                stall += 1
                # Random perturbation to escape
                if stall % 50 == 0:
                    n_pert = rng.integers(2, 6)
                    a_trial = a.copy()
                    for _ in range(n_pert):
                        i = rng.integers(n)
                        j = rng.integers(n)
                        if i != j and a_trial[i] > 1e-8:
                            frac = rng.uniform(0.001, 0.01)
                            amt = a_trial[i] * frac
                            a_trial[i] -= amt
                            a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-10:
                        a_trial = a_trial / st
                        sc = score_normalized(a_trial)
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                if stall > 500:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.8 + 0.15 * np.cos(2 * x) + 0.05 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.6 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (0.5 + 0.5 * np.cos(2 * np.pi * x * 3))
    elif choice == 7:
        # Symmetric raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 8:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)
    else:
        # Multi-bump
        x = np.linspace(0, 1, n)
        a = np.exp(-20*(x-0.2)**2) + np.exp(-20*(x-0.5)**2) + np.exp(-20*(x-0.8)**2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    1. LP Nesterov at base length (primary workhorse, extended time)
    2. Top-k LSE with Adam and cosine annealing (wider peaks)
    3. LP Nesterov at longer length
    4. LSE at longer length
    5. Coordinate descent polish (sparsity-aware)
    6. Try shorter length
    7. Final LSE + CD polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 100.0, 1700.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s budget) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, max_iters=500000)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: LSE at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: LSE n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.14, 220.0)
    if t2 > 30:
        a_out, _ = adam_lse_optimize(best_seq, t2, T_start=80.0, T_end=0.1, n_stages=30, topk=50)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: LP Nesterov at n+150 ===
    n_long = n_base + 150
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: LP Nesterov n={n_long} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t3, solver, max_iters=300000)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 4: LSE at n+150 ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t4 = min(remaining * 0.10, 120.0)
        if t4 > 20:
            print(f"=== Phase 4: LSE n={n_long} ({remaining:.0f}s left) ===")
            a_out, _ = adam_lse_optimize(best_seq, t4, T_start=50.0, T_end=0.2, n_stages=20, topk=40)
            # Interpolate result back to n_long for evaluation
            a_out_long = interpolate_sequence(a_out, n_long)
            try_update(a_out_long, f"lse_n{n_long}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: CD polish ({remaining:.0f}s left) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        a_out, _ = coordinate_descent_polish(best_seq, t5, seed=77)
        try_update(a_out, "cd_polish_1")

    # === Phase 6: Try shorter length ===
    n_short = n_base - 150
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 40:
        t6 = min(remaining * 0.10, 120.0)
        if t6 > 30:
            print(f"=== Phase 6: LSE n={n_short} ({remaining:.0f}s left) ===")
            init_short = interpolate_sequence(best_seq, n_short)
            a_out, _ = adam_lse_optimize(init_short, t6, T_start=40.0, T_end=0.3, n_stages=15, topk=30)
            try_update(a_out, f"lse_n{n_short}")

    # === Phase 7: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t7 = min(remaining * 0.12, 100.0)
        if t7 > 15:
            print(f"=== Phase 7: Final LSE polish ({remaining:.0f}s left) ===")
            a_out, _ = adam_lse_optimize(best_seq, t7, T_start=5.0, T_end=0.05, n_stages=12, topk=20)
            try_update(a_out, "final_lse")

    # === Phase 8: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.4, 60.0)
        if t8 > 8:
            print(f"=== Phase 8: Final CD ({remaining:.0f}s left) ===")
            a_out, _ = coordinate_descent_polish(best_seq, t8, seed=2024)
            try_update(a_out, "cd_polish_2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
