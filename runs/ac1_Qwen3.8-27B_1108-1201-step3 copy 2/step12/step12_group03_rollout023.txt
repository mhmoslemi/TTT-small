1. FFT-accelerated Nesterov subgradient with adaptive log-sum-exp smoothing and dual tracking

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, p):
    """Cross-correlation: result[m] = sum_j a[j]*p[j+m] for m=0..n-1."""
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    A_f = rfft(np.asarray(a, dtype=np.float64), N)
    P_f = rfft(np.asarray(p, dtype=np.float64), N)
    return irfft(np.conj(A_f) * P_f, N)[:n]

def compute_lse_weights(c, tau):
    """Compute softmax(tau * c) weights, numerically stable."""
    c_shifted = c - c.max()
    log_w = tau * c_shifted
    max_lw = log_w.max()
    if max_lw > 500:
        log_w = log_w - max_lw
    w = np.exp(log_w)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w


def nesterov_fft(a0, time_limit, verbose=True, tau_start=30.0, tau_end=500.0):
    """
    Nesterov-accelerated subgradient method with log-sum-exp smoothing.
    Tracks primal and dual running averages for convergence monitoring.
    Uses FFT for all convolutions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()

    # Dual tracking for lower bound
    dual_sum = np.zeros(2 * n - 1)
    dual_count = 0

    t0 = time.monotonic()
    N_fft = 1 << (3 * n - 1).bit_length()

    # Primal running average (for upper bound)
    primal_avg = a.copy()
    primal_count = 0

    iter_count = 0
    stall = 0
    stall_limit = 20000
    best_stall_val = best_val

    # Adaptive step size
    step_mult = 1.0
    last_func_val = float('inf')
    bad_steps = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Smoothness parameter: start moderate, increase over time
        progress = min(1.0, t / 50000.0)
        tau = tau_start + (tau_end - tau_start) * progress

        # Compute autocorrelation at y
        c_y = fast_autocorr(y)
        max_c = c_y.max()
        val_y = 2.0 * n * max_c

        # Compute LSE weights (dual variable)
        w = compute_lse_weights(c_y, tau)

        # Update dual running average
        dual_sum += w
        dual_count += 1

        # Compute gradient: grad[m] = 2 * sum_k w[k] * y[k-m]
        grad = 2.0 * fast_cross_corr(y, w)

        # Nesterov schedule
        alpha_t = 2.0 / (t + 2.0)

        # Update y
        y_new = (1.0 - alpha_t) * y + alpha_t * a
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s

        # Recompute gradient at y_new for the step
        c_yn = fast_autocorr(y_new)
        w_n = compute_lse_weights(c_yn, tau)
        grad_n = 2.0 * fast_cross_corr(y_new, w_n)

        # Adaptive step size
        grad_norm = np.linalg.norm(grad_n)
        if grad_norm < 1e-15:
            step = 0.0
        else:
            # Step size: c / (tau * (t+1)^0.7) - betweensqrt and 1/t schedules
            step = 0.8 * step_mult / (tau * (t + 1.0) ** 0.7)
            # Cap step to avoid divergence
            max_step = 1.0 / (4.0 * np.max(y_new) + 1e-10)
            step = min(step, max_step)

        a_new = y_new - step * grad_n
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            a_new = y_new.copy()
            a_new_s = 1.0
        a_new = a_new / a_new_s

        y = y_new
        a = a_new
        iter_count += 1

        # Update primal running average (exponential for recency)
        if primal_count < 1:
            primal_avg = a_new.copy()
        else:
            rate = min(0.05, 2.0 / (t + 10.0))
            primal_avg = (1.0 - rate) * primal_avg + rate * a_new
        primal_count += 1

        # Score
        val_a = 2.0 * n * fast_autocorr(a).max()
        if val_a < best_val:
            best_val = val_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if val_y < best_val:
            best_val = val_y
            best_a = y.copy()
            stall = 0

        # Check primal average
        if t > 100 and t % 100 == 0:
            val_avg = 2.0 * n * fast_autocorr(primal_avg).max()
            if val_avg < best_val:
                best_val = val_avg
                best_a = primal_avg.copy()
                stall = 0

        # Adaptive step: if function is increasing, reduce step
        if val_a > last_func_val * 1.01:
            bad_steps += 1
            if bad_steps > 5:
                step_mult *= 0.7
                bad_steps = 0
        else:
            bad_steps = max(0, bad_steps - 1)
        last_func_val = val_a

        # Periodic verbose output
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            # Lower bound from dual
            dual_avg = dual_sum / dual_count
            lower_bound = 2.0 * n * np.dot(dual_avg, fast_autocorr(primal_avg))
            print(f"    NS t={t} ({elapsed:.1f}s): best={best_val:.8f}, lb={lower_bound:.8f}, tau={tau:.1f}")

        # Early stopping
        if stall > stall_limit and t > 20000:
            if verbose:
                print(f"    NS: Stalled at t={t}, best={best_val:.8f}")
            break

    return best_a, best_val, iter_count


def qp_active_refine(a0, time_limit, verbose=True):
    """
    QP refinement on active constraint set with adaptive expansion.
    Minimize M s.t. c_k(a) <= M for k in active_set, sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    c = fast_autocorr(a)
    max_c = c.max()
    best_a = a.copy()
    best_val = 2.0 * n * max_c
    t0 = time.monotonic()

    # Get solver
    solver = None
    for sname in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=sname)
            if p.status in ('optimal', 'optimal_inaccurate'):
                solver = sname
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        # Identify active set: lags near the maximum
        sorted_lags = np.argsort(c)[::-1]
        # Include lags within 2% of max
        threshold = max_c * 0.98
        active = sorted_lags[c[sorted_lags] >= threshold]
        # Limit to top 40
        if len(active) > 40:
            active = sorted_lags[:40]

        # Build QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, M_var >= 0]

        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other])
            constraints.append(q_expr <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, max_iter=20000)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=20000)
            except Exception:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt, max_iter=20000)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        solver = alt
                        break
                except Exception:
                    continue
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        c_new = fast_autocorr(a_new)
        max_new = c_new.max()
        val_new = 2.0 * n * max_new

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            max_c = max_new
            if verbose:
                print(f"    QP r={round_num}: improved to {best_val:.8f}")
        else:
            # Try blending
            improved_blend = False
            for frac in [0.3, 0.5, 0.7, 0.2, 0.8]:
                a_blend = (1 - frac) * a + frac * a_new
                a_blend = np.maximum(a_blend, 0)
                a_blend_s = a_blend.sum()
                if a_blend_s < 1e-15:
                    continue
                a_blend = a_blend / a_blend_s
                val_blend = 2.0 * n * fast_autocorr(a_blend).max()
                if val_blend < best_val:
                    best_val = val_blend
                    best_a = a_blend.copy()
                    a = a_blend
                    c = fast_autocorr(a)
                    max_c = c.max()
                    improved_blend = True
                    if verbose:
                        print(f"    QP r={round_num} blend={frac}: {best_val:.8f}")
                    break

            if not improved_blend and round_num >= 3:
                break

    return best_a, best_val


def coordinate_descent(a0, time_limit, n_restarts=3, verbose=True):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate j, the optimal value minimizes max_k c_k over c_k that involve a_j.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        if restart > 0:
            # Perturb
            a = a.copy()
            n_pert = rng.integers(5, 20)
            srcs = rng.choice(n, n_pert, replace=False)
            dsts = rng.choice(n, n_pert, replace=False)
            for si, di in zip(srcs, dsts):
                amt = a[si] * rng.uniform(0.01, 0.05)
                a[si] -= amt
                a[di] += amt
            a = np.maximum(a, 0)
            a = a / a.sum()

        for sweep in range(50):
            if time.monotonic() - t0 > time_limit:
                break

            # Random order for this sweep
            order = rng.permutation(n)

            for j in order:
                if time.monotonic() - t0 > time_limit:
                    break

                # c_k involving a_j: c_k = sum_i a_i * a_{k-i}
                # a_j appears in c_k when k-j is valid (as first factor) or j-k is valid (as second)
                # c_k = a_j * a_{k-j} + (other terms) for k = j, j+1, ..., j+n-1 (first factor)
                #      c_k = a_{k-j} * a_j for k = j-n+1, ..., j (second factor)
                # So a_j affects c_k for k in [max(0, j-n+1), min(2n-2, j+n-1)]
                # Simplify: k in [0, 2n-2] where either k-j in [0,n-1] or j-k in [0,n-1]

                # Find affected lags
                k_lo = max(0, j - (n - 1))
                k_hi = min(2 * n - 2, j + (n - 1))
                affected = np.arange(k_lo, k_hi + 1)

                # Current values at affected lags
                c_full = fast_autocorr(a)
                base_c = c_full[affected].copy()

                # The part of c_k not involving a_j:
                # c_k = a_j * a_{k-j} + a_{k-j} * a_j - (correction if k-j == j, i.e., a_j^2 counted twice)
                # Actually: c_k = sum_i a_i * a_{k-i}
                # The terms involving a_j: i=j gives a_j * a_{k-j}; k-i=j gives a_{k-j} * a_j
                # If j != k-j: contribution = 2 * a_j * a_{k-j}
                # If j == k-j (k=2j): contribution = a_j^2
                # Base (without a_j): c_k - contribution

                a_j = a[j]
                contributions = np.zeros(len(affected))
                for idx, k in enumerate(affected):
                    other = k - j
                    if 0 <= other < n:
                        if other == j:
                            contributions[idx] = a_j * a_j
                        else:
                            contributions[idx] = 2.0 * a_j * a[other]
                base_c = base_c - contributions

                # Now optimize a_j: c_k(a_j) = base_c[k] + contrib_k(a_j)
                # where contrib_k uses the new a_j
                # We want to minimize max_k (base_c[k] + contrib_k(a_j))
                # subject to a_j >= 0, and sum constraint (a_j changes sum)

                # For simplicity, fix the sum: if a_j changes by delta, sum changes by delta
                # We can either allow the sum to change (and renormalize) or fix it.
                # Let's try: new_a_j in [0, 3*max(a)], evaluate after renormalization

                best_jj = a_j
                best_val_jj = 2.0 * n * c_full.max()

                # Sample candidate values for a_j
                max_aj = 3.0 * np.max(a)
                candidates = np.concatenate([
                    np.linspace(0, max_aj, 10),
                    np.array([a_j * 0.5, a_j * 0.7, a_j * 1.3, a_j * 1.5])
                ])
                candidates = np.unique(np.clip(candidates, 0, 100.0))

                for aj_new in candidates:
                    if abs(aj_new - a_j) < 1e-10:
                        continue
                    a_trial = a.copy()
                    # Adjust to maintain sum: reduce others proportionally
                    old_sum = a.sum()
                    new_sum = old_sum - a_j + aj_new
                    if new_sum < 1e-10:
                        continue
                    # Scale others to maintain sum
                    if new_sum != old_sum:
                        scale = old_sum / new_sum
                        a_trial[1:] *= 0  # placeholder
                        # Actually, just set a_j and renormalize
                        a_trial[j] = aj_new
                        a_trial = a_trial / a_trial.sum()

                    val = 2.0 * n * fast_autocorr(a_trial).max()
                    if val < best_val_jj:
                        best_val_jj = val
                        best_jj = aj_new

                if best_jj != a_j:
                    a[j] = best_jj
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    val_new = 2.0 * n * fast_autocorr(a).max()
                    if val_new < best_val:
                        best_val = val_new
                        best_a = a.copy()

    return best_a, best_val


def mirror_descent(a0, time_limit, verbose=True):
    """
    Mirror descent (exponential gradient) on the simplex.
    Uses top-peak weighted subgradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    N_fft = 1 << (3 * n - 1).bit_length()

    lr = 0.1
    n_iters = 20000

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / max(1, n_iters)
        lr_curr = lr * (0.01) ** progress

        c = fast_autocorr(a)
        max_c = c.max()

        # Top-5 peak weights
        top_idx = np.argsort(c)[::-1][:5]
        weights = np.exp((c[top_idx] - max_c) * 20.0)
        weights /= weights.sum()

        # Gradient = weighted sum of 2*flip_shift(a, k)
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            # dc_k/da_m = 2*a_{k-m} if 0<=k-m<=n-1
            other_idx = k - np.arange(n)
            valid = (other_idx >= 0) & (other_idx < n)
            grad[valid] += 2.0 * w * a[other_idx[valid]]

        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - lr_curr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_s = a_new.sum()
        if a_new_s < 1e-30:
            break
        a_new = a_new / a_new_s

        a = a_new
        val = 2.0 * n * fast_autocorr(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.1f}s): best={best_val:.8f}")

    return best_a, best_val


def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_symmetric(a):
    """Make sequence symmetric: a[j] = a[n-1-j]."""
    n = len(a)
    a_sym = np.zeros(n)
    for j in range(n):
        a_sym[j] = (a[j] + a[n - 1 - j]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

    if t3 > 30:
        print(f"\n=== Phase 3: Nesterov round 2 at n={n_base} ({t3:.0f}s) ===")
        a_out, sc_out, iters = nesterov_fft(best_seq, t3, verbose=True,
                                            tau_start=100.0, tau_end=1000.0)
        print(f"  Done: {iters} iters, score={sc_out:.8f}")
        try_update(a_out, f"ns2_n{n_base}")

    # --- Phase 4: Mirror Descent ---
    remaining = time_budget - (time.monotonic() - t_start)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Mirror Descent at n={n_base} ({t4:.0f}s) ===")
        a_out, sc_out = mirror_descent(best_seq, t4, verbose=True)
        print(f"  Done: score={sc_out:.8f}")
        try_update(a_out, f"md_n{n_base}")

    # --- Phase 5: Try symmetric version ---
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 60:
        print(f"\n=== Phase 5: Symmetric exploration ===")
        a_sym = make_symmetric(best_seq)
        t5 = min(remaining * 0.10, 100.0)
        if t5 > 20:
            a_out, sc_out, iters = nesterov_fft(a_sym, t5, verbose=False)
            print(f"  Symmetric NS: {iters} iters, score={sc_out:.8f}")
            try_update(a_out, "ns_sym")

    # --- Phase 6: Multi-length exploration ---
    for n_try in [n_base + 200, n_base - 200, n_base + 400]:
        if n_try < 200:
            continue
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 90:
            break
        print(f"\n=== Phase 6: Nesterov at n={n_try} ===")
        t6 = min(remaining * 0.12, 120.0)
        if t6 < 30:
            continue
        init_try = interpolate_seq(best_seq, n_try)
        a_out, sc_out, iters = nesterov_fft(init_try, t6, verbose=False)
        print(f"  n={n_try}: {iters} iters, score={sc_out:.8f}")
        try_update(a_out, f"ns_n{n_try}")

        # Quick QP at this length
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 40:
            t_qp = min(remaining * 0.05, 50.0)
            if t_qp > 10:
                a_qp, sc_qp = qp_active_refine(a_out, t_qp, verbose=False)
                try_update(a_qp, f"qp_n{n_try}")

    # --- Phase 7: Coordinate descent polishing ---
    remaining = time_budget - (time.monotonic() - t_start)
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Coordinate Descent at n={len(best_seq)} ({t7:.0f}s) ===")
        a_out, sc_out = coordinate_descent(best_seq, t7, n_restarts=2, verbose=False)
        print(f"  Done: score={sc_out:.8f}")
        try_update(a_out, "coord_desc")

    # --- Phase 8: Final QP polish ---
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 20:
        print(f"\n=== Phase 8: Final QP ===")
        t8 = min(remaining * 0.4, 60.0)
        if t8 > 10:
            a_out, sc_out = qp_active_refine(best_seq, t8, verbose=False)
            try_update(a_out, "final_qp")

    # --- Final verification ---
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
