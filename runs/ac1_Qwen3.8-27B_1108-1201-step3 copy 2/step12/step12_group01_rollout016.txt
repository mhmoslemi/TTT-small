def compute_gradient_lse(a, T):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*c_k)) w.r.t. a.
    Returns (value, gradient) where gradient has same shape as a.
    """
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    value = (max_Tc + np.log(exp_Tc.sum())) / T
    
    # Gradient: g_m = 2 * sum_k w_k * a_{k-m} (valid indices)
    # This is 2 * cross-correlation of w and a
    # Using FFT: 
    fft_size = 1 << (2*n - 1 + n - 1 - 1).bit_length()  # need at least 3n-2
    # Actually, let's think: w has length 2n-1, a has length n
    # We want g_m = sum_{k} w_k * a_{k-m} for m = 0,...,n-1
    # k ranges from m to m+n-1 (so that k-m is in [0,n-1])
    # and k is in [0, 2n-2]
    # So k ranges from max(m, 0) to min(m+n-1, 2n-2)
    # Since m >= 0: k from m to min(m+n-1, 2n-2)

def compute_lse_value_and_grad(a, T):
    """
    Compute log-sum-exp value and gradient w.r.t. a.
    F(a) = (1/T) * log(sum_k exp(T * c_k)) where c = a*a

    value = (max_Tc + np.log(exp_Tc.sum())) / T

    # Gradient: g_m = 2 * sum_{k in top} w_k * a_{k-m}
    N = 1 << (3 * n - 1).bit_length()
    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = w_top
    W = np.fft.fft(w_full, N)
    A = np.fft.fft(a, N)
    g_full = np.fft.ifft(W * np.conj(A)).real[:n]
    gradient = 2.0 * g_full

    return value, gradient


def topk_lse_obj_grad_z(z, T, topk):
    """Top-k LSE objective and gradient in z-space."""
    a = softmax_fn(z)
    val, grad_a = compute_topk_lse_value_and_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return val, grad_z


def lbfgs_lse_optimize(a0, time_limit, T_start=10.0, T_end=50000.0, n_stages=20, topk=None):
    """
    L-BFGS on log-sum-exp (full or top-k) with temperature annealing.
    Uses softmax parameterization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 20:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 8:
            break

        # Allocate more iterations for later (higher T) stages
        if si < n_stages // 3:
            maxiter = 300
            gtol = 1e-8
        elif si < 2 * n_stages // 3:
            maxiter = 500
            gtol = 1e-10
        else:
            maxiter = 800
            gtol = 1e-12

        try:
            if topk is not None:
                result = minimize(
                    fun=lambda z: topk_lse_obj_grad_z(z, T, topk),
                    x0=z.copy(),
                    jac=True,
                    method='L-BFGS-B',
                    options={'maxiter': maxiter, 'gtol': gtol, 'maxls': 30}
                )
            else:
                result = minimize(
                    fun=lambda z: lse_obj_grad_z_full(z, T),
                    x0=z.copy(),
                    jac=True,
                    method='L-BFGS-B',
                    options={'maxiter': maxiter, 'gtol': gtol, 'maxls': 30}
                )
            z = result.x
        except Exception:
            pass

        a_new = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        elapsed = time.monotonic() - t0
        print(f"    LSE-LBFGS stage {si+1}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_inner=50):
    """
    Coordinate descent with exact 1D optimization.
    For each coordinate, minimize the max of top peaks via 1D search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    stall_outer = 0

    outer = 0
    while time.monotonic() - t0 < time_limit - 5:
        outer += 1
        c = fast_autocorr(a)
        cmax = c.max()
        # Focus on top peaks
        threshold = cmax * 0.999
        active_peaks = np.where(c >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.argsort(c)[::-1][:5]
        active_peaks = active_peaks[:30]

        improved = False
        for pass_num in range(n_inner):
            for m in range(n):
                if time.monotonic() - t0 > time_limit - 5:
                    break

                a_m = a[m]
                if a_m < 1e-12:
                    continue

                # Compute base autocorrelation (with a_m = 0)
                a_trial = a.copy()
                a_trial[m] = 0.0
                c_base = fast_autocorr(a_trial)

                # (a*a)_k = c_base_k + 2*a_m * a_{k-m} + a_m^2 * [k=2m]
                # For active peaks:
                best_new_m = a_m
                best_new_score = best_score

                # Try to find better a_m by minimizing max over active peaks
                # For each active peak k: f_k(x) = c_base_k + 2*x*a_{k-m} + x^2*[k=2m]
                # We want to find x >= 0 minimizing max_k f_k(x)
                # (ignoring the /sum^2 factor for now, will renormalize)

                def peak_func(x):
                    max_val = 0.0
                    for k in active_peaks:
                        km = k - m
                        if 0 <= km <= n - 1:
                            val = c_base[k] + 2.0 * x * a[km]
                            if k == 2 * m:
                                val += x * x
                        else:
                            val = c_base[k]
                        if val > max_val:
                            max_val = val
                    return max_val

                # Current value
                cur_val = peak_func(a_m)

                # Try: reduce a_m (move towards 0)
                # The optimal x is where the active peaks are balanced
                # Use bisection: find x where the max peak value is minimized
                # The function peak_func(x) is convex in x (max of quadratics with positive leading coeffs)
                # So we can use ternary search or golden section

                # Golden section search on [0, a_m * 2] (allowing increase too)
                lo = 0.0
                hi = max(2.0 * a_m, 1e-6)

                # Check if peak_func is decreasing at a_m (if so, increase)
                # or increasing (if so, decrease)
                test_lo = peak_func(a_m * 0.5)
                test_hi = peak_func(a_m * 1.5)

                if test_lo < cur_val:
                    hi = a_m
                    lo = 0.0
                elif test_hi < cur_val:
                    lo = a_m
                    hi = max(a_m * 3, 1e-6)
                else:
                    lo = a_m * 0.5
                    hi = a_m * 1.5

                # Golden section
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)
                f1 = peak_func(x1)
                f2 = peak_func(x2)

                for _ in range(40):
                    if f1 < f2:
                        hi = x2
                        x2 = x1
                        f2 = f1
                        x1 = hi - gr * (hi - lo)
                        f1 = peak_func(x1)
                    else:
                        lo = x1
                        x1 = x2
                        f1 = f2
                        x2 = lo + gr * (hi - lo)
                        f2 = peak_func(x2)

                best_m = (lo + hi) / 2.0
                best_m = max(0.0, best_m)

                # Apply and check
                a_trial2 = a.copy()
                a_trial2[m] = best_m
                s2 = a_trial2.sum()
                if s2 < 1e-15:
                    continue
                a_trial2 = a_trial2 / s2
                score = 2.0 * n * fast_autocorr(a_trial2).max()

                if score < best_score - 1e-12:
                    a = a_trial2
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            stall_outer += 1
            if stall_outer > 10:
                break
        else:
            stall_outer = 0

        if outer % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent outer {outer} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """
    Nesterov-accelerated LP best-response with adaptive restart.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_long = np.zeros(n)
    n_long = 0
    stall_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        avg_long += a
        n_long += 1

        # Check a
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Check y
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # Check running average periodically
        if n_long % 50 == 0 and n_long >= 100:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        # Adaptive restart
        if stall_count > 500 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            stall_count = 0

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = np.where(np.abs(x) <= 1, 0.5 * (1 + np.cos(np.pi * x)), 0)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (1.0 + 0.3 * np.sin(4 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
