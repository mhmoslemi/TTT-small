"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage LogSumExp optimization in softmax z-space with FFT-based gradients
   - Full smooth gradient (not subgradient) via weighted cross-correlation
   - O(n log n) per gradient evaluation
   - Adaptive temperature annealing
2. Coordinate descent peak shaving on a-space
3. Multi-start with diverse structured initializations
4. Length extension via interpolation + re-optimization
Key innovation: smooth LSE gradient in z-space avoids subgradient noise,
enabling L-BFGS to exploit curvature information effectively.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full autocorrelation a*a via FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.zeros(1)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_objective_and_grad_z(z, T):
    """
    LSE objective and gradient in z-space where a = softmax(z).
    Objective: (1/T)*log(sum(exp(T*b_k)))
    Gradient computed via FFT-based cross-correlation.
    """
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    obj = (Tb_max + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # grad_a[j] = 2 * sum_{k=0}^{n-1} w[j+k] * a[k] (valid indices)
    # Computed as: c = conv(a_flip, w)[n-1 : n-1+n]
    a_flip = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    AF = rfft(a_flip, fft_size)
    WF = rfft(w, fft_size)
    conv_full = irfft(AF * WF, fft_size)
    c = conv_full[n - 1: n - 1 + n]
    grad_a = 2.0 * c

    # Convert to z-gradient: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lse_optimize_multi_stage(a0, total_time, T_start=200.0, T_end=0.1,
                             n_stages=20, max_iter_per_stage=50,
                             verbose=True, label=""):
    """
    Multi-stage LSE optimization in z-space using L-BFGS-B.
    Anneals temperature from T_start to T_end over n_stages.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(a_init)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > total_time:
            break
        remaining = total_time - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_z(z_in):
            val, _ = lse_objective_and_grad_z(z_in, stage_T)
            return val

        def jac_z(z_in):
            _, g = lse_objective_and_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func_z, x0=z, jac=jac_z, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_stage,
                                    'ftol': 1e-18, 'gtol': 1e-12,
                                    'maxfun': max_iter_per_stage * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception as ex:
            if verbose:
                print(f"    LSE stage {si} error: {ex}")

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE[{label}] stage {si+1}/{n_stages} "
                  f"(T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.8f}")

    # Ensure we return at least the initial if nothing improved
    if best_score >= fast_score(a_init):
        best_a = a_init.copy()
        best_score = fast_score(a_init)

    return best_a, best_score


def coordinate_descent_refine(a0, total_time, seed=42, verbose=True, label=""):
    """
    Coordinate descent: reduce top autocorrelation peaks by redistributing mass.
    Uses line search and random restarts to escape local minima.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < total_time:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(8, n)]

        improved = False
        for k_idx in top_k:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-18:
                continue

            src = int(idx[np.argmax(contrib)])

            # Find good destination: low contribution to all top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:4]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                break

            # Pick lowest-a candidates (least impact on other peaks)
            sorted_cands = candidates[np.argsort(a[candidates])[:max(10, len(candidates) // 5)]]
            if len(sorted_cands) == 0:
                continue

            dst = int(sorted_cands[rng.integers(len(sorted_cands))])

            for frac in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 10 == 0:
                n_perturb = rng.integers(1, 5)
                for _ in range(n_perturb):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.03)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        if a_trial[i] >= 0:
                            a_trial = a_trial / a_trial.sum()
                            sc = fast_score(a_trial)
                            if sc < best_score:
                                a = a_trial.copy()
                                best_score = sc
                                best_a = a.copy()
                                stall = 0
                                break

            if stall > 500:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDesc[{label}] ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    elif variant == 4:

        a = x ** 2 + (1 - x) ** 2 + 0.1

        x = np.linspace(0, 2 * np.pi, n)

    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.15

        # Multi-peak random
        a = np.ones(n) * 0.1
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.25)
            amp = rng.uniform(0.5, 2.0)

    elif variant == 11:

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        top_idx = np.argsort(b)[::-1][:k]

            grad[j_start:j_end + 1] += w_m * a[m - j_start:m - j_start + (j_end - j_start + 1)][::-1]

        a_out, _ = lse_optimize_multi_stage(best_seq, t_f1,
                                            T_start=20.0, T_end=0.005,
                                            n_stages=15, max_iter_per_stage=80,
                                            label="final_lse")
        try_update(a_out, "final_lse")

    # Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    t_f2 = min(remaining * 0.35, 100.0)
    if t_f2 > 10:
        a_out, _ = coordinate_descent_refine(best_seq, t_f2, seed=999, label="final_cd")
        try_update(a_out, "final_cd")

    # One more LSE pass
    remaining = time_budget - (time.monotonic() - t_global)
    t_f3 = min(remaining * 0.30, 80.0)
    if t_f3 > 10:
        a_out, _ = lse_optimize_multi_stage(best_seq, t_f3,
                                            T_start=10.0, T_end=0.001,
                                            n_stages=12, max_iter_per_stage=100,
                                            label="final_lse2")
        try_update(a_out, "final_lse2")

    # Final output
    elapsed = time.monotonic() - t_global
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.0f}s")

    if best_seq is not None:
        result = [float(x) for x in best_seq]
        # Sanity check
        assert all(x >= 0 and x < 1000 for x in result)
        assert sum(result) >= 0.01
        return result
    else:
        return [1.0]
