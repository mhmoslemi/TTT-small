        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver via benchmarking."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    best_solver = None
    best_time = float('inf')
    test_n = 200
    a_test = np.ones(test_n) / test_n
    T_test = build_conv_matrix(a_test)
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            b_var = cp.Variable(test_n, nonneg=True)
            M_var = cp.Variable()
            constraints = [T_test @ b_var <= M_var, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate') and elapsed < best_time:
                best_time = elapsed
                best_solver = solver_name
        except Exception:
            continue
    if best_solver is None:
        best_solver = cp.ECOS
    _lp_solver = best_solver
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)*b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = None
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                b_var2 = cp.Variable(n, nonneg=True)
                M_var2 = cp.Variable()
                constraints2 = [T @ b_var2 <= M_var2, cp.sum(b_var2) == 1]
                prob2 = cp.Problem(cp.Minimize(M_var2), constraints2)
                prob2.solve(solver=alt)
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_var = b_var2
                    M_var = M_var2
                    break
            except Exception:
                continue
        else:
            return None, None
    if prob is None:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=100000, verbose=False):
    """Nesterov-accelerated LP best-response iteration with adaptive restart."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    stall_count = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * max(1, it) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0

        # Adaptive restart: if stalled for too long, reduce momentum
        if stall_count > 500:
            y = a.copy()
            stall_count = 0

        if verbose and (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsbfgs_lse(a0, time_limit, T_start=60.0, T_min=0.15, n_stages=30, iters_per_stage=40):
    """Adaptive LogSumExp L-BFGS with progress-based temperature control."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            relative_gap = (sc - best_score) / (best_score + 1e-15)
            if relative_gap > 0.003:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 1)

            if stagnant >= 4:
                T = max(T, T * 1.5)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            if (stage + 1) % 10 == 0:
                print(f"    LSE stage={stage+1} T={T:.2f}: cur={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_shave_second_order(a0, time_limit, n_steps=200000):
    """Peak-directed subgradient with second-order correction and Nesterov momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a
    a = a / a.sum()
    z = z_from_a(a).copy()
    z_prev = z.copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks (top 10% threshold)
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        # Compute subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Project to tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * max(1, step) ** 2)) / 2.0
        alpha_nr = 2.0 / (t_k + 1.0)
        momentum_nr = (t_k - 1.0) / t_k

        # Polyak step
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.3)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Backtracking if needed
        if sc_new > cur_score * 1.005:
            for _ in range(8):
                eta *= 0.5
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                sc_new = score_normalized(a_new)
                if sc_new < cur_score:
                    break

        # Nesterov extrapolation
        z_prev_old = z_prev.copy()
        z = z_new
        a = a_new
        z_prev = z + momentum_nr * (z - z_prev_old)
        z_prev = np.maximum(z_prev, -100)  # Keep logs bounded

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
    elif choice == 4:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    elif choice == 5:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 4 * x) + 0.15 * np.cos(2 * np.pi * 9 * x)
    elif choice == 6:
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 7:
        a = np.exp(-(x - 0.3) ** 2 / 0.03) + 0.5 * np.exp(-(x - 0.7) ** 2 / 0.03)
    elif choice == 8:
        a = 1.0 / (1.0 + 2.0 * np.sin(np.pi * x) ** 2)
    else:
        a = 0.6 + 0.4 * np.cos(2 * np.pi * x) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def annealed_restart_search(best_a, best_score, time_limit, solver, rng):
    """Annealed Nesterov LP restarts: perturb and re-optimize."""
    n = len(best_a)
    t0 = time.monotonic()
    current_best_a = best_a.copy()
    current_best_score = best_score
    temperature = 0.05

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        # Generate perturbation
        scale = temperature * np.sqrt(n) / np.sqrt(n)
        noise = rng.normal(0, scale, n)
        perturbed = best_a + noise
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-15:
            perturbed = best_a + 0.01 * rng.normal(0, 1, n)
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
        perturbed = perturbed / ps

        # Run short Nesterov LP from perturbed start
        sub_time = min(remaining * 0.3, 60.0)
        a_try, sc_try = nesterov_lp(perturbed, sub_time, solver=solver, max_iters=50000)

        if sc_try < current_best_score:
            current_best_score = sc_try
            current_best_a = a_try.copy()
            temperature *= 0.7  # Accept: cool down
            print(f"    Anneal ACCEPT: score={sc_try:.8f} T={temperature:.4f}")
        else:
            # Metropolis acceptance
            delta = sc_try - current_best_score
            if delta < 1e-10:
                temperature = max(0.001, temperature * 0.9)
            else:
                accept_prob = np.exp(-delta / (temperature * 0.01))
                if rng.random() < accept_prob:
                    current_best_a = a_try.copy()
                    temperature = max(0.001, temperature * 0.9)
                else:
                    temperature = max(0.001, temperature * 0.85)

        # If temperature is very low, try a bigger perturbation (escapement)
        if temperature < 0.005:
            temperature = 0.03
            big_noise = rng.normal(0, 0.1, n)
            perturbed = best_a + big_noise
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps > 1e-15:
                perturbed = perturbed / ps
                sub_time = min(remaining * 0.2, 45.0)
                a_try2, sc_try2 = nesterov_lp(perturbed, sub_time, solver=solver, max_iters=30000)
                if sc_try2 < current_best_score:
                    current_best_score = sc_try2
                    current_best_a = a_try2.copy()
                    temperature = 0.02

    return current_best_a, current_best_score


def final_propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective remaining: {global_deadline - time.monotonic():.0f}s")

    # Get initial sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = score_normalized(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = score_normalized(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_normalized(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP main run (35% time) ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score:.8f}) ===")
    t1 = 0.35 * (budget_s - 30)
    a_out, _ = nesterov_lp(best_seq, t1, solver=solver, verbose=True)
    try_update(a_out, "nesterov_main")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: Annealed restarts (20% time) ===
    print(f"\n=== Phase 2: Annealed restarts (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.20 * (budget_s - 30), remaining * 0.5)
    if t2 > 30:
        a_anneal, sc_anneal = annealed_restart_search(best_seq, best_score, t2, solver, rng)
        try_update(a_anneal, "annealed")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: LSE optimization (15% time) ===
    print(f"\n=== Phase 3: LSE (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t3 > 20:
        # Multi-temperature LSE
        a_lse1, _ = lsbfgs_lse(best_seq, t3 * 0.5, T_start=80.0, T_min=0.1, n_stages=25, iters_per_stage=35)
        try_update(a_lse1, "lse_high_T")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_lse2, _ = lsbfgs_lse(best_seq, remaining * 0.6, T_start=15.0, T_min=0.05, n_stages=20, iters_per_stage=30)
            try_update(a_lse2, "lse_low_T")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Peak shaving (10% time) ===
    print(f"\n=== Phase 4: Peak Shaving (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t4 > 15:
        a_ps, _ = peak_shave_second_order(best_seq, t4, n_steps=500000)
        try_update(a_ps, "peak_shave")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.10 * (budget_s - 30), remaining * 0.6)
    if t5_total > 20:
        lengths = [n_base + 300, n_base - 300, n_base + 600]
        for n_try in lengths:
            if n_try < 100:
                continue
            remaining = global_deadline - time.monotonic()
            if remaining < 25:
                break
            t5 = min(t5_total / float(len(lengths)), remaining * 0.45)
            if t5 < 12:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t5 * 0.7, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Final combined polish (10% time) ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        # Short Nesterov
        t6a = remaining * 0.35
        if t6a > 10:
            a_f1, _ = nesterov_lp(best_seq, t6a, solver=solver, max_iters=30000)
            try_update(a_f1, "final_nlp")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Short LSE low T
            t6b = remaining * 0.4
            if t6b > 8:
                a_f2, _ = lsbfgs_lse(best_seq, t6b, T_start=8.0, T_min=0.05, n_stages=15, iters_per_stage=25)
                try_update(a_f2, "final_lse")

        remaining = global_deadline - time.monotonic()
        if remaining > 8:
            # Short peak shave
            t6c = remaining * 0.5
            if t6c > 5:
                a_f3, _ = peak_shave_second_order(best_seq, t6c, n_steps=100000)
                try_update(a_f3, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
