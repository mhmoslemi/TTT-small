    z_prev = z.copy()
    
    # Temperature schedule: geometric from T_start to T_end

    last_improve = 0
    stall_count = 0

        if time.monotonic() - t0 > time_limit:
            break
        if it - last_improve > 10000:
            break
        
        # Current temperature
        frac = min(1.0, it / max(1, n_total // 2))
        T_cur = T_start * (T_end / T_start) ** frac

        max_b = b.max()
        
        # Soft active set weights
        if T_cur > 1.0:
            Tb = T_cur * (b - max_b)
            Tb -= Tb.max()  # already 0 at max
            w = np.exp(Tb)
            w = w / w.sum()
        else:
            # Near-binary: top peaks get more weight
            threshold = max_b * (1.0 - 0.05 / T_cur)
            w = np.where(b >= threshold, 1.0, 0.0)
            if w.sum() < 1e-10:
                w = np.zeros(len(b))
                w[np.argmax(b)] = 1.0
            else:
                w = w / w.sum()

        grad_a = np.zeros(n)
        # For efficiency, only process non-negligible weights
        significant = w > 1e-10
        for k in np.where(significant)[0]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]

def fast_convolve(a):

def compute_gradient_from_weights(w, a):
    """Compute grad_a[j] = sum_k w[k] * a[k-j] using FFT cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return corr


def build_conv_sparse(a_fixed):
    """Build sparse convolution matrix T where T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        js = j
        je = j + n
        for k in range(js, je):
            rows.append(k)
            cols.append(j)
            vals.append(a_fixed[k - j])
    T = sp.csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T

def quick_mw_screen(a0, n_iters=2000):

        if len(topk) == 0:

        w[topk] = 1.0 / len(topk)

def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):

        if t_nest > 1.0:

            y = np.maximum(y, 0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

            best_score = sc_new

        else:
            patience += 1

        # Improved restart: use score history to detect oscillation
        score_history.append(sc_new)
        if len(score_history) > 50:
            score_history.pop(0)

        sc_cur = score_normalized(a)

        if len(score_history) >= 20:

        if time.monotonic() - t0 > time_limit:
            break
        if it - last_improve > 15000:
            break

        # Anneal K
        frac = min(1.0, it / max(1, max_iters * 0.7))
        K_cur = int(K_start + (K_end - K_start) * frac)
        K_cur = max(1, K_cur)

        b = fast_convolve(a)
        max_b = b.max()

        # Top-K peaks with equal weight
        if K_cur >= len(b):
            w = np.ones(len(b)) / len(b)
        else:
            idx = np.argpartition(b, -K_cur)[-K_cur:]
            w = np.zeros(len(b))
            w[idx] = 1.0 / K_cur

        # FFT-based gradient
        grad_a = compute_gradient_from_weights(w, a)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_prev)
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            z_prev = z.copy()
            a = a_new
            z = z_new
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            t_nest = t_new
            last_improve = it
            eta = min(eta * 1.002, 0.35)
        else:
            if sc > best_score * 1.002:
                t_nest = 1.0
                eta *= 0.93
            else:
                z_prev = z.copy()
                t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                t_nest = t_new
                a = a_new
                z = z_new
            eta = max(eta * 0.9997, 0.03)

        if (it + 1) % 3000 == 0:
            print(f"    SoftKW iter={it+1}: score={best_score:.8f} K={K_cur} eta={eta:.4f}")

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient via FFT
    Tb_shifted = T_val * (b - b.max())
    exp_Tb_s = np.exp(Tb_shifted)
    w_s = exp_Tb_s / exp_Tb_s.sum()
    grad_a = compute_gradient_from_weights(w_s, a)
    S = np.dot(grad_a, a)
    grad_z = a * (T_val * grad_a - T_val * S)
    return obj, grad_z


def cyclic_lse(a0, time_limit, n_cycles=3, T_high=150.0, T_low=0.02):
    """
    Cyclic LogSumExp L-BFGS: multiple heating + cooling cycles.
    Each cycle: start at T_high, cool to T_low, then reheat slightly.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    for cycle in range(n_cycles):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        n_stages = 12
        temps = np.geomspace(T_high, T_low, n_stages)
        # Add a slight reheat at the end (except last cycle)
        if cycle < n_cycles - 1:
            temps = np.concatenate([temps, [T_high * 0.3]])

        for si, stage_T in enumerate(temps):
            if time.monotonic() - t0 > time_limit:
                break

            def func_lse(z_val):
                obj, _ = lse_objective_grad(z_val, stage_T, n)
                return obj

            def jac_lse(z_val):
                _, grad = lse_objective_grad(z_val, stage_T, n)
                return grad

            iters = 30 if stage_T > 10 else 50
            try:
                res = minimize(
                    fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                    options={'maxiter': iters, 'ftol': 1e-24,
                             'gtol': 1e-20, 'maxfun': iters * 500,
                             'disp': False}
                )
                z_new = res.x
                a_new = softmax_from_z(z_new)
                sc = score_normalized(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                z = z_new
            except Exception:
                pass

        print(f"    CyclicLSE cycle={cycle+1}/{n_cycles}: best_score={best_score:.8f}")

    return best_a, best_score


def perturb_restart(a, rng, strength=0.05):
    """Perturb a solution to escape local minima."""
    n = len(a)
    noise = rng.normal(0, strength, n)
    a_pert = a + noise
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    a_pert = a_pert / s
    return a_pert


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # ============ PHASE 0: Get initial sequences ============
    best_seq = None
    best_score = float('inf')

    n_base = 1200

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            candidate = p / p.sum()
            sc = evaluate_safe(candidate)
            if np.isfinite(sc):
                best_seq = candidate
                n_base = len(best_seq)
                best_score = sc
                print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    # Screen diverse initializations (quick)
    n_initials = 6
    init_candidates = []
    for i in range(n_initials):
        if best_seq is not None and i == 0:
            init_candidates.append((best_seq, best_score))
        else:
            a_init = diverse_init(n_base, rng, variant=i)
            a_quick, sc_quick = quick_mw_screen(a_init, n_iters=1000)
            init_candidates.append((a_quick, sc_quick))

    # Find the best initialization
    best_init = min(init_candidates, key=lambda x: x[1])
    if best_init[1] < best_score:
        best_seq = best_init[0].copy()
        best_score = best_init[1]
    print(f"Best init: n={n_base}, score={best_score:.8f}")

    if best_seq is None:
        best_seq = diverse_init(n_base, rng, 0)
        best_score = evaluate_safe(best_seq)

    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # ============ PHASE 1: LP Nesterov (main workhorse, ~550s) ============
    print(f"\n=== Phase 1: LP Nesterov (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t1 = min(550.0, remaining * 0.35)
    if t1 > 30:
        a_out, _ = nesterov_lp_adaptive(best_seq, t1, solver=solver)
        try_update(a_out, "lp_nest_1")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # ============ PHASE 2: Soft-K MW with annealed K (~250s) ============
    print(f"\n=== Phase 2: Soft-K MW (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(250.0, remaining * 0.2)
    if t2 > 15:
        a_out2, _ = soft_k_mw(best_seq, t2, K_start=1, K_end=60)
        try_update(a_out2, "soft_k_mw")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # ============ PHASE 3: Cyclic LSE (~350s) ============
    print(f"\n=== Phase 3: Cyclic LSE (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(350.0, remaining * 0.25)
    if t3 > 20:
        a_lse = cyclic_lse(best_seq, t3, n_cycles=3, T_high=100.0, T_low=0.02)
        try_update(a_lse, "cyclic_lse")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # ============ PHASE 4: Perturbation + LP Nesterov restart (~200s) ============
    print(f"\n=== Phase 4: Perturb + LP Nesterov (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(200.0, remaining * 0.18)
    if t4 > 20:
        # Try a perturbed version
        a_pert = perturb_restart(best_seq, rng, strength=0.02)
        sc_pert = score_normalized(a_pert)
        print(f"  Perturbed score: {sc_pert:.8f}")
        a_out4, _ = nesterov_lp_adaptive(a_pert, t4 * 0.7, solver=solver)
        try_update(a_out4, "perturb_lp")

        # Also try on the current best
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            a_out4b, _ = nesterov_lp_adaptive(best_seq, min(60.0, remaining * 0.5), solver=solver)
            try_update(a_out4b, "lp_r2")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # ============ PHASE 5: Final Soft-K + LSE polish (~150s) ============
    print(f"\n=== Phase 5: Final polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 20:
        t5a = min(60.0, remaining * 0.4)
        if t5a > 10:
            a_s5, _ = soft_k_mw(best_seq, t5a, K_start=5, K_end=100)
            try_update(a_s5, "soft_k_final")
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            t5b = min(60.0, remaining * 0.6)
            if t5b > 10:
                a_l5 = cyclic_lse(best_seq, t5b, n_cycles=2, T_high=30.0, T_low=0.01)
                try_update(a_l5, "lse_final")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # ============ PHASE 6: Multi-n exploration ============
    print(f"\n=== Phase 6: Multi-n exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 40:
        n_values = [best_seq.shape[0] - 150, best_seq.shape[0] + 150,
                    best_seq.shape[0] - 300, best_seq.shape[0] + 300]
        n_values = [n for n in n_values if 400 <= n <= 3000]
        # Sort by how close to current best n
        n_values.sort(key=lambda x: abs(x - best_seq.shape[0]))

        for n_try in n_values[:3]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 25:
                break
            t6 = min(45.0, remaining * 0.35)
            if t6 < 10:
                break
            print(f"  Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_adaptive(init_try, t6, solver=solver, max_iters=100000)
            try_update(a_try, f"len_{n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        return [float(x) for x in best_seq]

    # Fallback
    fallback = np.ones(n_base) / n_base
    return [float(x) for x in fallback]
