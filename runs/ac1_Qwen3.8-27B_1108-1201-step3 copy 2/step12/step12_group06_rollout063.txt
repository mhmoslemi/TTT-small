2. QP polishing using dual-weighted quadratic form with FFT-accelerated matrix-vector products

import numpy as np
import cvxpy as cp

from scipy.fft import rfft, irfft


def fast_convolve(a, b=None):
    """Fast linear convolution via FFT. If b is None, computes a*a."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def q_w_multiply(w, a):
    """Compute Q_w @ a in O(n log n) using the Toeplitz/convolution structure.
    Q_w[j,l] = w[j+l], so (Q_w @ a)_j = conv(w, reverse(a))[j + n - 1].
    """
    n = len(a)
    a_rev = a[::-1].copy()
    c = fast_convolve(w, a_rev)
    # c has length len(w) + n - 1 = (2n-1) + n - 1 = 3n - 2
    # We need c[n-1 : n-1+n]
    return c[n - 1: n - 1 + n].copy()


def quadratic_form_w(w, a):
    """Compute a^T Q_w a efficiently."""
    q = q_w_multiply(w, a)
    return float(np.dot(a, q))


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

        lo = max(0, j - n + 1)
        hi = min(n, j + 1)
        T[lo:hi, j] = a_fixed[hi - 1:lo - 1:-1] if lo < hi else 0
    # Actually let's just do it the simple way
    T = np.zeros((size, n))
    for j in range(n):
        for k in range(n):
            T[k + j, j] = a_fixed[k]
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0.
    Returns (b_val, M_val, dual_weights)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    conv_constraints = T @ b_var <= M_var
    sum_constraint = cp.sum(b_var) == 1
    constraints = [conv_constraints, sum_constraint]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, **({'eps_abs': 1e-9, 'eps_rel': 1e-9} if solver == cp.SCS else {}))
    except Exception:
        prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s

    # Extract dual variables (weights over lags)
    dual_w = np.zeros(2 * n - 1)
    try:
        dual_vals = np.asarray(conv_constraints.dual_value, dtype=np.float64)
        dual_vals = np.maximum(dual_vals, 0)
        dsum = dual_vals.sum()
        if dsum > 1e-15:
            dual_w = dual_vals / dsum
        else:
            dual_w = np.ones(2 * n - 1) / (2 * n - 1)
    except Exception:
        # Fallback: use empirical distribution based on convolution peaks
        conv_vals = fast_convolve(a_fixed, b_val)
        dual_w = conv_vals / conv_vals.sum()

    M_val = float(M_var.value) if M_val is not None else float(fast_convolve(a_fixed, b_val).max())
    return b_val, M_val, dual_w


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response iteration. Returns (best_a, best_score, last_dual_w)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    last_dual_w = np.ones(2 * n - 1) / (2 * n - 1)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp, dual_w = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        last_dual_w = dual_w

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")

    return best_a, best_score, last_dual_w


def qp_polish(a0, dual_w, time_limit, max_iters=100000):
    """Polish using the dual-weighted QP: minimize a^T Q_w a projected on simplex.
    Uses FFT-accelerated Q_w @ a computation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Update dual_w periodically
    update_interval = max(10, max_iters // 50)
    eta = 1.0 / (n * 0.5)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient of a^T Q_w a = 2 * Q_w @ a
        grad = 2.0 * q_w_multiply(dual_w, a)

        # Projected gradient step with backtracking
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break

        # Try a few step sizes
        accepted = False
        cur_val = quadratic_form_w(dual_w, a)
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = eta * shrink
            a_trial = a - step * grad
            a_trial = project_simplex(a_trial)
            val_trial = quadratic_form_w(dual_w, a_trial)
            if val_trial < cur_val - 1e-15:
                a = a_trial
                accepted = True
                break

        if not accepted:
            # Try even smaller steps
            for shrink in [0.005, 0.001, 0.0001]:
                step = eta * shrink
                a_trial = a - step * grad
                a_trial = project_simplex(a_trial)
                val_trial = quadratic_form_w(dual_w, a_trial)
                if val_trial < cur_val:
                    a = a_trial
                    accepted = True
                    break
            if not accepted:
                # Stuck, try updating dual
                pass

        # Update dual weights periodically based on current a
        if (it + 1) % update_interval == 0:
            conv_vals = fast_convolve(a)
            # Softmax weighting emphasizing peaks
            max_cv = conv_vals.max()
            if max_cv > 1e-15:
                ratio = conv_vals / max_cv
                # Weight proportional to how close to max
                new_dual = ratio ** 10.0
                ns = new_dual.sum()
                if ns > 1e-15:
                    # Combine with old dual (weighted average)
                    dual_w = 0.5 * dual_w + 0.5 * (new_dual / ns)
                    dual_w = dual_w / dual_w.sum()
                # Also try pure peak weighting
                # dual_w = new_dual / ns

        # Track best
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QP-Polish iter={it+1} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f}")

    return best_a, best_score


def subgradient_refinement(a0, time_limit, max_iters=50000):
    """Subgradient method targeting the max convolution peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0  # Theoretical lower bound
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        # Compute subgradient (average of gradients at active peaks)
        # grad_k[j] = 2 * a[k-j] for valid j
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Project gradient to tangent space
        S = np.dot(g, a)
        grad_proj = a * (g - S)

        grad_norm_sq = np.dot(grad_proj, grad_proj)
        if grad_norm_sq < 1e-25:
            break

        # Polyak step
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm_sq)
        eta = min(eta, 1.0 / n)

        # Step
        z = z_from_a(a) - eta * grad_proj
        a_new = softmax_from_z(z)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        sc_new = score_normalized(a_new)
        if sc_new < cur_score * 1.001:
            a = a_new
        else:
            # Halve step
            z = z_from_a(a) - eta * 0.5 * grad_proj
            a_new = softmax_from_z(z)
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
            if score_normalized(a_new) < cur_score:
                a = a_new

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(5 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.4) ** 2) / 0.08) + 0.5 * np.exp(-((x - 0.7) ** 2) / 0.08)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def active_set_polish(a0, time_limit, max_iters=5000):
    """Polish using active-set LP direction finding.
    At each step, solve a small LP to find the best first-order descent direction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    step_size = 0.1 / n

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Identify top active set (top 10 lags)
        top_k = min(10, 2 * n - 1)
        active = np.argsort(b)[::-1][:top_k]

        # Build LP: minimize t
        # s.t. conv(a,a)_k + 2*(a*d)_k <= t for k in active
        #      conv(a,a)_k + 2*(a*d)_k <= max_b for k not in active
        #      sum(d) = 0
        #      -a_j <= d_j <= step_size
        #
        # (a*d)_k = sum_j a_{k-j} * d_j (valid j)
        # So constraint k: sum_j a_{k-j} * d_j <= (t - conv(a,a)_k) / 2

        d_var = cp.Variable(n)
        t_var = cp.Variable()

        # Build constraint matrix for active set
        # For each k in active: sum_j A[k,j] * d_j <= (t - b[k])/2
        # where A[k,j] = a_{k-j} if valid, else 0

        constraints = [
            cp.sum(d_var) == 0,
            d_var >= -a + 1e-12,
            d_var <= step_size
        ]

        # Add convolution constraints (active + inactive)
        # For efficiency, only add active constraints as tight,
        # and a few nearby inactive ones
        for k in active:
            # Build row: A[k,j] = a_{k-j} for valid j
            row = np.zeros(n)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                row[j] = a[k - j]
            # Constraint: sum_j row[j]*d_j <= (t - b[k])/2
            # i.e., sum_j row[j]*d_j - t/2 <= -b[k]/2
            constraints.append(cp.sum(cp.multiply(row, d_var)) - 0.5 * t_var <= -0.5 * b[k])

        # Add a few inactive constraints (next 10 after active)
        remaining = np.setdiff1d(np.arange(2 * n - 1), active)
        if len(remaining) > 0:
            # Take the ones with highest b values among remaining
            inactive_top = remaining[np.argsort(b[remaining])[::-1][:5]]
            for k in inactive_top:
                row = np.zeros(n)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    row[j] = a[k - j]
                constraints.append(cp.sum(cp.multiply(row, d_var)) - 0.5 * t_var <= 0.5 * (max_b - b[k]))

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        d_val = np.asarray(d_var.value, dtype=np.float64)
        t_val = float(t_var.value)

        # Check if this gives improvement
        if t_val >= max_b * 0.999:
            # No improvement found, shrink step size
            step_size *= 0.5
            if step_size < 1e-12:
                break
            continue

        # Take the step
        a_new = a + d_val
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
        else:
            break

        sc_new = score_normalized(a_new)
        if sc_new < cur_score:
            a = a_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
        else:
            # Step was too aggressive, halve it
            a_new = a + 0.5 * d_val
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
            sc_new = score_normalized(a_new)
            if sc_new < cur_score:
                a = a_new
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a.copy()
            else:
                step_size *= 0.5
                if step_size < 1e-12:
                    break

        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveSet iter={it+1} ({elapsed:.0f}s): score={score_normalized(a):.8f} best={best_score:.8f}")

    return best_a, best_score


def final_propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective time: {global_deadline - time.monotonic():.0f}s")

    best_seq = None
    best_score = float('inf')
    n_base = 1200

    # Get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    global best_seq, best_score

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP (45% time) ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score:.8f}) ===")
    t1 = 0.45 * (budget_s - 30)
    remain = global_deadline - time.monotonic()
    t1 = min(t1, remain * 0.9)
    a_out, _, last_dual = nesterov_lp(best_seq, t1, solver=solver)
    try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: QP Polishing (25% time) ===
    print(f"\n=== Phase 2: QP Polishing (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.25 * (budget_s - 30), remaining * 0.6)
    if t2 > 30 and last_dual is not None:
        # First QP polish pass
        a_qp, _ = qp_polish(best_seq, last_dual, t2 * 0.5, max_iters=200000)
        try_update(a_qp, "qp_polish1")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            # Second QP polish from updated dual
            conv_vals = fast_convolve(best_seq)
            max_cv = conv_vals.max()
            updated_dual = (conv_vals / max_cv) ** 20.0
            updated_dual = updated_dual / updated_dual.sum()
            a_qp2, _ = qp_polish(best_seq, updated_dual, remaining * 0.5, max_iters=100000)
            try_update(a_qp2, "qp_polish2")

    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Subgradient refinement (15% time) ===
    print(f"\n=== Phase 3: Subgradient (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t3 > 20:
        a_sg, _ = subgradient_refinement(best_seq, t3, max_iters=50000)
        try_update(a_sg, "subgradient")

        # Also try from a slightly perturbed starting point
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            perturb = 0.02 * rng.standard_normal(len(best_seq))
            perturbed = best_seq + perturb
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_sg2, _ = subgradient_refinement(perturbed, remaining * 0.5, max_iters=30000)
            try_update(a_sg2, "subgradient_perturbed")

    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Active set polishing (10% time) ===
    print(f"\n=== Phase 4: Active Set Polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.6)
    if t4 > 15:
        a_as, _ = active_set_polish(best_seq, t4, max_iters=2000)
        try_update(a_as, "active_set")

    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Multi-length exploration (5% time) ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.05 * (budget_s - 30), remaining * 0.7)
    if t5_total > 30:
        lengths_to_try = [n_base + 200, n_base + 400]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 20:
                break
            t5 = min(t5_total / 2.0, remaining * 0.45)
            if t5 < 15:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _, _ = nesterov_lp(init_try, t5 * 0.7, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")

            remaining = global_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = qp_polish(a_try, np.ones(2 * n_try - 1) / (2 * n_try - 1),
                                       remaining * 0.5, max_iters=50000)
                try_update(a_try2, f"len_{n_try}_qp")

    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final combined polish (remaining time) ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        # Short Nesterov
        a_f1, _, _ = nesterov_lp(best_seq, remaining * 0.3, solver=solver, max_iters=20000)
        try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # QP polish with updated dual
            conv_vals = fast_conf = fast_convolve(best_seq)
            max_cv = conv_vals.max()
            if max_cv > 0:
                dual_final = (conv_vals / max_cv) ** 30.0
                dual_final = dual_final / dual_final.sum()
            else:
                dual_final = np.ones(2 * len(best_seq) - 1) / (2 * len(best_seq) - 1)
            a_f2, _ = qp_polish(best_seq, dual_final, remaining * 0.5, max_iters=50000)
            try_update(a_f2, "final_qp")

        remaining = global_deadline - time.monotonic()
        if remaining > 8:
            a_f3, _ = subgradient_refinement(best_seq, remaining * 0.6, max_iters=20000)
            try_update(a_f3, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (global_deadline - (budget_s - 30)):.0f}s")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
