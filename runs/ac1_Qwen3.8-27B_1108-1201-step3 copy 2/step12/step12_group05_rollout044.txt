def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_grad_z(z, T):
    """Compute objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectrally shape to flatten power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=25, lr=0.008):
    """LogSumExp-smoothed optimization with Adam optimizer."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(30, min(200, int(remaining * 100 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad = compute_lse_grad_z(z, stage_T)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            lr_t = lr / (1 + 0.002 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000, adaptive_step=True):
    """
    Nesterov-accelerated LP best-response with adaptive step size.
    Tracks upper/lower bounds for convergence detection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    lower_bound = 0.0

    # Adaptive step size parameters
    step_scale = 1.0
    gap_history = []

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        lower_bound = max(lower_bound, M_val)

        # Adaptive step size based on gap
        if adaptive_step:
            score_y = 2.0 * n * fast_autocorr(y).max()
            gap = score_y / max(M_val, 1e-15) - 1.0
            gap_history.append(gap)
            if len(gap_history) > 100:
                gap_history = gap_history[-100:]
            avg_gap = np.mean(gap_history)
            # Increase step when gap is large, decrease when small
            target_scale = np.clip(1.0 / (1.0 + max(avg_gap, 0.0) * 5.0), 0.3, 2.0)
            step_scale = 0.9 * step_scale + 0.1 * target_scale

        alpha_t = 2.0 * step_scale / (t + 2.0)

        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LB={2*n*lower_bound:.6f}, gap={best_score/(2*n*lower_bound)-1:.4f}")

    return best_a, best_score

def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-12:
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def equate_peaks_step(a, n_top=5, n_iter=200):
    """
    Iterative peak equalization: find top autocorrelation peaks and
    adjust a to make them more equal (water-filling in real space).
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for it in range(n_iter):
        b = fast_autocorr(a)
        top_idx = np.argsort(b)[::-1][:n_top]
        peak_vals = b[top_idx]
        peak_mean = peak_vals.mean()

        # For each top peak, find which a_j contributes most and reduce it
        for k in top_idx:
            if b[k] < peak_mean:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.sum() < 1e-30:
                continue
            # Find the maximum contributor
            src_pos = np.argmax(contrib)
            src = idx[src_pos]
            # Move a small amount to a low-value position far from this peak
            excess = b[k] - peak_mean
            if excess < 1e-15:
                continue
            # Find a good destination
            dist = np.abs(np.arange(n) - src)
            # Prefer far positions
            far_mask = dist > n // 4
            if far_mask.sum() > 0:
                far_indices = np.where(far_mask)[0]
                # Pick the one with lowest a value
                dst = far_indices[np.argmin(a[far_indices])]
            else:
                dst = (src + n // 2) % n

            # Compute how much to move
            # Reducing a[src] by delta reduces b[k] by approximately 2*a[k-src]*delta
            reduction_rate = 2.0 * a[k - src] if 0 <= k - src < n else 2.0 * a[src - k + n - 1]
            if reduction_rate < 1e-15:
                continue
            delta = min(excess / reduction_rate * 0.5, a[src] * 0.02)
            if delta < 1e-15:
                continue

            a_trial = a.copy()
            a_trial[src] -= delta
            a_trial[dst] += delta
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization with improved time allocation:
    Phase 1: Spectral shaping (adaptive, fast)
    Phase 2: Adam LSE at base length
    Phase 3: LP Nesterov (main workhorse, most time)
    Phase 4: Peak equalization
    Phase 5: Second LP round from perturbed start
    Phase 6: Peak shaving + final polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    print(f"Starting optimization at n={n_base}")

    # Phase 1: Adaptive spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Adaptive spectral shaping (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.05, 40.0)
    if t1 > 5:
        a_out = spectral_shaping_adaptive(best_seq, n_iter=2000)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [spectral]: {best_score:.6f}")
        # But keep the better of original vs spectral for next phase
        a_for_next = a_out if sc < evaluate_safe(best_seq) or sc <= best_score else best_seq
        # Actually just use whichever is better
        sc_best = evaluate_safe(best_seq)
        if sc < sc_best:
            a_for_next = a_out
        else:
            a_for_next = best_seq
        a_for_next = a_for_next / a_for_next.sum()
    else:
        a_for_next = best_seq.copy()

    # Phase 2: Adam LSE
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Adam LSE (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_out, _ = adam_lse_optimize(a_for_next, t2, T_start=80.0, T_end=0.02, n_stages=25, lr=0.006)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [adam_lse]: {best_score:.6f}")
        a_for_next = a_out if sc < evaluate_safe(best_seq) else best_seq
        a_for_next = a_for_next / a_for_next.sum()
    else:
        a_for_next = best_seq.copy()

    # Phase 3: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP Nesterov (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.40, 500.0)
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(a_for_next, t3, solver, max_iters=500000, adaptive_step=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [lp_nesterov]: {best_score:.6f}")
        a_for_next = a_out if sc < evaluate_safe(best_seq) else best_seq
        a_for_next = a_for_next / a_for_next.sum()
    else:
        a_for_next = best_seq.copy()

    # Phase 4: Peak equalization
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak equalization (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 60.0)
    if t4 > 10:
        n_peq_iter = min(2000, int(t4 * 5))
        a_out = equate_peaks_step(a_for_next, n_top=8, n_iter=n_peq_iter)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [peak_eq]: {best_score:.6f}")
        a_for_next = a_out if sc < evaluate_safe(best_seq) else best_seq
        a_for_next = a_for_next / a_for_next.sum()
    else:
        a_for_next = best_seq.copy()

    # Phase 5: Second LP round (perturbed restart)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Second LP round (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 30:
        # Perturb the current best
        perturbation = 0.05
        a_perturbed = a_for_next + perturbation * rng.standard_normal(n_base)
        a_perturbed = np.maximum(a_perturbed, 0)
        a_perturbed = a_perturbed / a_perturbed.sum()
        a_out, _ = nesterov_lp_optimize(a_perturbed, t5, solver, max_iters=300000, adaptive_step=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [lp_perturbed]: {best_score:.6f}")
        a_for_next = a_out if sc < evaluate_safe(best_seq) else best_seq
        a_for_next = a_for_next / a_for_next.sum()
    else:
        a_for_next = best_seq.copy()

    # Phase 6: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 120.0)
    if t6 > 10:
        a_out, _ = peak_shaving(a_for_next, t6, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [peak_shave]: {best_score:.6f}")
        a_for_next = a_out if sc < evaluate_safe(best_seq) else best_seq
        a_for_next = a_for_next / a_for_next.sum()

    # Phase 7: Final short LP polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final LP polish (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.30, 100.0)
    if t7 > 15:
        a_out, _ = nesterov_lp_optimize(a_for_next, t7, solver, max_iters=100000, adaptive_step=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [final_lp]: {best_score:.6f}")

    # Phase 8: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 5:
            a_out, _ = peak_shaving(best_seq, t8, seed=789)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  *** NEW BEST [final_peak_shave]: {best_score:.6f}")

    # Phase 9: Final spectral micro-adjustment
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 5:
        a_out = spectral_shaping_adaptive(best_seq, n_iter=200)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [final_spectral]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
