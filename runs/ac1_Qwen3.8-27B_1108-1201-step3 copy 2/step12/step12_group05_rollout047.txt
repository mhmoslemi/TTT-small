from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):

def compute_max_gradient(a, top_k=3, weight_temp=5.0):
    """
    Compute the gradient of the top-k weighted max of conv(a,a).
    Returns (gradient, peak_positions, peak_values).
    Gradient of b_k w.r.t. a_m = 2*a_{k-m} if 0<=k-m<n, else 0.
    """
    n = len(a)
    b = fast_autocorr(a)
    
    # Get top-k peaks
    top_indices = np.argsort(b)[::-1][:top_k]
    peak_vals = b[top_indices]
    
    # Exponential weighting (soft-max over top-k)
    weights = np.exp(weight_temp * (peak_vals - peak_vals[0]))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_indices):
        # Valid m range: 0 <= m <= n-1 and 0 <= k-m <= n-1
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            m_range = np.arange(m_lo, m_hi + 1)
            grad[m_range] += weights[i] * 2.0 * a[k - m_range]
    
    return grad, top_indices, peak_vals


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    mu = np.maximum(v - theta, 0)
    return mu


def nesterov_max_descent(a0, time_limit, top_k=5, weight_temp=5.0,
                          lr_init=None, restart_window=50, max_iters=500000):
    """
    Nesterov accelerated gradient descent on the top-k weighted max autocorrelation.
    Uses exact sparse gradient with adaptive step size and restarts.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr_init is None:
        lr_init = 1.0 / (4.0 * n)
    
    lr = lr_init
    y = a.copy()
    x_prev = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    f_history = []
    last_restart = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Evaluate at Nesterov lookahead point y
        b_y = fast_autocorr(y)
        top_k_actual = min(top_k, 2 * n - 1)
        grad_y, peaks_y, vals_y = compute_max_gradient(y, top_k_actual, weight_temp)
        
        f_y = float(np.sum(vals_y * np.exp(weight_temp * (vals_y - vals_y[0]))) /
                    np.sum(np.exp(weight_temp * (vals_y - vals_y[0]))))
        
        # Nesterov momentum
        if t == 1:
            y_new = y - lr * grad_y
        else:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = y + momentum * (y - x_prev) - lr * grad_y
        
        # Project to simplex
        x_new = project_simplex(y_new)
        
        # Evaluate score at x_new
        b_new = fast_autocorr(x_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = x_new.copy()
        
        # Nesterov restart condition
        f_x_prev = 2.0 * n * fast_autocorr(x_prev).max()
        f_y_val = 2.0 * n * fast_autocorr(y).max()
        if f_y_val - f_x_prev > -10.0 * f_new_to_prev := score_new / (2.0 * n) - f_x_prev / (2.0 * n):
            # Simplified restart: if we haven't improved in restart_window iterations
            if t - last_restart > restart_window:
                y = x_prev.copy()
                last_restart = t
                lr = min(lr * 1.5, 1.0 / (2.0 * n))
        
        # Adaptive step size: reduce if no improvement recently
        f_history.append(score_new)
        if len(f_history) > 100:
            f_history.pop(0)
        
        if t % 50 == 0 and len(f_history) > 50:
            recent = f_history[-25:]
            if max(recent) - min(recent) < 1e-10:
                lr *= 0.7
        
        # Update
        x_prev = x_new
        y = y_new
        
        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NAG t={t} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6e}")
    
    return best_a, best_score


def nesterov_max_descent_v2(a0, time_limit, top_k=5, weight_temp=5.0,
                             lr_init=None, max_iters=500000):
    """
    Improved Nesterov with proper restart logic and multi-scale step adjustment.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr_init is None:
        lr_init = 2.0 / (n + 10.0)
    
    lr = lr_init
    y = a.copy()
    x_prev = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    no_improve_count = 0
    best_in_window = best_score
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Gradient at Nesterov lookahead point
        b_y = fast_autocorr(y)
        top_k_act = min(top_k, 2 * n - 1)
        grad_y, _, _ = compute_max_gradient(y, top_k_act, weight_temp)
        
        # Nesterov step
        if t == 1:
            y_new = y - lr * grad_y
        else:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = y + momentum * (y - x_prev) - lr * grad_y
        
        x_new = project_simplex(y_new)
        
        # Fast score check
        b_new = fast_autocorr(x_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = x_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Periodic best-in-window tracking and restart
        if t % 100 == 0:
            if score_new < best_in_window:
                best_in_window = score_new
                no_improve_count = 0
            elif t % 500 == 0 and best_in_window >= best_score * (1 - 1e-8):
                # Stuck - restart Nesterov
                y = best_a.copy()
                x_prev = best_a.copy()
                lr = min(lr * 1.3, 5.0 / (n + 10.0))
                best_in_window = best_score
                no_improve_count = 0
        
        x_prev = x_new
        y = y_new
        
        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NAG2 t={t} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6e}, stall={no_improve_count}")
    
    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def spectral_shaping_v2(a, n_iter=1000, alpha=0.2, balance=0.5):
    """
    Improved spectral shaping with balance between peak reduction and smoothness.
    Uses a target spectrum that's a geometric mean of flat and current.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        # Target: geometric mean of flat and current (balance parameter controls)
        non_dc_current = B[1:]
        flat_val = np.sum(B) / len(B)
        
        # Target spectrum: interpolation between flat and current
        target = np.power(flat_val, balance) * np.power(non_dc_current, 1.0 - balance)
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.1, 10.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=200000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    a_prev = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s
        
        y = y_new
        a_prev = a
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-NAG t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving: directly reduce top autocorrelation peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) < 2:
                continue
            contrib = a[idx] * a[k - idx]
            
            src = idx[np.argmax(contrib)]
            
            # Find destination: minimize impact on other peaks
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Try the smallest-value candidates first
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            
            for dst in dst_pool[:5]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 200:
                break
    
    return best_a, best_score


def multi_seed_nesterov(a0, time_limit, n_seeds=3, top_k_range=[3, 5, 7]):
    """
    Run Nesterov max-gradient with different random perturbations and parameters.
    Returns the best result across all runs.
    """
    n = len(a0)
    rng = np.random.default_rng(42)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_autocorr(a0).max()
    t0 = time.monotonic()
    
    time_per_seed = time_limit / n_seeds
    
    for seed_i in range(n_seeds):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        t_alloc = min(time_per_seed, remaining)
        if t_alloc < 10:
            break
        
        # Create perturbed starting point
        a_start = a0.copy() * (1 + rng.normal(0, 0.01, n))
        a_start = np.maximum(a_start, 0)
        s = a_start.sum()
        if s > 1e-15:
            a_start = a_start / s
        
        top_k = top_k_range[seed_i % len(top_k_range)]
        weight_temp = 3.0 + seed_i * 2.0
        
        a_out, sc = nesterov_max_descent_v2(a_start, t_alloc, top_k=top_k,
                                             weight_temp=weight_temp)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
    
    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=15, lr=0.01):
    """
    LogSumExp-smoothed optimization with Adam optimizer.
    Uses z-parameterization (a = softmax(z)) to maintain positivity.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        iters = max(30, min(200, int(remaining * 100 / (n_stages - si))))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_fn(z)
            A = rfft(a, fft_size)
            b = irfft(A * A, fft_size)[:2 * n - 1]
            
            Tb = T_start * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # Gradient via convolution
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]
            
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_t = lr / np.sqrt(1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 20 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-Adam stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Direct Nesterov max-gradient (primary, most effective)
    Phase 2: Spectral shaping (frequency-domain global improvement)
    Phase 3: Second Nesterov round from spectral result
    Phase 4: LP zero-sum game (powerful but slow)
    Phase 5: LSE-Adam (smooth optimization)
    Phase 6: Peak shaving (fine polish)
    Phase 7: Final Nesterov + spectral polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)
    
    # Phase 1: Direct Nesterov max-gradient (most effective, allocate 35%)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Nesterov max-gradient at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_out, sc = nesterov_max_descent_v2(best_seq, t1, top_k=5, weight_temp=5.0)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [nesterov1]: {best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 2: Spectral shaping (allocate 10%)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Spectral shaping at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        a_out = spectral_shaping_v2(best_seq, n_iter=800, alpha=0.15, balance=0.5)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [spectral1]: {best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 3: Second Nesterov round from spectral result (allocate 20%)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Nesterov round 2 (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 280.0)
    if t3 > 20:
        # Multi-seed for diversity
        a_out, sc = multi_seed_nesterov(best_seq, t3, n_seeds=3, top_k_range=[3, 5, 7])
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [nesterov2]: {best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 4: LP zero-sum game (allocate 20%)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_out, sc = nesterov_lp_optimize(best_seq, t4, solver)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [lp_nesterov]: {best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 5: LSE-Adam (allocate 10%)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: LSE-Adam at n={n_base} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_out, sc = lse_adam_optimize(best_seq, t5, T_start=50.0, T_end=1.0, n_stages=12, lr=0.005)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [lse_adam]: {best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 6: Peak shaving (allocate 10%)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        a_out, sc = peak_shaving(best_seq, t6, seed=456)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [peak_shave1]: {best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 7: Final polished Nesterov (remaining time / 2)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final Nesterov polish (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.5, 100.0)
    if t7 > 10:
        a_out, sc = nesterov_max_descent_v2(best_seq, t7, top_k=7, weight_temp=10.0)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [final_nesterov]: {best_score:.8f} (t={elapsed:.0f}s)")
    
    # Phase 8: Final spectral + peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final polish (remaining={remaining:.0f}s) ===")
    t8a = min(remaining * 0.4, 40.0)
    if t8a > 5:
        a_out = spectral_shaping_v2(best_seq, n_iter=300, alpha=0.05, balance=0.6)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [final_spectral]: {best_score:.8f} (t={elapsed:.0f}s)")
    
    t8b = min((time_budget - (time.monotonic() - t_global_start)) * 0.5, 30.0)
    if t8b > 5:
        a_out, sc = peak_shaving(best_seq, t8b, seed=789)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [final_shave]: {best_score:.8f} (t={elapsed:.0f}s)")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.maximum(best_seq, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
