def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    a_pad = np.asarray(a, dtype=np.float64)
    if len(a_pad) < fft_size:
        a_pad = np.pad(a_pad, (0, fft_size-len(a_pad)), 'constant')
    A = rfft(a_pad, fft_size)
    return irfft(A * A, fft_size)[:size]
