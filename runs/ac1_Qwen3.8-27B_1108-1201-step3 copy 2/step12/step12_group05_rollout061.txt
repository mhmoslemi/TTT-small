def fast_autocorr(a):

def solve_lp_best_response(a_fixed, M, solver):

"""

2. Spectral shaping for initialization and reformation

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time
from scipy.fft import rfft, irfft


def fast_conv(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * b.max()


def normalize(a):
    """Normalize to simplex (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(arr):
    """Safely evaluate a sequence using the actual evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def spectral_shaping(a, n_iter=800, alpha=0.25):
    """Spectral shaping: flatten power spectrum to reduce convolution peaks."""
    n = len(a)
    a = normalize(a.copy())
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = compute_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_sparse_conv_matrix(a_fixed):
    """Build sparse (2n-1) x n matrix T where (T@b)[k] = sum_j a[k-j]*b[j]."""
    n = len(a_fixed)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        if je >= js:
            js_arr = np.arange(js, je + 1)
            rows.extend(k * len(js_arr))
            cols.extend(js_arr.tolist())
            vals.extend((a_fixed[k - js_arr]).tolist())
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_scipy(a_fixed, M):
    """
    Solve LP: maximize sum(b) s.t. T(a) @ b <= M, b >= 0
    using scipy HiGHS solver.
    Returns (b* normalized, optimal_value) or (None, None).
    """
    n = len(a_fixed)
    T = build_sparse_conv_matrix(a_fixed)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')

    if res.status != 0:
        return None, None

    b_val = np.asarray(res.x, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    opt_val = -res.fun
    return b_val, float(opt_val)


def nesterov_lp_optimize(a0, time_limit, max_iters=100000):
    """
    Nesterov-accelerated LP best-response iteration using scipy HiGHS.
    """
    n = len(a0)
    a = normalize(a0.copy())
    y = a.copy()

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    linspace_check = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_conv = fast_conv(a)
        M = b_conv.max()
        if M < 1e-15:
            break

        b_star, opt_val = solve_lp_scipy(a, M)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        score_a = compute_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = compute_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = compute_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, "
                  f"LP_val={opt_val:.4f}")

    return best_a, best_score


def lse_optimize(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=20, lr=0.005):
    """
    LogSumExp-smoothed gradient optimization with Adam in softmax parameterization.
    """
    n = len(a0)
    a_init = normalize(a0.copy())
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = compute_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(10, min(120, int(remaining * 60 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            n_a = len(a)
            b = fast_conv(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            fft_size = 1 << (2 * n_a - 1).bit_length()
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n_a - 1:n_a - 1 + n_a]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 20 == 0:
                a_check = softmax_fn(z)
                score = compute_score(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a = softmax_fn(z)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Enhanced coordinate-descent peak shaving with multi-coefficient moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        top_k = np.argsort(b)[::-1][:12]
        max_b = b[top_k[0]]

        improved = False
        for k in top_k[:6]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_indices = set()
            for k2 in top_k[:8]:
                js2 = max(0, k2 - (n - 1))
                je2 = min(n - 1, k2)
                peak_indices.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_indices]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = normalize(a_trial)
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Try multi-coefficient move: reduce 2 peaks simultaneously
            if len(top_k) >= 4:
                k1, k2 = top_k[0], top_k[1]
                js1 = max(0, k1 - (n - 1))
                je1 = min(n - 1, k1)
                idx1 = np.arange(js1, je1 + 1)
                contrib1 = a[idx1] * a[k1 - idx1]
                src1 = idx1[np.argmax(contrib1)]

                js2 = max(0, k2 - (n - 1))
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                contrib2 = a[idx2] * a[k2 - idx2]
                src2 = idx2[np.argmax(contrib2)]

                if src1 != src2 and a[src1] > 1e-10 and a[src2] > 1e-10:
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        a_trial = a.copy()
                        amt1 = a_trial[src1] * frac
                        amt2 = a_trial[src2] * frac
                        a_trial[src1] -= amt1
                        a_trial[src2] -= amt2
                        d1 = rng.integers(n)
                        d2 = rng.integers(n)
                        a_trial[d1] += amt1
                        a_trial[d2] += amt2
                        a_trial = normalize(a_trial)
                        score = compute_score(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

            if not improved:
                stall += 1
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.04)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-15:
                        a_trial = normalize(a_trial)
                        score = compute_score(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

                if stall > 500:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):

    print(f"\n=== Phase 6: Peak shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 5:
        a_out, sc = peak_shaving(best_seq, t6, seed=789)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [peak_n{len(best_seq)}]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 7: Another LP round on best found
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final LP polish at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.30, 350.0)
    if t7 > 20:
        a_out, sc = nesterov_lp_optimize(best_seq, t7)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [lp_final_n{len(best_seq)}]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 8: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final peak shaving (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.30, 100.0)
    if t8 > 5:
        a_out, sc = peak_shaving(best_seq, t8, seed=1011)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [peak_final]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 9: One more LP if time remains
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t9 = min(remaining * 0.5, 150.0)
        print(f"\n=== Phase 9: Extra LP ({t9:.0f}s) ===")
        a_out, sc = nesterov_lp_optimize(best_seq, t9)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [extra_lp]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Final peak shaving with whatever time is left
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 5:
        t10 = min(remaining * 0.6, 50.0)
        if t10 > 3:
            a_out, sc = peak_shaving(best_seq, t10, seed=3333)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                elapsed = time.monotonic() - t_global_start
                print(f"  *** NEW BEST [peak_extra]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Final verification with actual evaluator
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 1e-15:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
