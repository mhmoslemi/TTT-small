    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix_rows(a_fixed, row_indices):
    """Build specific rows of the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    rows = np.zeros((len(row_indices), n))
    for i, k in enumerate(row_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            rows[i, lo:hi+1] = a_fixed[k-hi:k-lo+1]
    return rows


def get_lp_solver():
    """Find the best available LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_active(a_fixed, solver, max_constraints=200):
    """Solve LP with active-set constraints for speed."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Identify active constraints using current max pattern
    T_full = build_conv_matrix_rows(a_fixed, np.arange(size))
    b_init = np.ones(n) / n
    conv_init = T_full @ b_init
    threshold = conv_init.max() * 0.85
    active = np.where(conv_init >= threshold)[0]
    if len(active) > max_constraints:
        top_idx = np.argsort(conv_init[active])[-max_constraints:]
        active = active[top_idx]

    T_active = T_full[active]
    constraints = [T_active @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Fallback: full LP
        constraints = [T_full @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_lp_full(a_fixed, solver):
    """Solve the full LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_rows(a_fixed, np.arange(size))
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver, max_iters=100000, use_active=False):
    """Nesterov-accelerated LP best-response iteration with convergence detection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Convergence tracking
    prev_score = best_score
    stall_count = 0
    stall_threshold = 0.0001
    restart_count = 0
    max_restarts = 5

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if use_active:
            b_new, M_lp = solve_lp_active(y, solver)
        else:
            b_new, M_lp = solve_lp_full(y, solver)

        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        # Convergence detection
        improvement = prev_score - best_score
        if improvement < stall_threshold:
            stall_count += 1
        else:
            stall_count = 0
        prev_score = best_score

        # Restart if stalled too long
        if stall_count > 200 and restart_count < max_restarts:
            restart_count += 1
            stall_count = 0
            # Perturb and restart Nesterov
            rng_local = np.random.default_rng(it)
            perturb = best_a + 0.02 * rng_local.normal(0, 1, n)
            perturb = np.maximum(perturb, 0)
            perturb = perturb / perturb.sum()
            a = perturb
            y = perturb.copy()
            it = 0  # Reset Nesterov counter
            t_k = 1.0

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_polish(a0, time_limit, T_start=50.0, T_min=0.05, n_stages=40, iters_per_stage=50):
    """Adaptive LogSumExp L-BFGS with progress-based temperature control."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0
    prev_score = best_score

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            improvement = prev_score - sc
            if improvement < 1e-6:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 2)

            if stagnant >= 4:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            prev_score = sc

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_subgradient_fast(a0, time_limit, n_steps=500000):
    """Peak-targeted subgradient with adaptive step and averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0
    prev_score = best_score
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks
        threshold = max_b * 0.97
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.3)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.005:
            # Bisection
            lo_e, hi_e = 0.0, eta
            for _ in range(4):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        # Moving average
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 3000:
            avg_count = 3000
            avg_a = (avg_a * 3000 + a) / 3001

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
            stall = 0

        if stall > 5000:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.02)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining Nesterov LP, LSE polish, and multi-length exploration."""
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    global_deadline = total_start + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remain={remain:.0f}s)")

    # === Phase 1: Nesterov LP with active set (55% time) ===
    print(f"\n=== Phase 1: Nesterov LP active-set at n={n_base} ===")
    t1 = 0.55 * (budget_s - 20)
    a_out, sc_out = nesterov_lp(best_seq, t1, solver, use_active=True)
    try_update(a_out, "nesterov_active")
    print(f"  Phase 1 score: {best_score:.8f}")

    # === Phase 2: Nesterov LP full (10% time) ===
    remaining = global_deadline - time.monotonic()
    t2 = min(0.10 * (budget_s - 20), remaining * 0.4)
    if t2 > 30:
        print(f"\n=== Phase 2: Nesterov LP full ===")
        a_out2, _ = nesterov_lp(best_seq, t2, solver, use_active=False)
        try_update(a_out2, "nesterov_full")
        print(f"  Phase 2 score: {best_score:.8f}")

    # === Phase 3: LSE polish (20% time) ===
    remaining = global_deadline - time.monotonic()
    t3 = min(0.20 * (budget_s - 20), remaining * 0.5)
    if t3 > 30:
        print(f"\n=== Phase 3: LSE Polish ===")
        # First LSE pass with moderate temperature
        a_lse1, _ = lse_polish(best_seq, t3 * 0.5, T_start=80.0, T_min=0.1, n_stages=30, iters_per_stage=45)
        try_update(a_lse1, "lse_pass1")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            # Second LSE pass from low temperature for fine polishing
            a_lse2, _ = lse_polish(best_seq, remaining * 0.5, T_start=15.0, T_min=0.05, n_stages=25, iters_per_stage=40)
            try_update(a_lse2, "lse_pass2")

        # LSE from perturbed point
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            perturbed = best_seq + 0.03 * np.random.default_rng(777).normal(0, 1, len(best_seq))
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_lse3, _ = lse_polish(perturbed, remaining * 0.5, T_start=30.0, T_min=0.1, n_stages=20, iters_per_stage=35)
            try_update(a_lse3, "lse_perturbed")

        print(f"  Phase 3 score: {best_score:.8f}")

    # === Phase 4: Peak subgradient (10% time) ===
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 20), remaining * 0.5)
    if t4 > 20:
        print(f"\n=== Phase 4: Peak Subgradient ===")
        a_sg, _ = peak_subgradient_fast(best_seq, t4)
        try_update(a_sg, "peak_sg")
        print(f"  Phase 4 score: {best_score:.8f}")

    # === Phase 5: Multi-length exploration (3% time) ===
    remaining = global_deadline - time.monotonic()
    t5 = min(0.03 * (budget_s - 20), remaining * 0.6)
    if t5 > 30:
        print(f"\n=== Phase 5: Multi-length ===")
        for n_try in [n_base + 200, n_base - 200]:
            remaining = global_deadline - time.monotonic()
            if remaining < 30:
                break
            t5_sub = min(t5 / 2.0, remaining * 0.4)
            if t5_sub < 15:
                break
            print(f"  Trying n={n_try}")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t5_sub * 0.6, solver, use_active=True, max_iters=50000)
            try_update(a_try, f"len_{n_try}")
            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = lse_polish(a_try, remaining * 0.5, T_start=20.0, T_min=0.1, n_stages=10, iters_per_stage=30)
                try_update(a_try2, f"len_{n_try}_lse")

    # === Phase 6: Final combined polish (2% time) ===
    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 6: Final Polish ===")
        # Quick Nesterov
        a_f1, _ = nesterov_lp(best_seq, remaining * 0.4, solver, use_active=True, max_iters=30000)
        try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Quick LSE at very low T
            a_f2, _ = lse_polish(best_seq, remaining * 0.5, T_start=5.0, T_min=0.03, n_stages=15, iters_per_stage=30)
            try_update(a_f2, "final_lse_low")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Quick subgradient
            a_f3, _ = peak_subgradient_fast(best_seq, remaining * 0.6, n_steps=100000)
            try_update(a_f3, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    total_time = time.monotonic() - total_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {total_time:.0f}s")
    return [float(x) for x in best_seq]
