Now, for the spectral approach. The idea is:
1. Compute the DFT of a: A = DFT(a)
2. The power spectrum is P = |A|^2
3. We want P to be as flat as possible
4. But we need a >= 0

One approach: 
- Start with a target flat power spectrum
- Find the non-negative a that best matches it

This is related to the "phase retrieval" or "spectral factorization" problem, which is hard in general.

A simpler approach: iterative projection.
1. Start with some a >= 0.
2. Compute A = DFT(a), P = |A|^2.
3. Modify P to be flatter (e.g., clip the peaks of P, or apply a concave function).
4. Recover a new power spectrum P'.
5. Find a' >= 0 with |DFT(a')|^2 ≈ P'. (This is the hard part.)

For step 5, one approach is to keep the phase of A and adjust the magnitude:
A' = A * (sqrt(P') / |A|) (where |A| > 0)
a' = IDFT(A')
Then project to a' >= 0.

This is similar to the Gerchberg-Saxton algorithm (used in phase retrieval).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Temperature-annealed softmax subgradient descent (multiplicative weights)
   with proper O(n log n) gradient computation via FFT
2. Spectral power-spectrum flattening (iterative magnitude adjustment)
3. Multi-length exploration (n = 800 to 4000)
4. Diverse initializations + parent sequence
5. Final coordinate-descent polish on top peaks
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

# --- Core computation functions ---

def fast_conv(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A*A, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = fast_conv(a)
    return 2.0*n*b.max()/(s*s)

def compute_grad(a, w):
    """grad[i] = 2 * sum_j w[i+j]*a[j] (cross-correlation of w and a)"""
    n = len(a)
    a_flipped = a[::-1]
    fft_size = 1 << (3*n).bit_length()
    AF = rfft(a_flipped, fft_size)
    WF = rfft(w, fft_size)
    conv = irfft(AF * WF, fft_size)
    return 2.0 * conv[n-1 : 2*n-2]

def softmax_descent(a0, time_limit, T_start=50.0, T_end=0.1, 
                    eta0=0.05, decay=0.6, seed=42, verbose=False):
    """Temperature-annealed softmax subgradient descent on the simplex."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0).astype(float)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)/n
    else:
        a = a/s
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    # Estimate max iterations
    # Each iteration is O(n log n), roughly 1ms for n=2000
    # So we can do about 1000 iterations per second for n=2000
    est_iters_per_sec = 5000000 / n  # rough estimate
    
    for t in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Temperature
        frac = min(1.0, t / (est_iters_per_sec * time_limit * 0.8))
        T = T_start * (T_end / T_start) ** frac
        
        b = fast_conv(a)
        Tb = T * b
        Tb -= Tb.max()
        w = np.exp(Tb)
        w /= w.sum()
        
        grad = compute_grad(a, w)
        
        # Step size
        eta = eta0 / (1.0 + t / (est_iters_per_sec * time_limit * 0.1)) ** decay
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a + 0.05*rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        a = a_new / s_new
        
        # Periodic score check
        interval = 2 if t < 500 else (10 if t < 5000 else 30)
        if t % interval == 0:
            sc = score_of(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and t % 10000 == 0:
            print(f"      t={t}, T={T:.2f}, score={best_score:.6f}")
    
    return best_a, best_score

def peak_polish(a0, time_limit, seed=42):
    """Coordinate descent: reduce the top autocorrelation peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0).astype(float)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        # Find the argmax
        k_max = int(np.argmax(b))
        
        # The peak at k_max is b[k_max] = sum_j a[j]*a[k_max-j]
        # To reduce it, move mass from contributing positions
        js = np.arange(max(0, k_max-n+1), min(n-1, k_max)+1)
        contrib = a[js] * a[k_max - js]
        
        if contrib.sum() < 1e-15:
            break
        
        # Move mass from the highest contributor
        src = int(js[np.argmax(contrib)])
        
        # Find a good destination: low a value, far from peak
        # Try random positions
        improved = False
        for _ in range(50):
            dst = int(rng.integers(n))
            if dst == src:
                continue
            for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    break
                a_trial = a_trial / a_trial.sum()
                sc = score_of(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.05)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_of(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                elif rng.random() < 0.1:
                    # Occasionally accept worse solutions (simulated annealing)
                    a = a_trial
    
    return best_a, best_score

def interp_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-10:
        return np.ones(new_n)/new_n
    a_norm = a/s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss < 1e-10:
        a_new = np.ones(new_n)/new_n
    else:
        a_new = a_new/ss
    return a_new
