def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

4. Update: `b_new = project_simplex(b - eta * g)`

This is O(n log n) per step! And we only need ~10-50 steps for a good approximation.

    c = conv_matvec_combined(a_fixed, b)
    max_c = c.max()

    lower_bound = np.sum(a_fixed * b)  # This is a lower bound on max_c? No...

    # So max(c) >= sum(c)/(2n-1) = sum(a_fixed)/(2n-1)
    lb = sum(a_fixed) / (2 * n - 1)

        threshold = max_c - 0.01 * max_c  # 99% of max
        active = np.where(c >= threshold)[0]

            # d/d b_i [(a_fixed * b)_k] = a_fixed[k-i] if 0<=k-i<n

        if gap < 1e-12:

            break

        eta = min(eta, 1.0)  # cap
