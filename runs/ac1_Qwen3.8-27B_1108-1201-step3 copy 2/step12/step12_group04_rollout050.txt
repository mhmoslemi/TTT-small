import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    b = fast_autocorr(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * b.max() / (s * s)

def lse_gradient(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*b_k)) w.r.t. a."""

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Short Nesterov LP best-response for initialization
2. LogSumExp optimization with Adam optimizer and FFT-based gradients (main workhorse)
3. Active-set peak equalization via LP
4. Fine projected gradient local search
5. Multi-length exploration and final polishing
Key innovations: Adam in log-domain with adaptive LR, FFT-based gradient for O(n log n)
per iteration, aggressive multi-stage annealing, and direct peak-targeting LP.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    b = fast_autocorr(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * b.max() / (s * s)


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_gradient(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*b_k)) w.r.t. a.
    grad_a[j] = 2 * sum_m W[m+j] * a[m]
    Computed via FFT in O(n log n).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    W = exp_Tb / exp_Tb.sum()

    # grad_a[j] = 2 * (W * a_rev)[n-1+j]
    a_rev = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1  # 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    Wf = rfft(W, fft_size)
    arev_f = rfft(a_rev, fft_size)
    conv = irfft(Wf * arev_f, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, W


def lse_value(a, T):
    """Compute LogSumExp value (for monitoring)."""
    b = fast_autocorr(a)
    Tb = T * b
    return (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum())) / T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def adam_lse_optimize(a0, time_limit, T_start=300.0, T_end=0.005, n_stages=60):

    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    else:
        a = a / s

    # Initialize z in log-domain
    z = np.log(np.maximum(a, 1e-30)).copy()
    # Center z (invariance of softmax to constant shifts)
    z = z - z.mean()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # More iterations at later (colder) stages
        progress = (si + 1) / n_stages
        iters = max(30, min(300, int(100 + 200 * progress)))

        # Learning rate scales inversely with temperature
        base_lr = 0.5 / (1.0 + T * 0.01)

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a_cur = softmax_fn(z)
            grad_a, W = lse_gradient(a_cur, T)

            # Transform to z-space
            S = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - S)

            t_adam = si * 10000 + it + 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            # Adaptive LR: decay within stage
            lr = base_lr / (1.0 + 0.005 * it)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            # Keep z centered
            z = z - z.mean()

            # Periodic evaluation

                a_check = softmax_fn(z)
                sc = compute_score(a_check)

                            a_trial = a_trial / ss
                            sc = compute_score(a_trial)
                            if sc < best_score:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                stall = 0
                                break
                stall += 1
                if stall > 200:
                    break

        if (stall // 100) > 0 and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FLR stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
