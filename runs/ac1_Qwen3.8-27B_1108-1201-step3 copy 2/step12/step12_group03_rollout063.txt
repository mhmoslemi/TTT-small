The subgradient of max_k conv(a,a)[k] is 2*shift_{k*}(a), enabling fast O(n log n) iterations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def geometric_seq(n, c=1.0):
    """Generate a geometric sequence a[j] = C*(1-c/n)^j normalized to sum=1."""
    r = 1.0 - c / n
    if r <= 0:
        r = 0.5
    a = r ** np.arange(n)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def triangular_seq(n):
    """Generate a triangular (tent) sequence normalized to sum=1."""
    mid = n // 2
    if n % 2 == 0:
        a = np.concatenate([np.arange(1, mid + 1), np.arange(mid, 0, -1)])
    else:
        a = np.concatenate([np.arange(1, mid + 1), [mid + 1], np.arange(mid, 0, -1)])
    s = a.sum()
    return a / s


def raised_cosine_seq(n):
    """Generate a raised cosine (Hann-like) window normalized to sum=1."""
    j = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * j / n))
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s


def onesided_exp_seq(n, c=2.0):
    """One-sided exponential decay from the left, normalized."""
    j = np.arange(n)
    a = np.exp(-c * j / n)
    s = a.sum()
    return a / s


def twosided_exp_seq(n, c=2.0):
    """Two-sided exponential decay from center, normalized."""
    mid = (n - 1) / 2.0
    j = np.arange(n)
    a = np.exp(-c * np.abs(j - mid) / n)
    s = a.sum()
    return a / s


def nesterov_subgradient_optimize(a0, time_limit, verbose=True, top_k=5, concentration=5.0):
    """
    Nesterov-accelerated subgradient method for min max_k conv(a,a)[k] s.t. sum(a)=1, a>=0.
    
    Much faster than LP (O(n log n) per step vs O(n^omega)).
    Uses top-K peaks with concentration weights for smoother gradients.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_b = fast_autocorr(a)
    best_score = 2.0 * n * best_b.max()
    
    y = a.copy()  # Nesterov look-ahead point
    t0 = time.monotonic()
    iter_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation of y
        by = fast_autocorr(y)
        max_by = by.max()
        
        # Top-K peaks with concentration weights
        sorted_idx = np.argsort(by)[::-1][:top_k]
        weights = np.exp(concentration * (by[sorted_idx] - max_by))
        weights = weights / weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            # subgrad of conv(a,a)[k] = 2 * a shifted
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += 2.0 * w * y[other]
            grad[other] += 2.0 * w * y[idx]
        
        # Nesterov step size
        alpha_t = 2.0 / (t + 2.0)
        
        # Nesterov update: y_new = (1-alpha)*y - alpha*grad (normalized)
        y_new = y - alpha_t * grad
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new
        
        # Momentum step for a
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
                s_a = 1.0
            a_new = a_new / s_a
        
        y = y_new
        a = a_new
        iter_count += 1
        
        # Check score periodically (every 50 iters to save FFT cost)
        if iter_count % 50 == 0 or iter_count < 100:
            ba = fast_autocorr(a)
            score_a = 2.0 * n * ba.max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            
            score_y = 2.0 * n * max_by
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
        
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NS iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}")
    
    # Final check
    ba = fast_autocorr(a)
    score_a = 2.0 * n * ba.max()
    if score_a < best_score:
        best_score = score_a
        best_a = a.copy()
    
    return best_a, best_score, iter_count


def lp_best_response(a_fixed):
    """
    Solve LP: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0.
    Uses multiplicative weights as a fast approximation to avoid LP solver overhead.
    """
    n = len(a_fixed)
    # Use multiplicative weights for the column player
    # Build convolution matrix T (2n-1) x n implicitly
    b = np.ones(n) / n  # initial distribution
    M = 0.0
    
    # Just use argmax as best response (greedy)
    # This is faster and often sufficient
    conv_val = fast_autocorr(a_fixed)
    k_star = np.argmax(conv_val)
    
    # Return the "best response" as a delta at the contributing positions
    # For the game interpretation, the best response is the lag that maximizes
    return k_star, conv_val[k_star]


def coordinate_descent_refine(a0, time_limit, verbose=False, seed=42):
    """
    Aggressive coordinate descent with:
    - Top peak identification
    - Contribution analysis
    - Bisection line search for optimal move amount
    - Multiple move types (transfer, split, merge)
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * n
    a = a / s
    
    best_a = a.copy()
    best_b = fast_autocorr(a)
    best_score = 2.0 * n * best_b.max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Top 5 peaks
        n_peaks = min(5, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        
        for k in top_peaks:
            # Get contribution pairs for this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue
            
            # Top contributors
            c_sorted = np.argsort(contrib)[::-1][:8]
            
            for ci in c_sorted:
                src = idx[ci]
                src_other = k - src
                
                # Try moving mass from src (and maybe src_other) to other positions
                # Score candidate destinations by distance from peak-contributing positions
                
                # Build set of "bad" positions (those that contribute to top peaks)
                bad_positions = set()
                for k2 in top_peaks[:3]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    bad_positions.update(range(js2, je2 + 1))
                
                # Candidate destinations: positions NOT in bad set, or with low a[j]
                candidates = []
                for j in range(n):
                    if j == src and src_other == src:
                        continue
                    if j in bad_positions:
                        continue
                    # Score: prefer low a[j] (easy to add mass) and far from peaks
                    score_j = -a[j] * n
                    candidates.append((score_j, j))
                
                candidates.sort(reverse=True)
                
                # Try top 15 candidates with bisection
                for _, dst in candidates[:15]:
                    if dst == src and dst == src_other:
                        continue
                    
                    # Determine max movable amount
                    max_amt = a[src] * 0.5
                    if src != src_other:
                        max_amt = min(max_amt, a[src_other] * 0.25)
                    
                    # Bisection line search: find best fraction to move
                    lo, hi = 0.0, 1.0
                    best_frac = 0.0
                    best_trial_score = best_score
                    
                    for _ in range(12):
                        mid = (lo + hi) / 2.0
                        amt = max_amt * mid
                        if amt < 1e-15:
                            break
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        if src != src_other:
                            a_trial[src_other] -= amt * 0.3
                        a_trial[dst] += amt * (1.3 if src != src_other else 1.0)
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            break
                        a_trial = a_trial / s_t
                        
                        score_trial = 2.0 * n * fast_autocorr(a_trial).max()
                        if score_trial < best_trial_score:
                            best_trial_score = score_trial
                            best_frac = mid
                            lo = mid
                        else:
                            hi = mid
                    
                    if best_frac > 0.05 and best_trial_score < best_score - 1e-14:
                        amt = max_amt * best_frac
                        a[src] -= amt
                        if src != src_other:
                            a[src_other] -= amt * 0.3
                        a[dst] += amt * (1.3 if src != src_other else 1.0)
                        a = np.maximum(a, 0)
                        s_a = a.sum()
                        if s_a < 1e-15:
                            break
                        a = a / s_a
                        best_score = best_trial_score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation every 5 stalls
            if stall % 5 == 0:
                n_pert = rng.integers(2, 6)
                sources = rng.choice(n, size=n_pert, replace=False)
                targets = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for src_p, dst_p in zip(sources, targets):
                    frac = rng.uniform(0.005, 0.03)
                    amt = a_trial[src_p] * frac
                    a_trial[src_p] -= amt
                    a_trial[dst_p] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    sc = 2.0 * n * fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            
            if stall > max_stall:
                # Try larger random jumps
                if stall > max_stall * 3:
                    break
                # Big perturbation
                n_pert = rng.integers(5, 20)
                sources = rng.choice(n, size=n_pert, replace=False)
                targets = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for src_p, dst_p in zip(sources, targets):
                    frac = rng.uniform(0.01, 0.1)
                    amt = a_trial[src_p] * frac
                    a_trial[src_p] -= amt
                    a_trial[dst_p] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    sc = 2.0 * n * fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_score


def simulated_annealing_refine(a0, time_limit, seed=123, T_start=0.01, T_end=1e-6):
    """
    Simulated annealing to escape local minima.
    Moves: random mass redistribution between 2-4 positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    
    # Estimate iterations possible
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        # Random move: pick 2-4 source positions and redistribute
        n_move = rng.integers(2, 5)
        sources = rng.choice(n, size=n_move, replace=False)
        amounts = np.zeros(n_move)
        for i in range(n_move):
            amounts[i] = a[sources[i]] * rng.uniform(0.05, 0.3)
        
        targets = rng.choice(n, size=n_move, replace=False)
        
        a_trial = a.copy()
        for src, dst, amt in zip(sources, targets, amounts):
            amt_actual = min(amt, a_trial[src])
            a_trial[src] -= amt_actual
            a_trial[dst] += amt_actual
        
        a_trial = np.maximum(a_trial, 0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        
        score_trial = 2.0 * n * fast_autocorr(a_trial).max()
        
        if score_trial < best_score:
            best_score = score_trial
            best_a = a_trial.copy()
        
        # Accept/reject
        if score_trial <= best_score or rng.random() < np.exp(-(score_trial - best_score) / max(T, 1e-10)):
            a = a_trial
        
        n_iters += 1
        if n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA iter={n_iters} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.2e}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def final_coordinate_polish(a0, time_limit, seed=999):
    """
    Very fine-grained coordinate descent for final polishing.
    Uses smaller steps and more careful evaluation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Precompute peak structure
    b = fast_autocorr(a)
    top_peaks = np.argsort(b)[::-1][:3]
    
    # For each pair of adjacent positions, try small transfers
    step_sizes = [0.001, 0.0005, 0.0002, 0.0001]
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:3]
        
        improved = False
        
        # Try all pairs of positions (for smaller n) or top contributors
        if n <= 500:
            positions = range(n)
        else:
            # Only try positions with significant mass
            positions = np.where(a > a.max() * 0.01)[0]
            if len(positions) > 200:
                positions = np.argsort(a)[::-1][:200]
        
        for i in positions:
            if a[i] < 1e-10:
                continue
            # Check if i contributes to any top peak
            contributes = False
            for k in top_peaks:
                if 0 <= k - i < n and a[k - i] > 1e-10:
                    contributes = True
                    break
            if not contributes:
                continue
            
            for j in positions:
                if j == i:
                    continue
                for step in step_sizes:
                    a_trial = a.copy()
                    a_trial[i] -= step
                    a_trial[j] += step
                    if a_trial[i] < -1e-12 or a_trial[j] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                break
        if not improved and time.monotonic() - t0 > time_limit * 0.5:
            break
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization combining:
    1. Multi-shape initialization
    2. Fast Nesterov subgradient (bulk of optimization)
    3. Aggressive coordinate descent
    4. Simulated annealing
    5. Length exploration
    6. Final polishing
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)  # Leave buffer
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 0: Get parent and determine base length
    parent = None
    n_base = 1600
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_base = len(parent)
    except Exception:
        pass
    
    elapsed = time.monotonic() - t_global_start
    print(f"Initial: best_score={best_score:.6f}, n_base={n_base}, elapsed={elapsed:.1f}s")
    
    # Phase 1: Try multiple initializations (30s)
    print(f"\n=== Phase 1: Multi-shape initialization (30s) ===")
    init_time = min(30.0, time_budget * 0.05)
    t1_end = t_global_start + init_time
    
    inits = []
    for name, seq in [
        ("geometric_c1", geometric_seq(n_base, c=1.0)),
        ("geometric_c2", geometric_seq(n_base, c=2.0)),
        ("onesided_exp", onesided_exp_seq(n_base, c=2.0)),
        ("twosided_exp", twosided_exp_seq(n_base, c=2.0)),
        ("triangular", triangular_seq(n_base)),
        ("raised_cos", raised_cosine_seq(n_base)),
    ]:
        elapsed = time.monotonic() - t_global_start
        if elapsed > t1_end:
            break
        # Quick Nesterov warm-up (200 iterations)
        a_warm, sc_warm, _ = nesterov_subgradient_optimize(seq, 2.0, verbose=False, top_k=3, concentration=3.0)
        try_update(a_warm, f"init_{name}")
        inits.append(a_warm)
    
    # Also try parent with quick Nesterov
    if parent is not None:
        elapsed = time.monotonic() - t_global_start
        if elapsed < t1_end:
            a_warm, sc_warm, _ = nesterov_subgradient_optimize(parent, 5.0, verbose=False, top_k=5, concentration=5.0)
            try_update(a_warm, "init_parent_ns")
            inits.append(a_warm)
    
    # Use best as starting point for main optimization
    start_a = best_seq.copy()
    
    # Phase 2: Main Nesterov subgradient optimization (40% of time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Main Nesterov Subgradient (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 650.0)
    if t2 > 30:
        print(f"  Time allocated: {t2:.0f}s")
        a_out, sc_out, iters = nesterov_subgradient_optimize(start_a, t2, verbose=True, top_k=5, concentration=5.0)
        print(f"  NS done: {iters} iters, score={sc_out:.6f}")
        try_update(a
