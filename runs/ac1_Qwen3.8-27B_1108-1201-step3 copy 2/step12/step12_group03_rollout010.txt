def binomial_init(n):

def solve_lp_best_response(a_fixed, solver, max_iter=50000):

def mirror_descent_refine(a0, time_limit, lr_start=0.05, lr_end=0.0001, n_iters=None):
    """Mirror descent with entropy kernel on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    if n_iters is None:
        n_iters = 10000

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / max(1, n_iters)
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()

        # Subgradient: weighted average over top peaks
        n_top = min(8, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:n_top]
        gaps = max_b - b[top_k]
        weights = np.exp(-gaps * 20.0 / max(max_b, 1e-15))
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights = weights / w_sum

        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += w * a[other]
            grad[other] += w * a[idx]
        grad *= 2.0

        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_s = a_new.sum()
        if a_new_s < 1e-30:
            break
        a_new = a_new / a_new_s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if it % 2000 == 0:
            print(f"      MD it={it}: best={best_score:.6f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine starting length
    n_base = len(best_seq) if best_seq is not None else 1600

    # Create binomial initialization for comparison
    binom_init = binomial_init(n_base)
    binom_score = evaluate_safe(binom_init)
    print(f"Binomial init: n={n_base}, score={binom_score:.6f}")

    # Use the better of parent and binomial as starting point
    if parent is None:
        best_seq = binom_init
        best_score = binom_score
    elif binom_score < best_score:
        best_seq = binom_init
        best_score = binom_score
        print(f"  Using binomial init as better starting point")

    # Also try a few different initializations and pick the best
    inits_to_try = [best_seq]

    # Try a "smoothed" version: mix binomial with the LP-optimized parent
    if parent is not None:
        mixed = 0.5 * binom_init + 0.5 * parent
        mixed = np.maximum(mixed, 0)
        mix_s = mixed.sum()
        if mix_s > 1e-15:
            mixed = mixed / mix_s
            inits_to_try.append(mixed)

    # Try a slightly different shape: a^p for different p
    for p in [0.7, 1.3]:
        shaped = binom_init ** p
        shaped = np.maximum(shaped, 0)
        sh_s = shaped.sum()
        if sh_s > 1e-15:
            shaped = shaped / sh_s
            inits_to_try.append(shaped)

    # Evaluate all inits and pick best
    for i, init in enumerate(inits_to_try):
        sc = evaluate_safe(init)
        if sc < best_score:
            best_score = sc
            best_seq = init.copy()
            print(f"  Better init {i}: score={sc:.6f}")

    print(f"Starting optimization: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t1, solver)
        print(f"  LP Nesterov done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_base}")

        # Second round of LP Nesterov (shorter, to exploit improvements)
        remaining = time_budget - (time.monotonic() - t_global_start)
        t1b = min(remaining * 0.15, 250.0)
        if t1b > 40:
            print(f"\n=== Phase 1b: LP Nesterov round 2 (remaining={remaining:.0f}s) ===")
            a_out2, sc_out2, iters2 = nesterov_lp_optimize(best_seq, t1b, solver, verbose=False)
            print(f"  LP round 2: {iters2} iters, score={sc_out2:.6f}")
            try_update(a_out2, f"lp2_n{n_base}")

    # === Phase 2: QP Active Set Refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: QP Refinement (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_out, sc_out = qp_active_set_refinement(best_seq, t2, n_active=25)
        print(f"  QP done: score={sc_out:.6f}")
        try_update(a_out, f"qp_n{n_base}")

    # === Phase 3: Mirror Descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Mirror Descent (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        a_out, sc_out = mirror_descent_refine(best_seq, t3, lr_start=0.03, lr_end=0.0001)
        print(f"  MD done: score={sc_out:.6f}")
        try_update(a_out, f"md_n{n_base}")

    # === Phase 4: Multi-length exploration ===
    for n_try in [n_base - 200, n_base + 100, n_base + 300, n_base - 400]:
        if n_try < 400:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 150:
            break
        print(f"\n=== Phase 4: LP Nesterov at n={n_try} (remaining={remaining:.0f}s) ===")
        t4 = min(remaining * 0.18, 180.0)
        if t4 < 30:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp_optimize(init_try, t4, solver, verbose=False)
        print(f"  LP at n={n_try}: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_try}")

        # Quick QP at this length
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 40:
            t_qp = min(remaining * 0.08, 50.0)
            if t_qp > 10:
                a_qp, sc_qp = qp_active_set_refinement(a_out, t_qp, n_active=20)
                try_update(a_qp, f"qp_n{n_try}")

    # === Phase 5: Final polishing ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Final Polish at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.3, 120.0)
    if t5 > 10:
        a_out, sc_out = coordinate_descent_polish(best_seq, t5, seed=999)
        print(f"  Coord descent: score={sc_out:.6f}")
        try_update(a_out, "coord_final")

    # Final LP Nesterov short burst
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t6 = min(remaining * 0.5, 100.0)
        if t6 > 15:
            print(f"\n=== Phase 6: Final LP burst (remaining={remaining:.0f}s) ===")
            a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t6, solver, verbose=False)
            try_update(a_out, "lp_final_burst")

    # Final QP
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t7 = min(remaining * 0.6, 60.0)
        if t7 > 5:
            a_out, sc_out = qp_active_set_refinement(best_seq, t7, n_active=15)
            try_update(a_out, "qp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
