**Better approach: Use the LP to get a good starting point, then use a second-order method for refinement.**

The refinement problem is: minimize `max_k f_k(a)` where `f_k(a) = (a*a)[k]` (with `sum a = 1`, `a >= 0`).

This can be reformulated as: minimize `t` subject to `f_k(a) <= t` for all k, `sum a = 1`, `a >= 0`.

The KKT conditions give us: at the optimum, there exist multipliers `λ_k >= 0` such that `grad_a (sum_k λ_k f_k(a)) - ν * 1 = 0` (for the equality constraint `sum a = 1`), with complementary slackness `λ_k * (t - f_k(a)) = 0`.

So at the optimum, the gradient of the weighted sum of active quadratic forms is constant (equal to ν*1). This gives us a system of equations that we can solve iteratively.

**Algorithm: Active Set Newton**
1. Start with a good `a` (from Nesterov LP)
2. Identify active set: `A = {k : f_k(a) >= t - ε}`
3. Solve: find `a_new` and `t_new` that minimize `t` subject to `f_k(a) <= t` for `k in A`, `sum a = 1`, `a >= 0`
   (This is a QP!)
4. Update active set, repeat

The QP in step 3 has a nice structure: the constraints are quadratic (each `f_k` is a quadratic form in `a`). So it's a quadratically constrained quadratic program (QCQP), which is non-convex in general.

However, if we linearize the constraints around the current point, we get a convex QP:
`f_k(a + d) ≈ f_k(a) + grad_k(a)^T d`
so the constraint becomes `f_k(a) + grad_k(a)^T d <= t_new`.

This gives a linearly constrained QP, which is convex! And we can solve it efficiently.

This is essentially a "sequential quadratic programming" (SQP) approach applied to our problem.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Top-k LogSumExp L-BFGS with aggressive temperature annealing

4. Multi-length exploration and final polishing

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def symmetric_constraint(a):
    """Enforce palindromic symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a[i]
        a_sym[n - 1 - i] = a[i]
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, use_symmetry=False):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    if use_symmetry:
        a = symmetric_constraint(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        y_eval = symmetric_constraint(y) if use_symmetry else y
        b_new, M_lp = solve_lp_best_response(y_eval, solver)
        if b_new is None:
            continue
        if use_symmetry:
            b_new = symmetric_constraint(b_new)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        if use_symmetry:
            a_new = symmetric_constraint(a_new)

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        if use_symmetry:
            y_new = symmetric_constraint(y_new)

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def topk_lse_objective_grad(z_in, T_val, n, top_k=20):
    """Compute Top-k LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)

    # Select top-k peaks
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[top_idx]
    else:
        b_top = b

    max_bt = b_top.max()
    if max_bt > 700:
        max_bt = 700
    exp_bt = np.exp(T_val * b_top - T_val * max_bt)
    w = exp_bt / exp_bt.sum()
    log_sum_exp = max_bt + np.log(exp_bt.sum()) / T_val
    obj = 2.0 * n * log_sum_exp

    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        wv = T_val * w[idx_i]
        if wv < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += wv * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsebfgs_topk(a0, time_limit, T_start=50.0, T_min=0.1, n_stages=30,
                 iters_per_stage=50, top_k_start=5, top_k_end=50):
    """Top-k LogSumExp L-BFGS with adaptive temperature."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0
    prev_score = best_score

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Interpolate top_k
        frac = stage / max(1, n_stages - 1)
        top_k = max(top_k_start, int(top_k_start + frac * (top_k_end - top_k_start)))

        def func_staged(z_val):
            obj, _ = topk_lse_objective_grad(z_val, T, n, top_k)
            return obj

        def jac_staged(z_val):
            _, grad = topk_lse_objective_grad(z_val, T, n, top_k)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            if prev_score - sc > 1e-10:
                stagnant = 0
                T = max(T_min, T * 0.7)
            else:
                stagnant += 1
                if stagnant >= 3:
                    T = max(T_min, T * 0.9)
                    stagnant = 0

            prev_score = sc
        except Exception:
            T = max(T_min, T * 0.8)

        if (stage + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TopK-LSE stage={stage+1}/{n_stages} T={T:.3f} k={top_k}: score={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=10):
    """Direct peak equalization using trust-region subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    step_size = 0.1

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find top peaks
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        b_top = b[top_idx]
        w_top = (b_top - b_top.min()) / max(1e-15, b_top.max() - b_top.min())
        w_top = w_top / w_top.sum()

        # Weighted gradient
        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w_top[i] * 2.0 * a[k - idx]

        # Project to tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)
        grad_norm = np.linalg.norm(grad_z)

        if grad_norm < 1e-15:
            break

        # Adaptive step with backtracking
        eta = step_size
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.001:
            # Backtrack
            for _ in range(10):
                eta *= 0.5
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                sc_new = score_normalized(a_new)
                if sc_new < cur_score:
                    break

        if eta < 1e-10:
            step_size *= 0.5
        else:
            step_size = min(0.5, step_size * 1.02)

        z = z_new
        a = a_new

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def hankel_solve(lam, rhs, n, max_cg_iter=500, tol=1e-10):
    """Solve L @ x = rhs where L[j,l] = lam[j+l] using CG.
    Falls back to dense solve if CG fails.
    """
    def matvec(x):
        result = np.zeros(n)
        for j in range(n):
            result[j] = np.dot(x, lam[j:j+n])
        return result

    # CG
    x = np.zeros(n)
    r = rhs - matvec(x)
    p = r.copy()
    rs_old = np.dot(r, r)

    if rs_old < tol * tol:
        return x

    for i in range(max_cg_iter):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-30:
            break
        alpha_cg = rs_old / pAp
        x += alpha_cg * p
        r -= alpha_cg * Ap
        rs_new = np.dot(r, r)
        if rs_new < tol * tol:
            break
        p = r + (rs_new / rs_old) * p
        rs_old = rs_new

    # Check convergence
    if np.dot(r, r) > tol * tol:
        # Fallback: regularized dense solve
        L = np.zeros((n, n))
        for j in range(n):
            L[j, :] = lam[j:j+n]
        try:
            x = np.linalg.solve(L + 1e-8 * np.eye(n), rhs)
        except np.linalg.LinAlgError:
            x = np.linalg.lstsq(L + 1e-8 * np.eye(n), rhs, rcond=None)[0]

    return x


def active_set_fixed_point(a0, time_limit, tol=1e-8, max_iters=100, epsilon=1e-4):
    """Active-set fixed-point iteration exploiting Hankel structure.
    
    At the optimum, the KKT conditions require:
    (sum lambda_k S_k) @ a = (nu/2) * 1
    where S_k are the anti-diagonal matrices for the active peaks.
    
    This gives a = L^{-1} 1 / (1^T L^{-1} 1) where L = sum lambda_k S_k (Hankel).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Identify active set (peaks within epsilon of max)
        threshold = max_b * (1.0 - epsilon)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > 50:
            top_idx = np.argsort(b[active])[-50:]
            active = active[top_idx]

        # Compute gradients g_k = S_k @ a for active peaks
        # g_k[j] = a[k-j] if valid (for the convolution structure)
        # Actually: (S_k @ a)[j] = a[k - (j - (n-1))] ... let me use the direct formula
        # f_k(a) = sum_i a[i] * a[k-i] (with bounds)
        # grad f_k[j] = a[k-j] + a[j-k] (with bounds) ... 
        # Actually grad f_k[j] = sum over l where the term involves a[j]:
        # f_k = sum_i a[i]*a[k-i], so d/d a[j] = a[k-j] (if k-j in range) + a[j-k] (if j-k in range)
        # But since f_k = sum_i a[i]*a[k-i], the only term involving a[j] is when i=j (giving a[k-j])
        # Wait no: f_k = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
        # d f_k / d a[j] = a[k-j] (if 0<=k-j<n) [from the i=j term]
        #                    + a[j-(k-j)] ... no
        # Actually: f_k = sum_i a[i]*a[k-i], treating a as a vector
        # d f_k / d a[j] = a[k-j] * (1 if 0<=k-j<n) + a[j] * (1 if 0<=j-j<n) ... 
        # I'm confusing myself. Let me use the matrix form.
        # f_k(a) = a^T S_k a where S_k[j,l] = 1 if j+l = k (0-indexed, with j,l in [0,n-1])
        # grad f_k = 2 * S_k @ a (since S_k is symmetric)
        # (S_k @ a)[j] = sum_l S_k[j,l] * a[l] = a[k-j] if 0 <= k-j < n, else 0
        
        m = len(active)
        # Build the Hankel "lam" vector: lam[k] = lambda_k for active k, 0 otherwise
        # First, solve for lambda: find λ in simplex such that sum λ_k g_k is as constant as possible
        
        # Compute g_k for each active peak
        # g_k[j] = a[k-j] if 0 <= k-j <= n-1
        G = np.zeros((m, n))
        for i, k in enumerate(active):
            for j in range(n):
                val = k - j
                if 0 <= val < n:
                    G[i, j] = a[val]
                else:
                    # Also check the "other side" for symmetric convolution
                    # Actually S_k[j,l] = 1 iff j+l=k, so (S_k @ a)[j] = a[k-j]
                    pass
        
        # We want: sum_i λ_i G[i] = c * 1 (constant vector)
        # Minimize ||sum_i λ_i (G[i] - mean(G[i]))||^2 s.t. sum λ = 1, λ >= 0
        # This is: min ||H λ||^2 s.t. 1^T λ = 1, λ >= 0
        # where H[i] = G[i] - G[i].mean()
        
        H = G - G.mean(axis=1, keepdims=True)
        # QP: min (λ^T H^T H λ) s.t. 1^T λ = 1, λ >= 0
        # Use scipy minimize for small m
        if m <= 1:
            lam_opt = np.ones(m) / max(m, 1)
        else:
            def qp_obj(lam_v):
                Hv = H.T @ lam_v
                return np.dot(Hv, Hv)
            
            def qp_grad(lam_v):
                return 2.0 * H @ (H.T @ lam_v)
            
            from scipy.optimize import minimize as sp_minimize
            lam_init = np.ones(m) / m
            res = sp_minimize(qp_obj, lam_init, jac=qp_grad, method='L-BFGS-B',
                             bounds=[(0, 1)] * m,
                             options={'maxiter': 100, 'ftol': 1e-16})
            lam_opt = np.maximum(res.x, 0)
            lam_sum = lam_opt.sum()
            if lam_sum < 1e-15:
                lam_opt = np.ones(m) / m
            else:
                lam_opt /= lam_sum
        
        # Build Hankel vector: lam_hankel[k] = lam_opt[i] if k == active[i], else 0
        lam_hankel = np.zeros(2 * n - 1)
        for i, k in enumerate(active):
            lam_hankel[k] = lam_opt[i]
        
        # Solve L @ x = 1 where L[j,l] = lam_hankel[j+l]
        rhs = np.ones(n)
        x = hankel_solve(lam_hankel, rhs, n)
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            break
        a_new = x / sx
        
        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        # Check convergence
        if np.linalg.norm(a_new - a) < tol:
            break
        
        a = a_new
        
        if (it + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveSetFP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} |Δa|={np.linalg.norm(a_new-a):.6e}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.02)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
        a = a + 0.1 * np.sin(4 * np.pi * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.sin(2 * np.pi * x) ** 2 + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function with active-set refinement and multi-phase optimization."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective deadline: {global_deadline - t_start:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP (unconstrained) - 25% time ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score:.8f}) ===")
    t1 = 0.25 * (budget_s - 30)
    a_out, _ = nesterov_lp(best_seq, t1, solver=solver)
    try_update(a_out, "nesterov_lp")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Nesterov LP with symmetry - 10% time ===
    print(f"\n=== Phase 2: Symmetric Nesterov LP (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.10 * (budget_s - 30), remaining * 0.4)
    if t2 > 20:
        a_sym = symmetric_constraint(best_seq)
        a_out_sym, _ = nesterov_lp(a_sym, t2, solver=solver, use_symmetry=True)
        try_update(a_out_sym, "nest_sym")
        # Continue unsymmetrically from symmetric result
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_out_sym2, _ = nesterov_lp(a_out_sym, remaining * 0.3, solver=solver)
            try_update(a_out_sym2, "nest_sym_to_free")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Top-k LSE optimization - 25% time ===
    print(f"\n=== Phase 3: Top-k LSE (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.25 * (budget_s - 30), remaining * 0.5)
    if t3 > 30:
        # Pass 1: high temperature, few peaks
        a_lse1, _ = lsebfgs_topk(best_seq, t3 * 0.4, T_start=80.0, T_min=1.0,
                                  n_stages=20, iters_per_stage=40, top_k_start=5, top_k_end=20)
        try_update(a_lse1, "lse_high_T")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            # Pass 2: medium temperature
            a_lse2, _ = lsebfgs_topk(a_lse1 if a_lse1 is not None else best_seq,
                                     remaining * 0.4, T_start=20.0, T_min=0.5,
                                     n_stages=20, iters_per_stage=40, top_k_start=10, top_k_end=30)
            try_update(a_lse2, "lse_med_T")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # Pass 3: very low temperature
            a_lse3, _ = lsebfgs_topk(best_seq, remaining * 0.5, T_start=5.0, T_min=0.1,
                                     n_stages=15, iters_per_stage=35, top_k_start=5, top_k_end=15)
            try_update(a_lse3, "lse_low_T")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Active Set Fixed Point - 20% time ===
    print(f"\n=== Phase 4: Active Set Fixed Point (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.20 * (budget_s - 30), remaining * 0.5)
    if t4 > 15:
        a_fp, _ = active_set_fixed_point(best_seq, t4, tol=1e-9, max_iters=50, epsilon=1e-4)
        try_update(a_fp, "active_fp")
        # Refine with tighter epsilon
        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            a_fp2, _ = active_set_fixed_point(best_seq, remaining * 0.6, tol=1e-10, max_iters=30, epsilon=1e-5)
            try_update(a_fp2, "active_fp_tight")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Peak Equalization - 10% time ===
    print(f"\n=== Phase 5: Peak Equalization (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t5 > 10:
        a_pe, _ = peak_equalization(best_seq, t5, n_peaks=10)
        try_update(a_pe, "peak_eq")
        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            a_pe2, _ = peak_equalization(best_seq, remaining * 0.5, n_peaks=20)
            try_update(a_pe2, "peak_eq20")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Multi-length exploration - 5% time ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t6_total = min(0.05 * (budget_s - 30), remaining * 0.6)
    if t6_total > 30:
        lengths_to_try = [n_base + 400, n_base - 200, n_base + 800]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 20:
                break
            t6 = min(t6_total / 3.0, remaining * 0.45)
            if t6 < 15:
                break
            print(f"  Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t6 * 0.5, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}_nest")
            remaining = global_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = lsebfgs_topk(a_try, remaining * 0.5, T_start=20.0, T_min=0.5,
                                         n_stages=10, iters_per_stage=25, top_k_start=5, top_k_end=20)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        # Short Nesterov
        a_f1, _ = nesterov_lp(best_seq, remaining * 0.3, solver=solver, max_iters=30000)
        try_update(a_f1, "final_nest")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Active set with very tight tolerance
            a_f2, _ = active_set_fixed_point(best_seq, remaining * 0.4, tol=1e-11, max_iters=20, epsilon=1e-6)
            try_update(a_f2, "final_fp")

        remaining = global_deadline - time.monotonic()
        if remaining > 8:
            # Very low T LSE
            a_f3, _ = lsebfgs_topk(best_seq, remaining * 0.5, T_start=2.0, T_min=0.05,
                                   n_stages=10, iters_per_stage=20, top_k_start=3, top_k_end=10)
            try_update(a_f3, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {elapsed:.0f}s / {budget_s:.0f}s")
    return [float(x) for x in best_seq]
