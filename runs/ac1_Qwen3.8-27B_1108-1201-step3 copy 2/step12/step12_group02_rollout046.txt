Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on softmax parameterization with LogSumExp (LSE) objective
2. Multi-stage temperature annealing (high T for smooth start, low T for sharp max)
3. FFT-based O(n log n) gradient computation
4. Diverse structured initializations (U-shaped, W-shaped, sinc, parent)
5. Multi-length exploration with interpolation
6. Coordinate descent + spectral shaping polish

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (max(1, size - 1)).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))

    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

    len_w = len(b)

    exp_Tb = np.exp(Tb_shift)
    w = exp_Tb / exp_Tb.sum()

    h1 = h1_full[len_w - n : len_w][::-1]
    h2 = h2_full[0 : n]

            g_z = a * (g_a - dot_ga_a)
            
            t_count += 1
            m = beta1 * m + (1 - beta1) * g_z
            v = beta2 * v + (1 - beta2) * g_z * g_z
            m_hat = m / (1 - beta1 ** t_count)
            v_hat = v / (1 - beta2 ** t_count)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 20 == 0 or it == iters_per_stage - 1:
                z_shift = z - z.max()
                a_check = np.exp(z_shift) / np.exp(z_shift).sum()
                sc = fast_score(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
        
        if verbose and si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam LSE stage {si+1}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score
