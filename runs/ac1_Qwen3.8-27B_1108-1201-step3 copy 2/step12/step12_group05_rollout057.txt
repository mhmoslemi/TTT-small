(grad = 2*shifted_a) for O(n) or O(n log n) per step without LP solvers.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def score_fn(a):
    """Compute the objective 2*n*max_b/(sum_a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-300:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_shaping(a, n_iter=5000, alpha_start=0.5, alpha_end=0.05, target_power=0.0):
    """
    Adaptive spectral shaping: flatten the power spectrum to reduce autocorrelation peaks.
    alpha decays from alpha_start to alpha_end over iterations.
    target_power: if > 0, target |A|^2 = target_power for non-DC bins.
    """
    n = len(a)
    a = normalize_nonneg(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = score_fn(a)

    for i in range(n_iter):
        # Adaptive alpha
        frac = i / max(1, n_iter - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        if target_power > 0:
            target = target_power
        else:
            # Geometric mean of non-DC
            non_dc = B[1:]
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize_nonneg(a_new)

        sc = score_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def mirror_descent(a0, time_limit, eta=None, n_max_iter=200000):
    """
    Mirror descent (multiplicative weight updates) for min max_k b_k over the simplex.
    
    The gradient of b_{k*} w.r.t. a_i is 2*a_{k*-i} (for valid indices).
    Update: a_i <- a_i * exp(-eta * 2*a_{k*-i}) / Z
    
    This is O(n log n) per step due to FFT for autocorrelation.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    FFT_SIZE = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    if eta is None:
        eta = 1.0 / np.sqrt(2.0 * n)  # initial guess
    
    # Adaptive eta: geometric schedule
    eta_decay = 0.99995  # very slow decay
    
    for it in range(n_max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        A = rfft(a, FFT_SIZE)
        b = irfft(A * A, FFT_SIZE)[:2*n-1]
        
        # Find peak
        k_star = np.argmax(b)
        max_b = b[k_star]
        
        # Compute gradient: g_i = 2 * a_{k_star - i} for valid i
        # Valid i: 0 <= i < n and 0 <= k_star - i < n
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            grad = np.zeros(n)
            # For i in [i_start, i_end]: grad[i] = 2 * a[k_star - i]
            idx = np.arange(i_start, i_end + 1)
            grad[idx] = 2.0 * a[k_star - idx]
        else:
            grad = np.zeros(n)
        
        # Mirror descent update
        # a_i <- a_i * exp(-eta * grad_i) / Z
        # For numerical stability, work in log space
        log_a = np.where(a > 0, np.log(a + 1e-300), -30.0)
        log_a_new = log_a - eta * grad
        # Normalize (soft)
        log_a_new -= log_a_new.max()
        exp_a = np.exp(log_a_new)
        Z = exp_a.sum()
        if Z > 0 and np.isfinite(Z):
            a_new = exp_a / Z
        else:
            a_new = a.copy()
        
        sc = 2.0 * n * max_b  # Since sum(a) = 1, score = 2*n*max_b
        # Actually let me recompute properly
        # score = 2*n*max_b / sum(a)^2 = 2*n*max_b (since sum=1)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        eta *= eta_decay
        
        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD iter={it} ({elapsed:.0f}s): score={best_score:.6f}, max_b={max_b:.8f}, eta={eta:.6f}")
    
    return best_a, best_score


def frank_wolfe(a0, time_limit, n_max_iter=500000):
    """
    Frank-Wolfe (conditional gradient) for min max_k b_k over the simplex.
    
    At each step:
    1. Compute grad = 2 * a shifted (at argmax of b)
    2. Find s = e_j where j = argmin grad (vertex of simplex)
    3. Update: a <- (1-t) * a + t * s
    
    Each step is O(n log n) for the FFT + O(n) for the update.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    FFT_SIZE = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    for t in range(1, n_max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        A = rfft(a, FFT_SIZE)
        b = irfft(A * A, FFT_SIZE)[:2*n-1]
        k_star = np.argmax(b)
        max_b = b[k_star]
        
        # Compute gradient
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad = np.zeros(n)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad[idx] = 2.0 * a[k_star - idx]
        
        # FW step: find vertex with minimum grad
        j = np.argmin(grad)
        
        # Step size: t/(t+2) for FW (or 2/(t+2) for average FW)
        gamma = 2.0 / (t + 2.0)
        
        a_new = a.copy()
        a_new[j] += gamma * (1.0 - a_new[j])  # Move toward e_j by gamma
        # Actually: a_new = (1-gamma)*a + gamma*e_j
        # a_new[i] = (1-gamma)*a[i] + gamma*(1 if i==j else 0)
        # a_new[j] = (1-gamma)*a[j] + gamma = a[j] + gamma*(1-a[j])
        # a_new[i≠j] = (1-gamma)*a[i]
        
        a_new = (1.0 - gamma) * a
        a_new[j] += gamma
        
        sc = 2.0 * n * b.max()  # quick score (sum=1)
        # But we should recompute since a changed slightly
        if sc < best_score:
            # Verify with actual computation
            actual_score = score_fn(a_new)
            if actual_score < best_score:
                best_score = actual_score
                best_a = a_new.copy()
        
        a = a_new
        
        if t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FW iter={t} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def multi_peak_mirror_descent(a0, time_limit, n_peaks=5, eta=None):
    """
    Mirror descent targeting multiple peaks simultaneously.
    Uses a weighted combination of gradients at top-k peaks.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    FFT_SIZE = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    if eta is None:
        eta = 0.5 / np.sqrt(n)
    
    eta_decay = 0.9999
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, FFT_SIZE)
        b = irfft(A * A, FFT_SIZE)[:2*n-1]
        
        # Get top peaks
        sorted_idx = np.argsort(b)[::-1]
        top_k = sorted_idx[:n_peaks]
        
        # Build combined gradient with weights proportional to (b_k - mean_b)^+
        mean_b = b.mean()
        weights = np.maximum(b[top_k] - mean_b, 0)
        if weights.sum() > 0:
            weights = weights / weights.sum()
        else:
            weights = np.ones(len(top_k)) / len(top_k)
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                grad[idx] += w * 2.0 * a[k - idx]
        
        # Mirror descent update
        log_a = np.where(a > 0, np.log(a + 1e-300), -30.0)
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        exp_a = np.exp(log_a_new)
        Z = exp_a.sum()
        if Z > 0 and np.isfinite(Z):
            a_new = exp_a / Z
        else:
            continue
        
        sc = 2.0 * n * b.max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        eta *= eta_decay
        
        if it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MPMD iter={it} ({elapsed:.0f}s): score={best_score:.6f}, eta={eta:.6f}")
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    FFT_SIZE = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        A = rfft(a, FFT_SIZE)
        b = irfft(A * A, FFT_SIZE)[:2*n-1]
        top_k = np.argsort(b)[::-1][:8]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=(lambda j: a[j]))
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = normalize_nonneg(a_trial)
                score = score_fn(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = normalize_nonneg(a_trial)
                    score = score_fn(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lp_constraint_generation(a0, time_limit, solver=None):
    """
    LP with constraint generation for the zero-sum game.
    min M s.t. (T(a) @ b)_k <= M for all k, sum(b) = 1, b >= 0
    
    Uses constraint generation: start with few constraints, add violated ones.
    """
    import cvxpy as cp
    
    n = len(a0)
    a = normalize_nonneg(a0)
    
    # We'll alternate between:
    # 1. Finding best response b given a (solve LP with constraints for current a)
    # 2. Updating a given b (the a that minimizes max_k (T(b) @ a)_k, which is the primal of a different LP)
    
    # Actually, let's do the standard LP best response:
    # Given a, find b that minimizes max_k (conv(a,b)_k) subject to sum(b)=1, b>=0
    
    # Build the convolution matrix T(a): (T(a) @ b)_k = sum_j a_j * b_{k-j}
    # This is a (2n-1) x n matrix
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    # Try to find a solver
    if solver is None:
        for s in [cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solver = s
                    break
            except:
                continue
        if solver is None:
            solver = cp.ECOS
    
    y = a.copy()  # running average
    a_cur = a.copy()
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # LP: min M s.t. T(y) @ b <= M, sum(b)=1, b >= 0
        # T(y) is (2n-1) x n
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        # Build constraints using the structure of T(y)
        # (T(y) @ b)_k = sum_j y_j * b_{k-j} for valid indices
        # Instead of building the full matrix, we can use cvxpy expressions
        constraints = [cp.sum(b_var) == 1]
        for k in range(2 * n - 1):
            # (T(y) @ b)_k = sum_{j} y_j * b_{k-j} where 0<=j<n and 0<=k-j<n
            # Valid j: max(0, k-n+1) <= j <= min(n-1, k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                expr = sum(y[j] * b_var[k - j] for j in range(j_start, j_end + 1))
                constraints.append(expr <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b
        M_val = float(M_var.value)
        
        # Update y (average)
        alpha_t = 2.0 / (t + 2.0)
        y = (1 - alpha_t) * y + alpha_t * b_val
        y = normalize_nonneg(y)
        
        # Nesterov extrapolation
        if t >= 2:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y + momentum * (y - a_cur)
            a_new = normalize_nonneg(a_new)
        else:
            a_new = b_val.copy()
        
        a_cur = a_new
        
        # Check scores
        sc_y = score_fn(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = score_fn(a_new)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a_new.copy()
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-CG iter={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization combining:
    1. Spectral shaping (frequency domain)
    2. Mirror descent (multiplicative weights)
    3. Frank-Wolfe
    4. Multi-peak mirror descent
    5. LP constraint generation
    6. Peak shaving
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except:
        pass
    
    # Determine working lengths
    if best_seq is not None:
        n_parent = len(best_seq)
        n_base = max(n_parent, 1600)
    else:
        n_base = 2000
        best_seq = structured_init(n_base, rng)
        best_score = score_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    # Ensure we work at n_base length
    if len(best_seq) != n_base:
        best_seq = interpolate_sequence(best_seq, n_base)
        best_score = score_fn(best_seq)
        print(f"Interpolated to n={n_base}, score={best_score:.6f}")
    
    def update_best(a, tag):
        nonlocal best_seq, best_score
        a = normalize_nonneg(a)
        sc = score_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{tag}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # Phase 1: Spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.12, 120.0)
    if t1 > 10:
        a_out, _ = spectral_shaping(best_seq, n_iter=15000, alpha_start=0.4, alpha_end=0.03)
        update_best(a_out, "spectral_1")
    
    # Phase 2: Mirror descent (single peak)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Mirror descent at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 200.0)
    if t2 > 10:
        a_out, _ = mirror_descent(best_seq, t2)
        update_best(a_out, "mirror_1")
    
    # Phase 3: Multi-peak mirror descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Multi-peak MD at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 10:
        a_out, _ = multi_peak_mirror_descent(best_seq, t3, n_peaks=5)
        update_best(a_out, "multi_md_1")
    
    # Phase 4: Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global_start)
