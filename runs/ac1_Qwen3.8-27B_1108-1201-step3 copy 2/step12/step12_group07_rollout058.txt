Lagrangian: $M + \lambda^T(Tb - M\mathbf{1}) - \nu^T b + \mu(1 - e^Tb)$
$= M(1 - \sum \lambda_k) + b^T(T^T\lambda - \nu - \mu e) + \mu$

For boundedness below: $\sum \lambda_k = 1$ and $T^T\lambda - \nu - \mu e = 0$ (i.e., $\nu = T^T\lambda - \mu e \geq 0$).

Dual: $\max_{\lambda \in \Delta, \mu} \mu$ s.t. $T^T\lambda \geq \mu e$.

This means $\mu \leq \min_j (T^T\lambda)_j = \min_j \sum_k \lambda_k T_{kj}$.

So the dual is: $\max_{\lambda \in \Delta} \min_j \sum_k \lambda_k T_{kj} = \max_{\lambda \in \Delta} \min_j (T^T\lambda)_j$

This is a "max-min" problem! And it's actually a linear program in $\lambda$:
$\max_{\lambda, \mu} \mu$ s.t. $\lambda \in \Delta$, $T^T\lambda \geq \mu e$

This dual LP has $2n-1$ variables ($\lambda_0, ..., \lambda_{2n-2}, \mu$) and $n$ constraints ($T^T\lambda \geq \mu e$) plus the simplex constraint.

The dual LP is smaller in terms of constraints (n vs 2n-1) but larger in terms of variables (2n vs n). For our sizes, they're comparable.

The key insight is: **the dual LP gives us a lower bound on the primal optimal value**. And the primal LP gives an upper bound. The gap between them tells us how close we are to optimality.

**Strategy: Use the dual LP to guide the primal LP.**

1. Solve the dual LP to get $\lambda^*$ (a distribution over positions).
2. Use $\lambda^*$ to identify the "important" positions.
3. Solve the primal LP with only the important positions (reduced LP).
4. Iterate.

def solve_lp_dual(a_fixed, solver=None, eps=1e-10):
    """Solve the dual LP: max mu s.t. T^T @ lambda >= mu*1, lambda in Delta."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    lam_var = cp.Variable(size, nonneg=True)
    mu_var = cp.Variable()
    # T^T @ lambda >= mu * ones(n)
    # (T^T @ lambda)_j = sum_k lambda_k * T_{k,j} = sum_k lambda_k * a_{k-j}
    # We can compute this as: for each j, sum over k of lambda_k * a_{k-j} (if valid)
    # This is the convolution of lambda with reversed a, but we need individual constraints.
    # Let's build T^T explicitly (n x size matrix)
    TT = np.zeros((n, size))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        TT[js:je + 1, k] = a_fixed[k - je:k - js + 1]
    constraints = [TT @ lam_var >= mu_var, cp.sum(lam_var) == 1]
    prob = cp.Problem(cp.Maximize(mu_var), constraints)
    try:
        prob.solve(solver=solver, eps_abs=eps, eps_rel=eps)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, eps_abs=eps, eps_rel=eps)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    lam_val = np.asarray(lam_var.value, dtype=np.float64)
    lam_val = np.maximum(lam_val, 0)
    s = lam_val.sum()
    if s < 1e-15:
        return None, None
    lam_val = lam_val / s
    return lam_val, float(mu_var.value)

def spectral_init(n, rng):
    """Initialize using shaped spectrum for potentially flatter auto-convolution."""
    # Create a spectrum that's close to flat but with non-negative inverse
    num_freqs = n // 2 + 1
    # Raised cosine spectrum (ensures smoothness and non-negativity of time domain)
    k = np.arange(num_freqs)
    # Main lobe: raised cosine
    w = 0.5 * (1 + np.cos(np.pi * k / (n * 0.3)))
    w = np.maximum(w, 0.01)
    # Add some random perturbation
    w += 0.05 * np.abs(rng.standard_normal(num_freqs))
    # Make it Hermitian
    spectrum = np.zeros(n, dtype=np.complex128)
    spectrum[:num_freqs] = w
    spectrum[n - (num_freqs - 1):] = w[1:][::-1]
    # Inverse FFT
    a = np.real(irfft(spectrum, n))
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_adaptive_v2(a0, time_limit, solver=None, max_iters=1000000, K_start=30, K_end=100):
    """
    Nesterov LP with progressive active set growth and dual convergence check.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    epoch_start_score = best_score
    epoch_start_a = a.copy()
    epoch_iter = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Progressive K: start small, grow over time
        progress = min(1.0, it / (max_iters * 0.5))
        K = int(K_start + (K_end - K_start) * progress)

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        # Get active rows and solve LP
        active_rows = get_active_rows(y, K)
        b_new, M_lp = solve_lp_best_response_reduced(y, active_rows, solver)

        if b_new is None:
            # Fallback to full LP
            b_new, M_lp = solve_lp_best_response_full(y, solver)
            if b_new is None:
                continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)
        M_new = fast_convolve(a_new).max()

        if sc_new < best_score:
            best_score = sc_new
            best_M = M_new
            best_a = a_new.copy()

        # Adaptive restart: if we're not making progress in this epoch
        sc_cur = score_normalized(a)
        epoch_iter += 1
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        elif sc_new > epoch_start_score * 1.005 and epoch_iter > 200:
            # Significant regression from epoch start - full restart from best
            a = best_a.copy()
            a_prev = a.copy()
            t_nest = 1.0
            epoch_start_score = score_normalized(a)
            epoch_start_a = a.copy()
            epoch_iter = 0
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        # Dual convergence check every 500 iterations
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            remaining = time_limit - elapsed
            if remaining > 30:
                # Solve dual LP for convergence check
                lam_val, mu_val = solve_lp_dual(best_a, solver)
                if lam_val is not None:
                    primal_bound = best_M  # max of convolution for best_a
                    dual_bound = mu_val
                    gap = primal_bound - dual_bound
                    if elapsed - last_print > 10:
                        print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} "
                              f"t={t_nest:.1f} K={K} restarts={restart_count} "
                              f"gap={gap:.2e}")
                        last_print = elapsed
                    if gap < 1e-6 * primal_bound:
                        print(f"    Converged! gap={gap:.2e}")
                        break

        if (it + 1) % 500 == 0 and (it + 1) % 500 != 0 or (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} K={K}")
                last_print = elapsed

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, max_steps=100000):
    """Fast subgradient Nesterov using exact convolution gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.3
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 8000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.95)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.002:
                t_nest = 1.0
                eta *= 0.9
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.998, 0.01)

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=25):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 50 if si < n_stages // 3 else (30 if si < 2 * n_stages // 3 else 20)
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: score={best_score:.8f}")
        except Exception:
            pass

    return best_a, best_score


def peak_targeted_refine_v2(a0, time_limit, K=30, max_steps=80000):
    """Improved multiplicative weights update targeting top K peaks with adaptive eta."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve_step = 0
    consecutive_fail = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 8000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Use a slightly wider threshold to include more peaks
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        topk = topk[:K]  # Limit to K peaks

        grad = np.zeros(n)
        weights = b[topk]
        weights = weights / weights.sum()  # Weight by peak height
        for i, k in enumerate(topk):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += weights[i] * a[k - idx]

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            consecutive_fail = 0
            # Slightly increase eta if improving
            eta = min(eta * 1.005, 0.12)
        else:
            eta *= 0.88
            consecutive_fail += 1
            if consecutive_fail > 200:
                break

    return best_a, best_score


def multiplicative_weights_fast(a0, time_limit, max_steps=500000):
    """Very fast multiplicative weights for initial progress."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.15
    last_improve = 0
    batch_size = 10

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 15000:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Target only the single maximum
        k_max = int(np.argmax(b))

        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        grad[idx] = a[k_max - idx]

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.001, 0.3)
        else:
            eta *= 0.95
            if eta < 0.005:
                eta = 0.005

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def add_noise_and_refine(a0, noise_level, time_limit, solver):
    """Add noise, then do a short Nesterov LP to potentially find a better basin."""
    n = len(a0)
    rng = np.random.default_rng()
    a_noisy = a0 + noise_level * rng.standard_normal(n)
    a_noisy = np.maximum(a_noisy, 0)
    s = a_noisy.sum()
    if s < 1e-15:
        return a0, score_normalized(a0)
    a_noisy = a_noisy / s
    a_out, sc = nesterov_lp_adaptive_v2(a_noisy, time_limit, solver, max_iters=20000, K_start=20, K_end=50)
    return a_out, sc
