        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x) + 0.3 * np.cos(4 * np.pi * x)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.3))
    elif choice == 2:
        a = np.exp(-3.0 * (x - 0.5)**2) + 0.3
    elif choice == 3:
        a = 0.7 + 0.3 * np.sin(4 * np.pi * x + 1.0)
    elif choice == 4:
        a = (1.0 + np.cos(2 * np.pi * x)) * 0.6 + 0.2
    elif choice == 5:
        a = np.exp(-6.0 * (x - 0.2)**2) + 0.5 * np.exp(-6.0 * (x - 0.8)**2) + 0.1
    elif choice == 6:
        a = 0.8 + 0.2 * np.cos(6 * np.pi * x)
    elif choice == 7:
        a = 1.0 + 0.4 * np.sin(3 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.maximum(0.5 + 0.5 * np.tanh(3.0 * (x - 0.5)), 0.05)
    else:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + 0.5)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    patience = 0
    max_patience = 300
    last_improve_iter = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            patience += 1
            if patience > max_patience:
                break
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_a = a_new.copy()
            patience = 0
            last_improve_iter = it
        else:
            patience += 1

        if patience > max_patience:
            t_nest = 1.0
            patience = 0
            max_patience = max(100, max_patience // 2)
        elif sc_new > score_normalized(a) * 1.001 and t_nest > 10.0:
            t_nest = 1.0
            patience = 0
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f} pat={patience}")

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 60 if si < n_stages // 3 else 30
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            z = z_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

        if (si + 1) % 5 == 0:
            print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_rounds=200):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit * 0.7 and best_score < score_normalized(a) - 1e-8:
            break

        perm = rng.permutation(n)
        improved = False
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            a_old = a[i]
            if a_old < 1e-15:
                a[i] = 1e-8
                s_new = a.sum()
                a = a / s_new
            factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
            best_local_score = best_score
            best_local_factor = 1.0
            for f in factors:
                a[i] = a_old * f
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = a_old
                    continue
                a_temp = a / s_new
                sc = score_normalized(a_temp)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local_factor = f
            a[i] = a_old * best_local_factor
            s_new = a.sum()
            if s_new < 1e-15:
                a[i] = a_old
                continue
            a = a / s_new
            if best_local_factor != 1.0:
                improved = True
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()

        if not improved:
            break

    return best_a, best_score


def peak_targeted_refine(a0, time_limit, max_steps=30000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve_step = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.005, 0.1)
        else:
            eta *= 0.8
            if eta < 1e-12:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_subgrad(a0, time_limit, max_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.2
    last_improve = 0
    patience = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.95)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a.copy()
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            patience = 0
            last_improve = step
        else:
            patience += 1
            if sc > best_score * 1.002:
                t_nest = 1.0
                eta *= 0.95
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.998, 0.02)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):

    print(f"Using LP solver: {solver}")

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:

        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            elapsed = time.monotonic()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)}) t={elapsed - _global_deadline + budget_s - 30:.0f}s")

    print(f"\n=== Phase 1: Subgrad Nesterov n={n_base} score={best_score:.8f} ===")

    if t1 > 5:
        a_out, _ = nesterov_subgrad(best_seq, t1)

        a_lse = lse_lbfgs(best_seq, t2, T_start=40.0, T_end=2.0, n_stages=10)
        try_update(a_lse, "lse_quick")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Main LP-Nesterov (750s) ===
    print(f"\n=== Phase 3: LP-Nesterov main n={best_n} score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(750.0, remaining * 0.45)
    if t3 > 30:
        a_out, _ = nesterov_lp_adaptive(best_seq, t3, solver=solver)
        try_update(a_out, "lp_nesterov_main")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Coordinate descent (120s) ===
    print(f"\n=== Phase 4: Coordinate descent ===")
    remaining = _global_deadline - time.monotonic()

            a_try, _ = nesterov_lp_adaptive(a_warm, t9 * 0.8, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")

    print(f"  Phase 9 done: best={best_score:.8f}")

    # === Phase 10: Final polish (100s) ===

    # Final verification
    if best_seq is not None:

        if np.isfinite(fs):
            best_score = fs
            print(f"Verified score: {best_score:.8f}")

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
