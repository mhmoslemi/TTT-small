def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def cross_corr_grad(a, w):
    """
    Compute grad[j] = 2 * sum_m a[m] * w[m+j] for j=0,...,n-1.
    This is the gradient of sum_k w[k]*b[k] w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    w = np.asarray(w, dtype=np.float64)
    # We need: result[j] = sum_m a[m] * w[m+j]

    w = exp_Tb / exp_Tb.sum()

                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PeakShave ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0) + 0.01
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.05
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.01
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.1
    elif variant == 10:
        # Multi-bump random
        a = np.ones(n) * 0.1
        for _ in range(7):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(0.5, 2.0)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
    else:
        # Raised cosine window (good spectral properties)
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a ** 1.5  # Sharpen slightly

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def spectral_init(n, rng, seed=0):
    """
    Generate initialization by specifying a target flat-ish power spectrum shape
    and finding a non-negative sequence that approximates it.
    Uses iterative projection: alternate between time-domain (non-negative, fixed sum)
    and frequency-domain (target spectrum shape).
    """
    rng2 = np.random.default_rng(seed)
    a = np.ones(n) / n
    fft_size = 1 << (2 * n - 1).bit_length()

    # Target: slightly raised-cosine shaped power spectrum (avoids perfect flatness
    # which is unachievable, but encourages low sidelobes)
    f = np.arange(fft_size // 2 + 1)
    # Target weight: emphasize low frequencies slightly
    target_weight = 1.0 + 0.5 * np.exp(-(f / (fft_size * 0.1)) ** 2)

    for it in range(200):
        # Time domain: project to non-negative with fixed sum
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        # Compute current autocorrelation
        b = irfft(A * A, fft_size)[:2 * n - 1]
        # Adjust power spectrum toward target shape
        P_target_shape = target_weight
        P_norm = P / P.mean()
        P_blend = 0.7 * P_norm + 0.3 * P_target_shape
        # Reconstruct with adjusted amplitude, keep phase
        if np.abs(A).sum() > 1e-30:
            phase = A / np.abs(A + 1e-30)
        else:
            phase = np.ones_like(A)
        A_new = np.sqrt(P_blend) * phase
        a = irfft(A_new, fft_size)[:n]
        # Project to non-negative
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization with full-subgradient MD,
    Adam LSE, peak shaving, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1750.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = fast_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine working lengths
    n_parent = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = diverse_init(n_parent, rng, variant=4)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Full-subgradient Mirror Descent at parent length ===
    print(f"\n=== Phase 1: Full-Subgradient MD at n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_md1 = min(remaining * 0.20, 350.0)
    if t_md1 > 30:
        a_out, _ = mirror_descent_full(best_seq, t_md1, eta0=0.04, decay=0.55,
                                        T_sub=40.0, seed=1, verbose=True)
        try_update(a_out, f"mdfull_n{n_parent}")

    # === Phase 2: Adam LSE at parent length ===
    print(f"\n=== Phase 2: Adam LSE at n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse1 = min(remaining * 0.18, 300.0)
    if t_lse1 > 30 and best_seq is not None:
        a_out, _ = adam_lse(best_seq, t_lse1, T_start=100.0, T_end=1.5,
                            lr_start=0.015, n_restarts=3, seed=7, verbose=True)
        try_update(a_out, f"adam_lse_n{n_parent}")

    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving at n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_ps = min(remaining * 0.08, 100.0)
    if t_ps > 10 and best_seq is not None:
        a_out, _ = peak_shaving_global(best_seq, t_ps, seed=99)
        try_update(a_out, f"ps_n{n_parent}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Length Exploration ===")
    lengths_to_try = []
    for mult in [0.8, 1.2, 1.4, 1.6]:
        n_try = int(round(n_parent * mult))
        n_try = max(200, min(n_try, 5000))
        if n_try != n_parent:
            lengths_to_try.append(n_try)

    for n_try in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_phase = min(remaining * 0.15, 180.0)
        if t_phase < 30:
            break

        print(f"  Trying n={n_try}...")
        init_long = interpolate_to_length(best_seq, n_try)

        # Quick MD
        t_md = t_phase * 0.5
        a_out, _ = mirror_descent_full(init_long, t_md, eta0=0.03, decay=0.5,
                                        T_sub=30.0, seed=n_try % 100, verbose=False)
        try_update(a_out, f"md_n{n_try}")

        # Quick Adam LSE
        remaining_phase = time_budget - (time.monotonic() - t_global)
        t_lse = min(remaining_phase * 0.10, t_phase * 0.4)
        if t_lse > 20 and best_seq is not None:
            init2 = interpolate_to_length(best_seq, n_try)
            a_out, _ = adam_lse(init2, t_lse, T_start=60.0, T_end=2.0,
                                lr_start=0.01, n_restarts=2, seed=n_try % 200, verbose=False)
            try_update(a_out, f"adam_n{n_try}")

    # === Phase 5: Diverse initializations + optimization ===
    print(f"\n=== Phase 5: Diverse Inits ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for v in [0, 4, 9, 10]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_div = min(remaining * 0.06, 80.0)
            if t_div < 15:
                break
            init = diverse_init(n_parent, rng, variant=v)
            a_out, _ = mirror_descent_full(init, t_div, eta0=0.06, decay=0.6,
                                            T_sub=30.0, seed=v * 13 + 5, verbose=False)
            try_update(a_out, f"div_v{v}")

    # === Phase 6: Spectral initialization ===
    print(f"\n=== Phase 6: Spectral Init ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t_spec = min(remaining * 0.05, 50.0)
        if t_spec > 10:
            spec_init = spectral_init(n_parent, rng, seed=999)
            t_md_spec = t_spec * 0.7
            a_out, _ = mirror_descent_full(spec_init, t_md_spec, eta0=0.05, decay=0.6,
                                            T_sub=30.0, seed=777, verbose=False)
            try_update(a_out, "spectral")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 20:
        # Adam LSE with low temperature
        t_pol1 = min(remaining * 0.30, 100.0)
        if t_pol1 > 15:
            a_out, _ = adam_lse(best_seq, t_pol1, T_start=30.0, T_end=0.5,
                                lr_start=0.005, n_restarts=2, seed=333, verbose=True)
            try_update(a_out, "polish_adam")

        # Peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        t_pol2 = min(remaining * 0.30, 80.0)
        if t_pol2 > 10:
            a_out, _ = peak_shaving_global(best_seq, t_pol2, seed=555)
            try_update(a_out, "polish_ps")

        # Very slow MD with high T (smooth)
        remaining = time_budget - (time.monotonic() - t_global)
        t_pol3 = min(remaining * 0.30, 60.0)
        if t_pol3 > 10:
            a_out, _ = mirror_descent_full(best_seq, t_pol3, eta0=0.01, decay=0.7,
                                            T_sub=100.0, seed=444, verbose=False)
            try_update(a_out, "polish_md")

    # Final output
    if best_seq is not None:
        # Ensure validity
        final = np.maximum(best_seq, 0.0)
        final = np.minimum(final, 1000.0)
        s = final.sum()
        if s < 0.01:
            final = np.ones(len(final)) / len(final)
        else:
            final = final / s
        print(f"\nFINAL: score={fast_score(final):.6f}, n={len(final)}")
        return [float(x) for x in final]
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]
