    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def power_norm_adam(a0, time_limit, p_start=4, p_end=50, n_p_stages=8, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    # Power annealing schedule
    p_values = np.geomspace(p_start, p_end, n_p_stages)
    
    for pi, p in enumerate(p_values):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Step size schedule for this stage
        lr_start = 0.1 / (n ** 0.5)
        n_steps = max(50, min(500, int(remaining * 3)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad = power_norm_gradient(a, p)
            
            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr = lr_start / (1.0 + 0.01 * it)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()

                a_new = a.copy()

                a_new = a_new * norm / s_new
            
            a = a_new

def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Aggressive coordinate descent: try replacing each coordinate with 0 or redistributing mass."""
    n = len(a0)

        a = a * norm / s0

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Precompute which positions participate in which convolution peaks
    b = fast_autocorr(a)
    top_peaks = np.argsort(b)[::-1][:20]

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False
        
        # Try zeroing out small coordinates that contribute to peaks
        for k in top_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            # Find the largest contributor
            max_i = idx[np.argmax(contrib)]
            
            # Try reducing max_i and spreading to non-peak positions
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) > 0:
                # Pick candidate with lowest value
                dst = candidates[np.argmin(a[candidates])]
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    amt = a[max_i] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[max_i] -= amt
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
        
        # Random perturbation
        if not improved and iteration % 10 == 0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = a[i] * rng.uniform(0.05, 0.3)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:

                    if s_t > 1e-15:

            elapsed = time.monotonic() - t0
            print(f"    CD iter={iteration} ({elapsed:.0f}s): best={best_score:.8f}")

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng, style=0):
    if style == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif style == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif style == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif style == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        # Uniform with random perturbations
        a = np.ones(n) * (1.0 + 0.3 * rng.standard_normal(n))
        a = np.maximum(a, 0.0)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.ones(n) * (1.0 + 0.2 * np.sin(4.0 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
