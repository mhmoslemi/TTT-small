                # If crossover rate is high, add randomness elsewhere; otherwise, mutate
                if np.random.random() <= 0.1:  # Mutation threshold
                    new_child = mutation(new_child, len(new_child), np.random)
                new_pop[i] = new_child
        
        # Update population
        pop = new_pop
        
        # Increment generation count
        gen_count += 1
    
    # Evaluate last generation
    for seq in pop:
        s = evaluate_safe(seq)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = seq
    
    # Return sequence as list (explicit non-negative and bounded)
    validated_best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    total = sum(validated_best_seq)
    # Normalization if needed (to ensure sum >= 0.01, with the same scaling)
    if total < 0.01:
        validated_best_seq = [0.01 / total * x for x in best_seq]  # This is a fallback within the allowed bound
        best_score = evaluate_safe(np.array(validated_best_seq))
    
    # Format final output
    return_list = [x for x in validated_best_seq]
    final_score = best_score
    return return_list, final_score

def main():
    height_sequence_1 = None  # This will be provided by the environment
    time_budget = 1800.0  # Available time for the search
    try:
        candidate, score = propose_candidate(height_sequence_1, time_budget)
    except:
        candidate = structured_init(1600, np.random)
        score = evaluate_safe(candidate)
    
    print(f"Search completed. Objective value (score): {score:.6f}")
    print(f"Generated sequence length: {len(candidate)}")
    # Return sequence as list of non-negative floats
    final_candidate = candidate
    # If time runs out, fallback is at least this evaluated candidate
    return final_candidate

if __name__ == "__main__":
    main()
