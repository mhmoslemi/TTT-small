def fast_conv(a):
    """Linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]

def score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
