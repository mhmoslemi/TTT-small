    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) > (cssv - 1))
    theta = (cssv[rho] - 1) / (rho + 1)
    a = np.maximum(v - theta, 0)
    return a / a.sum()

def fast_autocorr(a):
    """Full self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Compute score efficiently (assumes a is normalized to sum=1)."""

def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""

def logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=15):
    """LogSumExp-smoothed optimization with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s_init = a_init.sum()
    if s_init < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s_init
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = eval_score(a_init)
    
    def lse_obj(z_in, T):
        a = softmax_fn(z_in)
        b = fast_autocorr(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        return (max_Tb + np.log(exp_Tb.sum())) / T
    
    def lse_grad(z_in, T):
        a = softmax_fn(z_in)
        b = fast_autocorr(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        a_rev = a[::-1].copy()
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        if conv_len > fft_size:
            fft_size *= 2
        W = rfft(w, fft_size)
        A_rev = rfft(a_rev, fft_size)
        conv = irfft(W * A_rev, fft_size)
        grad_a = 2.0 * conv[n - 1:n - 1 + n]
        
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return grad_z
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        iters_per_stage = max(20, min(80, int(10 + 50 * (si / max(1, n_stages - 1)))))
        
        def func(z_in):
            return lse_obj(z_in, stage_T)
        
        def jac(z_in):
            return lse_grad(z_in, stage_T)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = eval_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass
        
        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score_fast(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""

def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.argmax(cond)
    if rho == 0 and not cond[0]:
        return np.ones(n) / n
    theta = (cssv[rho] - 1) / (rho + 1)
    a = np.maximum(v - theta, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

        score_new = eval_score_fast(a_new)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            best_step = step
            no_improve = 0
        elif score_new < current_score:
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive step: shrink when no improvement, slowly grow otherwise
        if no_improve > 30:
            step *= 0.85
            no_improve = 0
            if step < 1e-6:
                break
        elif it % 200 == 0 and it > 0:
            step = min(step * 1.02, best_step * 2.0)

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): score={best_score:.8f}, step={step:.6f}")

    return best_a, best_score


def lse_objective(z_in, T, n):
    """LogSumExp objective value."""
    a = softmax_fn(z_in)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (max_Tb + np.log(exp_Tb.sum())) / T


def lse_gradient(z_in, T, n, fft_size):
    """LogSumExp gradient in z-space."""
    a = softmax_fn(z_in)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    a_rev = a[::-1].copy()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=15):
    """LogSumExp multi-stage annealing with L-BFGS."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s_init = a_init.sum()
    if s_init < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s_init
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = eval_score_fast(a_init)

    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters_per_stage = max(15, min(100, int(10 + 60 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            return lse_objective(z_in, stage_T, n)

        def jac(z_in):
            return lse_gradient(z_in, stage_T, n, fft_size)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-22,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = eval_score_fast(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def local_search(a0, time_limit, seed=42):
    """Local search focusing on peak contributors with adaptive moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = eval_score_fast(a)
    t0 = time.monotonic()
    stall = 0
    K = 6

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k_idx = np.argsort(b)[::-1][:K]

        improved = False
        for k in top_k_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            top_contrib = np.argsort(contrib)[::-1][:4]

            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                for attempt in range(4):
                    dst = rng.integers(n)
                    if dst == src:
                        continue
                    frac = rng.uniform(0.002, 0.02)
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = eval_score_fast(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration with varying step sizes
            for _ in range(15):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score = eval_score_fast(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 800:
                break

        if stall > 0 and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LS stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def coordinate_descent(a0, time_limit, n_rounds=50000):
    """Pair-coordinate descent: try moving mass from i to j with optimal fraction."""
    n = len(a0)
    rng = np.random.default_rng(777)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = eval_score_fast(a)
    t0 = time.monotonic()

    # Precompute which pairs to try
    # Focus on pairs where i is a peak contributor and j is a "quiet" position
    b = fast_autocorr(a)
    top_lags = np.argsort(b)[::-1][:5]

    for rd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        if rd % 500 == 0:
            b = fast_autocorr(a)
            top_lags = np.argsort(b)[::-1][:5]

        # Pick a peak and its top contributor
        k = top_lags[rd % len(top_lags)]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            continue
        contrib = a[idx] * a[k - idx]
        tc = np.argmax(contrib)
        src = idx[tc]
        src_val = a[src]
        if src_val < 1e-12:
            continue

        # Try several target positions
        n_targets = 5
        for _ in range(n_targets):
            dst = rng.integers(n)
            if dst == src:
                continue
            # Binary search for optimal fraction
            lo, hi = 0.0, 0.2
            base_score = best_score
            for _ in range(8):
                mid = (lo + hi) / 2.0
                amt = src_val * mid
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    hi = mid
                    continue
                a_trial = a_trial / ss
                sc = eval_score_fast(a_trial)
                if sc < base_score:
                    lo = mid
                    base_score = sc
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                else:
                    hi = mid

    return best_a, best_score
