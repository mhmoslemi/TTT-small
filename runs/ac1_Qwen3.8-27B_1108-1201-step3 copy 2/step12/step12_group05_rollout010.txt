def subgradient_mirror_descent(a0, time_limit, n_lags=5):
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2*n*fast_autocorr(a).max()
    
    # Use averaged iterate
    avg_a = np.zeros(n)
    count = 0
    
    eta = 1.0  # learning rate
    
    for t in range(1, max_iters+1):
        # Compute convolution
        c = fast_autocorr(a)
        
        # Find top lags
        top_lags = np.argsort(c)[::-1][:n_lags]
        
        # Compute weighted subgradient
        # Use softmax over top lags for soft subgradient
        top_c = c[top_lags]
        weights = np.exp(top_c - top_c.max())
        weights = weights / weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_lags):
            # g[i] += w * 2 * a[k-i] (with bounds check)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * 2 * a[k - idx]
        
        # Adaptive step size
        eta_t = eta / np.sqrt(t)  # or 1/sqrt(t)
        
        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta_t * g
        log_a -= log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        a = a_new
        avg_a += a
        count += 1
        
        # Check improvement periodically
        if t % 100 == 0:
            avg_candidate = avg_a / count
            score = 2*n*fast_autocorr(avg_candidate).max()
            if score < best_score:
                best_score = score
                best_a = avg_candidate.copy()
            
            score_last = 2*n*fast_autocorr(a).max()
            if score_last < best_score:
                best_score = score_last
                best_a = a.copy()
    
    return best_a, best_score

def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_mirror_descent(a0, time_limit, n_top_start=3, n_top_end=15,
                                eta0=2.0, decay_power=0.5, avg_power=0.7,
                                check_interval=50, restart_interval=3000):
    """
    Mirror descent on the simplex with subgradient of max(conv(a,a)).
    Uses KL-geometry (softmax) mirror map for natural handling of non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    log_a = np.log(a)

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    avg_log_a = log_a.copy()
    weight_sum = 1.0

    t0 = time.monotonic()
    t = 0
    stall = 0
    current_eta = eta0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Adaptive n_top: start small (focused), grow larger (smoother)
        frac = min(1.0, t / (restart_interval * 2))
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(1, min(n_top, len(a) * 2 - 1))

        # Compute convolution
        c = fast_autocorr(a)

        # Find top lags with softmax weights
        if n_top >= len(c):
            top_k = np.arange(len(c))
            weights = np.ones(len(c)) / len(c)
        else:
            top_k = np.argsort(c)[::-1][:n_top]
            top_c = c[top_k]
            # Temperature-controlled softmax: higher temp = more uniform
            temp = 1.0 / (n_top * top_c.max() + 1e-10)
            exps = np.exp(temp * (top_c - top_c.max()))
            weights = exps / exps.sum()

        # Compute weighted subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

        # Mirror descent update

        a_new = np.exp(log_a)

        if s < 1e-300:

        else:

            log_a = np.log(np.maximum(a_new, 1e-300))
        a = a_new

        # Update weighted geometric average
        w_t = t ** avg_power
        avg_log_a = avg_log_a * (weight_sum / (weight_sum + w_t)) + log_a * (w_t / (weight_sum + w_t))
        weight_sum += w_t

        # Periodic evaluation
        if t % check_interval == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += check_interval

            avg_a = np.exp(avg_log_a - avg_log_a.max())
            avg_a = avg_a / avg_a.sum()
            score_avg = 2.0 * n * fast_autocorr(avg_a).max()

                    log_a = np.log(np.maximum(a, 1e-300))
                    avg_log_a = log_a.copy()
                    weight_sum = 1.0
                    current_eta = eta0 * 0.1
                    stall = 0

def peak_shaving(a0, time_limit, seed=42):
    """Coordinate-descent peak shaving with adaptive fraction search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        top_k = np.argsort(c)[::-1][:6]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Build exclusion zone
            excl = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                excl.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in excl]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        # Chirp-like
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    elif choice == 4:
        # Multi-bump
        x = np.linspace(0, 1, n)
        a = np.exp(-10*(x-0.2)**2) + np.exp(-10*(x-0.5)**2) + np.exp(-10*(x-0.8)**2)
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3*np.cos(6*np.pi*x) + 0.15*np.cos(10*np.pi*x)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Spectral shaping (fast preprocessing)
    Phase 2: Subgradient mirror descent (main new method)
    Phase 3: Hybrid subgradient + coordinate descent
    Phase 4: Peak shaving polish
    Phase 5: Multi-length exploration
    Phase 6: Final polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 5:
        a_out = spectral_shaping(best_seq, n_iter=800, alpha=0.25)
        sc = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST [spectral]: {best_score:.6f}")

    # Phase 2: Main subgradient mirror descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Subgradient Mirror Descent (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 550.0)
    if t2 > 30:
        a_out, sc = subgradient_mirror_descent(
            best_seq, t2, n_top_start=3, n_top_end=12,
            eta0=2.0, decay_power=0.5, avg_power=0.7,
            check_interval=30, restart_interval=2000
        )
        sc = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST [subgrad_md]: {best_score:.6f}")
        else:
            print(f"  subgrad_md: {sc:.6f} (no improvement over {best_score:.6f})")

    # Phase 3: Hybrid subgradient + coordinate descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Hybrid SubGrad+CD (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 300.0)
    if t3 > 20:
        a_out, sc = subgradient_cd_hybrid(best_seq, t3, n_top=5)
        sc = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST [hybrid]: {best_score:.6f}")
        else:
            print(f"  hybrid: {sc:.6f} (no improvement)")

    # Phase 4:Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak Shaving (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 200.0)
    if t4 > 10:
        a_out, sc = peak_shaving(best_seq, t4, seed=777)
        sc = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST [peak_shave]: {best_score:.6f}")
        else:
            print(f"  peak_shave: {sc:.6f}")

    # Phase 5: Try slightly different lengths
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Length exploration (remaining={remaining:.0f}s) ===")
    for length_factor in [1.02, 0.98, 1.05]:
        if remaining < 30:
            break
        n_try = int(round(n_base * length_factor))
        n_try = max(100, min(n_try, 5000))
        if n_try == n_base:
            continue
        t5 = min(remaining * 0.15, 120.0)
        if t5 < 15:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        # Quick spectral + subgradient
        a_spec = spectral_shaping(init_try, n_iter=200, alpha=0.2)
        a_out, sc = subgradient_mirror_descent(
            a_spec, t5 * 0.6, n_top_start=3, n_top_end=10,
            eta0=1.5, decay_power=0.5, avg_power=0.7,
            check_interval=30, restart_interval=1000
        )
        sc = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST [len={n_try}]: {sc:.6f}")
        else:
            print(f"  len={n_try}: {sc:.6f} (no improvement)")
        remaining = time_budget - (time.monotonic() - t_global_start)

    # Phase 6: Final polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final Polish (remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t6a = min(remaining * 0.4, 80.0)
        a_out, sc = peak_shaving(best_seq, t6a, seed=999)
        sc = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST [final_shave]: {best_score:.6f}")

    if remaining > 15:
        t6b = min(remaining * 0.4, 60.0)
        a_out, sc = subgradient_mirror_descent(
            best_seq, t6b, n_top_start=2, n_top_end=5,
            eta0=0.5, decay_power=0.7, avg_power=0.8,
            check_interval=20, restart_interval=1000
        )
        sc = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST [final_subgrad]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
