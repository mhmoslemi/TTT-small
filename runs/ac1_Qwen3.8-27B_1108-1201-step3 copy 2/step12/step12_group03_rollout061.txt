"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Nesterov accelerated method with weighted running average
2. Smooth log-sum-exp gradient refinement with adaptive tau
3. Aggressive peak shaving via multiplicative weight updates
4. Multi-length exploration with interpolation

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely call the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_to_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max_k (a*a)_k using softmax weights."""
    n = len(a)
    b = fast_autocorr(a)
    b_shifted = b - b.max()
    w = np.exp(b_shifted / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w_rev, fft_size)
    conv_result = irfft(A * W, fft_size)[:conv_len]
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return grad


def get_lp_solver():
    """Find a working LP solver by testing on a small problem."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        T[j:end, j] = a_fixed[:end - j]
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated LP best-response method.
    Tracks weighted running average for O(1/T) convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Weighted running average (exponential weighting, favoring recent iterates)
    weights_sum = 0.0
    weighted_avg = np.zeros(n)
    base_weight = 1.0

    # Also track simple running average
    simple_avg = np.zeros(n)
    n_avg = 0

    iter_count = 0
    stall_count = 0
    last_improvement_iter = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1

        # Nesterov schedule
        if t < 50:
            alpha_t = 2.0 / (t + 1.0)
        else:
            alpha_t = 2.0 / (t + 2.0)

        # Nesterov update for y
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s

        y = y_new
        a = a_new

        # Score current iterate
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
            last_improvement_iter = t
        else:
            stall_count += 1

        # Score y
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
            last_improvement_iter = t

        # Weighted running average (geometric weighting)
        w_t = base_weight * (1.001 ** t)
        weighted_avg = (weights_sum * weighted_avg + w_t * a_new) / (weights_sum + w_t)
        weights_sum += w_t

        if t % 10 == 0:
            score_wavg = 2.0 * n * fast_autocorr(weighted_avg).max()
            if score_wavg < best_score:
                best_score = score_wavg
                best_a = weighted_avg.copy()
                stall_count = 0
                last_improvement_iter = t

        # Simple running average
        simple_avg += a_new
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = simple_avg / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
                last_improvement_iter = t

        if t % 200 == 0 and verbose:
            elapsed = time.monotonic() - t0

        # Periodic restart from best if stalled

            if verbose:
                print(f"    Restart at t={t} from best (stall={stall_count})")
            y = best_a.copy()

            weighted_avg = best_a.copy()
            weights_sum = 1.0
            simple_avg = best_a.copy()
            n_avg = 1
            stall_count = 0
            base_weight = 1.0

    a = project_to_simplex(a)

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    tau_schedule = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        phase_time = min((time_limit - (time.monotonic() - t0)) / len(tau_schedule), 60.0)
        n_iters = max(50, int(1200 * tau))  # More iters for larger tau
        lr = tau * 0.5 / n  # Initial learning rate
        print(f"    GradRefine: tau={tau}, lr={lr:.6f}, iters~{n_iters}")

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            grad = smoothed_gradient(a, tau)
            cur_score = 2.0 * n * fast_autocorr(a).max()

            # Backtracking line search
            improved = False
            for ls in range(15):
                a_try = project_to_simplex(a - lr * grad)
                a_try_s = a_try.sum()
                if a_try_s < 1e-15:
                    break
                a_try = a_try / a_try_s
                sc = 2.0 * n * fast_autocorr(a_try).max()
                if sc < cur_score - 1e-12:
                    a = a_try
                    improved = True
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    break
                lr *= 0.5
            if not improved:
                # Try a very small step
                a_try = project_to_simplex(a - lr * grad)
                a_try_s = a_try.sum()
                if a_try_s > 1e-15:
                    a_try = a_try / a_try_s
                    a = a_try

            # Adaptive lr: increase slightly if we keep improving
            if improved and it % 20 == 19:
                lr *= 1.1

    return best_a, best_score


def peak_shave_multiplicative(a0, time_limit, seed=42):
    """
    Peak shaving via multiplicative weight updates focused on top peaks.
    For each top peak, reduce contributions by multiplicatively decreasing
    the responsible coefficients and redistributing to "safe" positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = project_to_simplex(a)

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        n_peaks = min(8, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue

            # Find top contributors
            sorted_ci = np.argsort(contrib)[::-1][:min(10, len(idx))]
            for ci in sorted_ci:
                src = idx[ci]
                src2 = k - src

                # Find "safe" target positions (far from all top peaks)
                peak_lags = top_peaks[:4]
                safe_scores = np.zeros(n)
                for j in range(n):
                    min_dist = min(abs(j - k2) for k2 in peak_lags)
                    safe_scores[j] = min_dist - 0.5 * a[j] * n
                # Try top safe positions
                safe_targets = np.argsort(safe_scores)[::-1][:5]

                for dst in safe_targets:
                    if dst == src and dst == src2:
                        continue
                    # Multiplicative reduction
                    for factor in [0.99, 0.97, 0.95, 0.90, 0.85]:
                        a_try = a.copy()
                        red1 = a_try[src] * (1 - factor)
                        a_try[src] *= factor
                        if src2 != src:
                            red2 = a_try[src2] * (1 - factor)
                            a_try[src2] *= factor
                            red1 += red2
                        # Redistribute to dst
                        a_try[dst] += red1
                        if (a_try < -1e-12).any():
                            continue
                        a_try = np.maximum(a_try, 0)
                        a_try_s = a_try.sum()
                        if a_try_s < 1e-15:
                            continue
                        a_try = a_try / a_try_s
                        sc = 2.0 * n * fast_autocorr(a_try).max()
                        if sc < best_score - 1e-13:
                            a = a_try
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random kick
            if stall % 5 == 0:
                n_kick = rng.integers(1, 4)
                srcs = rng.choice(n, size=n_kick, replace=False)
                dsts = rng.choice(n, size=n_kick, replace=False)
                a_try = a.copy()
                for s_idx, d_idx in zip(srcs, dsts):
                    frac = rng.uniform(0.001, 0.01)
                    amt = a_try[s_idx] * frac
                    a_try[s_idx] -= amt
                    a_try[d_idx] += amt
                if (a_try < -1e-12).any():
                    continue
                a_try = np.maximum(a_try, 0)
                a_try = project_to_simplex(a_try)
                sc = 2.0 * n * fast_autocorr(a_try).max()
                if sc < best_score:
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interleaved_refine(a0, time_limit, solver, verbose=True):
    """
    Interleave short Nesterov bursts with gradient refinement.
    This helps escape local minima of either method.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Each round: 30s Nesterov + 15s gradient
    round_time = 45.0
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break
        round_num += 1

        # Nesterov burst
        nl_time = min(30.0, remaining * 0.6)
        if nl_time > 5:
            a_nl, sc_nl, _ = nesterov_lp_optimize(best_a, nl_time, solver, verbose=False)
            if sc_nl < best_score:
                best_score = sc_nl
                best_a = a_nl.copy()
                if verbose:
                    print(f"    InterRefine round {round_num} NLP: {best_score:.6f}")

        # Gradient burst
        gr_time = min(15.0, remaining * 0.3)
        if gr_time > 3 and time.monotonic() - t0 < time_limit - 5:
            a_gr, sc_gr = gradient_refine(best_a, gr_time)
            if sc_gr < best_score:
                best_score = sc_gr
                best_a = a_gr.copy()
                if verbose:
                    print(f"    InterRefine round {round_num} GRAD: {best_score:.6f}")

        # Peak shave burst
        ps_time = min(10.0, remaining * 0.1)
        if ps_time > 3 and time.monotonic() - t0 < time_limit - 3:
            a_ps, sc_ps = peak_shave_multiplicative(best_a, ps_time, seed=100 + round_num)
            if sc_ps < best_score:
                best_score = sc_ps
                best_a = a_ps.copy()
                if verbose:
                    print(f"    InterRefine round {round_num} PS: {best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining LP Nesterov,
    gradient refinement, and peak shaving at multiple lengths.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                n_base = len(parent)
                print(f"Parent: n={n_base}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        # Default initialization: cosine-shaped
        best_seq = np.cos(np.linspace(0, np.pi, n_base) / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Main Nesterov LP at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t1, solver, verbose=True)
        print(f"  NLP done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"nlp_n{n_base}")

    # Phase 2: Gradient refinement
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Gradient Refinement at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 30:
        a_out, sc_out = gradient_refine(best_seq, t2)
        print(f"  GradRefine done: score={sc_out:.6f}")
        try_update(a_out, f"grad_n{n_base}")

    # Phase 3: Interleaved refinement (Nesterov + gradient + peak shave)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Interleaved Refinement at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 200.0)
    if t3 > 30:
        a_out, sc_out = interleaved_refine(best_seq, t3, solver, verbose=True)
        print(f"  InterRefine done: score={sc_out:.6f}")
        try_update(a_out, f"inter_n{n_base}")

    # Phase 4: Peak shaving on best
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak Shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 120.0)
    if t4 > 10:
        a_out, sc_out = peak_shave_multiplicative(best_seq, t4, seed=789)
        print(f"  PeakShave done: score={sc_out:.6f}")
        try_update(a_out, f"ps_n{len(best_seq)}")

    # Phase 5: Try nearby lengths
    for n_try in [n_base - 200, n_base + 200, n_base + 400]:
        if n_try < 300:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: Explore n={n_try} (remaining={remaining:.0f}s) ===")
        t5 = min(remaining * 0.15, 150.0)
        if t5 < 40:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp_optimize(init_try, t5, solver, verbose=False)
        print(f"  NLP at n={n_try}: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"nlp_n{n_try}")

        # Quick gradient on this
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 50:
            t5g = min(remaining * 0.08, 60.0)
            if t5g > 10:
                a_out, sc_out = gradient_refine(a_out, t5g)
                try_update(a_out, f"grad_n{n_try}")

    # Phase 6: Final polish - short interleaved round
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final Polish at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.3, 120.0)
    if t6 > 20:
        a_out, sc_out = interleaved_refine(best_seq, t6, solver, verbose=True)
        try_update(a_out, "final_polish")

    # Final: peak shave with long budget
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t_final = min(remaining * 0.5, 60.0)
        if t_final > 5:
            a_out, sc_out = peak_shave_multiplicative(best_seq, t_final, seed=999)
            try_update(a_out, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
