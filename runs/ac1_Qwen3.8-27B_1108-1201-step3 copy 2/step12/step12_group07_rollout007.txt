    def get_toeplitz_k(k):
        """Get the matrix B_k such that a^T B_k a = (a*a)_k"""
        B = np.zeros((n, n))
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                B[i, j] = 1.0
        return B

    def solve_reduced_qcqP(active_lags, a_init):
        """Solve QCQP with given active lags using trust-constr"""
        # Variables: x = [a_0, ..., a_{n-1}, M]
        m = n + 1
        
        # Precompute B_k matrices
        B_mats = [get_toeplitz_k(k) for k in active_lags]
        
        def obj(x):
            a_vec = x[:n]
            M_val = x[n]
            return M_val
        
        def obj_grad(x):
            a_vec = x[:n]
            M_val = x[n]
            g = np.zeros(m)
            g[:n] = 0.0  # dM/da = 0 (M is the last variable)
            g[n] = 1.0
            return g
        
        # Constraints: a^T B_k a - M <= 0 for each k
        def constraints_fun(x):
            a_vec = x[:n]
            M_val = x[n]
            vals = np.zeros(len(active_lags))
            for idx, B in enumerate(B_mats):
                vals[idx] = a_vec @ B @ a_vec - M_val
            return vals
        
        def constraints_jac(x):
            a_vec = x[:n]
            M_val = x[n]
            J = np.zeros((len(active_lags), m))
            for idx, B in enumerate(B_mats):
                J[idx, :n] = 2.0 * (B @ a_vec)
                J[idx, n] = -1.0
            return J
        
        # Equality constraint: sum(a) = 1
        def eq_fun(x):
            return np.array([x[:n].sum() - 1.0])
        
        def eq_jac(x):
            J = np.zeros((1, m))
            J[0, :n] = 1.0
            return J
        
        bounds = [(0.0, None)] * n + [(1e-10, None)]
        
        x0 = np.append(a_init, best_score / (2.0 * n))
        
        from scipy.optimize import NonlinearConstraint, LinearConstraint
        
        con1 = NonlinearConstraint(constraints_fun, -np.inf, 0.0, jac=constraints_jac)
        con2 = LinearConstraint(eq_jac(x0), 0.0, 0.0)
        
        try:
            res = minimize(
                fun=obj, x0=x0, args=(), jac=obj_grad, method='trust-constr',
                constraints=[con1, con2], bounds=bounds,
                options={'maxiter': 200, 'gtol': 1e-10, 'xtol': 1e-12,
                         'verbose': 0}
            )
            a_new = np.maximum(res.x[:n], 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            return a_new
        except Exception:
            return a_init
    
    # Active set iterations
    for outer_it in range(10):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve QCQP with current active set
        a_new = solve_reduced_qcqP(sorted(active_set), a)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        
        # Update active set
        b = fast_convolve(a)
        max_b = b.max()
        # Keep lags where conv is within 1% of max
        threshold = max_b * 0.99
        new_active = set(np.where(b >= threshold)[0].tolist())
        
        # Also ensure the true max is always in the active set
        new_active.add(int(np.argmax(b)))
        
        # Limit size

            new_active2.add(int(np.argmax(b)))

                break

        active_set = new_active
        print(f"    QCQP outer={outer_it}: active_set_size={len(active_set)}, score={best_score:.8f}")
    
    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lse_lbgfs(a0, time_limit, T_start=80.0, T_end=0.02, n_stages=25):
    n = len(a0)

    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        iters = 50 if si < n_stages // 3 else 30 if si < 2 * n_stages // 3 else 20
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22,

def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a
        
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0005 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new
        
        a = a_new
    
    return best_a, best_score

def peak_targeted_refine(a0, time_limit, max_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.05
    last_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.92
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)
        
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa
        
        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.002, 0.15)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturb_sequence(a, rng, scale=0.1):
    """Add structured perturbation to escape local minima."""
    n = len(a)
    # Random smooth perturbation
    noise = np.zeros(n)
    # Low-frequency noise
    freqs = rng.integers(1, 10, size=10)
    amps = rng.uniform(0.5, 1.0, size=10)
    phases = rng.uniform(0, 2 * np.pi, size=10)
    x = np.arange(n)
    for f, a0, p in zip(freqs, amps, phases):
        noise += a0 * np.sin(2 * np.pi * f * x / n + p)
    # High-frequency noise
    noise += rng.normal(0, 0.1, n)
    
    perturbed = a + scale * noise / (np.abs(noise).max() + 1e-10)
    perturbed = np.maximum(perturbed, 0)
    s = perturbed.sum()
    if s < 1e-15:
        return a.copy()
    return perturbed / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov LP mirror descent (fast convergence to local Nash equilibrium)
2. Active-set QCQP refinement via trust-constr (exploits that few lags are critical)
3. LogSumExp L-BFGS multi-stage annealing (smooth polishing)
4. Peak-targeted multiplicative weights (complementary attack on peaks)
5. Perturbation-based restarts with structured low+high frequency noise
6. Systematic length exploration around the parent length

import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None

    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:

def get_toeplitz_diag(k, n):
    """Return diag such that a^T diag a / 2 = (a*a)_k for the self-convolution at lag k."""
    diag = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        j = k - i
        if 0 <= j < n:
            if i == j:
                diag[i] += 1.0
            else:
                diag[i] += 1.0
    return diag


def active_set_qcqP(a0, time_limit, K_max=60, max_outer=12):
    """Active-set QCQP: minimize M subject to a^T B_k a <= M for active lags k, sum(a)=1, a>=0."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    b = fast_convolve(a)
    max_b = b.max()
    active_set = set(np.where(b >= max_b * 0.95)[0].tolist())
    active_set.add(int(np.argmax(b)))

    # Precompute Toeplitz diagonals
    def get_dist_matrix(k, A):
        diag = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                if i == j:
                    diag[i] += 1.0
                else:
                    diag[i] += 1.0
        return diag * A

    def solve_reduced(active_lags, a_init):
        m = n + 1
        B_diags = [get_toeplitz_diag(k, n) for k in active_lags]
        n_act = len(active_lags)

        def obj(x):
            return x[n]

        def obj_grad(x):
            g = np.zeros(m)
            g[n] = 1.0
            return g

        def con_fun(x):
            a_v = x[:n]
            M_v = x[n]
            vals = np.zeros(n_act)
            for idx, d in enumerate(B_diags):
                vals[idx] = np.dot(a_v, d * a_v) / 2.0 - M_v
            return vals

        def con_jac(x):
            a_v = x[:n]
            M_v = x[n]
            J = np.zeros((n_act, m))
            for idx, d in enumerate(B_diags):
                J[idx, :n] = d * a_v
                J[idx, n] = -1.0
            return J

        def eq_fun(x):
            return np.array([x[:n].sum() - 1.0])

        def eq_jac(x):
            J = np.zeros((1, m))
            J[0, :n] = 1.0
            return J

        bounds = [(0.0, None)] * n + [(1e-10, None)]
        x0 = np.append(a_init.copy(), best_score / (2.0 * n) * 1.1)

        con1 = NonlinearConstraint(con_fun, -np.inf, 0.0, jac=con_jac)
        con2 = LinearConstraint(lambda x: np.array([x[:n].sum() - 1.0]), 0.0, 0.0,
                                jac=lambda x: np.vstack([np.append(np.ones(n), 0.0)]))

        try:
            res = minimize(
                fun=obj, x0=x0, jac=obj_grad, method='trust-constr',
                constraints=[con1, con2], bounds=bounds,
                options={'maxiter': 300, 'gtol': 1e-9, 'xtol': 1e-11}
            )
            a_new = np.maximum(res.x[:n], 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                return a_new / s_new
        except Exception:
            pass
        return a_init

    for outer_it in range(max_outer):
        if time.monotonic() - t0 > time_limit:
            break

        active_lags = sorted(active_set)
        if len(active_lags) > K_max:
            b_cur = fast_convolve(a)
            top = np.argsort(b_cur)[-K_max:]
            active_lags = sorted(set(top.tolist()))

        a_new = solve_reduced(active_lags, a)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.98
        new_active = set(np.where(b >= threshold)[0].tolist())
        new_active.add(int(np.argmax(b)))

        if len(new_active) > K_max:
            top = np.argsort(b)[-K_max:]
            new_active = set(top.tolist())

        if new_active == set(active_lags):
            threshold = max_b * 0.96
            new_active2 = set(np.where(b >= threshold)[0].tolist())
            new_active2.add(int(np.argmax(b)))
            if new_active2 == set(active_lags):
                break
            new_active = new_active2
            if len(new_active) > K_max:
                top = np.argsort(b)[-K_max:]
                new_active = set(top.tolist())

        active_set = new_active
        print(f"    QCQP outer={outer_it}: |active|={len(active_set)}, score={best_score:.8f}")

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbgfs(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=30):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = 50 if si < n_stages // 3 else 30 if si < int(2 * n_stages / 3) else 20
        try:
            res = minimize(
                fun=lambda zv: lse_objective_grad(zv, stage_T, n)[0],
                x0=z,
                jac=lambda zv: lse_objective_grad(zv, stage_T, n)[1],
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22, 'gtol': 1e-18,
                         'maxfun': iters * 300, 'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass
    return best_a, best_score


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            y = y / sy if sy > 1e-15 else a.copy()
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        a_new = a_new / sa if sa > 1e-15 else b_new.copy()

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0005 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new
        a = a_new

    return best_a, best_score


def peak_targeted_refine(a0, time_limit, max_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.05
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.92
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.002, 0.15)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, scale=0.1):
    n = len(a)
    noise = np.zeros(n)
    freqs = rng.integers(1, 12, size=10)
    amps = rng.uniform(0.5, 1.0, size=10)
    phases = rng.uniform(0, 2 * np.pi, size=10)
    x = np.arange(n)
    for f, a0, p in zip(freqs, amps, phases):
        noise += a0 * np.sin(2 * np.pi * f * x / n + p)
    noise += rng.normal(0, 0.1, n)
    perturbed = a + scale * noise / (np.abs(noise).max() + 1e-10)
    perturbed = np.maximum(perturbed, 0)
    s = perturbed.sum()
    if s < 1e-15:
        return a.copy()
    return perturbed / s


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n = n_base
        x = np.linspace(0, 1, n)
        best_seq = 0.5 + 0.4 * np.cos(2 * np.pi * (x + 0.1)) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    if t1 > 30:
        a_out, _ = nesterov_lp(best_seq, t1, solver=solver)
        try_update(a_out, "nesterov_lp_p1")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # Phase 2: Active-set QCQP
    print(f"\n=== Phase 2: Active-set QCQP (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(450.0, remaining * 0.3)
    if t2 > 30:
        a_qc, _ = active_set_qcqP(best_seq, t2, K_max=60, max_outer=12)

    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # Phase 3: LSE L-BFGS

    remaining = _global_deadline - time.monotonic()
    t3 = min(250.0, remaining * 0.2)
    if t3 > 20:
        a_lse = lse_lbgfs(best_seq, t3, T_start=100.0, T_end=0.02, n_stages=30)

    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # Phase 4: Peak-targeted + second QCQP

    if t4b > 20:

    remaining = _global_deadline - time.monotonic()

        for pert in range(4):
