import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    # Use FFT-based approach for speed: T @ b = convolve(a_fixed, b)
    # But for LP we need the explicit matrix. For n up to ~2000,
    # the matrix is (2n-1) x n which is manageable.
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    stall_count = 0
    stall_threshold = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_threshold = best_score
            stall_count = 0
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_threshold = best_score
            stall_count = 0

        # Track stalling
        if best_score > stall_threshold - 1e-10:
            stall_count += 1
        else:
            stall_count = 0

        # If stalled for too long, restart momentum
        if stall_count > 3000:
            y = a.copy()
            stall_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsbfgs_lse_adaptive(a0, time_limit, T_start=80.0, T_min=0.3, n_stages=25, iters_per_stage=40):
    """Adaptive LogSumExp L-BFGS with progress-based temperature control."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0
    prev_obj = None

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            if prev_obj is not None:
                obj_ratio = (sc - best_score) / (best_score + 1e-15)
                if obj_ratio > 0.005:
                    stagnant += 1
                else:
                    stagnant = max(0, stagnant - 1)

            if stagnant >= 3:
                T = max(T, T * 1.2)
                stagnant = 0
            else:
                T = max(T_min, T * 0.75)

            prev_obj = sc

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_flatten(a0, time_limit, n_rounds=50000):
    """
    Novel: Directly target the top peaks of the convolution.
    At each step, identify the top-k peaks and perform a gradient step
    that specifically reduces them, using a carefully chosen step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    for step in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks (top 10% or at least 5)
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) < 3:
            active = np.argsort(b)[-10:]
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        # Weight peaks by their value (emphasize the highest)
        peak_vals = b[active]
        weights = peak_vals ** 2
        weights = weights / weights.sum()

        # Compute weighted subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += weights[active == k] * a[k - idx]

        # Project to tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Polyak step with damping
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = 0.5 * gap / (grad_norm * grad_norm)
        eta = min(eta, 0.3)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Backtrack if needed
        if sc_new > cur_score * 1.005:
            for _ in range(8):
                eta *= 0.5
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                sc_new = score_normalized(a_new)
                if sc_new < cur_score:
                    break

        z = z_new
        a = a_new

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakFlatten step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def polynomial_init(n, degree=4, rng=None):
    """
    Generate a sequence using Bernstein polynomial basis.
    a[i] = sum_{j=0}^{d} b_j * C(d,j) * (i/n)^j * (1-i/n)^(d-j)
    with random positive coefficients b_j.
    """
    if rng is None:
        rng = np.random.default_rng()

    # Random Bernstein coefficients (positive)
    b = rng.uniform(0.1, 2.0, size=degree + 1)

    x = np.arange(n) / (n - 1)
    a = np.zeros(n)
    for j in range(degree + 1):
        binom = 1.0
        # C(degree, j)
        from math import comb
        binom = float(comb(degree, j))
        a += b[j] * binom * (x ** j) * ((1 - x) ** (degree - j))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remain={remain:.0f}s)")

    # === Phase 0: Quick polynomial optimization for multiple n ===
    print(f"\n=== Phase 0: Polynomial init scan ===")
    t0_phase0 = time.monotonic()
    n_candidates = [1200, 1500, 1800, 2000, 2500]
    poly_inits = {}

    for n_try in n_candidates:
        if time.monotonic() > global_deadline - 600:
            break
        best_poly_score = float('inf')
        best_poly_a = None
        for trial in range(8):
            a_try = polynomial_init(n_try, degree=4, rng=rng)
            sc = score_normalized(a_try)
            if sc < best_poly_score:
                best_poly_score = sc
                best_poly_a = a_try.copy()
            # Also try with different degree
            a_try2 = polynomial_init(n_try, degree=5, rng=rng)
            sc2 = score_normalized(a_try2)
            if sc2 < best_poly_score:
                best_poly_score = sc2
                best_poly_a = a_try2.copy()

        if best_poly_a is not None:
            poly_inits[n_try] = best_poly_a
            print(f"    n={n_try}: poly_init score={best_poly_score:.6f}")

    # Also create inits from parent by interpolation
    if best_seq is not None:
        for n_try in n_candidates:
            if n_try not in poly_inits:
                poly_inits[n_try] = interpolate_sequence(best_seq, n_try)
            else:
                # Keep the better one
                parent_interp = interpolate_sequence(best_seq, n_try)
                if score_normalized(parent_interp) < score_normalized(poly_inits[n_try]):
                    poly_inits[n_try] = parent_interp

    # === Phase 1: Nesterov LP on parent length (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP at parent n (score={best_score:.8f}) ===")
    if best_seq is not None:
        t1 = min(400.0, (global_deadline - time.monotonic()) * 0.25)
        a_out, _ = nesterov_lp(best_seq, t1, solver=solver)
        try_update(a_out, "nesterov_parent")

    # === Phase 2: Nesterov LP on best polynomial init at larger n ===
    print(f"\n=== Phase 2: Nesterov LP multi-n (score={best_score:.8f}) ===")
    # Sort n by current best score for that n
    n_score_pairs = []
    for n_try, a_init in poly_inits.items():
        sc = score_normalized(a_init)
        n_score_pairs.append((sc, n_try, a_init))
    n_score_pairs.sort(key=lambda x: x[0])

    # Try the top 3 n values
    for sc_init, n_try, a_init in n_score_pairs[:4]:
        if time.monotonic() > global_deadline - 200:
            break
        t2 = min(200.0, (global_deadline - time.monotonic()) * 0.2)
        if t2 < 30:
            break
        print(f"  Nesterov at n={n_try} (init_score={sc_init:.6f}), budget={t2:.0f}s")
        a_out, _ = nesterov_lp(a_init, t2, solver=solver)
        try_update(a_out, f"nesterov_n{n_try}")

    # === Phase 3: Peak flattening on best sequence ===
    print(f"\n=== Phase 3: Peak Flattening (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 60:
        t3 = min(150.0, remaining * 0.3)
        a_pf, _ = peak_flatten(best_seq, t3, n_rounds=100000)
        try_update(a_pf, "peak_flatten")

        # Peak flatten from the result
        remaining = global_deadline - time.monotonic()
        if remaining > 30:
            a_pf2, _ = peak_flatten(best_seq, remaining * 0.4, n_rounds=50000)
            try_update(a_pf2, "peak_flatten2")

    # === Phase 4: LSE refinement ===
    print(f"\n=== Phase 4: LSE Refinement (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 60:
        t4 = min(150.0, remaining * 0.3)
        a_lse, _ = lsbfgs_lse_adaptive(best_seq, t4, T_start=50.0, T_min=0.1, n_stages=20, iters_per_stage=35)
        try_update(a_lse, "lse_main")

        remaining = global_deadline - time.monotonic()
        if remaining > 30:
            a_lse2, _ = lsbfgs_lse_adaptive(best_seq, remaining * 0.4, T_start=15.0, T_min=0.05, n_stages=15, iters_per_stage=30)
            try_update(a_lse2, "lse_cold")

    # === Phase 5: Combined short passes ===
    print(f"\n=== Phase 5: Combined polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 30:
        # Short Nesterov
        t5a = min(60.0, remaining * 0.3)
        if t5a > 20:
            a_f1, _ = nesterov_lp(best_seq, t5a, solver=solver, max_iters=30000)
            try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        # Peak flatten again
        if remaining > 30:
            t5b = min(60.0, remaining * 0.4)
            a_f2, _ = peak_flatten(best_seq, t5b, n_rounds=50000)
            try_update(a_f2, "final_peak")

        remaining = global_deadline - time.monotonic()
        # LSE from low temp
        if remaining > 20:
            t5c = min(40.0, remaining * 0.5)
            a_f3, _ = lsbfgs_lse_adaptive(best_seq, t5c, T_start=10.0, T_min=0.05, n_stages=10, iters_per_stage=25)
            try_update(a_f3, "final_lse")

    # === Phase 6: Try one more n with best sequence interpolated ===
    print(f"\n=== Phase 6: One more n attempt (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 60:
        # Try n = len(best_seq) + 300
        n_extra = len(best_seq) + 300
        a_interp = interpolate_sequence(best_seq, n_extra)
        t6 = min(80.0, remaining * 0.5)
        print(f"  Trying n={n_extra}, budget={t6:.0f}s")
        a_out, _ = nesterov_lp(a_interp, t6 * 0.6, solver=solver, max_iters=20000)
        try_update(a_out, f"extra_n{n_extra}")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_pf3, _ = peak_flatten(a_out, remaining * 0.5, n_rounds=30000)
            try_update(a_pf3, f"extra_n{n.extra}_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    elapsed_total = time.monotonic() - (global_deadline - (budget_s - 30))
    print(f"Time used: {elapsed_total:.0f}s / {budget_s:.0f}s")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
