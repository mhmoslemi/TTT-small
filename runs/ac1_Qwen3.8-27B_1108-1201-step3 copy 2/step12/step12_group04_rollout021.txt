Actually wait - computing `max_k (a[k-j] - a[k-i])^+` is O(n) and we do it for each candidate (i,j) pair. If we check many pairs, it's O(n^2) total per iteration, which might be too slow.

**Better approach: just try the best pair and compute eps_max for it.**

1. For peak k*, find i* = argmax a[k*-i] and j* = argmin a[k*-j]. These are O(n) to find.
2. Compute eps_max by scanning all k: O(n).
3. Apply the move.
4. Update b incrementally (or batch and recompute via FFT every 100 moves).

Total per move: O(n). For n=1600, that's very fast. We can do millions of moves.

**But there's a subtlety**: after the move, the peak might shift to a different position. So we need to re-identify the peak periodically.

**Algorithm: Fast coordinate descent with batching**

    diff = a_shifted(j) - a_shifted(i)  # diff[k] = a[k-j] - a[k-i]
    positive_diff = diff[diff > 0]
    if len(positive_diff) > 0:

        eps_max = min(ratios[ratios > 0])  # careful with b_k > b_max (shouldn't happen)
    else:
        eps_max = a[i_src]  # can move all mass
    
    eps = min(eps_max * 0.9, a[i_src] * 0.5)  # safety
    
    a[i_src] -= eps
    a[i_dst] += eps
    
    # Periodically recompute b via FFT (every 50 moves)
    if iteration % 50 == 0:
        b = autocorr(a)

import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_and_grad(z, T):
    """Compute LogSumExp objective and gradient in z-space."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def logsumexp_polish(a_init, time_limit, T_stages=None):
    """
    LogSumExp-smoothed optimization using L-BFGS with multi-stage annealing.
    """
    from scipy.optimize import minimize

    n = len(a_init)
    a0 = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / s
    z = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [40.0, 25.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5]

    best_a = a0.copy()
    best_score = 2.0 * n * fast_autocorr(a0).max()

    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(30, min(150, int(remaining * 15)))

        def func(z_in):
            val, _ = lse_obj_and_grad(z_in, T)
            return val

        def jac(z_in):
            _, g = lse_obj_and_grad(z_in, T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14})
            z = res.x
            a_new = softmax_fn(z)
            sc = 2.0 * n * fast_autocorr(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        if si % 2 == 0:
            print(f"      LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search: multi-phase optimization combining fast coordinate descent,
    multi-peak refinement, LogSumExp polishing, and multi-length exploration.
    """
    rng = np.random.default_rng(777)
    t_global = time.monotonic()
    time_budget = 1740.0

    # Initialize from parent
    best_a = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            n_base = len(p)
            best_a = p / p.sum()
            b = fast_autocorr(best_a)
            best_score = 2.0 * n_base * b.max()
            print(f"Parent: n={n_base}, score={best_score:.10f}")
    except Exception:
        best_a = np.ones(n_base) / n_base
        best_score = 2.0 * n_base * fast_autocorr(best_a).max()
        print(f"Default init: n={n_base}, score={best_score:.10f}")

    def try_update(a, label=""):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        b = fast_autocorr(a)
        sc = 2.0 * len(a) * b.max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Fast coordinate descent (primary workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Coordinate descent n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_out, _ = coordinate_descent(best_a, t1, rng, label="main")
    try_update(a_out, "cd1")

    # === Phase 2: Multi-peak descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multi-peak descent (budget={remaining:.0f}s) ===")
    t2 = remaining * 0.25
    a_out, _ = multi_peak_descent(best_a, t2, rng, num_peaks=5)
    try_update(a_out, "mp")

    # === Phase 3: LogSumExp polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LogSumExp polish (budget={remaining:.0f}s) ===")
    t3 = remaining * 0.25
    a_out, _ = logsumexp_polish(best_a, t3)
    try_update(a_out, "lse")

    # === Phase 4: Another coordinate descent round on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: CD round 2 (budget={remaining:.0f}s) ===")
    t4 = remaining * 0.35
    a_out, _ = coordinate_descent(best_a, t4, rng, batch=50, safety=0.82, label="r2")
    try_update(a_out, "cd2")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (budget={remaining:.0f}s) ===")
    n_best = len(best_a)

    for offset in [120, -120, 240, -240]:
        nl = n_best + offset
        if nl < 800 or nl > 2800:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        tl = min(remaining * 0.4, 200.0)

        a_interp = interpolate_sequence(best_a, nl)
        a_out, _ = coordinate_descent(a_interp, tl, rng, batch=60, label=f"n{nl}")
        try_update(a_out, f"len{nl}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 15:
        n_fin = len(best_a)
        t6a = remaining * 0.5
        a_out, _ = coordinate_descent(best_a, t6a, rng, batch=30, safety=0.75, label="fin")
        try_update(a_out, "fin_cd")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_out, _ = logsumexp_polish(best_a, remaining * 0.8, T_stages=[3.0, 1.5, 0.8, 0.4])
        try_update(a_out, "fin_lse")

    # === Phase 7: Simulated annealing restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Annealed restarts (budget={remaining:.0f}s) ===")
        n_best = len(best_a)
        n_restarts = min(5, int(remaining // 60))

        for ri in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            # Perturb best solution
            a_pert = best_a.copy()
            noise_scale = 0.02 * (1.0 + 0.5 * ri)
            noise = rng.normal(0, noise_scale, n_best)
            a_pert = a_pert + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            tl = min(remaining * 0.6, 120.0)
            a_out, _ = coordinate_descent(a_pert, tl, rng, batch=50, label=f"ann{ri}")
            try_update(a_out, f"restart{ri}")

    # Final evaluation
    result_a = best_a
    result_a = np.maximum(result_a, 0.0)
    result_a = result_a / result_a.sum()
    final_b = fast_autocorr(result_a)
    final_score = 2.0 * len(result_a) * final_b.max()
    print(f"\nFINAL: score={final_score:.10f}, n={len(result_a)}")

    result = [float(max(0.0, min(1000.0, x))) for x in result_a]
    return result
