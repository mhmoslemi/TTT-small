def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

    # This is the cross-correlation of a and w at shift j

def mirror_descent_adaptive(a0, time_limit, T_start=50.0, T_end=1.0,
                            eta0=0.04, decay_rate=0.5, seed=42, verbose=True):
    """
    Mirror descent with adaptive temperature subgradient.
    Temperature anneals from T_start (flat weights) to T_end (peak-focused).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0
    iteration = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        iteration = t

        b = fast_autocorr(a)
        b_max = b.max()
        
        # Adaptive temperature: anneal from T_start to T_end
        total_iters = max(1, int(time_limit / 0.05))  # assume ~50ms per iter
        frac = min(1.0, t / float(max(100, total_iters // 2)))
        T = T_start * (T_end / T_start) ** frac
        
        # Exponential weights on all lags
        b_shift = T * (b - b.max())
        exp_b = np.exp(b_shift)
        w = exp_b / exp_b.sum()
        
        # Compute full subgradient using FFT
        grad = cross_corr_w_b(w, a)
        grad = np.maximum(grad, 0)  # ensure non-negative gradient
        
        # Learning rate with power decay
        eta = eta0 / (1.0 + t / 500.0) ** decay_rate
        
        # Multiplicative update (mirror map for simplex)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        a_new = a_new / s_new
        a = a_new

        # Check score periodically
        if t < 200:
            check_interval = 5
        elif t < 2000:
            check_interval = 10
        else:
            check_interval = 20

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Stall handling: adaptive perturbation
        if stall_count > 300:
            noise_scale = min(0.03, 0.005 * math.sqrt(stall_count / 50.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall_count = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.1f}, bmax={b_max:.6e}")

    return best_a, best_score

    Objective: (1/T)*log(sum(exp(T*b_k)))  (LSE approximation of max b)
    """
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    obj = (Tb_max + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # grad_a = 2 * cross_corr_w_b(w, a)
    grad_a = cross_corr_w_b(w, a)

    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lse_optimize(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=15, seed=42):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Scale iterations with remaining time
        frac_remaining = remaining / time_limit
        iters = max(10, int(60 * (0.5 + 0.5 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            val, _ = lse_obj_grad(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            a = np.maximum(a, 0)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 4 == 0 or si == n_stages - 1:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lse_obj_grad(z_in, stage_T):
    """Wrapper to avoid closure issues."""
    return lse_obj_grad_z(z_in, stage_T)


def targeted_peak_shaving(a0, time_limit, seed=123, verbose=True):
    """
    Advanced peak shaving: for each dominant peak lag, identify the top contributing
    positions and move mass to positions with minimum impact on all peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Get top peaks
        n_peaks = min(5, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        
        for k in top_peaks[:3]:  # Focus on top 3 peaks
            k = int(k)
            # Find contributing positions
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Source: position with max contribution
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            
            if a[src] < 1e-10:
                continue

            # Find best destination: minimize impact on all top peaks
            # Moving delta from src to dst changes b[k'] by:
            #   For peak at lag k': delta * (a[k'-dst] - a[k'-src]) (approx)
            # We want this to be negative (reduce peak) for the dominant peaks
            
            # Pre-compute "impact scores" for each potential destination
            # impact[dst] = sum over top peaks of a[peak - dst] (normalized)
            # We want to minimize this (prefer destinations that don't contribute to peaks)
            
            # For efficiency, only consider a subset of candidates
            n_candidates = max(20, n // 20)
            # Prefer positions with low a value and low peak contribution
            peak_contribution = np.zeros(n)
            for k2 in top_peaks[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                peak_contribution[idx2] += a[k2 - idx2]
            
            # Score: prefer low a and low peak contribution
            candidate_score = a + 0.5 * peak_contribution / max(peak_contribution.max(), 1e-15)
            # Exclude source
            candidate_score[src] = np.inf
            # Exclude positions very close to source (they'd just shift the peak)
            for d in range(-3, 4):
                pos = src + d
                if 0 <= pos < n:
                    candidate_score[pos] = np.inf
            
            # Pick bottom candidates
            sorted_candidates = np.argsort(candidate_score)[:n_candidates]
            
            if len(sorted_candidates) == 0:
                continue
                
            # Try a few destinations with line search
            best_trial = None
            best_trial_score = best_score
            
            for dst in sorted_candidates[:min(10, len(sorted_candidates))]:
                dst = int(dst)
                # Try different fractions
                for frac in [0.001, 0.003, 0.01, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_trial_score - 1e-14:
                        best_trial_score = sc
                        best_trial = a_trial.copy()

            if best_trial is not None:
                a = best_trial
                if best_trial_score < best_score:
                    best_score = best_trial_score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break  # Restart peak loop with new a

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 10 == 0:
                n_perturb = rng.integers(1, 4)
                for _ in range(n_perturb):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial / s_t
                            sc = fast_score(a_trial)
                            if sc < best_score:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                stall = 0
                                break

            if stall > 500:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PeakShave ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def local_search(a0, time_limit, seed=456, verbose=True):
    """
    Directional + random local search.
    Makes small perturbations targeted at reducing the peak autocorrelation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    no_improve = 0
    step_scale = 0.01

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_lag = int(np.argmax(b))
        
        # Determine if move is directional or random
        if rng.random() < 0.6:
            # Directional: move mass from peak contributor to low-impact position
            js = max(0, peak_lag - n + 1)
            je = min(n - 1, peak_lag)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[peak_lag - idx]
            
            # Pick top 3 contributors randomly
            top_k = min(3, len(idx))
            top_local = np.argsort(contrib)[::-1][:top_k]
            src = int(idx[rng.integers(top_k)])
            
            if a[src] < 1e-8:
                continue
            
            # Destination: random position far from peak contributors
            dst = int(rng.integers(n))
            if abs(dst - src) < 2:
                dst = (dst + n // 2) % n
        else:
            # Random pair
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
            if src == dst or a[src] < 1e-8:
                continue

        # Adaptive step size
        frac = step_scale * rng.uniform(0.5, 2.0)
        amt = a[src] * frac
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0)
        s_t = a_trial.sum()
        if s_t > 1e-15:
            a_trial = a_trial / s_t
            sc = fast_score(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                no_improve = 0
                step_scale = min(0.05, step_scale * 1.1)  # Slightly increase
            else:
                no_improve += 1
                if no_improve > 200:
                    step_scale = max(0.001, step_scale * 0.8)  # Decrease step
                    no_improve = 0

        if verbose and (int(time.monotonic() - t0) % 30 == 0):
            elapsed = time.monotonic() - t0
            print(f"    LocalSearch ({elapsed:.0f}s): best={best_score:.6f}, step={step_scale:.4f}")

    return best_a, best_score


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        # Multi-bump random
        a = np.ones(n)
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.5, 1.0)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)
    else:
        # Frequency-domain structured: flat spectrum with random phases
        freqs = np.zeros(n)
        freqs[0] = 1.0
        for f in range(1, n // 2 + 1):
            if rng.random() < 0.5:
                freqs[f] = rng.uniform(0.3, 1.0)
                freqs[n - f] = freqs[f]
        a = irfft(freqs, n)
        a = np.maximum(a, 0)

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Load parent, quick diverse exploration
    Phase 2: Adaptive-temperature mirror descent (main workhorse)
    Phase 3: LogSumExp L-BFGS refinement
    Phase 4: Targeted peak shaving
    Phase 5: Local search (directional + random)
    Phase 6: Cross-length exploration
    Phase 7: Final combined polish
    """
    import math
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1750.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = fast_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_parent = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = diverse_init(n_parent, rng, variant=0)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Quick diverse exploration (5% of time)
    print(f"\n=== Phase 1: Quick Exploration ===")
    t_explore = min(time_budget * 0.05, 80.0)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > t_explore > 10:
        # Try a few diverse inits with quick MD
        per_init = t_explore / 4
        for v in [4, 9, 10, 11]:
            if time.monotonic() - t_global > time_budget:
                break
            init = diverse_init(n_parent, rng, variant=v)
            a_out, _ = mirror_descent_adaptive(init, per_init, T_start=10.0, T_end=2.0,
                                                eta0=0.06, decay_rate=0.6, seed=v * 7 + 1, verbose=False)
            try_update(a_out, f"explore_v{v}")

    # Phase 2: Main mirror descent (30% of time)
    print(f"\n=== Phase 2: Main Mirror Descent (n={n_parent}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_md = min(remaining * 0.30, 520.0)
    if t_md > 30:
        a_out, _ = mirror_descent_adaptive(best_seq, t_md, T_start=80.0, T_end=0.8,
                                            eta0=0.035, decay_rate=0.5, seed=100, verbose=True)
        try_update(a_out, f"main_md_n{n_parent}")
        # Second MD run with different params
        remaining = time_budget - (time.monotonic() - t_global)
        t_md2 = min(remaining * 0.10, 160.0)
        if t_md2 > 20:
            # Perturb and re-optimize
            a_pert = best_seq + 0.02 * rng.standard_normal(n_parent)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a_out, _ = mirror_descent_adaptive(a_pert, t_md2, T_start=30.0, T_end=1.0,
                                                eta0=0.03, decay_rate=0.55, seed=200, verbose=False)
            try_update(a_out, f"perturb_md_n{n_parent}")

    # Phase 3: LSE L-BFGS (20% of time)
    print(f"\n=== Phase 3: LogSumExp L-BFGS (n={n_parent}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse = min(remaining * 0.20, 340.0)
    if t_lse > 30:
        a_out, _ = lse_optimize(best_seq, t_lse, T_start=80.0, T_end=0.2, n_stages=15, seed=300)

                a_out, _ = lse_optimize(a_pert2, t_lse2, T_start=20.0, T_end=0.1, n_stages=10, seed=400)
                try_update(a_out, f"lse_pert_n{n_parent}")

    # Phase 4: Targeted peak shaving (25% of time)

    # Phase 5: Local search (15% of time)
    print(f"\n=== Phase 5: Local Search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_ls = min(remaining * 0.15, 250.0)
    if t_ls > 10:
        a_out, _ = local_search(best_seq, t_ls, seed=600)
        try_update(a_out, f"ls_n{n_parent}")

    # Phase 6: Cross-length exploration (10% of time)

    remaining = time_budget - (time.monotonic() - t_global)
    t_cross = min(remaining * 0.10, 150.0)
    if t_cross > 20:
        for length_factor in [1.1, 0.9]:
            if time.monotonic() - t_global > time_budget:
                break
            new_n = int(round(n_parent * length_factor))
            if new_n < 100:
                continue
            init_new = interpolate_to_length(best_seq, new_n)
            t_sub = t_cross / 2
            a_out, _ = mirror_descent_adaptive(init_new, t_sub * 0.6, T_start=50.0, T_end=1.0,
                                                eta0=0.03, decay_rate=0.5, seed=700 + int(length_factor*100), verbose=False)
            try_update(a_out, f"cross_n{new_n}")
            if time.monotonic() - t_global < time_budget:
                a_out, _ = lse_optimize(a_out, t_sub * 0.4, T_start=30.0, T_end=0.3, n_stages=8, seed=800)
                try_update(a_out, f"cross_lse_n{new_n}")

    # Phase 7: Final combined polish (remaining time)
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 20:
        # Quick MD with low temperature (peak-focused)
        t_f1 = min(remaining * 0.3, 60.0)
        if t_f1 > 5:
            a_out, _ = mirror_descent_adaptive(best_seq, t_f1, T_start=10.0, T_end=0.5,
                                                eta0=0.02, decay_rate=0.7, seed=900, verbose=False)
            try_update(a_out, "final_md")

        # Quick LSE with very low temperature
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.3, 50.0)
        if t_f2 > 5:
            a_out, _ = lse_optimize(best_seq, t_f2, T_start=10.0, T_end=0.05, n_stages=8, seed=950)
            try_update(a_out, "final_lse")

        # Final peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.25, 40.0)
        if t_f3 > 5:
            a_out, _ = targeted_peak_shaving(best_seq, t_f3, seed=999)
            try_update(a_out, "final_ps")

        # Final local search
        remaining = time_budget - (time.monotonic() - t_global)
        t_f4 = min(remaining * 0.2, 30.0)
        if t_f4 > 3:
            a_out, _ = local_search(best_seq, t_f4, seed=1000, verbose=False)

    # Final evaluation with official function
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            if fs < best_score:
                best_score = fs
            print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
        else:
            print(f"\nWARNING: Official eval returned {fs}, using fast_score={best_score:.6f}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
