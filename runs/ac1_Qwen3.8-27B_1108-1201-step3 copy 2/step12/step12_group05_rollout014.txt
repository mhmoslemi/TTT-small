"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_normalized(a):
    """Score for a normalized (sum=1) sequence."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def spectral_shaping(a, n_iter=500, alpha_start=0.5, alpha_end=0.05):
    """Spectral shaping with annealing alpha."""
    n = len(a)
    a = normalize(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = score_normalized(a)

    alphas = np.geomspace(alpha_start, alpha_end, n_iter)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alphas[i] / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = score_normalized(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b)_k = sum_j a_fixed[k-j]*b[j]."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, T_matrix=None):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    if T_matrix is None:
        T_matrix = build_conv_matrix(a_fixed)
    constraints = [T_matrix @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_game_mwu(a0, time_limit, solver, max_iters=50000):
    """
    LP game with multiplicative weight updates (MWU) on the simplex.
    Uses exponential weights for better convergence properties.
    Tracks both primal iterate and running average.
    """
    n = len(a0)
    a = normalize(a0)
    y = a.copy()  # The "mirror" variable for the dual player
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Running average for both players
    avg_a = a.copy()
    avg_y = a.copy()
    n_avg = 1

    eta = 1.0 / np.sqrt(np.log(n))  # MWU step size

    T_mat_cache = None
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP best response: given y, find b* that maximizes min over a
        # Actually: given y (which represents the current "a" in the game),
        # find b* = argmax_b min_a b^T T(a) ... no.
        #
        # The game: Player A picks a to minimize max_k (a*a)_k
        #          Player B picks b (prob over k) to maximize sum_k b_k (a*a)_k
        # Best response of B to a: b* = argmax_b sum_k b_k (a*a)_k s.t. b in simplex
        #   This is just b* concentrated on argmax_k (a*a)_k
        # Best response of A to b: a* = argmin_a sum_k b_k (a*a)_k s.t. a in simplex
        #   This is a QP: min a^T (sum b_k S_k) a
        #
        # Alternative LP formulation (from the previous code):
        # Given a_fixed, find b* and M minimizing M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0
        # This gives the best response of B to a in the sense of the LP relaxation.

        b_star, M_val = solve_lp_best_response(y, solver, T_mat_cache)
        if b_star is None:
            if time.monotonic() - t0 > time_limit - 5:
                break
            continue

        # Update y using MWU (multiplicative weights for the dual player)
        # y represents the "certifying" distribution
        alpha_t = 2.0 / (np.log(it + 2) + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        # Nesterov extrapolation for the primal
        momentum = (it - 1.0) / (it + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / s

        y = y_new
        a = a_new

        # Track averages
        avg_a += a
        avg_y += y
        n_avg += 1

        # Evaluate
        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            score_avg = score_normalized(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

            avg_y_check = avg_y / n_avg
            score_avy = score_normalized(avg_y_check)
            if score_avy < best_score:
                best_score = score_avy
                best_a = avg_y_check.copy()

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-MWU it={it} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def top_k_peak_optimization(a0, time_limit, K=5, n_subgrad_iters=20000):
    """
    Direct subgradient optimization focusing on top-K peaks.
    Uses a "soft max" of the top K autocorrelation values as a smooth approximation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0

    # Temperature for soft-max over top-K
    T_start = 50.0
    T_end = 1.0
    n_total = min(n_subgrad_iters, int(time_limit * 200))

    for it in range(n_total):
        if time.monotonic() - t0 > time_limit - 1:
            break

        # Adaptive temperature
        frac = it / max(1, n_total)
        T = T_start * (T_end / T_start) ** frac

        b = fast_autocorr(a)

        # Get top-K peaks
        top_idx = np.argsort(b)[::-1][:K]
        top_vals = b[top_idx]

        # Soft max of top-K
        w = np.exp((top_vals - top_vals.max()) * T)
        w = w / w.sum()

        # Weighted combination of peak positions
        # Gradient of sum_k w_k * (a*a)_k w.r.t. a
        # (a*a)_k = sum_j a[j] * a[k-j]
        # d/d(a[i]) (a*a)_k = a[k-i] + a[i-k] (if valid)
        # = contribution from position i to peak k

        grad = np.zeros(n)
        for kk, ww in zip(top_idx, w):
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            for j in range(js, je + 1):
                partner = kk - j
                if 0 <= partner < n:
                    grad[j] += ww * a[partner]
                    if partner != j:
                        grad[partner] += ww * a[j]

        # Normalize gradient
        grad = 2.0 * grad
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm

        # Project gradient onto tangent space of simplex
        mean_grad = grad.mean()
        grad_tangent = grad - mean_grad

        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad_tangent
        v = beta2 * v + (1 - beta2) * grad_tangent ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        lr = 0.01 / np.sqrt(1 + 0.01 * it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project back to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a = a_new

        if it % 50 == 0:
            score = score_normalized(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      TopK-Opt it={it} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.1f}")

    return best_a, best_score


def peak_shaving_aggressive(a0, time_limit, seed=123):
    """Aggressive peak shaving with multi-scale moves."""
    n = len(a0)
    rng = np.random.default rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src = idx[np.argmax(contrib)]

            # Find positions NOT involved in any top peak
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j in range(js2, je2 + 1):
                    peak_areas.add(j)
                    peak_areas.add(k2 - j)

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Sort by a[j] ascending (prefer sparse targets)
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                dst = dst_pool[rng.integers(len(dst_pool))]
                amt = a[src] * frac
                if amt < 1e-15 or a[src] - amt < -1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = score_normalized(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.06)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-15:
                    a_trial = a_trial / s
                    score = score_normalized(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 400:
                # Larger perturbation to escape
                n_perturb = rng.integers(3, 10)
                sources = rng.choice(n, size=n_perturb, replace=False)
                targets = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for si, ti in zip(sources, targets):
                    amt = a_trial[si] * 0.1
                    a_trial[si] -= amt
                    a_trial[ti] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-15:
                    a_trial = a_trial / s
                    score = score_normalized(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                if stall > 600:

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping with annealed alpha
2. LP game with multiplicative weight updates and running averages
3. Top-K peak subgradient optimization with Adam

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):

def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_normalized(a):
    """Score for a normalized (sum=1) sequence: 2*n*max_conv."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def spectral_shaping(a, n_iter=500, alpha_start=0.5, alpha_end=0.05):
    """Spectral shaping with annealed alpha for flat spectrum."""
    n = len(a)
    a = normalize(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_score = score_normalized(a)

    if n_iter <= 1:
        alphas = np.array([alpha_start])
    else:
        alphas = np.geomspace(alpha_start, alpha_end, n_iter)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        if non_dc.max() < 1e-30:
            break
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b)_k = (a_fixed * b)_k."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def lp_game_mwu(a0, time_limit, solver, max_iters=50000):
    """

def top_k_peak_optimization(a0, time_limit, K=5, n_total_iters=30000):
    """
    Subgradient optimization on a soft-max of top-K autocorrelation peaks.
    Uses Adam optimizer in the z-space (log parameterization).
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Work in log-space for positivity
    z = np.log(np.maximum(a, 1e-30))

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    T_start = 100.0
    T_end = 1.0

    for it in range(n_total_iters):
        if time.monotonic() - t0 > time_limit - 1:
            break

        frac = it / max(1, n_total_iters)
        T = T_start * (T_end / T_start) ** frac

        # Recover a from z
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()

        b = fast_autocorr(a)

        # Top-K peaks
        top_idx = np.argsort(b)[::-1][:K]
        top_vals = b[top_idx]

        # Soft weights
        w = np.exp((top_vals - top_vals.max()) * T)
        w = w / w.sum()

        # Gradient: d/d(a[i]) [sum_k w_k * (a*a)_k]
        grad = np.zeros(n)
        for ki in range(K):
            k = top_idx[ki]
            ww = w[ki]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += 2.0 * ww * a[partner]

        # Project to tangent space of simplex
        mean_g = grad.mean()
        grad_t = grad - mean_g

        # Adam update on z
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad_t
        v = beta2 * v + (1 - beta2) * grad_t ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        lr = 0.01 / np.sqrt(1 + 0.01 * it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 50 == 0:
            score = score_normalized(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      TopK it={it} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.1f}")

    return best_a, best_score
