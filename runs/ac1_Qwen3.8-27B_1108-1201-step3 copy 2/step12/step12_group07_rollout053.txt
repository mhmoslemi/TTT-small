- Payoff is (p*p)_k

= max_{q} min_{p} sum_k q_k * sum_i p_i * p_{k-i}

= max_{q} min_{p} p^T (sum_k q_k * Q_k) p

```python
def adam_lse(a0, time_limit, T_start=50.0, T_end=0.01, n_epochs=50, lr=0.01):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_step = 0
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()
    
    # Annealing schedule
    temps = np.geomspace(T_start, T_end, n_epochs)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        iters_per_stage = max(50, 2000 // n_epochs)
        
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            t_step += 1
            a = softmax_from_z(z)
            b = fast_convolve(a)
            
            # Compute LSE weights
            T_shifted = stage_T * b - stage_T * b.max()
            exp_Tb = np.exp(T_shifted)
            w = exp_Tb / exp_Tb.sum()
            
            # Compute gradient
            grad = compute_grad(a, w, n)
            
            # Project onto simplex tangent (sum of gradient components should be 0 for z-space)
            S = np.dot(grad, a)
            grad_z = a * (grad - S)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z**2
            m_hat = m / (1 - beta1**t_step)
            v_hat = v / (1 - beta2**t_step)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Evaluate
            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    
    return best_a, best_score
```

```python
    n = len(a)
    b = np.ones(n) / n  # Start from uniform

        ab = fast_convolve(a, b)  # conv(a, b)
        # LSE weights
        Tb = T * ab - T * ab.max()
        w = np.exp(Tb) / np.exp(Tb).sum()
        # Gradient: grad_j = sum_k w_k * a_{k-j} (with a out of range = 0)
        # This is the correlation of w and a, but...
        # grad_j = sum_{k: 0<=k-j<n} w_k * a_{k-j} = sum_{i: 0<=i<n} w_{j+i} * a_i
        # = (w cross-correlated with reversed a) ... 
        
        # Actually: grad_j = sum_k w_k * a_{k-j}
        # Let i = k-j: grad_j = sum_i w_{i+j} * a_i (where w is defined on 0..2n-2)
        # This is similar to the LSE gradient I computed before!

        for i in range(n):
            grad[i] = np.dot(w[i:i+n], a)  # w[i:i+n] has length n (since i < n and w has length 2n-1)
        
        # Wait, that's the same formula as before but with roles of a and w swapped in a sense.

        # = [dot(w[0:n], a), dot(w[1:n+1], a), ..., dot(w[n-1:2n-1], a)]
        
        # Projected gradient step: b ← proj_simplex(b - lr * grad)
        lr = 1.0 / (T * a.max() * 2 * n)  # Lipschitz constant related
        b_new = b - lr * grad
        b_new = project_simplex(b_new)
        
        # Line search or fixed step
        sc = fast_convolve(a, b_new).max()
        sc_old = fast_convolve(a, b).max()
        if sc < sc_old:
            b = b_new
        else:
            break
    
    return b
```
