    n = len(y)

    q = np.ones(2*n-1) / (2*n-1)
    b = np.ones(n) / n
    
    for step in range(n_inner_steps):
        # Column player's loss: for each i, loss_i = sum_k q_k * y_{k-i}
        # Compute loss for all i
        loss = np.zeros(n)
        for i in range(n):
            # loss_i = sum_k q_k * A[k,i] = sum_k q_k * y_{k-i} (where y_j=0 if j outside [0,n-1])

            # = sum_{j=0}^{n-1} y_j * q_{j+i}
            loss[i] = np.dot(y, q[i:i+n])

        i_star = np.argmin(loss)

The problem is: minimize `max_k (a*a)_k` subject to `a >= 0, sum(a) = 1`.

This is a convex optimization problem (minimize a convex function over a convex set). The function `f(a) = max_k (a*a)_k` is convex because each `(a*a)_k = sum_i a_i a_{k-i}` is... wait, is it convex?

`(a*a)_k = sum_i a_i a_{k-i}`. This is a quadratic form in `a`. Specifically, it's `a^T Q_k a / 2` where `Q_k` is a symmetric matrix. The Hessian of `(a*a)_k` is `2 * T_k^T T_k` (where `T_k` is the Toeplitz matrix for the k-th lag), which is positive semi-definite. So `(a*a)_k` is convex in `a`.

Therefore `f(a) = max_k (a*a)_k` is convex (max of convex functions). And the constraint set `{a >= 0, sum(a) = 1}` is convex. So we have a convex optimization problem!

This means:
- Any local minimum is a global minimum
- The KKT conditions are necessary and sufficient
- The subgradient method converges (albeit slowly)

The KKT conditions at the optimum `a*`:
- There exists a subgradient `g` of `f` at `a*` and a Lagrange multiplier `λ` such that:
  `g_i + λ >= 0` for all i (from `a_i >= 0`)
  `g_i + λ = 0` for all i with `a_i > 0` (complementary slackness)
  `sum(a_i) = 1`

The subgradient `g` is a convex combination of the gradients of the active `(a*a)_k` terms:
`g = sum_{k in K*} μ_k * ∇(a*a)_k` where `μ_k >= 0, sum μ_k = 1`

And `∇(a*a)_k = 2 * shift(a, k)` (shifted by k, with zeros outside).

So at the optimum:
`sum_{k in K*} μ_k * 2 * a_{k-i} + λ = 0` for all i with `a_i > 0`
`sum_{k in K*} μ_k * 2 * a_{k-i} + λ >= 0` for all i with `a_i = 0`

This gives us: for all i with `a_i > 0`: `sum_{k in K*} μ_k * a_{k-i} = constant`

This is a very useful characterization! It means that at the optimum, the "weighted sum of shifted a's" (weighted by the dual variables μ_k) is constant on the support of a.

This is exactly what the LP best response gives us: the LP finds `b` (which is related to `μ`) such that the weighted sum is minimized. At the Nash equilibrium, `a = b` and the weighted sum is constant on the support.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE-based Nesterov subgradient (main workhorse, replaces LP)
2. Exact LP Nesterov for polishing
3. Multi-stage LSE L-BFGS annealing
4. Peak-targeted multiplicative refinement
5. Length exploration with interpolation
Key improvements over previous: 50-100x more iterations via fast LSE Nesterov,
better temperature schedule, adaptive inner parameters.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_global_deadline = 0.0

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def normalize_a(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_obj_grad_a(a, T, n):
    """Compute LSE objective and gradient w.r.t. a (not z)."""
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
        Tb = Tb - Tb.max() + max_Tb  # shift to prevent overflow
    else:
        Tb = Tb - max_Tb
    exp_Tb = np.exp(Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Gradient: g_j = (2/T) * sum_k w_k * a_{k-j}
    grad_a = np.zeros(n)
    # For efficiency, only process significant w_k
    thresh = 1e-15
    active = w > thresh
    for k in np.where(active)[0]:
        wk = w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * wk * a[k - idx]
    
    return obj, grad_a

def lse_obj_grad_z(z_val, T, n):
    """Compute LSE objective and gradient w.r.t. z (log-space)."""
    a = softmax_from_z(z_val)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    thresh = 1e-15
    active = w > thresh
    for k in np.where(active)[0]:
        wk = w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * wk * a[k - idx]
    
    # Convert to z-gradient: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def nesterov_lse(a0, time_limit, T_start=200.0, T_end=2.0, max_iters=1000000):
    """Fast Nesterov with LSE smooth approximation."""
    n = len(a0)
    a = normalize_a(np.asarray(a0, dtype=np.float64))
    z = z_from_a(a).copy()
    a_prev = a.copy()
    z_prev = z.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 1.0  # step size in z-space
    last_print = t0
    restart_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov extrapolation in z-space
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_prev)
        else:
            z_extrap = z
        
        # Compute gradient at extrapolated point
        a_extrap = softmax_from_z(z_extrap)
        _, grad_z = lse_obj_grad_z(z_extrap, T, n)
        
        # Step
        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)
        
        # Evaluate
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        # Restart logic
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            z_prev = z_new.copy()
            restart_count += 1
        else:
            z_prev = z.copy()
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        
        z = z_new
        a = a_new
        
        # Adaptive step size
        if sc_new < best_score:
            eta = min(eta * 1.01, 3.0)
        else:
            eta *= 0.95
            eta = max(eta, 0.01)
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    LSE-Nest iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} T={T:.1f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed
    
    return best_a, best_score

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):

    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov LP with adaptive restarts (for polishing)."""
    n = len(a0)
    a = normalize_a(np.asarray(a0, dtype=np.float64))
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = normalize_a(y)
        else:
            y = a
        
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = normalize_a(a_new)
        
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0005 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new
        
        a = a_new
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed
    
    return best_a, best_score

def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.5, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_obj_grad_z(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_obj_grad_z(z_val, stage_T, n)
            return grad
        
        iters = 50 if si < n_stages // 2 else 30
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = z_new
            else:
                z = z_new
        except Exception:
            pass
    
    return best_a

def peak_targeted_refine(a0, time_limit, max_steps=50000):
    n = len(a0)
    a = normalize_a(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve_step = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 10000:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.95
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)
        
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa
        
        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.005, 0.1)
        else:
            eta *= 0.9
            if eta < 1e-12:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 1:
        a = np.exp(-2.0 * (x - 0.5)**2) + 0.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(8 * np.pi * x)
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.3)**2) + 0.5 * np.exp(-5.0 * (x - 0.7)**2) + 0.1
    elif choice == 4:
        a = 1.0 + 0.4 * np.sin(3 * np.pi * x)
    else:
        a = (1.0 - x)**0.3 + 0.2
    a = np.maximum(a, 0.01)
    return normalize_a(a)
