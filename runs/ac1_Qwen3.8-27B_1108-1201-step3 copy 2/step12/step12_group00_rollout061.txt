import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def multi_block_init(n, p, gap_frac, seed=42):
    """Create a multi-block initialization."""
    # p blocks with gaps
    # Each block has length m, gap length g
    # p*m + (p-1)*g = n
    # gap_frac = g/m
    m = (n / (p + (p-1)*gap_frac))
    g = gap_frac * m
    # Adjust for rounding
    # ...
    a = np.zeros(n)
    # Fill blocks
    for i in range(p):
        start = int(round(i * (m + g)))
        end = min(start + int(round(m)), n)
        a[start:end] = 1.0
    s = a.sum()
    if s > 0:
        a = a / s
    return a

def lp_solve(a, M):
    """Solve the LP best response."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # Build constraint matrix T (2n-1 x n)
    # T[k, i] = a[k-i] if 0 <= k-i < n, else 0

    # Constraint: (convolve(a, b))[k] <= M for all k
    # This is a linear constraint on b
    T = np.zeros((2*n-1, n))
    for i in range(n):
        T[i:i+n, i] = a
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    prob.solve(solver=BEST_SOLVER, verbose=False)
    if prob.status in ('optimal', 'optimal_inaccurate') and b.value is not None:
        return np.maximum(np.asarray(b.value, dtype=np.float64), 0)
    return None

def nesterov_lp(a0, time_limit):

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    count = 0
    t0 = time.monotonic()

        if b is None:

        s_b = b.sum()
        if s_b < 1e-15:
            continue
        g = b * norm / s_b

        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

Wait, actually T @ b IS the full convolution of a and b! Because:
(T @ b)[k] = sum_i T[k,i] * b[i] = sum_i a[k-i] * b[i] (where a[j] is defined for j in [0,n-1] and 0 otherwise)
= sum_i a[k-i] * b[i]
= (a * b)[k] (full convolution)

So the constraint is conv(a, b)[k] <= M for all k, which is exactly what we want.

So I can use FFT to compute T @ b in O(n log n) instead of O(n^2).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-block structured initialization (key insight: separated blocks reduce peak)
2. LP-based Nesterov acceleration (main optimizer)
3. Fast iterative LP approximation for longer sequences
4. Spectral peak shaving for final polish
5. Adaptive multi-length search (n=2000, 3000, 4000)
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

# Global state
_best_solver = None

def get_solver():
    global _best_solver
    if _best_solver is not None:
        return _best_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _best_solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _best_solver = cp.ECOS
    print(f"  Fallback solver: ECOS")
    return _best_solver


def fast_full_conv(a, b=None):
    """Full linear convolution using FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Compute the objective value locally (fast, no external call)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_full_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def multi_block_init(n, p, gap_ratio, ramp_frac=0.1):
    """
    Create a multi-block initialization.
    p: number of blocks
    gap_ratio: gap_width / block_width
    ramp_frac: fraction of block width used for ramp (smooth edges)
    """
    # Total: p * m + (p-1) * g = n, where g = gap_ratio * m
    # p*m + (p-1)*gap_ratio*m = n
    # m * (p + (p-1)*gap_ratio) = n
    m = n / (p + (p - 1) * gap_ratio)
    g = gap_ratio * m
    
    a = np.zeros(n)
    for i in range(p):
        start = int(round(i * (m + g)))
        end = min(int(round(start + m)), n)
        if start >= n:
            break
        # Add a ramp at the edges for smoothness
        block_len = end - start
        if block_len <= 0:
            continue
        a[start:end] = 1.0
        # Ramp down at edges
        ramp = max(1, int(ramp_frac * block_len))
        for j in range(ramp):
            frac = (j + 1) / (ramp + 1)
            if j < block_len:
                a[start + j] = min(a[start + j], frac)
            if block_len - 1 - j >= 0:
                a[end - 1 - j] = min(a[end - 1 - j], frac)
    
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def sin_weighted_init(n, freq=1.0, seed=42):
    """Initialize with a sine-weighted pattern."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
    # Add a second harmonic
    base += 0.3 * np.sin(2 * np.pi * 2 * freq * x + rng.uniform(0, 2*np.pi))
    base = np.maximum(base, 0)
    s = base.sum()
    if s > 0:
        base = base / s
    return base


def lp_best_response(a, M, solver):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build T matrix: T[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # Instead of explicit matrix, we can use the convolution structure
    # But cvxpy needs explicit constraints, so let's build T
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i + n, i] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
