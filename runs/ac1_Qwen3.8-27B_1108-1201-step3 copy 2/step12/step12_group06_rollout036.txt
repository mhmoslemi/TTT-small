**But wait**: The standard FW uses a single step (move towards one vertex). We can use:
- **Smoothed FW**: Use a smoothed objective (LSE) and the smooth gradient
- **Away-step FW**: Also allow moving away from the current iterate
- **Pairwise FW**: Move from one vertex to another (increase one coordinate, decrease another)

**Pairwise Frank-Wolfe** is particularly interesting:
- At each step, find a pair (i, j) such that moving mass from i to j reduces the objective
- The subgradient tells us: g_i > g_j means coordinate i contributes more to the peak than j
- So we want to move mass from high-g coordinates to low-g coordinates
- The optimal step size can be found by line search

This is essentially a "bottleneck" approach: identify the most harmful coordinate and the most beneficial coordinate, and transfer mass between them.

    """Compute new convolution after moving delta from i_src to i_tgt."""
    n = len(a)
    size = 2*n - 1
    new_b = b.copy()
    # Add 2*delta*(a[k-i_tgt] - a[k-i_src]) for all k
    for k in range(size):
        val = 0.0
        if 0 <= k - i_tgt < n:
            val += a[k - i_tgt]
        if 0 <= k - i_src < n:
            val -= a[k - i_src]

    if 2*i_src < size:

    isum = i_src + i_tgt
    if isum < size:
        new_b[isum] -= 2*delta*delta
    return new_b

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Pairwise coordinate optimization with O(n) line search (new method)
2. Nesterov-accelerated LP best-response (Nash equilibrium search)
3. Adaptive LogSumExp L-BFGS with progress-based temperature scheduling
4. Multi-length exploration and final polishing

The key new contribution is the pairwise method which uses the structure
of the convolution to achieve O(n) per line search evaluation, enabling
many more iterations than FFT-based approaches.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def compute_subgradient(a, peaks):
    """Compute the subgradient of max at given peak positions."""
    n = len(a)
    size = 2 * n - 1
    g = np.zeros(n)
    for k in peaks:
        for i in range(n):
            if 0 <= k - i < n:
                g[i] += a[k - i]
            if 0 <= i - k < n:
                g[i] += a[i - k]
    g /= len(peaks)
    return g


def compute_subgradient_fast(a, k):
    """Compute subgradient at a single peak position k (vectorized)."""
    n = len(a)
    size = 2 * n - 1
    g = np.zeros(n)
    # a[k-i] for valid i: k-i in [0, n-1] means i in [k-n+1, k]
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] += a[k - idx]
    # a[i-k] for valid i: i-k in [0, n-1] means i in [k, k+n-1], intersect [0, n-1]
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] += a[idx2 - k]
    return g

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find top peaks

        g /= len(peaks)

        # Find source and target
        i_src = int(np.argmax(g))

        lo1 = max(0, i_tgt - n + 1)
        hi1 = min(size - 1, i_tgt)  # k range where k-i_tgt is valid

        if 2 * i_src < size:
            quad[2 * i_src] -= 1.0
        isum = i_src + i_tgt
        if isum < size:
            quad[isum] -= 2.0

        # Use ternary search or bisection on the derivative
        lo = 0.0
        hi = a[i_src] * 0.9999

        # First check if moving helps at all
        mid_check = (lo + hi) / 2
        b_check = b + mid_check * linear + mid_check * mid_check * quad

                    a_new = a_new / a_new.sum()
                    sc = score_normalized(a_new)

                        best_a = a_new.copy()
                        stagnation_count = 0
                    a = a_new

        best_val = max_b
        for _ in range(20):

            b_mid = b + mid * linear + mid * mid * quad
            val_mid = b_mid.max()
            if val_mid < best_val:

                best_delta = mid
                lo = mid
            else:
                hi = mid

            a[i_tgt] += best_delta
            # Re-normalize (should already sum to 1)
            s_new = a.sum()
            if abs(s_new - 1.0) > 1e-10:

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        if stagnation_count > stagnation_limit:

            pert = rng.normal(0, 0.05, n)
            a_new = a + pert * a

            sc_new = score_normalized(a_new)
            if sc_new < best_score * 1.01:  # Accept if not too bad
                a = a_new
                stagnation_count = 0

    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Pairwise coordinate optimization with O(n) incremental line search (new)
2. Nesterov-accelerated LP best-response (Nash equilibrium search)
3. Adaptive LogSumExp L-BFGS with progress-based temperature scheduling
4. Multi-length exploration and final polishing

Key innovation: The pairwise method maintains the convolution incrementally,
achieving O(n) per iteration instead of O(n log n), enabling millions of
updates. Combined with exact line search and stagnation-escape mechanisms,
this provides a powerful local optimizer complementary to the LP-based methods.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def compute_subgradient_at_peak(a, k):
    """Compute subgradient of (a*a)[k] w.r.t. a, vectorized."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] += a[idx2 - k]
    return g


def pairwise_optimize(a0, time_limit, max_iters=1000000, stagnation_limit=2000):
    """
    Pairwise coordinate optimization with incremental convolution update.
    Moves mass from the coordinate contributing most to the peak to the one
    contributing least, using exact line search.
    """
    n = len(a0)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    b = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    last_print = t0
    stagnation_count = 0
    refresh_counter = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - last_print > 15:
            print(f"    Pairwise iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} stagnation={stagnation_count}")
            last_print = time.monotonic()

        # Periodic refresh to avoid numerical drift
        refresh_counter += 1
        if refresh_counter > 2000:
            b = fast_convolve(a)
            refresh_counter = 0

        max_b = b.max()
        if max_b < 1e-15:
            break

        # Find top peaks
        threshold = max_b * 0.9995
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])
        if len(peaks) > 30:
            top_idx = np.argsort(b[peaks])[-30:]
            peaks = peaks[top_idx]

        # Compute averaged subgradient over peaks
        g = np.zeros(n)
        for k in peaks:
            g += compute_subgradient_at_peak(a, int(k))
        g /= len(peaks)

        # Find source (highest contribution) and target (lowest)
        i_src = int(np.argmax(g))
        i_tgt = int(np.argmin(g))

        diff = g[i_src] - g[i_tgt]
        if diff < 1e-14:
            stagnation_count += 10
            continue

        # Precompute incremental update terms
        linear = np.zeros(size)
        linear[i_tgt:i_tgt + n] += 2.0 * a
        linear[i_src:i_src + n] -= 2.0 * a

        quad = np.zeros(size)
        quad[2 * i_tgt] += 1.0
        quad[2 * i_src] -= 1.0
        quad[i_src + i_tgt] -= 2.0

        # Line search: minimize max(b + delta*linear + delta^2*quad) for delta in [0, a[i_src]]
        # The function is convex in delta, so we can bisection on the subgradient
        lo = 0.0
        hi = a[i_src] * 0.9999

        # Check if function is decreasing at 0
        k0 = int(np.argmax(b))
        subgrad_0 = linear[k0]  # derivative at delta=0 for the active constraint
        if subgrad_0 >= 0:
            # Function is increasing at 0, no improvement
            stagnation_count += 1
            if stagnation_count > stagnation_limit:
                _try_perturbation(a, b, best_a, best_score, it)
                stagnation_count = 0
            continue

        # Bisection: find delta where subgradient crosses 0
        best_delta = 0.0
        best_val = max_b
        for _ in range(25):
            mid = (lo + hi) / 2.0
            b_mid = b + mid * linear + mid * mid * quad
            val_mid = b_mid.max()
            if val_mid < best_val:
                best_val = val_mid
                best_delta = mid
            # Check subgradient at mid for bisection direction
            k_mid = int(np.argmax(b_mid))
            sg_mid = linear[k_mid] + 2.0 * mid * quad[k_mid]
            if sg_mid < 0:
                lo = mid
            else:
                hi = mid
            if hi - lo < a[i_src] * 1e-12:
                break

        if best_delta > 1e-15:
            a[i_src] -= best_delta
            a[i_tgt] += best_delta
            b = b + best_delta * linear + best_delta * best_delta * quad

            sc = 2.0 * n * b.max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1

            # Ensure non-negativity (should be automatic but safety check)
            if a[i_src] < -1e-15:
                a[i_src] = 0.0
        else:
            stagnation_count += 1

        if stagnation_count > stagnation_limit:
            _try_perturbation(a, b, best_a, best_score, it)
            stagnation_count = 0

    return best_a, best_score


def _try_perturbation(a, b, best_a, best_score, seed_val):
    """Apply a random perturbation to escape local minima. Returns updated a, b."""
    n = len(a)
    size = 2 * n - 1
    rng = np.random.default_rng(int(seed_val) + 7919)
    scale = 0.03
    pert = rng.normal(0, scale, n)
    a_new = a + pert * a
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
        b_new = fast_convolve(a_new)
        sc_new = 2.0 * n * b_new.max()
        if sc_new < best_score * 1.02:
            # Accept (update in place via return)
            return a_new, b_new
    return a, b


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_bfgs_adaptive(a0, time_limit, T_start=80.0, T_min=0.2, n_stages=30, iters_per_stage=35):
    """Adaptive LogSumExp L-BFGS with progress-based temperature control."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            if stage > 0:
                if sc > best_score * 1.005:
                    stagnant += 1
                else:
                    stagnant = max(0, stagnant - 1)

            if stagnant >= 3:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            if (stage + 1) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE-Adaptive stage={stage+1}/{n_stages} T={T:.2f} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def natural_gradient_descent(a0, time_limit, n_steps=500000):
    """
    Natural gradient descent with Polyak step size and averaging.
    Different from the subgradient method: uses the Fisher information
    metric and a more careful step size schedule.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0
    last_print = t0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step % 5000 == 0 and time.monotonic() - last_print > 15:
            print(f"    NGD step={step+1} ({time.monotonic()-t0:.0f}s): cur={2*n*fast_convolve(a).max():.8f} best={best_score:.8f}")
            last_print = time.monotonic()

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Use top-K peaks for robustness
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 40:
            top_idx = np.argsort(b[active])[-40:]
            active = active[top_idx]

        # Compute subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            g += compute_subgradient_at_peak(a, int(k))
        g /= len(active)

        # Natural gradient in simplex: project to tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm_sq = np.dot(grad_z, grad_z)
        if grad_norm_sq < 1e-20:
            break

        # Polyak step size with damping
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / grad_norm_sq
        eta = min(eta, 0.5)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Backtracking if needed
        if sc_new > cur_score * 1.005:
            for _ in range(8):
                eta *= 0.5
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                sc_new = score_normalized(a_new)
                if sc_new < cur_score:
                    break

        z = z_new
        a = a_new

        # Ergodic average (last 3000 iterates)
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 3000:
            avg_count = 3000
            avg_a = (avg_a * 3000 + a) / 3001

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        if step % 500 == 0:
            sc_avg = score_normalized(avg_a)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies:
    1. Pairwise coordinate optimization (fast O(n) per iteration)
    2. Nesterov-accelerated LP best-response
    3. Adaptive LogSumExp L-BFGS
    4. Natural gradient descent with Polyak step
    5. Multi-length exploration
    6. Final combined polish
    """
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective deadline: {global_deadline - time.monotonic():.0f}s")

    # Get starting sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Pairwise optimization (35% time) ===
    remaining = global_deadline - time.monotonic()
    t1 = min(0.35 * (budget_s - 30), remaining * 0.55)
    print(f"\n=== Phase 1: Pairwise optimization at n={n_base} (budget={t1:.0f}s) ===")
    a_out, _ = pairwise_optimize(best_seq, t1)
    try_update(a_out, "pairwise_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Nesterov LP (30% time) ===
    remaining = global_deadline - time.monotonic()
    t2 = min(0.30 * (budget_s - 30), remaining * 0.5)
    print(f"\n=== Phase 2: Nesterov LP at n={n_base} (budget={t2:.0f}s) ===")
    a_out, _ = nesterov_lp(best_seq, t2, solver=solver)
    try_update(a_out, "nesterov_lp")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Pairwise again on improved sequence (10% time) ===
    remaining = global_deadline - time.monotonic()
    t3 = min(0.10 * (budget_s - 30), remaining * 0.3)
    if t3 > 20:
        print(f"\n=== Phase 3: Pairwise round 2 (budget={t3:.0f}s) ===")
        a_out, _ = pairwise_optimize(best_seq, t3)
        try_update(a_out, "pairwise_r2")
        print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: LSE optimization (15% time) ===
    remaining = global_deadline - time.monotonic()
    t4 = min(0.15 * (budget_s - 30), remaining * 0.4)
    if t4 > 30:
        print(f"\n=== Phase 4: LSE optimization (budget={t4:.0f}s) ===")
        a_lse1, _ = lse_bfgs_adaptive(best_seq, t4 * 0.5, T_start=60.0, T_min=0.2, n_stages=20, iters_per_stage=30)
        try_update(a_lse1, "lse_pass1")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse2, _ = lse_bfgs_adaptive(best_seq, remaining * 0.6, T_start=15.0, T_min=0.1, n_stages=15, iters_per_stage=25)
            try_update(a_lse2, "lse_pass2")

        # LSE from perturbed start
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            perturbed = best_seq + 0.03 * rng.normal(0, 1, len(best_seq))
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_lse3, _ = lse_bfgs_adaptive(perturbed, remaining * 0.5, T_start=40.0, T_min=0.2, n_stages=12, iters_per_stage=25)
            try_update(a_lse3, "lse_perturbed")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Natural gradient descent (10% time) ===
    remaining = global_deadline - time.monotonic()
    t5 = min(0.10 * (budget_s - 30), remaining * 0.4)
    if t5 > 20:
        print(f"\n=== Phase 5: Natural gradient descent (budget={t5:.0f}s) ===")
        a_ngd, _ = natural_gradient_descent(best_seq, t5)
        try_update(a_ngd, "ngd")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Multi-length exploration (5% time) ===
    remaining = global_deadline - time.monotonic()
    t6_total = min(0.05 * (budget_s - 30), remaining * 0.5)
    if t6_total > 20:
        print(f"\n=== Phase 6: Multi-length exploration (budget={t6_total:.0f}s) ===")
        lengths_to_try = [n_base - 100, n_base + 100, n_base + 300]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 20:
                break
            t6 = min(t6_total / len(lengths_to_try), remaining * 0.45)
            if t6 < 15:
                break
            print(f"  Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = pairwise_optimize(init_try, t6 * 0.5)
            try_update(a_try, f"len_{n_try}_pairwise")
            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = nesterov_lp(a_try, remaining * 0.5, solver=solver, max_iters=50000)
                try_update(a_try2, f"len_{n_try}_nesterov")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Final combined polish (5% time) ===
    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 7: Final polish (budget={remaining:.0f}s) ===")
        # Quick pairwise
        a_f1, _ = pairwise_optimize(best_seq, remaining * 0.4)
        try_update(a_f1, "final_pairwise")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Quick Nesterov
            a_f2, _ = nesterov_lp(best_seq, remaining * 0.5, solver=solver, max_iters=30000)
            try_update(a_f2, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Quick LSE from low T
            a_f3, _ = lse_bfgs_adaptive(best_seq, remaining * 0.6, T_start=8.0, T_min=0.1, n_stages=8, iters_per_stage=20)
            try_update(a_f3, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (global_deadline - (budget_s - 30)):.0f}s")
    return [float(x) for x in best_seq]
