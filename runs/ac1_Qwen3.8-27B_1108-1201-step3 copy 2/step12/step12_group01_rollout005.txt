import numpy as np
import cvxpy as cp

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

                # Quick timing test
                t0 = time.monotonic()
                for _ in range(3):
                    prob.solve(solver=solver_name)
                elapsed = (time.monotonic() - t0) / 3
                print(f"  Solver {solver_name}: {elapsed*1000:.1f}ms per solve (20-var test)")
                return solver_name
        except Exception:
            continue
    return cp.SCS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    Returns (b_star, M_value) or (None, None) on failure.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, **({'verbose': False} if solver != cp.PDLP else {}))
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try fallback solvers
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, **({'verbose': False} if alt != cp.PDLP else {}))
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000):
    """
    Nesterov-accelerated LP best-response iteration.
    Uses ergodic average (theoretically optimal) + short window average.
    Adaptive restart with random perturbation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Ergodic average (theoretically optimal for Nesterov)
    ergodic_avg = np.zeros(n)
    n_iter_count = 0
    
    # Short window circular buffer
    WINDOW = 50
    short_buf = [None] * WINDOW
    short_idx = 0
    short_sum = np.zeros(n)
    
    stall_count = 0
    score_check_interval = 20  # Check score every 20 iterations
    rng = np.random.default_rng(12345)

    iter_count = 0
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new
        iter_count += 1
        
        # Update ergodic average
        ergodic_avg += a
        n_iter_count += 1
        
        # Update short window
        if short_buf[short_idx] is not None:
            short_sum -= short_buf[short_idx]
        short_buf[short_idx] = a.copy()
        short_sum += a
        short_idx = (short_idx + 1) % WINDOW

        # Periodic score checks
        if iter_count % score_check_interval == 0:
            # Check a
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += score_check_interval

            # Check y
            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall_count = 0

            # Check ergodic average (every 100 iterations it's worth checking)
            if n_iter_count % 100 == 0 and n_iter_count >= 50:
                ergodic_check = ergodic_avg / n_iter_count
                score_e = 2.0 * n * fast_autocorr(ergodic_check).max()
                if score_e < best_score:
                    best_score = score_e
                    best_a = ergodic_check.copy()
                    stall_count = 0

            # Check short window average
            if n_iter_count >= WINDOW and n_iter_count % 50 == 0:
                short_avg = short_sum / WINDOW
                score_s = 2.0 * n * fast_autocorr(short_avg).max()
                if score_s < best_score:
                    best_score = score_s
                    best_a = short_avg.copy()
                    stall_count = 0

        # Adaptive restart with perturbation
        if stall_count > 1500 and t > 500:
            # Reset to best with random perturbation
            pert_scale = 0.02 + 0.03 * (stall_count / 3000.0)
            a = best_a.copy()
            noise = rng.normal(0, pert_scale, n)
            a = a + noise * (a + 1e-6)
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            y = a.copy()
            ergodic_avg = np.zeros(n)
            n_iter_count = 0
            short_buf = [None] * WINDOW
            short_idx = 0
            short_sum = np.zeros(n)
            stall_count = 0
            if t % 5000 == 0:
                print(f"      NLP restart at t={t}, best={best_score:.8f}")

        # Progress logging
        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.8f}")

    # Final check on ergodic average
    if n_iter_count > 0:
        ergodic_final = ergodic_avg / n_iter_count
        score_f = 2.0 * n * fast_autocorr(ergodic_final).max()
        if score_f < best_score:
            best_score = score_f
            best_a = ergodic_final.copy()

    return best_a, best_score


def compute_topk_grad(a, T, topk):
    """Compute top-k LogSumExp objective and gradient."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_topk_lse(a0, time_limit, T_start=80.0, T_end=0.3, n_stages=20, topk=10):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        base_lr = 5e-3 * min(1.0, 80.0 / n)
        lr = base_lr * (0.7 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 100.0) * 4))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 50 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def smart_peak_shaving(a0, time_limit, seed=42, n_top=10):
    """
    Smart peak shaving using 'safe move' criterion.
    For each top peak, find the source position with max contribution,
    then find destination that minimizes impact on ALL top peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:n_top]
        peak_threshold = b[top_peaks[-1]] * 1.001

        improved = False
        
        # For each top peak, try to reduce it
        for k in top_peaks[:5]:
            # Find the pair (i, k-i) with maximum contribution to this peak
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib_idx = np.argmax(contrib)
            src = idx[max_contrib_idx]
            
            # Compute "impact vector" for src: how much does each position q
            # help/hurt when we move mass from src to q?
            # Change in peak k' when moving from src to q:
            #   delta[k'] = 2 * amt * (a[k'-q] - a[k'-src])
            # We want delta[k'] <= 0 for all top peaks k'
            # i.e., a[k'-q] <= a[k'-src] for all top peaks k'
            
            # Compute the vector of a[k'-src] for all top peaks
            src_impact = np.zeros(len(top_peaks))
            for pi, pk in enumerate(top_peaks):
                src_offset = pk - src
                if 0 <= src_offset < n:
                    src_impact[pi] = a[src_offset]
                else:
                    src_impact[pi] = 0.0
            
            # Find candidates q where a[k'-q] <= a[k'-src] for all top peaks
            # Score: how much it helps the worst peak
            candidates = []
            all_q = np.arange(n)
            all_q = all_q[all_q != src]
            
            # For efficiency, precompute a[k'-q] for all top peaks
            # impact_matrix[pi, q] = a[top_peaks[pi] - q] if valid else 0
            impact_matrix = np.zeros((len(top_peaks), n))
            for pi, pk in enumerate(top_peaks):
                for q in range(n):
                    off = pk - q
                    if 0 <= off < n:
                        impact_matrix[pi, q] = a[off]
            
            # Find q where impact_matrix[:, q] <= src_impact (element-wise)
            # This ensures we don't increase any top peak
            valid_mask = np.all(impact_matrix[:, all_q] <= src_impact[:, None] + 1e-15, axis=0)
            valid_q = all_q[valid_mask]
            
            if len(valid_q) == 0:
                # Relax: allow small increase on non-max peaks
                valid_mask = np.all(impact_matrix[:, all_q] <= src_impact[:, None] + bmax * 0.01, axis=0)
                valid_q = all_q[valid_mask]
            
            if len(valid_q) == 0:
                valid_q = all_q
            
            # Among valid candidates, prefer those that reduce the worst peak most
            # and have low current value (less disruption)
            reduction = src_impact[0] - impact_matrix[0, valid_q]  # reduction on worst peak
            combined_score = reduction - 0.1 * a[valid_q]  # prefer low-value positions
            top_candidates = valid_q[np.argsort(combined_score)[::-1][:20]]
            
            if len(top_candidates) == 0:
                continue
            
            # Try multiple step sizes on the best candidate
            for dst in top_candidates[:5]:
                for frac in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                n_pert = rng.integers(2, 6)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 500:
                break

    return best_a, best_score


def fine_peak_shaving(a0, time_limit, seed=777, n_top=5):
    """Very fine peak shaving with tiny steps for final polish."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        bmax = b[k_star]
        
        # Focus on top few peaks
        top_k = np.argsort(b)[::-1][:n_top]
        
        improved = False
        for k in top_k:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            
            # Top contributing sources
            top_src = idx[np.argsort(contrib)[::-1][:10]]
            
            for src in top_src:
                # Try moving to random positions
                for _ in range(3):
                    dst = rng.integers(n)
                    if dst == src:
                        continue
                    for frac in [0.0001, 0.0003, 0.001, 0.003]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < 0:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-16:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try a slightly larger random move
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.005)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    else:
        x = np.linspace(-1, 1, n)
        a = np.where(np.abs(x) <= 1, 0.5 * (1 + np.cos(np.pi * x)), 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_init(a0, rng, scale=0.05):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization for minimum peak autocorrelation sequence.
    
    Strategy:
    - 55% time: LP Nesterov (main workhorse)
    - 15% time: Smart peak shaving
    - 10% time: LSE gradient descent
    - 15% time: Multi-length exploration
    - 5% time: Final fine polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1680.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # ============ Phase 1: LP Nesterov (55% time) ============
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.55
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget {t1:.0f}s) ===")
    if t1 > 60:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}_main")

    # ============ Phase 2: Smart Peak Shaving (15%) ============
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Smart Peak Shaving (budget {t2:.0f}s) ===")
    if t2 > 20:
        a_out, sc_out = smart_peak_shaving(best_seq, t2, seed=42, n_top=8)
        try_update(a_out, "smart_shave")

    # ============ Phase 3: LSE Gradient Descent (10%) ============
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: LSE Adam (budget {t3:.0f}s) ===")
    if t3 > 15:
        a_out, sc_out = adam_topk_lse(best_seq, t3, T_start=30.0, T_end=0.3, n_stages=15, topk=8)
        try_update(a_out, f"lse_n{n_base}")

    # ============ Phase 4: Multi-length exploration (15%) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-length exploration (budget {remaining * 0.15:.0f}s) ===")
    
    # Try shorter length
    n_short = max(800, n_base - 200)
    t4a = remaining * 0.08
    if t4a > 30:
        init_short = interpolate_sequence(best_seq, n_short)
        a_out, _ = nesterov_lp_optimize(init_short, t4a, solver)
        try_update(a_out, f"lp_n{n_short}")

    # Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 200
    t4b = remaining * 0.07
    if t4b > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t4b, solver)
        try_update(a_out, f"lp_n{n_long}")

    # ============ Phase 5: Final fine polish (5%) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Final fine polish (budget {remaining * 0.4:.0f}s) ===")
    t5 = remaining * 0.4
    if t5 > 10:
        a_out, _ = fine_peak_shaving(best_seq, t5, seed=999, n_top=5)
        try_update(a_out, "fine_polish")

    # Another round of LSE at very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t6 = min(remaining * 0.3, 60.0)
        print(f"\n=== Phase 6: Final LSE (budget {t6:.0f}s) ===")
        a_out, _ = adam_topk_lse(best_seq, t6, T_start=5.0, T_end=0.1, n_stages=8, topk=6)
        try_update(a_out, "final_lse")

    # Final peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t7 = min(remaining * 0.4, 40.0)
        a_out, _ = fine_peak_shaving(best_seq, t7, seed=31337, n_top=3)
        try_update(a_out, "final_shave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        print(f"{'='*60}")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
