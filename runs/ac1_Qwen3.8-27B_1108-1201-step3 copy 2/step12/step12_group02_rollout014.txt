    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def cross_corr_w_b(b, a):
    """
    Compute grad of conv[k] w.r.t. a for weighted peaks.
    b: weights for each lag position (length 2n-1 or subset)
    Returns gradient array of length n.
    """
    n = len(a)
    # For each peak at position m with weight w_m:
    # d(conv[m])/d(a[j]) = a[m-j] for valid j (0 <= m-j < n, 0 <= j < n)
    # So grad[j] = sum_m w_m * a[m-j] (where valid)
    # This is a cross-correlation of b with a (reversed)
    
    # Efficient computation via FFT
    b_full = np.zeros(2 * n - 1)
    b_full[:len(b)] = b  # b is already the full or partial weight array
    
    # Actually, let me just do it directly for efficiency when k is small
    # Or use FFT for large k
    
    # Using FFT:
    # grad[j] = sum_m b_full[m] * a[m-j] for valid (m-j)
    # Let a_rev[i] = a[n-1-i]
    # Then grad[j] = sum_m b_full[m] * a[m-j]
    # = (b_full * a_rev)[j + n - 1] ... hmm this is getting complicated
    
    # Let me just use the direct approach with FFT
    # conv of b_full with a: result[j] = sum_m b_full[m] * a[j-m]
    # We want: grad[j] = sum_m b_full[m] * a[m-j]
    # = sum_m b_full[m] * a[(-j)+m] ... 
    
    # Actually: let's define c[m] = b_full[m] and d[j] = a[j]
    # We want e[j] = sum_m c[m] * d[m-j]
    # This is the cross-correlation of c and d, which is conv(c, d_rev)
    # where d_rev[i] = d[len(d)-1-i]
    
    d_rev = a[::-1].copy()
    conv_len = len(b_full) + n - 1
    fft_size2 = 1 << (conv_len - 1).bit_length()
    B = rfft(b_full, fft_size2)
    D = rfft(d_rev, fft_size2)
    conv = irfft(B * D, fft_size2)
    # conv[k] = sum_i b_full[i] * d_rev[k-i] = sum_i b_full[i] * a[n-1-(k-i)]
    # = sum_i b_full[i] * a[n-1-k+i]
    # We want e[j] = sum_m b_full[m] * a[m-j]
    # Let n-1-k+i = m-j => i = m-j-n+1+k
    # This is getting confusing. Let me just do it directly.
    
    # Direct approach (O(n*k) where k is number of non-zero weights):
    grad = np.zeros(n)
    nonzero = np.where(np.abs(b_full) > 1e-15)[0]
    for m in nonzero:
        m = int(m)
        w = b_full[m]
        # d(conv[m])/d(a[j]) = a[m-j] for 0<=m-j<n and 0<=j<n
        # => j ranges from max(0, m-n+1) to min(n-1, m)
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        if j_start <= j_end:
            grad[j_start:j_end+1] += w * a[m - np.arange(j_start, j_end + 1)]
    return grad


def mirror_descent_v2(a0, time_limit, eta0=0.06, decay_rate=0.5,
                      k_min=1, k_max=20, k_ramp=500, seed=42, verbose=True):
    """
    Improved mirror descent with sharpened peak weighting.
    Uses power-law weighting instead of Gaussian to focus more on the worst peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        
        # Determine current k with ramp
        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))

        # Top-k indices
        top_idx = np.argsort(b)[::-1][:k]
        
        # Power-law weights: focus more on the worst peaks
        excess = b[top_idx] - b[top_idx][0] + 1.0  # shift so worst is 1
        # Actually, use: weight = (b[m] / b_max)^p for p > 1
        # This gives more weight to peaks closer to the max
        ratio = b[top_idx] / (b_max + 1e-30)
        weights = ratio ** 3  # cubic weighting to emphasize worst peaks
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / w_sum

        # Compute subgradient
        grad = np.zeros(n)
        for ii in range(k):
            m = int(top_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            if j_start <= j_end:
                grad[j_start:j_end+1] += w_m * a[m - np.arange(j_start, j_end + 1)]

        # Learning rate
        eta = eta0 / (1.0 + t / 400.0) ** decay_rate

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        a = a_new

        # Check score periodically
        if t < 200:
            check_interval = 2
        elif t < 2000:
            check_interval = 5
        else:
            check_interval = 15

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Stall handling
        if stall_count > 300:
            noise_scale = min(0.03, 0.005 * (stall_count / 50.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall_count = 0

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD2 t={t} ({elapsed:.0f}s): best={best_score:.6f}, k={k}")

    return best_a, best_score


def lse_objective_grad(z, T):
    """
    LogSumExp objective and gradient in z-space where a = softmax(z).
    Returns (obj, grad_z).
    """
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    log_sum_exp = Tb_max + np.log(np.sum(np.exp(Tb - Tb_max)))
    obj = log_sum_exp / T

    # Softmax weights for the peaks
    exp_Tb = np.exp(Tb - Tb_max)
    w = exp_Tb / exp_Tb.sum()

    # Gradient of LSE w.r.t. a: grad_a = 2 * cross_corr(w, a)
    # where cross_corr(w, a)[j] = sum_m w[m] * a[m-j]
    grad_a = np.zeros(n)
    nonzero = np.where(w > 1e-10)[0]
    for m in nonzero:
        m = int(m)
        wm = 2.0 * w[m]
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        if j_start <= j_end:
            grad_a[j_start:j_end+1] += wm * a[m - np.arange(j_start, j_end + 1)]

    # Convert to z-space gradient
    # a = softmax(z), so da_j/dz_k = a_j*(delta_jk - a_k)
    # grad_z_k = sum_j grad_a_j * da_j/dz_k = sum_j grad_a_j * a_j*(delta_jk - a_k)
    # = a_k * (grad_a_k - sum_j grad_a_j * a_j)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lse_optimize_v2(a0, time_limit, T_start=200.0, T_end=0.1, n_stages=15, seed=42):
    """
    Multi-stage LogSumExp L-BFGS with very sharp final temperature.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # More iterations for sharper temperatures
        if stage_T > 50:
            iters = 30
        elif stage_T > 10:
            iters = 60
        else:
            iters = 100

        def func(z_in):
            val, _ = lse_objective_grad(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_objective_grad(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            a = np.maximum(a, 0)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if (si + 1) % 3 == 0 or si == n_stages - 1:
            elapsed = time.monotonic() - t0
            print(f"    LSE2 stage {si+1}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """
    Coordinate descent: for each position, optimize its value by line search.
    This is a new technique that can make fine adjustments.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    no_improve = 0

    # Pre-compute the "base" convolution (with a[j] = 0 for the position being varied)
    # Actually, it's easier to just recompute each time for n < 5000

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            old_val = a[j]
            if old_val < 1e-15:
                continue
            
            # Try different values: scale by factors
            factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]
            best_factor = 1.0
            best_local_score = best_score
            
            for f in factors:
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                # Renormalize
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                sc = fast_score(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_factor = f
            
            if best_factor != 1.0:
                a[j] = old_val * best_factor
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if verbose and no_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.6f}, no_improve={no_improve}")
        
        if no_improve > 3 * n:
            break
        if not improved and no_improve > n:
            # Try random perturbations when stuck
            for _ in range(10):
                i = int(rng.integers(n))
                f = rng.uniform(0.5, 2.0)
                a_trial = a.copy()
                a_trial[i] *= f
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    sc = fast_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
            if not improved and no_improve > 5 * n:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CD ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_v2(a0, time_limit, seed=123, verbose=True):
    """
    Improved peak shaving: move mass from high-contributing positions 
    to positions that reduce the worst peak the most.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        top_k = np.argsort(b)[::-1][:min(8, n)]

        improved = False
        for k_idx in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            # Find the position that contributes most to this peak
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            if a[src] < 1e-10:
                continue

            # Find good target positions
            # Score each position by how much reducing it would help
            # and how much adding to it would hurt
            scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                scores[idx2] += a[k2 - idx2]
            
            # We want to move to positions with low scores (not contributing to peaks)
            # and low current values
            target_score = scores + 10.0 * a  # combined penalty
            
            # Mask out the source and nearby positions
            target_score[src] = np.inf
            
            # Get candidates
            n_candidates = max(10, n // 10)
            candidates = np.argsort(target_score)[:n_candidates]
            
            for dst in candidates[:min(5, len(candidates))]:
                dst = int(dst)
                for frac in [0.001, 0.003, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            if rng.integers(5) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0

            if stall > 200:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PS2 ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.05
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.1
    elif variant == 10:
        # Strong U-shape
        x = np.linspace(0, 1, n)
        a = (x * (1 - x)) ** 0.3
    elif variant == 11:
        # Double peak at edges
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * x ** 2) + np.exp(-50 * (x - 1) ** 2) + 0.01
    else:
        a = np.ones(n)
        for _ in range(5):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.1, n * 0.4)
            amp = rng.uniform(-0.5, 1.0)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

    Phase 2: LogSumExp L-BFGS (sharper than before)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based mirror descent with cubic power-law peak weighting
2. Multi-stage LogSumExp L-BFGS with very sharp final temperature (T_end=0.05)
3. Coordinate descent polish (new technique for fine per-coordinate adjustments)
4. Improved peak shaving with contribution-based target selection
5. Multi-length exploration with diverse initializations
6. Adaptive restarts to escape local minima
Key innovations over previous: coordinate descent, sharper LSE annealing,
cubic peak weighting, more diverse structured initializations.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full linear convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def lse_objective_grad(z, T):
    """
    LogSumExp objective and gradient in z-space where a = softmax(z).
    Returns (obj, grad_z).
    """
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    log_sum_exp = Tb_max + np.log(np.sum(np.exp(Tb - Tb_max)))
    obj = log_sum_exp / T

    exp_Tb = np.exp(Tb - Tb_max)
    w = exp_Tb / exp_Tb.sum()

    # Gradient w.r.t. a: grad_a[j] = 2 * sum_m w[m] * a[m-j]
    grad_a = np.zeros(n)
    nonzero = np.where(w > 1e-12)[0]
    for m in nonzero:
        m = int(m)
        wm = 2.0 * w[m]
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        if j_start <= j_end:
            grad_a[j_start:j_end + 1] += wm * a[m - np.arange(j_start, j_end + 1)]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


class LSEObjFunc:
    def __init__(self, T):
        self.T = T
    def __call__(self, z):
        val, _ = lse_objective_grad(z, self.T)
        return val


class LSEObjGrad:
    def __init__(self, T):
        self.T = T
    def __call__(self, z):
        _, g = lse_objective_grad(z, self.T)
        return g


def mirror_descent_v2(a0, time_limit, eta0=0.06, decay_rate=0.5,

    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()

            if j_start <= j_end:

        eta = eta0 / (1.0 + t / 400.0) ** decay_rate

def lse_optimize_v2(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=15, seed=42):

    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()

def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """
    Coordinate descent: for each position, try scaling it by various factors
    and keep the best. New technique for fine adjustments.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    no_improve = 0
    sweep_count = 0

    factors_coarse = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 3.0]
    factors_fine = [0.9, 0.95, 1.05, 1.1]

    while time.monotonic() - t0 < time_limit:
        sweep_count += 1
        order = rng.permutation(n)

        # Use fine factors after first few sweeps
        if sweep_count <= 3:
            factors = factors_coarse
        else:
            factors = factors_fine

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_factor = 1.0
            best_local = best_score

            for f in factors:
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                sc = fast_score(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_factor = f

            if best_factor != 1.0:
                a[j] = old_val * best_factor
                a = a / a.sum()
                if best_local < best_score:
                    best_score = best_local
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if verbose and sweep_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep {sweep_count} ({elapsed:.0f}s): best={best_score:.6f}, no_imp={no_improve}")

        if no_improve > n * 4:
            # Try random perturbations
            found = False
            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                f = rng.uniform(0.5, 2.0)
                a_trial = a.copy()
                a_trial[i] *= f
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    sc = fast_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0
                        found = True
                        break
            if not found and no_improve > n * 6:
                break
        if no_improve == 0:
            continue
        if no_improve > n * 8:
            break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CD done ({elapsed:.0f}s): best={best_score:.6f}, sweeps={sweep_count}")

    return best_a, best_score

            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            if a[src] < 1e-10:
                continue

            # Find best targets: low penalty, low current value
            penalty = target_penalty + 10.0 * a
            penalty[src] = np.inf

            n_cand = max(10, n // 15)
            candidates = np.argsort(penalty)[:n_cand]

            for dst in candidates[:min(5, len(candidates))]:
                dst = int(dst)
                for frac in [0.001, 0.003, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            if rng.integers(4) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 150:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PS2 ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.05
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.1
    elif variant == 10:
        x = np.linspace(0, 1, n)
        a = (x * (1 - x)) ** 0.3
    elif variant == 11:
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * x ** 2) + np.exp(-50 * (x - 1) ** 2) + 0.01
    else:
        a = np.ones(n)
        for _ in range(5):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.1, n * 0.4)
            amp = rng.uniform(-0.5, 1.0)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization pipeline:
    Phase 1: Mirror descent + diverse initializations
    Phase 2: LogSumExp L-BFGS with sharp annealing (T: 200 -> 0.05)
    Phase 3: Coordinate descent polish (NEW)
    Phase 4: Peak shaving
    Phase 5: Cross-length transfer (interpolate longer, re-optimize)
    Phase 6: Final combined polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1770.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = fast_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    n_parent = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = diverse_init(n_parent, rng, variant=4)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    # Helper to try updating best
    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Mirror descent at parent length
    print(f"\n=== Phase 1: Mirror Descent at n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_md1 = min(remaining * 0.16, 280.0)
    if t_md1 > 30:
        a_out, _ = mirror_descent_v2(best_seq, t_md1, eta0=0.06, decay_rate=0.5,
                                      k_min=1, k_max=20, k_ramp=400, seed=1)
        do_try_update(a_out, f"md_n{n_parent}")

    # Diverse initializations
    remaining = time_budget - (time.monotonic() - t_global)
    t_md2 = min(remaining * 0.06, 80.0)
    if t_md2 > 15:
        for v in [4, 10, 11]:
            init = diverse_init(n_parent, rng, variant=v)
            a_out, _ = mirror_descent_v2(init, t_md2 / 3.0, eta0=0.08, decay_rate=0.6,
                                          k_min=1, k_max=15, k_ramp=150, seed=v * 7 + 3)
            do_try_update(a_out, f"md_v{v}")

    # Phase 2: LogSumExp L-BFGS (sharper annealing)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LogSumExp L-BFGS at n={n_parent} ===")
    t_lse1 = min(remaining * 0.20, 350.0)
    if t_lse1 > 30 and best_seq is not None:
        a_out, _ = lse_optimize_v2(best_seq, t_lse1, T_start=200.0, T_end=0.05,
                                    n_stages=15, seed=7)
        do_try_update(a_out, f"lse_main_n{n_parent}")

    # Second LSE from perturbed starting point
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse1b = min(remaining * 0.08, 100.0)
    if t_lse1b > 15 and best_seq is not None:
        perturbed = best_seq + 0.015 * rng.standard_normal(len(best_seq))
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        a_out, _ = lse_optimize_v2(perturbed, t_lse1b, T_start=50.0, T_end=0.05,
                                    n_stages=10, seed=13)
        do_try_update(a_out, "lse_perturbed")

    # Phase 3: Coordinate descent (NEW technique)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coordinate Descent at n={n_parent} ===")
    t_cd1 = min(remaining * 0.12, 150.0)
    if t_cd1 > 10 and best_seq is not None:
        a_out, _ = coordinate_descent(best_seq, t_cd1, seed=99)
        do_try_update(a_out, f"cd_n{n_parent}")

    # Phase 4: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Peak Shaving ===")
    t_ps1 = min(remaining * 0.07, 90.0)
    if t_ps1 > 10 and best_seq is not None:
        a_out, _ = peak_shaving_v2(best_seq, t_ps1, seed=55)
        do_try_update(a_out, f"ps_n{len(best_seq)}")

    # Phase 5: Try longer length
    n_long = int(round(n_parent * 1.3))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Longer length n={n_long} ===")
    t_long_total = min(remaining * 0.22, 320.0)
    if t_long_total > 40 and best_seq is not None:
        init_long = interpolate_to_length(best_seq, n_long)

        t_long_md = t_long_total * 0.35
        a_out, _ = mirror_descent_v2(init_long, t_long_md, eta0=0.03, decay_rate=0.5,
                                      k_min=1, k_max=20, k_ramp=300, seed=2)
        do_try_update(a_out, f"md_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        t_long_lse = min(remaining * 0.10, 160.0)
        if t_long_lse > 20:
            init_long2 = interpolate_to_length(best_seq, n_long)
            a_out, _ = lse_optimize_v2(init_long2, t_long_lse, T_start=100.0,
                                        T_end=0.05, n_stages=12, seed=11)
            do_try_update(a_out, f"lse_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        t_long_cd = min(remaining * 0.05, 60.0)
        if t_long_cd > 5:
            a_out, _ = coordinate_descent(best_seq, t_long_cd, seed=77)
            do_try_update(a_out, "cd_long")

    # Phase 6: Final polish on best solution found
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None and remaining > 20:
        # Coordinate descent
        t_f1 = min(remaining * 0.30, 60.0)
        if t_f1 > 5:
            a_out, _ = coordinate_descent(best_seq, t_f1, seed=444)
            do_try_update(a_out, "final_cd")

        # Very sharp LSE
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.40, 80.0)
        if t_f2 > 10:
            a_out, _ = lse_optimize_v2(best_seq, t_f2, T_start=15.0, T_end=0.03,
                                        n_stages=10, seed=33)
            do_try_update(a_out, "final_lse")

        # Peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.30, 40.0)
        if t_f3 > 5:
            a_out, _ = peak_shaving_v2(best_seq, t_f3, seed=222)
            do_try_update(a_out, "final_ps")

    # Final evaluation with official function
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
