def fast_autocorr(a):

def compute_score(a):

    s = a.sum()

def project_onto_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()


def lse_objective_and_grad(a, T):

def lse_objective_grad_z(z, T):
    """LSE objective and gradient in z-space (a = softmax(z) / sum(softmax(z)))."""
    z = z - z.max()
    a = np.exp(z)
    a = a / a.sum()
    n = len(a)

def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Iteratively flatten the power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def solve_lp_best_response(a_fixed):
    """
    Solve LP: min M s.t. conv(a_fixed, b)[k] <= M for all k, sum(b)=1, b>=0.
    Uses scipy HiGHS solver.
    Returns (b_opt, M_opt) or (None, None) on failure.
    """
    n = len(a_fixed)
    # Decision variables: x = [b_0, ..., b_{n-1}, M]
    # Objective: minimize M -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: (conv(a_fixed, b))[k] - M <= 0 for k = 0, ..., 2n-2
    # A_ub @ x <= b_ub
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, n] = -1.0  # -M column
    for j in range(n):
        A_ub[j:j + n, j] = a_fixed
    b_ub = np.zeros(2 * n - 1)
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    b_opt = np.maximum(res.x[:n], 0)
    s = b_opt.sum()
    if s < 1e-15:
        return None, None
    b_opt = b_opt / s
    M_opt = float(res.x[n])
    return b_opt, M_opt

        n_avg += 1
        if n_avg >= 10 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")
    
    return best_a, best_score


def subgradient_optimize(a0, time_limit, lr_init=0.01, n_print=1000):
    """
    Subgradient method with projection onto simplex and Polyak-style step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Lower bound on optimal value (for Polyak step size)
    # Theoretical lower bound: 2n/(2n-1)
    lower_bound = 2.0 * n / (2 * n - 1)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        current_val = b.max()
        score = 2.0 * n * current_val  # since sum(a)=1
        
        # Find subgradient: average of subgradients of active constraints
        active = np.where(b >= current_val - 1e-10)[0]
        
        # Subgradient of (conv(a,a))[k] at a is: 
        # for each i: 2 * sum_{k in active} a[k-i] (where 0 <= k-i < n)
        # = 2 * sum_{k in active} a[|k-i| ... ] -- this is just 2 * conv(a, indicator_active)[i]
        
        # Actually, let's compute it via FFT
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        
        fft_size = 1 << (2 * n - 1).bit_length()
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]
        
        # Polyak step size
        gap = score - lower_bound
        if gap < 1e-10:
            break
        lr_t = lr_init / np.sqrt(t)  #或使用 Polyak: gap / ||grad||^2
        
        # Update
        a_new = a - lr_t * grad
        a_new = project_onto_simplex(a_new)
        
        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        
        if t % n_print == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad t={t} ({elapsed:.0f}s): score={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    else:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
