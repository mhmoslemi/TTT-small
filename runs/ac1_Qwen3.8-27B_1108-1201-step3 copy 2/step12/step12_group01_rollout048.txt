**Deflation step:**
After LP Nesterov converges, we have a good `a`. The peaks are at certain positions. We want to reduce the second, third, etc. peaks without increasing the maximum.

Approach:
1. Compute `c = conv(a, a)`
2. Sort peaks: `c[k_1] >= c[k_2] >= ...`
3. For the second peak `c[k_2]`, find a direction `d` such that:
   - `d` reduces `c[k_2]`
   - `d` doesn't increase `c[k_1]` (or increases it minimally)
   - `d` is feasible (keeps `a >= 0, sum(a) = 1`)
4. Move in direction `d` by a small step

This is essentially a constrained optimization problem. We can solve it as an LP:
- Variables: `d` (the direction)
- Objective: minimize `d^T grad_{k_2}` (reduce the second peak)
- Constraints: `d^T grad_{k_i} <= 0` for i=1 (don't increase the first peak)
- `sum(d) = 0` (maintain normalization)
- `a + d >= 0` (non-negativity)

But `c[k]` is quadratic in `a`, so `grad_k` depends on `a`. For a small step, we can linearize: `c[k](a+εd) ≈ c[k](a) + ε * grad_k[a]^T d`.

In the Nesterov LP algorithm, the iterate `a_t` is a convex combination of past `b_s` values. **Another insight:** The LP gives us not just `b*` but also the dual variables (shadow prices) of the constraints. The dual variables tell us which constraints are binding and how much "pressure" they exert. We can use this information to guide a gradient step.

The dual variables give us the "optimal mixture" of peaks to focus on. Specifically, if `λ_k` is the dual value for constraint k, then the optimal `a` should satisfy:
`sum_k λ_k * (conv(a, e_k))[gradient] = constant`

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def topk_lse_adam(a0, time_limit, T_start=50.0, T_end=0.2, n_stages=30, topk=15):
    """Top-k LogSumExp with Adam optimizer."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    a_init = a_init / s if s > 1e-15 else np.ones(n) / n
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    def softmax_fn(z):
        z = z - z.max()
        e = np.exp(z)
        s = e.sum()
        return e / s if s > 1e-300 else np.ones(len(z)) / len(z)
    
    def compute_obj_grad(a, T, topk):
        c = fast_autocorr(a)
        k_use = min(topk, len(c))
        top_idx = np.argsort(c)[::-1][:k_use]
        top_vals = c[top_idx]
        Tc = T * top_vals
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
        w = exp_Tc / exp_Tc.sum()
        
        grad_a = np.zeros(n)
        for idx in range(k_use):
            K = top_idx[idx]
            wK = w[idx]
            js = max(0, K - (n - 1))
            je = min(n - 1, K)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
        return lse_val, grad_a
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        base_lr = 3e-3
        lr = base_lr * (0.85 ** si)
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        b1, b2, eps = 0.8, 0.999, 1e-8
        
        max_it = max(20, int(min(remaining, 100.0) * 4))
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break
            a = softmax_fn(z)
            obj, grad_a = compute_obj_grad(a, stage_T, topk)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m_z = b1 * m_z + (1 - b1) * grad_z
            v_z = b2 * v_z + (1 - b2) * grad_z ** 2
            m_hat = m_z / (1 - b1 ** t_adam)
            v_hat = v_z / (1 - b2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
    
    return best_a, best_score

def peak_deflation(a0, time_limit, seed=42):
    """Reduce second/third peaks without increasing the maximum."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        c = fast_autocorr(a)
        cmax = c.max()
        # Find peaks near the max
        threshold = cmax * 0.999
        peaks = np.where(c >= threshold)[0]
        
        if len(peaks) == 0:
            break
        
        # For each peak, try to reduce it by moving mass
        improved = False
        for k in peaks[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_ci = np.argmax(contrib)
            src = idx[max_ci]
            
            # Find safe destination
            # "Danger" positions: those that contribute to any peak
            danger = np.zeros(n, dtype=bool)
            for k2 in peaks[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True
            
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            
            # Try a few destinations with small steps
            candidates = safe[np.argsort(a[safe])][:10]
            for dst in candidates[:5]:
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(1, 4)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            st = a.sum()
            if st > 1e-10:
                a = a / st
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            if stall > 200:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
