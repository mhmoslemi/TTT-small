def fast_convolve(a):
    """Fast auto-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return result


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)

def softmax_from_z(z):
    """Compute softmax numerically stably."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:

def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T @ b = a_fixed * b."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, use_active_set=False, epsilon=0.05):
    """Nesterov LP with adaptive restarts and optional active set restriction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        # Optionally use active set
        active = None
        if use_active_set:
            b_y = fast_convolve(y)
            max_b_y = b_y.max()
            if max_b_y > 0:
                active = np.where(b_y >= max_b_y * (1.0 - epsilon))[0]
                if len(active) == 0:
                    active = None

        b_new, M_lp = solve_lp_best_response(y, solver, active)
        if b_new is None:
            # Fallback: use uniform
            b_new = np.ones(n) / n

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        # Adaptive restart
        if sc_new > best_score * 1.001 and t_nest > 10.0:
            t_nest = 1.0
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """LogSumExp objective and gradient for the normalized problem."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient: d/dz of obj
    # dLSE/dz = a * (dLSE/da - <dLSE/da, a>)
    # dLSE/da = (2n/T) * sum_k w_k * a_{k-i} (for each i)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def peak_targeted_refine(a0, time_limit, max_steps=50000):
    """Multiplicative weights update targeting peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Gradient: sum over active peaks of a[k-i]
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.001, 0.1)
        else:
            eta *= 0.88
            if eta < 1e-12:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):

    size = 2 * n - 1

        return np.ones(len(z)) / len(z)

    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None, active_set=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    if active_set is not None:
        constraints = [T[active_set] @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def structured_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 1.0 + 0.8 * (2.0 * x - 1.0)
    elif mode == 1:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-5.0 * (x - 0.7) ** 2) + 0.1
    elif mode == 2:
        a = 0.6 + 0.3 * np.cos(2 * np.pi * (x + 0.15)) + 0.15 * np.cos(4 * np.pi * x + 0.3)
    elif mode == 3:
        a = np.exp(-4.0 * x) + 0.4 * np.exp(-4.0 * (1 - x)) + 0.15
    elif mode == 4:
        a = (x * (1 - x)) ** 0.25 + 0.1
    elif mode == 5:
        a = 1.0 + 0.4 * np.sin(6 * np.pi * x + 0.2)
    elif mode == 6:
        a = 0.7 + 0.25 * np.cos(2 * np.pi * x) + 0.2 * np.cos(5 * np.pi * x + 0.7)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, use_active_set=False, epsilon=0.04):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        active = None
        if use_active_set:
            b_y = fast_convolve(y)
            max_b_y = b_y.max()
            if max_b_y > 0:
                active = np.where(b_y >= max_b_y * (1.0 - epsilon))[0]
                if len(active) == 0:
                    active = None

        b_new, M_lp = solve_lp_best_response(y, solver, active)
        if b_new is None:
            b_new = np.ones(n) / n

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        if sc_new > best_score * 1.0008 and t_nest > 8.0:
            t_nest = 1.0
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new
        a = a_new

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def hinge_objective_grad(z_in, target, n, weight=1.0):
    """Squared hinge loss: sum_k max(0, (a*a)_k - target)^2, scaled by weight."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    violations = np.maximum(b - target, 0)
    obj = weight * np.sum(violations ** 2)

    # Gradient: d/dz = a * (d_obj/da - <d_obj/da, a>)
    # d_obj/da_i = 2 * weight * sum_k violations_k * a_{k-i} (for k where b_k > target)
    active_mask = b > target
    if not np.any(active_mask):
        return obj, np.zeros(n)

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if not active_mask[k]:
            continue
        coeff = 2.0 * weight * violations[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += coeff * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbgfs(a0, time_limit, T_start=40.0, T_end=0.003, n_stages=35):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_lse(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_lse(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        if si < n_stages // 4:
            iters = 25
        elif si < n_stages // 2:
            iters = 35
        else:
            iters = 50

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25, 'gtol': 1e-20,
                         'maxfun': iters * 400, 'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a, best_score


def hinge_lbgfs(a0, time_limit, n_stages=15):
    """Hinge-loss based L-BFGS: gradually lower the target."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    # Start with target at current max, gradually decrease
    b_init = fast_convolve(a0)
    current_max = b_init.max()
    # We want to push below current_max, so start target slightly below
    target_start = current_max * 0.999
    target_end = current_max * 0.985

    targets = np.geomspace(target_start, target_end, n_stages)

    for si, tgt in enumerate(targets):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_hinge(z_val):
            obj, _ = hinge_objective_grad(z_val, tgt, n, weight=1.0)
            return obj

        def jac_hinge(z_val):
            _, grad = hinge_objective_grad(z_val, tgt, n, weight=1.0)
            return grad

        iters = 40 if si >= n_stages // 2 else 25
        try:
            res = minimize(
                fun=func_hinge, x0=z, jac=jac_hinge, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25, 'gtol': 1e-20,
                         'maxfun': iters * 400, 'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a, best_score


def active_set_lp_refine(a0, time_limit, solver=None, max_rounds=100):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for round_idx in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        for eps in [0.0, 0.01, 0.03]:
            if time.monotonic() - t0 > time_limit:
                break
            threshold = max_b * (1.0 - eps)
            active = np.where(b >= threshold)[0]
            if len(active) == 0 or len(active) >= len(b):
                continue

            b_new, M_lp = solve_lp_best_response(a, solver, active)
            if b_new is None:
                continue

            T = build_conv_matrix(a)
            full_conv = T @ b_new
            if full_conv.max() <= M_lp * 1.002:
                for alpha in [0.05, 0.1, 0.2]:
                    a_new = (1 - alpha) * a + alpha * b_new
                    a_new = np.maximum(a_new, 0)
                    sns = a_new.sum()
                    if sns < 1e-15:
                        a_new = b_new.copy()
                    else:
                        a_new = a_new / sns
                    sc = score_normalized(a_new)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        a = a_new
                        break

    return best_a, best_score


def peak_targeted_refine(a0, time_limit, max_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.02
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 4000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        threshold = max_b * 0.88
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.001, 0.08)
        else:
            eta *= 0.87
            if eta < 1e-12:
                break

    return best_a, best_score


def boost_search(a0, time_limit, n_trials=20, step_size=0.02):
    """Try random perturbations + quick local fix to escape local minima."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        # Random perturbation in log-space
        z = z_from_a(a)
        perturbation = rng.normal(0, step_size, size=n)
        z_pert = z + perturbation
        a_pert = softmax_from_z(z_pert)
        sc = score_normalized(a_pert)
        if sc < best_score:
            best_score = sc
            best_a = a_pert.copy()
        # Try a few quick LSE iterations from perturbed point
        if sc < best_score * 1.005:
            # Quick subgradient step
            b = fast_convolve(a_pert)
            max_b = b.max()
            threshold = max_b * 0.90
            topk = np.where(b >= threshold)[0]
            if len(topk) == 0:
                topk = np.array([int(np.argmax(b))])
            grad = np.zeros(n)
            for k in topk:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad[idx] += a_pert[k - idx]
            grad /= len(topk)
            for eta_try in [0.01, 0.005, 0.002]:
                exponent = -eta_try * grad
                exponent -= exponent.max()
                a_fixed = a_pert * np.exp(exponent)
                sf = a_fixed.sum()
                if sf < 1e-15:
                    continue
                a_fixed = a_fixed / sf
                sc_f = score_normalized(a_fixed)
                if sc_f < best_score:
                    best_score = sc_f
                    best_a = a_fixed.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        for n_try in [best_n - 50, best_n + 100]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 12:
                break
            if n_try < 200:
                continue
            t10 = min(20.0, remaining * 0.4)
            if t10 < 8:
                break
            print(f"  Trying n={n_try}")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t10, solver=solver)
            try_update(a_try, f"len_{n_try}")

    # Final check with LSE on best
    remaining = _global_deadline - time.monotonic()
    if remaining > 10 and best_seq is not None:
        a_final = lse_lbgfs(best_seq, min(remaining - 5, 30.0), T_start=5.0, T_end=0.001, n_stages=10)
        try_update(a_final, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
