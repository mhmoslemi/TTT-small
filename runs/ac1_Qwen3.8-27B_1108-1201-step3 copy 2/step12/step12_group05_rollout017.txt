import numpy as np

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute score without calling evaluate_sequence."""
    n = len(a)
    b = fast_autocorr(a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    return float(2 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

                stall += 1

            if stall > stall_threshold and it % 50 == 0:
                # Restart column player
                b = np.ones(m) / m
                eta = max(eta, 0.1)
                stall = 0

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            sc_now = eval_score(a)
            print(f"      MW t={it} ({elapsed:.0f}s): best={best_score:.6f}, cur={sc_now:.6f}, eta={eta:.4f}")

    return best_a, best_score


def frank_wolfe_optimize(a0, time_limit, use_accel=False):
    """
    Frank-Wolfe (conditional gradient) for minimizing max_k (a*a)_k over simplex.
    Each step is O(n) after FFT. Very fast.
    """
    n = len(a0)
    a = normalize(a0)
    if use_accel:
        a_prev = a.copy()
        y = a.copy()
    else:
        y = a.copy()

    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute autocorrelation of y
        autoc = fast_autocorr(y)
        M = autoc.max()
        
        # Find active peaks (within 5% of max)
        tol = 0.05 * M
        active_mask = autoc >= (M - tol)
        active_k = np.where(active_mask)[0]
        
        if len(active_k) == 0:
            active_k = np.array([np.argmax(autoc)])
        
        # Compute subgradient: weighted average of gradients from active peaks
        # For peak k: grad[j] = 2*y[k-j] for valid j
        g = np.zeros(n)
        for k in active_k:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # g[lo:hi+1] += 2 * y[k-hi : k-lo+1][::-1]
            segment = y[k - hi: k - lo + 1][::-1]
            g[lo:hi + 1] += 2.0 * segment
        g /= len(active_k)
        
        # Find vertex: j* = argmin g
        j_star = np.argmin(g)
        
        # Step size
        alpha = 2.0 / (it + 2.0)
        
        # FW step
        a_new = (1.0 - alpha) * y.copy()
        a_new[j_star] += alpha
        
        if use_accel:
            if it > 1:
                momentum = (it - 1.0) / (it + 2.0)
                a_new = a_new + momentum * (a_new - a)
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s > 1e-15:
                    a_new /= s
            a = a_new
        else:
            a = a_new
        a_prev = a
        y = a

        # Check score every 100 steps
        if it % 100 == 0:
            sc = eval_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FW t={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    m = 2 * n - 1
    T = np.zeros((m, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    a_prev = a.copy()

    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a_prev = a
        a = a_new

        score_a = eval_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = eval_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 40 == 0:
            avg_check = avg_a / n_avg
            score_avg = eval_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=(lambda j: a[j]))
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = eval_score(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = eval_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def adam_lse_polish(a0, time_limit, T_start=20.0, T_end=0.5, lr=0.003):
    """Quick Adam LSE polish for final improvement."""
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = eval_score(a_init)

    m_mom = np.zeros(n)
    v_mom = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(10, min(100, int(remaining * 50 / (n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            a = softmax_fn(z)
            b = fast_autocorr(a)
            Tb = T * 0 + stage_T * b  # avoid modifying
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            fft_size = 1 << (2 * n - 1).bit_length()
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_mom = beta1 * m_mom + (1 - beta1) * grad_z
            v_mom = beta2 * v_mom + (1 - beta2) * grad_z ** 2
            m_hat = m_mom / (1 - beta1 ** t_adam)
            v_hat = v_mom / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 20 == 0:
                a = softmax_fn(z)
                score = eval_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = eval_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Spectral shaping (fast preprocessing)
    Phase 2: MW game optimization (main workhorse, very fast per step)
    Phase 3: Frank-Wolfe (alternative fast method)
    Phase 4: LP Nesterov (powerful polishing)
    Phase 5: Try longer length with MW
    Phase 6: Peak shaving + Adam LSE final polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        x = np.linspace(0, 2 * np.pi, n_base)
        best_seq = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = eval_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 5:
        a_out = spectral_shaping(best_seq, n_iter=800, alpha=0.3)
        sc = eval_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [spectral]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 2: MW game optimization (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: MW game at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 20:
        a_out, sc_out = mw_game_optimize(best_seq, t2, eta0=0.3, eta_decay=0.9997)
        sc = eval_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [mw_game]: {best_score:.6f} (t={elapsed:.0f}s)")
        else:
            print(f"  MW game done: score={sc:.6f} (no improvement over {best_score:.6f})")

    # Phase 3: Frank-Wolfe (different optimization direction)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Frank-Wolfe at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_out, sc_out = frank_wolfe_optimize(best_seq, t3, use_accel=False)
        sc = eval_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [fw]: {best_score:.6f} (t={elapsed:.0f}s)")
        else:
            print(f"  FW done: score={sc:.6f}")

    # Phase 4: LP Nesterov (powerful but slower)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t4, solver)
        sc = eval_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [lp_nesterov]: {best_score:.6f} (t={elapsed:.0f}s)")
        else:
            print(f"  LP Nesterov done: score={sc:.6f}")

    # Phase 5: Try longer length
    n_long = int(round(n_base * 1.25))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Longer n={n_long} - MW game (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_spec = spectral_shaping(init_long, n_iter=300, alpha=0.2)
        a_out, sc_out = mw_game_optimize(a_spec, t5 * 0.7, eta0=0.3, eta_decay=0.9998)
        sc = eval_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [mw_long_n{n_long}]: {best_score:.6f} (t={elapsed:.0f}s)")
        else:
            print(f"  Long MW done: score={sc:.6f}")

    # Phase 6: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_out, sc_out = peak_shaving(best_seq, t6, seed=456)
        sc = eval_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [peak_shave]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 7: Final Adam LSE polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Adam LSE polish (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 120.0)
    if t7 > 10:
        a_out, sc_out = adam_lse_polish(best_seq, t7, T_start=10.0, T_end=0.5, lr=0.002)
        sc = eval_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [adam_lse_polish]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 8: Second MW pass with different parameters
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        print(f"\n=== Phase 8: Second MW pass (remaining={remaining:.0f}s) ===")
        t8 = min(remaining * 0.4, 200.0)
        a_out, sc_out = mw_game_optimize(best_seq, t8, eta0=0.1, eta_decay=0.9999)
        sc = eval_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [mw_pass2]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 9: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t9 = min(remaining * 0.5, 60.0)
        a_out, sc_out = peak_shaving(best_seq, t9, seed=789)
        sc = eval_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [final_peak_shave]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.maximum(best_seq, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
