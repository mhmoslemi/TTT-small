```python
def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}"""
    n = len(v)
    if n == 0:
        return v
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) + s - cssv > 0
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0)
```

```python
def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s} using binary search."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def softmax_gradient(ac, beta):
    """Compute weighted subgradient using softmax over the autocorrelation peaks."""
    n = len(ac) // 2 + 1  # Only need n values for the gradient
    # Actually we need the full ac to weight all lags
    # But the gradient w.r.t. a[i] is 2*sum_k w_k * a[k-i] (with bounds)
    # For efficiency, use the top few lags
    top_k = min(10, len(ac))
    idx = np.argsort(ac)[::-1][:top_k]
    weights = np.zeros(len(ac))
    for i in idx:
        weights[i] = ac[i] ** beta  # temperature parameter
    w = weights / weights.sum()
    return w, idx

def prox_gradient_optimize(a0, time_limit, beta=2.0, verbose=True):
    """Proximal gradient method with simplex projection."""
    n = len(a0)
    c = np.sqrt(2.0 * n)  # target sum
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * c / s0
    a = proj_simplex(a, c)
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Momentum
    v = a.copy()
    mu = 0.9
    
    # Adaptive step size
    step = c / (n * best_m + 1e-10)
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        ac = fast_autocorr(v)
        M = ac.max()
        
        # Compute gradient (subgradient)
        # For the max: k* = argmax ac, grad_i = 2*a[k*-i] (with bounds)
        # For soft-max: weighted combination
        k_star = np.argmax(ac)
        grad = np.zeros(n)
        
        # Use top-k lags with soft weighting
        top_positions = np.argsort(ac)[::-1][:min(5, len(ac))]
        top_vals = ac[top_positions]
        # Temperature-scaled weights
        w = top_vals ** beta
        w = w / w.sum()
        
        for wi, k in zip(w, top_positions):
            # grad_i += 2 * wi * v[k-i] (with valid index check)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            j = k - idx
            grad[idx] += 2.0 * wi * v[j]
        
        # Step with backtracking
        for _ in range(20):
            v_new = v - step * grad
            v_new = proj_simplex(v_new, c)
            ac_new = fast_autocorr(v_new)
            M_new = ac_new.max()
            if M_new < M - 1e-10 * step * np.abs(grad).max():
                break
            step *= 0.5
        
        # Check improvement
        if M_new < best_m:
            best_m = M_new
            best_a = v_new.copy()
        
        # Momentum update
        v_new_momentum = v_new + mu * (v_new - v)
        v = proj_simplex(v_new_momentum, c)
        
        # Periodic check
        if t % 1000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    PG iter={t} ({elapsed:.1f}s): best_M={best_m:.8f}, J={2*n*best_m/c**2:.6f}")
        
        # Adaptive step size
        step = c / (n * M + 1e-10)
    
    return best_a, best_m

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint: conv(a, b)_k <= M for all k
    # conv(a, b)[k] = sum_i a[i]*b[k-i]
    # This is (a * b)_k <= M
    # We can express this using FFT or directly
    # For cvxpy, we need to express as a matrix constraint
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # So T @ b <= M
    
    # Build T sparsely - for efficiency, only use non-zero rows
    # T has 2n-1 rows and n columns
    # T[k, j] = a[k-j] if 0 <= k-j < n
    
    # For large n, building the full matrix is expensive
    # Use the structure: each row k has min(k+1, n, 2n-1-k) non-zeros
    
    # Alternative: use the fact that T @ b = conv(a, b) and use FFT
    # But cvxpy needs explicit constraints
    
    # For n=1600, T is 3199 x 1600. Let's build it.
    T = np.zeros((2*n-1, n))
    for j in range(n):
        # T[k, j] = a[k-j] for valid k
        # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        T[j:j+n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        # Check both y and x
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2 * n * best_m / c**2
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=False):
    """Direct coordinate descent: for each coordinate, find optimal value."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    # Precompute the autocorrelation
    ac = fast_autocorr(a)
    M = ac.max()
    
    # For coordinate i, the contribution to ac[k] is:
    # ac[k] includes a[i] * a[k-i] (if valid)
    # Changing a[i] by delta changes ac[k] by delta * a[k-i] for each valid k
    
    n_rounds = 0
    while time.monotonic() - t0 < time_limit:
        n_rounds += 1
        improved = False
        
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Current value
            ai = a[i]
            if ai < 1e-15:
                continue
            
            # The contribution of a[i] to ac[k] is a[i] * a[k-i] for valid k
            # If we change a[i] to ai + delta:
            # ac[k] changes by delta * a[k-i]
            # We want to minimize max(ac)
            
            # Simple approach: try reducing a[i] and adding to a random other position
            # Or: try setting a[i] = 0 and redistributing
            
            # More efficient: compute the "sensitivity" of M to a[i]
            # d(max_ac)/da[i] ≈ sum of a[k-i] for k in argmax(ac)
            
            k_star = np.argmax(ac)
            # Sensitivity
            j = k_star - i
            if 0 <= j < n:
                sens = a[j]
            else:
                sens = 0.0
            
            if sens > 1e-10:
                # Try reducing a[i]
                frac = 0.1
                delta = ai * frac
                a[i] -= delta
                # Add to a low-sensitivity position
                # Find position with lowest contribution to the peak
                j2 = k_star - (i + 1) % n  # rough alternative
                # Better: add to position with a[k-i] = 0 for the peak
                # Simple: add to a random position
                dst = rng.integers(n)
                if dst != i:
                    a[dst] += delta
                    s = a.sum()
                    if s > 1e-15:
                        a = a * c / s
                        ac = fast_autocorr(a)
                        M_new = ac.max()
                        if M_new < M:
                            M = M_new
                            if M < best_m:
                                best_m = M
                                best_a = a.copy()
                                improved = True
                        else:
                            # Revert
                            a[i] += delta
                            a[dst] -= delta
                    else:
                        a[i] += delta
                        a[dst] -= delta
                else:
                    a[i] += delta
        
        if not improved and n_rounds > 50:
            break
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
```

```python
def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def compute_soft_subgrad(a, ac, top_k=5, beta=3.0):
    """Compute weighted subgradient using top-k lags with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    top_vals = np.maximum(top_vals, 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        i_start = max(0, int(k) - n + 1)
        i_end = min(n, int(k) + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = int(k) - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def prox_gradient_optimize(a0, time_limit, top_k=3, beta=2.0, verbose=True):

    n = len(a0)

    x_prev = a.copy()
    mu = 0.85
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(v)
        M = ac.max()
        
        g = compute_soft_subgrad(v, ac, top_k=top_k, beta=beta)
        
        # Step size: use L-smoothness estimate
        L = 2.0 * np.linalg.norm(g, ord=np.inf) + 1e-10
        step = 1.0 / L
        
        # Backtracking
        for _ in range(10):
            v_new = v - step * g
            v_new = proj_simplex(v_new, c)
            ac_new = fast_autocorr(v_new)
            M_new = ac_new.max()
            if M_new <= M - 0.5 * step * np.dot(g, v - v_new) + 1e-12:
                break
            step *= 0.5

            if M_new < best_m:
                best_m = M_new
                best_a = v_new.copy()
            
            # Nesterov momentum
            beta_mom = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
            x_new = v_new + (mu * (1.0 - beta_mom) / beta_mom) * (v_new - x_prev)
            x_new = proj_simplex(x_new, c)
            x_prev = v_new
            v = x_new
            M_v = fast_autocorr(v).max()

            break

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:8]

            if time.monotonic() - t0 > time_limit:

            idx = np.arange(js, je + 1)

            src = idx[np.argmax(contrib)]

            for k2 in top_k:

            cands_sorted = candidates[np.argsort(a[candidates])]
            dst = cands_sorted[rng.integers(min(len(cands_sorted), max(1, len(cands_sorted)//3)))]
            
            if dst == src:
                continue

                if s_t < 1e-15:

                a_trial = a_trial * c / s_t
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    ac = ac_trial
                    M = M_trial
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > c * 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                if a[i] >= 0:

                    if s_t > 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    # Phase 1: Mirror descent at parent length (fast, ~30s)
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror Descent n={n0} ===")
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 15:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = mirror_descent_optimize(a_norm, t1, top_k=5, beta=3.0, verbose=True)
        try_update(a_out, f"md_n{n0}")
    
    # Phase 2: Proximal gradient at parent length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: ProxGrad n={n0} ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 10:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = prox_gradient_optimize(a_norm, t2, top_k=3, beta=2.0, verbose=True)
        try_update(a_out, f"pg_n{n0}")
    
    # Phase 3: LP Nesterov at parent length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n0} ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 30:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n0}")
    
    # Phase 4: Scale to n=2400, mirror descent
    n1 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Mirror Descent n={n1} ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 15:
        a_long = interpolate_sequence(best_seq, n1)
        c1 = np.sqrt(2.0 * n1)
        a_norm = a_long * c1 / a_long.sum()
        a_out, _ = mirror_descent_optimize(a_norm, t4, top_k=5, beta=2.5, verbose=True)
        try_update(a_out, f"md_n{n1}")
    
    # Phase 5: LP at n=2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n1} ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30:
        a_long = interpolate_sequence(best_seq, n1)
        c1 = np.sqrt(2.0 * n1)
        a_norm = a_long * c1 / a_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        try_update(a_out, f"lp_n{n1}")
    
    # Phase 6: Scale to n=3200, mirror descent
    n2 = 3200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Mirror Descent n={n2} ===")
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 15:
        a_long2 = interpolate_sequence(best_seq, n2)
        c2 = np.sqrt(2.0 * n2)
        a_norm = a_long2 * c2 / a_long2.sum()
        a_out, _ = mirror_descent_optimize(a_norm, t6, top_k=4, beta=2.0, verbose=True)
        try_update(a_out, f"md_n{n2}")
    
    # Phase 7: ProxGrad at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: ProxGrad polish n={n_best} ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = prox_gradient_optimize(a_norm, t7, top_k=3, beta=1.5, verbose=True)
        try_update(a_out, f"pg_polish_n{n_best}")
    
    # Phase 8: Coordinate polish at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Coordinate polish n={n_best} ===")
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = coordinate_polish(a_norm, t8, verbose=True)
        try_update(a_out, f"coord_n{n_best}")
    
    # Phase 9: Final mirror descent with aggressive params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final MD n={n_best} ===")
    t9 = min(remaining * 0.30, 200.0)
    if t9 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = mirror_descent_optimize(a_norm, t9, top_k=3, beta=1.5, verbose=True)
        try_update(a_out, f"md_final_n{n_best}")
    
    # Phase 10: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t10 = min(remaining * 0.5, 100.0)
        if t10 > 5:
            c_best = np.sqrt(2.0 * n_best)
            a_norm = best_seq * c_best / best_seq.sum()
            a_out, _ = coordinate_polish(a_norm, t10, verbose=True)
            try_update(a_out, f"coord_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
```
