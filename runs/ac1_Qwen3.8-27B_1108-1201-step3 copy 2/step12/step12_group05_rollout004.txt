def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(arr):

def normalize(a):
    """Normalize to sum to 1, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix where T@b gives conv(a_fixed, b)."""
    n = len(a_fixed)
    m = 2 * n - 1
    T = np.zeros((m, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver, previous_solution=None):
    """
    Solve LP: min M s.t. (a_fixed * b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    objective = cp.Minimize(M_var)
    
    if previous_solution is not None:
        b_var.value = np.array(previous_solution[0])
        M_var.value = previous_solution[1]
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                b_var.value = None
                M_var.value = None
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.ECOS

def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000):
    """
    Nesterov-accelerated LP best-response iteration.
    At each step: solve LP to get best response b*, update via Nesterov.
    """
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    a_prev = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average for additional candidate
    avg_a = np.zeros(n)
    n_avg = 0
    
    prev_lp_sol = None
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver, prev_lp_sol)
        if b_star is None:
            continue
        prev_lp_sol = (b_star, M_val)
        
        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = normalize(y_new)
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize(a_new)
        
        y = y_new
        a_prev = a
        a = a_new
        
        # Track best
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Periodic average check
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")
    
    return best_a, best_score

def spectral_shaping(a, n_iter=500, alpha=0.3):
    """
    Spectral shaping: iteratively flatten the power spectrum.
    """
    n = len(a)
    a = normalize(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        # Target: geometric mean of non-DC spectral components
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        # Scale non-DC components toward flat spectrum
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize(a_new)
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)
