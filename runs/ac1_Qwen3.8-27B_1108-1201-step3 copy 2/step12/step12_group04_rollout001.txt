def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    grad_a, w = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=20):

    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    """

    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters_per_stage = max(15, min(80, int(30 + 50 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-22,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Find target positions that minimize contribution to all top peaks
                def candidate_score(j):
                    total = 0.0
                    for k2 in top_k:
                        if max(0, k2 - n + 1) <= j <= min(n - 1, k2):
                            partner = k2 - j
                            if 0 <= partner < n:
                                total += a[j] * a[partner]
                    return total + 0.005 * a[j]

                # Sample a subset of candidates for speed
                all_idx = np.arange(n)
                all_idx = all_idx[all_idx != src]
                scores = np.array([candidate_score(j) for j in all_idx[:200]])
                if len(all_idx) > 200:
                    scores = np.concatenate([scores, np.array([candidate_score(j) for j in all_idx[::10]])])
                    all_idx_sampled = np.concatenate([all_idx[:200], all_idx[::10]])
                else:
                    all_idx_sampled = all_idx

                best_dst_local = np.argsort(scores)[:max(1, len(scores) // 5)]
                dst_pool = all_idx_sampled[best_dst_local][:5]

                for dst in dst_pool:
                    for frac in [0.001, 0.003, 0.008, 0.015, 0.03, 0.05, 0.08]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            n_trials = 8
            for _ in range(n_trials):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-15:
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break

            if stall > 500:
                # Try a larger random restart perturbation
                noise = rng.normal(0, 0.01, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                score = 2.0 * n * fast_autocorr(a_pert).max()
                if score < best_score:
                    a = a_pert
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 1500:
                break

        if stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"      PS stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.5 + 0.5 * np.sin(10 * np.pi * x))
    else:
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * abs(x - 0.5)) + 0.2 * np.sin(5 * np.pi * x)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def dual_guided_refine(a0, time_limit, solver, n_lp_solves=50):
    """
    Novel: Use LP dual information to guide quadratic refinement.
    After each LP solve, use the dual weights to do gradient steps on a quadratic.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(n_lp_solves):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP to get best response and dual info
        b_star, M_val = solve_lp_best_response(a, solver)
        if b_star is None:
            continue

        # Update a using a small step toward b_star
        step = 0.1 / (1 + 0.05 * it)
        a_new = (1 - step) * a + step * b_star
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = b_star.copy()
        else:
            a_new = a_new / ss

        # Now do a few gradient steps on the log-sum-exp with high temperature
        b_conv = fast_autocorr(a_new)
        T_local = max(50.0, 200.0 / (it + 1))
        max_b = b_conv.max()
        exp_Tb = np.exp(T_local * (b_conv - max_b))
        w = exp_Tb / exp_Tb.sum()

        # Gradient of a^T Q a where Q is defined by w (circulant-like)
        a_rev = a_new[::-1]
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A_rev = rfft(a_rev, fft_size)
        conv = irfft(W * A_rev, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Gradient step with projection onto simplex
        lr = 0.01 / (it + 1)
        grad = grad - np.mean(grad)  # Remove component along 1 (sum constraint)
        a_new = a_new - lr * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # Still update to track the dual evolution

        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      DGR it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search function. Multi-phase optimization with aggressive multi-length
    exploration and dual-guided refinement.
    """
    rng = np.random.default_rng(42)
    t_global_start = time.monotonic()
    time_budget = 1740.0  # Leave 60s buffer

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default initialization")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov LP at parent length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"nlp_n{n_base}")

    # === Phase 2: Dual-guided refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Dual-guided refinement at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.1, 150.0)
    if t2 > 20:
        a_out, _ = dual_guided_refine(best_seq, t2, solver, n_lp_solves=30)
        try_update(a_out, f"dgr_n{n_base}")

    # === Phase 3: LogSumExp at parent length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LogSumExp at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.1, 150.0)
    if t3 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t3, T_start=100.0, T_end=0.01, n_stages=20)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Multi-length exploration (remaining={remaining:.0f}s) ===")

    # Try a range of lengths, prioritizing longer ones (theoretical advantage)
    candidate_lengths = []
    # Always try near n_base and significantly longer
    for offset in [-200, 0, 200, 400, 800, 1200, 1600, 2200]:
        nl = n_base + offset
        if 800 <= nl <= 3500:
            candidate_lengths.append(nl)

    # Remove duplicates, sort by distance from n_base (try closest first for warm start)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_base))

    # Allocate time: longer lengths get more time (they have better potential)
    total_weight = sum(1.0 + 0.3 * max(0, nl - n_base) / 1000.0 for nl in candidate_lengths)
    time_per = {}
    for nl in candidate_lengths:
        weight = 1.0 + 0.3 * max(0, nl - n_base) / 1000.0
        time_per[nl] = remaining * 0.75 * weight / total_weight

    for nl in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 30:
            break
        tl = min(time_per[nl], remaining * 0.3)
        if tl < 15:
            continue

        print(f"  Trying n={nl} (alloc={time_per[nl]:.0f}s, using={tl:.0f}s)")
        init_long = interpolate_sequence(best_seq, nl)

        # Nesterov LP (main workhorse)
        t_lp = tl * 0.55
        a_out, _ = nesterov_lp_optimize(init_long, t_lp, solver)

        # Quick LSE refinement
        rem2 = time_budget - (time.monotonic() - t_global_start)
        t_lse = max(5, min(tl * 0.3, rem2 * 0.1))
        if t_lse > 5:
            a_out2, _ = logsumexp_optimize(a_out, t_lse, T_start=60.0, T_end=0.03, n_stages=12)
            try_update(a_out2, f"lse_n{nl}")
        try_update(a_out, f"nlp_n{nl}")

    # === Phase 5: Refine best found ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Refinement at n={n_best} (remaining={remaining:.0f}s) ===")

    # Nesterov LP on best
    t5a = min(remaining * 0.25, 200.0)
    if t5a > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5a, solver)
        try_update(a_out, f"nlp_refine_n{n_best}")

    # LSE with very tight temperature
    remaining = time_budget - (time.monotonic() - t_global_start)
    t5b = min(remaining * 0.2, 120.0)
    if t5b > 10:
        a_out, _ = logsumexp_optimize(best_seq, t5b, T_start=30.0, T_end=0.005, n_stages=15)
        try_update(a_out, f"lse_refine_n{n_best}")

    # Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    t5c = min(remaining * 0.35, 200.0)
    if t5c > 10:
        a_out, _ = peak_shaving_advanced(best_seq, t5c, seed=12345)
        try_update(a_out, f"ps_n{n_best}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final polish (remaining={remaining:.0f}s) ===")

    # Another round of peak shaving with different seed
    t6a = min(remaining * 0.4, 100.0)
    if t6a > 5:
        a_out, _ = peak_shaving_advanced(best_seq, t6a, seed=777)
        try_update(a_out, "final_ps1")

    # Quick LSE
    remaining = time_budget - (time.monotonic() - t_global_start)
    t6b = min(remaining * 0.3, 60.0)
    if t6b > 5:
        a_out, _ = logsumexp_optimize(best_seq, t6b, T_start=10.0, T_end=0.003, n_stages=10)
        try_update(a_out, "final_lse")

    # One more peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    t6c = min(remaining * 0.3, 50.0)
    if t6c > 5:
        a_out, _ = peak_shaving_advanced(best_seq, t6c, seed=9999)
        try_update(a_out, "final_ps2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure output is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
