from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping for fast initial improvement
2. Exact cyclic coordinate descent (main workhorse) - for each coordinate,
   solves the 1D subproblem optimally using grid search + golden section
3. Multi-start with diverse initializations
4. Periodic LP best-response for escaping local minima
5. Final polish with peak shaving
Key innovation: The coordinate descent solves the EXACT 1D subproblem
(analytically derived), making each step maximally informative.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the evaluation score directly."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def coord_score(x, b0, D, pos_2i, S0, n):
    """Compute score for a single x value in the 1D subproblem."""
    denom_s = S0 + x
    if denom_s < 1e-15:
        return float('inf')
    linear_max = (b0 + x * D).max()
    quad_val = b0[pos_2i] + x * x
    max_val = max(linear_max, quad_val)
    return 2.0 * n * max_val / (denom_s * denom_s)


def solve_1d_subproblem(b0, D, pos_2i, S0, n, x_current):
    """Solve the 1D subproblem: find x >= 0 minimizing the score."""
    # Grid search
    if x_current < 1e-15:
        x_max = max(S0 / n * 5.0, 1.0)
    else:
        x_max = max(x_current * 5.0, S0 / n * 5.0, 1.0)
    
    # Create grid: log-spaced from very small to x_max
    x_min = max(x_current * 0.01, 1e-10)
    if x_current > 1e-10:
        x_grid = np.geomspace(x_min, x_max, 40)
    else:
        x_grid = np.linspace(1e-10, x_max, 40)
    
    # Include 0 and x_current
    x_grid = np.unique(np.concatenate([np.array([0.0]), x_grid, np.array([x_current])]))
    
    # Evaluate score at all grid points
    best_x = x_grid[0]
    best_sc = float('inf')
    for x in x_grid:
        sc = coord_score(x, b0, D, pos_2i, S0, n)
        if sc < best_sc:
            best_sc = sc
            best_x = x
    
    # Golden section refinement
    if best_x > 1e-10:
        span = (x_max - x_grid[0]) / 40.0
        lo_g = max(0.0, best_x - span * 2)
        hi_g = best_x + span * 2
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_g = hi_g - gr * (hi_g - lo_g)
        b_g = lo_g + gr * (hi_g - lo_g)
        fa = coord_score(a_g, b0, D, pos_2i, S0, n)
        fb = coord_score(b_g, b0, D, pos_2i, S0, n)
        for _ in range(40):
            if hi_g - lo_g < 1e-12 * max(1.0, best_x):
                break
            if fa < fb:
                hi_g = b_g
                b_g = a_g
                fb = fa
                a_g = hi_g - gr * (hi_g - lo_g)
                fa = coord_score(a_g, b0, D, pos_2i, S0, n)
            else:
                lo_g = a_g
                a_g = b_g
                fa = fb
                b_g = lo_g + gr * (hi_g - lo_g)
                fb = coord_score(b_g, b0, D, pos_2i, S0, n)
        refined_x = (lo_g + hi_g) / 2.0
        refined_sc = coord_score(refined_x, b0, D, pos_2i, S0, n)
        if refined_sc < best_sc:
            best_x = refined_x
            best_sc = refined_sc
    
    return best_x, best_sc


def coordinate_descent(a, time_limit, seed=42):
    """
    Exact cyclic coordinate descent.
    For each coordinate i, solves the 1D subproblem to find the optimal value.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    S = a.sum()
    if S < 1e-15:
        a[:] = 1.0 / n
        S = 1.0
    
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    cycle = 0
    no_improve_cycles = 0
    
    while time.monotonic() - t0 < time_limit:
        prev_score = best_score
        perm = rng.permutation(n)
        
        for i in perm:
            x_old = a[i]
            if x_old < 1e-15:
                continue
            
            # Set a[i] = 0
            a[i] = 0.0
            S0 = S - x_old
            
            # Compute b0
            b0 = fast_autocorr(a)
            
            # Compute D
            D = np.zeros(2 * n - 1)
            lo = i
            hi = min(i + n, 2 * n - 1)
            if hi > lo:
                D[lo:hi] = 2.0 * a[0:hi-lo]
            pos_2i = 2 * i
            if pos_2i < 2 * n - 1:
                D[pos_2i] = 0.0
            
            # Solve 1D subproblem
            x_opt, sc_opt = solve_1d_subproblem(b0, D, pos_2i, S0, n, x_old)
            
            a[i] = x_opt
            S = S0 + x_opt
            
            if sc_opt < best_score - 1e-12:
                best_score = sc_opt
                best_a = a.copy()
        
        cycle += 1
        if best_score < prev_score - 1e-10:
            no_improve_cycles = 0
        else:
            no_improve_cycles += 1
        
        if cycle % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CCD cycle {cycle} ({elapsed:.0f}s): best={best_score:.8f}")
        
        # Restart if stuck
        if no_improve_cycles > 5:
            # Add small random perturbation
            a = best_a + rng.uniform(0, 0.01 * best_a.max(), n)
            a = np.maximum(a, 0)
            S = a.sum()
            if S > 0:
                a = a / S
            no_improve_cycles = 0
        
        if no_improve_cycles > 15:
            break
    
    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: iteratively flatten the power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def peak_shaving(a0, time_limit, seed=123):

        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)

            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping for fast initial improvement
2. Exact cyclic coordinate descent (primary method) - for each coordinate,
   solves the 1D subproblem (minimize score as a function of that coordinate
   with all others fixed) using vectorized grid search + golden section
3. LP best-response for escaping local minima
4. Peak shaving for final polish
Key innovation: The coordinate descent exploits the structure that fixing all
coordinates except one reduces the problem to a 1D optimization of a max of
linears + quadratic over a quadratic, solved exactly per step.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the evaluation score directly."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def coord_score_val(x, b0, D, pos_2i, S0, n):
    """Compute score for a single x value in the 1D subproblem."""
    denom_s = S0 + x
    if denom_s < 1e-15:
        return float('inf')
    linear_max = (b0 + x * D).max()
    quad_val = b0[pos_2i] + x * x
    max_val = max(linear_max, quad_val)
    return 2.0 * n * max_val / (denom_s * denom_s)


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: iteratively flatten the power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            # Sort by a value ascending
            candidates_sorted = sorted(candidates, key=(lambda jj: a[jj]))
            dst_pool_size = max(1, len(candidates_sorted) // 4)
            dst = candidates_sorted[rng.integers(dst_pool_size)]

            for frac in [0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.01, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break

    return best_a, best_score


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]

def coord_descent_fast(a, time_limit, seed=42):
    """
    Optimized exact cyclic coordinate descent.
    For each coordinate i, computes the 1D subproblem and solves it
    via vectorized grid search + golden section refinement.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    S = a.sum()
    if S < 1e-15:
        a[:] = 1.0 / n
        S = 1.0

    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    cycle = 0
    no_improve_cycles = 0
    prev_cycle_best = best_score

    # Preallocate D vector
    D = np.zeros(2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        perm = rng.permutation(n)

        for i in perm:
            x_old = a[i]
            if x_old < 1e-14:
                continue

            a[i] = 0.0
            S0 = S - x_old

            b0 = fast_autocorr(a)

            # Compute D[m] = 2*a[m-i] for m in [i, i+n-1], D[2i] = 0
            D[:] = 0.0
            lo = i
            hi = min(i + n, 2 * n - 1)
            if hi > lo:
                D[lo:hi] = 2.0 * a[:hi - lo]
            pos_2i = 2 * i
            if pos_2i < 2 * n - 1:
                D[pos_2i] = 0.0

            # Solve 1D subproblem via vectorized grid search
            if x_old > 1e-10:
                x_max = max(x_old * 4.0, S0 / n * 2.0, 0.05)
                x_min = x_old * 0.01
                x_grid = np.geomspace(x_min, x_max, 25)
            else:
                x_max = max(S0 / n * 2.0, 0.05)
                x_grid = np.linspace(1e-8, x_max, 25)

            # Include current value
            x_grid = np.sort(np.unique(np.concatenate([x_grid, np.array([x_old])])))

            # Vectorized evaluation: score(x) = 2n * max(max(b0+x*D), b0[2i]+x^2) / (S0+x)^2
            linear = b0[None, :] + x_grid[:, None] * D[None, :]
            quad = b0[pos_2i] + x_grid ** 2
            max_part = np.maximum(linear, quad[:, None])
            max_val = max_part.max(axis=1)
            denom = (S0 + x_grid) ** 2
            denom = np.maximum(denom, 1e-30)
            scores = 2.0 * n * max_val / denom

            idx_best = np.argmin(scores)
            best_x = x_grid[idx_best]
            best_x_score = scores[idx_best]

            # Golden section refinement around best
            if best_x > 1e-8 and x_grid.size > 3:
                idx = idx_best
                lo_g = x_grid[max(0, idx - 1)] if idx > 0 else 0.0
                hi_g = x_grid[min(x_grid.size - 1, idx + 1)] if idx < x_grid.size - 1 else best_x * 2.0
                if hi_g <= lo_g:
                    hi_g = lo_g + max(best_x * 0.5, 1e-6)

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a_g = hi_g - gr * (hi_g - lo_g)
                b_g = lo_g + gr * (hi_g - lo_g)
                fa = coord_score_val(a_g, b0, D, pos_2i, S0, n)
                fb = coord_score_val(b_g, b0, D, pos_2i, S0, n)
                for _ in range(20):
                    if hi_g - lo_g < 1e-12 * max(1.0, best_x):
                        break
                    if fa < fb:
                        hi_g = b_g
                        b_g = a_g
                        fb = fa
                        a_g = hi_g - gr * (hi_g - lo_g)
                        fa = coord_score_val(a_g, b0, D, pos_2i, S0, n)
                    else:
                        lo_g = a_g
                        a_g = b_g
                        fa = fb
                        b_g = lo_g + gr * (hi_g - lo_g)
                        fb = coord_score_val(b_g, b0, D, pos_2i, S0, n)
                refined_x = (lo_g + hi_g) / 2.0
                refined_sc = coord_score_val(refined_x, b0, D, pos_2i, S0, n)
                if refined_sc < best_x_score:
                    best_x = refined_x
                    best_x_score = refined_sc

            a[i] = best_x
            S = S0 + best_x

            if best_x_score < best_score - 1e-13:
                best_score = best_x_score
                best_a = a.copy()

        cycle += 1
        if best_score < prev_cycle_best - 1e-10:
            no_improve_cycles = 0
            prev_cycle_best = best_score
        else:
            no_improve_cycles += 1

        if cycle % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CCD cycle {cycle} ({elapsed:.0f}s): best={best_score:.8f}")

        # Perturbation if stuck
        if no_improve_cycles > 6:
            a = best_a.copy()
            rng2 = np.random.default_rng(seed + cycle * 7)
            mask = rng2.random(n) < 0.05
            n_mask = mask.sum()
            if n_mask > 0:
                a[mask] *= (1.0 + rng2.uniform(-0.05, 0.05, n_mask))
            a = np.maximum(a, 0)
            S = a.sum()
            if S > 1e-15:
                a = a / S
                S = 1.0
            no_improve_cycles = 0

        if no_improve_cycles > 15:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses exact coordinate descent as the primary method,
    combined with spectral shaping, LP, and peak shaving for polish.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n = len(best_seq)
    print(f"Starting at n={n}, score={best_score:.6f}")

    # Phase 1: Spectral shaping (fast, ~40s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 10:
        a_out = spectral_shaping(best_seq, n_iter=1000, alpha=0.3)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [spectral]: {best_score:.6f}")

    # Phase 2: Main coordinate descent (~1200s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Coordinate descent (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.65, 1200.0)
    if t2 > 30:
        a_out, _ = coord_descent_fast(best_seq, t2, seed=100)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [ccd_main]: {best_score:.6f}")

    # Phase 3: LP best-response for escape (~120s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP steps (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 30:
        solver = get_lp_solver()
        print(f"  Using LP solver: {solver}")
        a_work = best_seq.copy()
        for step in range(15):
            if time.monotonic() - t_global_start > t_global_start + time_budget:
                break
            b_star, M_val = solve_lp_best_response(a_work, solver)
            if b_star is None:
                break
            t_mix = 0.25
            a_new = (1.0 - t_mix) * a_work + t_mix * b_star
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            sc = evaluate_safe(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_new.copy()
                a_work = a_new.copy()
                print(f"  *** NEW BEST [lp_step{step}]: {best_score:.6f}")
            a_work = a_new

    # Phase 4: More coordinate descent with different seed (~200s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: CCD round 2 (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 250.0)
    if t4 > 30:
        a_out, _ = coord_descent_fast(best_seq, t4, seed=777)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [ccd_round2]: {best_score:.6f}")

    # Phase 5: Peak shaving polish (~60s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak shaving (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 70.0)
    if t5 > 10:
        a_out, _ = peak_shaving(best_seq, t5, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [peak_shave]: {best_score:.6f}")

    # Phase 6: Final short CCD
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        print(f"\n=== Phase 6: Final CCD (remaining={remaining:.0f}s) ===")
        t6 = min(remaining * 0.85, 80.0)
        if t6 > 10:
            a_out, _ = coord_descent_fast(best_seq, t6, seed=1234)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [final_ccd]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
