import numpy as np

import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_lse_grad(a, beta):
    """
    Compute gradient of logsumexp(beta * conv(a,a)) w.r.t. a.
    Returns (grad_a, lse_value).
    """
    n = len(a)
    c = fast_autocorr(a)  # length 2n-1
    c_max = c.max()
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    w = exp_z / exp_z.sum()
    lse_val = beta * c_max + z.max() + np.log(exp_z.sum())
    
    weight = beta * c * w  # length 2n-1
    
    # grad_a[i] = 2 * sum_j weight[i+j] * a[j] for i=0,...,n-1
    # Compute via FFT: irfft(fft(weight) * conj(fft(a)))[:n]
    N = 1 << (len(weight) + n - 1 - 1).bit_length()
    W = rfft(weight, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)
    grad_a = 2.0 * corr[:n]
    
    return grad_a, lse_val


def lse_adam_optimize(a0, time_limit, beta_start=20.0, beta_end=500.0, lr_start=0.05, lr_end=0.001):
    """
    Optimize max(conv(a,a)) using LSE smoothing + Adam on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    # Adam state
    m = np.zeros(n)
    v_hat = np.zeros(n)
    t_count = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    n_iters = 0
    no_improve = 0
    max_no_improve = 3000
    
    while time.monotonic() - t0 < time_limit:
        # Beta schedule: increase over time
        progress = min(1.0, (time.monotonic() - t0) / (time_limit * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        grad_a, _ = compute_lse_grad(a, beta)
        
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad_a
        v_hat = beta2 * v_hat + (1 - beta2) * grad_a ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat_hat = v_hat / (1 - beta2 ** t_count)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat_hat) + eps)
        a_new = project_simplex(a_new, 1.0)
        
        # Check score
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        n_iters += 1
        
        if n_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-Adam iter={n_iters} ({elapsed:.0f}s): best={best_score:.6f}, beta={beta:.1f}, lr={lr:.5f}")
        
        if no_improve > max_no_improve:
            # Try a restart with perturbation
            pert = np.random.default_rng(n_iters).normal(0, 0.01, n)
            a_pert = project_simplex(a + pert, 1.0)
            a = a_pert
            no_improve = 0
    
    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def peak_shaving(a0, time_limit, seed=123):
    """Coordinate descent peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    t0 = time.monotonic()
    stall = 0
    max_stall = 800
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        n_peaks = min(15, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            contrib_sorted = np.argsort(contrib)[::-1][:8]
            for ci in contrib_sorted:
                src = idx[ci]
                src_other = k - src
                
                # Score candidate positions
                peak_positions = set()
                for k2 in top_peaks[:5]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for pos in range(js2, je2 + 1):
                        peak_positions.add(pos)
                
                # Try moving mass to positions far from peaks
                candidates = []
                for j in range(n):
                    if j == src or j == src_other:
                        continue
                    min_dist = min(abs(j - ps) for ps in peak_positions) if peak_positions else n
                    dist_to_src = abs(j - src)
                    score_j = min_dist * 0.5 + dist_to_src * 0.1 - a[j] * n * 0.05
                    candidates.append((score_j, j))
                candidates.sort(reverse=True)
                
                for _, dst in candidates[:15]:
                    max_frac = min(a[src], a[src_other] if src != src_other else a[src])
                    if max_frac < 1e-15:
                        continue
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.1]:
                        amt = max_frac * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        if src != src_other:
                            a_trial[src_other] -= amt * 0.5
                        a_trial[dst] += amt
                        if (a_trial < -1e-12).any():
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial_s = a_trial.sum()
                        if a_trial_s < 1e-15:
                            continue
                        a_trial = a_trial / a_trial_s
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 5 == 0:
                n_perturb = rng.integers(2, 8)
                sources = rng.choice(n, size=n_perturb, replace=False)
                targets = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for src_i, dst_i in zip(sources, targets):
                    frac = rng.uniform(0.005, 0.03)
                    amt = a_trial[src_i] * frac
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s < 1e-15:

                a_trial = a_trial / a_trial_s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:

                    best_score = score

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_initialization(n, shape='cosine', rng=None):
    """Create an initial sequence with a given shape."""
    x = np.linspace(0, 1, n)
    if shape == 'uniform':
        a = np.ones(n)
    elif shape == 'cosine':
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif shape == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a[-1] = 0  # avoid discontinuity
    elif shape == 'gaussian':
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif shape == 'u_shape':
        a = 1.0 / (0.1 + np.abs(x - 0.5) * 4)
    elif shape == 'w_shape':
        a = np.exp(-12 * (x - 0.25) ** 2) + np.exp(-12 * (x - 0.75) ** 2)
    elif shape == 'flat_top':
        a = np.ones(n)
        fade = max(1, n // 10)
        for i in range(fade):
            frac = (i + 1) / fade
            a[i] = frac
            a[n - 1 - i] = frac
    elif shape == 'parent':
        return None  # signal to use parent
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        a_out, sc_out = lse_adam_optimize(best_seq, t1, beta_start=20, beta_end=300, lr_start=0.03, lr_end=0.0005)
        print(f"  LSE-Adam done: score={sc_out:.6f}")
        try_update(a_out, f"lse_n{n_base}")
    
    # === Phase 2: LP Nesterov refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LP Nesterov at n={n_base} (rem={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 420.0)
    if t2 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        print(f"  LP Nesterov done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_base}")
    
    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Peak shaving at n={len(best_seq)} (rem={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 20:
        a_out, sc_out = peak_shaving(best_seq, t3, seed=777)
        print(f"  Peak shave 1: score={sc_out:.6f}")
        try_update(a_out, "peak_shave_1")
    
    # === Phase 4: Try nearby lengths ===
    for n_try in [n_base - 100, n_base + 100, n_base - 200, n_base + 200, n_base + 100, n_base - 100]:
        if n_try < 300:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 100:
            break
        print(f"\n=== Phase 4: Exploration at n={n_try} (rem={remaining:.0f}s) ===")
        t4 = min(remaining * 0.18, 180.0)
        if t4 < 40:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        
        # Quick LSE-Adam at new length
        a_out, sc_out = lse_adam_optimize(init_try, t4 * 0.5, beta_start=30, beta_end=200, lr_start=0.02, lr_end=0.001)
        print(f"  LSE at n={n_try}: score={sc_out:.6f}")
        try_update(a_out, f"lse_n{n_try}")
        
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 60:
            # Quick LP Nesterov
            t4b = min(remaining * 0.12, 100.0)
            if t4b > 30:
                a_out2, sc_out2, _ = nesterov_lp_optimize(a_out, t4b, solver, verbose=False)
                try_update(a_out2, f"lp_n{n_try}")
        
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 30:
            t4c = min(remaining * 0.08, 50.0)
            if t4c > 15:
                a_out3, sc_out3 = peak_shaving(best_seq, t4c, seed=n_try)
                try_update(a_out3, f"ps_n{n_try}")
    
    # === Phase 5: Second LSE-Adam round (exploit improvements) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: LSE-Adam round 2 at n={len(best_seq)} (rem={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 250.0)
    if t5 > 40:
        a_out, sc_out = lse_adam_optimize(best_seq, t5, beta_start=50, beta_end=500, lr_start=0.01, lr_end=0.0003)
        print(f"  LSE-Adam round 2: score={sc_out:.6f}")
        try_update(a_out, "lse_round2")
    
    # === Phase 6: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: LP Nesterov round 2 (rem={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 30:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t6, solver, verbose=False)
        print(f"  LP round 2: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "lp_round2")
    
    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final peak shave (rem={remaining:.0f}s) ===")
    if remaining > 20:
        t7 = min(remaining * 0.6, 120.0)
        if t7 > 10:
            a_out, sc_out = peak_shaving(best_seq, t7, seed=999)
            print(f"  Final peak shave: score={sc_out:.6f}")
            try_update(a_out, "final_ps")
    
    # === Phase 8: Very fine peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t8 = min(remaining * 0.7, 60.0)
        if t8 > 5:
            a_out, sc_out = peak_shaving(best_seq, t8, seed=1234)
            try_update(a_out, "fine_ps")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
