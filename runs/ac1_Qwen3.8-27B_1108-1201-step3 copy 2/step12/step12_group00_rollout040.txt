    term2 = irfft(W * A, fft_size)[:n]
    return 2.0 * (term1 + term2)

def _lse_value(a, T):
    b = _fast_autocorr(a)
    return T * np.log(np.sum(np.exp(b / T - b.max()))) + b.max()

def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint matrix: conv(a,b)[k] <= M for all k
    # conv(a,b)[k] = sum_j a[j] * b[k-j] (valid indices)
    # For efficiency, build the sparse matrix
    # T[k, j] = a[j + k - (n-1)] ... hmm, let me think
    # Actually: conv(a,b)[k] = sum_j a[j] * b[k-j] where j in [0,n-1] and k-j in [0,n-1]
    # So for a given k, the valid j range is [max(0, k-(n-1)), min(k, n-1)]
    # T[k, j] = a[j] if max(0, k-(n-1)) <= j <= min(k, n-1), else 0
    # Wait, that's not right either. Let me re-derive.
    # conv(a,b)[k] = sum_{j} a[j] * b[k-j] where the sum is over j such that 0<=j<n and 0<=k-j<n
    # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * b[k-j]
    # Let l = k-j, so j = k-l:
    # = sum_{l=max(0,k-n+1)}^{min(k,n-1)} a[k-l] * b[l]
    # So the coefficient of b[l] in conv(a,b)[k] is a[k-l] (if 0<=k-l<n)
    # T[k, l] = a[k-l] if 0 <= k-l < n, else 0
    #
    # For k in [0, 2n-2] and l in [0, n-1]:
    # T[k, l] = a[k-l] if 0 <= k-l <= n-1, else 0
    # = a[k-l] if l in [k-(n-1), k] intersect [0, n-1]
    
    # Build as sparse or dense? For n=1600, the matrix is 3199 x 1600. Dense is 5M entries, which is fine.
    # But building it densely is O(n^2). Let me use a smarter approach.
    #
    # Actually, I can compute T @ b efficiently without explicitly building T:
    # (T @ b)[k] = conv(a, b)[k] = _fast_conv(a, b)[k]
    # So I can just use the fast convolution in the constraint evaluation.
    #
    # But cvxpy needs explicit constraints. Let me build the matrix.
    # For n=1600, T is 3199 x 1600. That's about 5M floats = 40MB. Should be OK.
    
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a  # T[k, j] = a[k-j] for k in [j, j+n-1], i.e., a[k-j] where 0<=k-j<n
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def _nesterov_lse_optimize(a0, time_limit, T_start=10.0, T_min=0.1, solver=None, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = float(2 * n * _fast_autocorr(a).max())  # score with sum=1
    
    y = a.copy()
    v = np.zeros(n)  # momentum
    avg_a = np.zeros(n)
    n_avg = 0
    
    t0 = time.monotonic()
    n_iters = 0
    T = T_start
    
    # Annealing schedule: T decreases geometrically
    n_total = 100000  # expected iterations
    decay = (T_min / T_start) ** (1.0 / max(1, n_total))
    
    # Step size: start with a reasonable value
    b0 = _fast_autocorr(a)
    M0 = b0.max()
    eta = 0.5 / (M0 + 1e-10)  # initial step size
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update temperature
        T = max(T_min, T_start * (decay ** t))
        
        # Compute gradient
        grad = _lse_gradient(y, T)
        
        # Nesterov update
        v = 0.9 * v + grad  # simple momentum (or use proper Nesterov)
        # Actually, let me use proper Nesterov:
        # v_t = (t-1)/(t+2) * v_{t-1} + grad_t
        # y_{t+1} = a_t - eta * v_t
        # a_{t+1} = y_{t+1} (with projection)
        
        beta = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        v = beta * v + (1.0 - beta) * grad  # Hmm, this is not standard
        
        # Let me use the standard Nesterov scheme:
        # momentum = (t-1)/(t+2)
        # v = momentum * v_prev + grad
        # y_new = y - eta * v
        # a_new = project(y_new)
        # y = a_new
        
        # Actually, the standard Nesterov accelerated gradient for minimizing f:
        # theta_t = (1 + sqrt(1 + 4*theta_{t-1}^2)) / 2
        # y_t = a_t + (theta_{t-1} - 1)/theta_t * (a_t - a_{t-1})
        # a_{t+1} = proj(y_t - (eta/theta_t) * grad f(y_t))
        
        # Let me use a simpler momentum-based approach:
        # v = mu * v + (1-mu) * grad  (or v = mu * v + grad)
        # a_new = a - eta * v
        # a_new = clip(a_new, 0, inf)
        # a_new = a_new / sum(a_new)  (normalize to simplex)
        
        mu = min(0.9, 0.5 + 0.4 * t / 1000)  # increasing momentum
        v = mu * v + (1.0 - mu) * grad
        a_new = y - eta * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        # Check score
        score = _fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        y = a_new
        n_iters += 1
        
        # Maybe adapt step size
        if t % 100 == 0:
            b_check = _fast_autocorr(y)
            M_check = b_check.max()
            eta = 0.3 / (M_check + 1e-10)
        
        # Averaging
        avg_a += a_new
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            avg_check = avg_check / avg_check.sum()
            score_avg = _fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if verbose and n_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE iter={n_iters} ({elapsed:.0f}s): best={2*n*best_score:.6f}, T={T:.4f}")
    
    return best_a, 2.0 * n * best_score

def _mirror_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = float(2 * n * _fast_autocorr(a).max())
    
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    
    # Mirror descent with negative entropy: a_new = a * exp(-eta * grad) / Z
    # where grad is a subgradient of max(conv(a,a))
    
    eta0 = 1.0
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = _fast_autocorr(a)
        k_star = int(np.argmax(b))
        M = b[k_star]
        
        # Subgradient: grad[i] = 2 * (a[k_star - i] if valid + a[i - k_star] if valid)
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]
        grad *= 2.0
        
        eta = eta0 / (1.0 + 0.1 * t)
        log_a = np.log(a + 1e-30) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        score = _fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = avg_check / avg_check.sum()
            score_avg = _fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={t} ({elapsed:.0f}s): best={2*n*best_score:.6f}")
    
    return best_a, 2.0 * n * best_score

def _coord_descent_polish(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = float(2 * n * _fast_autocorr(a).max())
    t0 = time.monotonic()
    
    # Precompute: for each coordinate i, the coefficients c_k and d_k
    # conv(a,a)[k] = sum_j a[j]*a[k-j]
    # = a[i]*a[k-i] + a[k-i]*a[i] + sum_{j!=i} a[j]*a[k-j]
    # = 2*a[i]*a[k-i] + rest_k  (if i != k-i, i.e., k != 2i)
    # = a[i]^2 + rest_k  (if k = 2i)
    #
    # Actually: conv(a,a)[k] = sum_j a[j]*a[k-j]
    # The terms involving a[i] are: j=i → a[i]*a[k-i] (if k-i is valid)
    # and j=k-i → a[k-i]*a[i] (if k-i is valid and k-i != i)
    # So: conv(a,a)[k] = a[i]*(a[k-i] + a[k-i]) + rest (if k-i valid and k-i != i)
    #                        = a[i]*2*a[k-i] + rest_k
    # Or if k-i = i (k=2i): a[i]*a[i] + rest = a[i]^2 + rest
    #
    # For a 1D search on a[i] = x:
    # conv(x,a)[k] = c_k * x + d_k (for most k)
    # where c_k = a[k-i] + a[k-i] = 2*a[k-i] (if k-i valid and k-i≠i)
    #        or c_k = a[k-i] = a[i] (if k-i=i, i.e., k=2i) → but then it's x, not x*const
    #
    # Actually this is getting complicated because when k=2i, the term is x^2, not linear.
    # Let me just use a numerical approach: for each coordinate, try a few values and pick the best.
    
    rng = np.random.default_rng(123)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        b_current = _fast_autocorr(a)
        peak_k = int(np.argmax(b_current))
        
        # Identify coordinates that contribute most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributors and adding to non-contributors
        for _ in range(5):
            # Pick a coordinate to reduce (high contribution to peak)
            src_candidates = idx[np.argsort(contrib)[::-1][:10]]
            # Pick a coordinate to add to (low contribution to peak, low value)
            non_peak = np.setdiff1d(np.arange(n), idx)
            if len(non_peak) == 0:
                non_peak = np.arange(n)
            dst_candidates = non_peak[np.argsort(a[non_peak])[:10]]
            
            src = src_candidates[rng.integers(len(src_candidates))]
            dst = dst_candidates[rng.integers(len(dst_candidates))]
            
            if a[src] < 1e-10:
                continue
            
            # Try several transfer amounts
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                if a_trial[src] < 0:
                    continue
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 2000:
                break
    
    return best_a, 2.0 * n * best_score

def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE gradient with Nesterov momentum and adaptive temperature annealing
2. Periodic LP best-response refinement with proper Sargos averaging
3. Mirror descent (multiplicative weights) with geometric averaging
4. Spectral-guided peak shaving and coordinate descent polish
5. Multi-length exploration with interpolation transfer
Key: avoids expensive per-iteration LP in favor of fast FFT-based gradients,
uses proper Nesterov-Sargos iteration-average for convergence to saddle point.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

_global_solver = None


def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def _fast_conv(a, b):
    n_out = len(a) + len(b) - 1
    fft_size = 1 << (n_out - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n_out]


def _fast_autocorr(a):
    return _fast_conv(a, a)


def _score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = _fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def _lse_gradient(a, T):
    n = len(a)
    b = _fast_autocorr(a)
    w = np.exp(b / T - b.max())
    w = w / w.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    term1 = irfft(np.conj(W) * A, fft_size)[:n]
    term2 = irfft(W * A, fft_size)[:n]
    return 2.0 * (term1 + term2)


def _build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP-based Nesterov with proper Sargos averaging on y-iterates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()

    y = a.copy()
    a_prev = a.copy()
    avg_y = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    theta = 1.0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov: theta update
        theta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * theta * theta))
        # y_{t+1} = a_t + (theta_t - 1)/theta_{t+1} * (a_t - a_{t-1})
        # a_{t+1} = y_{t+1} - (1/theta_{t+1}) * (y_{t+1} - g_t)  [gradient step direction]
        # Simplified: a_{t+1} = (1 - 1/theta_new) * y + (1/theta_new) * g
        # Then y_{t+1} for next iteration

        # Standard Nesterov for the game:
        # y_new = a + (theta - 1)/theta_new * (a - a_prev)
        # a_new = (1 - 1/theta_new) * y_new + (1/theta_new) * g
        # a_prev = a; a = a_new

        alpha_mom = (theta - 1.0) / theta_new
        y_new = a + alpha_mom * (a - a_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a.copy()
            s_y = norm
        y_new = y_new * norm / s_y

        a_new = (1.0 - 1.0 / theta_new) * y_new + (1.0 / theta_new) * g
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
            s_a = norm
        a_new = a_new * norm / s_a

        a_prev = a
        a = a_new
        y = y_new
        theta = theta_new

        # Track best on a and y
        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Sargos averaging on a-iterates
        avg_y += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_y / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent with negative entropy (multiplicative weights) and subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * _fast_autocorr(a).max()

    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = _fast_autocorr(a)
        k_star = int(np.argmax(b))
        M = b[k_star]

        # Subgradient of max_k conv(a,a)[k] at a
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]
        grad *= 2.0

        # Mirror descent step: a_new = a * exp(-eta * grad) / Z
        eta = 1.0 / (1.0 + 0.05 * t)
        log_ratio = -eta * grad
        log_ratio -= log_ratio.max()
        a_new = a * np.exp(log_ratio)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        score = 2.0 * n * _fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = avg_check / avg_check.sum()
            score_avg = 2.0 * n * _fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def _lse_annealed(a0, time_limit, T_start=5.0, T_min=0.05, verbose=True):
    """LSE gradient descent with Nesterov momentum and temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * _fast_autocorr(a).max()

    y = a.copy()
    a_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    theta = 1.0
    t0 = time.monotonic()

    b0 = _fast_autocorr(a)
    M0 = max(b0.max(), 1e-6)
    T = T_start * M0
    decay = (T_min / T) ** (1.0 / 50000.0)

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        T_curr = max(T_min * M0, T * (decay ** t))
        grad = _lse_gradient(y, T_curr)

        theta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * theta * theta))
        eta = 1.0 / (theta_new * (T_curr + 1e-10))

        alpha_mom = (theta - 1.0) / theta_new
        y_new = y + alpha_mom * (y - a_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            s_y = 1.0
        y_new = y_new / s_y

        a_new = y_new - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
            s_a = 1.0
        a_new = a_new / s_a

        a_prev = y
        y = y_new
        a = a_new
        theta = theta_new

        score = _fast_autocorr(a).max()
        if score < best_score / (2.0 * n) * (2.0 * n):
            pass
        if 2.0 * n * score < best_score:
            best_score = 2.0 * n * score
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            avg_check = avg_check / avg_check.sum()
            score_avg = 2.0 * n * _fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE iter={t} ({elapsed:.0f}s): best={best_score:.6f}, T={T_curr:.4f}")

    return best_a, best_score


def _peak_shave(a0, time_limit, seed=42):
    """Spectral peak shaving: reduce contributions to dominant peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        M_curr = b[top_k[0]]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Find low-contribution, low-value positions
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.003, 0.01, 0.03, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * _fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.02, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                break

    return best_a, best_score


def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    else:
        a = np.ones(n)
        a[:n // 2] = 0.8
        a[n // 2:] = 1.2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    t_global = time.monotonic()
    time_budget = min(budget_s - 60.0, 1700.0)
    rng = np.random.default_rng(seed)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, np.pi, n_base)
        init = np.cos(x / 2.0) ** 2
        init = np.maximum(init, 0.0)
        best_seq = init / init.sum()
        best_score = _score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LP Nesterov at base length (primary method)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_update(a_out, f"lp_n{n_base}")

    # Phase 2: Mirror descent at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror Descent n={n_base} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 10:
        a_out, _ = _mirror_descent(best_seq, t2, verbose=True)
        do_update(a_out, f"md_n{n_base}")

    # Phase 3: LSE annealed at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE Annealed n={n_base} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 10:
        a_out, _ = _lse_annealed(best_seq, t3, T_start=3.0, T_min=0.03, verbose=True)
        do_update(a_out, f"lse_n{n_base}")

    # Phase 4: LP Nesterov at longer length
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.22, 300.0)
    if t4 > 30:
        init_long = _interpolate(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_update(a_out, f"lp_n{n2}")

    # Phase 5: LP Nesterov at even longer length
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 30:
        init_long3 = _interpolate(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_update(a_out, f"lp_n{n3}")

    # Phase 6: Peak shaving on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Peak Shave n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 180.0)
    if t6 > 5:
        a_out, _ = _peak_shave(best_seq, t6, seed=777)
        do_update(a_out, "peak_shave")

    # Phase 7: LP Nesterov polish on best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP Polish n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
        do_update(a_out, f"lp_polish_n{n_best}")

    # Phase 8: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_best = len(best_seq)
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Final Peak Shave n={n_best} ===")
            a_out, _ = _peak_shave(best_seq, t8, seed=999)
            do_update(a_out, "peak_shave_final")

    # Final validation
    if best_seq is not None:
        final_score = _score(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
