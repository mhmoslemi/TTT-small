def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]

def lse_objective_and_grad(a, T):
    """
    Compute LSE objective and gradient (in a-space) for the max autocorrelation.
    Returns (lse_value, grad_a)
    """
    n = len(a)
    size = 2 * n - 1
    N_ac = 1 << (size - 1).bit_length()
    
    # Autocorrelation
    A = rfft(a, N_ac)
    b = irfft(A * A, N_ac)[:size]
    
    # LSE value
    max_b = b.max()
    Tb = T * b
    Tb_shifted = Tb - T * max_b
    exp_Tb = np.exp(Tb_shifted)
    lse_val = max_b + np.log(exp_Tb.sum()) / T
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient: grad_a[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * cross_correlation(w, a)[i]
    # Using FFT
    N_grad = 1 << (2 * size - 1).bit_length()  # need >= size + n - 1 = 3n-2
    W = rfft(w, N_grad)
    A2 = rfft(a, N_grad)
    corr = irfft(np.conj(A2) * W, N_grad)
    grad_a = 2.0 * corr[:n]
    
    return lse_val, grad_a, b

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_grad_z(z, T):
    """LSE objective and gradient in z-space (log-softmax parameterization)."""
    a = softmax_fn(z)
    lse_val, grad_a, b = lse_objective_and_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return lse_val, grad_z

"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def lse_obj_grad_z(z, T):
    """
    Compute LSE objective (approximating max autocorrelation) and gradient in z-space.
    a = softmax(z), objective = (1/T)*log(sum_k exp(T*(a*a)_k))
    Returns (lse_value, grad_z, autocorrelation_array)
    """
    n = len(z)
    a = z - z.max()
    e = np.exp(a)
    a = e / e.sum()

    # Autocorrelation
    size = 2 * n - 1
    N_ac = 1 << (size - 1).bit_length()
    A = rfft(a, N_ac)
    b = irfft(A * A, N_ac)[:size]

    # LSE
    max_b = b.max()
    Tb = T * (b - max_b)
    exp_Tb = np.exp(Tb)
    lse_val = max_b + np.log(exp_Tb.sum()) / T
    w = exp_Tb / exp_Tb.sum()

    # Gradient in a-space: grad_a[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * cross_correlation(w, a)[i]
    # cross_correlation(w, a)[m] = sum_n w[n+m] * a[n]
    # FFT: IFFT(conj(FFT(a)) * FFT(w))
    N_grad = 1 << (size + n - 1 - 1).bit_length()
    W = rfft(w, N_grad)
    A2 = rfft(a, N_grad)
    corr = irfft(np.conj(A2) * W, N_grad)
    grad_a = 2.0 * corr[:n]

    # Gradient in z-space: grad_z = a * (grad_a - dot(a, grad_a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return lse_val, grad_z, b


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):

        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)

        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()

            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if stall_count > 500 and t > 100:

def peak_shaving(a0, time_limit, seed=123):
    """
    Adaptive multi-scale peak shaving via coordinate moves.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    else:

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    current_score = best_score
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.99
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            # Find safe destinations
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(2, len(safe_sorted) // 8)]
            if len(dst_pool) == 0:
                continue

            # Try several destinations and step sizes
            best_trial = None
            best_trial_score = best_score
            for dst in dst_pool[:5]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_trial_score:
                        best_trial_score = score
                        best_trial = a_trial.copy()

            if best_trial is not None:
                a = best_trial
                best_score = best_trial_score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Random exploration
            n_pert = rng.integers(1, 5)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 200:
                break

    return best_a, best_score


def structured_init(n, rng):

def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version for multi-start."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: Multi-start ({remaining:.0f}s left) ===")
    t3_total = min(remaining * 0.15, 200.0)
    if t3_total > 30:
        n_starts = 3
        t3_each = t3_total / n_starts
        for si in range(n_starts):
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            pert = perturbed_init(best_seq, rng, scale=0.08 * (si + 1))
            a_out, _ = lse_adam_optimize(pert, t3_each, T_start=30.0, T_end=3000.0, n_stages=20, lr_start=0.5, lr_end=0.02)
            try_update(a_out, f"multi_{si}")

    # === Phase 4: Longer length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        n_long = n_base + 200
        print(f"=== Phase 4: LSE n={n_long} ({remaining:.0f}s left) ===")
        t4 = min(remaining * 0.12, 150.0)
        if t4 > 20:
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = lse_adam_optimize(init_long, t4, T_start=30.0, T_end=3000.0, n_stages=25, lr_start=0.5, lr_end=0.02)
            try_update(a_out, f"lse_n{n_long}")

    # === Phase 5: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        print(f"=== Phase 5: Peak shaving ({remaining:.0f}s left) ===")
        t5 = min(remaining * 0.5, 100.0)
        if t5 > 10:
            a_out, _ = peak_shaving(best_seq, t5, seed=456)
            try_update(a_out, "peak_shave")

    # === Phase 6: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        print(f"=== Phase 6: Final LSE polish ({remaining:.0f}s left) ===")
        t6 = min(remaining * 0.6, 80.0)
        if t6 > 10:
            a_out, _ = lse_adam_optimize(best_seq, t6, T_start=2000.0, T_end=50000.0, n_stages=15, lr_start=0.2, lr_end=0.005)
            try_update(a_out, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
