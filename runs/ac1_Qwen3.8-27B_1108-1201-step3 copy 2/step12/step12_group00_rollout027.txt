def lp_best_response(a, M, solver, K=None):
    """Solve the LP best-response. Optionally use only top-K constraints."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if K is not None and K < 2 * n - 1:
        # Use only the top-K constraints (heuristic: where a*a is largest)
        conv_est = fast_autocorr(a)
        top_k = np.argsort(conv_est)[::-1][:K]
        # Also always include k=0 (the sum constraint area)
        top_k_set = set(top_k.tolist())
        top_k_set.add(0)
        top_k_list =(sorted(top_k_set)[:K + 1])
        
        constraints = []
        for k in top_k_list:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            row = np.zeros(n)
            row[idx] = a[k - idx]
            constraints.append(cp.sum(row * b_var) <= M)
    else:

        T = np.zeros((2 * n - 1, n))

        b_var_vec = b_var
        constraints = [T @ b_var_vec <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:15]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-30:
                continue
            src = idx[np.argmax(contrib)]

            # Find positions with low contribution to all top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 3)]
            if len(candidates_sorted) == 0:
                continue
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.01, 0.06)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    else:
        # Frequency-based: try to make spectrum flat
        k = np.arange(n)
        a = np.ones(n) + 0.3 * np.cos(2 * np.pi * 3 * k / n) + 0.1 * np.cos(2 * np.pi * 7 * k / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
