    n = len(y)

    prob.solve(solver=cp.ECOS, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        prob.solve(solver=cp.SCS, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        prob.solve(solver=cp.PDLP, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

    s = a.sum()
    if s < 1e-15:

        a = a / s

    n = len(y)

    size = 2 * n - 1

    b_val = uniform.copy()
    M_val = Tu.max()

        active_list = sorted(active)
        if len(active_list) == 0:
            active = set(np.argsort(Tu)[-20:].tolist())
            continue

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T[active_list] @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return b_val, M_val
        
        new_b = np.asarray(b_var.value, dtype=np.float64)
        new_b = np.maximum(new_b, 0)
        new_b /= new_b.sum()
        new_M = float(M_var.value)
        
        # Check for violated constraints
        full = T @ new_b
        violated = np.where(full > new_M + tol)[0]
        
        if len(violated) == 0:
            return new_b, new_M
        
        # Add most violated
        most = violated[np.argmax(full[violated])]
        active.add(int(most))
        
        # If active set is too large, remove least active
        if len(active) > max_active:
            # Remove the one with smallest full[k] - new_M
            active_list = sorted(active)
            margins = {k: full[k] - new_M for k in active_list}
            least = min(margins, key=margins.get)
            active.remove(least)
        
        b_val = new_b
        M_val = new_M
    
    return b_val, M_val
