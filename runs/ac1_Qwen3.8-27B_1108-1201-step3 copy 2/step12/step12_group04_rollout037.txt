def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_conv_mid(a, b):
    """Cross-convolution of a and b, return middle n elements (aligned with a)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    result = irfft(A * B, fft_size)[:size]
    return result[n - 1:n - 1 + n]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:

        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if sum(a_list) < 0.01:

        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def get_lp_solver():

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) gives convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        end = min(j + n, size)
        T[j:end, j] = a_fixed[:end - j]
    return T


def solve_lp_best_response(a_fixed, solver, n_active_constraints=None):
    """
    Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0.
    Returns (b, M, active_indices).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        prob = None

    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)

    # Find active constraints
    conv_vals = T @ b_val
    threshold = M_val - 1e-8
    active = np.where(conv_vals >= threshold)[0]
    if len(active) == 0:
        active = np.argmax(conv_vals)
        active = np.array([active])

    return b_val, M_val, active

def mirror_descent_step(y, a_fixed, eta):
    """One step of mirror descent for best response computation."""
    n = len(a_fixed)
    c = cross_conv_mid(a_fixed, y)
    # Exponentiated gradient update
    log_B = -eta * c
    log_B = log_B - log_B.max()
    B = np.exp(log_B)
    B = np.maximum(B, 1e-300)
    B = B / B.sum()
    return B, c.max()

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_objective_and_grad(z, T_stage):
    """Compute LSE-smoothed objective and gradient in z-space."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T_stage * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T_stage

    w = exp_Tb / exp_Tb.sum()

    # Gradient w.r.t. a: 2 * (w * a_rev)[n-1 : n-1+n]
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=25):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters_per_stage = max(15, min(80, int(20 + 60 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            val, _ = lse_objective_and_grad(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_objective_and_grad(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500,
                         'disp': False, 'maxls': 50}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, seed=42):
    """
    Subgradient descent on the exact max objective with Nesterov acceleration.
    Uses multiple subgradient selections for robustness.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Nesterov state
    y = a.copy()
    t = 0
    eta = 1.0 / math.sqrt(n)

    for iteration in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        b_max = b.max()
        # Find all near-max positions
        threshold = b_max - 1e-6
        max_idx = np.where(b >= threshold)[0]

        # Random subgradient from the max set
        if len(max_idx) > 0:
            k = max_idx[rng.integers(len(max_idx))]
        else:
            k = int(np.argmax(b))

        # Compute subgradient: 2 * (e_k * a_rev) middle part
        # (e_k convolved with a gives shift of a)
        a_rev = y[::-1].copy()
        # The subgradient at position j is 2 * y[k-j] (if valid)
        grad = np.zeros(n)
        # grad[j] = 2 * y[k - j] for valid k-j
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] = 2.0 * y[idx]
            # Also the "other" term: 2 * y[j + k - (n-1)]... 
            # Actually for self-convolution, the gradient is 2 * shift(a, k)
            # where shift means (shift(a,k))[j] = a[k-j] if valid

        # Nesterov update
        t += 1
        alpha_t = 2.0 / (t + 2.0)
        step_size = eta / math.sqrt(t)

        y_new = y - step_size * grad
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / ss

        if t == 1:
            a_new = y_new
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss

        y = y_new
        a = a_new

        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Adaptive step size
        if iteration % 500 == 0 and iteration > 0:
            current = 2.0 * n * fast_autocorr(a).max()
            if current > best_score * 1.001:
                eta *= 0.9
            elif current < best_score * 0.999:
                eta *= 1.1

        if iteration % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG it={iteration} ({elapsed:.0f}s): best={best_score:.8f}, cur={2.0*n*fast_autocorr(a).max():.8f}")

    return best_a, best_score

        name, param = strategies[rng.integers(len(strategies))]

        if name == 'power':
            P_new = P ** param + 1e-10
        else:  # linear
            P_new = P + param * (mean_P - P)
            P_new = np.maximum(P_new, 1e-10)

        # Compute new time-domain signal
        mag_new = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            continue
        a_new = a_new / ss

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Coordinate descent with smart pair selection for final polishing.
    Focuses on reducing the maximum autocorrelation peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()

        # Focus on top peaks
        top_k_count = min(5, len(b))
        top_peaks = np.argsort(b)[::-1][:top_k_count]

        improved = False
        for k in top_peaks:
            if improved:
                break
            # Find the pair (i, k-i) that contributes most to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if len(contrib) == 0:
                continue
            # Top contributor
            tc = np.argmax(contrib)
            src = idx[tc]
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Find good destination: position that minimizes impact on all top peaks
            peak_positions = set()
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_positions.update(range(js2, je2 + 1))

            # Score all candidate destinations
            candidates = []
            for j in range(n):
                if j == src:
                    continue
                impact = 0.0
                for k2 in top_peaks:
                    if max(0, k2 - n + 1) <= j <= min(n - 1, k2):
                        partner = k2 - j
                        if 0 <= partner < n:
                            impact += a[j] * a[partner] + a[src_val] * 0.01
                candidates.append((impact + 0.001 * a[j], j))

            candidates.sort(key=lambda x: x[0])
            top_dst = [j for _, j in candidates[:5]]

            for dst in top_dst:
                if improved:
                    break
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

        if not improved:
            stall += 1
            # Random exploration
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > 200:
                break

        if stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"      CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, seed=0):
    """Generate a structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.8 * np.exp(-5 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def dual_nesterov_optimize(a0, time_limit, solver):
    """
    Dual Nesterov: alternates between optimizing a and b (both sides).
    Uses the game structure more directly.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    b = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 30000):
        if time.monotonic() - t0 > time_limit:
            break

        # Best response for b given a
        b_resp, M_a, _ = solve_lp_best_response(a, solver)
        if b_resp is not None:
            alpha = 2.0 / (t + 2.0)
            b = (1 - alpha) * b + alpha * b_resp
            b = np.maximum(b, 0)
            b = b / b.sum()

        # Best response for a given b
        a_resp, M_b, _ = solve_lp_best_response(b, solver)
        if a_resp is not None:
            alpha = 2.0 / (t + 2.0)
            a = (1 - alpha) * a + alpha * a_resp
            a = np.maximum(a, 0)
            a = a / a.sum()

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_b = 2.0 * n * fast_autocorr(b).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()

        # Average
        if t >= 20:
            avg_a += a
            n_avg += 1
            if n_avg % 30 == 0:
                avg_check = avg_a / n_avg
                score_avg = 2.0 * n * fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      DNL t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search function: multi-phase optimization combining LP, gradient,
    spectral, and coordinate descent methods.
    """
    t_global_start = time.monotonic()
    time_budget = 1740.0

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default initialization")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        best_seq = structured_init(n_base, seed=123)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov LP at parent length (primary workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 420.0)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"nlp_n{n_base}")

    # === Phase 2: Dual Nesterov (two-sided) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Dual Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 160.0)
    if t2 > 20:
        a_out, _ = dual_nesterov_optimize(best_seq, t2, solver)
        try_update(a_out, f"dnl_n{n_base}")

    # === Phase 3: LogSumExp optimization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LogSumExp at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t3, T_start=80.0, T_end=0.01, n_stages=25)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 4: Mirror-descent Nesterov (faster per step) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Mirror-descent Nesterov (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        a_out, _ = nesterov_mirror_optimize(best_seq, t4)
        try_update(a_out, f"nmr_n{n_base}")

    # === Phase 5: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Subgradient polish (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_out, _ = subgradient_polish(best_seq, t5, seed=77)
        try_update(a_out, f"sg_n{n_base}")

    # === Phase 6: Spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Spectral shaping (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 10:
        a_out, _ = spectral_shaping(best_seq, t6, seed=55)
        try_update(a_out, f"spec_n{n_base}")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Multi-length exploration (remaining={remaining:.0f}s) ===")

    candidate_lengths = []
    for offset in [-200, 100, 200, 400]:
        nl = n_base + offset
        if 800 <= nl <= 3000:
            candidate_lengths.append(nl)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_base))

    time_per_length = min(remaining * 0.5 / max(1, len(candidate_lengths)), 120.0)

    for nl in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 30:
            break
        tl = min(time_per_length, remaining * 0.3)
        if tl < 15:
            continue

        print(f"  Trying n={nl} (time={tl:.0f}s)")
        init_nl = interpolate_sequence(best_seq, nl)

        # Quick Nesterov LP
        t_sub = tl * 0.5
        a_out, _ = nesterov_lp_optimize(init_nl, t_sub, solver)
        try_update(a_out, f"nlp_n{nl}")

        # Quick LSE
        rem2 = time_budget - (time.monotonic() - t_global_start)
        t_lse = max(5, min(tl * 0.3, rem2 * 0.05))
        if t_lse > 5:

            try_update(a_out2, f"lse_n{nl}")

    # === Phase 8: Final polishing on best sequence ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish at n={n_best} (remaining={remaining:.0f}s) ===")

    # Coordinate descent
    t8a = min(remaining * 0.30, 150.0)
    if t8a > 10:
        a_out, _ = coordinate_descent_polish(best_seq, t8a, seed=111)
        try_update(a_out, f"cd_n{n_best}")

    # More LogSumExp with very tight parameters
    remaining = time_budget - (time.monotonic() - t_global_start)
    t8b = min(remaining * 0.25, 100.0)
    if t8b > 10:
        a_out, _ = logsumexp_optimize(best_seq, t8b, T_start=15.0, T_end=0.005, n_stages=20)
        try_update(a_out, f"lse_tight_n{n_best}")

    # Subgradient again
    remaining = time_budget - (time.monotonic() - t_global_start)
    t8c = min(remaining * 0.20, 80.0)
    if t8c > 5:
        a_out, _ = subgradient_polish(best_seq, t8c, seed=222)
        try_update(a_out, f"sg_tight_n{n_best}")

    # Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    t8d = min(remaining * 0.20, 80.0)
    if t8d > 5:
        a_out, _ = coordinate_descent_polish(best_seq, t8d, seed=333)
        try_update(a_out, f"cd_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * n_base

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
