def interpolate_length(seq, n_new, n_old):
    """ Linear interpolation between sequence points to change length without scrambling structure """
    n_old = len(seq)
    n_new = int(n_new)
    # Cannot interpolate if the same length
    if n_old == n_new:
        return seq
    # Use linear interpolation between uniform sampling points
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    # Interpolate and enforce structure
    seq_full = np.interp(x_new, x_old, np.sort(seq))
    seq_full = np.clip(seq_full, 0, 1000)
    s_val = np.sum(seq_full)
    if s_val > 0 and s_val < 0.01:
        seq_full = seq_full + 0.01
    return seq_full.tolist()
