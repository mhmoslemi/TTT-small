def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None, rel_tol=None, abs_tol=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    kwargs = {}
    if rel_tol is not None:
        kwargs['rel_tol'] = rel_tol
    if abs_tol is not None:
        kwargs['abs_tol'] = abs_tol
    try:
        prob.solve(solver=solver, **kwargs)

            prob.solve(solver=solver)

            pass

    b_val = np.asarray(b_var.value, dtype=np.float64)

    s = b_val.sum()

def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=500000, perturb_interval=8000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive tolerance: tighter as we progress
        progress = min(1.0, it / 10000.0)
        if progress < 0.3:
            rel_tol = 1e-5
            abs_tol = 1e-7
        elif progress < 0.7:
            rel_tol = 1e-6
            abs_tol = 1e-8
        else:
            rel_tol = 1e-7
            abs_tol = 1e-9

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()

                y = y / sy

            y = a

        if b_new is None:

        if sa < 1e-15:

            a_new = a_new / sa

        sc_new = score_normalized(a_new)

            best_a = a_new.copy()

def dual_guided_subgradient(a0, time_limit, solver=None, max_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.4
    last_improve = 0
    reroll_count = 0
    dual_refresh = 500

    y_weights = None

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 10000:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Periodically refresh dual weights
        if y_weights is None or (step + 1) % dual_refresh == 0:
            y_dual = solve_lp_dual_weights(a, solver)
            if y_dual is not None:
                y_weights = y_dual / y_dual.sum()
            else:
                y_weights = np.ones(len(b)) / len(b)

        # Compute weighted subgradient
        Tb = b * y_weights
        max_Tb = Tb.max()
        if max_Tb > 500:
            exp_Tb = np.exp(Tb - Tb.max())
        else:
            exp_Tb = np.exp(Tb - max_Tb)
        p = exp_Tb / exp_Tb.sum()

        # Gradient: grad_a_i = 2 * sum_k p_k * a_{k-i}
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if p[k] < 1e-25:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * p[k] * a[k - idx]

        # Projected gradient step in log-space
        z = z_from_a(a)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            last_improve = step
            eta = min(eta * 1.001, 0.8)
        else:
            eta *= 0.95
            reroll_count += 1
            if eta < 1e-12:
                break

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-22:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        print(f"Default init: n={n_base}, score={best_score:.10f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(arr)})")

    # === Phase 1: Fast subgradient Nesterov (60s) ===
    print(f"\n=== Phase 1: Subgradient Nesterov at n={n_base} (score={best_score:.10f}) ===")
    t1 = min(60.0, (_global_deadline - time.monotonic()) * 0.05)
    if t1 > 5:
        a_out, _ = subgradient_nesterov_adaptive(best_seq, t1, max_steps=200000)
        try_update(a_out, "subgrad_p1")
    print(f"  Phase 1 done: score={best_score:.10f}")

    # === Phase 2: LP-based Nesterov main (700s) ===
    print(f"\n=== Phase 2: Nesterov LP at n={best_n} (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(700.0, remaining * 0.5)
    if t2 > 30:
        a_out, _ = nesterov_lp_adaptive(best_seq, t2, solver=solver)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 2 done: score={best_score:.10f}")

    # === Phase 3: Dual-guided subgradient (200s) ===
    print(f"\n=== Phase 3: Dual-guided refinement (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(200.0, remaining * 0.18)
    if t3 > 15:
        a_dg, _ = dual_guided_subgradient(best_seq, t3, solver=solver)
        try_update(a_dg, "dual_guided")
    print(f"  Phase 3 done: score={best_score:.10f}")

    # === Phase 4: Second LP round (250s) ===
    print(f"\n=== Phase 4: Nesterov LP round 2 (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(250.0, remaining * 0.22)
    if t4 > 20:
        a_out2, _ = nesterov_lp_adaptive(best_seq, t4, solver=solver, perturb_interval=12000)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 4 done: score={best_score:.10f}")

    # === Phase 5: LSE L-BFGS polishing (250s) ===
    print(f"\n=== Phase 5: LSE L-BFGS polish (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(250.0, remaining * 0.25)
    if t5 > 20:
        a_lse = lbgfs_logsumexp(best_seq, t5, T_start=100.0, T_end=0.02, n_stages=30)
        try_update(a_lse, "lse_polish")
    print(f"  Phase 5 done: score={best_score:.10f}")

    # === Phase 6: Direct gradient fine-tuning (100s) ===
    print(f"\n=== Phase 6: Direct gradient fine-tune (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6 = min(100.0, remaining * 0.15)
    if t6 > 10:
        a_dg = direct_gradient_finetune(best_seq, t6, T=3.0, max_steps=50000)
        try_update(a_dg, "direct_grad")
    print(f"  Phase 6 done: score={best_score:.10f}")

    # === Phase 7: Peak-targeted multiplicative (60s) ===
    print(f"\n=== Phase 7: Peak-targeted refinement (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t7 = min(60.0, remaining * 0.1)
    if t7 > 5:
        a_pt, _ = peak_targeted_refine(best_seq, t7, max_steps=30000)
        try_update(a_pt, "peak_targeted")
    print(f"  Phase 7 done: score={best_score:.10f}")

    # === Phase 8: Final LSE + LP polish (remaining) ===
    print(f"\n=== Phase 8: Final polish (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        t8a = min(40.0, remaining * 0.4)
        if t8a > 5:
            a_lse2 = lbgfs_logsumexp(best_seq, t8a, T_start=30.0, T_end=0.01, n_stages=20)
            try_update(a_lse2, "lse_final")
        remaining = _global_deadline - time.monotonic()
        if remaining > 20:
            t8b = min(40.0, remaining * 0.6)
            if t8b > 10:
                a_out3, _ = nesterov_lp_adaptive(best_seq, t8b, solver=solver, perturb_interval=0)
                try_update(a_out3, "nlp_final")
    print(f"  Phase 8 done: score={best_score:.10f}")

    # === Phase 9: Length exploration ===
    print(f"\n=== Phase 9: Length exploration (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 60:
        for n_try in [best_n - 100, best_n + 100, best_n + 200]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 25:
                break
            if n_try < 100:
                continue
            t9 = min(45.0, remaining * 0.35)
            if t9 < 10:
                break
            print(f"  Trying n={n_try}, budget={t9:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_adaptive(init_try, t9, solver=solver)
            try_update(a_try, f"len_{n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
