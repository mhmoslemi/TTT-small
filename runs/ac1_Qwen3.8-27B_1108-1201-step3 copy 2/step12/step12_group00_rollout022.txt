from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv2(a, b):
    """Full convolution of two sequences via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    B = rfft(b.astype(np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def normalize(a, norm_val):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * norm_val / len(a)
    return a * norm_val / s


def build_conv_matrix(a):
    """Build the convolution matrix T where (T*b)_k = conv(a,b)_k."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve(a, M, n_active=None):
    """
    Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Uses active set if n_active is specified.
    """
    n = len(a)
    size = 2 * n - 1
    
    if n_active is not None and n_active < size:
        # Use only top n_active constraints
        aa = fast_autocorr(a)
        idx = np.argsort(aa)[::-1][:n_active]
        T = build_conv_matrix(a)
        T_act = T[idx, :]
    else:
        T = build_conv_matrix(a)
        T_act = T
    
    c = -np.ones(n)
    rhs = np.full(T_act.shape[0], M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=T_act, b_ub=rhs, bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    b_val = np.maximum(res.x, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_active_set(a, M, tol=1e-8, max_rounds=15):
    """
    LP with active set: start with top constraints, grow as needed.
    """
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    
    # Initial active set: top 50 peaks of autocorrelation
    aa = fast_autocorr(a)
    n_init = min(50, size)
    active = np.argsort(aa)[::-1][:n_init]
    
    b_val = None
    for round_num in range(max_rounds):
        T_act = T[active, :]
        c = -np.ones(n)
        rhs = np.full(len(active), M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=T_act, b_ub=rhs, bounds=bounds, method='highs')
        except Exception:
            return b_val  # Return best so far
        
        if res.status != 0:
            break
        
        b_val = np.maximum(res.x, 0.0)
        if b_val.sum() < 1e-15:
            break
        
        # Check all constraints
        r = T @ b_val
        violated = np.where(r > M + tol)[0]
        
        if len(violated) == 0:
            return b_val
        
        # Grow active set
        active = np.unique(np.concatenate([active, violated]))
        
        # Safety: if active set is too large, use all
        if len(active) > size * 0.8:
            active = np.arange(size)
    
    return b_val


def nesterov_lp(a0, time_limit, verbose=True):
    """
    LP Nesterov with active set LP.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = sqrt(2n).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = normalize(a, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    last_improve = best_score
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current peak
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break
        
        # Solve LP
        b = lp_active_set(y, M_curr)
        if b is None:
            # Fallback: uniform
            b = np.ones(n)
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n)
            b_sum = n
        
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Polyak average
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            avg_check = normalize(avg_check, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_score  # Since sum = sqrt(2n), score = max_conv, obj = 2n*max_conv/(2n) = max_conv
            # Actually: obj = 2*n*max_b/(sum_a)^2 = 2*n*best_score/norm^2 = 2*n*best_score/(2*n) = best_score
            print(f"    NL iter={n_iters} ({elapsed:.0f}s): max_conv={best_score:.8f}, obj={best_score:.8f}")
    
    return best_a, best_score


def lse_gradient_descent(a0, time_limit, T_start=20.0, T_end=0.1):

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_stages = 12

            break
