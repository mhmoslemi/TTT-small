import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def spectral_flatten(a, alpha=0.5, iterations=100):
    """
    Iteratively flatten the power spectrum of a.
    At each step:
    1. Compute FFT of a
    2. Apply spectral flattening: A -> A / |A|^alpha
    3. IFFT back
    4. Clip to non-negative
    5. Renormalize
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        return a
    a = a / a.sum()
    
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(iterations):
        A = rfft(a, fft_size)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-30)
        # Flatten: reduce peaks, boost valleys
        A_flat = A / (mag ** alpha)
        a_new = irfft(A_flat, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
    
    return a

def lse_obj_grad_z(z, T, n):
    """
    Objective and gradient in z-space where a = softmax(z).
    Objective: (1/T)*log(sum(exp(T*b_k)))
    """
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (when valid)
    # This is a cross-correlation
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z

def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=15, seed=42):
    """
    LSE optimization with Adam optimizer and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # More iterations for later stages (lower T = more precise)
        iters = max(30, min(200, int(50 + 150 * si / max(1, n_stages - 1))))
        lr = 0.01 * (1.0 + 0.1 * si)  # Slightly increasing lr for later stages

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            if it % 50 == 49:
                remaining = time_limit - (time.monotonic() - t0)
                if remaining < 2:
                    break

            obj, grad = lse_obj_grad_z(z, stage_T, n)

            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            # Remove mean to maintain softmax invariance
            m_hat_centered = m_hat - np.mean(m_hat)

            z -= lr * m_hat_centered / (np.sqrt(v_hat) + eps)

            # Renormalize z (remove mean for numerical stability)
            z -= z.mean()

            sc = fast_score(np.exp(z - z.max()) / np.sum(np.exp(z - z.max())))
            if sc < best_score:
                best_score = sc
                best_a = np.exp(z - z.max())
                best_a = best_a / best_a.sum()

    return best_a, best_score

def subgradient_momentum(a0, time_limit, eta0=0.03, momentum=0.9, 
                          temp_start=5.0, temp_end=0.5, seed=42):
    """
    Subgradient descent with heavy-ball momentum.
    Uses softmax-weighted all-peak subgradient (not just top-k).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    
    # Momentum buffer
    v = np.zeros(n)
    stall_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        
        # Softmax weights over ALL peaks (temperature anneals)
        temp = temp_start * (temp_end / temp_start) ** (min(t, 5000) / 5000.0)
        b_shift = b - b.max()
        w = np.exp(b_shift / temp)
        w = w / w.sum()

        # Compute subgradient: grad[j] = 2 * sum_k w[k] * a[k-j]
        # Use FFT for efficiency
        a_rev = a[::-1].copy()
        conv_len = n + len(w) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_rev, fft_size)
        conv = irfft(W * A, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Learning rate decay
        eta = eta0 / (1.0 + t / 500.0) ** 0.7

        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / max(s_new, 1e-15)
        else:
            a_new = a_new / s_new

        a = a_new

        # Check score periodically

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Stall handling

            noise_scale = min(0.03, 0.005 * (stall_count / 20.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v *= 0.5  # Reset momentum partially
            stall_count = 0

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            b_max = b.max()
            b_ratio = b_max / (b.sum() / len(b))
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}, bmax/mean={b_ratio:.4f}")

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:

    elif variant == 2:

        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)

    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        # Alternating with noise
        a = np.ones(n) * (1 + 0.3 * np.cos(2 * np.pi * n / 4 * np.arange(n) / n))

def spectral_flatten_iter(a0, iterations=200, alpha_start=0.3, alpha_end=0.7, seed=42):
    """
    Spectral flattening with adaptive alpha.
    Start with mild flattening, increase over time.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        return a
    a = a / a.sum()
    
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = fast_score(a)
    
    for it in range(iterations):
        alpha = alpha_start + (alpha_end - alpha_start) * it / max(1, iterations - 1)
        
        A = rfft(a, fft_size)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-30)
        
        # Flatten: A_new = A * (geomean(mag) / mag)^alpha
        # This is equivalent to A / mag^alpha * geomean(mag)^alpha
        log_mag = np.log(mag)
        log_geomean = log_mag.mean()
        flatten_factor = np.exp(alpha * (log_geomean - log_mag))
        A_flat = A * flatten_factor
        
        a_new = irfft(A_flat, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        
        if it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

def conjugate_grad_lse(a0, time_limit, T_start=30.0, T_end=0.1, n_stages=10):
    """
    Conjugate gradient method for LSE objective.
    More memory-efficient than L-BFGS for large n.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 10:
            break

        iters = max(20, min(100, int(30 + 70 * si / max(1, n_stages - 1))))
        
        # CG in z-space
        _, g = lse_obj_grad_z(z, stage_T, n)
        d = -g  # Initial search direction
        g = g.copy()
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            if it % 20 == 19 and (time_limit - (time.monotonic() - t0)) < 3:
                break
            
            # Line search (backtracking)
            alpha = 0.1
            obj_prev, _ = lse_obj_grad_z(z, stage_T, n)
            
            for ls in range(20):
                z_trial = z + alpha * d
                z_trial -= z_trial.mean()  # Center for stability
                obj_trial, g_trial = lse_obj_grad_z(z_trial, stage_T, n)
                if obj_trial < obj_prev - 1e-4 * alpha * np.dot(g, d):
                    break
                alpha *= 0.5
            else:
                break
            
            z += alpha * d
            z -= z.mean()
            
            # Check score
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = fast_score(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
            
            if it < iters - 1:
                g_new = g_trial
                beta = np.dot(g_new - g, g_new) / (np.dot(g, g) + 1e-30)
                beta = max(0, min(beta, 1.0))  # Polak-Ribière with clipping
                d = -g_new + beta * d
                g = g_new

    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral domain flattening (NOVEL: iteratively flatten power spectrum to reduce time-domain peaks)
2. Adam-optimized LogSumExp with progressive temperature annealing
3. Subgradient descent with heavy-ball momentum and all-peak softmax weighting
4. Conjugate gradient on LSE objective
5. Peak shaving coordinate descent
6. Multi-length exploration with spectral transfer
7. Population-based blending for diversity
Key innovations: spectral-domain initialization, Adam adaptive learning, all-peak softmax subgradient,
conjugate gradient alternative, and iterative power-spectrum flattening.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def cross_corr_w_a(w, a):
    """Compute cross-correlation: result[j] = sum_i w[j+i]*a[i]."""
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def lse_obj_grad_z(z, T):
    """
    Objective and gradient in z-space where a = softmax(z).
    Objective: (1/T)*log(sum(exp(T*b_k)))
    """
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    cc = cross_corr_w_a(w, a)
    grad_a = 2.0 * cc

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def spectral_flatten_iter(a0, iterations=200, alpha_start=0.2, alpha_end=0.6, time_limit=None, t0=None):
    """
    Iteratively flatten the power spectrum of a to reduce time-domain autocorrelation peaks.
    At each step:
    1. Compute rFFT of a
    2. Apply spectral flattening: multiply by (geomean(|A|)/|A|)^alpha
    3. IFFT back
    4. Clip to non-negative
    5. Renormalize
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        return a, fast_score(a)
    a = a / a.sum()

    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = fast_score(a)

    for it in range(iterations):
        if time_limit is not None and t0 is not None:
            if time.monotonic() - t0 > time_limit:
                break

        alpha = alpha_start + (alpha_end - alpha_start) * it / max(1, iterations - 1)

        A = rfft(a, fft_size)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-30)

        # Flatten: boost valleys, suppress peaks
        log_mag = np.log(mag)
        log_geomean = log_mag.mean()
        flatten_factor = np.exp(alpha * (log_geomean - log_mag))
        A_flat = A * flatten_factor

        a_new = irfft(A_flat, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

        if it % 50 == 0 or it == iterations - 1:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=15, seed=42):
    """
    LSE optimization with Adam optimizer and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    z -= z.mean()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        iters = max(30, min(200, int(40 + 160 * si / max(1, n_stages - 1))))
        lr = 0.008 * (1.0 + 0.15 * si)

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            if it % 100 == 99 and (time.monotonic() - t0 > time_limit - 3):
                break

            obj, grad = lse_obj_grad_z(z, stage_T)

            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            # Center to maintain softmax invariance
            m_hat_centered = m_hat - np.mean(m_hat)

            z -= lr * m_hat_centered / (np.sqrt(v_hat) + eps)
            z -= z.mean()

            # Periodic score check
            if it % 20 == 19:
                a_cur = np.exp(z - z.max())
                a_cur = a_cur / a_cur.sum()
                sc = fast_score(a_cur)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()

    # Final check
    a_cur = np.exp(z - z.max())
    a_cur = a_cur / a_cur.sum()
    sc = fast_score(a_cur)
    if sc < best_score:
        best_score = sc
        best_a = a_cur.copy()

    return best_a, best_score


def subgradient_momentum(a0, time_limit, eta0=0.02, momentum=0.85,
                         temp_start=3.0, temp_end=0.2, seed=42):
    """
    Subgradient descent with heavy-ball momentum.
    Uses softmax-weighted all-peak subgradient.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    stall_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()

        # Softmax weights over peaks (temperature annealing)
        progress = min(1.0, t / 3000.0)
        temp = temp_start + (temp_end - temp_start) * progress
        b_shift = b - b_max
        w = np.exp(b_shift / temp)
        w = w / w.sum()

        # Compute subgradient via cross-correlation
        grad = 2.0 * cross_corr_w_a(w, a)

        # Learning rate
        eta = eta0 / (1.0 + t / 400.0) ** 0.7

        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / max(s_new, 1e-15)
        else:
            a_new = a_new / s_new

        a = a_new

        # Score check
        check_interval = 5 if t < 500 else 20
        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Stall handling
        if stall_count > 80:
            noise_scale = min(0.02, 0.003 * (stall_count / 20.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v *= 0.3
            stall_count = 0

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def conjugate_grad_lse(a0, time_limit, T_start=30.0, T_end=0.1, n_stages=10):
    """
    Conjugate gradient method for LSE objective.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    z -= z.mean()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 10:
            break

        iters = max(20, min(80, int(25 + 55 * si / max(1, n_stages - 1))))

        _, g = lse_obj_grad_z(z, stage_T)
        d = -g
        g = g.copy()

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            if it % 30 == 29 and (time_limit - (time.monotonic() - t0)) < 3:
                break

            # Backtracking line search
            alpha = 0.05
            obj_prev, _ = lse_obj_grad_z(z, stage_T)
            slope = np.dot(g, d)

            accepted = False
            for ls in range(15):
                if slope >= 0:
                    # Direction is not descent, restart
                    z_trial = z
                    obj_trial, g_trial = lse_obj_grad_z(z_trial, stage_T)
                    accepted = True
                    break
                z_trial = z + alpha * d
                z_trial -= z_trial.mean()
                obj_trial, g_trial = lse_obj_grad_z(z_trial, stage_T)
                if obj_trial <= obj_prev + 1e-4 * alpha * slope:
                    accepted = True
                    break
                alpha *= 0.5

            if not accepted:
                break

            z = z_trial
            z -= z.mean()

            # Score check
            if it % 10 == 9:
                a_cur = np.exp(z - z.max())
                a_cur = a_cur / a_cur.sum()
                sc = fast_score(a_cur)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()

            if it < iters - 1:
                g_new = g_trial
                if np.dot(g, g) > 1e-30:
                    beta = np.dot(g_new - g, g_new) / np.dot(g, g)
                    beta = max(0.0, min(beta, 1.5))
                else:
                    beta = 0.0
                d = -g_new + beta * d
                g = g_new

    a_cur = np.exp(z - z.max())
    a_cur = a_cur / a_cur.sum()
    sc = fast_score(a_cur)
    if sc < best_score:
        best_score = sc
        best_a = a_cur.copy()

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Coordinate descent: reduce top autocorrelation peaks by moving mass
    from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k_idx in top_k:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            src = int(idx[np.argmax(contrib)])

            # Find safe target positions
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]
            sorted_candidates = candidates[np.argsort(a[candidates])[:max(5, len(candidates) // 10)]]
            if len(sorted_candidates) == 0:
                continue
            dst = int(sorted_candidates[rng.integers(len(sorted_candidates))])

            for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            if rng.integers(4) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 250:
                break

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * np.arange(n) / n)
    else:
        a = np.ones(n) * 0.5
        for _ in range(3 + rng.integers(0, 4)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_start_spectral(n, time_limit, seeds=(1, 2, 3)):
    """
    Multiple spectral flattening runs from different starts.
    Returns best result.
    """
    best_a = None
    best_score = float('inf')
    t0 = time.monotonic()
    per_seed_time = time_limit / len(seeds)

    for si, seed in enumerate(seeds):
        if time.monotonic() - t0 > time_limit:
            break
        rng = np.random.default_rng(seed)
        variant = si % 8

        a_init = diverse_init(n, rng, variant=variant)
        per_limit = time_limit - (time.monotonic() - t0)
        per_limit = min(per_limit, per_seed_time * (len(seeds) - si))
        if per_limit < 10:
            continue

        a_out, sc = spectral_flatten_iter(a_init, iterations=300,
                                           alpha_start=0.15, alpha_end=0.5,
                                           time_limit=per_limit, t0=t0)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining:
    1. Spectral domain flattening (iterative power spectrum equalization)
    2. Adam-optimized LSE with progressive annealing
    3. Subgradient descent with momentum (all-peak softmax weighting)
    4. Conjugate gradient on LSE
    5. Peak shaving coordinate descent
    6. Multi-length exploration
    Key innovations over previous: spectral-domain ops, Adam adaptive rates,
    all-peak subgradient, CG alternative, multi-start diversity.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1760.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = fast_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    n_parent = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = diverse_init(n_parent, rng, variant=4)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Spectral Flattening (novel) ===
    print(f"\n=== Phase 1: Spectral Flattening (n={n_parent}) ===")
    t_sf1 = min(0.10 * time_budget, 170.0)
    if t_sf1 > 20:
        a_out, sc = spectral_flatten_iter(best_seq, iterations=250,
                                           alpha_start=0.15, alpha_end=0.5,
                                           time_limit=t_sf1, t0=t_global)
        try_update(a_out, "spectral_main")

    # Multi-start spectral from diverse inits
    remaining = time_budget - (time.monotonic() - t_global)
    t_sf2 = min(remaining * 0.06, 80.0)
    if t_sf2 > 15:
        a_out, sc = multi_start_spectral(n_parent, t_sf2, seeds=(42, 123, 456))
        try_update(a_out, "spectral_multi")

    # === Phase 2: Adam LSE ===
    print(f"\n=== Phase 2: Adam LSE (n={n_parent}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse1 = min(remaining * 0.18, 300.0)
    if t_lse1 > 30:
        a_out, _ = adam_lse_optimize(best_seq, t_lse1, T_start=60.0, T_end=0.2,
                                      n_stages=15, seed=7)
        try_update(a_out, f"adam_lse_n{n_parent}")

    # === Phase 3: Subgradient + Momentum ===
    print(f"\n=== Phase 3: Subgrad+Momentum (n={n_parent}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_sg = min(remaining * 0.14, 240.0)
    if t_sg > 20:
        a_out, _ = subgradient_momentum(best_seq, t_sg, eta0=0.02, momentum=0.82,
                                         temp_start=4.0, temp_end=0.2, seed=13)
        try_update(a_out, f"subgrad_n{n_parent}")

    # === Phase 4: CG LSE ===
    print(f"\n=== Phase 4: CG LSE (n={n_parent}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_cg = min(remaining * 0.10, 150.0)
    if t_cg > 20:
        a_out, _ = conjugate_grad_lse(best_seq, t_cg, T_start=40.0, T_end=0.15, n_stages=10)
        try_update(a_out, f"cg_n{n_parent}")

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (n={n_parent}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_ps1 = min(remain
