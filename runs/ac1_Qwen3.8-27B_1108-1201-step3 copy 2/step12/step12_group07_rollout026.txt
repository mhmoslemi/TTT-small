    return b_val, float(M_var.value)


def structured_init_asymmetric(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.8 * (2.0 * x - 1.0)
    elif choice == 1:
        a = x + 0.1
    elif choice == 2:
        a = 2.0 * np.exp(-3.0 * x) + 0.3
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.2
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (x + 0.1)) + 0.15 * np.cos(4 * np.pi * x)
    elif choice == 5:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.7) ** 2) + 0.1
    elif choice == 6:
        a = (x + 0.05) ** 0.5
    else:
        a = 0.8 + 0.4 * np.sin(2 * np.pi * x + 0.5)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0005 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad_fft(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.max())
    obj = (2.0 * n / T_val) * log_sum_exp

    # FFT-accelerated gradient: grad_i = 2 * sum_k w[k] * a[k-i]
    # This is 2 * cross_correlation(w, a)[i]
    # cross_correlation(w, a)[i] = sum_j w[i+j] * a[j]
    # = IFFT(FFT(w) * conj(FFT(a))) -- but need careful indexing
    # Alternative: use the direct vectorized approach
    # For each i: sum_{j=0}^{n-1} w[i+j] * a[j]
    # Build using dot products (vectorized over i)
    # w has length 2n-1
    # grad[i] = w[i:i+n] dot a
    # Use stride tricks or just a loop with vectorized inner product
    grad_a = np.empty(n)
    for i in range(n):
        grad_a[i] = np.dot(w[i:i + n], a)
    grad_a *= 2.0

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_objective_grad_direct(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.max())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.005, n_stages=60,
                    iters_high=30, iters_low=50):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    # Use a power-law schedule: more stages at low temperature
    # T_i = T_start * (T_end/T_start)^(i/n_stages) but with bias toward low T
    # Use: T_i = T_start * (T_end/T_start)^(i^beta/n_stages) with beta > 1 for more low-T
    temps = np.array([T_start * (T_end / T_start) ** ((i / n_stages) ** 1.3) for i in range(n_stages)])

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # More iterations at low temperatures
        if stage_T > T_start * 0.1:
            iters = iters_high
        elif stage_T > T_start * 0.01:
            iters = (iters_high + iters_low) // 2
        else:
            iters = iters_low

        def func_staged(z_val):
            obj, _ = lse_objective_grad_direct(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad_direct(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
            if (si + 1) % 10 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: score={best_score:.8f}")
        except Exception:
            pass

    return best_a, best_score


def peak_targeted_refine(a0, time_limit, K=30, max_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.05
    last_improve_step = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.92

        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.002, 0.15)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, max_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.3
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.95)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.001:
                t_nest = 1.0
                eta *= 0.9
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.999, 0.01)

    return best_a, best_score


def perturb_sequence(a, rng, strength=0.1):
    """Perturb sequence with smooth noise to escape local optima."""
    n = len(a)
    # Generate smooth noise via low-frequency sine components
    x = np.linspace(0, 1, n)
    noise = np.zeros(n)
    for freq in [1, 2, 3, 5, 7, 11]:
        noise += rng.normal(0, 1) * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    noise /= np.sqrt(len(noise) * sum(1 for _ in [1, 2, 3, 5, 7, 11]))
    perturbed = a * (1 + strength * noise)
    perturbed = np.maximum(perturbed, 0)
    s = perturbed.sum()
    if s < 1e-15:
        return a.copy()
    return perturbed / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            best_seq = p / p.sum()

            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()

    print(f"\n=== Phase 1: Subgradient Nesterov at n={n_base} (score={best_score:.8f}) ===")
