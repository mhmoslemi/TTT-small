g_j = sum_k w_k * T_{k,j} = sum_{k=j}^{j+n-1} w_k * a_j = a_j * sum_{k=j}^{j+n-1} w_k

So g_j = a_j * (W[j] + W[j+1] + ... + W[j+n-1]) where W = w (the softmax weights over lags).

This can be computed in O(n) using a prefix sum of w! And w itself can be computed from a*b (which is O(n log n) via FFT).

So the total cost per mirror descent step is O(n log n) (dominated by the FFT for a*b).

This is MUCH faster than solving an LP (which is O(n^3) or more in the worst case, though in practice with good solvers it's much less).

def fast_convolve(a, b=None):
    """Fast convolution using FFT. If b is None, computes a*a."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)

def fast_best_response(a, n_iters=50, tau=200.0, eta=0.1):
    """
    Approximate best-response using mirror descent with softmax-smoothed max.
    Given fixed a, find b (probability vector) that minimizes max_k (a*b)_k.
    """
    n = len(a)
    b = np.ones(n) / n
    best_b = b.copy()
    best_val = np.inf
    
    for it in range(n_iters):
        # Compute a * b
        ab = fast_convolve(a, b)
        # Softmax weights
        max_ab = ab.max()
        if max_ab > 500:
            max_ab = 500
        exp_ab = np.exp(tau * (ab - max_ab))
        w = exp_ab / exp_ab.sum()
        
        # Compute gradient: g_j = a_j * sum_{k=j}^{j+n-1} w_k
        # Using prefix sum of w (padded to length 3n-2 to handle boundaries)
        w_ext = np.zeros(3 * n)
        w_ext[:len(w)] = w
        w_prefix = np.cumsum(w_ext)
        
        g = np.zeros(n)
        for j in range(n):
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            w_sum = w_prefix[k_end + 1] - w_prefix[k_start]
            g[j] = a[j] * w_sum
        
        # Mirror descent update
        exponent = -eta * g
        exponent -= exponent.max()
        new_b = b * np.exp(exponent)
        sb = new_b.sum()
        if sb < 1e-300:
            break
        new_b = new_b / sb
        
        # Check improvement
        ab_new = fast_convolve(a, new_b)
        val = ab_new.max()
        if val < best_val:
            best_val = val
            best_b = new_b.copy()
        b = new_b
        
        # Adaptive eta
        if val > best_val * 1.01:
            eta *= 0.5
        else:
            eta = min(eta * 1.01, 0.5)
    
    return best_b, best_val

```python
def fast_best_response(a, n_iters=50, tau=200.0, eta=0.1):
    n = len(a)
    b = np.ones(n) / n
    best_b = b.copy()
    best_val = np.inf
    
    for it in range(n_iters):
        ab = fast_convolve(a, b)
        max_ab = ab.max()
        if max_ab > 500:
            max_ab = 500
        exp_ab = np.exp(tau * (ab - max_ab))
        w = exp_ab / exp_ab.sum()
        
        # g_j = a_j * sum_{k=j}^{j+n-1} w_k (capped at len(w)-1 = 2n-2)
        # Use prefix sum
        w_pad = np.concatenate([w, np.zeros(n-1)])  # length 3n-2
        w_prefix = np.concatenate([[0], np.cumsum(w_pad)])  # length 3n-1
        
        # For j=0,...,n-1: sum_{k=j}^{min(j+n-1, 2n-2)} w_k
        # = w_prefix[min(j+n, 2n-1)] - w_prefix[j]
        k_end = np.minimum(np.arange(n) + n, 2 * n - 1)
        w_sums = w_prefix[k_end] - w_prefix[np.arange(n)]
        
        g = a * w_sums
        
        # Mirror descent
        exponent = -eta * g
        exponent -= exponent.max()
        new_b = b * np.exp(exponent)
        sb = new_b.sum()
        if sb < 1e-300:
            break
        new_b /= sb
        
        ab_new = fast_convolve(a, new_b)
        val = ab_new.max()
        if val < best_val:
            best_val = val
            best_b = new_b.copy()
        b = new_b
        eta = min(eta * 1.005, 0.5)
    
    return best_b, best_val
```

```python
def nesterov_fast(a0, time_limit, n_inner=30, tau_start=100.0, tau_end=500.0):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y /= sy
        else:
            y = a
        
        # Fast best-response
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / 1000.0)
        b_new, _ = fast_best_response(y, n_inner=n_inner, tau=tau)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new /= sa
        
        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        # Adaptive restart
        if sc_new > best_score * 1.001 and t_nest > 5.0:
            t_nest = 1.0
        else:
            t_nest = t_new
        
        a_prev = a.copy()
        a = a_new
    
    return best_a, best_score
```
