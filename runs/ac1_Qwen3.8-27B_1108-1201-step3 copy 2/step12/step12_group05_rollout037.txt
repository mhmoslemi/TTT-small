    # For efficiency, let's compute:
    # Part1[j] = sum_m a[m]*w[j+m] = cross-correlation of a and w at "sub-lag" j
    # Part2[j] = sum_m a[m]*w[j-m] = standard convolution of a and w at j
    
    # Part2 is just conv(a, w)[j] (with appropriate padding)
    # Part1: sum_m a[m]*w[j+m]. Let's define w_rev[k] = w[2n-2-k]. Then:
    #   sum_m a[m]*w[j+m] = sum_m a[m]*w_rev[2n-2-j-m] = conv(a, w_rev)[n-1-j + n-1]... 
    # This is getting too complicated. Let me just compute it directly for moderate n.
    
    # For n <= 3000, direct computation of the gradient takes O(n^2) which is 9M operations.
    # That's about 10ms in numpy. Acceptable.
    
    # Actually, let's just use the FFT approach properly:
    # Part1[j] = sum_{m=0}^{n-1} a[m] * w[j+m] (w zero-padded)
    # This is the "anti-convolution": (a * w_flip_full)[j + 2n - 2] where w_flip_full has length 2n-1
    # Let's define: a_pad = [a, 0, ..., 0] (length 3n-2), w_pad = [w, 0, ..., 0] (length 3n-2)
    # Then (a_pad * w_pad)[j + 2n - 2] = sum_k a_pad[k] * w_pad[j+2n-2-k]
    # = sum_{k=0}^{n-1} a[k] * w[j+2n-2-k] (since a_pad[k]=0 for k>=n)
    # Let m = 2n-2-k: = sum_{m=2n-1-n}^{2n-2} a[2n-2-m] * w[j+m] ... this is getting circular.
    
    # Let me just do it the simple O(n^2) way for correctness, and optimize later if needed.
    # Actually for n=1600, O(n^2) = 2.56M which is fast enough.
    
    # Even simpler: use the fact that the gradient is a "blurred" version of a.
    # grad_f[j] = f^(p-1) * (h * a)[j] where h[k] = w[k] + w[-k] (the "symmetrized" weight)
    # Let's verify: (h * a)[j] = sum_k h[k]*a[j-k] = sum_k (w[k]+w[-k])*a[j-k]
    # = sum_k w[k]*a[j-k] + sum_k w[-k]*a[j-k]
    # Second term: let m=-k: sum_m w[m]*a[j+m]
    # So (h*a)[j] = sum_k w[k]*a[j-k] + sum_m w[m]*a[j+m] = Part2[j] + Part1[j] ✓
    
    # So the gradient is: grad_f[j] = f^(p-1) * (h * a)[j] where h[k] = w[k] + w[-k]
    # and h is defined on [-2n+2, 2n-2] (with h[k]=0 outside [0,2n-2] for w[k]).
    
    # h[k] = w[k] + w[-k] for k in [-(2n-2), 2n-2], with w out of range = 0.
    # For k >= 0: h[k] = w[k] + (w[-k] if -k >= 0 else 0) = w[k] + (w[-k] if k <= 0 else 0)
    #   For k=0: h[0] = w[0] + w[0] = 2*w[0]
    #   For k>0: h[k] = w[k]
    # For k < 0: h[k] = (w[k] if k>=0 else 0) + w[-k] = w[-k]
    #   So h[-k] = w[k] for k > 0, meaning h is symmetric: h[-k] = h[k] for k > 0.
    #   And h[0] = 2*w[0].
    
    # So h is: h[0] = 2*w[0], h[k] = w[k] for k=1,...,2n-2, h[-k] = w[k] for k=1,...,2n-2.
    # Length of h: from -(2n-2) to 2n-2, total 4n-3 elements.
    
    # (h * a)[j] = sum_k h[k]*a[j-k] = 2*w[0]*a[j] + sum_{k=1}^{min(2n-2, j)} w[k]*a[j-k] + sum_{k=1}^{min(2n-2, 2n-1-j)} w[k]*a[j+k]
    
    # This is: 2*w[0]*a[j] + (w * a)[j] (truncated) + (w * a_rev_shifted)[j]
    # In FFT terms: (h * a) can be computed as IFFT(FFT(h) * FFT(a)) with appropriate padding.
    
    # h has length 4n-3 (from index -(2n-2) to 2n-2).
    # a has length n.
    # The convolution h*a has length (4n-3) + n - 1 = 5n-3.
    # We only need positions corresponding to j=0,...,n-1.
    
    # Let's set up the FFT properly.
    h = np.zeros(4 * n - 3)
    h[n - 2 + 1:] = w  # h[k] for k=0,...,2n-2 starts at index (2n-2) in the array
    # h array index i corresponds to offset i - (2n-2)
    # h[2n-2] = h[offset 0] = 2*w[0] -- wait, I defined h[0] = 2*w[0]
    # Let me redefine: h[offset] where offset = i - (2n-2)
    # h[offset=0] = 2*w[0]
    # h[offset=k] = w[k] for k=1,...,2n-2
    # h[offset=-k] = w[k] for k=1,...,2n-2
    h_idx = 2 * n - 2
    h[h_idx] = 2 * w[0]
    h[h_idx + 1:h_idx + (2 * n - 1)] = w[1:]
    h[h_idx - (2 * n - 2):h_idx] = w[(2 * n - 2): 0 : -1]
    
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    h_pad = np.zeros(fft_size)
    # h has indices from 0 to 4n-4 in the array, corresponding to offsets -(2n-2) to 2n-2
    # We need to place h into h_pad such that the convolution works out.
    # (h*a)[j] where j is the "true" index (0 to n-1)
    # In the padded array, a is at positions 0 to n-1.
    # h is centered at position h_idx = 2n-2 (offset 0).
    # The convolution (h*a)[true_j] = sum_offset h[offset] * a[true_j - offset + ...]
    # This is getting confusing. Let me just use scipy.signal.fftconvolve equivalent.
    
    conv_result = np.fft.irfft(np.fft.rfft(h_pad, fft_size) * np.fft.rfft(a_pad, fft_size), fft_size)
    
    # The conv_result[t] corresponds to (h_pad * a_pad)[t] = sum_i h_pad[i]*a_pad[t-i]
    # a_pad[i] = a[i] for i in [0,n-1], 0 otherwise.
    # h_pad[i] = h[i] for i in [0, 4n-4] (if we placed it correctly)
    # We want (h * a)[j] = sum_offset h[offset]*a[j-offset] for j=0,...,n-1
    # In terms of the array: sum_i h_pad[i]*a_pad[j-i] where the "true" j corresponds to...
    # h_pad[h_idx + offset] = h[offset] (by construction)
    # So sum_i h_pad[i]*a_pad[t-i] = sum_offset h[offset]*a_pad[t - h_idx - offset]
    # = sum_offset h[offset]*a[t - h_idx - offset] (where a is zero outside [0,n-1])
    # We want this to equal sum_offset h[offset]*a[j-offset] when t - h_idx = j, i.e., t = j + h_idx.
    
    grad_f = f ** (p - 1) * conv_result[h_idx:h_idx + n]
    
    return grad_f, f


def compute_lp_grad_direct(a, p):
    """Direct O(n^2) computation of L^p gradient. Slower but simpler and correct."""
    n = len(a)
    c = fast_autocorr(a)
    c = np.maximum(c, 1e-30)
    f = (np.sum(c ** p)) ** (1.0 / p)
    w = c ** (p - 1)
    
    # grad[j] = f^(p-1) * sum_k w[k] * (a[k-j] if 0<=k-j<n else 0 + a[j-k] if 0<=j-k<n else 0)
    grad = np.zeros(n)
    for j in range(n):
        # First part: sum over k where 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_lo = max(0, j)
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            grad[j] += np.dot(a[k_lo - j:k_hi - j + 1], w[k_lo:k_hi + 1])
        # Second part: sum over k where 0 <= j-k < n, i.e., j-n+1 <= k <= j
        k_lo2 = max(0, j - n + 1)
        k_hi2 = min(2 * n - 2, j)
        if k_lo2 <= k_hi2:
            grad[j] += np.dot(a[j - k_lo2:j - k_hi2 + 1][::-1], w[k_lo2:k_hi2 + 1])
    
    grad *= f ** (p - 1)
    return grad, f


def lp_optimize(a0, time_limit, p_start=4.0, p_end=64.0, n_stages=8):
    """
    L^p-norm optimization with Adam and p-annealing.
    Minimizes (sum_k c[k]^p)^(1/p) as p→∞ approaches max_k c[k].
    Uses softmax parameterization.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    p_values = np.geomspace(p_start, p_end, n_stages)
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for si, p_val in enumerate(p_values):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        iters = max(10, min(200, int(remaining * 120 / max(1, n_stages - si))))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_fn(z)
            # Compute gradient of L^p objective w.r.t. z
            a_for_grad = np.maximum(a, 1e-15)
            c = fast_autocorr(a_for_grad)
            c = np.maximum(c, 1e-30)
            f = (np.sum(c ** p_val)) ** (1.0 / p_val)
            w = c ** (p_val - 1)
            
            # Gradient w.r.t. a: grad_a[j] = f^(p-1) * (h * a)[j]
            # Use FFT: h is the symmetrized w
            h = np.zeros(4 * n - 3)
            h_idx = 2 * n - 2
            h[h_idx] = 2 * w[0]
            h[h_idx + 1:h_idx + (2 * n - 1)] = w[1:]
            h[h_idx - (2 * n - 2):h_idx] = w[(2 * n - 2): 0 : -1]
            
            fft_size = 1 << (5 * n - 2).bit_length()
            h_pad = np.zeros(fft_size)
            h_pad[:len(h)] = h
            a_pad = np.zeros(fft_size)
            a_pad[:n] = a_for_grad
            
            conv_result = np.fft.irfft(np.fft.rfft(h_pad) * np.fft.rfft(a_pad), fft_size)
            grad_a = f ** (p_val - 1) * conv_result[h_idx:h_idx + n]
            
            # Chain rule: grad_z = a * (grad_a - dot(grad_a, a))
            S = np.dot(grad_a, a_for_grad)
            grad_z = a_for_grad * (grad_a - S)
            
            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_t = 0.01 / (1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 20 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      L^p stage {si}/{n_stages} (p={p_val:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_fast(a_fixed, solver, warm_start=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response with warm starts and adaptive rate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    no_improve_count = 0
    last_best_score = best_score
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_fast(y, solver)
        if b_star is None:
            no_improve_count += 1
            if no_improve_count > 50:
                break
            continue
        
        # Adaptive step size: start with larger steps, decrease over time
        if t < 100:
            alpha_t = 0.3
        elif t < 500:
            alpha_t = 2.0 / (t + 2.0)
        else:
            alpha_t = 1.0 / np.sqrt(t)
        
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            no_improve_count = 0
        
        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                no_improve_count = 0
        
        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")
        
        if no_improve_count > 200:
            # Try a restart with perturbation
            rng = np.random.default_rng(t)
            pert = rng.random(n) * 0.1
            a_pert = a + pert
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            y = a_pert.copy()
            no_improve_count = 0
            avg_a = np.zeros(n)
            n_avg = 0
    
    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def multi_peak_shaving(a0, time_limit, seed=123):
    """
    Multi-peak coordinate descent: simultaneously target multiple peaks.
    Uses smart candidate selection based on contribution analysis.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:12]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.sum() < 1e-15:
                continue
            
            # Find the pair with maximum contribution
            src_local = np.argmax(contrib)
            src = idx[src_local]
            partner = k - src
            
            # Also try second and third largest contributions
            contrib_sorted = np.argsort(contrib)[::-1]
            
            for ci in range(min(3, len(contrib_sorted))):
                src_c = idx[contrib_sorted[ci]]
                partner_c = k - src_c
                
                # Find target: positions that don't contribute to any top peak
                peak_areas = set()
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for j2 in range(js2, je2 + 1):
                        peak_areas.add(j2)
                        if 0 <= k2 - j2 < n:
                            peak_areas.add(k2 - j2)
                
                candidates = [j for j in range(n) if j not in peak_areas and j != src_c]
                if not candidates:
                    candidates = [j for j in range(n) if j != src_c]
                
                # Sort by a value (ascending) - move to low-a positions
                candidates_sorted = sorted(candidates, key=lambda j: a[j])
                dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
                
                for _ in range(3):  # Try a few random targets
                    dst = dst_pool[rng.integers(len(dst_pool))]
                    
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                        amt = a[src_c] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src_c] -= amt
                        a_trial[dst] += amt
                        if a_trial[src_c] < 0:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 * x
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.15 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.25) ** 2) + 0.7 * np.exp(-4 * (x - 0.75) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.4 + 0.3 * np.cos(6 * np.pi * x ** 1.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        # Anti-symmetric peaked structure (good for reducing center autocorrelation)
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.5) ** 2) + 0.3 * np.exp(-50 * (x - 0.2) ** 2) + 0.3 * np.exp(-50 * (x - 0.8) ** 2)
    elif choice == 8:
        # Broad peaks at edges
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * x ** 2) + np.exp(-30 * (1 - x) ** 2) + 0.5 * np.exp(-10 * (x - 0.5) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.4 * np.abs(np.sin(5 * np.pi * x)) + 0.2 * np.cos(20 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
