"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Smoothed FISTA (log-sum-exp approximation)
3. KKT-guided coordinate descent

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def smoothed_fista(a0, time_limit, beta_start=50.0, beta_end=500.0, n_iters=20000):
    """
    FISTA on the log-sum-exp smoothed objective.
    F(a) = (1/beta) * log(sum_k exp(beta * (a*a)_k))
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    z = a.copy()  # FISTA momentum variable

    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive beta
        progress = t / n_iters
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute smoothed objective gradient at z
        b = fast_autocorr(z)
        b_max = b.max()
        b_shifted = b - b_max
        weights = np.exp(beta * b_shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(len(b))
            weights[np.argmax(b)] = 1.0
        else:
            weights = weights / w_sum

        # Gradient: sum_k w_k * 2 * z[k-i] for coordinate i
        # This is a weighted sum of shifts of z
        grad = np.zeros(n)
        for k_idx in range(len(b)):
            if weights[k_idx] < 1e-10:
                continue
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k_idx - idx
            grad[idx] += 2 * weights[k_idx] * z[other]
            grad[other] += 2 * weights[k_idx] * z[idx]

        # Step size: use 1/L where L is Lipschitz constant bound
        # L ~ 2 * max_eig of the weighted sum of Toeplitz matrices
        # Upper bound: 2 * sum_k w_k * sum_j |z_{k-j}| = 2 * (sum z) = 2 (since sum=1)
        # More conservatively: L = 4 (safe upper bound)
        step = 1.0 / 4.0

        a_new = z - step * grad
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            a_new = z.copy()
            a_new_s = 1.0
        a_new = a_new / a_new_s

        # FISTA update
        t_plus = (1 + np.sqrt(1 + 4 * t * t)) / 2
        theta = (1 + t) / t_plus
        z_new = a_new + theta * (a_new - a)
        z_new = np.maximum(z_new, 0)
        z_new_s = z_new.sum()
        if z_new_s < 1e-15:
            z_new = a_new.copy()
            z_new_s = 1.0
        z_new = z_new / z_new_s

        a = a_new
        z = z_new

        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FISTA t={t} ({elapsed:.0f}s): best={best_score:.6f}, beta={beta:.1f}")

    return best_a, best_score


def kkt_coordinate_descent(a0, time_limit, n_sweeps=200):
    """
    Coordinate descent guided by KKT conditions.
    At optimum: sum_k lambda_k * a[k-i] = constant for all i in support.
    We iterate: identify active set, compute multipliers, adjust a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()

        # Identify active lags (within 1% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]

        # Compute KKT "residual": for each i, compute sum_k lambda_k * a[k-i]
        # where lambda_k are proportional to (b_k - second_max)^2 (heuristic)
        # Better: use softmax weights
        b_active = b[active]
        w = np.exp(b_active / (max_b * 0.01))
        w = w / w.sum()

        # Compute the "target" function: g(i) = sum_k w_k * a[k-i]
        g = np.zeros(n)
        for wi, ki in zip(w, active):
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = ki - idx
            g[idx] += wi * a[other]
            g[other] += wi * a[idx]

        # At KKT, g(i) should be constant on support
        # If g(i) is high, we should increase a_i; if low, decrease
        support = a > 1e-8
        if support.sum() > 0:
            g_mean = g[support].mean()
            # Perturb: move mass from low-g to high-g positions
            diff = g - g_mean
            # Scale perturbation
            scale = 0.01  # Small step
            a_trial = a + scale * diff
            a_trial = np.maximum(a_trial, 0)
            a_trial_s = a_trial.sum()
            if a_trial_s > 1e-15:
                a_trial = a_trial / a_trial_s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial

        # Also try direct coordinate descent: for each i, optimize a_i
        # (keeping sum=1 by proportionally scaling others)
        if sweep % 10 == 0:
            # Try moving mass from worst to best positions
            b_new = fast_autocorr(a)
            peak_lag = np.argmax(b_new)
            # Gradient at this peak: 2 * a[peak_lag - i]
            grad_peak = 2.0 * np.zeros(n)
            js = max(0, peak_lag - n + 1)
            je = min(n - 1, peak_lag)
            idx = np.arange(js, je + 1)
            other = peak_lag - idx
            grad_peak[idx] = 2.0 * a[other]
            grad_peak[other] = 2.0 * a[idx]

            # Move mass from high-gradient to low-gradient positions
            if grad_peak.max() > 0:
                top = np.argsort(grad_peak)[::-1][:n // 4]
                bottom = np.argsort(grad_peak)[:n // 4]
                for i in top[:5]:
                    for j in bottom[:5]:
                        if a[i] < 1e-10:
                            continue
                        for frac in [0.001, 0.005, 0.01]:
                            amt = a[i] * frac
                            a_trial = a.copy()
                            a_trial[i] -= amt
                            a_trial[j] += amt
                            a_trial = np.maximum(a_trial, 0)
                            a_trial_s = a_trial.sum()
                            if a_trial_s < 1e-15:
                                continue
                            a_trial = a_trial / a_trial_s
                            score = 2.0 * n * fast_autocorr(a_trial).max()
                            if score < best_score:
                                best_score = score
                                best_a = a_trial.copy()
                                a = a_trial
                                break

    return best_a, best_score

                n_pert = rng.integers(5, 20)
                sources = rng.choice(n, size=n_pert, replace=False)
                targets = rng.choice(n, size=n_pert, replace=False)

                for src_idx, dst_idx in zip(sources, targets):

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.PDLP


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b)[k] = sum_j T[k,j]*b[j] = conv(a_fixed,b)[k]."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0. Returns (b, M_val)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def subgradient_descent_smoothed(a0, time_limit, beta_start=10.0, beta_end=5000.0):
    """
    FISTA on log-sum-exp smoothed objective with adaptive beta.
    Each iteration is O(n log n) via FFT, so we can do many iterations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    z = a.copy()
    n_iters = 50000

    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(t / n_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute subgradient via softmax-weighted average of shift gradients
        b = fast_autocorr(z)
        b_max = b.max()
        b_shifted = beta * (b - b_max)
        # Clip for numerical stability
        b_shifted = np.clip(b_shifted, -500, 0)
        weights = np.exp(b_shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(len(b))
            weights[np.argmax(b)] = 1.0
        else:
            weights = weights / w_sum

        # Gradient computation: for each lag k, gradient contribution is 2*weights[k]*shift_k(z)
        grad = np.zeros(n)
        # Only process lags with significant weight
        significant = weights > 1e-6
        for k_idx in np.where(significant)[0]:
            w = weights[k_idx]
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k_idx - idx
            grad[idx] += 2.0 * w * z[other]
            grad[other] += 2.0 * w * z[idx]

        # Step size: conservative to ensure convergence
        # Lipschitz constant of the smoothed gradient is bounded by 2 * sum_k w_k * (sum of |z|) = 2
        # Use step = 1/4 for safety
        step = 1.0 / 4.0

        a_new = z - step * grad
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            a_new = z.copy()
            a_new_s = 1.0
        a_new = a_new / a_new_s

        # FISTA momentum
        t_plus = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (1.0 + t) / t_plus
        z_new = a_new + theta * (a_new - a)
        z_new = np.maximum(z_new, 0)
        z_new_s = z_new.sum()
        if z_new_s < 1e-15:
            z_new = a_new.copy()
            z_new_s = 1.0
        z_new = z_new / z_new_s

        a = a_new
        z = z_new

        if t % 1000 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"      FISTA t={t} ({elapsed:.0f}s): best={best_score:.7f}, beta={beta:.0f}")

    # Final score
    score = 2.0 * n * fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, seed=123):
    """
    Local refinement via random 2-coordinate swaps with adaptive step sizes.
    Focuses on reducing the maximum convolution peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()

        # Identify the top contributing lag
        peak_lag = np.argmax(b)

        # Gradient at peak: g[i] = 2 * a[peak_lag - i] (for valid indices)
        grad_peak = np.zeros(n)
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        if js <= je:
            idx = np.arange(js, je + 1)
            other = peak_lag - idx
            grad_peak[idx] = 2.0 * a[other]
            grad_peak[other] = 2.0 * a[idx]

        # Strategy: move mass from high-grad to low-grad positions
        if grad_peak.max() > 1e-12:
            # Sort by gradient
            order = np.argsort(grad_peak)
            # Sources: high gradient (contributing to peak)
            # Targets: low gradient (not contributing)
            n_move = max(1, n // 20)
            sources = order[::-1][:n_move]
            targets = order[:n_move]

            improved = False
            for src in sources:
                if a[src] < 1e-12:
                    continue
                for dst in targets:
                    if src == dst:
                        continue
                    for frac in [0.001, 0.003, 0.01, 0.03]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial_s = a_trial.sum()
                        if a_trial_s < 1e-15:
                            continue
                        a_trial = a_trial / a_trial_s
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

            if not improved:
                # Try random swaps
                for _ in range(10):
                    src = rng.integers(0, n)
                    dst = rng.integers(0, n)
                    if src == dst or a[src] < 1e-12:
                        continue
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

            if not improved:
                no_improve += 1
                if no_improve > 300:
                    break
        else:
            no_improve += 1
            if no_improve > 100:
                break

    return best_a, best_score


def kkt_alignment_refine(a0, time_limit):
    """
    Refine using KKT conditions: at optimum, the weighted sum of shifted
    sequences should be constant on the support.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()

        # Active lags: within 2% of max
        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]

        if len(active) == 0:
            break

        # Compute weights (softmax over active lags)
        b_active = b[active]
        temp = max_b * 0.005  # Temperature for weight concentration
        w = np.exp((b_active - b_active.max()) / temp)
        w = w / w.sum()

        # Compute target vector: g(i) = sum_k w_k * a[k-i]
        g = np.zeros(n)
        for wi, ki in zip(w, active):
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = ki - idx
            g[idx] += wi * a[other]
            g[other] += wi * a[idx]

        # At KKT: g(i) should be constant on support
        support = a > 1e-8
        if support.sum() < 2:
            break
        g_target = g[support].mean()

        # Direction: increase a where g > target, decrease where g < target
        # But we need to maintain sum(a) = 1 and a >= 0
        diff = g - g_target
        # Scale step
        step_size = 0.005
        a_trial = a + step_size * diff
        a_trial = np.maximum(a_trial, 0)
        a_trial_s = a_trial.sum()
        if a_trial_s < 1e-15:
            break
        a_trial = a_trial / a_trial_s
        score = 2.0 * n * fast_autocorr(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller step
            step_size *= 0.5
            a_trial = a + step_size * diff
            a_trial = np.maximum(a_trial, 0)
            a_trial_s = a_trial.sum()
            if a_trial_s < 1e-15:
                break
            a_trial = a_trial / a_trial_s
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
            else:
                # No improvement, try opposite direction (might help with non-convexity in numerics)
                a_trial = a - step_size * diff
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s < 1e-15:
                    break
                a_trial = a_trial / a_trial_s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.PDLP


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses LP Nesterov as primary method with
    KKT-guided refinement and multi-length exploration.
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default uniform init: n={n_base}, score={best_score:.7f}")

    # Helper to check updates (no closure - use a list container)
    best_container = [best_seq.copy(), best_score]

    def check_update_func(a, label, container):
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < container[1]:
            container[1] = sc
            container[0] = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LP Nesterov - main workhorse (70% of time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.70, 1150.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t1, solver)
        print(f"  LP Nesterov done: {iters} iters, score={sc_out:.7f}")
        check_update_func(a_out, f"lp_n{n_base}", best_container)

    # Phase 2: KKT alignment refinement (10% of time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: KKT Refinement (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 120.0)
    if t2 > 15:
        a_out, sc_out = kkt_alignment_refine(best_container[0], t2)
        print(f"  KKT refine: score={sc_out:.7f}")
        check_update_func(a_out, "kkt_refine", best_container)

    # Phase 3: Smoothed FISTA (10% of time) - fast iterations
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Smoothed FISTA (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, sc_out = subgradient_descent_smoothed(best_container[0], t3, beta_start=5.0, beta_end=2000.0)
        print(f"  FISTA: score={sc_out:.7f}")
        check_update_func(a_out, "fista", best_container)

    # Phase 4: Multi-length exploration (8% of time)
    for n_try in [n_base + 50, n_base - 50, n_base + 100, n_base - 100, n_base + 200]:
        if n_try < 100:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 60:
            break
        print(f"\n=== Phase 4: Explore n={n_try} (budget={remaining:.0f}s) ===")
        t4 = min(remaining * 0.12, 80.0)
        if t4 < 20:
            continue
        init_try = interpolate_sequence(best_container[0], n_try)
        a_out, sc_out, iters = nesterov_lp_optimize(init_try, t4, solver, verbose=False)
        print(f"  LP at n={n_try}: {iters} iters, score={sc_out:.7f}")
        check_update_func(a_out, f"lp_n{n_try}", best_container)

    # Phase 5: Coordinate descent polish (5% of time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Coord Descent Polish (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 80.0)
    if t5 > 10:
        a_out, sc_out = coordinate_descent_refine(best_container[0], t5, seed=789)
        print(f"  CD polish: score={sc_out:.7f}")
        check_update_func(a_out, "cd_polish", best_container)

    # Phase 6: Final LP Nesterov round on current best (remaining time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final LP Nesterov (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.70, 120.0)
    if t6 > 20:
        a_out, sc_out, iters = nesterov_lp_optimize(best_container[0], t6, solver, verbose=False)
        print(f"  Final LP: {iters} iters, score={sc_out:.7f}")
        check_update_func(a_out, "final_lp", best_container)

    # Phase 7: Final KKT polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t7 = min(remaining * 0.6, 40.0)
        if t7 > 5:
            a_out, sc_out = kkt_alignment_refine(best_container[0], t7)
            check_update_func(a_out, "final_kkt", best_container)

    # Final verification
    if best_container[0] is not None:
        fs = evaluate_safe(best_container[0])
        if np.isfinite(fs):
            best_container[1] = fs
        print(f"\nFINAL SCORE: {best_container[1]:.7f} (n={len(best_container[0])})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_container[0]]
