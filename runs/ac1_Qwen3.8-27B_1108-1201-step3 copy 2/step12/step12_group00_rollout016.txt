        # = sum_i a[i] * a[k-i] * 1(0<=i<n) * 1(0<=k-i<n)
        
        # d/da[j] conv(a,a)[k] = a[k-j] * 1(0<=k-j<n) + a[j-k] * 1(0<=j-k<n)
        #                         = a[k-j] * 1(0<=k-j<n) + a[|j-k|] * 1(0<=|j-k|<n) [for k!=j]
        
        # Actually: d/da[j] of sum_i a[i]*a[k-i] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # For j=k: contribution is a[0] (from i=j term: d/da[j] a[j]*a[0] = a[0])
        #          + a[0] (from i=k-j=j term: d/da[j] a[j]*a[0] = a[0])
        # Wait, I'm confusing myself. Let me use the direct formula.
        
        # Let's just compute it via FFT for efficiency.
        # grad[j] = 2 * sum_k y[k] * (a[k-j] if 0<=k-j<n else 0 + a[j-k] if 0<=j-k<n else 0)
        #         = 2 * (convolve(y, a)[j + n - 1] + convolve(y, a_rev)[j])  -- not quite
        
        # Simpler: grad[j] = 2 * sum_{k} y[k] * a[k-j] * 1[0<=k-j<n]
        #              + 2 * sum_{k} y[k] * a[j-k] * 1[0<=j-k<n]
        # Note: the second sum with substitution k' = j-k gives:
        #              2 * sum_{k'} y[j-k'] * a[k'] * 1[0<=k'<n]
        # And the first sum with substitution k' = k-j gives:
        #              2 * sum_{k'} y[k'+j] * a[k'] * 1[0<=k'<n]
        # So: grad[j] = 2 * sum_{k'=0}^{n-1} a[k'] * (y[k'+j] + y[j-k'] if valid)
        
        # This is: grad = 2 * (a * y_full) where y_full is the "circular-like" sum
        # Actually it's: grad[j] = 2 * sum_i a[i] * y[j+i] (first term, j+i < num_lags)
        #                            + 2 * sum_i a[i] * y[i-j] (second term, i-j >= 0)
        
        # Let me just compute it directly with the FFT trick:
        # Define y_padded = y (length 2n-1), a (length n)
        # conv(y_padded, a) has length 3n-2
        # (conv(y,a))[k] = sum_i y[k-i] * a[i] for valid i
        # = sum_{i=0}^{n-1} y[k-i] * a[i] (where y is 0 outside [0, 2n-2])
        
        # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * (y[j+i] + y[|i-j|] if i>=j... )
        # Hmm this is getting complicated. Let me just use the direct matrix approach for small n.
        
        # Actually, the simplest way: 
        # grad[j] = 2 * (T^T y)[j] where T is the conv matrix
        # (T^T y)[j] = sum_k T[k,j] * y[k] = sum_k a[k-j] * y[k] (where 0<=k-j<n)
        # = sum_{k=j}^{j+n-1} a[k-j] * y[k] (with y[k]=0 for k<0 or k>=2n-1)
        # = sum_{i=0}^{n-1} a[i] * y[i+j] (with y=0 outside valid range)
        
        # So (T^T y)[j] = sum_{i=0}^{n-1} a[i] * y[i+j]
        # This is just the cross-correlation of a and y, shifted!
        # = (a * y)[j + n - 1] in the standard cross-correlation sense
        # Or equivalently: if we define y_ext as y padded to length 3n-2:
        # (T^T y)[j] = sum_i a[i] * y[i+j]
        
        # We can compute this with FFT:
        # Let A = fft(a, size=3n-2), Y = fft(y, size=3n-2)
        # Then corr = ifft(A * conj(Y)) gives cross-correlation
        # Actually, cross-correlation: (a ⋆ y)[k] = sum_i a[i+k] * conj(y[i])
        # Hmm, let me just use direct convolution with a flipped signal.
        
        # (T^T y)[j] = sum_i a[i] * y[i+j] = (a reversed * y)[j + n - 1]
        # where * denotes standard convolution
        
        # a_rev = a[::-1], length n
        # conv(a_rev, y) has length n + 2n-1 - 1 = 3n-2
        # conv(a_rev, y)[j + n - 1] = sum_i a_rev[i] * y[j+n-1-i] = sum_i a[n-1-i] * y[j+n-1-i]
        # Let k = n-1-i: sum_k a[k] * y[j+k]  ✓
        
        # So: (T^T y) = conv(a[::-1], y)[n-1 : n-1+n]
        
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 1).bit_length()
        A_r = rfft(a_rev, fft_size)
        Y = rfft(y, fft_size)
        conv_ay = irfft(A_r * Y, fft_size)
        Ty = conv_ay[n-1:n-1+n]
        
        grad = 2.0 * Ty
        
        # Projected gradient step on the simplex-like constraint
        # We normalize to sum = norm after each step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_a
        
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualNR t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def multi_peak_coordinate_descent(a0, time_limit, seed=42):
    """
    Targeted coordinate descent that attacks multiple peaks simultaneously.
    For each top peak, find the pair (source, dest) that most reduces it,
    while tracking the effect on all other top peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    n_top = 8  # number of top peaks to track
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:n_top]
        current_max = b[top_k[0]]
        
        if current_max <= best_score + 1e-12:
            pass  # already at best
        
        improved = False
        
        # For each top peak, try to reduce it
        for k in top_k:
            # Find which pairs (i, j) with i+j = k contribute most
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if len(idx) == 0 or contrib.max() < 1e-15:
                continue
            
            # Sort by contribution descending
            order = np.argsort(contrib)[::-1]
            
            # Try moving mass from the top contributors
            for oi in order[:5]:
                src = idx[oi]
                src_val = a[src]
                if src_val < norm * 1e-8:
                    continue
                
                # Find best destination: position that minimizes impact on all top peaks
                best_dst = -1
                best_gain = 0
                
                # Sample candidate destinations (for efficiency)
                if n > 200:
                    candidates = rng.choice(n, size=min(n // 2, 500), replace=False)
                else:
                    candidates = np.arange(n)
                
                for dst in candidates:
                    if dst == src:
                        continue
                    # Estimate effect on peak k
                    # Reducing a[src] by amt reduces peak k by ~2*a[k-src]*amt (if k-src valid)
                    # Increasing a[dst] by amt increases peak k by ~2*a[k-dst]*amt (if k-dst valid)
                    amt = src_val * 0.02
                    net_k = 0.0
                    # Effect on peak k:
                    # a[src] decreases by amt: reduces conv[k] by a[k-src]*amt + (a[dst... no
                    # conv[k] = sum_i a[i]*a[k-i]
                    # After: a'[src] = a[src]-amt, a'[dst] = a[dst]+amt
                    # Delta conv[k] = -amt*a[k-src] (if valid) - amt*a[src-k] (if valid)
                    #                + amt*a[k-dst] (if valid) + amt*a[dst-k] (if valid)
                    dc = 0.0
                    if 0 <= k - src < n:
                        dc -= a[k - src] * amt
                    if 0 <= src - k < n:
                        dc -= a[src - k] * amt
                    if 0 <= k - dst < n:
                        dc += a[k - dst] * amt
                    if 0 <= dst - k < n:
                        dc += a[dst - k] * amt
                    net_k = dc
                    
                    if net_k < best_gain * -1:  # i.e., dc < -best_gain
                        best_gain = -net_k
                        best_dst = dst
                
                if best_dst >= 0 and best_gain > 1e-15:
                    # Try the move
                    amt = src_val * 0.02
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
            if improved:
                break
        
        if not improved:
            # Try random perturbations
            stall += 1
            n_perturb = min(50, stall * 5)
            for _ in range(n_perturb):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < norm * 1e-8:
                    continue
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if not improved and stall > 500:
                break
    
    return best_a, best_score


def symmetric_spectral_method(a0, time_limit, verbose=True):
    """
    Exploit the spectral decomposition. If a is symmetric (a[i] = a[n-1-i]),
    then conv(a,a) is also symmetric and the peak is at the center = sum a[i]^2.
    
    For asymmetric a, we can think of a = s + d where s is the symmetric part
    and d is the antisymmetric part. The center peak of conv depends on the
    overlap of a with its reverse.
    
    Strategy: parameterize a as a function that has controlled overlap with its reverse.
    Use gradient descent on the parameters.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Use projected subgradient with adaptive step size
    step = 1.0 / n
    decay = 0.999
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]
        
        # Subgradient of max_k conv(a,a)_k
        # d/da[j] conv(a,a)_{k*} = a[k*-j] (if valid) + a[j-k*] (if valid)
        grad = np.zeros(n)
        # conv(k*) = sum_i a[i] * a[k*-i]
        # d/da[j] = a[k*-j] * 1[0<=k*-j<n] + a[j-k*] * 1[0<=j-k*<n]
        # (the first term is from a[i] when i=j, the second from a[k*-i] when k*-i=j)
        
        # More carefully:
        # conv(k*) = sum_{i: 0<=i<n, 0<=k*-i<n} a[i] * a[k*-i]
        # d/da[j]:
        #   term 1: a[j] * a[k*-j] if 0<=k*-j<n
        #   term 2: a[k*-j] * a[j] if 0<=j<n (always true) and 0<=k*-j<n wait...
        #   
        # Let me use: conv(k) = sum_i a[i] a[k-i]
        # d/da[j] = a[k-j] (from the a[i] factor, i=j) + a[j-k] (from the a[k-i] factor, k-i=j, i=k-j)
        # Both terms are valid if 0<=k-j<n and 0<=j-k<n respectively
        # But k-j and j-k can't both be >= 0 unless j=k
        # If j < k: only first term valid (k-j > 0), need k-j < n
        # If j > k: only second term valid (j-k > 0), need j-k < n  
        # If j = k: both terms = a[0], so contribution = 2*a[0]
        
        vec = np.zeros(n)
        # First term: a[k*-j] for j in [max(0,k*-n+1), min(n-1,k*)]
        j1_start = max(0, k_star - n + 1)
        j1_end = min(n - 1, k_star)
        if j1_start <= j1_end:
            vec[j1_start:j1_end+1] = a[k_star - j1_start:k_star - j1_end + 1][::-1]
        
        # Second term: a[j-k*] for j in [max(0,k*), min(n-1,k*+n-1)]
        j2_start = max(0, k_star)
        j2_end = min(n - 1, k_star + n - 1)
        if j2_start <= j2_end:
            vec[j2_start:j2_end+1] += a[j2_start - k_star:j2_end - k_star + 1]
        
        grad = 2.0 * vec
        
        # Projected gradient step with adaptive step size
        for factor in [1.0, 0.5, 0.25, 0.1]:
            s = step * factor
            a_new = a - s * grad
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                continue
            a_new = a_new * norm / s_a
            sc = fast_autocorr(a_new).max()
            if sc < M_curr:
                a = a_new
                break
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        step *= decay
        if step < 1e-10:
            step = 1.0 / n  # reset
            decay = 0.999
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpecM t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)[:-1]
    new_x = np.linspace(0.0, 1.0, new_n + 1)[:-1]
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        # Asymmetric ramp
        a = x ** 0.5 * (1 - x) ** 0.3
    elif choice == 6:
        # Multi-frequency asymmetric
        a = 0.4 + 0.3 * np.cos(2*np.pi*3*x) * x + 0.2 * np.sin(2*np.pi*7*x) * (1-x)
    else:
        # Modulated Gaussian
        a = np.exp(-3.0 * x**2) * (1 + 0.3*np.cos(2*np.pi*4*x))
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Dual no-regret method ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Dual No-Regret n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = dual_no_regret(a_norm, t2, verbose=True)
        try_update(a_out, f"dual_nr_n{n_best}")

    # === Phase 3: Longer length n2 ===
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n2}")

    # === Phase 4: Even longer n3 ===
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp_n{n3}")

    # === Phase 5: Multi-peak coordinate descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-peak CD n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_coordinate_descent(a_norm, t5, seed=777)
        try_update(a_out, "cd_peak")

    # === Phase 6: Spectral gradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Spectral method n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = symmetric_spectral_method(a_norm, t6, verbose=True)
        try_update(a_out, "spectral")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP polish n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 120.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
        try_update(a_out, "final_lp")

    # === Phase 8: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_best = len(best_seq)
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 5:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = multi_peak_coordinate_descent(a_norm, t8, seed=999)
            try_update(a_out, "final_cd")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        y_score_curr = fast_autocorr(y).max()
        best_y_new = None
        best_y_score = y_score_curr

        for alpha_try in [2.0 / (t + 2.0), 3.0 / (t + 3.0), 2.0 / (t + 1.0), 1.0 / (t + 1.0)]:

        else:

            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

        # Nesterov momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)

def dual_no_regret(a0, time_limit, verbose=True):
    """
    No-regret algorithm: maintain dual distribution y over lags,
    update a via projected gradient on the weighted quadratic.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_norm(a, n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    num_lags = 2 * n - 1
    y = np.ones(num_lags) / num_lags
    eta = 2.0 / n
    best_eta = eta

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)

        # Update dual: multiplicative weights toward high lags
        max_b = b.max()
        y_new = y * np.exp(eta * (b - max_b))
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = np.ones(num_lags) / num_lags
        else:
            y_new = y_new / y_sum
        y = y_new

        # Compute gradient of sum_k y[k] * conv(a,a)[k]
        # (T^T y)[j] = sum_i a[i] * y[i+j] (valid range)
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 1).bit_length()
        A_r = rfft(a_rev, fft_size)
        Y = rfft(y, fft_size)
        conv_ay = irfft(A_r * Y, fft_size)
        Ty = conv_ay[n - 1:n - 1 + n]
        grad = 2.0 * Ty

        # Backtracking line search
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            s_step = eta * factor
            a_cand = a - s_step * grad
            a_cand = np.maximum(a_cand, 0.0)
            s_a = a_cand.sum()
            if s_a < 1e-15:
                continue
            a_cand = a_cand * norm / s_a
            sc = fast_autocorr(a_cand).max()
            if sc < b.max() - 1e-14:
                a = a_cand
                improved = True
                if factor < 1.0:
                    best_eta = eta * factor
                break

        if not improved:
            eta = max(eta * 0.98, 1e-8)
            if eta < 1.0 / (n * 1000):
                eta = 2.0 / n
                best_eta = eta

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualNR t={t} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score


def multi_peak_cd(a0, time_limit, seed=42):
    """Multi-peak targeted coordinate descent with smart destination selection."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_norm(a, n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    n_top = 6

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:n_top]

        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue

            order = np.argsort(contrib)[::-1]

            for oi in order[:4]:
                src = idx[oi]
                src_val = a[src]
                if src_val < norm * 1e-8:
                    continue

                # Compute gradient effect for candidate destinations
                # delta_conv[k'] for each top peak k' when moving amt from src to dst
                best_dst = -1
                best_metric = 0.0

                n_cand = min(n, 800)
                if n_cand < n:
                    candidates = rng.choice(n, size=n_cand, replace=False)
                else:
                    candidates = np.arange(n)

                for dst in candidates:
                    if dst == src:
                        continue
                    # Effect on the primary peak k:
                    dc_k = 0.0
                    if 0 <= k - src < n:
                        dc_k -= a[k - src]
                    if 0 <= src - k < n:
                        dc_k -= a[src - k]
                    if 0 <= k - dst < n:
                        dc_k += a[k - dst]
                    if 0 <= dst - k < n:
                        dc_k += a[dst - k]

                    # We want dc_k < 0 (reduce the peak)
                    # Also want to not increase other peaks too much
                    if dc_k < 0:
                        # Metric: reduction in peak k, penalize increases elsewhere
                        other_increase = 0.0
                        for k2 in top_k:
                            if k2 == k:
                                continue
                            dc_k2 = 0.0
                            if 0 <= k2 - src < n:
                                dc_k2 -= a[k2 - src]
                            if 0 <= src - k2 < n:
                                dc_k2 -= a[src - k2]
                            if 0 <= k2 - dst < n:
                                dc_k2 += a[k2 - dst]
                            if 0 <= dst - k2 < n:
                                dc_k2 += a[dst - k2]
                            if dc_k2 > 0:
                                other_increase += dc_k2

                        metric = -dc_k - 0.3 * other_increase
                        if metric > best_metric:
                            best_metric = metric
                            best_dst = dst

                if best_dst >= 0 and best_metric > 0:
                    amt = src_val * 0.02
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = normalize_to_norm(a_trial, n)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbations
            n_pert = min(30, stall * 3)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < norm * 1e-7:
                    continue
                frac = rng.uniform(0.003, 0.04)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                a_trial = normalize_to_norm(a_trial, n)
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if not improved and stall > 400:
                break

    return best_a, best_score


def subgrad_nesterov_fast(a0, time_limit, verbose=False):
    """Fast subgradient Nesterov with Frank-Wolfe vertices."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_norm(a, n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        k_star = int(np.argmax(b))

        # Compute subgradient
        vec = np.zeros(n)
        j1s = max(0, k_star - n + 1)
        j1e = min(n - 1, k_star)
        if j1s <= j1e:
            vec[j1s:j1e + 1] = y[k_star - j1e:k_star - j1s + 1][::-1]
        j2s = max(0, k_star)
        j2e = min(n - 1, k_star + n - 1)
        if j2s <= j2e:
            vec[j2s:j2e + 1] += y[j2s - k_star:j2e - k_star + 1]
        grad = 2.0 * vec

        # FW vertex: e_{argmin(grad)} scaled to norm
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to_norm(y_new, n)

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_norm(a_new, n)

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
