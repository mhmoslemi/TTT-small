def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)

def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu - (cv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = np.max(rho[cond])
    theta = (cv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):

    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):

    return T

    b_val = np.asarray(b_var.value, dtype=np.float64)

def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it > 0 and it % 2000 == 0 and no_improve > 3000:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:

            else:

        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new

            a_prev = a_new.copy()

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=80.0, T_end=0.02, n_stages=25):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 50 if si < n_stages // 3 else (30 if si < 2 * n_stages // 3 else 20)
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-24,
                         'gtol': 1e-20, 'maxfun': iters * 400,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a, best_score


def peak_targeted_refine(a0, time_limit, max_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.1
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step > 0 and step - last_improve > 8000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.003, 0.2)
        else:
            eta *= 0.8
            if eta < 1e-12:
                break

    return best_a, best_score


def subgradient_line_search(a0, time_limit, max_steps=1000000):
    """Direct subgradient method with backtracking line search on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.5 / n
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step > 0 and no_improve > 50000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_threshold = max_b * 0.95
        topk = np.where(b >= peak_threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Subgradient
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(topk)

        # Projected gradient step with line search
        step_size = eta
        found = False
        for _ in range(20):
            candidate = a - step_size * grad
            candidate = project_simplex(candidate)
            sc = score_normalized(candidate)
            if sc < best_score or sc < score_normalized(a):
                a = candidate
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                found = True
                eta = min(eta * 1.01, 2.0 / n)
                break
            step_size *= 0.5

        if not found:
            no_improve += 1
            eta *= 0.9

    return best_a, best_score


def coordinate_descent(a0, time_limit, max_rounds=5000):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        if rnd > 0 and no_improve > 500:
            break

        # Random coordinate
        i = rng.integers(0, n)

        # Try small perturbations in coordinate i
        # Keep sum = 1 by distributing change to a random other coordinate
        j = rng.integers(0, n)
        if j == i:
            j = (j + 1) % n

        base_score = score_normalized(a)
        improved = False

        for delta in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001, 0.02, -0.02]:
            a[i] += delta
            a[j] -= delta
            if a[i] < 0 or a[j] < 0:
                a[i] -= delta
                a[j] += delta
                continue
            sc = score_normalized(a)
            if sc < base_score:
                base_score = sc
                improved = True
                break
            a[i] -= delta
            a[j] += delta

        if improved:
            if base_score < best_score:
                best_score = base_score
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if (rnd + 1) % 1000 == 0:
            print(f"    CoordDesc round={rnd+1}: score={best_score:.10f}")

    return best_a, best_score


def nesterov_subgradient(a0, time_limit, max_steps=500000):
    """Nesterov-accelerated subgradient in z-space."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    z_prev = z_from_a(a_prev).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 1.0 / np.sqrt(n)
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step > 0 and no_improve > 30000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_threshold = max_b * 0.97
        topk = np.where(b >= peak_threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_prev)
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            a_prev = a
            z_prev = z
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            eta = max(eta * 0.9995, 0.01 / np.sqrt(n))
        else:
            no_improve += 1
            if sc > score_normalized(a) * 1.0005:
                t_nest = 1.0
                eta *= 0.95
            else:
                a_prev = a
                z_prev = z
                a = a_new
                z = z_new
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            eta = max(eta * 0.999, 0.005 / np.sqrt(n))

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.10f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(arr)})")

    # === Phase 1: LSE L-BFGS rapid convergence (150s) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 1: LSE L-BFGS at n={best_n} (score={best_score:.10f}) ===")
    t1 = min(150.0, remaining * 0.12)
    if t1 > 10:
        a_out = lbgfs_logsumexp(best_seq, t1, T_start=60.0, T_end=0.05, n_stages=20)
        try_update(a_out, "lse_p1")
    print(f"  Phase 1 done: best_score={best_score:.10f}")

    # === Phase 2: Nesterov subgradient pre-optimization (100s) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 2: Nesterov subgradient (score={best_score:.10f}) ===")
    t2 = min(100.0, remaining * 0.08)
    if t2 > 10:
        a_out, _ = nesterov_subgradient(best_seq, t2, max_steps=200000)
        try_update(a_out, "nesterov_sg_p2")
    print(f"  Phase 2 done: best_score={best_score:.10f}")

    # === Phase 3: LP-based Nesterov (main workhorse, 600s) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 3: Nesterov LP main at n={best_n} (score={best_score:.10f}) ===")
    t3 = min(600.0, remaining * 0.4)
    if t3 > 30:
        a_out, _ = nesterov_lp_adaptive(best_seq, t3, solver=solver)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 3 done: best_score={best_score:.10f}")

    # === Phase 4: Subgradient line search (200s) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 4: Subgradient line search (score={best_score:.10f}) ===")
    t4 = min(200.0, remaining * 0.15)
    if t4 > 15:
        a_out, _ = subgradient_line_search(best_seq, t4, max_steps=2000000)
        try_update(a_out, "sg_linsearch")
    print(f"  Phase 4 done: best_score={best_score:.10f}")

    # === Phase 5: Peak-targeted refinement (150s) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 5: Peak-targeted (score={best_score:.10f}) ===")
    t5 = min(150.0, remaining * 0.12)
    if t5 > 10:
        a_pt, _ = peak_targeted_refine(best_seq, t5, max_steps=50000)
        try_update(a_pt, "peak_targeted")
    print(f"  Phase 5 done: best_score={best_score:.10f}")

    # === Phase 6: LP round 2 (200s) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 6: Nesterov LP round 2 (score={best_score:.10f}) ===")
    t6 = min(200.0, remaining * 0.15)
    if t6 > 15:
        a_out2, _ = nesterov_lp_adaptive(best_seq, t6, solver=solver)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 6 done: best_score={best_score:.10f}")

    # === Phase 7: Coordinate descent polish (80s) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 7: Coordinate descent (score={best_score:.10f}) ===")
    t7 = min(80.0, remaining * 0.08)
    if t7 > 10:
        a_cd, _ = coordinate_descent(best_seq, t7, max_rounds=20000)
        try_update(a_cd, "coord_desc")
    print(f"  Phase 7 done: best_score={best_score:.10f}")

    # === Phase 8: LSE final polish (60s) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 8: LSE final polish (score={best_score:.10f}) ===")
    t8 = min(60.0, remaining * 0.06)
    if t8 > 10:
        a_lse2 = lbgfs_logsumexp(best_seq, t8, T_start=15.0, T_end=0.01, n_stages=12)
        try_update(a_lse2, "lse_final")
    print(f"  Phase 8 done: best_score={best_score:.10f}")

    # === Phase 9: Length exploration ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 9: Length exploration (score={best_score:.10f}) ===")
    if remaining > 60:
        n_candidates = [best_n - 100, best_n - 200, best_n + 100, best_n + 200,
                        best_n + 400, best_n + 800]
        # Filter valid candidates
        n_candidates = [n for n in n_candidates if 200 <= n <= 5000]
        # Remove duplicates
        seen = set()
        n_candidates = [n for n in n_candidates if n not in seen and not seen.add(n)]

        for n_try in n_candidates:
            remaining = _global_deadline - time.monotonic()
            if remaining < 30:
                break
            t_try = min(50.0, remaining * 0.4)
            if t_try < 10:
                break
            print(f"  Trying n={n_try}, budget={t_try:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick LSE + LP
            a_tmp = lbgfs_logsumexp(init_try, t_try * 0.3, T_start=40.0, T_end=0.1, n_stages=10)
            try_update(a_tmp, f"len_{n_try}_lse")
            remaining = _global_deadline - time.monotonic()
            if remaining > 15:
                a_tmp2, _ = nesterov_lp_adaptive(a_tmp, remaining * 0.7, solver=solver, max_iters=50000)
                try_update(a_tmp2, f"len_{n_try}_lp")

    # === Phase 10: Final combined polish ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 10: Final polish (score={best_score:.10f}) ===")
    if remaining > 30:
        t_final = remaining * 0.6
        a_f1, _ = subgradient_line_search(best_seq, t_final * 0.4, max_steps=500000)
        try_update(a_f1, "final_sg")
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            a_f2, _ = nesterov_lp_adaptive(best_seq, remaining * 0.7, solver=solver, max_iters=50000)
            try_update(a_f2, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
