import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        a = a_new
        y = y_new
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
    
    return best_a, best_score

def lse_objective_grad(z_in, T, n, m_top=None):
    """Compute LSE objective and gradient. If m_top is set, only use top m peaks."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    
    if m_top is not None and m_top < len(b):
        # Get indices of top m
        top_idx = np.argsort(b)[-m_top:]
        b_sel = b[top_idx]
    else:
        b_sel = b
    
    max_b = b_sel.max()
    if max_b > 700 / T:
        max_b = 700 / T
    exp_Tb = np.exp(T * (b_sel - max_b))
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_b + np.log(exp_Tb.sum()) / T
    obj = (2.0 * n) * log_sum_exp
    
    # Gradient
    grad_a = np.zeros(n)
    if m_top is not None and m_top < len(b):
        for i_idx in range(len(top_idx)):
            k = top_idx[i_idx]
            if w[i_idx] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[i_idx] * a[k - idx]
    else:
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lse_polish(a0, time_limit, T_start=20.0, T_min=0.3, m_top=50, n_stages=50, iters_per_stage=50):
    """LSE polish with top-m peak selection and aggressive cooling."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    T = T_start
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n, m_top)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n, m_top)
            return grad
        
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            
            # Adaptive cooling
            T = max(T_min, T * 0.72)
            
            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.3f} m={m_top}: score={sc:.10f} best={best_score:.10f}")
                
        except Exception:
            T = max(T_min, T * 0.8)
    
    return best_a, best_score

def peak_equalize(a0, time_limit, m_top=30):
    """Try to equalize the top m peaks using direct optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a, score_normalized(a)
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Iteratively: find top m peaks, compute gradient to push them down, update a
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        top_idx = np.argsort(b)[-m_top:]
        top_vals = b[top_idx]
        M = top_vals.max()
        
        # Gradient of max of top m peaks (weighted by how close to M)
        weights = np.exp((top_vals - M) * 50)  # Sharp weighting
        weights /= weights.sum()
        
        g = np.zeros(n)
        for i_idx in range(m_top):
            k = top_idx[i_idx]
            if weights[i_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i_idx] * a[k - idx]
        
        # Project gradient to sum=0 (tangent to simplex)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Step size: use the gap to the lower bound
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        gap = 2.0 * n * M - lower_bound
        eta = gap / (g_norm * g_norm + 1e-30)
        eta = min(eta, 1.0)
        
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Accept if improved, otherwise halve step
        if sc >= 2.0 * n * M:
            eta /= 2
            a_new = a - eta * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            if score_normalized(a_new) < 2.0 * n * M:
                a = a_new
            else:
                # Don't update, just reduce step
                pass
        else:
            a = a_new
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response for initial convergence
2. Top-m peak LogSumExp L-BFGS polish with aggressive multi-stage cooling
3. Direct peak-equalization subgradient for final refinement
4. Multi-length exploration with interpolation
Key: very low LSE temperatures (T<0.5) with many L-BFGS iterations,
and explicit top-peak tracking to sharpen the max approximation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        a = a_new
        y = y_new

            best_a = a.copy()
        sc_y = score_normalized(y)

            best_a = y.copy()
        if (it + 1) % 1000 == 0:

            if elapsed - last_print > 20:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.10f}")
                last_print = elapsed

    return best_a, best_score


def lse_obj_grad(z_in, T, n, m_top):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    if m_top is not None and m_top < len(b):
        top_idx = np.argsort(b)[-m_top:]
        b_sel = b[top_idx]
        use_top = True
    else:
        b_sel = b
        top_idx = None
        use_top = False
    max_b = b_sel.max()
    scaled = T * (b_sel - max_b)
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    lse_val = max_b + np.log(exp_s.sum()) / T
    obj = 2.0 * n * lse_val
    grad_a = np.zeros(n)
    if use_top:
        for ii in range(len(top_idx)):
            k = int(top_idx[ii])
            if w[ii] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[ii] * a[k - idx]
    else:
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def make_lse_func(T, n, m_top):
    def func(z_val):
        val, _ = lse_obj_grad(z_val, T, n, m_top)
        return val
    return func


def make_lse_jac(T, n, m_top):
    def jac(z_val):
        _, g = lse_obj_grad(z_val, T, n, m_top)
        return g
    return jac


def lse_polish(a0, time_limit, T_start, T_min, m_top, n_stages, iters_per_stage):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    T = T_start

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        func = make_lse_func(T, n, m_top)
        jac = make_lse_jac(T, n, m_top)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            T = max(T_min, T * 0.7)
            if (stage + 1) % 5 == 0:
                print(f"    LSE st={stage+1}/{n_stages} T={T:.3f} m={m_top}: cur={sc:.10f} best={best_score:.10f}")
        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_equalize(a0, time_limit, m_top):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a, score_normalized(a)
    a = a / s
    best_a = a.copy()

    t0 = time.monotonic()

    for it in range(200000):

            break
        b = fast_convolve(a)
        top_idx = np.argsort(b)[-m_top:]
        top_vals = b[top_idx]
        M = top_vals.max()
        sharp = 100.0
        weights = np.exp(sharp * (top_vals - M))
        sw = weights.sum()
        if sw < 1e-300:
            break
        weights = weights / sw
        g = np.zeros(n)
        for ii in range(m_top):
            k = int(top_idx[ii])
            if weights[ii] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[ii] * a[k - idx]
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        gap = 2.0 * n * M - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (g_norm * g_norm + 1e-30)
        eta = min(eta, 0.5)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:

        sc = score_normalized(a_new)
        if sc < best_score:

        if sc >= 2.0 * n * M:
            for _ in range(8):

                    break
                a_test = a_test / st
                if score_normalized(a_test) < 2.0 * n * M:
                    a = a_test
                    break
            else:
                if eta < 1e-15:
                    break
        else:
            a = a_new
        if (it + 1) % 5000 == 0:
            print(f"    PE it={it+1}: cur={2.0*n*M:.10f} best={best_score:.10f} eta={eta:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def try_update(state, a, label):
    if a is None:
        return
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a.copy()
        remain = state['deadline'] - time.monotonic()
        print(f"  ** NEW BEST [{label}]: {sc:.10f} (n={len(a)}, rem={remain:.0f}s)")

            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)

    return [float(x) for x in state['best_seq']]
