def fast_autocorr(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return irfft(A * A, fft_size)[:size]


def smooth_max_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*c_k)) w.r.t. a.
    where c = convolve(a, a).
    Returns gradient of length n.
    """
    n = len(a)
    c = fast_autocorr(a)
    # Compute softmax weights
    beta_c = beta * c
    beta_c -= beta_c.max()
    w = np.exp(beta_c)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    # Compute g_k = 2 * sum_m a[m] * w[m+k] for k = 0..n-1
    # = 2 * (w cross-correlated with a) [first n elements]
    size = 2 * n - 1
    fft_size = 1 << (2 * size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g_full = irfft(W * np.conj(A), fft_size)
    grad = 2.0 * g_full[:n]
    return grad, w


def smooth_max_value(c, beta):
    """Compute (1/beta)*log(sum exp(beta*c_k))."""
    beta_c = beta * c
    beta_c -= beta_c.max()
    return (beta_c.max() + np.log(np.exp(beta_c).sum())) / beta


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

- `c[n-1] = sum_j a[j]*a[n-1-j]` (the "central" lag)
- By Cauchy-Schwarz: `c[n-1] >= (sum_j sqrt(a[j]*a[n-1-j]))^2 / n`... not helpful.

Actually, a useful lower bound: consider the sum of c_k for k in a certain range. For example, sum of all c_k = 1. And c_k >= 0. So max(c_k) >= 1/(2n-1).

def evaluate_safe(a):

        sc = evaluate_sequence(a_list)

def compute_smooth_max_and_grad(a, beta):
    """Compute smooth max value and gradient."""
    n = len(a)
    c = fast_autocorr(a)
    beta_c = beta * c - (beta * c).max()
    w = np.exp(beta_c)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
        smooth_val = c.max()
    else:
        w /= w_sum
        smooth_val = (beta * c).max() / beta + np.log(w_sum) / beta
    # Gradient: g_i = 2 * sum_m a[m] * w[m+i]
    size = 2 * n - 1

def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new


def peak_shaving(a0, time_limit, seed=42):
    """Targeted peak shaving: identify peaks and try to reduce them."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        max_c = c.max()
        top_k = np.argsort(c)[::-1][:8]
        improved = False
        for k in top_k:
            # c[k] = sum_j a[j]*a[k-j]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            top_contrib = np.argsort(contrib)[::-1][:3]
            for ci in top_contrib:
                i1 = idx[ci]
                i2 = k - i1
                # Try reducing a[i1] and/or a[i2] by moving mass elsewhere
                for fraction in [0.01, 0.02, 0.05, 0.1]:
                    amt = min(a[i1], a[i2] if i1 != i2 else a[i1]) * fraction
                    if amt < 1e-15:
                        continue
                    # Find a good target: position that minimizes contribution to top peaks
                    a_trial = a.copy()
                    a_trial[i1] -= amt
                    if i1 != i2:
                        a_trial[i2] -= amt * 0.5
                    # Distribute to a random set of positions
                    targets = rng.choice(n, size=3, replace=False)
                    for t in targets:
                        a_trial[t] += amt / 3.0
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                n_pert = rng.integers(2, 8)
                srcs = rng.choice(n, size=n_pert, replace=False)
                dsts = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for s_idx, d_idx in zip(srcs, dsts):
                    frac = rng.uniform(0.01, 0.05)
                    amt = a_trial[s_idx] * frac
                    a_trial[s_idx] -= amt
                    a_trial[d_idx] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s > 1e-15:
                    a_trial = a_trial / a_trial_s
                    sc = 2.0 * n * fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a_trial.copy()
                        stall = 0
            if stall > 500:
                break
    return best_a, best_score


def qp_refinement(a0, time_limit):
    """QP active-set refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for rnd in range(10):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_autocorr(a)
        max_c = c.max()
        active = np.where(c >= max_c * 0.95)[0]
        if len(active) > 20:
            active = np.argsort(c[active])[::-1][:20] + np.where(c >= max_c * 0.95)[0][:20]
            active = np.unique(active[:20])
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k - idx
            q = cp.sum(a_var[idx] * a_var[other])
            constraints.append(q <= M_var)
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, max_iter=5000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            print(f"      QP round {rnd}: {best_score:.6f}")
        else:
            break
    return best_a, best_score
