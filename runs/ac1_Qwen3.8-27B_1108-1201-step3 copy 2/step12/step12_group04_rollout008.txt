import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def subgradient(a, active_set):
    """Compute subgradient of max_k b[k] at point a, given active set."""
    n = len(a)
    g = np.zeros(n)
    a_rev = a[::-1]
    for k in active_set:
        # g[j] += 2 * a[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += 2 * a[k - j]
    return g

    best_score = fast_autocorr(a).max()

    f_star = 1.0 / (2 * n - 1)  # theoretical lower bound
    gap = best_score - f_star
    
    # Averaging
    avg_a = np.zeros(n)
    n_avg = 0
    start_avg = 50  # start averaging after 50 iterations
    restart_count = 0

        if time.monotonic() - t0 > time_limit:

        b = fast_autocorr(a)
        bmax = b.max()
        
        if bmax < best_score:
            best_score = bmax
            best_a = a.copy()

        active = np.where(b >= threshold)[0]

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgradient(a, active_indices):
    """g[j] = 2 * sum_{k in active} a[k-j] for valid k-j"""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += 2.0 * a[k - j]
    return g

            y_new = a_new.copy()

            momentum = (t - 1.0 - 1.0) / beta_t  # (beta_{t-1} - 1) / beta_t

            y_new = project_simplex(y_new)

        a = a_new
        y = y_new

        if score_a < best_score:

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
