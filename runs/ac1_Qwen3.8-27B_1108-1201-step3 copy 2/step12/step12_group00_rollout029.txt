Let me think about a practical approach that could beat 1.5043:

**Approach: Fast Mirror Descent with Entropy Regularization**

The key idea is to use multiplicative weight updates, which are:
1. Very fast (no LP needed)
2. Naturally handle non-negativity
3. Can be accelerated with Nesterov

Algorithm:
1. Normalize a so sum(a) = sqrt(2n)
2. Compute autocorrelation c = convolve(a, a)
3. Find the peak(s) of c
4. Compute subgradient: for each i, grad[i] = sum over active peaks k of (a[k-i] + a[i-k]) / |active_peaks|
5. Multiplicative update: a_new[i] = a[i] * exp(-eta * grad[i])
6. Renormalize
7. Repeat with decreasing eta

The choice of eta is critical. We can use eta = c/t (standard mirror descent rate) or adapt it.

**Additional trick: "Active set" focusing**

Instead of just the single peak, use the top-K peaks with exponential weighting. This gives a smoother gradient and can escape local optima.

**Another trick: Periodic restarts with different initializations**

Since the landscape is non-convex (max of linear functions is convex, but the constraint set with normalization makes it tricky), we can try:
- Different starting points
- Periodic random perturbations
- "Simulated annealing" style: occasionally accept worse solutions

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast mirror descent with entropy (multiplicative weight updates)
2. Nesterov acceleration for convergence
3. Active-set peak weighting for smoother subgradients
4. Multi-length search with interpolation transfer
5. Periodic spectral refinements and coordinate polishing
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

_global_best = None

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the evaluation score without calling evaluate_sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)

def mirror_descent_fast(a0, time_limit, eta_start=0.1, eta_decay=0.999, 
                        top_k=5, verbose=False, seed=42):
    """Fast mirror descent with entropy (multiplicative weights) and Nesterov."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)  # velocity for multiplicative updates (in log space)
    
    iter_count = 0
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation of y (Nesterov lookahead point)
        c = fast_autocorr(y)
        M = c.max()
        if M < 1e-12:
            break
        
        # Find top-k peaks for active set
        if top_k > 1:
            top_indices = np.argsort(c)[::-1][:top_k]
            # Exponential weighting on peaks
            peak_vals = c[top_indices]
            weights = np.exp(peak_vals / (peak_vals[0] * 0.01 + 1e-18))
            weights /= weights.sum()
        else:
            top_indices = np.array([np.argmax(c)])
            weights = np.array([1.0])
        
        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            for i in range(n):
                j1 = k - i
                if 0 <= j1 < n:
                    grad[i] += w * y[j1]
                j2 = i - k
                if 0 <= j2 < n:
                    grad[i] += w * y[j2]
        
        # Scale gradient
        grad_scale = grad.sum() / n
        if grad_scale > 1e-18:
            grad = grad / grad_scale
        
        # Learning rate
        # eta = eta_start / (1 + t * eta_start)  # harmonic decay
        eta = eta_start / (1.0 + 0.01 * t)
        
        # Nesterov momentum in multiplicative space
        # v_new = momentum * v - eta * grad
        momentum = 0.9
        v = momentum * v - eta * grad
        
        # Multiplicative update: a_new = a * exp(v)
        a_new = a * np.exp(v)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_a
        
        # Nesterov update for y
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        alpha = 2.0 / (1.0 + alpha)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a_new.copy()
        else:
            y = y * norm / s_y
        
        a = a_new
        iter_count += 1
        
        # Check improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and iter_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def autocorr_max(a):
    """Return only the max of the autocorrelation."""
    return fast_autocorr(a).max()


def subgrad_single(a, k):
    """Subgradient of (a*a)[k]: grad[j] = 2*a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    grad = np.zeros(n)
    if lo <= hi:
        grad[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def subgrad_multi(a, peaks, weights):
    """Weighted sum of subgradients for multiple peaks."""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, peaks):
        if w < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def get_peak_weights(c, top_k, temp_factor=0.05):
    """Get top-k peak indices with exponential weights."""
    n_peaks = len(c)
    k_use = min(top_k, n_peaks)
    top_idx = np.argsort(c)[::-1][:k_use]
    pv = c[top_idx]
    temp = max(pv[0] * temp_factor, 1e-12)
    w = np.exp((pv - pv[0]) / temp)
    w /= w.sum()
    return top_idx, w


def normalize_seq(a, norm_val):
    """Normalize sequence to have sum = norm_val, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return norm_val / len(a) * np.ones(len(a))
    return a * norm_val / s


def mirror_descent_optimize(a0, time_limit, eta0=0.5, decay_rate=0.9998,
                            top_k=3, momentum=0.85, temp_factor=0.05,
                            verbose=True, seed=42, eta_cycle=2000):
    """
    Fast mirror descent with entropy (multiplicative weights) + Nesterov.
    Minimize max_k (a*a)[k] subject to a>=0, sum(a)=sqrt(2n).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_seq(a, norm_val)

    best_a = a.copy()
    best_score = autocorr_max(a)
    t0 = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    # Dual-averaging accumulator
    dual_avg = np.zeros(n)
    dual_count = 0

    iter_count = 0
    best_eta0 = eta0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(y)
        M = c.max()
        if M < 1e-12:
            break

        peaks, weights = get_peak_weights(c, top_k, temp_factor)
        grad = subgrad_multi(y, peaks, weights)

        gm = grad.max()
        if gm < 1e-15:
            grad = np.zeros(n)
        else:
            grad = grad / gm

        eta = best_eta0
        v = momentum * v - eta * grad

        # Safety: clip log-space velocity to prevent explosion
        v = np.clip(v, -5.0, 5.0)

        a_new = a * np.exp(v)
        a_new = np.maximum(a_new, 1e-20)
        a_new = normalize_seq(a_new, norm_val)

        # Nesterov update
        t_val = iter_count + 1
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha_ns = 2.0 / (1.0 + beta)
        y = (1.0 - alpha_ns) * y + alpha_ns * a_new
        y = normalize_seq(y, norm_val)

        a = a_new
        iter_count += 1

        # Track best
        sc_a = autocorr_max(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        sc_y = autocorr_max(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        # Dual averaging
        dual_avg += a
        dual_count += 1
        if dual_count % 100 == 0:
            da = dual_avg / dual_count
            da = normalize_seq(da, norm_val)
            sc_da = autocorr_max(da)
            if sc_da < best_score:
                best_score = sc_da
                best_a = da.copy()

        # Eta cycling: occasionally increase eta for exploration
        if iter_count % eta_cycle == 0:
            best_eta0 = max(0.01, best_eta0 * 0.7)
            # Also try a boost
            if rng.random() < 0.3:
                best_eta0 = min(3.0, best_eta0 * 2.0)
            v *= 0.5  # Dampen velocity on restart

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={iter_count} ({elapsed:.0f}s): score={best_score:.8f}, eta={best_eta0:.4f}")

def lse_gradient_descent(a0, time_limit, T_start=20.0, T_end=0.5,
                         n_stages=8, verbose=False, seed=42):
    """
    Gradient descent on LogSumExp smooth surrogate of max autocorrelation.
    Faster convergence near optimum than mirror descent.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_seq(a, norm_val)

    best_a = a.copy()
    best_score = autocorr_max(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break
        n_steps = max(10, min(200, int(remaining * 3)))
        step_base = 0.5 / (T * np.sqrt(n))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            c = fast_autocorr(a)
            # LSE surrogate gradient
            c_shifted = c - c.max()
            w = np.exp(c_shifted / T)
            w /= w.sum()

            # grad[j] = 2 * sum_k w[k] * a[k-j] * 1[0<=k-j<n]
            # Compute via correlation
            # For efficiency, use top weighted peaks
            top_idx = np.argsort(w)[::-1][:10]
            top_w = w[top_idx]
            tw_sum = top_w.sum()
            if tw_sum < 1e-15:
                break
            top_w = top_w / tw_sum
            grad = subgrad_multi(a, top_idx, top_w)

            step = step_base / (1.0 + 0.05 * it)
            a_new = a - step * grad
            a_new = normalize_seq(a_new, norm_val)

            sc = autocorr_max(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} T={T:.2f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(a, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif choice == 3:
        a = (1.0 - np.abs(2.0 * np.arange(n) / n - 1.0)) ** 2
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    else:
        a = np.ones(n) * (1.0 + 0.3 * np.cos(4.0 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_local(a):
    """Local evaluation (avoids calling external evaluate_sequence for speed)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = len(a)
    c = fast_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))
