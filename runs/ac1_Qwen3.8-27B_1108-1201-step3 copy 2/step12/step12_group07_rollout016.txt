def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Compute cross-convolution a*b."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def normalize_simplex(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def build_conv_matrix_vec(fixed):
    """Build the convolution matrix T where (T @ x)_k = (fixed * x)(k)."""
    n_fixed = len(fixed)
    n_x = n_fixed  # We assume same length
    size = 2 * n_fixed - 1
    T = np.zeros((size, n_x))
    for j in range(n_x):
        T[j:j + n_fixed, j] = fixed
    return T


_lp_solver = None

def solve_lp_best_response(fixed, solver=None):

def symmetric_lp_nesterov(a0, b0, time_limit, solver=None, max_iters=100000):
    """Symmetric Nesterov for the zero-sum convolution game.
    Maintains both a and b with Nesterov acceleration.
    """
    n = len(a0)
    a = normalize_simplex(np.asarray(a0, dtype=np.float64))
    b = normalize_simplex(np.asarray(b0, dtype=np.float64))
    
    a_prev = a.copy()
    b_prev = b.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            a_y = a + momentum * (a - a_prev)
            b_y = b + momentum * (b - b_prev)
            a_y = np.maximum(a_y, 0)
            b_y = np.maximum(b_y, 0)
            sy_a = a_y.sum()
            sy_b = b_y.sum()
            if sy_a < 1e-15:
                a_y = a.copy()
            else:
                a_y = a_y / sy_a
            if sy_b < 1e-15:
                b_y = b.copy()
            else:
                b_y = b_y / sy_b
        else:
            a_y = a.copy()
            b_y = b.copy()
        
        # Solve LPs
        b_new, M_b = solve_lp_best_response(a_y, solver)
        a_new, M_a = solve_lp_best_response(b_y, solver)
        
        if b_new is None or a_new is None:
            continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        # Update both
        a_upd = (1.0 - alpha) * a + alpha * a_new
        b_upd = (1.0 - alpha) * b + alpha * b_new
        a_upd = normalize_simplex(a_upd)
        b_upd = normalize_simplex(b_upd)
        
        # Evaluate
        sc_a = score_normalized(a_upd)
        sc_b = score_normalized(b_upd)
        
        if sc_a < best_score:
            best_score = sc_a
            best_a = a_upd.copy()
            no_improve = 0
        elif sc_b < best_score:
            best_score = sc_b
            best_a = b_upd.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive restart
        if no_improve > 200 or (sc_a > best_score * 1.002 and t_nest > 20):
            t_nest = 1.0
            a_prev = a_upd.copy()
            b_prev = b_upd.copy()
            restart_count += 1
            no_improve = 0
        else:
            a_prev = a.copy()
            b_prev = b.copy()
            t_nest = t_new
        
        a = a_upd
        b = b_upd
        
        if (it + 1) % 300 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    SymLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed
    
    return best_a, best_score


def multiplicative_weights_adaptive(a0, time_limit, T_start=10.0, T_end=500.0, 
                                     max_steps=50000, eta_start=0.1):
    """Multiplicative weights with adaptive temperature for the convolution game."""
    n = len(a0)
    a = normalize_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_start
    last_improve = 0
    no_improve = 0
    
    # Temperature schedule: geometric
    total_steps = max_steps
    temps = np.geomspace(T_start, T_end, total_steps)
    
    for step in range(total_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > max(500, total_steps // 10):
            break
        
        T = temps[min(step, len(temps) - 1)]
        
        # Compute convolution
        b = fast_convolve(a)
        max_b = b.max()
        
        # Softmax weights on convolution
        if T > 100:
            # Use top-K approximation for large T
            threshold = T * 20  # Only keep entries within 20/T of max
            w = np.exp(T * (b - max_b))
            s = w.sum()
            if s < 1e-300:
                w = np.zeros(len(b))
                w[int(np.argmax(b))] = 1.0
            else:
                w = w / s
        else:
            w = np.exp(T * (b - max_b))
            s = w.sum()
            if s < 1e-300:
                w = np.zeros(len(b))
                w[int(np.argmax(b))] = 1.0
            else:
                w = w / s
        
        # Compute gradient for a (effectively the "cost" of each coordinate)
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]
        
        # Multiplicative update
        # a_new(i) = a(i) * exp(-eta * grad_a(i)) / Z
        exponent = -eta * grad_a
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa
        
        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc

            eta = min(eta * 1.001, 0.3)
        else:

            eta *= 0.95
            if eta < 1e-8:

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """LSE approximation of the max-convolution objective with gradient."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    
    # LSE of b
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    s = exp_b.sum()
    if s < 1e-300:
        log_sum_exp = max_b
        w = np.zeros(len(b))
        w[int(np.argmax(b))] = 1.0
    else:
        log_sum_exp = max_b + np.log(s)
        w = exp_b / s
    
    obj = 2.0 * n * log_sum_exp / T_val
    
    # Gradient: d/dz [LSE(b)] where b = a*a
    # d(b_k)/da_i = a_{k-i} for valid i
    # d(LSE)/da_i = sum_k w_k * a_{k-i}
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    # d/dz via softmax: grad_z = a * (grad_a - sum(a * grad_a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start=80.0, T_end=0.03, n_stages=20):
    """LSE L-BFGS with multi-stage temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        iters = max(20, int(60 * (1.0 - si / n_stages)))
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass
    
    return best_a


def local_coordinate_search(a0, time_limit, n_trials=200000):
    """Local search by perturbing small groups of coordinates."""
    n = len(a0)
    a = normalize_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    T_cool = 0.1  # Temperature for simulated annealing
    T_start_sa = 0.5
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Choose a random window of coordinates to perturb
        window_size = rng.integers(3, 20)
        start = rng.integers(0, n - window_size)
        
        # Create perturbation
        perturbation = rng.normal(0, 0.1, window_size)
        a_trial = a.copy()
        a_trial[start:start+window_size] += perturbation
        
        # Project
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        
        sc_trial = score_normalized(a_trial)
        
        # Accept/reject (simulated annealing)
        progress = trial / n_trials
        T_curr = T_start_sa * (T_cool / T_start_sa) ** progress
        
        if sc_trial < best_score:
            best_score = sc_trial
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < np.exp(-(sc_trial - best_score) / max(T_curr, 1e-10)):
            a = a_trial.copy()
        
        if (trial + 1) % 10000 == 0:
            print(f"    LocalSearch trial={trial+1}: best={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x * (1 + rng.integers(1, 5)))
    elif choice == 1:
        # Multi-frequency
        freqs = rng.uniform(1, 10, size=3)
        a = 1.0
        for f in freqs:
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        # Gaussian mixture
        centers = rng.uniform(0, 1, size=rng.integers(2, 5))
        widths = rng.uniform(0.02, 0.1, size=len(centers))
        amps = rng.uniform(0.5, 2.0, size=len(centers))
        a = 0.1
        for c, w, amp in zip(centers, widths, amps):
            a += amp * np.exp(-((x - c) / w) ** 2)
    elif choice == 3:
        # Power law
        p = rng.uniform(0.3, 2.0)
        a = (x + 0.01) ** p
    elif choice == 4:
        # Cosine series
        a = 1.0
        for k in range(1, rng.integers(2, 6)):
            a += rng.uniform(-0.3, 0.3) * np.cos(2 * np.pi * k * x)
    elif choice == 5:
        # Sinc-like
        a = np.sinc(3.0 * (x - 0.5)) + 0.3
    elif choice == 6:
        # Exponential decay from random center
        center = rng.uniform(0, 1)
        rate = rng.uniform(1, 5)
        a = np.exp(-rate * np.abs(x - center)) + 0.1
    elif choice == 7:
        # Random smooth (Gaussian process sample)
        freqs = rng.uniform(0.5, 5, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        amps = rng.uniform(0.5, 1.5, size=5)
        a = 0.5
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
    elif choice == 8:
        # Tent / triangle
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    else:
        # Chebyshev-like
        a = np.cos(np.pi * x * rng.uniform(1, 3)) + 1.5
    
    a = np.maximum(a, 0.001)
    return normalize_simplex(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_fast(a0, time_limit, max_steps=100000):
    """Fast subgradient descent for quick initial improvement."""
    n = len(a0)
    a = normalize_simplex(np.asarray(a0, dtype=np.float64))
    z = z_from_a(a)
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.3
    last_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.90)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)
        
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z
        
        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.001:
                t_nest = 1.0
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.9995, 0.01)
    
    return best_a, best_score

Key improvements: symmetric game updates, better LSE annealing schedule,
coordinate-level local search, and diverse restarts to escape local minima.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_global_deadline = 0.0
_lp_solver = None


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix_vec(fixed):
    n_fixed = len(fixed)
    size = 2 * n_fixed - 1
    T = np.zeros((size, n_fixed))
    for j in range(n_fixed):
        T[j:j + n_fixed, j] = fixed
    return T


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(fixed)
    x_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_vec(fixed)
    constraints = [T @ x_var <= M_var, cp.sum(x_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    x_val = np.asarray(x_var.value, dtype=np.float64)
    x_val = np.maximum(x_val, 0)
    s = x_val.sum()
    if s < 1e-15:
        return None, None
    x_val = x_val / s
    return x_val, float(M_var.value)


def symmetric_lp_nesterov(a0, b0, time_limit, solver=None, max_iters=100000):
    n = len(a0)
    a = normalize_simplex(a0)
    b = normalize_simplex(b0)
    a_prev = a.copy()
    b_prev = b.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            a_y = a + momentum * (a - a_prev)
            b_y = b + momentum * (b - b_prev)
            a_y = np.maximum(a_y, 0)
            b_y = np.maximum(b_y, 0)
            sa = a_y.sum()
            sb = b_y.sum()
            if sa < 1e-15:
                a_y = a.copy()
            else:
                a_y = a_y / sa
            if sb < 1e-15:
                b_y = b.copy()
            else:
                b_y = b_y / sb
        else:
            a_y = a.copy()
            b_y = b.copy()

        b_new, _ = solve_lp_best_response(a_y, solver)
        a_new, _ = solve_lp_best_response(b_y, solver)

        if b_new is None or a_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_upd = (1.0 - alpha) * a + alpha * a_new
        b_upd = (1.0 - alpha) * b + alpha * b_new
        a_upd = normalize_simplex(a_upd)
        b_upd = normalize_simplex(b_upd)

        sc_a = score_normalized(a_upd)
        sc_b = score_normalized(b_upd)

        if sc_a < best_score:
            best_score = sc_a
            best_a = a_upd.copy()
            no_improve = 0
        elif sc_b < best_score:
            best_score = sc_b
            best_a = b_upd.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300 or (min(sc_a, sc_b) > best_score * 1.003 and t_nest > 25):
            t_nest = 1.0
            a_prev = a_upd.copy()
            b_prev = b_upd.copy()
            no_improve = 0
        else:
            a_prev = a.copy()
            b_prev = b.copy()
            t_nest = t_new

        a = a_upd
        b = b_upd

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    SymLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f}")
                last_print = elapsed

    return best_a, best_score


def multiplicative_weights_adaptive(a0, time_limit, T_start=5.0, T_end=400.0,
                                     max_steps=50000, eta_start=0.08):
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_start
    last_improve = 0
    no_improve = 0

    temps = np.geomspace(T_start, T_end, max_steps)

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > max(1000, max_steps // 8):
            break

        T = temps[min(step, len(temps) - 1)]

        b = fast_convolve(a)
        max_b = b.max()

        scaled = T * (b - max_b)
        scaled = np.maximum(scaled, -50.0)
        w = np.exp(scaled)
        s = w.sum()
        if s < 1e-300:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / s

        grad_a = np.zeros(n)
        active_k = np.where(w > 1e-12)[0]
        for k in active_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]

        exponent = -eta * grad_a
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            no_improve = 0
            eta = min(eta * 1.001, 0.25)
        else:
            no_improve += 1
            eta *= 0.96
            if eta < 1e-9:
                break

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b)
    scaled = np.maximum(scaled, -50.0)
    exp_b = np.exp(scaled)
    s = exp_b.sum()
    if s < 1e-300:
        log_sum_exp = max_b
        w = np.zeros(len(b))
        w[int(np.argmax(b))] = 1.0
    else:
        log_sum_exp = max_b + np.log(s)
        w = exp_b / s

    obj = 2.0 * n * log_sum_exp / T_val

    grad_a = np.zeros(n)
    active_k = np.where(w > 1e-18)[0]
    for k in active_k:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start=60.0, T_end=0.02, n_stages=25, reheat_every=8):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        stage_T = temps[si]

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        if si < n_stages // 3:
            iters = 50
        elif si < 2 * n_stages // 3:
            iters = 35
        else:
            iters = 25

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters * 400,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new

            # Reheating: every reheat_every stages, add small random perturbation
            if si % reheat_every == reheat_every - 1 and si < n_stages - 3:
                z += np.random.default_rng(si * 7 + 3).normal(0, 0.05, n)

        except Exception:
            pass

    return best_a


def local_coordinate_search(a0, time_limit, n_trials=200000, seed=12345):
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    T_sa = 0.05
    T_start_sa = 0.3

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        window_size = int(rng.integers(2, 15))
        start = int(rng.integers(0, max(1, n - window_size)))

        perturbation = rng.normal(0, 0.08, window_size)
        a_trial = a.copy()
        a_trial[start:start + window_size] += perturbation

        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s

        sc_trial = score_normalized(a_trial)

        progress = trial / n_trials
        T_curr = T_start_sa * (T_sa / T_start_sa) ** progress

        if sc_trial < best_score:
            best_score = sc_trial
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < np.exp(-(sc_trial - best_score) / max(T_curr, 1e-12)):
            a = a_trial.copy()

        if (trial + 1) % 20000 == 0:
            print(f"    LocalSearch trial={trial+1}: best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x * int(rng.integers(1, 5)))
    elif choice == 1:
        freqs = rng.uniform(1, 10, size=3)
        a = 1.0
        for f in freqs:
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        n_comp = int(rng.integers(2, 5))
        centers = rng.uniform(0, 1, size=n_comp)
        widths = rng.uniform(0.02, 0.12, size=n_comp)
        amps = rng.uniform(0.5, 2.0, size=n_comp)
        a = 0.1
        for c, w, amp in zip(centers, widths, amps):
            a += amp * np.exp(-((x - c) / w) ** 2)
    elif choice == 3:
        p = rng.uniform(0.3, 2.0)
        a = (x + 0.01) ** p
    elif choice == 4:
        a = 1.0
        n_harm = int(rng.integers(2, 6))
        for k in range(1, n_harm):
            a += rng.uniform(-0.3, 0.3) * np.cos(2 * np.pi * k * x)
    elif choice == 5:
        a = np.sinc(3.0 * (x - 0.5)) + 0.3
    elif choice == 6:
        center = rng.uniform(0.2, 0.8)
        rate = rng.uniform(1, 5)
        a = np.exp(-rate * np.abs(x - center)) + 0.1
    elif choice == 7:
        n_freq = 5
        freqs = rng.uniform(0.5, 5, size=n_freq)
        phases = rng.uniform(0, 2 * np.pi, size=n_freq)
        amps = rng.uniform(0.5, 1.5, size=n_freq)
        a = 0.5
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
    elif choice == 8:
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    else:
        a = np.cos(np.pi * x * rng.uniform(1, 3)) + 1.5
    a = np.maximum(a, 0.001)
    return normalize_simplex(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_fast(a0, time_limit, max_steps=100000):
    n = len(a0)
    a = normalize_simplex(a0)
    z = z_from_a(a)
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.3
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.88)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.001:
                t_nest = 1.0
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.9995, 0.01)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def do_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # Phase 1: Fast subgradient (40s)
    print(f"\n=== Phase 1: Subgradient at n={n_base} (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t1 = min(40.0, remaining * 0.04)
    if t1 > 5:
        a_out, _ = subgradient_fast(best_seq, t1, max_steps=100000)
        do_update(a_out, "subgrad_p1")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # Phase 2: Symmetric LP Nesterov (main, 550s)
    print(f"\n=== Phase 2: Symmetric LP Nesterov (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(550.0, remaining * 0.42)
    if t2 > 30:
        b_init, _ = solve_lp_best_response(best_seq, solver)
        if b_init is None:
            b_init = best_seq.copy()
        a_out, _ = symmetric_lp_nesterov(best_seq, b_init, t2, solver=solver)
        do_update(a_out, "sym_lp_p2")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # Phase 3: Multiplicative weights adaptive (200s)
    print(f"\n=== Phase 3: MW Adaptive (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(200.0, remaining * 0.18)
    if t3 > 15:
        a_out, _ = multiplicative_weights_adaptive(best_seq, t3, T_start=5.0, T_end=300.0, max_steps=40000)
        do_update(a_out, "mw_adaptive")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # Phase 4: LSE L-BFGS (250s)
    print(f"\n=== Phase 4: LSE L-BFGS (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(250.0, remaining * 0.22)
    if t4 > 20:
        a_out = lse_lbfgs(best_seq, t4, T_start=50.0, T_end=0.01, n_stages=28, reheat_every=7)
        do_update(a_out, "lse_p4")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # Phase 5: Second LP round (150s)
    print(f"\n=== Phase 5: LP Nesterov round 2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(150.0, remaining * 0.15)
    if t5 > 15:
        b_init2, _ = solve_lp_best_response(best_seq, solver)
        if b_init2 is None:
            b_init2 = best_seq.copy()
        a_out2, _ = symmetric_lp_nesterov(best_seq, b_init2, t5, solver=solver)
        do_update(a_out2, "sym_lp_p5")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # Phase 6: Local coordinate search (80s)
    print(f"\n=== Phase 6: Local coordinate search (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6 = min(80.0, remaining * 0.08)
    if t6 > 10:
        a_out, _ = local_coordinate_search(best_seq, t6, n_trials=80000, seed=999)
        do_update(a_out, "local_search")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # Phase 7: Multi-start diversification
    print(f"\n=== Phase 7: Multi-start (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 40:
        n_starts = min(4, int(remaining // 45))
        for si in range(n_starts):
            remaining = _global_deadline - time.monotonic()
            if remaining < 25:
                break

            if si == 0:
                n_try = n_base + int(rng.integers(-50, 50))
                if n_try < 100:
                    n_try = 100
                init_try = interpolate_sequence(best_seq, n_try)
            elif si == 1:
                # Blend with a structured init
                s_init = structured_init(n_base, rng)
                init_try = 0.7 * best_seq + 0.3 * s_init
                init_try = normalize_simplex(init_try)
            else:
                init_try = structured_init(n_base, rng)

            t_ms = min(40.0, remaining * 0.3)
            if t_ms < 10:
                break

            print(f"  Multi-start {si+1}: n={len(init_try)}, budget={t_ms:.0f}s")

            a_s, _ = subgradient_fast(init_try, t_ms * 0.4, max_steps=50000)
            do_update(a_s, f"ms_{si}_subgrad")

            remaining = _global_deadline - time.monotonic()
            if remaining > 20:
                b_ms, _ = solve_lp_best_response(a_s, solver)
                if b_ms is None:
                    b_ms = a_s.copy()
                a_ms, _ = symmetric_lp_nesterov(a_s, b_ms, min(25.0, remaining * 0.5), solver=solver)
                do_update(a_ms, f"ms_{si}_lp")
    print(f"  Phase 7 done: best={best_score:.8f}")

    # Phase 8: Final polish
    print(f"\n=== Phase 8: Final polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 20:
        t8a = min(30.0, remaining * 0.4)
        if t8a > 10:
            a_final = lse_lbfgs(best_seq, t8a, T_start=15.0, T_end=0.005, n_stages=12, reheat_every=6)
            do_update(a_final, "final_lse")

        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            a_final2, _ = local_coordinate_search(best_seq, min(20.0, remaining * 0.4), n_trials=30000, seed=777)
            do_update(a_final2, "final_local")

        # One more quick LP pass if time allows
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            b_init3, _ = solve_lp_best_response(best_seq, solver)
            if b_init3 is None:
                b_init3 = best_seq.copy()
            a_final3, _ = symmetric_lp_nesterov(best_seq, b_init3, min(15.0, remaining * 0.6), solver=solver)
            do_update(a_final3, "final_lp")
    print(f"  Phase 8 done: best={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"  Verified score: {best_score:.8f}")

    if best_seq is not None:
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        return [float(x) for x in best_seq]

    print("FALLBACK: uniform sequence")
    return [1.0] * n_base
