    """Line search along the direction from a to b_star."""
    n = len(a)
    def score_t(t):
        a_t = (1 - t) * a + t * b_star
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-15:
            return np.inf

    x2 = lo + gr * (hi - lo)
    f1 = score_t(x1)
    f2 = score_t(x2)

        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
