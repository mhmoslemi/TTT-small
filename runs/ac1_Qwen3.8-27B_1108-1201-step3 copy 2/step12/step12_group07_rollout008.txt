3. `sum_i a_i = 1` (1 equation)
4. `sum_{k in S} w_k = 1` (1 equation)

Unknowns: `a_i` for `i in I` (|I| unknowns), `M` (1), `λ` (1), `w_k` for `k in S` (|S|)
Total unknowns: |I| + 1 + 1 + |S| = |I| + |S| + 2

Total equations: |S| + |I| + 1 + 1 = |S| + |I| + 2 ✓

The Jacobian of this system:
- ∂(a*a)_k / ∂a_i = a_{k-i} + a_{i-k} (for valid indices)
- ∂(a*a)_k / ∂M = -1
- ∂(grad_i) / ∂a_j = sum_{k in S} w_k * (δ_{j,k-i} + δ_{j,i-k}) (where δ is Kronecker delta, with out-of-bounds = 0)
- ∂(grad_i) / ∂w_k = a_{k-i} + a_{i-k}
- ∂(grad_i) / ∂λ = -δ_{i in I} (wait, the equation is grad_i - λ = 0, so ∂/∂λ = -1)
- ∂(sum a) / ∂a_i = 1
- ∂(sum w) / ∂w_k = 1

This is a dense linear system of size |I| + |S| + 2. If |I| ≈ n and |S| is small (say 10-50), the system is about n+50 by n+50, which is 1250x1250 for n=1200. Newton steps would be O(n^3) ≈ 2e9 operations per step, which is feasible for a few steps.

Actually, the system is sparse! The gradient coupling is local (each `grad_i` only depends on `a_j` where `|i-j| < n`), but since the support of `w` is over `S` (which might be spread out), it's not necessarily very sparse.

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)

    return 2.0 * n * b.max()

def solve_lp_full(a_fixed, solver):
    """Solve the best-response LP with all constraints."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build sparse convolution matrix
    # T[k, m] = a[k-m] for 0 <= k-m < n
    rows = []
    cols = []
    vals = []
    for m in range(n):
        for k in range(m, m + n):  # k-m ranges from 0 to n-1
            rows.append(k)
            cols.append(m)
            vals.append(a_fixed[k - m])
    
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp(a0, time_limit, solver, use_reduced=False, reduce_threshold=0.95):
    """Nesterov LP with optional reduced constraints."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    last_print = t0 - 30
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a
        
        # Solve LP (full or reduced)
        if use_reduced and it > 10:
            # Compute current convolution to find active peaks
            b_conv = fast_convolve(y)
            max_b = b_conv.max()
            active_k = np.where(b_conv >= max_b * reduce_threshold)[0]
            if len(active_k) > 200:
                # Keep only top 200
                top_idx = np.argsort(b_conv)[::-1][:200]
                active_k = top_idx
            b_new, M_lp = solve_lp_reduced(y, active_k, solver)
        else:
            b_new, M_lp = solve_lp_full(y, solver)
        
        if b_new is None:
            continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        # Adaptive restart
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 10.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new
        
        a = a_new
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 30:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} cur={sc_new:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed
    
    return best_a, best_score


def kkt_newton_polish(a0, time_limit, max_newton_iters=100, active_tol=1e-4, support_tol=1e-6):
    """
    KKT Newton polishing.
    
    At the optimum of min max_k (a*a)_k s.t. sum(a)=1, a>=0:
    - Active peaks S: (a*a)_k = M for k in S
    - Support I: a_i > 0 for i in I
    - Weights w on S: sum w_k = 1, w_k >= 0
    - Gradient condition: sum_k w_k * (a_{k-i} + a_{i-k}) = lambda for i in I
    
    We solve this nonlinear system with Newton's method.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for restart in range(10):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Identify active set and support
        b_conv = fast_convolve(a)
        M_cur = b_conv.max()
        
        # Active peaks: those within active_tol of max
        active_mask = b_conv >= M_cur - active_tol
        S = np.where(active_mask)[0]
        
        # Limit S to top peaks if too many
        if len(S) > 20:
            top_idx = np.argsort(b_conv)[::-1][:20]
            S = top_idx
        S = np.sort(S)
        
        # Support: indices where a_i > support_tol
        I = np.where(a > support_tol)[0]
        if len(I) < 2:
            I = np.where(a > 0)[0]
        
        if len(S) == 0 or len(I) == 0:
            break
        
        # Set up the KKT system
        # Variables: a_I (|I|), M (1), lambda (1), w_S (|S|)
        # Total: |I| + |S| + 2
        
        nI = len(I)
        nS = len(S)
        nvars = nI + nS + 2
        
        def kkt_residual(x):
            # x = [a_I, M, lambda, w_S]
            a_full = np.zeros(n)
            a_full[I] = x[:nI]
            M_val = x[nI]
            lam_val = x[nI + 1]
            w = x[nI + 2:]
            
            # F1: (a*a)_k - M = 0 for k in S
            b_cur = fast_convolve(a_full)
            F1 = b_cur[S] - M_val
            
            # F2: sum_k w_k * (a_{k-i} + a_{i-k}) - lambda = 0 for i in I
            F2 = np.zeros(nI)
            for ii, i in enumerate(I):
                grad_i = 0.0
                for ks in range(nS):
                    k = S[ks]
                    # a_{k-i} is valid if 0 <= k-i < n
                    if 0 <= k - i < n:
                        grad_i += w[ks] * a_full[k - i]
                    # a_{i-k} is valid if 0 <= i-k < n
                    if 0 <= i - k < n:
                        grad_i += w[ks] * a_full[i - k]
                F2[ii] = grad_i - lam_val
            
            # F3: sum(a) - 1 = 0
            F3 = a_full.sum() - 1.0
            
            # F4: sum(w) - 1 = 0
            F4 = w.sum() - 1.0
            
            return np.concatenate([F1, F2, [F3, F4]])
        
        # Initialize
        x0 = np.concatenate([a[I], [M_cur], [M_cur], np.ones(nS) / nS])
        
        # Use fsolve or Newton
        from scipy.optimize import fsolve
        try:
            x_sol, info, ier, msg = fsolve(kkt_residual, x0, full_output=True, maxfev=10000)
            
            if ier == 1:
                a_sol = np.zeros(n)
                a_sol[I] = np.maximum(x_sol[:nI], 0)
                s_sol = a_sol.sum()
