    stall = 0
    best_dual = float('inf')
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1
        
        # Track dual objective (lower bound)
        dual_obj = 2.0 * n * M_val
        if dual_obj < best_dual:
            best_dual = dual_obj
        
        # Nesterov schedule with adaptive momentum
        alpha_t = 2.0 / (t + 2.0)
        
        # Nesterov lookahead
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s
        
        # Main iterate with momentum
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s
        
        y = y_new
        a = a_new
        
        # Score multiple candidates
        candidates = {
            'a': a_new,
            'y': y_new,
        }
        
        # Running average (theoretical guarantee is for this)
        avg_a += a_new
        n_avg += 1
        if n_avg >= 10:
            candidates['avg'] = avg_a / n_avg
        
        # EMA
        ema_a = (1 - ema_alpha) * ema_a + ema_alpha * a_new
        if t > 20:
            candidates['ema'] = ema_a
        
        for name, cand in candidates.items():
            sc = 2.0 * n * fast_autocorr(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                stall = 0
            else:
                stall += 1

        if stall > 500:

            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, dual={best_dual:.6f}, gap={best_score-best_dual:.6f}")
        
        # Early stopping
        if stall > 5000 and t > 1000:
            if verbose:
                print(f"    Early stop at t={t}")
            break
    
    return best_a, best_score, iter_count

def logsumexp_mirror_descent(a0, time_limit, beta_start=50.0, beta_end=500.0, verbose=True):
    """Log-sum-exp smoothed mirror descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_iters = 10000
    lr_start = 0.5
    lr_end = 0.005
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        progress = it / max(1, n_iters)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute autocorrelation
        b = fast_autocorr(a)
        
        # Softmax weights (stable)
        b_shifted = b - b.max()
        exp_b = np.exp(np.minimum(b_shifted * beta, 50))  # clip to avoid overflow
        w = exp_b / exp_b.sum()
        
        # Gradient: grad_j = 2 * sum_k w_k * a_{k-j} (with valid indices)
        # This is a weighted convolution of a with the weight sequence w
        # But w has length 2n-1, so we need to be careful
        grad = np.zeros(n)
        for j in range(n):
            # grad_j = 2 * sum_k w_k * a_{k-j} where 0 <= k-j < n
            # = 2 * sum_{j2=0}^{n-1} w_{j+j2} * a_{j2}
            # = 2 * (w convolve reversed_a)[... ]
            # Actually: grad_j = 2 * sum_{j2=0}^{n-1} a_{j2} * w_{j+j2}
            pass
        
        # Better: use the structure. The gradient of conv(a,a)_k w.r.t. a_j is:
        # 2 * a_{k-j} if 0 <= k-j < n, else 0
        # So grad_j = 2 * sum_k w_k * a_{k-j} (where k-j is valid)
        # = 2 * sum_{j2: 0<=j2<n} a_{j2} * w_{j+j2}
        
        # This is like a cross-correlation of a with w, evaluated at specific positions
        # Let's compute it efficiently:
        # grad_j = 2 * sum_{j2=0}^{n-1} a_{j2} * w_{j+j2}
        # If we define v = flip(a) = a[n-1::-1], then:
        # grad_j = 2 * (v cross-correlate w)[n-1-j] ... hmm this is getting complicated.
        
        # Let me just do it directly but vectorized:
        # For each j, grad_j = 2 * dot(a, w[j:j+n])
        # w has length 2n-1, so w[j:j+n] is valid for j=0,...,n-1
        # We can compute all of these at once using a matrix or FFT
        
        # Using FFT: grad_j = 2 * sum_{j2=0}^{n-1} a_{j2} * w_{j+j2}
        # Let's define u = a (length n), then grad_j = 2 * (u * w[j:j+n])
        # This is the "Aire" product or cross-correlation
        # Actually: if we define the full cross-correlation c = np.correlate(w, a, 'valid')
        # then c[j] = sum_i w[j+i] * a[i] for j=0,...,len(w)-len(a) = n-1
        # So grad = 2 * c
        
        c = np.correlate(w, a, 'valid')  # c[j] = sum_i w[j+i]*a[i], j=0..n-1
        grad = 2.0 * c
        
        # Learning rate
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Mirror descent update: a_new_j = a_j * exp(-lr * grad_j) / Z
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_s = a_new.sum()
        if a_new_s < 1e-30:
            break
        a_new = a_new / a_new_s
        
        sc = 2.0 * n * fast_autocorr(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 1000 == 0:
            cur_sc = 2.0 * n * fast_autocorr(a).max()
            print(f"    LSD it={it}: best={best_score:.6f}, cur={cur_sc:.6f}, beta={beta:.1f}")
    
    return best_a, best_score

def qp_refinement(a0, time_limit, n_active_max=50, verbose=True):
    """QP refinement on active constraint set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_active = 15
    
    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Identify active lags
        threshold = max_b * 0.995
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) > n_active:
            active_lags = active_lags[np.argsort(b[active_lags])[::-1]][:n_active]
        
        # Build QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iter=20000, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s
        
        new_max = fast_autocorr(a_new).max()
        
        if new_max < max_b:
            score = 2.0 * n * new_max
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"    QP round {round_num}: max={new_max:.8f}, score={score:.6f}")
        else:
            # Try mixing
            improved = False
            for frac in [0.2, 0.4, 0.6, 0.8]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                mix_max = fast_autocorr(a_mix).max()
                if mix_max < max_b:
                    score = 2.0 * n * mix_max
                    if score < best_score:
                        best_score = score
                        best_a = a_mix.copy()
                        a = a_mix
                        max_b = mix_max
                        improved = True
                        if verbose:
                            print(f"    QP mix {frac}: score={score:.6f}")
                        break
            
            if not improved and round_num >= 5:
                # Expand active set
                n_active = min(n_active + 10, n_active_max)
                continue
            elif not improved:
                break
    
    return best_a, best_score

def coordinate_descent_peak_shave(a0, time_limit, seed=42, verbose=True):
    """Aggressive coordinate descent peak shaving."""
    n = len(a0)
    rng = np.random.default_generator(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        top_k = np.argsort(b)[::-1][:8]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Find top contributing pairs
            top_ci = np.argsort(contrib)[::-1][:min(8, len(idx))]
            
            for ci in top_ci:
                src = idx[ci]
                other = k - src
                
                # Try moving mass from src (and other) to a random position
                n_attempts = 5
                for _ in range(n_attempts):
                    dst = rng.integers(0, n)
                    if dst == src:
                        continue
                    
                    for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        if src != other and src + other != k:
                            # Also reduce the "other" position
                            amt2 = min(a[other], amt * 0.5)
                            a_trial[other] -= amt2
                            a_trial[dst] += amt + amt2
                        else:
                            a_trial[dst] += amt
                        
                        if (a_trial < -1e-12).any():
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial_s = a_trial.sum()
                        if a_trial_s < 1e-15:
                            continue
                        a_trial = a_trial / a_trial_s
                        
                        trial_max = fast_autocorr(a_trial).max()
                        if trial_max < max_b:
                            score = 2.0 * n * trial_max
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                max_b = trial_max
                                stall = 0
                                break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Perturbation
            if stall % 50 == 0:
                n_move = rng.integers(2, 8)
                sources = rng.choice(n, size=n_move, replace=False)
                targets = rng.choice(n, size=n_move, replace=False)
                a_trial = a.copy()
                for src, dst in zip(sources, targets):
                    frac = rng.uniform(0.005, 0.03)
                    amt = a_trial[src] * frac
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s < 1e-15:
                    continue
                a_trial = a_trial / a_trial_s
                trial_max = fast_autocorr(a_trial).max()
                if trial_max < max_b:
                    a = a_trial
                    max_b = trial_max
                    score = 2.0 * n * max_b
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    stall = 0
            
            if stall > 2000:
                if verbose:
                    print(f"    CD: stalled at {best_score:.6f}")
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def generate_laplace_sequence(n, sigma=0.3):
    """Generate a Laplace-like centered sequence."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-np.abs(x - center) / (n * sigma))
    a = a / a.sum()
    return a
