def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))

def lp_best_response(a, M, solver):

    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]

    try:

        return None

        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    return b_val


def lp_best_response_minmax(a, solver):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

            frac = rng.uniform(0.05, 0.3)
            amt = a_trial[i] * frac
            a_trial[i] -= amt
            a_trial[j] += amt
            if a_trial[i] < 0:
                a_trial[i] = 0.0

        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial * norm / s_t
        score = fast_autocorr(a_trial).max()

        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial.copy()
        else:
            a = best_a.copy()

    return best_a, best_score


def cosine_param_optimize(a0, time_limit, n_harmonics=20):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a0, fast_autocorr(a0).max()
    a = a / s0

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    from scipy.optimize import minimize

    def coeffs_to_a(coeffs):
        x = np.arange(n) / n
        result = np.zeros(n)
        for k in range(len(coeffs)):
            if k == 0:
                result += coeffs[k]
            else:
                result += coeffs[k] * np.cos(2.0 * np.pi * k * x)
        result = np.maximum(result, 0.0)
        s = result.sum()
        if s < 1e-15:
            result = np.ones(n) / n
        else:
            result = result / s
        return result

    def objective(coeffs):
        aa = coeffs_to_a(coeffs)
        return 2.0 * n * fast_autocorr(aa).max()

    x_init = np.zeros(n_harmonics)
    x_init[0] = 1.0
    for k in range(1, n_harmonics):
        x_init[k] = np.mean(a * np.cos(2.0 * np.pi * k * np.arange(n) / n))

    res = minimize(objective, x_init, method='Nelder-Mead',
                   options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10, 'adaptive': True})

    if time.monotonic() - t0 < time_limit:
        a_opt = coeffs_to_a(res.x)
        score = 2.0 * n * fast_autocorr(a_opt).max()
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()

    return best_a * norm, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a_raw, label):
        nonlocal best_seq, best_score
        if a_raw is None:
            return
        a = np.maximum(np.asarray(a_raw, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov (maxsum) at base length - primary
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov (maxsum) n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 30:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_base) / s_a
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_minmax=False)
        do_try_update(a_out, f"lp_maxsum_n{n_base}")

    # Phase 2: LP Nesterov (minmax) at base length - different inner problem
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Nesterov (minmax) n={n_base} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 240.0)
    if t2 > 30:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_base) / s_a
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, use_minmax=True)
        do_try_update(a_out, f"lp_minmax_n{n_base}")

    # Phase 3: Try longer length n2
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 320.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, use_minmax=False)
        do_try_update(a_out, f"lp_n{n2}")

    # Phase 4: Try even longer n3
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 240.0)
    if t4 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, use_minmax=False)
        do_try_update(a_out, f"lp_n{n3}")

    # Phase 5: Coordinate descent polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 200.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t5, seed=777)
        if a_out is not None:
            do_try_update(a_out, "coord_cd")

    # Phase 6: Subgradient FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: SubGrad FW n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgrad_fw_optimize(a_norm, t6, verbose=False)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # Phase 7: Perturbation search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Perturbation n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.10, 80.0)
    if t7 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = perturbation_search(a_norm, t7, solver, seed=456)
        do_try_update(a_out, "perturb")

    # Phase 8: Final LP polish with remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP polish n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.40, 150.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True, use_minmax=False)
        do_try_update(a_out, "final_lp")

    # Phase 9: Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.6, 80.0)
        if t9 > 5:
            print(f"\n=== Phase 9: Final coord descent n={n_best} ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coordinate_descent_polish(a_norm, t9, seed=999)
            if a_out is not None:
                do_try_update(a_out, "final_cd")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
