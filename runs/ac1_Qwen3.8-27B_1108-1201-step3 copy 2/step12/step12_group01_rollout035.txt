def fast_conv(a, b):

    m = len(b)

from scipy.optimize import minimize


def fast_conv(a, b):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def obj_val(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def nesterov_fast_game(a0, time_limit, n_inner=20, lr_inner=1.5, seed=42):
    """
    Fast Nesterov zero-sum game with multiplicative weights inner loop.
    Minimizes max_k conv(a,a)[k] subject to a in simplex.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a, a).max()
    t0 = time.monotonic()
    t = 0
    y = a.copy()
    running_sum = np.zeros(n)
    count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        alpha = 2.0 / (t + 2.0)

        # Inner loop: multiplicative weights to approximate best response b given y
        b = y.copy()
        for _ in range(n_inner):
            conv_yb = fast_conv(y, b)
            k_star = int(np.argmax(conv_yb))
            g = np.zeros(n)
            m_lo = max(0, k_star - n + 1)
            m_hi = min(n - 1, k_star)
            if m_lo <= m_hi:
                g[m_lo:m_hi + 1] = y[k_star - m_lo:k_star - m_hi - 1: -1]
            b = b * np.exp(-lr_inner * g)
            b = np.maximum(b, 1e-300)
            b = b / b.sum()

        # Nesterov extrapolation
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        ys = y_new.sum()
        if ys > 1e-15:
            y_new = y_new / ys
        else:
            y_new = b.copy()

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = y_new.copy()

        y = y_new
        a = a_new

        # Running average
        running_sum += a
        count += 1

        # Periodic evaluation
        if t % 5 == 0:
            sc = 2.0 * n * fast_conv(a, a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Check running average
        if count >= 50 and t % 50 == 0:
            avg_a = running_sum / count
            sc_avg = 2.0 * n * fast_conv(avg_a, a).max() if False else 2.0 * n * fast_conv(avg_a, avg_a).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_a.copy()
                stall = 0

        # Adaptive restart
        if stall > 2000 and t > 500:
            y = best_a.copy()
            a = best_a.copy()
            running_sum = np.zeros(n)
            count = 0
            stall = 0
            if t % 5000 == 0:
                print(f"      NFG restart at t={t}, best={best_score:.6f}")

        if t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NFG t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lse_obj_grad_z(z, T):
    """
    LSE-smoothed objective and gradient in z-space (a = softmax(z)).
    obj = (1/T)*log(sum_k exp(T*(a*a)[k]))
    """
    a = softmax_fn(z)
    n = len(a)
    conv_vals = fast_conv(a, a)
    Tc = T * conv_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    Z = exp_Tc.sum()
    obj = (max_Tc + np.log(Z)) / T
    w = exp_Tc / Z

    # grad_a[j] = (2/T) * sum_k w[k] * a[k-j]
    a_flipped = a[::-1]
    conv_w_af = fast_conv(w, a_flipped)
    grad_a = (2.0 / T) * conv_w_af[n - 1:2 * n - 1]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbfgs_lse_polish(a0, time_limit, T_schedule=None, max_iter_total=None):
    """
    L-BFGS optimization of LSE-smoothed objective with temperature annealing.
    """
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()

    if T_schedule is None:
        T_schedule = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.4, 0.25, 0.15]

    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a, a).max()
    total_iters = 0

    for si, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        max_iter_stage = max(20, int(remaining * 2))
        if max_iter_total is not None:
            max_iter_stage = min(max_iter_stage, max_iter_total)

        def fun_and_grad(z_vec):
            return lse_obj_grad_z(z_vec, T)

        try:
            res = minimize(fun_and_grad, z, method='L-BFGS-B',
                           jac=True, options={'maxiter': max_iter_stage,
                                              'gtol': 1e-10, 'maxls': 50})
            z = res.x
            total_iters += res.nit
        except Exception:
            pass

        a_stage = softmax_fn(z)
        sc = 2.0 * n * fast_conv(a_stage, a_stage).max()
        if sc < best_score:
            best_score = sc
            best_a = a_stage.copy()

        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE T={T:.1f} ({elapsed:.0f}s): score={sc:.6f}, best={best_score:.6f}")

    return best_a, best_score
