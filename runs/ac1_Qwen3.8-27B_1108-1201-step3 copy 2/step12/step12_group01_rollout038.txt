import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_projection(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    v = v / v.sum()  # If already on simplex
    # Actually, proper projection:
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def topk_subgradient(a, c, K=5):
    """Compute top-K weighted subgradient of max(c) w.r.t. a."""
    n = len(a)
    # Get top K indices
    top_idx = np.argsort(c)[::-1][:K]
    # Softmax weights
    top_vals = c[top_idx]
    T = 1.0 / max(c.max(), 1e-10)  # Temperature
    w = np.exp(T * (top_vals - top_vals.max()))
    w = w / w.sum()
    
    g = np.zeros(n)
    for idx in range(K):
        k = top_idx[idx]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx_range = np.arange(js, je + 1)
        g[idx_range] += w[idx] * 2.0 * a[k - idx_range]
    return g

    best_a = a.copy()

    t = 0
    step = step_init

        c = fast_autocorr(y)
        c_max = c.max()
        
        g = topk_subgradient(y, c, K)

        y_new = y - step * g

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative-weights subgradient with Nesterov (no LP, FFT-based)
2. Top-k LogSumExp gradient descent with Adam for refinement
3. Multi-length exploration with interpolation
4. Direct peak-coordinate descent for final polish
5. Fixed-point peak flattening iteration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    else:
        result = np.ones(n) / n
    return result


def compute_topk_grad(a, c, K):
    """Compute top-K weighted subgradient of max(c) w.r.t. a."""
    n = len(a)
    k_use = min(K, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    # Softmax weights with temperature based on spread
    spread = top_vals[0] - top_vals[-1]
    if spread > 1e-15:
        T = spread * 0.3
        w = np.exp((top_vals - top_vals[0]) / T)
        w = w / w.sum()
    else:
        w = np.ones(k_use) / k_use
    g = np.zeros(n)
    for idx in range(k_use):
        k = int(top_idx[idx])
        wK = w[idx]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            g[idx_range] += wK * 2.0 * a[k - idx_range]
    return g


def mp_subgradient_nesterov(a0, time_limit, K=15, eta0=0.5, n_restarts=5):
    """
    Multiplicative weights + top-K subgradient with Nesterov extrapolation.
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    no_improve = 0
    restart_interval = max(5000, int(time_limit * 20))
    used_restarts = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_autocorr(y)
        c_max = c.max()

        g = compute_topk_grad(y, c, K)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Adaptive step size
        eta_t = eta0 / (np.sqrt(t) + 1.0) * (n / max(g_norm * n, 1.0))
        eta_t = min(eta_t, 2.0)

        # Multiplicative weights (mirror descent on simplex)
        a_new = y * np.exp(-eta_t * g)
        a_new = np.maximum(a_new, 1e-30)
        a_sum = a_new.sum()
        if a_sum > 1e-15:
            a_new = a_new / a_sum
        else:
            a_new = y.copy()

        # Nesterov extrapolation
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum > 1e-15:
            y_new = y_new / y_sum
        else:
            y_new = a_new.copy()

        y = y_new
        a = a_new

        # Check improvement every few steps
        if t % 5 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 5

        # Restart if stuck
        if no_improve > restart_interval and used_restarts < n_restarts:
            y = best_a.copy()
            a = best_a.copy()
            no_improve = 0
            used_restarts += 1
            t = 0
            if used_restarts <= 3:
                eta0 *= 0.7

    # Final check
    score_a = 2.0 * n * fast_autocorr(a).max()
    if score_a < best_score:
        best_score = score_a
        best_a = a.copy()

    return best_a, best_score


def fixed_point_flatten(a0, time_limit, n_iters=5000):
    """
    Fixed-point iteration: a_i <- 1/a[k*-i] targeting the dominant peak.
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    damping = 0.3

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_autocorr(a)
        c_max = c.max()
        k_star = int(np.argmax(c))

        # Fixed point: a_i proportional to 1 / a[k_star - i]
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        a_fp = a.copy()
        if js <= je:
            idx_range = np.arange(js, je + 1)
            vals = a[k_star - idx_range]
            a_fp[idx_range] = 1.0 / np.maximum(vals, 1e-10)
            # For positions not contributing to this peak, keep as-is but reduced
            mask = np.ones(n, dtype=bool)
            mask[idx_range] = False
            a_fp[mask] = a[mask] * 0.5

        # Damped update
        a_new = (1.0 - damping) * a + damping * a_fp
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
        else:
            a_new = a.copy()
        a = a_new

        if (it + 1) % 50 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    score = 2.0 * n * fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def lse_gradient_adam(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=20, topk=12):
    """
    Adam-optimized top-k LogSumExp with multi-stage annealing.
    """
    n = len(a0)
    a_init = np.asarray(a0, dtype=np.float64).copy()
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break

        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.85 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(50, int(remaining * 4))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()
            n_a = len(a)
            c = fast_autocorr(a)
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]

            Tc = stage_T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()

            # Gradient w.r.t. a
            grad_a = np.zeros(n_a)
            for idx in range(k_use):
                K = int(top_idx[idx])
                wK = w[idx]
                js = max(0, K - n_a + 1)
                je = min(n_a - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

            # Gradient w.r.t. z (softmax parameterization)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_check = np.exp(z - z.max())
                a_check = a_check / a_check.sum()
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        # End of stage
        a_stage = np.exp(z - z.max())
        a_stage = a_stage / a_stage.sum()
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
            z = np.log(np.maximum(best_a, 1e-30))

    return best_a, best_score

            # Try multiple step sizes to best safe destinations
            safe_sorted = safe[np.argsort(a[safe])][:max(5, len(safe) // 10)]
            for dst in safe_sorted[:5]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(1, 3)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.8 + 0.15 * np.cos(3 * x) + 0.05 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.6 * np.exp(-4 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5 + 0.3 * x ** 1.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    1. Fast subgradient with multiplicative weights (main workhorse)
    2. Fixed-point peak flattening
    3. LSE gradient descent with Adam
    4. Multi-length exploration
    5. Peak coordinate descent for final polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1680.0)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a_arr, label):
        nonlocal best_seq, best_score
        if a_arr is None:
            return
        a_arr = np.maximum(np.asarray(a_arr, dtype=np.float64), 0)
        s = a_arr.sum()
        if s < 0.01:
            return
        a_arr = a_arr / s
        sc = evaluate_safe(a_arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_arr.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_arr)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Fast subgradient with multiplicative weights ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: MP Subgradient n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.30, 550.0)
    if t1 > 60:
        a_out, _ = mp_subgradient_nesterov(best_seq, t1, K=15, eta0=0.8)
        try_update(a_out, f"mpsub_n{n_base}")

    # === Phase 2: Fixed-point peak flattening ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: Fixed-point flatten n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 30:
        a_out, _ = fixed_point_flatten(best_seq, t2, n_iters=10000)
        try_update(a_out, f"fp_n{n_base}")

    # === Phase 3: LSE gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: LSE Adam n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        a_out, _ = lse_gradient_adam(best_seq, t3, T_start=60.0, T_end=0.3, n_stages=20, topk=12)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Multi-length ({remaining:.0f}s left) ===")
    lengths_to_try = []
    deltas = [100, 200, 300, -100, -200, 400, 500]
    for d in deltas:
        nl = n_base + d
        if nl >= 200 and nl <= 5000:
            lengths_to_try.append(nl)

    t4_each = min(remaining * 0.08, 100.0)
    for nl in lengths_to_try[:5]:
        if time.monotonic() - t_global_start > time_budget - 100:
            break
        if remaining < 30:
            break
        remaining = time_budget - (time.monotonic() - t_global_start)
        t4 = min(t4_each, remaining * 0.5)
        if t4 < 20:
            break
        init_long = interpolate_sequence(best_seq, nl)
        # Quick subgradient optimization
        a_out, _ = mp_subgradient_nesterov(init_long, t4, K=10, eta0=0.5)
        try_update(a_out, f"mpsub_n{nl}")
        # Quick LSE
        remaining = time_budget - (time.monotonic() - t_global_start)
        t4b = min(remaining * 0.06, 80.0)
        if t4b > 20:
            a_out2, _ = lse_gradient_adam(a_out, t4b, T_start=30.0, T_end=1.0, n_stages=10, topk=10)
            try_update(a_out2, f"lse_n{nl}")

    # === Phase 5: Peak coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: Peak descent n={len(best_seq)} ({remaining:.0f}s left) ===")
    t5 = min(remaining * 0.25, 200.0)
    if t5 > 10:
        a_out, _ = peak_coordinate_descent(best_seq, t5, seed=777)
        try_update(a_out, "peakdesc1")

    # === Phase 6: Second round of subgradient with different params ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"=== Phase 6: MP Subgradient round 2 ({remaining:.0f}s left) ===")
        t6 = min(remaining * 0.30, 200.0)
        if t6 > 20:
            a_out, _ = mp_subgradient_nesterov(best_seq, t6, K=8, eta0=0.3, n_restarts=3)
            try_update(a_out, "mpsub_r2")

    # === Phase 7: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        print(f"=== Phase 7: Final LSE polish ({remaining:.0f}s left) ===")
        t7 = min(remaining * 0.30, 150.0)
        if t7 > 15:
            a_out, _ = lse_gradient_adam(best_seq, t7, T_start=5.0, T_end=0.1, n_stages=12, topk=8)
            try_update(a_out, "final_lse")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        print(f"=== Phase 8: Final peak shave ({remaining:.0f}s left) ===")
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 10:
            a_out, _ = peak_coordinate_descent(best_seq, t8, seed=999)
            try_update(a_out, "peakdesc2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
