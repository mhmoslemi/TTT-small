def mirror_descent_nesterov(a0, time_limit, T_initial=5.0, T_final=0.1, n_iters=None, topk=20):
    """
    Mirror descent with Nesterov acceleration for minimizing max_k conv(a,a)[k].
    Uses LogSumExp soft-subgradient with annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    if n_iters is None:
        n_iters = int(time_limit * 50)  # estimate
    
    y = a.copy()  # Nesterov lookahead
    avg = np.zeros(n)
    
    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        # Anneal temperature
        progress = t / n_iters
        T = T_initial * (T_final / T_initial) ** progress
        
        # Compute subgradient at y
        grad_y, _ = lse_subgradient(y, T, topk=topk)
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        
        # Mirror descent step: a_new = a * exp(-eta * grad) / Z
        # Step size for mirror descent
        eta = alpha * 0.5  # scale factor
        
        # Regularize gradient to avoid numerical issues
        grad_reg = grad_y - grad_y.mean()  # center the gradient
        scale = np.abs(grad_reg).max()
        if scale > 1e-10:
            grad_reg = grad_reg / scale
        
        a_new = a * np.exp(-eta * grad_reg)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov update for y
        momentum = (t - 1.0) / (t + 2.0)
        y_new = a_new + momentum * (a_new - y)
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / s_y
        
        y = y_new
        a = a_new
        
        # Running average (Polyak)
        avg += a
        
        # Evaluate periodically
        if t % 200 == 0 or t == 1:
            avg_candidate = avg / t
            score_avg = compute_score(avg_candidate)
            score_a = compute_score(a)
            
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_candidate.copy()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            
            if t % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      MD-Nest t={t} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.3f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-3 * x)  # Exponential decay
    elif choice == 1:
        a = np.exp(-x**2 * 5)  # Gaussian
    elif choice == 2:
        a = (1 - x)**3  # Polynomial decay
    elif choice == 3:
        a = np.exp(-5 * (x - 0.3)**2) + 0.3 * np.exp(-5 * (x - 0.8)**2)
    elif choice == 4:
        a = np.exp(-2 * x) * (1 + 0.3 * np.sin(10 * x))
    else:
        a = np.exp(-x**2 * 3) * (1 + 0.2 * np.cos(5 * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def convolutional_ipf(a0, time_limit, n_iters=None, damping=0.3):
    """
    Convolutional IPF / multiplicative peak redistribution.
    Iteratively adjusts a to flatten its autocorrelation.

        threshold = bmean * 2.0

        n_peaks = peak_mask.sum()
        if n_peaks == 0:
            break

        # Compute multiplicative correction factor for each position i
        # Position i contributes to b[k] with weight a[i] * a[k-i]
        # We want to reduce positions that contribute most to peaks
        correction = np.ones(n)

def spectral_flatten(a0, time_limit, n_iters=None, strength=0.3):
    """
    Spectral flattening: work in frequency domain to reduce autocorrelation peaks.
    Iterates: FFT -> modify magnitude -> IFFT -> project to simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    if n_iters is None:
        n_iters = int(time_limit * 40)

    # Use zero-padded FFT for linear convolution
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 2:
            break

        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Flatten magnitude: mag_new = mag^(1-strength) * mag_mean^strength
        mag_mean = mag.mean()
        if mag_mean < 1e-15:
            break
        mag_new = np.power(np.maximum(mag, 1e-15), 1.0 - strength) * np.power(mag_mean, strength)

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # Gradually reduce strength
            strength = max(0.05, strength * 0.98)
        else:
            strength = max(0.05, strength * 0.95)

        if it % 1000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"      SpecFlatten it={it} ({elapsed:.0f}s): best={best_score:.7f}, strength={strength:.3f}")

    return best_a, best_score

def safe_mask(indices, a):
    """Filter indices to valid range."""
    n = len(a)
    return (indices >= 0) & (indices < n)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):

            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            state['best_seq'] = parent.copy()
            state['best_score'] = p_score
    except Exception:
        pass

    n_base = len(state['best_seq']) if state['best_seq'] is not None else 1600

    if parent is None:
        n_base = 1600
        state['best_seq'] = structured_init(n_base, rng)
        state['best_score'] = compute_score(state['best_seq'])
        print(f"Default init: n={n_base}, score={state['best_score']:.7f}")

    # Helper to update best
    def try_update(a, label):
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < state['best_score']:
            state['best_score'] = sc
            state['best_seq'] = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.32, 530.0)
    if t1 > 40:
        a_out, _ = nesterov_lp_optimize(state['best_seq'], t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Convolutional IPF (NEW) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: Convolutional IPF n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 15:
        a_out, _ = convolutional_ipf(state['best_seq'], t2, damping=0.25)
        try_update(a_out, f"cipf_n{n_base}")

    # === Phase 3: Spectral flattening (NEW) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: Spectral Flatten n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 10:
        a_out, _ = spectral_flatten(state['best_seq'], t3, strength=0.3)
        try_update(a_out, f"spec_n{n_base}")

    # === Phase 4: Enhanced peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Peak Shaving n={n_base} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 15:
        a_out, _ = enhanced_peak_shaving(state['best_seq'], t4, seed=123)
        try_update(a_out, f"peak_n{n_base}")

    # === Phase 5: LP Nesterov at longer length ===
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: LP Nesterov n={n_long} ({remaining:.0f}s left) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 40:
        init_long = interpolate_sequence(state['best_seq'], n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t5, solver)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 6: Peak shaving at longer length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t6 = min(remaining * 0.10, 100.0)
        if t6 > 10:
            print(f"=== Phase 6: Peak Shave n={n_long} ({remaining:.0f}s left) ===")
            a_out, _ = enhanced_peak_shaving(state['best_seq'], t6, seed=456)
            try_update(a_out, f"peak_n{n_long}_2")

    # === Phase 7: Micro-refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 7: Micro Refine ({remaining:.0f}s left) ===")
    t7 = min(remaining * 0.20, 180.0)
    if t7 > 10:
        a_out, _ = micro_refine(state['best_seq'], t7, seed=789)
        try_update(a_out, "micro_refine")

    # === Phase 8: Final LP Nesterov polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 20:
            print(f"=== Phase 8: Final LP Polish ({remaining:.0f}s left) ===")
            a_out, _ = nesterov_lp_optimize(state['best_seq'], t8, solver)
            try_update(a_out, "final_lp")

    # === Phase 9: Final micro-refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t9 = min(remaining * 0.50, 80.0)
        if t9 > 8:
            print(f"=== Phase 9: Final Micro ({remaining:.0f}s left) ===")
            a_out, _ = micro_refine(state['best_seq'], t9, seed=101)
            try_update(a_out, "micro_final")

    # Final verification
    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = fs
        print(f"\nFINAL SCORE: {state['best_score']:.7f} (n={len(state['best_seq'])})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in state['best_seq']]
