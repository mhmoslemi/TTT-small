    best_a = a.copy()
    t0 = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        for i in range(n):
            a_i = a[i]
            S_minus = S - a_i
            
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Base max
            base_max = 0.0
            if k_lo > 0:
                base_max = max(base_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                base_max = max(base_max, float(conv[k_hi+1:].max()))
            
            ks = np.arange(k_lo, k_hi + 1)
            offsets = ks - i
            
            candidates = [0.0]
            if a_i > 0:
                candidates.append(a_i)
            
            # Analytical
            k_peak = int(ks[np.argmax(conv[ks])])
            if k_peak != 2 * i:
                m = a[k_peak - i]
                B = conv[k_peak]
                if m > 1e-15:
                    # f(x) = 2n*(B + 2m*(x-ai))/(S_minus+x)^2
                    # df/dx = 0 => m*(S_minus+x) = B + 2m*(x-ai)
                    # m*S_minus + m*x = B + 2m*x - 2m*ai
                    # m*S_minus - B + 2m*ai = m*x
                    # x = S_minus - B/m + 2*ai
                    x_an = S_minus - B / m + 2.0 * a_i
                    if 0 < x_an < 100.0:
                        candidates.append(x_an)
            
            # Water-filling: bring peak to base_max
            if base_max > 0 and k_peak != 2 * i:
                m = a[k_peak - i]
                B = conv[k_peak]
                if m > 1e-15 and B > base_max:
                    x_wf = a_i + (base_max - B) / (2.0 * m)
                    if 0 < x_wf < 100.0:
                        candidates.append(x_wf)
            
            best_x = a_i
            best_sc = 2.0 * n * max(base_max, float(conv[ks].max())) / (S * S)
            
            for x in candidates:
                new_vals = conv[ks].copy()
                mask_2i = (ks == 2 * i)
                if mask_2i.any():
                    new_vals[mask_2i] = conv[ks[mask_2i]] + x * x - a_i * a_i
                mask_not = ~mask_2i
                if mask_not.any():
                    new_vals[mask_not] = conv[ks[mask_not]] + 2.0 * (x - a_i) * a[offsets[mask_not]]
                new_vals = np.maximum(new_vals, 0.0)
                
                new_max = max(base_max, float(new_vals.max()))
                new_S = S_minus + x
                if new_S < 0.01:
                    continue
                sc = 2.0 * n * new_max / (new_S * new_S)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            
            if best_x != a_i:
                S = S - a_i + best_x
                conv[ks[(ks == 2*i)]] += best_x**2 - a_i**2
                mask_not = (ks != 2*i)
                conv[ks[mask_not]] += 2.0 * (best_x - a_i) * a[offsets[mask_not]]
                a[i] = best_x
        
        cur_score = 2.0 * n * conv.max() / (S * S)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            
        if (sweep + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep+1} ({elapsed:.0f}s): score={cur_score:.8f} best={best_score:.8f}")
    
    return best_a, best_score
