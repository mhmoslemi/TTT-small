**Algorithm: Iterative Peak Reduction (IPR)**

1. Start with an initial sequence a (e.g., slightly non-uniform).
2. Compute conv(a, a) and identify the peak position k*.
3. Compute the gradient of c_{k*} with respect to a: grad_j = a_{k*-j} (with appropriate boundary handling).

5. Project back to the feasible set (a_i >= 0, sum a = 1).
6. Repeat.

def grad_conv_at_k(a, k):
    n = len(a)
    grad = np.zeros(n)
    # Part 1: grad[m] = a[k-m] for 0 <= k-m < n
    # This means m ranges from max(0, k-n+1) to min(k, n-1)
    m_start = max(0, k - n + 1)
    m_end = min(k, n - 1)
    if m_start <= m_end:
        indices = np.arange(m_start, m_end + 1)
        grad[indices] = a[k - indices]
    # Part 2: if k is even and k//2 < n, add a[k//2] to grad[k//2]
    if k % 2 == 0 and k // 2 < n:
        grad[k // 2] += a[k // 2]
    return grad

def fast_autocorr(a):

def compute_smooth_max_subgrad(a, c, T):
    n = len(a)
    max_c = c.max()
    weights = np.exp(T * (c - max_c))
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(c))
        weights[np.argmax(c)] = 1.0
    else:
        weights = weights / w_sum

    N = 1 << (3 * n).bit_length()
    u = np.zeros(N)
    u[:2 * n - 1] = weights
    v = np.zeros(N)
    v[:n] = a
    U = rfft(u, N)
    V = rfft(v, N)
    corr = irfft(np.conj(U) * V, N)
    grad = 2.0 * corr[:n]
    return grad

        top_peaks = np.argsort(c)[::-1][:n_top]
        max_c = c[0]
        improved = False

        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k - idx

            for pos in range(len(idx)):
                j = idx[pos]
                o = other[pos]

                c_k_at_j = a[j] * a[o]
                if j == o:
                    c_k_at_j = a[j] ** 2

                if c_k_at_j < 1e-15:
                    continue

                for target in range(n):
                    if target == j:
                        continue
                    for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04, 0.07, 0.1]:
                        amt = a[j] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[j] = a[j] - amt
                        if j != o:
                            a_trial[o] = a[o] * (1.0 - frac * 0.3)
                        a_trial[target] += amt
                        if (a_trial < -1e-12).any():
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial_s = a_trial.sum()
                        if a_trial_s < 1e-15:
                            continue
                        a_trial = a_trial / a_trial_s
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            c = fast_autocorr(a)
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                n_perturb = rng.integers(1, 6)
                src = rng.choice(n, size=n_perturb, replace=False)
                dst = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for s_i, d_i in zip(src, dst):
                    frac = rng.uniform(0.002, 0.015)
                    amt = a_trial[s_i] * frac
                    a_trial[s_i] -= amt
                    a_trial[d_i] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s > 1e-15:
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        c = fast_autocorr(a)
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 3000:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_initializations(n, parent, seed=42):
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n)
    center = (n - 1) / 2.0
    norm_center = center if center > 0 else 1.0

    uniform = np.ones(n) / n
    inits.append(("uniform", uniform))

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits.append(("parent", p / ps))

    u_shaped = 1.0 / (1.0 + 0.3 * ((x - center) / norm_center) ** 2)
    inits.append(("u_shaped", u_shaped / u_shaped.sum()))

    u_shaped2 = 1.0 / (1.0 + 1.0 * ((x - center) / norm_center) ** 2)
    inits.append(("u_shaped2", u_shaped2 / u_shaped2.sum()))

    chirp = np.cos(np.pi * (x - center) ** 2 / (2.0 * n))
    chirp = chirp ** 2
    chirp = np.maximum(chirp, 0.001)
    inits.append(("chirp", chirp / chirp.sum()))

    chirp2 = np.cos(np.pi * (x - center) ** 2 / (4.0 * n))
    chirp2 = (1.0 + chirp2) / 2.0
    inits.append(("chirp2", chirp2 / chirp2.sum()))

    gauss = np.exp(-2.0 * ((x - center) / norm_center) ** 2)
    inits.append(("gauss", gauss / gauss.sum()))

    gauss2 = np.exp(-4.0 * ((x - center) / norm_center) ** 2)
    inits.append(("gauss2", gauss2 / gauss2.sum()))

    ramp = (x + 1).astype(np.float64)
    inits.append(("ramp", ramp / ramp.sum()))

    ramp2 = (n - x).astype(np.float64)
    inits.append(("ramp2", ramp2 / ramp2.sum()))

    for i in range(8):
        rand = rng.random(n) + 0.05
        inits.append(f"rand_{i}", rand / rand.sum())

    mixed1 = 0.5 * uniform + 0.5 * (1.0 / (1.0 + 0.5 * ((x - center) / norm_center) ** 2))
    inits.append(("mixed1", mixed1 / mixed1.sum()))

    mixed2 = 0.3 * uniform + 0.3 * gauss + 0.4 * (u_shaped / u_shaped.sum())
    inits.append(("mixed2", mixed2 / mixed2.sum()))

    sine_mod = 1.0 + 0.2 * np.sin(2.0 * np.pi * x / n)
    inits.append(("sine_mod", sine_mod / sine_mod.sum()))

    sine_mod2 = 1.0 + 0.4 * np.cos(4.0 * np.pi * x / n)
    sine_mod2 = np.maximum(sine_mod2, 0.01)
    inits.append(("sine_mod2", sine_mod2 / sine_mod2.sum()))

    tri = 1.0 - np.abs(2.0 * x / (n - 1) - 1.0)
    inits.append(("triangular", tri / tri.sum()))

    inv_tri = 1.0 / (0.5 + tri)
    inits.append(("inv_triangular", inv_tri / inv_tri.sum()))

    return inits
