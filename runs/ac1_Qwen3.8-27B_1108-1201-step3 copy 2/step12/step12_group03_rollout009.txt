def subgradient_descent(a, time_limit, lr_start=0.01, lr_end=0.0001, n_top=5):
    """Subgradient descent on max_l (a*a)[l] with top-peak averaging."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()

    best_a = a.copy()
    best_max = fast_conv(a).max()
    t0 = time.monotonic()
    n_iters = 0
    n_total_iters = 50000
    
    # Polyak averaging
    avg_a = a.copy()
    
    for it in range(n_total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = it / n_total_iters
        lr = lr_start * (lr_end / lr_start) ** progress
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Top peaks with softmax weights
        top_idx = np.argsort(b)[::-1][:n_top]
        w = b[top_idx] - max_b
        w = np.exp(w * 5.0)  # sharpen
        w = w / w.sum()
        
        # Compute subgradient: grad[j] = 2 * sum_l w_l * a[l-j] * I(0<=l-j<n)
        # = 2 * correlate(w_full, a, 'valid')[j] where w_full is b-weighted
        # Build full weight vector
        w_full = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_idx):
            w_full[idx] = w[i]
        
        grad = 2.0 * np.correlate(w_full, a, mode='valid')
        
        # Gradient step with projection to simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        # Polyak average
        avg_a = avg_a + (a_new - avg_a) / (it + 1)
        a = a_new
        n_iters += 1
        
        cur_max = fast_conv(a).max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        
        # Also check averaged iterate

            if avg_max < best_max:

            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.6f}")
        
        if stall > 2000 and t > 500:
            break
    
    # Final: compute the average of all iterates
    if len(iterates) > 10:
        avg_all = np.mean(iterates, axis=0)
        avg_all = np.maximum(avg_all, 0)
        avg_all = avg_all / avg_all.sum()
        avg_score = 2.0 * n * fast_conv(avg_all).max()
        if avg_score < best_score:
            best_score = avg_score
            best_a = avg_all.copy()
    
    return best_a, best_score


def qp_refine(a0, time_limit, n_active=20):
    """QP active-set refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    b = fast_conv(a)
    max_b = b.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * max_b
    n_active_current = n_active
    
    for rd in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        
        threshold = max_b * (1.0 - 0.005)
        active = np.where(b >= threshold)[0]
        if len(active) > n_active_current:
            # Keep top n_active_current by value
            active = np.argsort(b[active])[::-1][:n_active_current] + np.where(b >= threshold)[0]
            # Actually just take the top ones
            all_above = np.where(b >= threshold)[0]
            active = all_above[np.argsort(b[all_above])[::-1][:n_active_current]]
        
        # Build QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k - idx
            constraints.append(cp.sum(a_var[idx] * a_var[other]) <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iter=10000)
            except Exception:
                break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                break
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_conv(a)
            max_b = b.max()
        else:
            # Try mixing
            improved = False
            for frac in [0.1, 0.3, 0.5, 0.7]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix = a_mix / a_mix.sum()
                s_mix = 2.0 * n * fast_conv(a_mix).max()
                if s_mix < best_score:
                    best_score = s_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_conv(a)
                    max_b = b.max()
                    improved = True
                    break
            if not improved and rd >= 5:
                break
            if not improved:
                n_active_current = min(n_active_current + 10, 60)
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=0):
    """Coordinate descent: try moving mass from one position to another."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    
    # Pre-compute which positions affect which convolution lags
    # Position i contributes to conv lags i through i+n-1 (as a[i]*a[lag-i])
    # and also to lags 0 through i (as a[lag]*(a's contribution))
    # Actually position i contributes to conv[k] for k in [i, i+n-1] (via a[i]*a[k-i])
    # and for k in [0, i+n-1] (via a[k-i]*a[i])... 
    # Simplified: position i contributes to conv[k] where max(0,k-n+1) <= i <= min(n-1,k)
    # i.e., k in range [i, i+n-1] (since k-i must be in [0,n-1], so k in [i, i+n-1])
    # AND k in range [i-(n-1), i] (since i-k must be in [0,n-1]... wait)
    
    # Actually conv[k] = sum_i a[i]*a[k-i] where 0<=i<n and 0<=k-i<n
    # So a[i] appears in conv[k] when 0<=k-i<n, i.e., i<=k<=i+n-1
    # So position i affects conv lags from i to i+n-1
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        max_b = b.max()
        peak_lags = np.argsort(b)[::-1][:5]
        
        improved = False
        # Try moving mass from positions that contribute to peaks to positions that don't
        for k in peak_lags:
            # Positions contributing to this lag: i in [max(0,k-n+1), min(n-1,k)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            for src in range(js, je + 1):
                src_other = k - src
                contrib = a[src] * a[src_other]
                if contrib < 1e-8:
                    continue
                
                # Find target: position that contributes least to top peaks
                best_target = -1
                best_gain = 0
                for _ in range(20):  # random targets
                    dst = rng.integers(0, n)
                    if dst == src or dst == src_other:
                        continue
                    # Estimate: reducing contribution to lag k by amt*(a[src_other]+a[src])
                    # and adding to other lags
                    amt = min(a[src], a[src_other]) * 0.01
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    if src != src_other:
                        a_trial[src_other] -= amt * 0.5
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_conv(a_trial).max()
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 50 == 0:
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
                amt = a[src] * 0.05
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_conv(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_initialization(n, mode='cosine'):
    """Get an initial sequence of given length."""
    if mode == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 'gaussian':
        x = np.arange(n) - n / 2
        a = np.exp(-x**2 / (2 * (n/4)**2))
    elif mode == 'super_gaussian':
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 4
    elif mode == 'sinc_clipped':
        x = np.linspace(-1, 1, n)
        a = np.sinc(x * 2)
        a = np.maximum(a, 0)
    elif mode == 'kaiser_like':
        # Approximate Kaiser window
        from scipy.special import i0
        x = np.linspace(-1, 1, n)
        beta = 5.0
        a = i0(beta * np.sqrt(np.maximum(1 - x**2, 0))) / i0(beta)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a
