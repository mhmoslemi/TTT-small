    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_obj_grad(z, T):
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z

def direct_max_optimize(a0, time_limit, n_active_frac=0.05):
    """
    Optimize the max convolution peak directly using subgradient descent
    with exponential (mirror) steps. Only consider the top peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 0.1  # initial step size
    n_active = max(1, int(n_active_frac * (2*n-1)))
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Find active peaks
        threshold = b_max - 1e-4 * b_max  # peaks within 0.01% of max
        active = np.where(b >= threshold)[0]
        if len(active) < n_active:
            active = np.argsort(b)[::-1][:n_active]
        
        # Compute subgradient: average of gradients from active peaks
        # grad[i] = 2 * sum_{k in active, k-i valid} a[k-i]
        # This is 2 * (1/|active|) * sum_k active a_rev shifted
        grad = np.zeros(n)
        a_rev = a[::-1]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je+1] += a_rev[k-je:k-js+1]
        grad *= 2.0 / len(active)
        
        # Multiplicative update (mirror descent with entropy)
        eta_new = eta * (1.0 / (1.0 + 0.01))  # slightly decrease step size
        a = a * np.exp(-eta_new * grad)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        eta = eta_new
        
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

To reduce the max of `b`, I want to find `δa` (with `sum(δa) = 0`, `a + δa >= 0`) such that the maximum of `b + 2*convolve(δa, a_rev)` is minimized.

This is exactly the linearized peak LP from the previous code! But it's slow because it requires solving an LP.

**Faster approximation**: Instead of the full LP, I can use a greedy approach:
1. Identify the peak positions
2. Find the direction `δa` that most reduces the peak (by looking at the gradient)
3. Take a step in that direction

The gradient of `max(b)` with respect to `a` (at the active peaks) gives the direction to move. The mirror descent step `a * exp(-η * grad)` naturally moves mass from high-gradient positions to low-gradient positions.

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive step size
    eta = 1.0
    decay = 0.999
    patience = 0
    best_so_far = best_score
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        score = 2.0 * n * b_max
        
        # Select active peaks (top fraction)
        n_active = max(1, int(n_active_frac * (2*n - 1)))
        active = np.argsort(b)[::-1][:n_active]
        
        # Compute subgradient at active peaks
        # grad[i] = 2 * mean_{k in active} a[k-i] (for valid k-i)
        grad = np.zeros(n)
        a_rev = a[::-1]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # a[k-j] for j in [js, je] = a_rev[je-k+1 ... k-js]
            # Actually a[k-j] for j in [js, je]: when j=js, a[k-js]; when j=je, a[k-je]
            # a_rev[m] = a[n-1-m], so a[k-j] = a_rev[n-1-(k-j)] = a_rev[n-1-k+j]
            grad[js:je+1] += a_rev[n-1-k+js : n-1-k+je+1]
        grad /= len(active)
        
        # Mirror descent step
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        
        # Check improvement
        b_new = fast_autocorr(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            # Backtrack: take a smaller step
            for factor in [0.5, 0.25, 0.1]:

                a_trial = np.maximum(a_trial, 1e-30)

                s_trial = 2.0 * n * fast_autocorr(a_trial).max()
                if s_trial < score:
                    a_new = a_trial
                    break
            else:
                a_new = a  # no improvement, keep current
        
        a = a_new
        eta *= decay
        
        if patience > 200:

            a = a + 0.01 * rng.normal(size=n)

            a = a / a.sum()
            patience = 0
            eta = 0.5

`b_k = sum_j a_j * a_{k-j}`

`∂b_k/∂a_i`: The term where `j=i` gives `a_{k-i}`. The term where `k-j=i`, i.e., `j=k-i`, gives `a_i`. If `i = k-i` (i.e., `k=2i`), these are the same term and the derivative is `2*a_i`. Otherwise, the derivative is `a_{k-i} + a_{k-i}`... no wait.

`b_k = sum_j a_j * a_{k-j}`

If I vary `a_i`:
- In the sum, the term with `j=i` is `a_i * a_{k-i}`, derivative w.r.t. `a_i` is `a_{k-i}`
- The term with `k-j=i` means `j=k-i`, which is `a_{k-i} * a_i`, derivative w.r.t. `a_i` is `a_{k-i}`
- If `i = k-i` (i.e., `k = 2i`), these are the same term `a_i * a_i = a_i^2`, derivative is `2*a_i`
- If `i ≠ k-i`, we get two separate terms, total derivative is `a_{k-i} + a_{k-i} = 2*a_{k-i}`

So: `∂b_k/∂a_i = 2 * a_{k-i}` if `0 ≤ k-i < n` and `k ≠ 2i`
                    `2 * a_i` if `k = 2i` (which is the same as `2 * a_{k-i}` since `k-i = i`)

So in all cases: `∂b_k/∂a_i = 2 * a_{k-i}` if `0 ≤ k-i < n`, else 0.

This is much simpler! The gradient of `b_k` w.r.t. `a_i` is just `2 * a_{k-i}` (when valid).

So the subgradient of `max_k b_k` at `a` is:
`g_i = 2 * (1/|S|) * sum_{k in S} a_{k-i}` where `S` is the set of active peaks (those achieving the max).

Or for the smoothed version with weights `w_k`:
`g_i = 2 * sum_k w_k * a_{k-i}`

This can be computed as: `g = 2 * convolve(w, a_rev)[n-1 : n-1+n]` where `a_rev = a[::-1]`.

Wait, let me verify. `convolve(w, a_rev)[k] = sum_j w_j * a_rev[k-j] = sum_j w_j * a[n-1-(k-j)] = sum_j w_j * a[n-1-k+j]`.

I want `sum_k w_k * a_{k-i}`. Let me substitute: this equals `sum_k w_k * a_rev[n-1-(k-i)] = sum_k w_k * a_rev[n-1-k+i]`.

Now, `convolve(w, a_rev)[m] = sum_k w_k * a_rev[m-k]`. Setting `m-k = n-1-k+i` gives `m = n-1+i`.

So `sum_k w_k * a_{k-i} = convolve(w, a_rev)[n-1+i]`.

Therefore: `g_i = 2 * convolve(w, a_rev)[n-1+i]` for `i = 0, ..., n-1`.

This is `g = 2 * convolve(w, a_rev)[n-1 : n-1+n]`.

Great, this can be computed with FFT in O(n log n).

Now, for the mirror descent step:
`a_i_new = a_i * exp(-η * g_i) / Z`

where `Z = sum_j a_j * exp(-η * g_j)`.

This is the exponential gradient (ExponGrad) or multiplicative weights update.

The step size η is crucial. For the max problem, I should use a decreasing step size like `η_t = c / sqrt(t)` (standard for subgradient methods) or `η_t = c / t` (faster convergence but requires strong convexity).

Actually, since we're using the mirror map (entropy), the convergence rate for smooth objectives is O(log(n)/T) and for non-smooth (like max) is O(sqrt(n)/T) where T is the number of iterations.

For n=1600, to get convergence to 0.001 precision, we'd need about `sqrt(1600) / 0.001 = 40,000` iterations. With each iteration being O(n log n) ≈ O(16000) operations, that's about 640 million operations, which should take a few seconds in numpy. So it's feasible!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LogSumExp-smoothed L-BFGS optimization (primary method)
2. Mirror descent with exponential steps for direct max optimization
3. Multi-length exploration with interpolation
4. Coordinate descent pair moves for final polish

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

TOTAL_TIME = 1800.0
SAFETY_MARGIN = 30.0  # seconds

def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv(a, b):
    """Full convolution of a and b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def normalize(a):
    """Normalize to sum=1, clip non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def lse_obj_and_grad(z, T):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    # Gradient
    a_rev = a[::-1]
    g = 2.0 * fast_conv(w, a_rev)[n-1:n-1+n]
    S = np.dot(g, a)
    grad_z = a * (g - S)
    return obj_val, grad_z

def logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=12):
    """LogSumExp-smoothed optimization with L-BFGS and geometric annealing."""
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    
    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        stage_T = temps[si]
        
        # Allocate iterations proportional to remaining time
        frac_remaining = remaining / time_limit if time_limit > 0 else 1
        iters = max(5, int(30 + 100 * si / max(1, n_stages - 1)))
        
        def func(z_in):
            val, _ = lse_obj_and_grad(z_in, stage_T)
            return val
        
        def jac(z_in):
            _, g = lse_obj_and_grad(z_in, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 100,
                         'maxls': 50,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception as e:
            pass
        
        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/{n_stages} T={stage_T:.4f} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def mirror_descent_max(a0, time_limit):
    """
    Direct max optimization using mirror descent (exponential gradient).
    Uses soft active peak selection with decreasing temperature.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Parameters
    T_act = 1.0  # Temperature for active peak weighting
    T_act_min = 0.01
    T_act_decay = 0.999
    eta = 0.5  # Step size
    eta_decay = 0.9995
    patience = 0
    max_patience = 500
    
    a_sum = a.copy()  # For averaged iterate
    n_accepted = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        score = 2.0 * n * b_max
        
        # Soft active peak selection
        Tb = T_act * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # Compute subgradient
        a_rev = a[::-1]
        g = 2.0 * fast_conv(w, a_rev)[n-1:n-1+n]
        
        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Evaluate
        b_new = fast_autocorr(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            a = a_new
            a_sum += a_new
            n_accepted += 1
        else:
            patience += 1
            # Try smaller step
            for factor in [0.5, 0.1, 0.01]:
                log_a_trial = log_a - eta * factor * g
                log_a_trial -= log_a_trial.max()
                a_trial = np.exp(log_a_trial)
                a_trial = a_trial / a_trial.sum()
                s_trial = 2.0 * n * fast_autocorr(a_trial).max()
                if s_trial < score:
                    a = a_trial
                    break
            else:
                a = a_new  # accept anyway to continue
                a_sum += a_new
                n_accepted += 1
        
        # Schedule
        T_act = max(T_act_min, T_act * T_act_decay)
        eta = max(0.001, eta * eta_decay)
        
        if patience > max_patience:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 10000))
            a = a + 0.005 * rng.normal(size=n)
            a = normalize(a)
            patience = 0
            eta = 0.3
    
    # Also check the averaged iterate
    if n_accepted > 100:
        a_avg = a_sum / n_accepted
        score_avg = compute_score(a_avg)
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum() if f.sum() > 1e-15 else f
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)

def window_init(n, kind='blackman'):
    """Generate a window function of length n, normalized to sum=1."""
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'triangular':
        a = np.minimum(np.arange(n), n-1-np.arange(n)).astype(float) + 1
        # Actually use scipy's triangle
        a = np.minimum(2.0 * np.arange(n) / n, 2.0 - 2.0 * np.arange(n) / n)
    elif kind == 'hann':
        a = np.hanning(n)
    elif kind == 'hamming':
        a = np.hamming(n)
    elif kind == 'blackman':
        a = np.blackman(n)
    elif kind == 'blackmanharris':
        a = np.array([0.35875 - 0.48829*np.cos(2*np.pi*i/n) + 0.14128*np.cos(4*np.pi*i/n) - 0.01168*np.cos(6*np.pi*i/n) for i in range(n)])
    elif kind == 'kaiser':
        from scipy.signal import get_window
        a = get_window('kaiser', n, beta=5.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    return normalize(a)

def coordinate_descent_pairs(a0, time_limit, n_trials_per_iter=20):
    """
    Coordinate descent: try moving mass from one position to another.
    Focus on positions that contribute to peaks.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Find peak positions
        peak_positions = np.argsort(b)[::-1][:10]
        
        # Find positions contributing to peaks
        contrib_set = set()
        for k in peak_positions:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                contrib_set.add(j)
        
        # Candidate sources: positions with high values that contribute to peaks
        sources = [j for j in contrib_set if a[j] > 0.001]
        # Candidate targets: positions that contribute least to peaks
        non_contrib = [j for j in range(n) if j not in contrib_set]
        
        improved = False
        for _ in range(n_trials_per_iter):
            if not sources or not non_contrib:
                break
            
            src = sources[rng.integers(len(sources))]
            dst = non_contrib[rng.integers(len(non_contrib))]
            
            frac = rng.uniform(0.001, 0.02)
            amt = a[src] * frac
            
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a
