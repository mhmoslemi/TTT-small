    """Solve min_b max_k (a*b)_k using first-order mirror descent."""
    n = len(a_fixed)
    b = np.ones(n) / n
    fft_size = 1 << (2*n-1).bit_length()
    A = rfft(a_fixed, fft_size)
    
    best_b = b.copy()
    best_M = (a_fixed * b).max() # need to compute convolution
    
    for it in range(n_iter):
        # Compute convolution a*b using FFT
        B = rfft(b, fft_size)
        conv = irfft(A * B, fft_size)[:2*n-1]
        M_val = conv.max()
        
        if M_val < best_M:
            best_M = M_val
            best_b = b.copy()
        
        # Subgradient: find the argmax lag
        k_star = np.argmax(conv)
        # Gradient of (a*b)_{k*} w.r.t. b is: a[k*-j] for valid j
        # This is a shifted/reversed version of a
        js = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        grad = np.zeros(n)
        if len(js) > 0:
            grad[js] = a_fixed[k_star - js]
        
        # Mirror descent step on the simplex
        eta = eta0 / (1 + 0.1 * it)
        b_new = np.exp(-eta * grad)
        b_new = b_new / b_new.sum()
        b = b_new
    
    return best_b, best_M

def fast_autocorr(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def evaluate_safe(arr):

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_value_grad_a(a, T):
    """Compute LogSumExp value and gradient w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    return val, grad_a


def lse_lbfgs_wrap(z, T, n):
    """Objective and gradient for L-BFGS-B optimizer (z-parametrization)."""
    a = softmax_fn(z)
    val, grad_a = compute_lse_value_grad_a(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return val, grad_z

def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: iteratively flatten power spectrum to reduce peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fast_first_order_lp(a_fixed, n_iter=5000, eta0=1.0):
    """Fast first-order method for min_b max_k (a*b)_k using mirror descent."""
    n = len(a_fixed)
    b = np.ones(n) / n
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a_fixed, fft_size)

    best_b = b.copy()
    best_M = np.max(irfft(A * rfft(b, fft_size), fft_size)[:2*n-1])

    for it in range(n_iter):
        B = rfft(b, fft_size)
        conv = irfft(A * B, fft_size)[:2*n-1]
        M_val = conv.max()

        if M_val < best_M:
            best_M = M_val
            best_b = b.copy()

        k_star = int(np.argmax(conv))
        js = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        grad = np.zeros(n)
        if len(js) > 0:
            grad[js] = a_fixed[k_star - js]

        eta = eta0 / (1.0 + 0.05 * it)
        log_b = -eta * grad
        log_b -= log_b.max()
        b_new = np.exp(log_b)
        s = b_new.sum()
        if s < 1e-300:
            break
        b = b_new / s

    return best_b, best_M


def ftrl_lp_optimize(a0, time_limit, solver, max_iters=100000, use_fast_lp_every=5):
    """FTRL-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        if t % use_fast_lp_every == 0:
            b_star, M_val = fast_first_order_lp(y, n_iter=2000, eta0=1.0)
        else:
            b_star, M_val = solve_lp_best_response(y, solver)
            if b_star is None:
                b_star, M_val = fast_first_order_lp(y, n_iter=2000, eta0=1.0)

        if b_star is None:
            continue

        # FTRL update
        eta_t = 1.0 / np.sqrt(t)
        y_new = y + eta_t * (b_star - y)
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        # Extrapolation step
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL-LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def adaptive_peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with golden-section line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    golden = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = int(idx[src_local])

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j in range(js2, je2 + 1):
                    peak_areas.add(j)

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = int(dst_pool[rng.integers(len(dst_pool))])

            # Triangle search for optimal transfer fraction
            if a[src] < 1e-12:
                continue
            f_min = 0.0
            f_max = 0.3
            c = f_max - golden * (f_max - f_min)
            d = f_min + golden * (f_max - f_min)

            def score_at_frac(f):
                amt = a[src] * f
                if amt < 1e-15:
                    return best_score
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    return best_score
                a_trial = a_trial / a_trial.sum()
                return 2.0 * n * fast_autocorr(a_trial).max()

            fc = score_at_frac(c)
            fd = score_at_frac(d)

            for _ in range(12):
                if fc < fd:
                    f_max = d
                    d = c
                    fd = fc
                    c = f_max - golden * (f_max - f_min)
                    fc = score_at_frac(c)
                else:
                    f_min = c
                    c = d
                    fc = fd
                    d = f_min + golden * (f_max - f_min)
                    fd = score_at_frac(d)

            f_opt = (f_min + f_max) / 2
            amt = a[src] * f_opt
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall < 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            else:
                break

    return best_a, best_score


def subgradient_max_descent(a0, time_limit, n_iter=50000):
    """Direct subgradient descent on max autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_val = b.max()
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Find all lags achieving (near) the max
        threshold = max_val - 1e-10 * max_val
        peak_lags = np.where(b >= threshold)[0]

        # Compute subgradient: average gradient over peak lags
        grad = np.zeros(n)
        for k in peak_lags:
            js = np.arange(max(0, k - n + 1), min(n - 1, k) + 1)
            if len(js) > 0:
                grad[js] += a[k - js]
        grad = 2.0 * grad / len(peak_lags)

        # Step size (Polyak type)
        eta = 0.01 / (1.0 + 0.001 * it)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        a = a_new

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad it={it} ({elapsed:.0f}s): score={score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [peak_shave_1]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 9: One more subgradient pass (5% of time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t9 = min(remaining * 0.05, 40.0)
        if t9 > 3:
            a_out, _ = subgradient_max_descent(best_seq, t9, n_iter=20000)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                elapsed = time.monotonic() - t_global_start
                print(f"  *** NEW BEST [subgrad_final]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 10: Final peak shaving (5% of time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 5:
        t10 = min(remaining * 0.05, 40.0)
        if t10 > 3:
            a_out, _ = adaptive_peak_shaving(best_seq, t10, seed=789)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                elapsed = time.monotonic() - t_global_start
                print(f"  *** NEW BEST [peak_shave_2]: {best_score:.6f} (t={elapsed:.0f}s)")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.maximum(best_seq, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
