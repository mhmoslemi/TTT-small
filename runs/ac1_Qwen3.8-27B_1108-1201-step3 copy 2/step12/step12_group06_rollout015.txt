import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar

def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None

def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]

            except Exception:

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def subgradient_at_peak(a, k):
    """Compute subgradient of (a*a)_k w.r.t. a. O(n) operation.
    g[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        g[i] = 2.0 * a[k - i]
    return g


def subgradient_at_peak_fast(a, k):
    """Faster vectorized subgradient computation."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lse_objective_grad(z_in, T, n):
    """Compute LogSumExp objective and gradient w.r.t. z (softmax params)."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    # Gradient of max_k (a*a)_k w.r.t. a, weighted by softmax weights
    grad_a = np.zeros(n)
    # Only iterate over significant weights
    threshold = w.max() * 1e-10
    active = np.where(w > threshold)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsbfgs_lse_adaptive(a0, time_limit, T_start=80.0, T_min=0.3, n_stages=25, iters_per_stage=40):
    """Adaptive LogSumExp L-BFGS with progress-based temperature control."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            if best_score > 0:
                obj_ratio = (sc - best_score) / (best_score + 1e-15)
                if obj_ratio > 0.003:
                    stagnant += 1
                else:
                    stagnant = max(0, stagnant - 1)

            if stagnant >= 3:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            if (stage + 1) % 5 == 0:
                print(f"    LSE-Adaptive stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_subgradient_v2(a0, time_limit, n_steps=500000, top_fraction=0.05):
    """Efficient peak-targeted subgradient with Polyak step and iterate averaging.
    Uses O(n) subgradient computation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()

    # Exponential moving average of iterates
    avg_a = a.copy()
    avg_alpha = 0.995

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks
        threshold = max_b * (1.0 - top_fraction)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        # Compute averaged subgradient (O(n) per peak, but vectorized)
        g = np.zeros(n)
        for k in active:
            g += subgradient_at_peak_fast(a, k)
        g /= len(active)

        # Project to tangent space of simplex
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Polyak step size
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.3)

        # Step in z-space
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Backtrack if needed
        if sc_new > cur_score * 1.005:
            for _ in range(6):
                eta /= 2.0
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                sc_new = score_normalized(a_new)
                if sc_new <= cur_score:
                    break

        z = z_new
        a = a_new

        # Update average (exponential)
        avg_a = avg_alpha * avg_a + (1 - avg_alpha) * a
        avg_a = avg_a / avg_a.sum()

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_cycles=100):
    """Coordinate descent with exact 1D minimization per coordinate.
    For each coordinate i, minimize max_k [(a*a)_k with a_i = x] over x >= 0,
    then renormalize.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for cycle in range(n_cycles):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution
        b_cur = fast_convolve(a)
        cur_max = b_cur.max()

        # Process coordinates in order of potential impact (high gradient first)
        # Use a fixed cyclic order for simplicity, or randomize
        order = np.arange(n)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Compute the function f(x) = max_k g_k(x) where g_k is (a*a)_k with a_i=x
            # g_k(x) = h_k + l_k * x + q_k * x^2
            # where for k in [0, 2n-2]:
            #   if k = 2*i: q_k=1, l_k=0, h_k = b_cur[k] - a[i]^2
            #   if 0<=k-i<n and k!=2*i: q_k=0, l_k=2*a[k-i], h_k = b_cur[k] - 2*a[i]*a[k-i]
            #   else: q_k=0, l_k=0, h_k = b_cur[k]

            # Build the piecewise function parameters
            # The range of x is [0, x_max] where x_max = 1 - (sum(a) - a[i]) = a[i] (since sum=1)
            # Actually, we allow x in [0, 1] and renormalize at the end of the cycle
            # But to keep sum <= 1, x <= 1 - (sum(a) - a[i]) = a[i] + (1-sum(a)) = a[i]
            # Since sum(a) = 1, x_max = a[i]... that's too restrictive.
            # Instead, let's allow x in [0, 2*a[i]+0.1] and renormalize.
            x_max = max(a[i] * 3.0, 0.01)

            # Compute h, l, q arrays
            # h_k = b_cur[k] - (terms involving a[i])
            # l_k = coefficient of x
            # q_k = coefficient of x^2

            # For efficiency, compute the "affected" k values
            # k values where 0 <= k-i < n: k in [i, i+n-1]
            lo_k = i
            hi_k = min(i + n - 1, 2 * n - 2)

            h = b_cur.copy()
            l = np.zeros(2 * n - 1)
            q = np.zeros(2 * n - 1)

            for k in range(lo_k, hi_k + 1):
                ki = k - i  # This is the index in a: a[ki]
                if k == 2 * i:
                    h[k] = b_cur[k] - a[i] * a[i]
                    q[k] = 1.0
                else:
                    h[k] = b_cur[k] - 2.0 * a[i] * a[ki]
                    l[k] = 2.0 * a[ki]

            # Now minimize f(x) = max_k [h[k] + l[k]*x + q[k]*x^2] for x in [0, x_max]
            # This is a convex piecewise-quadratic function.
            # Use scipy's bounded scalar minimization.

            def f_1d(x_val):
                vals = h + l * x_val
                mask = (q > 0)
                if np.any(mask):
                    vals[mask] += q[mask] * x_val * x_val
                return vals.max()

            try:
                res = minimize_scalar(f_1d, bounds=(0.0, x_max), method='bounded',
                                      options={'xatol': 1e-12, 'maxiter': 50})
                x_opt = res.x
            except Exception:
                x_opt = a[i]

            # Update
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = max(0.0, x_opt)
                improved = True

        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if not improved:
            break

        if (cycle + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD cycle={cycle+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.15 * np.cos(2 * np.pi * 5 * np.arange(n) / n))
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.03)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_and_normalize(a, sigma, rng):
    """Add Gaussian noise and renormalize."""
    a = np.asarray(a, dtype=np.float64)
    noisy = a + sigma * rng.normal(0, 1, len(a))
    noisy = np.maximum(noisy, 0)
    s = noisy.sum()
    if s < 1e-15:
        return a.copy()
    return noisy / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    t_start = time.monotonic()

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(1200, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP at base length (25% time) ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} ===")
    remaining = global_deadline - time.monotonic()
    t1 = min(0.25 * (budget_s - 25), remaining * 0.6)
    if t1 > 15:
        a_out, _ = nesterov_lp(best_seq, t1, solver=solver)
        try_update(a_out, "nesterov_main")

    # === Phase 2: Adaptive LSE L-BFGS (20% time) ===
    print(f"\n=== Phase 2: Adaptive LSE ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.20 * (budget_s - 25), remaining * 0.5)
    if t2 > 20:
        a_lse1, _ = lsbfgs_lse_adaptive(best_seq, t2 * 0.5, T_start=60.0, T_min=0.2,
                                         n_stages=20, iters_per_stage=30)
        try_update(a_lse1, "lse_pass1")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # From perturbed start
            perturbed = perturb_and_normalize(best_seq, 0.03, rng)
            a_lse2, _ = lsbfgs_lse_adaptive(perturbed, remaining * 0.6, T_start=30.0,
                                             T_min=0.15, n_stages=15, iters_per_stage=25)
            try_update(a_lse2, "lse_perturbed")

    # === Phase 3: Peak-targeted subgradient (20% time) ===
    print(f"\n=== Phase 3: Peak Subgradient ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.20 * (budget_s - 25), remaining * 0.5)
    if t3 > 15:
        a_sg, _ = peak_subgradient_v2(best_seq, t3, n_steps=500000, top_fraction=0.03)
        try_update(a_sg, "peak_sg")

        # Second pass with different parameters
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_sg2, _ = peak_subgradient_v2(best_seq, remaining * 0.7, n_steps=300000,
                                            top_fraction=0.08)
            try_update(a_sg2, "peak_sg_wide")

    # === Phase 4: Multi-length exploration (20% time) ===
    print(f"\n=== Phase 4: Multi-length exploration ===")
    remaining = global_deadline - time.monotonic()
    t4_total = min(0.20 * (budget_s - 25), remaining * 0.6)
    if t4_total > 30:
        lengths_to_try = [n_base + 400, n_base + 800, n_base + 1500]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 20:
                break
            t_alloc = min(t4_total / 3.0, remaining * 0.45)
            if t_alloc < 10:
                break
            print(f"  Trying n={n_try}, budget={t_alloc:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)

            # Quick Nesterov
            a_try, _ = nesterov_lp(init_try, t_alloc * 0.5, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}_nest")

            # Quick subgradient
            remaining = global_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = peak_subgradient_v2(a_try, remaining * 0.5, n_steps=100000,
                                                 top_fraction=0.05)
                try_update(a_try2, f"len_{n_try}_sg")

    # === Phase 5: Coordinate descent polishing (10% time) ===
    print(f"\n=== Phase 5: CD Polishing ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.10 * (budget_s - 25), remaining * 0.6)
    if t5 > 10:
        # Run CD on the best sequence
        # Use a subset of coordinates for speed (top movers)
        a_cd, _ = coordinate_descent_polish(best_seq, t5 * 0.7, n_cycles=50)
        try_update(a_cd, "cd_polish")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            a_cd2, _ = coordinate_descent_polish(best_seq, remaining * 0.6, n_cycles=30)
            try_update(a_cd2, "cd_polish2")

    # === Phase 6: Final combined polish (5% time) ===
    print(f"\n=== Phase 6: Final polish ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 10:
        # Short Nesterov
        a_f1, _ = nesterov_lp(best_seq, remaining * 0.3, solver=solver, max_iters=20000)
        try_update(a_f1, "final_nest")

        remaining = global_deadline - time.monotonic()
        if remaining > 8:
            # Short LSE from low T
            a_f2, _ = lsbfgs_lse_adaptive(best_seq, remaining * 0.5, T_start=8.0,
                                           T_min=0.1, n_stages=8, iters_per_stage=20)
            try_update(a_f2, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in best_seq]
