                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lse_obj_grad(z_in, T_val, n, top_K=None):
    """LSE-smoothed objective and gradient."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    size = len(b)
    
    if top_K is not None and top_K < size:
        # Only smooth the top K values
        # Create a masked version
        idx = np.argsort(b)[-top_K:]
        b_masked = np.full(size, b.min() - 1.0)  # Set non-top values to very low
        b_masked[idx] = b[idx]
        # Actually, let's just use the top K in the LSE
        b_top = b[idx]
        max_b = b_top.max()
        exp_b = np.exp(b_top - max_b)
        w_top = exp_b / exp_b.sum()
        log_sum_exp = max_b + np.log(exp_b.sum())
        
        obj = (2.0 * n / T_val) * log_sum_exp
        
        # Gradient: only from top K peaks
        grad_a = np.zeros(n)
        for i, k in enumerate(idx):
            if w_top[i] < 1e-20:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx2 = np.arange(js, je + 1)
            grad_a[idx2] += 2.0 * w_top[i] * a[k - idx2]
    else:
        max_b = b.max()
        if max_b > 700:
            max_b = 700
        exp_b = np.exp(b - max_b)
        w = exp_b / exp_b.sum()
        log_sum_exp = max_b + np.log(exp_b.sum())
        
        obj = (2.0 * n / T_val) * log_sum_exp
        
        grad_a = np.zeros(n)
        for k in range(size):
            if w[k] < 1e-20:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx2 = np.arange(js, je + 1)
            grad_a[idx2] += 2.0 * w[k] * a[k - idx2]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lse_hessian_approx(z_in, T_val, n, top_K=None):
    """Approximate Hessian of LSE objective (diagonal approximation)."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    
    if top_K is not None and top_K < len(b):
        idx = np.argsort(b)[-top_K:]
        b_top = b[idx]
        max_b = b_top.max()
        exp_b = np.exp(b_top - max_b)
        w_top = exp_b / exp_b.sum()
    else:
        max_b = b.max()
        if max_b > 700:
            max_b = 700
        exp_b = np.exp(b - max_b)
        w = exp_b / exp_b.sum()
        # Diagonal of Hessian approximation
        H_diag = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-20:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx2 = np.arange(js, je + 1)
            contrib = 2.0 * w[k] * a[k - idx2]
            H_diag[idx2] += 2.0 * contrib * (1 - contrib / a[idx2].max())  # rough
        return H_diag
    
    return None

def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000, K_start=50, K_end=200):
    """Nesterov LP with adaptive top-K and momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    stall_count = 0
    best_score_history = []
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive K: start small, grow over time
        progress = min(1.0, it / 5000.0)
        K = int(K_start + (K_end - K_start) * progress)
        
        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a
        
        b_new, M_lp = solve_lp_topk(y, solver, K=K)
        if b_new is None:
            continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive restart based on stall
        if stall_count > 200 and t_nest > 3.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
            stall_count = 0
            # Small random perturbation to escape
            if restart_count % 3 == 0:
                eps = 0.01 * (1.0 / (1.0 + restart_count * 0.1))
                pert = np.random.default_rng(restart_count * 12345).normal(0, eps, n)
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0)
                sp = a_pert.sum()
                if sp > 1e-15:
                    a = a_pert / sp
                    a_prev = a.copy()
                    continue
        
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new
        
        a = a_new
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP it={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f} K={K} rsts={restart_count}")
                last_print = elapsed
    
    return best_a, best_score

def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=30, top_K=None):
    """LSE L-BFGS with adaptive stages."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    # Logarithmic temperature schedule
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func_staged(z_val):
            obj, _ = lse_obj_grad(z_val, stage_T, n, top_K)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_obj_grad(z_val, stage_T, n, top_K)
            return grad
        
        # More iterations for later stages (lower temperature)
        if si < n_stages // 3:
            iters = 30
        elif si < 2 * n_stages // 3:
            iters = 50
        else:
            iters = 80
        
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = z_new
            else:
                # Still update z for continuity but keep best
                z = z_new
            if (si + 1) % 10 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: score={best_score:.8f}")
        except Exception:
            pass
    
    return best_a, best_score

def peak_flatness_refine(a0, time_limit, max_steps=30000):
    """Minimize the sum of top K peaks (flatter peak distribution)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.02
    K = 20  # Number of top peaks to target
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Get top K peaks
        topk = np.argpartition(b, -K)[-K:]
        
        # Compute gradient targeting these peaks
        # Weight peaks by how much they exceed the "target" level
        target = b.max() * 0.95
        weights = np.maximum(b[topk] - target, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(K) / K
        else:
            weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for i, k in enumerate(topk):
            if weights[i] < 1e-20:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += weights[i] * a[k - idx]
        
        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa
        
        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            eta = min(eta * 1.005, 0.1)
        else:
            eta *= 0.9
            if eta < 1e-12:
                break
    
    return best_a, best_score

def subgradient_fast(a0, time_limit, max_steps=100000):
    """Fast subgradient descent in z-space."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.5
    last_improve = 0
    t_nest = 1.0
    z_prev = z.copy()
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.98)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)
        
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        # Nesterov in z-space
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_prev)
        else:
            z_extrap = z
        
        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z_prev = z
            z = z_new
            a = a_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.002:
                t_nest = 1.0
                eta *= 0.95
            else:
                z_prev = z
                z = z_new
                a = a_new
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            eta = max(eta * 0.9995, 0.05)
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
