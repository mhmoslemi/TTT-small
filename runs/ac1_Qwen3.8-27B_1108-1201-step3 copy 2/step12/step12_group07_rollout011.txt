def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):

    return T

def spectral_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    # Random phase in frequency domain
    fft_size = 1 << (n - 1).bit_length()
    phases = 2 * np.pi * rng.random(fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    # Make it Hermitian for real output
    full_spec = np.zeros(fft_size, dtype=complex)
    full_spec[:fft_size // 2 + 1] = spectrum
    for i in range(1, fft_size // 2 + 1):
        full_spec[fft_size - i] = np.conj(spectrum[i])
    a = np.real(ifft(full_spec))
    a = np.abs(a)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def nesterov_lp(a0, time_limit, solver=None, use_dual=False, max_iters=500000):
    """Nesterov acceleration with LP best response and adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    consecutive_bad = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        # Solve LP best response
        b_new = solve_lp_best_response(y, solver, use_dual=use_dual)
        if b_new is None:
            # Fallback: use current a
            b_new = a.copy()

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            consecutive_bad = 0
        else:
            consecutive_bad += 1

        # Adaptive restart
        sc_cur = score_normalized(a)
        if (sc_new > sc_cur * 1.001 and t_nest > 3.0) or consecutive_bad > 200:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
            consecutive_bad = 0
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > 10:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")

    return best_a, best_score


def peak_flatten_refine(a0, time_limit, top_k_frac=0.05, max_steps=30000):
    """Targeted refinement: reduce the top peaks of the autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve = 0
    stall_count = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Target peaks above threshold
        threshold = max_b * (1.0 - top_k_frac)
        peak_indices = np.where(b >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(b))])

        # Compute subgradient of max at peak points (unified over all top peaks)
        grad = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(peak_indices)

        # Multiplicative weights update in z-space
        z = z_from_a(a)
        S = np.dot(grad, a)
        grad_z = a * (grad - S)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            last_improve = step
            stall_count = 0
            eta = min(eta * 1.001, 0.1)
        else:
            stall_count += 1
            eta *= 0.95
            if eta < 1e-12:
                break

    return best_a, best_score


def lse_adam(a0, time_limit, T_start=50.0, T_end=0.01, max_iters=20000):
    """LSE objective optimized with Adam in z-space."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5

    n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(100, max_iters // n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            if max_Tb > 500:
                max_Tb = 500
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Adam update
            t_step = si * iters_per_stage + it + 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if (si + 1) % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam stage={si+1}/{n_stages} T={T_val:.3f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a


def lbgfs_logsumexp(a0, time_limit, T_start=80.0, T_end=0.03, n_stages=20):
    """LSE L-BFGS polishing (from previous, slightly modified)."""
    n = len(a0)

    t0 = time.monotonic()
    best_a = a0.copy()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            if max_Tb > 500:
                max_Tb = 500
            exp_Tb = np.exp(Tb - max_Tb)
            return (2.0 * n / stage_T) * (max_Tb + np.log(exp_Tb.sum()))

        def jac_staged(z_val):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            if max_Tb > 500:
                max_Tb = 500
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-20:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        iters = 30 if si < n_stages // 2 else 15
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            elapsed = time.monotonic() - time.monotonic()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # === Phase 1: Peak flatten refinement (quick warmup, 60s) ===
    print(f"\n=== Phase 1: Peak flatten at n={n_base} (score={best_score:.8f}) ===")
    t1 = min(60.0, (_global_deadline - time.monotonic()) * 0.04)
    if t1 > 5:
        a_out, _ = peak_flatten_refine(best_seq, t1, top_k_frac=0.1, max_steps=20000)
        try_update(a_out, "peak_flatten_p1")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: LP Nesterov - dual formulation (main, 700s) ===
    print(f"\n=== Phase 2: Nesterov LP (dual) at n={best_n} (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(700.0, remaining * 0.45)
    if t2 > 30:
        a_out, _ = nesterov_lp(best_seq, t2, solver=solver, use_dual=False)
        try_update(a_out, "nesterov_lp_dual")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Peak flatten refinement (200s) ===
    print(f"\n=== Phase 3: Peak flatten refinement (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(200.0, remaining * 0.15)
    if t3 > 10:
        a_pt, _ = peak_flatten_refine(best_seq, t3, top_k_frac=0.03, max_steps=30000)
        try_update(a_pt, "peak_flatten_p3")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: LSE Adam polish (250s) ===
    print(f"\n=== Phase 4: LSE Adam polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(250.0, remaining * 0.2)
    if t4 > 20:
        a_lse = lse_adam(best_seq, t4, T_start=40.0, T_end=0.005, max_iters=15000)
        try_update(a_lse, "lse_adam")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Second LP Nesterov round (200s) ===
    print(f"\n=== Phase 5: Nesterov LP round 2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(200.0, remaining * 0.15)
    if t5 > 15:
        a_out2, _ = nesterov_lp(best_seq, t5, solver=solver, use_dual=False)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: LSE L-BFGS polish (150s) ===
    print(f"\n=== Phase 6: LSE L-BFGS polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6 = min(150.0, remaining * 0.12)
    if t6 > 15:
        a_lse2 = lbgfs_logsumexp(best_seq, t6, T_start=30.0, T_end=0.01, n_stages=15)
        try_update(a_lse2, "lse_lbfgs")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 80:
        for n_try in [best_n + 200, best_n + 400]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 30:
                break
            t7 = min(50.0, remaining * 0.4)
            if t7 < 15:
                break
            print(f"  Trying n={n_try}, budget={t7:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t7, solver=solver, use_dual=False)
            try_update(a_try, f"len_{n_try}")
            # Also try peak flatten on the larger sequence
            remaining = _global_deadline - time.monotonic()
            if remaining > 20:
                a_try2, _ = peak_flatten_refine(interpolate_sequence(best_seq, n_try),
                                                min(30.0, remaining * 0.5),
                                                top_k_frac=0.05, max_steps=10000)
                try_update(a_try2, f"peakflatten_len_{n_try}")

    # === Phase 8: Final peak flatten + LSE (remaining time) ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 8: Final refinement (score={best_score:.8f}, {remaining:.0f}s left) ===")
        t8a = min(40.0, remaining * 0.4)
        if t8a > 5:
            a_final, _ = peak_flatten_refine(best_seq, t8a, top_k_frac=0.02, max_steps=20000)
            try_update(a_final, "final_peak")
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            t8b = min(40.0, remaining * 0.5)
            if t8b > 5:
                a_final2 = lse_adam(best_seq, t8b, T_start=10.0, T_end=0.001, max_iters=5000)
                try_update(a_final2, "final_lse")
    print(f"  Phase 8 done: best={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
