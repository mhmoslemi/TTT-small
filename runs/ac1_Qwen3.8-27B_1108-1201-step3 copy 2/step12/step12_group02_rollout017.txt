def lp_refine_step(a, n_top=100, solver=None):

def peak_shaving_fast(a, time_limit, seed=123):

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)

    rng = np.random.default_rng(seed)
    variant = int(rng.integers(0, 8))
    x = np.linspace(0, 1, n)

    if variant == 0:
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 1:
        a = np.exp(-4 * (x - 0.3) ** 2) + np.exp(-4 * (x - 0.7) ** 2)
    elif variant == 2:
        a = x ** 2 + (1 - x) ** 2 + 0.05
    elif variant == 3:
        a = 1 + 0.4 * np.cos(3 * 2 * np.pi * x) + 0.1 * np.cos(7 * 2 * np.pi * x)
    elif variant == 4:
        a = np.sinc(x - 0.5) ** 2
    elif variant == 5:
        a = np.maximum(1 - 3 * abs(x - 0.5), 0) + 0.05
    elif variant == 6:
        a = 1 / (1 + 10 * (x - 0.5) ** 2)
    else:
        a = np.ones(n)
        for _ in range(4):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            xx = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(xx - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)

    return normalize(a)


def mirror_descent_fast(a0, time_limit, eta0=0.04, k_max=10, seed=42):
    """Fast mirror descent as a backup/preprocessing step."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_sc = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    t_iter = 0
    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        b = fast_autocorr(a)
        k = min(k_max, max(1, t_iter // 100))
        top_idx = np.argsort(b)[::-1][:k]

        grad = np.zeros(n)
        for m in top_idx:
            m = int(m)
            js = max(0, m - n + 1)
            je = min(n - 1, m)
            js_arr = np.arange(js, je + 1)
            grad[js_arr] += a[m - js_arr]
        grad /= k

        eta = eta0 / (1.0 + t_iter / 200.0) ** 0.6
        a_new = a * np.exp(-eta * grad)
        a_new = normalize(a_new)
        a = a_new

        if t_iter % 20 == 0:
            sc = fast_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 100:
            a = a + 0.02 * rng.standard_normal(n)
            a = normalize(a)
            stall = 0

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase LP-based optimization with
    adaptive constraints, multi-length exploration, and local polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1760.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Load parent and create initializations
    print(f"=== Phase 0: Initialization (budget={time_budget:.0f}s) ===")
    n_base = 1600

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = fast_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_base = len(parent)
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(n_base, seed=0)
        best_score = fast_score(best_seq)
        print(f"  Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: LP refinement at base length
    print(f"\n=== Phase 1: LP Refinement n={n_base} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_p1 = min(remaining * 0.30, 500.0)
    if t_p1 > 30:
        a_out, sc_out = lp_multi_refine(best_seq, t_p1, n_top_start=30,
                                         n_top_end=200, n_iter_target=150, seed=1)
        try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Mirror descent from diverse inits at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MD + LP from diverse inits ===")
    t_p2 = min(remaining * 0.12, 180.0)
    if t_p2 > 30:
        t_per = t_p2 / 4
        for v in [1, 2, 4, 6]:
            if time.monotonic() - t_global > time_budget:
                break
            init = diverse_init(n_base, seed=v * 7 + 3)
            a_md, _ = mirror_descent_fast(init, t_per * 0.4, eta0=0.06, k_max=8, seed=v)
            a_lp, _ = lp_multi_refine(a_md, t_per * 0.6, n_top_start=30,
                                       n_top_end=120, n_iter_target=40, seed=v + 10)
            try_update(a_lp, f"mdlp_v{v}")

    # Phase 3: Try longer length
    n_long = int(round(n_base * 1.25))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Longer length n={n_long} ===")
    t_p3 = min(remaining * 0.22, 350.0)
    if t_p3 > 30 and best_seq is not None:
        init_long = interpolate_to_length(best_seq, n_long)
        a_out, _ = lp_multi_refine(init_long, t_p3, n_top_start=30,
                                    n_top_end=250, n_iter_target=100, seed=20)
        try_update(a_out, f"lp_n{n_long}")

    # Phase 4: Even longer length
    n_long2 = int(round(n_base * 1.5))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Longer length n={n_long2} ===")
    t_p4 = min(remaining * 0.15, 200.0)
    if t_p4 > 30 and best_seq is not None:
        init_long2 = interpolate_to_length(best_seq, n_long2)
        a_out, _ = lp_multi_refine(init_long2, t_p4, n_top_start=20,
                                    n_top_end=150, n_iter_target=60, seed=30)
        try_update(a_out, f"lp_n{n_long2}")

    # Phase 5: Peak shaving on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Peak Shaving ===")
    n_best = len(best_seq) if best_seq is not None else n_base
    t_p5 = min(remaining * 0.15, 180.0)
    if t_p5 > 10 and best_seq is not None:
        a_out, _ = peak_shaving_fast(best_seq, t_p5, seed=99)
        try_update(a_out, f"ps_n{n_best}")

    # Phase 6: Final LP polish with high n_top
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final LP Polish ===")
    t_p6 = min(remaining * 0.25, 300.0)
    if t_p6 > 20 and best_seq is not None:
        a_out, _ = lp_multi_refine(best_seq, t_p6, n_top_start=100,
                                    n_top_end=300, n_iter_target=100, seed=55)
        try_update(a_out, "final_lp")

    # Phase 7: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    t_p7 = min(remaining * 0.30, 150.0)
    if t_p7 > 10 and best_seq is not None:
        a_out, _ = peak_shaving_fast(best_seq, t_p7, seed=777)
        try_update(a_out, "final_ps")

    # Final MD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t_p8 = min(remaining * 0.20, 60.0)
    if t_p8 > 5 and best_seq is not None:
        a_out, _ = mirror_descent_fast(best_seq, t_p8, eta0=0.01, k_max=20, seed=88)
        try_update(a_out, "final_md")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
