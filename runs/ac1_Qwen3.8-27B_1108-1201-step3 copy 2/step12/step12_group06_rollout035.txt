Let me also add a **novel component**: a **direct max-reduction step** that works as follows:

After the Nesterov LP gives us a good `p`, we identify the top peaks in `conv(p,p)`. For each peak at index k, we know that `conv(p,p)[k] = sum_j p[j]*p[k-j]`. We can try to reduce this by finding a direction that decreases this specific coefficient while not increasing others too much.

Specifically, for a peak at k:
- The gradient of `conv(p,p)[k]` w.r.t. p is `2*flip(p)[k-·]`
- We want to move in a direction that decreases this gradient inner product
- But we also need to keep other peaks from increasing

This is essentially a **multi-objective** problem. We can use the **weighted sum** approach: minimize `sum_k w_k * conv(p,p)[k]` where `w_k` is proportional to how much each peak exceeds the target.

This is a quadratic program! `sum_k w_k * conv(p,p)[k]` is a quadratic form in p:
`= p^T * (sum_k w_k * T_k) * p` where T_k is the matrix for the k-th convolution coefficient.

The constraint is `sum(p) = 1, p >= 0`.

So we can solve:
`minimize p^T Q p`
`s.t. sum(p) = 1`
     `p >= 0`

where `Q = sum_k w_k * T_k`.

This is a QP (quadratic program) which can be solved efficiently! And it directly optimizes a weighted sum of the convolution coefficients, which is a smooth proxy for the max.

The weights `w_k` can be set adaptively: `w_k = max(0, conv(p,p)[k] - target)` or `w_k = exp(τ * conv(p,p)[k])` (the LSE weights).

Actually, this is very similar to the LSE approach but formulated as a QP rather than a non-linear optimization. The QP is:
`minimize (1/2) p^T Q p`
`s.t. 1^T p = 1, p >= 0`

This can be solved with any QP solver (cvxpy, scipy.optimize, etc.).

The key advantage: at each iteration, we solve a QP (which is convex!) rather than doing nonlinear optimization. The non-convexity is handled by the outer iteration (updating the weights).

This is essentially a **Dinkelbach-type** or **parametric** method for the max problem.

    """Project onto probability simplex {x: sum(x)=1, x>=0}."""

def qp_fista_polish(p0, time_limit, n_stages=30, T_start=200.0, T_min=20.0, 
                     fista_iters=2000, blend=0.5):
    """Polish using QP with LSE weights, solved via FISTA."""
    n = len(p0)
    p = p0.copy()
    p = p / p.sum()
    best_p = p.copy()
    best_score = score_normalized(p)
    t0 = time.monotonic()
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute LSE weights
        conv_p = fast_convolve(p)
        T = T_start * (T_min / T_start) ** (stage / max(n_stages - 1, 1))
        T = max(T, T_min)
        
        max_conv = conv_p.max()
        w = np.exp(T * (conv_p - max_conv))
        w = w / w.sum()
        
        # Compute Lipschitz constant estimate
        # L = 2 * ||Q||_2, where Q[i,j] = w[i+j]
        # Upper bound: L <= 2 * max_i sum_j w[i+j]
        row_sums = np.array([np.sum(w[i:i+n]) for i in range(n)])
        L = 2.0 * row_sums.max()
        if L < 1e-15:
            L = 1.0
        
        # FISTA
        p_k = p.copy()
        p_km1 = p.copy()
        t_k = 1.0
        
        for it in range(fista_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            t_km1 = t_k
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_km1 * t_km1)) / 2.0
            momentum = (t_km1 - 1.0) / t_k
            
            y_k = p_k + momentum * (p_k - p_km1)
            y_k = np.maximum(y_k, 0)
            sy = y_k.sum()
            if sy > 1e-15:
                y_k = y_k / sy
            
            # Gradient: 2 * Q @ y_k
            grad = 2.0 * hankel_matvec(w, y_k)
            
            # Projected gradient step
            step = 1.0 / L
            p_km1 = p_k
            p_k = project_simplex(y_k - step * grad)
            
            # Check score periodically
            if (it + 1) % 500 == 0:
                sc = score_normalized(p_k)
                if sc < best_score:
                    best_score = sc
                    best_p = p_k.copy()
        
        # After FISTA, check the result
        sc = score_normalized(p_k)
        if sc < best_score:
            best_score = sc
            best_p = p_k.copy()
        
        # Blend with previous
        p = blend * p + (1 - blend) * p_k
        p = np.maximum(p, 0)
        p = p / p.sum()
        
        # Also check blended
        sc_blend = score_normalized(p)
        if sc_blend < best_score:
            best_score = sc_blend
            best_p = p.copy()
    
    return best_p, best_score

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def hankel_matvec(w, p):
    """Compute (Q@p)[i] = sum_{j=0}^{n-1} w[i+j]*p[j] via FFT.
    
    Q is a Hankel matrix with Q[i,j] = w[i+j].
    """
    n = len(p)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w, fft_size)
    P = fft(p, fft_size)
    result = ifft(W * np.conj(P), fft_size).real[:n]
    return result

def build_conv_matrix(a_fixed):
    """Build convolution matrix T: (T@b)[k] = sum_j a_fixed[j]*b[k-j]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def get_lp_solver():
    """Find the best available LP solver."""
    candidates = [cp.CBC, cp.PDLP, cp.SCS, cp.ECOS]
    for solver_name in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, cp.sum(x**2) <= 2])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: minimize M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try fallback solvers
        for alt in [cp.CBC, cp.PDLP, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)
