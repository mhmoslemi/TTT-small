def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def entropy_fw_preopt(a0, time_limit, max_iters=500000):
    """Fast entropy-regularized Frank-Wolfe (multiplicative weights)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.5
    eta_min = 0.001
    patience = 0
    last_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max(200, it * 0.5):
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find peak indices
        threshold = max_b * 0.98
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([int(np.argmax(b))])
        # Cap number of peaks used
        if len(peak_idx) > 20:
            # Keep the top 20
            top_vals = np.argpartition(b, -20)[-20:]
            peak_idx = top_vals

        # Compute subgradient: average of a[k - i] over peak indices
        grad = np.zeros(n)
        for k in peak_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(peak_idx)

        # Entropy-regularized FW update (softmax of negative subgradient)
        shifted = -eta * grad
        shifted -= shifted.max()
        new_a = a * np.exp(shifted)
        sa = new_a.sum()
        if sa < 1e-30:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = it
            eta = min(eta * 1.01, 2.0)
        else:
            patience += 1
            eta *= 0.95
            if eta < eta_min:
                break

        if (it + 1) % 10000 == 0:
            print(f"    EntropyFW iter={it+1}: score={best_score:.8f} eta={eta:.4f}")

    return best_a, best_score


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, momentum_mode='adaptive'):
    """Nesterov-accelerated LP Frank-Wolfe with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    iter_since_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if iter_since_improve > 30000:
            break

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            iter_since_improve = 0

        # Adaptive restart
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0001 and t_nest > 3.0:
            t_nest = 1.0
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new
        iter_since_improve += 1

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f}")

    return best_a, best_score


def pnorm_objective_grad(z_in, p_val, n):
    """Compute p-norm of convolution and gradient w.r.t. z (log-space)."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    b = np.maximum(b, 1e-30)
    p_sum = np.sum(b ** p_val)
    obj = (2.0 * n) * p_sum ** (1.0 / p_val)

    # Gradient: d/dz_i [2n * (sum b_k^p)^(1/p)]
    # = 2n * (1/p) * (sum b_k^p)^(1/p - 1) * sum_k p * b_k^(p-1) * db_k/dz_i
    # = 2n * (sum b_k^p)^(1/p - 1) * sum_k b_k^(p-1) * db_k/dz_i
    # db_k/dz_i = db_k/da_i * da_i/dz_i = db_k/da_i * a_i
    # db_k/da_i = a[k-i] (for valid k-i)

    p_sum_half = p_sum ** ((1.0 - p_val) / p_val)
    if p_val == 2:
        weights = b
    else:
        weights = b ** (p_val - 1)

    grad_a = np.zeros(n)
    for k in range(len(b)):
        w = weights[k]
        if w < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (2.0 * n * p_sum_half * (grad_a - S))
    return obj, grad_z


def lse_objective_grad(z_in, T_val, n):
    """LogSumExp approximation to max, with gradient."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    exp_sum = exp_Tb.sum()
    w = exp_Tb / exp_sum
    log_sum_exp = max_Tb + np.log(exp_sum)
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def multi_stage_smooth_opt(a0, time_limit, use_pnorm=True):
    """Multi-stage smooth optimization (p-norm or LSE)."""
    n = len(a0)

    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

        n_stages = len(p_stages)
    else:
        p_stages = np.geomspace(200.0, 0.02, 20)
        n_stages = len(p_stages)

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        stage_param = p_stages[si]

        if use_pnorm:
            p_val = stage_param

            def func_staged(z_val):
                obj, _ = pnorm_objective_grad(z_val, p_val, n)
                return obj

            def jac_staged(z_val):
                _, grad = pnorm_objective_grad(z_val, p_val, n)
                return grad
        else:
            T_val = stage_param

            def func_staged(z_val):
                obj, _ = lse_objective_grad(z_val, T_val, n)
                return obj

            def jac_staged(z_val):
                _, grad = lse_objective_grad(z_val, T_val, n)
                return grad

        iters = 30 if si < n_stages // 2 else 15

            z_new = res.x
            a_new = softmax_from_z(z_new)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception as e:

            a_perturbed[i1] -= delta

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            if n_try < 200:
                continue
            t8 = min(50.0, remaining * 0.35)
            if t8 < 10:
                break
            print(f"  Trying n={n_try}, budget={t8:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t8, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
