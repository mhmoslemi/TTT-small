def fast_convolve(a):

    a = np.asarray(a, dtype=np.float64)

    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_sparse(a):
    """Build the convolution constraint matrix T where (T @ b)_k = (a*b)_k."""
    n = len(a)
    size = 2 * n - 1
    # T is (2n-1) x n, T[k,j] = a[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T

def subgradient_nesterov(a0, time_limit, max_steps=200000):
    """Fast subgradient Nesterov with smart peak targeting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.5
    last_improve = 0
    no_improve_count = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve_count > 8000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Use soft-top-k for smoother gradient
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        # Compute gradient
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        # Projected gradient in log-space (multiplicative update)
        # a * exp(-eta * grad_a / max(grad_a))
        grad_max = grad_a.max()
        if grad_max < 1e-15:
            break
        exponent = -eta * grad_a / grad_max
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score * (1 - 1e-10):
            best_score = sc
            best_a = new_a.copy()
            a_prev = a
            a = new_a
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            no_improve_count = 0
        else:
            # Nesterov extrapolation
            if t_nest > 1.5:
                momentum = (t_nest - 1.0) / t_nest
                a_extrap = a + momentum * (a - a_prev)
                a_extrap = np.maximum(a_extrap, 0)
                se = a_extrap.sum()
                if se > 1e-15:
                    a_extrap = a_extrap / se
                    a = a_extrap
            eta = max(eta * 0.995, 0.05)
            no_improve_count += 1
            if sc > best_score * 1.01 and t_nest > 5:
                t_nest = 1.0
                eta = min(eta * 1.5, 1.0)

    return best_a, best_score

            best_a = a_new.copy()

        # Adaptive restart: multiple criteria
        sc_cur = score_normalized(a)
        dist = np.linalg.norm(a_new - a) / n
        if sc_new > sc_cur * 1.001 and t_nest > 10.0:
            t_nest = 1.0
            consecutive_restarts += 1
            if consecutive_restarts >= 3:
                t_nest = 1.0
                # More aggressive restart: blend with best
                a_new = 0.5 * a_new + 0.5 * best_a
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
            restart_count += 1
        elif dist < 1e-6 and t_nest > 5.0:
            # Stagnation detected
            t_nest = 1.0
            consecutive_restarts += 1
            restart_count += 1
        else:
            consecutive_restarts = 0
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
            last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """LogSumExp objective and gradient in z-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient computation: grad_a = 2 * sum_k w_k * T_k @ a / (sum a)^2
    # But since we're in z-space with softmax parameterization:
    # grad_z = a * (grad_a - <grad_a, a>)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=25):
    """L-BFGS on the LogSumExp smoothed objective."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 60 if si < n_stages // 3 else (40 if si < 2 * n_stages // 3 else 25)
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a, best_score


def peak_shave(a0, time_limit, max_steps=100000):
    """Aggressive peak shaving using multiplicative updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve = 0
    stall_count = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Target the top peaks more aggressively
        # Use a ranked weighting
        sorted_b = np.sort(b)[::-1]
        K = max(5, min(50, len(b) // 20))
        threshold = sorted_b[K - 1]
        
        topk = np.where(b >= threshold)[0]
        
        # Weight by how much each peak exceeds the mean
        mean_b = b.mean()
        weights = np.maximum(b[topk] - mean_b, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(topk))
        weights = weights / weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for i, k in enumerate(topk):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += weights[i] * a[k - idx]
        
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        
        # Multiplicative update
        exponent = -eta * grad / grad_max
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score * (1 - 1e-9):
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.005, 0.1)
            stall_count = 0
        else:
            stall_count += 1
            eta = max(eta * 0.95, 0.005)
            if stall_count > 500:
                break

    return best_a, best_score
