import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autocorr(a):
    """Full self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate using evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list or np.sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    # Scale to sum=1 first, then scale back
    v_scaled = v / target_sum if target_sum != 1.0 else v
    # Project onto probability simplex
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v_scaled - theta, 0)
    return result * target_sum

def lse_value_and_grad(a, T):

    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum = max_Tb + np.log(exp_Tb.sum())
    val = log_sum / T
    
    # Weight for each peak
    w = exp_Tb / exp_Tb.sum()

    # More precisely: grad_i = 2 * (w * a_rev)[n-1+i] where a_rev[j] = a[n-1-j]
    # Let me verify: (w * a_rev)[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-(m-j)]
    # For m = n-1+i: sum_j w[j] * a[n-1-(n-1+i-j)] = sum_j w[j] * a[j-i]
    # We need a[j-i] to be valid: 0 <= j-i < n, i.e., i <= j < i+n

    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # FISTA iterations at this temperature
        x = a.copy()  # current
        z = a.copy()  # momentum
        t_fista = 1.0
        
        # Estimate Lipschitz constant: L = 2 * T * 2 * max(a) * (2n-1) roughly
        # More carefully: the gradient of LSE_T(a) has Lipschitz constant ~ 2T * max(a) * 2

        L = 2.0 * T * 2.0 * a.max() * 2.0
        if L < 1e-6:

            z = x + beta * (x - z)

            # Check improvement

def subgradient_method(a0, time_limit, n_iters=50000):
    """Subgradient method on the active set with averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # iterate
    avg_a = np.zeros(n)
    
    top_k = 10  # number of top peaks to use in subgradient
    
    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(y)
        b_max = b.max()
        
        # Find top peaks
        top_indices = np.argsort(b)[::-1][:top_k]
        top_indices = top_indices[b[top_indices] >= b_max * 0.999]
        
        # Compute subgradient: average of gradients at active peaks
        grad = np.zeros(n)
        for k in top_indices:
            # grad_i from peak k: a[k-i] (if valid) + a[i-k] (if valid)
            # Vectorized:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            partners = k - idx
            grad[idx] += y[partners]
            if k > 0:
                # a[i-k] for i > k
                idx2 = np.arange(k + 1, min(n, k + n))
                if len(idx2) > 0:
                    partners2 = idx2 - k
                    if (partners2 >= 0).all() and (partners2 < n).all():
                        grad[idx2] += y[partners2]
        
        grad /= len(top_indices)
        
        # Step size: eta = c / sqrt(t)
        eta = 1.0 / np.sqrt(t)
        
        # Update
        y_new = y - eta * grad
        y_new = project_simplex(y_new)
        
        # Average
        avg_a += y_new
        
        if t % 500 == 0:
            avg_check = avg_a / t
            score = 2.0 * n * fast_autocorr(avg_check).max()
            if score < best_score:
                best_score = score
                best_a = avg_check.copy()
        
        y = y_new
    
    # Final check
    avg_final = avg_a / min(t, n_iters)
    score = 2.0 * n * fast_autocorr(avg_final).max()
    if score < best_score:
        best_score = score
        best_a = avg_final.copy()
    
    return best_a, best_score

def mirror_descent(a0, time_limit, T=10.0, n_iters=50000):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    log_a = np.log(np.maximum(a, 1e-30))
    top_k = 10
    
    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_curr = np.exp(log_a - log_a.max())  # normalize later
        a_curr = a_curr / a_curr.sum()
        
        b = fast_autocorr(a_curr)
        b_max = b.max()
        top_indices = np.argsort(b)[::-1][:top_k]
        
        # Gradient at each coordinate
        grad = np.zeros(n)
        for k in top_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            partners = k - idx
            grad[idx] += a_curr[partners]
            idx2 = np.arange(k + 1, min(n, k + n))
            if len(idx2) > 0:
                partners2 = idx2 - k
                if (partners2 >= 0).all() and (partners2 < n).all():
                    grad[idx2] += a_curr[partners2]
        grad /= len(top_indices)
        
        # Multiplicative update: a_new_i ∝ a_curr_i * exp(-eta * grad_i)
        eta = 1.0 / np.sqrt(t)
        log_a = log_a - eta * grad
        log_a -= log_a.max()  # numerical stability
        
        if t % 1000 == 0:
            score = 2.0 * n * fast_autocorr(a_curr).max()
            if score < best_score:
                best_score = score
                best_a = a_curr.copy()
        
        a_curr = np.exp(log_a)
        a_curr = a_curr / a_curr.sum()
    
    score = 2.0 * n * fast_autocorr(a_curr).max()
    if score < best_score:
        best_score = score
        best_a = a_curr.copy()
    
    return best_a, best_score

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Precompute the autocorrelation
    b = fast_autocorr(a)
    b_max = b.max()
    
    stall = 0
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Choose source: prefer coordinates that contribute to peaks
        # Choose target: prefer coordinates that don't contribute much to peaks
        
        # Find top peaks
        top_peak = np.argmax(b)
        
        # Source: a coordinate that contributes to the top peak
        if rng.random() < 0.7:
            js = max(0, top_peak - n + 1)
            je = min(n - 1, top_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[top_peak - idx]
            # Probabilistically pick a high-contributor
            total = contrib.sum()
            if total < 1e-30:
                src = rng.integers(n)
            else:
                probs = contrib / total
                src = idx[rng.choice(len(idx), p=probs)]
        else:
            src = rng.integers(n)
        
        if a[src] < 1e-12:
            continue
        
        # Target: a coordinate that contributes least to peaks
        if rng.random() < 0.7:
            # Pick randomly from low-contribution coordinates
            dst = rng.integers(n)
        else:
            dst = rng.integers(n)
        
        if src == dst:
            continue
        
        # Try different fractions
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-15:
                continue
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                b = fast_autocorr(a)
                stall = 0
                break
        else:
            stall += 1
            if stall > 5000:
                # Try a bigger random perturbation
                src = rng.integers(n)
                dst = rng.integers(n)
                if src != dst and a[src] > 1e-10:
                    amt = a[src] * rng.uniform(0.01, 0.2)
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        b = fast_autocorr(a)
                        stall = 0
        
        if trial % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PM trial={trial} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
        
        if stall > 20000:
            break
    
    return best_a, best_score

def stochastic_subgradient(a0, time_limit, n_iters=100000):
    """Stochastic subgradient: randomly sample from top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    y = a.copy()
    avg_a = np.zeros(n)
    
    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation (expensive but necessary)
        if t % 10 == 1:  # every 10 iters
            b = fast_autocorr(y)
            b_max = b.max()
            # Sample from top peaks with probability proportional to their value
            top_n = min(50, len(b))
            top_idx = np.argsort(b)[::-1][:top_n]
            top_vals = b[top_idx]
            probs = top_vals / top_vals.sum()
        
        k = top_idx[rng.choice(len(top_idx), p=probs)]
        
        # Gradient at peak k
        grad = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        partners = k - idx
        grad[idx] = y[partners]
        idx2 = np.arange(k + 1, min(n, k + n))
        if len(idx2) > 0:
            partners2 = idx2 - k
            if (partners2 >= 0).all() and (partners2 < n).all():
                idx2_valid = idx2[(partners2 >= 0) & (partners2 < n)]
                partners2_valid = idx2_valid - k
                grad[idx2_valid] += y[partners2_valid]
        
        # Step
        eta = 0.5 / np.sqrt(t)
        y_new = y - eta * grad
        y_new = project_simplex(y_new)
        
        avg_a += y_new
        y = y_new
        
        if t % 5000 == 0:
            avg_check = avg_a / t
            score = 2.0 * n * fast_autocorr(avg_check).max()
            if score < best_score:
                best_score = score
                best_a = avg_check.copy()
            print(f"    SS t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    avg_final = avg_a / min(t, n_iters)
    score = 2.0 * n * fast_autocorr(avg_final).max()
    if score < best_score:
        best_score = score
        best_a = avg_final.copy()
    
    return best_a, best_score
