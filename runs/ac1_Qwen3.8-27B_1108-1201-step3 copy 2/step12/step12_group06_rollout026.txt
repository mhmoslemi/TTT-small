def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None, tol=1e-6):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, **({'eps_abs': tol, 'eps_rel': tol} if solver == cp.PDLP else {}))
    except TypeError:
        try:
            prob.solve(solver=solver)
        except Exception:
            pass
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, tol=1e-6):
    """Nesterov-accelerated LP best-response iteration with adaptive parameters."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        b_new, M_lp = solve_lp_best_response(y, solver, tol=tol)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) * (it + 1))) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        # Track stagnation
        if best_score < prev_best - 1e-8:
            stagnant_count = 0
            prev_best = best_score
        elif it > 100:
            stagnant_count += 1

        # If stagnant, loosen tolerance or restart
        if stagnant_count > 500:
            # Slightly perturb to escape
            rng = np.random.default_rng(it)
            pert = 0.01 * rng.normal(0, 1, n)
            y = np.maximum(y + pert, 0)
            y = y / y.sum()
            stagnant_count = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsep_lse_adaptive(a0, time_limit, T_start=80.0, T_min=0.1, n_stages=30, iters_per_stage=50):

        if remaining < 3:
            break

            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            if sc < prev_score - 1e-6:
                stagnant = max(0, stagnant - 2)
                T = max(T_min, T * 0.6)
            elif sc < prev_score + 1e-8:
                stagnant += 1
                T = max(T, T * 1.3)
            else:
                stagnant = 0
                T = max(T_min, T * 0.8)
            prev_score = best_score

            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.3f}: score={sc:.8f} best={best_score:.8f}")

        except Exception as e:
            T = max(T_min, T * 1.5)

    return best_a, best_score


def dual_averaging_subgradient(a0, time_limit, n_steps=500000, top_k=10):
    """Dual averaging subgradient method with iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    # Dual averaging accumulators
    s_acc = np.zeros(n)
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0
    best_avg_score = best_score

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks (top top_k)
        if top_k >= len(b):
            active = np.arange(len(b))
        else:
            active = np.argsort(b)[-top_k:]

        # Compute average subgradient over active peaks
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                g[i] += a[k - i]
        g /= len(active)

        # Project gradient to simplex tangent
        S = np.dot(g, a)
        grad_simplex = g - S

        # Dual averaging update
        step_num = step + 1
        s_acc += grad_simplex
        t = 1.0 / np.sqrt(step_num)  # Step size schedule
        a_new = project_simplex(a - t * s_acc / step_num)

        # Compute score
        sc_new = score_normalized(a_new)

        # Update average (exponential moving average, last 10000)
        avg_a = (avg_count * avg_a + a_new) / (avg_count + 1)
        avg_count += 1
        if avg_count > 10000:
            avg_count = 10000
            avg_a = (avg_a * 10000 + a_new) / 10001

        a = a_new

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_avg_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualAvg step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def accelerated_subgradient(a0, time_limit, n_steps=200000, top_k=5):

    n = len(a0)

    avg_count = 0

    y = a.copy()  # Nesterov momentum variable

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()

        # Find active peaks
        if top_k >= len(b):
            active = np.arange(len(b))
        else:
            active = np.argsort(b)[-top_k:]

        # Subgradient at y
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                g[i] += y[k - i]
        g /= len(active)

        # Nesterov acceleration
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (step + 1) * (step + 1))) / 2.0
        alpha = 2.0 / (t_k + 1.0)

        # Polyak step
        sc_y = 2.0 * n * max_b

        if gap < 1e-12:

        grad_norm_sq = np.dot(g, g)
        if grad_norm_sq < 1e-30:
            break
        eta = gap / (2.0 * grad_norm_sq)
        eta = min(eta, 2.0 / (np.sqrt(step + 1) + 1))

        y_new = project_simplex(y - eta * g)

        a_new = (1.0 - alpha) * a + alpha * y_new
        a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        sc_new = score_normalized(a)

        avg_a = (avg_count * avg_a + a) / (avg_count + 1)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def coordinate_polish(a0, time_limit, max_sweeps=1000):
    """Coordinate descent polish: optimize each coordinate while fixing others."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break

            # Try small perturbations of a[i]
            # Compute effect on convolution: changing a[i] by delta changes
            # conv[k] by delta * (a[k-i] + a[i-k]) for valid indices
            # (with a[x]=0 for x outside [0,n-1])
            old_val = a[i]
            if old_val < 1e-12:
                # Try setting to small positive value
                candidates = [old_val + 0.01 / n]
            else:
                # Try increasing and decreasing
                step_size = max(0.01 / n, old_val * 0.1)
                candidates = [old_val + step_size, old_val - step_size,
                              old_val + 2 * step_size, old_val - 2 * step_size]

            best_local = old_val
            best_local_score = best_score  # Use current score as reference

            # Compute current convolution for reference
            b = fast_convolve(a)
            cur_max = b.max()

            for new_val in candidates:
                if new_val < -1e-10:
                    continue
                new_val = max(0.0, min(new_val, 1.0))
                a[i] = new_val
                # Renormalize
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s
                sc = score_normalized(a)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = new_val

            a[i] = best_local
            s = a.sum()
            if s > 1e-15:
                a = a / s

            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                improved = True

        if not improved and sweep > 50:
            break

        if (sweep + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordPolish sweep={sweep+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    t1 = min(0.15 * (budget_s - 20), remaining * 0.25)
    if t1 > 10:
        a_out, _ = dual_averaging_subgradient(best_seq, t1, n_steps=1000000, top_k=8)
        try_update(a_out, "dual_avg_sg")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: Nesterov LP (40% time) ===
    print(f"\n=== Phase 2: Nesterov LP ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.40 * (budget_s - 20), remaining * 0.55)
    if t2 > 30:
        a_out, _ = nesterov_lp(best_seq, t2, solver=solver, max_iters=200000, tol=1e-7)
        try_update(a_out, "nesterov_lp")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Accelerated subgradient (15% time) ===
    print(f"\n=== Phase 3: Accelerated Subgradient ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 20), remaining * 0.35)
    if t3 > 10:
        a_out, _ = accelerated_subgradient(best_seq, t3, n_steps=500000, top_k=5)
        try_update(a_out, "accel_sg")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: LSE polishing (15% time) ===
    print(f"\n=== Phase 4: LSE Polishing ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.15 * (budget_s - 20), remaining * 0.4)
    if t4 > 20:
        # Multiple LSE passes with different temperatures
        a_lse1, _ = lsep_lse_adaptive(best_seq, t4 * 0.4, T_start=50.0, T_min=0.1, n_stages=25, iters_per_stage=40)
        try_update(a_lse1, "lse_main")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_lse2, _ = lsep_lse_adaptive(best_seq, remaining * 0.5, T_start=10.0, T_min=0.05, n_stages=20, iters_per_stage=35)
            try_update(a_lse2, "lse_cold")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # LSE from perturbed start
            perturbed = best_seq + 0.02 * rng.normal(0, 1, len(best_seq))
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_lse3, _ = lsep_lse_adaptive(perturbed, remaining * 0.5, T_start=30.0, T_min=0.1, n_stages=15, iters_per_stage=30)
            try_update(a_lse3, "lse_perturbed")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Nesterov LP again from improved start (8% time) ===
    print(f"\n=== Phase 5: Nesterov LP Round 2 ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.08 * (budget_s - 20), remaining * 0.4)
    if t5 > 15:
        a_out, _ = nesterov_lp(best_seq, t5, solver=solver, max_iters=100000, tol=1e-8)
        try_update(a_out, "nesterov_r2")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Multi-length exploration (6% time) ===
    print(f"\n=== Phase 6: Multi-length ===")
    remaining = global_deadline - time.monotonic()
    t6_total = min(0.06 * (budget_s - 20), remaining * 0.5)
    if t6_total > 20:
        for n_try in [n_base + 200, n_base - 200, n_base + 500]:
            remaining = global_deadline - time.monotonic()
            if remaining < 15:
                break
            t6 = min(t6_total / 3.0, remaining * 0.4)
            if t6 < 10:
                break
            print(f"  Trying n={n_try}")
            init_try = interpolate_sequence(best_seq, n_try)
            sc_init = score_normalized(init_try)
            if sc_init > best_score * 1.01:
                # Interpolation made it worse, skip
                continue
            # Quick subgradient
            a_try, _ = dual_averaging_subgradient(init_try, t6 * 0.5, n_steps=200000, top_k=5)
            try_update(a_try, f"len{n_try}_sg")
            # Quick LSE
            remaining = global_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = lsep_lse_adaptive(a_try, remaining * 0.4, T_start=20.0, T_min=0.2, n_stages=10, iters_per_stage=25)
                try_update(a_try2, f"len{n_try}_lse")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # === Phase 7: Final combined polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 10:
        # Short Nesterov
        a_f1, _ = nesterov_lp(best_seq, remaining * 0.3, solver=solver, max_iters=30000, tol=1e-8)
        try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Very cold LSE
            a_f2, _ = lsep_lse_adaptive(best_seq, remaining * 0.4, T_start=5.0, T_min=0.05, n_stages=15, iters_per_stage=30)
            try_update(a_f2, "final_cold_lse")

        remaining = global_deadline - time.monotonic()
        if remaining > 5:
            # Quick subgradient burst
            a_f3, _ = dual_averaging_subgradient(best_seq, remaining * 0.5, n_steps=100000, top_k=3)
            try_update(a_f3, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
