def lse_grad_fft(a, w, n):

    Compute gradient of LSE objective w.r.t. a using FFT-based cross-correlation.

    W = rfft(w, size)
    A = rfft(a, size)
    corr = irfft(np.conj(W) * A, size)
    return corr[:n]

def peak_targeted_refine(a0, time_limit, max_steps=100000):

        return f.copy()

        return np.ones(new_n) / new_n
