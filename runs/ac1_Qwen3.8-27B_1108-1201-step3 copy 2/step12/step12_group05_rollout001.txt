def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(arr):

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping to flatten power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        a = a_new
    return best_a

def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b

    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    S = np.dot(grad_a, a)

    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(20, min(150, int(remaining * 80 / (n_stages - si))))
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad = lse_obj_grad_z(z, stage_T)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            lr_t = lr / np.sqrt(1 + 0.005 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            if it % 15 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n convolution matrix."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp_adaptive(a0, time_limit, solver, max_iters=100000):
    """
    Nesterov-accelerated LP with adaptive restarts.
    Key improvement: periodically restart from best iterate when convergence stalls.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive restart: if no improvement in last 'patience' iterations, restart
    patience = 200
    last_improve_t = 0
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        # Adaptive step size: start with Nesterov, reduce when oscillating
        alpha_t = 2.0 / (t + 2.0)
        
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve_t = t
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve_t = t
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                last_improve_t = t
        
        # Adaptive restart
        if t - last_improve_t > patience and t > 500:
            # Reset iteration counter and restart from best
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            # Reduce effective t for the momentum schedule
            # (we just continue, but the fresh start from best_a helps)
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP-ADPT t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
    
    return best_a, best_score

def sa_peak_shaving(a0, time_limit, seed=123, t_start=1.0, t_end=0.01):
    """
    Simulated annealing peak shaving.
    Accepts worse moves with probability exp(-delta/T) to escape local minima.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    current_score = best_score
    t0 = time.monotonic()
    total_time = time_limit
    stall = 0
    
    n_iter = 0
    while time.monotonic() - t0 < time_limit:
        # Annealing schedule
        progress = (time.monotonic() - t0) / total_time
        T = t_start * (t_end / t_start) ** progress
        
        b = fast_autocorr(a)
        peak_idx = np.argmax(b)
        
        # Find contributing pairs
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        src = idx[np.argmax(contrib)]
        
        # Choose destination: prefer low-value positions not contributing to top peaks
        top_k_idx = np.argsort(b)[::-1][:6]
        peak_areas = set()
        for k2 in top_k_idx:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_areas.update(range(js2, je2 + 1))
        
        candidates = [j for j in range(n) if j not in peak_areas]
        if not candidates:
            candidates = list(range(n))
        # Weight by inverse value (prefer moving to low-value positions)
        cand_weights = np.array([1.0 / (a[j] + 1e-10) for j in candidates])
        cand_weights /= cand_weights.sum()
        dst = candidates[rng.choice(len(candidates), p=cand_weights)]
        
        # Try a few fractions
        improved = False
        for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            a_trial = a_trial / a_trial.sum()
            new_score = 2.0 * n * fast_autocorr(a_trial).max()
            
            delta = new_score - current_score
            if delta < 0:
                # Accept (improvement)
                a = a_trial
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    stall = 0
                improved = True
                break
            else:
                # Accept with probability exp(-delta/T)
                if rng.random() < np.exp(-delta / max(T, 1e-10)):
                    a = a_trial
                    current_score = new_score
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    new_score = 2.0 * n * fast_autocorr(a_trial).max()
                    delta = new_score - current_score
                    if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                        a = a_trial
                        current_score = new_score
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a_trial.copy()
                            stall = 0
                    else:
                        stall += 1
        
        n_iter += 1
        if stall > 500:
            # Reset to best and restart annealing
            a = best_a.copy()
            current_score = best_score
            stall = 0
        
        if n_iter % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA iter {n_iter} ({elapsed:.0f}s): best={best_score:.6f}, curr={current_score:.6f}, T={T:.4f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
