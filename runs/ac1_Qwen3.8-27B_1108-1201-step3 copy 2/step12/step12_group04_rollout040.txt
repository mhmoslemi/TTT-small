"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. Diverse initializations (parent, structured, ideal-autocorrelation)
2. Nesterov-accelerated LP best-response iteration (primary workhorse)
3. FFT-based LogSumExp gradient optimization with Adam + adaptive annealing
4. Multi-peak guided refinement with targeted mass redistribution
5. Fine peak shaving with local search
6. Multi-length exploration and interpolation transfer

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full self-convolution a*a via FFT, O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max_b/(sum_a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_solver():
    """Find a working LP solver."""
    for name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_br(a_fixed, solver):
    """Solve best-response LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.PDLP, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver, damp=1.0):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        b_star, M_val = solve_lp_br(y, solver)
        if b_star is None:
            continue

        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y) * damp
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss

        y = y_new
        a = a_new

        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        sc_y = score_of(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            sc_avg = score_of(avg_check)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()

        if t % 200 == 0:
            print(f"    NLP t={t}: best={best_score:.8f}, M={M_val:.10f}")

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_and_grad(z, T):
    """Compute LSE objective and gradient in z-space. Returns (obj, grad_z)."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    obj = (max_Tb + np.log(exp_Tb.sum())) / T

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_adam_stage(z_init, T, max_iter, lr):
    """Adam optimization for LSE objective at fixed temperature T."""
    n = len(z_init)
    z = z_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_obj = score_of(softmax_fn(z))
    best_z = z.copy()

    for t in range(1, max_iter + 1):
        obj, grad_z = lse_obj_and_grad(z, T)
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 20 == 0:
            sc = score_of(softmax_fn(z))
            if sc < best_obj:
                best_obj = sc
                best_z = z.copy()

    sc_final = score_of(softmax_fn(z))
    if sc_final < best_obj:
        best_obj = sc_final
        best_z = z.copy()
    return best_z, best_obj


def lse_optimize(a0, time_limit, T_start=80.0, T_end=0.01, n_stages=20):
    """Multi-stage LSE optimization with Adam."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = score_of(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        progress = si / max(1, n_stages - 1)
        iters = max(30, min(150, int(50 + 100 * progress))
        lr = max(0.005, 0.05 * (1.0 - 0.5 * progress))

        z, stage_score = lse_adam_stage(z, stage_T, iters, lr)
        a = softmax_fn(z)
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/T={stage_T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

                        if ss < 1e-15:

def structured_init(n, rng, method=0):
    """Generate diverse structured initializations."""
    if method == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif method == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif method == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif method == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif method == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.15 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
        a = np.maximum(a, 0)
    elif method == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * (x - 0.3) ** 2) + 0.8 * np.exp(-6 * (x - 0.7) ** 2)
    elif method == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.6 + 0.4 * np.sin(8 * np.pi * x))
        a = np.maximum(a, 0)
    elif method == 7:
        # "Ideal flat autocorrelation" approximation: heavier at edges
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.3 + 0.7 * np.abs(x - 0.5))
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        a = np.ones(n)

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interp_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            if it > 100 and not improved:
                break

        if it % 50 == 0:
            print(f"    TR it={it}: best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search function. Multi-phase optimization to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(2024)
    t_start = time.monotonic()
    time_budget = 1740.0  # Leave 60s buffer

    solver = get_solver()
    print(f"LP solver: {solver}")

    # Load parent
    best_seq = None
    best_score = float('inf')
    n_base = 1600
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = score_of(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent found.")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            el = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={el:.0f}s)")

    if parent is None:
        best_seq = structured_init(n_base, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Diverse initialization + quick LP (200s) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: Init + Quick LP (remaining={remaining:.0f}s) ===")

    # Try a few structured inits and pick the best
    inits = []
    for m in range(0, 8):
        a_try = structured_init(n_base, rng, m)
        sc = score_of(a_try)
        inits.append((sc, a_try, m))
        if m % 2 == 0:
            print(f"  Init {m}: score={sc:.8f}")
    inits.sort(key=lambda x: x[0])
    print(f"  Best init: method={inits[0][2]}, score={inits[0][0]:.8f}")

    # Use the better of parent and best structured init
    if inits[0][0] < best_score:
        best_seq = inits[0][1].copy()
        best_score = inits[0][0]

    # Quick Nesterov LP to warm up
    t1 = min(remaining * 0.25, 200.0)
    if t1 > 20:
        a_out, _ = nesterov_lp(best_seq, t1, solver, damp=1.0)
        try_update(a_out, "quick_lp")

    # === Phase 2: Deep Nesterov LP (500s) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: Deep Nesterov LP (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 30:
        a_out, _ = nesterov_lp(best_seq, t2, solver, damp=0.95)
        try_update(a_out, "deep_lp_1")

    # Second pass with different damping
    remaining = time_budget - (time.monotonic() - t_start)
    t2b = min(remaining * 0.15, 200.0)
    if t2b > 20:
        a_out, _ = nesterov_lp(best_seq, t2b, solver, damp=1.0)
        try_update(a_out, "deep_lp_2")

    # === Phase 3: LSE optimization (300s) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: LSE optimization (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_out, _ = lse_optimize(best_seq, t3, T_start=60.0, T_end=0.01, n_stages=25)
        try_update(a_out, "lse_1")

    # Second LSE with different params
    remaining = time_budget - (time.monotonic() - t_start)
    t3b = min(remaining * 0.10, 150.0)
    if t3b > 15:
        a_out, _ = lse_optimize(best_seq, t3b, T_start=20.0, T_end=0.005, n_stages=15)
        try_update(a_out, "lse_2")

    # === Phase 4: Targeted refinement (200s) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: Targeted refinement (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, _ = targeted_refine(best_seq, t4, n_peaks=3)
        try_update(a_out, "targeted")

    # === Phase 5: Peak shaving (200s) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 5: Peak shaving (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 10:
        a_out, _ = peak_shave(best_seq, t5, seed=123)
        try_update(a_out, "pshave_1")

    remaining = time_budget - (time.monotonic() - t_start)
    t5b = min(remaining * 0.08, 100.0)
    if t5b > 10:
        a_out, _ = peak_shave(best_seq, t5b, seed=456)
        try_update(a_out, "pshave_2")

    # === Phase 6: Multi-length exploration (200s) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 6: Multi-length exploration (remaining={remaining:.0f}s) ===")

    n_best = len(best_seq)
    candidate_ns = []
    for offset in [-200, 200, -400, 400]:
        nl = n_best + offset
        if 800 <= nl <= 3000:
            candidate_ns.append(nl)
    candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_best))

    if candidate_ns and remaining > 60:
        time_per_n = min(remaining * 0.6 / len(candidate_ns), 120.0)
        for nl in candidate_ns[:3]:
            remaining = time_budget - (time.monotonic() - t_start)
            if remaining < 30:
                break
            tl = min(time_per_n, remaining * 0.35)
            if tl < 15:
                continue
            print(f"  Trying n={nl} (time={tl:.0f}s)")
            init_n = interp_seq(best_seq, nl)

            # Quick LP
            t_lp = tl * 0.6
            a_out, _ = nesterov_lp(init_n, t_lp, solver, damp=0.9)
            try_update(a_out, f"lp_n{nl}")

            # Quick LSE
            remaining = time_budget - (time.monotonic() - t_start)
            t_lse = max(5, min(tl * 0.3, remaining * 0.1))
            if t_lse > 5:
                a_out2, _ = lse_optimize(a_out, t_lse, T_start=40.0, T_end=0.05, n_stages=10)
                try_update(a_out2, f"lse_n{nl}")

    # === Phase 7: Final polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish at n={n_best} (remaining={remaining:.0f}s) ===")

    if remaining > 30:
        # One more LP round
        t7a = min(remaining * 0.3, 100.0)
        if t7a > 10:
            a_out, _ = nesterov_lp(best_seq, t7a, solver, damp=1.0)
            try_update(a_out, "final_lp")

        # LSE with very low final temperature
        remaining = time_budget - (time.monotonic() - t_start)
        t7b = min(remaining * 0.3, 80.0)
        if t7b > 10:
            a_out, _ = lse_optimize(best_seq, t7b, T_start=10.0, T_end=0.003, n_stages=12)
            try_update(a_out, "final_lse")

        # Peak shaving
        remaining = time_budget - (time.monotonic() - t_start)
        t7c = min(remaining * 0.3, 60.0)
        if t7c > 5:
            a_out, _ = peak_shave(best_seq, t7c, seed=789)
            try_update(a_out, "final_ps")

        # Targeted refinement
        remaining = time_budget - (time.monotonic() - t_start)
        t7d = min(remaining * 0.2, 40.0)
        if t7d > 5:
            a_out, _ = targeted_refine(best_seq, t7d, n_peaks=5)
            try_update(a_out, "final_tr")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        print(f"Total time: {time.monotonic() - t_start:.1f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
