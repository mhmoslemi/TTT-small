def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 500:
        max_Tb = 500
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.003, 0.2)
        else:
            eta *= 0.9

    return best_a, best_score


def coordinate_polish(a0, time_limit, n_candidates=5):
    """Coordinate descent with candidate enumeration and incremental conv."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for sweep in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        conv = fast_convolve(a)
        S = a.sum()
        score_base = 2.0 * n * conv.max() / (S * S)

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                # Try small positive values
                candidates = [1e-4, 1e-3, 5e-3, 1e-2]
            else:
                candidates = [old_val * 0.3, old_val * 0.6, old_val, old_val * 1.5, old_val * 2.5]

            best_cand = old_val
            best_cand_score = score_base
            for x in candidates:
                x = max(0.0, x)
                a[i] = x
                S_new = S - old_val + x
                if S_new < 0.01:
                    a[i] = old_val
                    continue
                # Compute new conv incrementally
                # a[i] changed from old_val to x, delta = x - old_val
                delta = x - old_val
                if abs(delta) < 1e-15:
                    a[i] = old_val
                    continue
                # Affected k range: [i, min(i+n-1, 2n-2)]
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                conv_new = conv.copy()
                for k in range(k_start, k_end + 1):
                    ki = k - i
                    if 0 <= ki <= n - 1:
                        if k == 2 * i:
                            conv_new[k] += (x * x - old_val * old_val)
                        else:
                            conv_new[k] += 2.0 * delta * a[ki]
                max_new = conv_new.max()
                sc_new = 2.0 * n * max_new / (S_new * S_new)
                if sc_new < best_cand_score:
                    best_cand_score = sc_new
                    best_cand = x
                a[i] = old_val  # restore

            if best_cand != old_val and best_cand_score < score_base:
                a[i] = best_cand
                S = S - old_val + best_cand
                delta = best_cand - old_val
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                for k in range(k_start, k_end + 1):
                    ki = k - i
                    if 0 <= ki <= n - 1:
                        if k == 2 * i:
                            conv[k] += (best_cand * best_cand - old_val * old_val)
                        else:
                            conv[k] += 2.0 * delta * a[ki]
                score_base = 2.0 * n * conv.max() / (S * S)
                improved = True
                if score_base < best_score:
                    best_score = score_base
                    best_a = a.copy()

        if not improved:
            break

    return best_a, best_score


def subgradient_fast(a0, time_limit, max_steps=200000):
    """Fast subgradient with top-k peaks and Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    z_prev = z.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.5
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.90)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_prev)
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z_prev = z
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.002:
                t_nest = 1.0
                eta *= 0.85
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z_prev = z
                z = z_new
            eta = max(eta * 0.999, 0.05)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.9f}")
    except Exception:
        pass

    if best_seq is None:
        # Structured init
        n = n_base
        x = np.linspace(0, 1, n)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.9f}")

    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.9f} (n={len(arr)})")

    # === Phase 1: Quick subgradient (30s) ===
    remaining = _global_deadline - time.monotonic()
    t1 = min(30.0, remaining * 0.02)
    if t1 > 3:
        print(f"\n=== Phase 1: Subgradient (score={best_score:.9f}) ===")
        a_out, _ = subgradient_fast(best_seq, t1)
        try_update(a_out, "subgrad_p1")
    print(f"  Phase 1 done: {best_score:.9f}")

    # === Phase 2: LP-Nesterov main (550s) ===
    remaining = _global_deadline - time.monotonic()
    t2 = min(550.0, remaining * 0.5)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-Nesterov (score={best_score:.9f}) ===")
        a_out, _ = nesterov_lp_improved(best_seq, t2, solver=solver)
        try_update(a_out, "nlp_main")
    print(f"  Phase 2 done: {best_score:.9f}")

    # === Phase 3: Aggressive LSE (300s) ===
    remaining = _global_deadline - time.monotonic()
    t3 = min(300.0, remaining * 0.25)
    if t3 > 15:
        print(f"\n=== Phase 3: LSE aggressive (score={best_score:.9f}) ===")
        a_lse = lse_lbgfs_aggressive(best_seq, t3, T_start=40.0, T_end=0.001, n_stages=45)
        try_update(a_lse, "lse_agg")
    print(f"  Phase 3 done: {best_score:.9f}")

    # === Phase 4: Peak shaving adaptive (250s) ===
    remaining = _global_deadline - time.monotonic()
    t4 = min(250.0, remaining * 0.2)
    if t4 > 10:
        print(f"\n=== Phase 4: Peak shave (score={best_score:.9f}) ===")
        a_pt, _ = peak_shave_adaptive(best_seq, t4, max_steps=80000)
        try_update(a_pt, "peak_shave")
    print(f"  Phase 4 done: {best_score:.9f}")

    # === Phase 5: LP-Nesterov round 2 (250s) ===
    remaining = _global_deadline - time.monotonic()
    t5 = min(250.0, remaining * 0.2)
    if t5 > 15:
        print(f"\n=== Phase 5: LP-Nesterov r2 (score={best_score:.9f}) ===")
        a_out2, _ = nesterov_lp_improved(best_seq, t5, solver=solver)
        try_update(a_out2, "nlp_r2")
    print(f"  Phase 5 done: {best_score:.9f}")

    # === Phase 6: Final LSE polish (150s) ===
    remaining = _global_deadline - time.monotonic()
    t6 = min(150.0, remaining * 0.15)

        print(f"\n=== Phase 6: LSE final (score={best_score:.9f}) ===")
        a_lse2 = lse_lbgfs_aggressive(best_seq, t6, T_start=15.0, T_end=0.0005, n_stages=30)
        try_update(a_lse2, "lse_final")
    print(f"  Phase 6 done: {best_score:.9f}")

    # === Phase 7: Coordinate polish (100s) ===
    remaining = _global_deadline - time.monotonic()
    t7 = min(100.0, remaining * 0.1)
    if t7 > 10:
        print(f"\n=== Phase 7: Coordinate polish (score={best_score:.9f}) ===")
        a_cp, _ = coordinate_polish(best_seq, t7, n_candidates=4)
        try_update(a_cp, "coord_polish")
    print(f"  Phase 7 done: {best_score:.9f}")

    # === Phase 8: Length exploration ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Length exploration (score={best_score:.9f}) ===")
        for n_try in [len(best_seq) + 100, len(best_seq) - 100, len(best_seq) + 250]:
            remaining = _global_deadline - time.monotonic()

                break
            if n_try < 200:

    if remaining > 20:

            try_update(a_final, "final_polish")

    # Final verification

        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.9f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
