def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.array([1.0]) if n == 1 else v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u * ind > (cssv - 1.0)
    rho_list = np.nonzero(cond)[0]
    if len(rho_list) == 0:
        return np.ones(n) / n
    rho = rho_list[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad(a, w):
    """
    Compute gradient of LogSumExp(T * max-proxy) w.r.t. a.
    grad_j = 2 * sum_m w_{j+m} * a_m  (cross-correlation of w and a)
    w has length 2n-1, a has length n.
    """
    n = len(a)
    L = 2 * n - 1
    fft_size = 1 << (L + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(np.conj(W) * A, fft_size)
    return 2.0 * cross[:n]

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        else:
            t_prev = t_k
            t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
            momentum = (t_prev - 1.0) / t_k
            a_cand = project_simplex(y + momentum * (y - a) - lr * grad)

        # Backtracking line search (simplified: just accept if it improves the smoothed obj)
        val_old = lse_smooth_value(fast_autocorr(a), T)
        val_new = lse_smooth_value(fast_autocorr(a_cand), T)

        if val_new <= val_old + 1e-10:
            a_new = a_cand
        else:
            # Halve step
            lr = lr * 0.5
            a_new = project_simplex(y - lr * grad)
            val_new2 = lse_smooth_value(fast_autocorr(a_new), T)
            if val_new2 > val_old + 1e-10:
                a_new = a  # accept no change
                lr = lr * 2.0  # restore

        # Track best by true objective
        c_new = fast_autocorr(a_new)
        score_new = 2.0 * n * c_new.max()
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        y = a
        a = a_new

        # Restart if stalled for too long
        if stall_count > 3000:
            rng = np.random.default_rng(step + 1000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            y = a.copy()
            t_k = 1.0
            lr = 1.0 / (T * 3.0)
            stall_count = 0

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD step={step} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.3f}, lr={lr:.6f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search function: multi-phase optimization combining gradient descent,
    LP best-response, and targeted peak shaving.
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = 1740.0  # Leave 60s buffer

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using structured init")
        best_seq = structured_init(n_base, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Structured init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Nesterov LSE gradient descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov LSE at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 30:
        a_out, _ = nesterov_lse_optimize(best_seq, t1, T_start=50.0, T_end=3.0,
                                         n_steps=300000)
        try_update(a_out, f"nls_n{n_base}")

    # === Phase 2: Gradient descent with lower temperature ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: GD refine at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_out, _ = gradient_descent_refine(best_seq, t2, T_start=20.0, T_end=0.5,
                                           n_steps=200000)
        try_update(a_out, f"gd_n{n_base}")

    # === Phase 3: LP best-response ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP BR at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, n_steps=3000)
        try_update(a_out, f"nlp_n{n_base}")

    # === Phase 4: Targeted peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Peak shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 10:
        a_out, _ = peak_shave_targeted(best_seq, t4, seed=777)
        try_update(a_out, f"ps_n{len(best_seq)}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)
    candidate_lengths = [n_best - 200, n_best + 200, n_best - 400, n_best + 400]
    candidate_lengths = [nl for nl in candidate_lengths if 600 <= nl <= 2500]
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))

    for nl in candidate_lengths[:2]:  # Try at most 2 different lengths
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        tl = min(remaining * 0.25, 200.0)
        if tl < 20:
            continue

        print(f"  Trying n={nl} (time={tl:.0f}s)")
        init = interpolate_sequence(best_seq, nl)

        # Quick gradient descent
        a_out, _ = gradient_descent_refine(init, tl * 0.6, T_start=30.0, T_end=2.0,
                                           n_steps=150000)
        try_update(a_out, f"gd_n{nl}")

        # Quick peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        t_ps = min(remaining * 0.1, 60.0)
        if t_ps > 10:
            a_out2, _ = peak_shave_targeted(a_out, t_ps, seed=nl % 10000)
            try_update(a_out2, f"ps_n{nl}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish at n={n_best} (remaining={remaining:.0f}s) ===")

    t6a = min(remaining * 0.30, 120.0)
    if t6a > 10:
        a_out, _ = gradient_descent_refine(best_seq, t6a, T_start=10.0, T_end=0.3,
                                           n_steps=150000)
        try_update(a_out, f"gd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    t6b = min(remaining * 0.30, 100.0)
    if t6b > 10:
        a_out, _ = peak_shave_targeted(best_seq, t6b, seed=31415)
        try_update(a_out, f"ps_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    t6c = min(remaining * 0.30, 80.0)
    if t6c > 10:
        a_out, _ = nesterov_lse_optimize(best_seq, t6c, T_start=30.0, T_end=1.0,
                                         n_steps=100000)
        try_update(a_out, f"nls_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
