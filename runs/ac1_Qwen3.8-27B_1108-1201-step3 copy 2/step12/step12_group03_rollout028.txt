import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.fft import rfft, irfft


def fast_autocorr(a):

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_gradient(a, beta):
    """
    Compute gradient of LSE_beta(conv(a,a)) w.r.t. a.
    Returns (gradient, softmax_weights, conv).
    """
    n = len(a)
    conv = fast_autocorr(a)
    # Softmax of beta * conv
    bc = beta * conv
    bc -= bc.max()
    exp_bc = np.exp(bc)
    p = exp_bc / exp_bc.sum()

    # corr2[j] = sum_i a[i]*p[j-i] for j=0..n-1 (convolution a*p)
    full2 = fftconvolve(a, p, 'full')
    corr2 = full2[:n]

    # corr1[j] = sum_i a[i]*p[j+i] for j=0..n-1
    p_rev = p[::-1]
    full1 = fftconvolve(a, p_rev, 'full')
    corr1 = full1[n - 1:2 * n - 1][::-1]

    grad = (corr1 + corr2) / beta
    return grad, p, conv


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_lse_optimize(a0, time_limit, beta_start=5.0, beta_end=800.0,

        a = a / s

    v = np.zeros(n)

                stall_count = 0
            else:
                stall_count += 1

        # Periodic restart: perturb and reset momentum
        if t % restart_interval == 0 and stall_count > 5:

            n_pert = max(2, n // 100)
            src_idx = rng.choice(n, size=n_pert, replace=False)

            for si, di in zip(src_idx, dst_idx):

def nesterov_momentum_lse(a0, time_limit, beta_start=3.0, beta_end=400.0,
                          lr_start=0.1, lr_end=0.001, n_anneal=8000,
                          score_interval=500, seed=42):
    """
    Nesterov momentum method with LSE smoothing.
    Often better than Adam for convex/smooth problems.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    t = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1

        progress = min(1.0, t / n_anneal)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr_t = lr_start * (lr_end / lr_start) ** progress

        # Nesterov lookahead
        momentum = (t - 1.0) / (t + 1.9)
        y = a + momentum * (a - (y - momentum * (a - y)))  # approximate
        # Simpler: y = a + momentum*(a - prev_a), but let's use standard form
        # Standard Nesterov: y_{k+1} = x_k + (t_k - 1)/(t_{k+1} + 1) * (x_k - x_{k-1})
        # Let's just do: grad at y, step from x

        grad_y, _, _ = compute_lse_gradient(y, beta)
        a_new = y - lr_t * grad_y
        a_new = project_simplex(a_new)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        # Update Nesterov state
        if t == 1:
            y = a_new.copy()
        else:
            momentum_t = (t - 1.0) / (t + 1.5)
            y = a_new + momentum_t * (a_new - a)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy

        a = a_new

        if t % score_interval == 0:
            conv_cur = fast_autocorr(a)
            score_cur = 2.0 * n * conv_cur.max()
            if score_cur < best_score:
                best_score = score_cur
                best_a = a.copy()

    return best_a, best_score


def lp_best_response_step(a_fixed):
    """
    One LP best-response step: find b that minimizes max(conv(a_fixed, b))
    subject to sum(b)=1, b>=0.
    Uses scipy.optimize.linprog.
    Returns b, M.
    """
    from scipy.optimize import linprog
    n = len(a_fixed)
    n_cons = 2 * n - 1

    # Variables: b[0..n-1], M
    # Minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    # Inequality constraints: conv(a_fixed, b)[k] - M <= 0 for k=0..2n-2
    # conv(a_fixed, b)[k] = sum_j a_fixed[j] * b[k-j] (valid indices)
    A_ub = np.zeros((n_cons, n + 1))
    for k in range(n_cons):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a_fixed[j + (k - j) - (k - j)]  # a_fixed index
            # conv[k] = sum_j a[j]*b[k-j], so coefficient of b[j] in conv[k] is a[k-j]
            A_ub[k, j] = a_fixed[k - j]
        A_ub[k, -1] = -1.0  # -M

    b_ub = np.zeros(n_cons)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: b >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = res.x[-1]
    return b_val, M_val

            best_score = score_a
            best_a = a.copy()
            if verbose and t % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP t={t}: score={best_score:.8f}, M={M_val:.8f}")

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

    return best_a, best_score


def subgradient_peak_shave(a0, time_limit, n_subgrad_iters=2000, seed=42):
    """
    Subgradient method on the exact max-autocorrelation objective,
    with projection onto simplex and occasional random restarts.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    for t in range(n_subgrad_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_conv = conv.max()
        # Subgradient: subgradient of max_k conv[k] w.r.t. a
        # = weighted average of subgradients at active peaks
        # Find peaks within 1% of max
        threshold = max_conv * 0.99
        active = np.where(conv >= threshold)[0]

        # Weight by proximity to max
        weights = np.exp((conv[active] - max_conv) * 20.0)
        weights /= weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                # d(conv[k])/da[j] = a[k-j] + a[j-k] (if valid)
                g = 0.0
                if 0 <= k - j < n:
                    g += a[k - j]
                if 0 <= j - k < n:
                    g += a[j - k]
                grad[j] += w * g

        # Learning rate
        lr = 0.5 / (np.sqrt(t + 1) * max(1.0, np.linalg.norm(grad)))

        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        a = a_new

        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Random restart
        if stall > 200 and t % 50 == 0:
            n_pert = max(3, n // 50)
            src = rng.choice(n, size=n_pert, replace=False)
            dst = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for si, di in zip(src, dst):
                amt = a_trial[si] * rng.uniform(0.02, 0.1)
                a_trial[si] -= amt
                a_trial[di] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial /= st
                sc = 2.0 * n * fast_autocorr(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, seed=42):
    """
    Coordinate descent: for each coordinate j, find optimal a_j
    given all others fixed (1D optimization).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            # Fix all except a[j], optimize a[j] in [0, sum_others]
            # conv[k] as function of a[j] is quadratic (parabola opening up)
            # We want to minimize max_k conv[k]
            # Try bisection / golden section on the 1D problem

            a_fixed = a.copy()
            a_j = a[j]
            a_fixed[j] = 0.0
            remaining = a_fixed.sum()
            if remaining < 1e-15:
                break

            # conv[k] with a[j]=0
            conv_base = fast_autocorr(a_fixed)
            # Additional contribution from a[j]:
            # conv[k] += a[j] * (a_fixed[k-j] + a_fixed[j-k]) for valid indices
            # Let h[k] = a_fixed[k-j] + a_fixed[j-k] (with bounds)
            h = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                if 0 <= k - j < n:
                    val += a_fixed[k - j]
                if 0 <= j - k < n:
                    val += a_fixed[j - k]
                h[k] = val
            # But conv_base already has a[j]=0, so:
            # conv_full[k] = conv_base[k] + a[j]*h[k] + a[j]^2 * delta(k-2j) if 2j < 2n-1
            # Actually conv_base is computed with a[j]=0, so it doesn't include a[j] terms.
            # conv_full[k] = conv_base[k] + a[j]*h[k]  (since a[j]^2 would be in h at k=2j)
            # Wait: if k=2j, then h[2j] = a_fixed[2j-j] + a_fixed[j-2j] = a_fixed[j] + ... 
            # But a_fixed[j] = 0, so h[2j] = a_fixed[j] + a_fixed[-j] = 0 (if j>0)
            # Hmm, this is getting complicated. Let me just evaluate directly.

            # Golden section search for optimal a[j] in [0, remaining]
            gr = (np.sqrt(5) - 1) / 2
            lo, hi = 0.0, remaining

            def obj_1d(val):
                a_trial = a_fixed.copy()
                a_trial[j] = val
                return fast_autocorr(a_trial).max()

            # Evaluate at a few points first
            candidates = [a_j, remaining * 0.1, remaining * 0.3, remaining * 0.5,
                          remaining * 0.7, remaining * 0.9, remaining]
            best_val = a_j
            best_obj = obj_1d(a_j)
            for cv in candidates:
                ov = obj_1d(cv)
                if ov < best_obj:
                    best_obj = ov
                    best_val = cv

            # Local golden section around best_val
            span = remaining * 0.05
            lo2 = max(0.0, best_val - span)
            hi2 = min(remaining, best_val + span)
            if hi2 - lo2 > 1e-10:
                for _ in range(20):
                    c1 = hi2 - gr * (hi2 - lo2)
                    c2 = lo2 + gr * (hi2 - lo2)
                    f1 = obj_1d(c1)
                    f2 = obj_1d(c2)
                    if f1 < f2:
                        hi2 = c2
                    else:
                        lo2 = c1
                    if hi2 - lo2 < 1e-8:
                        break
                mid = (lo2 + hi2) / 2
                f_mid = obj_1d(mid)
                if f_mid < best_obj:
                    best_obj = f_mid
                    best_val = mid

            if best_val != a_j:
                a[j] = best_val
                # Renormalize
                a = np.maximum(a, 0)
                a = a / a.sum()
                sc = 2.0 * n * fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0

    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_lp_solver_check():
    """Check if cvxpy solvers are available."""
    try:
        import cvxpy as cp
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                x = cp.Variable(3, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.8])
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver
            except:
                continue
        return cp.ECOS
    except:
        return None


def solve_lp_cvxpy(a_fixed, solver):
    """Solve LP best response using cvxpy."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build conv constraint matrix
    # conv(a_fixed, b)[k] = sum_j a_fixed[k-j] * b[j] for valid j
    n_cons = 2 * n - 1
    A = np.zeros((n_cons, n))
    for k in range(n_cons):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a_fixed[k - j]

    constraints = [A @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=5000)
    except:
        try:
            prob.solve(solver=cp.ECOS, max_iter=5000)
        except:
            return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_cvxpy(a0, time_limit, solver, max_iters=300):
    """Nesterov LP best-response using cvxpy."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_cvxpy(y, solver)
        if b_star is None:
            continue

        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / sa

        y = y_new
        a = a_new

        sa_score = 2.0 * n * fast_autocorr(a).max()
        if sa_score < best_score:
            best_score = sa_score
            best_a = a.copy()

        sy_score = 2.0 * n * fast_autocorr(y).max()
        if sy_score < best_score:
            best_score = sy_score
            best_a = y.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    1. Adam LSE from diverse initializations
    2. Nesterov momentum LSE
    3. LP best-response refinement
    4. Subgradient peak shaving
    5. Coordinate descent polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence([float(x) for x in parent]):.6f}")
    except:
        pass

    n_base = len(parent) if parent is not None else 1600

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_sequence([float(x) for x in a])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is not None:
        best_a = parent.copy()
        best_score = evaluate_sequence([float(x) for x in parent])
    else:
        best_a = np.ones(n_base) / n_base
        best_score = evaluate_sequence([float(x) for x in best_a])
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # ===== Phase 1: Adam LSE from diverse inits =====
    print(f"\n=== Phase 1: Adam LSE (n={n_base}) ===")
    elapsed = time.monotonic() - t_global
    phase1_budget = min(time_budget * 0.40, 700.0)

    # Generate inits
    inits = []
    if parent is not None:
        inits.append(('parent', parent.copy()))

    x = np.arange(n_base) / n_base
    inits.append(('uniform', np.ones(n_base) / n_base))
    for sigma in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append((f'gauss_{sigma}', g / g.sum()))
    tri = 1.0 - np.abs(2.0 * x - 1.0)
    inits.append(('tri', tri / tri.sum()))
    # Bimodal
    bim = np.exp(-((x - 0.3) / 0.12) ** 2) + np.exp(-((x - 0.7) / 0.12) ** 2)
    inits.append(('bimodal', bim / bim.sum()))
    # Slight perturbation of parent
    if parent is not None:
        rng_p = np.random.default_rng(seed + 1)
        pert = parent + rng_p.normal(0, 0.02, n_base)
        pert = np.maximum(pert, 0)
        inits.append(('parent_pert', pert / pert.sum()))

    # Skip uniform if we have many inits (it's the worst starting point)
    # Priority: parent, parent_pert, gauss, bimodal, tri, uniform
    order = ['parent', 'parent_pert', 'gauss_0.15', 'gauss_0.25', 'bimodal',
             'gauss_0.4', 'tri', 'uniform']
    inits_sorted = []
    for name in order:
        for nm, arr in inits:
            if nm == name:
                inits_sorted.append((nm, arr))
                break
    # Add any remaining
    sorted_names = set(nm for nm, _ in inits_sorted)
    for nm, arr in inits:
        if nm not in sorted_names:
            inits_sorted.append((nm, arr))

    n_inits = len(inits_sorted)
    for idx, (name, init) in enumerate(inits_sorted):
        elapsed = time.monotonic() - t_global
        if elapsed > phase1_budget + t_global:
            break
        remaining = phase1_budget - (elapsed - t_global)
        if remaining < 30:
            break
        # Allocate time: more for early inits
        t_alloc = min(remaining * 0.4, 250.0)
        if t_alloc < 20:
            continue

        print(f"  Adam LSE from '{name}' (t={t_alloc:.0f}s)")
        a_out, sc_out = adam_lse_optimize(init, t_alloc,
                                           beta_start=3.0, beta_end=600.0,
                                           lr=0.06, n_anneal=8000,
                                           score_interval=100,
                                           restart_interval=4000,
                                           seed=seed + idx)
        try_update(a_out, f"adam_{name}")

    # ===== Phase 2: Nesterov momentum LSE from best =====
    print(f"\n=== Phase 2: Nesterov LSE (n={len(best_a)}) ===")
    elapsed = time.monotonic() - t_global
    remaining_total = time_budget - (elapsed - t_global)
    phase2_budget = min(remaining_total * 0.15, 250.0)
    if phase2_budget > 30:
        a_out, sc_out = nesterov_momentum_lse(best_a, phase2_budget,
                                               beta_start=3.0, beta_end=400.0,
                                               lr_start=0.05, lr_end=0.002,
                                               n_anneal=6000, score_interval=200,
                                               seed=seed + 100)
        try_update(a_out, "nesterov_lse")

    # ===== Phase 3: LP best-response =====
    print(f"\n=== Phase 3: LP Best-Response ===")
    elapsed = time.monotonic() - t_global
    remaining_total = time_budget - (elapsed - t_global)
    phase3_budget = min(remaining_total * 0.25, 400.0)

    if phase3_budget > 60:
        # Check cvxpy
        solver = get_lp_solver_check()
        if solver is not None:
            print(f"  Using cvxpy solver: {solver}")
            # Run LP from best_a
            t_lp1 = min(phase3_budget * 0.6, 250.0)
            a_out, sc_out = nesterov_lp_cvxpy(best_a, t_lp1, solver, max_iters=200)
            try_update(a_out, f"lp_{len(best_a)}")

            # Run from a slightly different init (mix with uniform)
            if time.monotonic() - t_global < t_global + time_budget - 120:
                mix_a = 0.7 * best_a + 0.3 * np.ones(len(best_a)) / len(best_a)
                mix_a = mix_a / mix_a.sum()
                t_lp2 = min(phase3_budget * 0.4, 150.0)
                a_out2, sc_out2 = nesterov_lp_cvxpy(mix_a, t_lp2, solver, max_iters=150)
                try_update(a_out2, f"lp_mix_{len(best_a)}")
        else:
            print("  cvxpy not available, using scipy linprog")
            n_cur = len(best_a)
            if n_cur <= 400:  # linprog is slow for large n
                a_out, sc_out = nesterov_lp_refine(best_a, phase3_budget, n_max_iters=100)
                try_update(a_out, f"lp_linprog_{n_cur}")

    # ===== Phase 4: Subgradient peak shaving =====
    print(f"\n=== Phase 4: Subgradient Peak Shaving ===")
    elapsed = time.monotonic() - t_global
    remaining_total = time_budget - (elapsed - t_global)
    phase4_budget = min(remaining_total * 0.15, 200.0)
    if phase4_budget > 20:
        a_out, sc_out = subgradient_peak_shave(best_a, phase4_budget,
                                               n_subgrad_iters=5000, seed=seed + 200)
        try_update(a_out, "subgrad_shave")

    # ===== Phase 5: Coordinate descent polish =====
    print(f"\n=== Phase 5: Coordinate Descent ===")
    elapsed = time.monotonic() - t_global
    remaining_total = time_budget - (elapsed - t_global)
    phase5_budget = min(remaining_total * 0.15, 200.0)
    if phase5_budget > 15:
        a_out, sc_out = coordinate_descent_refine(best_a, phase5_budget, seed=seed + 300)
        try_update(a_out, "coord_descent")

    # ===== Phase 6: Final Adam LSE (short, from best) =====
    elapsed = time.monotonic() - t_global
    remaining_total = time_budget - (elapsed - t_global)
    if remaining_total > 30:
        print(f"\n=== Phase 6: Final Adam LSE polish ===")
        t_final = min(remaining_total * 0.5, 100.0)
        if t_final > 15:
            a_out, sc_out = adam_lse_optimize(best_a, t_final,
                                               beta_start=50.0, beta_end=500.0,
                                               lr=0.02, n_anneal=3000,
                                               score_interval=50,
                                               restart_interval=2000,
                                               seed=seed + 400)
            try_update(a_out, "final_adam_polish")

    # ===== Final verification =====
    if best_a is not None:
        final_score = evaluate_sequence([float(x) for x in best_a])
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\n{'='*50}")
        print(f"FINAL SCORE: {best_score:.6f} (n={len(best_a)})")
        print(f"{'='*50}")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_a]
