import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, n=fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of a (for softmax parametrization)."""
    return np.log(np.maximum(a, 1e-300))


def convex_combination_gradient(p, peak_indices, n):
    """
    Compute weighted subgradient of max_k (p*p)[k].
    peak_indices: array of k values to use
    Returns gradient vector of length n.
    """
    g = np.zeros(n)
    for k in peak_indices:
        a_k = max(0, k - n + 1)
        b_k = min(n - 1, k)
        idx = np.arange(a_k, b_k + 1)
        # p[k-i] for i in idx
        rev_idx = k - idx
        g[idx] += 2.0 * p[rev_idx]
        # Subtract correction at center if k is even
        if k % 2 == 0:
            mid = k // 2
            if a_k <= mid <= b_k:
                g[mid] -= p[mid]
    return g


def subgradient_bb(p0, time_limit, K_peaks=5, restart_interval=2000,
                   patience=500, verbose=True):
    """
    Subgradient method with Barzilai-Borwein step size and restarts.
    Uses top-K peak weighting for a smoother direction.
    """
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_score = score_normalized(p)
    t0 = time.monotonic()

    # BB state
    p_prev = p.copy()
    g_prev = None
    bb_eta = 1.0 / np.sqrt(n)  # initial step size

    no_improve = 0
    restart_count = 0

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(p)
        max_c = c.max()
        cur_score = 2.0 * n * max_c

        # Find top-K peaks with quadratic weighting
        threshold = max_c * 0.92
        active_mask = c >= threshold
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask[np.argmax(c)] = True
            n_active = 1
        # Cap at K_peaks
        if n_active > K_peaks:
            active_indices = np.where(active_mask)[0]
            top_K = np.argsort(c[active_indices])[-K_peaks:]
            active_indices = active_indices[top_K]
        else:
            active_indices = np.where(active_mask)[0]

        # Weighted gradient
        weights = c[active_indices] ** 2
        weights /= weights.sum()
        g = np.zeros(n)
        for w, k in zip(weights, active_indices):
            a_k = max(0, k - n + 1)
            b_k = min(n - 1, k)
            idx = np.arange(a_k, b_k + 1)
            rev_idx = k - idx
            g[idx] += 2.0 * w * p[rev_idx]
            if k % 2 == 0:
                mid = k // 2
                if a_k <= mid <= b_k:
                    g[mid] -= w * p[mid]

        # BB step size update
        if g_prev is not None:
            s_vec = p - p_prev
            y_vec = g - g_prev
            s_dot_y = np.dot(s_vec, y_vec)
            y_norm_sq = np.dot(y_vec, y_vec)
            if y_norm_sq > 1e-20 and s_dot_y > 1e-10:
                bb_eta_new = s_dot_y / y_norm_sq
                bb_eta = 0.5 * bb_eta + 0.5 * bb_eta_new
                bb_eta = np.clip(bb_eta, 1e-6, 1.0 / np.sqrt(n))
        p_prev = p.copy()
        g_prev = g.copy()

        # Take step with projection
        p_trial = project_simplex(p - bb_eta * g)

        # Backtracking line search if needed
        trial_score = score_normalized(p_trial)
        if trial_score > cur_score * 1.05:
            # Reduce step
            for _ in range(8):
                bb_eta *= 0.5
                p_trial = project_simplex(p - bb_eta * g)
                trial_score = score_normalized(p_trial)
                if trial_score <= cur_score:
                    break

        p = p_trial

        if trial_score < best_score:
            best_score = trial_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve >= patience:
            restart_count += 1
            # Perturb from best
            rng = np.random.default_rng(hash((it, restart_count)) & 0xFFFFFFFF)
            noise_scale = 0.02 + 0.03 * min(restart_count, 5)
            p_pert = best_p + noise_scale * rng.normal(0, 1, n)
            p = project_simplex(p_pert)
            no_improve = 0
            bb_eta = 1.0 / np.sqrt(n)
            g_prev = None

        if verbose and (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradBB iter={it+1} ({elapsed:.0f}s): "
                  f"cur={cur_score:.8f} best={best_score:.8f} "
                  f"eta={bb_eta:.6f} restarts={restart_count}")

    return best_p, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs_adaptive(a0, time_limit, T_start=80.0, T_min=0.2,
                       n_stages=30, iters_per_stage=30, verbose=True):
    """Adaptive LogSumExp L-BFGS with progress-based temperature control."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0
    prev_score = None

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            if prev_score is not None:
                if sc > best_score * 1.003:
                    stagnant += 1
                else:
                    stagnant = max(0, stagnant - 1)

            if stagnant >= 3:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            prev_score = sc

            if verbose and (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: "
                      f"score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def randomized_perturb_search(p_best, time_limit, n_perturbs=20,
                              perturb_scale=0.03, opt_time_frac=0.7,
                              verbose=True):
    """
    Try random perturbations of the best solution, optimize each briefly.
    Returns the best found.
    """
    n = len(p_best)
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()
    best_p = p_best.copy()
    best_score = score_normalized(p_best)
    time_per = time_limit * opt_time_frac / n_perturbs

    for i in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        # Different perturbation strategies
        strategy = i % 4
        if strategy == 0:
            # Gaussian noise
            pert = p_best + perturb_scale * rng.normal(0, 1, n)
        elif strategy == 1:
            # Shift mass from one side to the other
            shift = rng.uniform(0.05, 0.15)
            pert = p_best.copy()
            n_shift = n // 10
            pert[:n_shift] *= (1 - shift)
            pert[-n_shift:] *= (1 + shift)
            pert = np.maximum(pert, 0)
        elif strategy == 2:
            # Frequency domain perturbation
            F = rfft(p_best)
            F[1:-1] *= (1 + 0.1 * rng.normal(0, 1, len(F[1:-1])))
            pert = irfft(F, len(p_best))
            pert = np.maximum(pert, 0)
        else:
            # Swap two random blocks
            pert = p_best.copy()
            b1 = rng.integers(0, n - n // 5)
            b2 = rng.integers(0, n - n // 5)
            sz = n // 10
            pert[b1:b1+sz], pert[b2:b2+sz] = pert[b2:b2+sz].copy(), pert[b1:b1+sz].copy()

        p = project_simplex(pert)
        sc = score_normalized(p)

        # Quick optimization
        if time.monotonic() - t0 + time_per < time_limit:
            p_opt, _ = subgradient_bb(p, time_per, K_peaks=5,
                                       restart_interval=500, patience=200,
                                       verbose=False)
            sc_opt = score_normalized(p_opt)
            if sc_opt < best_score:
                best_score = sc_opt
                best_p = p_opt.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    Perturb {i} ({elapsed:.0f}s): NEW BEST {best_score:.8f}")

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines subgradient, LSE, multi-length, and
    perturbation-based search to minimize the evaluation function.
    """
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    t_start = time.monotonic()

    print(f"Total budget: {budget_s}s, effective deadline: {global_deadline - time.monotonic():.0f}s")

    # Get starting sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remain={remain:.0f}s)")

    # === Phase 1: Subgradient BB optimization (35% time) ===
    print(f"\n=== Phase 1: Subgradient BB at n={n_base} (score={best_score:.8f}) ===")
    t1 = 0.35 * (budget_s - 35)
    remain = global_deadline - time.monotonic()
    t1 = min(t1, remain)
    if t1 > 30:
        a_out, sc_out = subgradient_bb(best_seq, t1, K_peaks=5,
                                        restart_interval=3000, patience=800,
                                        verbose=True)
        try_update(a_out, "subgrad_bb_main")
        # Second pass with different K
        remain = global_deadline - time.monotonic()
        if remain > 20:
            a_out2, _ = subgradient_bb(best_seq, remain * 0.6, K_peaks=10,
                                        restart_interval=2000, patience=500,
                                        verbose=False)
            try_update(a_out2, "subgrad_bb_K10")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: LSE L-BFGS polishing (25% time) ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t2 = min(0.25 * (budget_s - 35), remain)
    if t2 > 30:
        # High temperature pass
        a_lse1, _ = lse_lbfgs_adaptive(best_seq, t2 * 0.4, T_start=100.0,
                                        T_min=0.3, n_stages=20,
                                        iters_per_stage=30, verbose=True)
        try_update(a_lse1, "lse_high_T")

        # Low temperature pass
        remain = global_deadline - time.monotonic()
        if remain > 20:
            a_lse2, _ = lse_lbfgs_adaptive(best_seq, remain * 0.5, T_start=20.0,
                                            T_min=0.1, n_stages=15,
                                            iters_per_stage=25, verbose=False)
            try_update(a_lse2, "lse_low_T")

        # From perturbed start
        remain = global_deadline - time.monotonic()
        if remain > 15:
            pert = best_seq + 0.03 * rng.normal(0, 1, len(best_seq))
            pert = project_simplex(np.maximum(pert, 0))
            a_lse3, _ = lse_lbfgs_adaptive(pert, remain * 0.6, T_start=50.0,
                                            T_min=0.2, n_stages=10,
                                            iters_per_stage=25, verbose=False)
            try_update(a_lse3, "lse_perturbed")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Randomized perturbation search (20% time) ===
    print(f"\n=== Phase 3: Perturbation search (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t3 = min(0.20 * (budget_s - 35), remain)
    if t3 > 30:
        a_pert, _ = randomized_perturb_search(best_seq, t3, n_perturbs=15,
                                               perturb_scale=0.03,
                                               opt_time_frac=0.6, verbose=True)
        try_update(a_pert, "perturbation")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t4_total = min(0.15 * (budget_s - 35), remain)
    if t4_total > 30:
        # Try a few lengths around n_base
        lengths_to_try = []
        for delta in [200, 400, -200, 600, -400]:
            n_try = n_base + delta
            if 400 <= n_try <= 5000:
                lengths_to_try.append(n_try)

        t_per_len = t4_total / max(1, len(lengths_to_try))
        for n_try in lengths_to_try:
            remain = global_deadline - time.monotonic()
            if remain < 20:
                break
            t_alloc = min(t_per_len, remain * 0.8)
            if t_alloc < 10:
                break
            print(f"  Trying n={n_try}, budget={t_alloc:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            if t_alloc > 30:
                a_try, _ = subgradient_bb(init_try, t_alloc * 0.5, K_peaks=5,
                                           restart_interval=1000, patience=300,
                                           verbose=False)
                try_update(a_try, f"len_{n_try}_sg")
            remain = global_deadline - time.monotonic()
            if remain > 15:
                a_try2, _ = lse_lbfgs_adaptive(a_try, remain * 0.5, T_start=40.0,
                                                T_min=0.3, n_stages=8,
                                                iters_per_stage=20, verbose=False)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Final combined polish (5% time) ===
    print(f"\n=== Phase 5: Final polish (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    if remain > 15:
        # Alternating subgradient + LSE
        n_remaining_steps = 3
        step_time = remain / n_remaining_steps
        for step in range(n_remaining_steps):
            remain = global_deadline - time.monotonic()
            if remain < 5:
                break
            if step % 2 == 0:
                a_f, _ = subgradient_bb(best_seq, step_time * 0.8, K_peaks=3,
                                         restart_interval=500, patience=200,
                                         verbose=False)
                try_update(a_f, f"final_sg_{step}")
            else:
                a_f, _ = lse_lbfgs_adaptive(best_seq, step_time * 0.8, T_start=15.0,
                                             T_min=0.1, n_stages=8,
                                             iters_per_stage=20, verbose=False)
                try_update(a_f, f"final_lse_{step}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
