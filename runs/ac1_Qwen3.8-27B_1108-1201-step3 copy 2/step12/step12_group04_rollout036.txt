import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time
from scipy.optimize import minimize

def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T @ b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    n_avg = 0
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss
        y = y_new
        a = a_new
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 40 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.10f}")
    return best_a, best_score

def compute_pnorm_grad(a, p):
    """
    Compute gradient of (sum c_k^p)^(1/p) w.r.t. a.
    c = convolve(a,a), gradient[i] = sum_k c_k^(p-1) * dc_k/da_i
    dc_k/da_i = a_{k-i} + a_{i-k} (with bounds checking)
    Using FFT: grad = 2 * (c^(p-1) * a_rev)[n-1 : n-1+n]
    """
    n = len(a)
    c = fast_autocorr(a)
    c_clipped = np.maximum(c, 1e-15)
    w = c_clipped ** (p - 1)
    # Normalize weights
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum  # This gives direction; scaling doesn't matter for optimization
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad

def pnorm_optimize(a0, time_limit, p_start=50, p_end=200, n_stages=12):
    """p-norm smoothed optimization with L-BFGS."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    ps = np.geomspace(p_start, p_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, p_val in enumerate(ps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        iters = max(15, min(80, int(30 + 50 * (si / max(1, n_stages - 1)))))
        
        def func(z_in):
            a_s = np.exp(z_in - z_in.max())
            s_a = a_s.sum()
            a_s = a_s / s_a
            c = fast_autocorr(a_s)
            c_clip = np.maximum(c, 1e-15)
            return np.sum(c_clip ** p_val) ** (1.0 / p_val)
        
        def jac(z_in):
            a_s = np.exp(z_in - z_in.max())
            s_a = a_s.sum()
            a_s = a_s / s_a
            g_a = compute_pnorm_grad(a_s, p_val)
            # Transform to z-space: grad_z = a * (grad_a - dot(grad_a, a))
            S = np.dot(g_a, a_s)
            g_z = a_s * (g_a - S)
            return g_z
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-25, 'gtol': 1e-18,
                                   'maxfun': iters * 200, 'disp': False})
            z = res.x
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()
            score = 2.0 * n * fast_autocorr(a_s).max()
            if score < best_score:
                best_score = score
                best_a = a_s.copy()
        except Exception:
            pass
        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PNORM stage={si} p={p_val:.0f} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def greedy_polish(a0, time_limit, seed=42):

    n = len(a0)
    rng = np.random.default_rng(seed)

        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            # Find the largest contributor
            top_c = np.argsort(contrib)[::-1][:5]

                src = idx[tc]
                if a[src] < 1e-14:

                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:

                        best_score = score

                        improved = True
                        stall = 0

            if improved:
                break
        
        if not improved:

                i = rng.integers(n)

                if i == j or a[i] < 1e-12:

                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:

                a_trial = a_trial / ss

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n + 1)
    old_vals = np.concatenate([f_norm, [0]])
    new_x = np.linspace(0, 1, new_n + 1)
    new_vals = np.concatenate([np.interp(new_x[:-1], old_x, old_vals), [0]])
    # Actually let's do it simpler
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T @ b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)

        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:

def compute_pnorm_grad(a, p):
    """
    Compute gradient of (sum c_k^p)^(1/p) w.r.t. a using FFT.
    c = convolve(a,a); grad[i] proportional to sum_k c_k^(p-1) * (a[k-i] + a[i-k])
    Simplified as: grad = 2 * (w * a_rev)[n-1 : n-1+n] where w_k = c_k^(p-1)
    """
    n = len(a)
    c = fast_autocorr(a)
    c_clip = np.maximum(c, 1e-15)
    w = c_clip ** (p - 1)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size = 1 << (conv_len).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def pnorm_optimize(a0, time_limit, p_start=30, p_end=200, n_stages=12):
    """p-norm smoothed optimization using softmax parameterization and L-BFGS."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    ps = np.geomspace(p_start, p_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        p_val = ps[si]
        iters = max(15, min(80, int(30 + 50 * (si / max(1, n_stages - 1)))))

        def func_at_stage(z_in):
            a_s = np.exp(z_in - z_in.max())
            s_a = a_s.sum()
            a_s = a_s / s_a
            c = fast_autocorr(a_s)
            c_clip = np.maximum(c, 1e-15)
            return np.sum(c_clip ** p_val) ** (1.0 / p_val)

        def jac_at_stage(z_in):
            a_s = np.exp(z_in - z_in.max())
            s_a = a_s.sum()
            a_s = a_s / s_a
            g_a = compute_pnorm_grad(a_s, p_val)
            S = np.dot(g_a, a_s)
            g_z = a_s * (g_a - S)
            return g_z

        try:
            res = minimize(fun=func_at_stage, x0=z, jac=jac_at_stage,
                           method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-25,
                                    'gtol': 1e-18, 'maxfun': iters * 200,
                                    'disp': False})
            z = res.x
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()
            score = 2.0 * n * fast_autocorr(a_s).max()
            if score < best_score:
                best_score = score
                best_a = a_s.copy()
        except Exception:
            pass
        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PNORM stage={si} p={p_val:.0f} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def greedy_polish(a0, time_limit, seed=42):
    """Greedy coordinate descent with adaptive targeting using peak analysis."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    fracs = [0.001, 0.002, 0.005, 0.01, 0.02, 0.04]

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        top_peaks = np.argsort(c)[::-1][:12]
        improved = False

        for pk_idx in range(len(top_peaks)):
            k = top_peaks[pk_idx]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            top_c = np.argsort(contrib)[::-1][:5]
            for tc in top_c:
                src = idx[tc]
                if a[src] < 1e-14:
                    continue
                # Find best target: minimize total contribution to top peaks
                best_dst = -1
                best_dst_score = float('inf')
                n_candidates = min(30, max(5, n // 40))
                for _cand in range(n_candidates):
                    dst = rng.integers(n)
                    if dst == src:
                        continue
                    score = 0.0
                    for k2 in top_peaks[:6]:
                        js2 = max(0, k2 - n + 1)
                        je2 = min(n - 1, k2)
                        if js2 <= dst <= je2:
                            partner = k2 - dst
                            if 0 <= partner < n:
                                score += a[partner]
                    score += 0.005 * a[dst]
                    if score < best_dst_score:
                        best_dst_score = score
                        best_dst = dst
                if best_dst < 0:
                    continue
                for frac in fracs:
                    amt = a[src] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(15):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.001, 0.04)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    break
            else:
                stall += 1
                if stall > 300:
                    break
        if stall > 0 and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      GREEDY stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_optimize(a0, time_limit, T_start=80.0, T_end=0.01, n_stages=15):
    """LogSumExp-smoothed optimization (different from previous: more stages, finer schedule)."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        T_val = temps[si]
        iters = max(10, min(60, int(20 + 40 * (si / max(1, n_stages - 1)))))

        def func_lse(z_in):
            a_s = softmax_fn(z_in)
            c = fast_autocorr(a_s)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            return (max_Tc + np.log(exp_Tc.sum())) / T_val

        def jac_lse(z_in):
            a_s = softmax_fn(z_in)
            c = fast_autocorr(a_s)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            a_rev = a_s[::-1].copy()
            conv_len = n + (2 * n - 1) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            if conv_len > fft_size:
                fft_size = 1 << (conv_len).bit_length()
            W = rfft(w, fft_size)
            A_rev = rfft(a_rev, fft_size)
            conv = irfft(W * A_rev, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]
            S = np.dot(grad_a, a_s)
            grad_z = a_s * (grad_a - S)
            return grad_z

        try:
            res = minimize(fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-25,
                                    'gtol': 1e-18, 'maxfun': iters * 200,
                                    'disp': False})
            z = res.x
            a_s = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a_s).max()
            if score < best_score:
                best_score = score
                best_a = a_s.copy()
        except Exception:
            pass
        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si} T={T_val:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score
