"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping (MIA-style) for good initialization
2. Hedge algorithm with FFT-accelerated QP best-response (fast iterations)
3. LP-free mirror descent with top-K LSE for rapid convergence
4. Aggressive multi-peak shaving with binary-search step sizes
5. Multi-length exploration and final Adam polish

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def proj_simplex(v):
    """Project v onto the probability simplex {a: sum(a)=1, a>=0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fast_matvec_toeplitz(q, a):
    """
    Compute M_q @ a where M_q[i,j] = q[min(i+j, 2n-2)].
    Uses FFT-based convolution: (M_q @ a)[i] = conv(q, a_rev)[i + n - 1]
    """
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (len(q) + n - 1 - 1).bit_length()
    Q = rfft(q, fft_size)
    R = rfft(a_rev, fft_size)
    conv = irfft(Q * R, fft_size)
    # Extract elements at positions n-1, n, ..., 2n-2
    return conv[n - 1: 2 * n - 1]


def qp_best_response(q, n, n_iters=40, eta=None):
    """
    Fast QP solve: min a^T M_q a s.t. sum(a)=1, a>=0
    where M_q[i,j] = q[min(i+j, 2n-2)].
    Uses projected gradient with FFT-accelerated matvec.
    """
    a = np.ones(n) / n
    if eta is None:
        # Estimate Lipschitz constant ~ 2 * max_row_sum
        max_row = 0.0
        for i in range(n):
            s = np.sum(q[min(i, 2 * n - 2):min(i + n, 2 * n - 1) + 1] if i + n <= 2 * n - 1 else q[i:])
            # Simpler: just use a safe default
            max_row = max(max_row, s)
        eta = 0.9 / max(2.0 * max(max_row, 0.1), 0.5)
        eta = min(eta, 2.0)

    for _ in range(n_iters):
        grad = 2.0 * fast_matvec_toeplitz(q, a)
        a_new = proj_simplex(a - eta * grad)
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-10:
            a = a_new
            break
        a = a_new
    return a


def spectral_init(n, n_iters=50, seed=0):
    """
    MIA-style spectral initialization: flatten power spectrum while maintaining non-negativity.
    Produces a sequence with approximately flat autocorrelation sidelobes.
    """
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n))
    for _ in range(n_iters):
        # Forward FFT
        A = np.fft.fft(a)
        # Flatten magnitude (keep DC, reduce sidelobes)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: make magnitude approximately constant (but keep DC high enough for non-neg)
        target_mag = np.ones(n)
        target_mag[0] = mag[0]  # Keep DC to maintain sum
        # Blend: partially flatten
        blend = 0.8
        new_mag = (1 - blend) * mag + blend * target_mag
        new_mag = np.maximum(new_mag, 1e-10)
        # Inverse FFT
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new))
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    return a


def mirror_descent_topk(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=30, topk=15, lr_scale=1.0):
    """
    Mirror descent (multiplicative weights) with top-K LSE objective.
    Fast: one FFT per iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()

    best_a = a.copy()
    best_c = fast_autocorr(a).max()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Learning rate: smaller for larger n
        lr = lr_scale * 0.02 * min(1.0, 500.0 / n)
        # More iterations at lower temperature
        max_it = max(20, int(min(remaining, 60.0) * 5 * (T_start / T) ** 0.3))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break
            c = fast_autocorr(a)
            cmax = c.max()
            # Find top-K peaks
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]

            # LSE weight
            Tc = T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()

            # Gradient: grad[i] = 2 * sum_k w_k * a[k-i] (valid range)
            grad = np.zeros(n)
            for idx_ in range(k_use):
                K = top_idx[idx_]
                wK = w[idx_]
                js = max(0, K - n + 1)
                je = min(n - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad[idx_range] += 2.0 * wK * a[K - idx_range]

            # Mirror descent update in log-space
            # a_new[i] = a[i] * exp(-lr * grad[i])
            z_shift = -lr * grad
            # Renormalize to maintain sum(a) = 1
            log_a = np.log(np.maximum(a, 1e-30))
            log_a += z_shift
            log_a -= log_a.max()
            exp_la = np.exp(log_a)
            a_new = exp_la / exp_la.sum()
            a = a_new

            if (it + 1) % 20 == 0:
                c_now = fast_autocorr(a).max()
                if c_now < best_c:
                    best_c = c_now
                    best_a = a.copy()

        # End of stage
        c_stage = fast_autocorr(a).max()
        if c_stage < best_c:
            best_c = c_stage
            best_a = a.copy()

    return best_a


def hedge_qp_optimize(a0, time_limit, n_qp_iters=30, n_averages=5):
    """
    Hedge algorithm with FFT-accelerated QP best-response.
    Row player: solves QP min_a sum_k q[k]*(a*a)[k] via projected gradient.
    Column player: argmax of autocorrelation.
    Output: running average of iterates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Column player's empirical distribution
    q = np.ones(2 * n - 1) / (2 * n - 1)

    best_a = a.copy()
    best_c = fast_autocorr(a).max()

    # Running averages at different window sizes
    sums = [np.zeros(n) for _ in range(n_averages)]
    counts = 0
    t0 = time.monotonic()

    # Circular buffer for short window
    buf_size = 50
    buf = [a.copy() for _ in range(buf_size)]
    buf_idx = 0
    buf_sum = np.sum(buf, axis=0)

    c_full = fast_autocorr(a)
    k_star = int(np.argmax(c_full))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit - 3:
            break

        # Column player best response: argmax of autocorrelation of running average
        # (Using the current 'a' as a proxy for speed; refine periodically)
        # Actually, use the short-window average for column player's response
        avg_short = buf_sum / min(counts, buf_size)
        c_avg = fast_autocorr(avg_short)
        k_star = int(np.argmax(c_avg))

        # Update q (column player's mixing)
        eta_q = 1.0 / (t + 1.0)
        q = (1 - eta_q) * q + eta_q * np.zeros(2 * n - 1)
        q[k_star] += eta_q

        # Row player best response: solve QP min_a sum_k q[k] c_a[k]
        a_new = qp_best_response(q, n, n_iters=n_qp_iters)

        # Update running averages
        counts += 1
        for i in range(n_averages):
            sums[i] += a_new

        # Update circular buffer
        buf_sum -= buf[buf_idx]
        buf[buf_idx] = a_new.copy()
        buf_sum += a_new
        buf_idx = (buf_idx + 1) % buf_size

        a = a_new

        # Check best (every 50 iterations to save time)
        if counts % 50 == 0 or t < 10:
            c_now = fast_autocorr(a).max()
            if c_now < best_c:
                best_c = c_now
                best_a = a.copy()

            # Check running averages
            for i in range(n_averages):
                avg_i = sums[i] / counts
                c_avg_i = fast_autocorr(avg_i).max()
                if c_avg_i < best_c:
                    best_c = c_avg_i
                    best_a = avg_i.copy()

            # Check short window average
            if counts >= buf_size:
                avg_s = buf_sum / buf_size
                c_s = fast_autocorr(avg_s).max()
                if c_s < best_c:
                    best_c = c_s
                    best_a = avg_s.copy()

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Hedge t={t} ({elapsed:.0f}s): best_maxc={best_c:.8f}")

    return best_a


def adam_polish(a0, time_limit, T_start=5.0, T_end=0.1, n_stages=15, topk=8, lr_base=0.003):
    """
    Adam-optimized top-K LSE for fine-tuning.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_c = fast_autocorr(a_init).max()

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        lr = lr_base * min(1.0, 800.0 / n) * (0.85 ** si)
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(30, int(min(remaining, 80.0) * 4))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            a = softmax_fn(z)
            c = fast_autocorr(a)
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]

            Tc = T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()

            grad_a = np.zeros(n)
            for idx_ in range(k_use):
                K = top_idx[idx_]
                wK = w[idx_]
                js = max(0, K - n + 1)
                je = min(n - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                c_check = fast_autocorr(a_check).max()
                if c_check < best_c:
                    best_c = c_check
                    best_a = a_check.copy()

        a_stage = softmax_fn(z)
        c_stage = fast_autocorr(a_stage).max()
        if c_stage < best_c:
            best_c = c_stage
            best_a = a_stage.copy()

    return best_a


def peak_shave_advanced(a0, time_limit, n_peaks=10, seed=42):

    Advanced peak shaving with multi-peak awareness and binary-search step sizes.
    """

        top_peaks = np.argsort(c)[::-1][:k_use]

        # Identify danger zone: positions that contribute to any top peak
        danger_weight = np.zeros(n)
        for K in top_peaks:
            js = max(0, K - n + 1)
            je = min(n - 1, K)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                contrib = a[idx_range] * a[K - idx_range]
                danger_weight[idx_range] += contrib

        # Find source: position with highest danger weight and non-trivial value
        candidates = np.where(danger_weight > 0.01 * danger_weight.max())[0]
        candidates = candidates[a[candidates] > 1e-8]
        if len(candidates) == 0:
            candidates = np.where(a > 1e-8)[0]
        if len(candidates) == 0:
            break

        src = candidates[np.argmax(danger_weight[candidates])]

        # Find safe destinations: low danger weight
        safe_score = 1.0 / (danger_weight + 1e-10)
        # Also prefer positions with low current value (to avoid creating new peaks)
        safe_score = safe_score / (a + 0.1 * a.max() + 1e-10)
        safe_score[src] = -1
        safe_candidates = np.where(safe_score > 0)[0]
        if len(safe_candidates) == 0:
            safe_candidates = np.where(np.arange(n) != src)[0]

        # Pick a few candidates
        top_safe = safe_candidates[np.argsort(safe_score[safe_candidates])[::-1][:max(5, len(safe_candidates) // 10)]]
        if len(top_safe) == 0:
            break

        improved = False
        for dst in top_safe[:8]:
            # Binary search for maximum step that reduces the max
            lo, hi = 0.0, a[src] * 0.15
            best_step = 0.0
            best_trial_c = cmax

            for _ in range(8):
                mid = (lo + hi) / 2.0
                a_trial = a.copy()
                a_trial[src] -= mid
                a_trial[dst] += mid
                if a_trial[src] < 0:
                    hi = mid
                    continue
                a_trial = a_trial / a_trial.sum()
                c_trial_max = fast_autocorr(a_trial).max()
                if c_trial_max < best_trial_c:
                    best_trial_c = c_trial_max
                    best_step = mid
                    lo = mid
                else:
                    hi = mid

            if best_step > 1e-10 and best_trial_c < best_c - 1e-12:
                a = a.copy()
                a[src] -= best_step
                a[dst] += best_step
                a = a / a.sum()
                best_c = best_trial_c
                best_a = a.copy()
                improved = True
                stall = 0
                break

            # Random perturbation to escape

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def structured_init(n, seed=0):

    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 4
    elif choice == 2:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    elif choice == 4:
        a = (1 - np.abs(2 * x - 1)) ** 1.5
    else:
        a = np.ones(n) / n
        # Add structured noise
        a += 0.1 * np.sin(2 * np.pi * 5 * x)
        a += 0.05 * np.sin(2 * np.pi * 13 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, seed)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 0: Spectral shaping for better init ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t0_phase = min(remaining * 0.02, 20.0)
    if t0_phase > 5:
        print(f"=== Phase 0: Spectral init n={n_base} ({t0_phase:.0f}s) ===")
        a_spec = spectral_init(n_base, n_iters=80, seed=seed)
        try_update(a_spec, "spectral_init")

    # === Phase 1: Fast mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 1: Mirror descent n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.18, 250.0)
    if t1 > 20:
        a_out = mirror_descent_topk(best_seq, t1, T_start=40.0, T_end=0.3, n_stages=25, topk=12)
        try_update(a_out, "mirror_n1")

    # === Phase 2: Hedge with QP ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: Hedge+QP n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_out = hedge_qp_optimize(best_seq, t2, n_qp_iters=25, n_averages=4)
        try_update(a_out, "hedge_n1")

    # === Phase 3: Adam polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: Adam polish n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.12, 120.0)
    if t3 > 15:
        a_out = adam_polish(best_seq, t3, T_start=5.0, T_end=0.1, n_stages=12, topk=8)
        try_update(a_out, "adam_n1")

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Peak shaving n={n_base} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 15:
        a_out = peak_shave_advanced(best_seq, t4, n_peaks=10, seed=seed + 1)
        try_update(a_out, "peak_shave_1")

    # === Phase 5: Longer length - mirror descent ===
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: Mirror descent n={n_long} ({remaining:.0f}s left) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        init_long = interpolate_seq(best_seq, n_long)
        a_out = mirror_descent_topk(init_long, t5, T_start=30.0, T_end=0.5, n_stages=20, topk=10)
        try_update(a_out, f"mirror_n{n_long}")

    # === Phase 6: Longer length - Hedge ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_long2 = n_base + 400
    print(f"=== Phase 6: Hedge n={n_long2} ({remaining:.0f}s left) ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 20:
        init_long2 = interpolate_seq(best_seq, n_long2)
        a_out = hedge_qp_optimize(init_long2, t6, n_qp_iters=20, n_averages=3)
        try_update(a_out, f"hedge_n{n_long2}")

    # === Phase 7: Final Adam polish on best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 7: Final Adam polish n={len(best_seq)} ({remaining:.0f}s left) ===")
    t7 = min(remaining * 0.12, 100.0)
    if t7 > 10:
        a_out = adam_polish(best_seq, t7, T_start=2.0, T_end=0.05, n_stages=10, topk=6)
        try_update(a_out, "final_adam")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 8: Final peak shave n={len(best_seq)} ({remaining:.0f}s left) ===")
    t8 = min(remaining * 0.25, 150.0)
    if t8 > 10:
        a_out = peak_shave_advanced(best_seq, t8, n_peaks=15, seed=seed + 100)
        try_update(a_out, "final_peak_shave")

    # === Phase 9: One more round of mirror descent on best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t9 = min(remaining * 0.4, 100.0)
        if t9 > 10:
            print(f"=== Phase 9: Final mirror n={len(best_seq)} ({remaining:.0f}s left) ===")
            a_out = mirror_descent_topk(best_seq, t9, T_start=3.0, T_end=0.1, n_stages=10, topk=8, lr_scale=0.5)
            try_update(a_out, "final_mirror")

    # Final peak shaving with whatever time remains
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t10 = min(remaining * 0.7, 60.0)
        if t10 > 5:
            a_out = peak_shave_advanced(best_seq, t10, n_peaks=12, seed=seed + 200)
            try_update(a_out, "last_shave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
