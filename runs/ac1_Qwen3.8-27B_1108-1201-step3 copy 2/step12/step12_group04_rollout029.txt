def evaluate_safe(a):

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad_z(z, T, n):
    """LogSumExp objective and gradient in z-space."""
    a = softmax_fn(z)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    # Gradient w.r.t. a
    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1]
    conv_len = 2 * n - 1 + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv = irfft(W * Ar, fft_size)
    grad_a = 2.0 * conv[n - 1: 2 * n - 1]
    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def lse_func_z(z, T, n):
    """Wrapper for minimize (objective only)."""
    val, _ = lse_obj_grad_z(z, T, n)
    return val


def lse_jac_z(z, T, n):
    """Wrapper for minimize (gradient only)."""
    _, g = lse_obj_grad_z(z, T, n)
    return g


def spectral_flatten(a, alpha=0.5, n_iters=10):
    """Flatten the power spectrum of a to reduce peak autocorrelation."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a_cur = a.copy()
    for _ in range(n_iters):
        A = rfft(a_cur, N)
        P = np.abs(A) ** 2
        P_flat = P.mean() * np.ones(N // 2 + 1)
        # Mix: keep some of the original structure
        P_new = (1 - alpha) * P + alpha * P_flat
        P_new = np.maximum(P_new, 1e-20)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_cur = a_new / s
    return a_cur


def build_conv_matrix(a_fixed):
    """Build convolution matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed,b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def mw_best_response(a_fixed, n_iters=30, eta=None):
    """
    Approximate best response using multiplicative weights.
    min max_k (a_fixed * b)_k s.t. b >= 0, sum(b) = 1.
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    c_curr = fast_autocorr(a_fixed)  # This is a*a, not a*b
    # We need conv(a_fixed, b) for each iteration
    # Use FFT: conv(a_fixed, b) = ifft(fft(a_fixed) * fft(b))
    N = 1 << (2 * n - 1).bit_length()
    A = rfft(a_fixed, N)
    
    if eta is None:
        eta = 1.0
    
    for _ in range(n_iters):
        B = rfft(b, N)
        conv = irfft(A * B, N)[:2 * n - 1]
        # Multiplicative weights update
        log_b = np.log(b + 1e-30)
        log_b -= eta * conv / conv.max() if conv.max() > 0 else 0
        # Actually, for min max, we should increase weight on b where conv is small
        # MW for min max_k f_k(b): b_k *= exp(eta * min_j f_j(b))... 
        # Simpler: b *= exp(-eta * conv / max(conv)) then normalize
        # This pushes mass away from positions that contribute to high conv
        log_b = np.log(b + 1e-30) - eta * conv / (conv.max() + 1e-30)
        b = np.exp(log_b - log_b.max())
        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-15:
            b = np.ones(n) / n
        else:
            b = b / s
    return b


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            b_star = mw_best_response(y, n_iters=20)
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        y_new = y_new / (ss if ss > 1e-15 else 1)
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            a_new = a_new / (ss if ss > 1e-15 else 1)
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def dual_qp_optimize(a0, time_limit, n_stages=10):
    """
    Dual QP optimization: alternates between solving QP min a^T Q(λ) a
    and updating λ based on the current autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # For the QP: min a^T Q(λ) a s.t. a >= 0, sum(a) = 1
    # Q(λ)_{ij} = λ_{i+j}
    # Gradient: 2 * (Q @ a)_i = 2 * sum_j λ_{i+j} a_j
    # Compute Q@a via FFT: corr of a with λ
    N = 1 << (3 * n - 2).bit_length()  # Need size >= (2n-1) + (n-1) = 3n-2
    
    lambda_ = np.ones(2 * n - 1) / (2 * n - 1)  # Uniform init
    
    # Temperature schedule for λ update
    temp_schedule = np.geomspace(1.0, 0.05, n_stages)
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        T_lambda = temp_schedule[stage]
        
        # Solve QP using L-BFGS with softmax parameterization
        z = np.log(a + 1e-30)
        
        def qp_obj(z_in):
            a_in = softmax_fn(z_in)
            # Compute a^T Q(λ) a = sum_k λ_k (a*a)_k
            c = fast_autocorr(a_in)
            return np.dot(lambda_, c)
        
        def qp_grad(z_in):
            a_in = softmax_fn(z_in)
            # grad_a = 2 * (Q @ a) where (Q@a)_i = sum_j λ_{i+j} a_j
            # Use FFT: reverse a, convolve with lambda
            a_rev = a_in[::-1]
            fft_sz = 1 << (n + 2 * n - 2 - 1).bit_length()
            Ar = rfft(a_rev, fft_sz)
            L = rfft(lambda_, fft_sz)
            conv = irfft(Ar * L, fft_sz)
            grad_a = 2.0 * conv[n - 1: 2 * n - 1]
            # Transform to z-space
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)
        
        # Run L-BFGS
        iters = max(20, min(100, int((time_limit - (time.monotonic() - t0)) * 5)))
        try:
            res = minimize(qp_obj, z, jac=qp_grad, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14, 'disp': False})
            a_new = softmax_fn(res.x)
        except Exception:
            a_new = a.copy()
        
        # Update λ: emphasize peaks
        c_new = fast_autocorr(a_new)
        c_norm = c_new / (c_new.max() + 1e-30)
        lambda_new = np.exp(T_lambda * c_norm)
        lambda_new = lambda_new / lambda_new.sum()
        
        # Blend a
        a = a_new
        lambda_ = lambda_new
        
        score = 2.0 * n * c_new.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"      DQP stage={stage} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

- The "column player" plays b (the best response).

- We directly minimize the "diagonal payoff" using a dual variable λ.
- λ represents the "distribution over rows" (which k is the worst).

- Updating λ ∝ exp(η * c) is the multiplicative weights update for the "column player" in the dual game.

So the two approaches are related but different:
- Nesterov LP: accelerates the "row player's" best response.
- Dual QP: accelerates the "column player's" (λ's) update.

They should be complementary! Using both should give better results than either alone.
