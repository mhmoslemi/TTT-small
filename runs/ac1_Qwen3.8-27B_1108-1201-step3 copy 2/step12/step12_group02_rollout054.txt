    z_new = res.x
    z_s = z_new - z_new.max()
    a_new = np.exp(z_s)
    s = a_new.sum()
    if s < 1e-15:
        return a_current.copy()
    return a_new / s


def full_dist_subgrad(a0, time_limit, T_start, T_end, eta0, decay_rate,
                      perturbation_interval, noise_scale, seed, verbose):
    """Subgradient descent with full-peak-distribution softmax weighting."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / 5000.0)
        T = T_start * (T_end / T_start) ** frac
        eta = eta0 / (1.0 + it / 300.0) ** decay_rate

        # Compute full-distribution gradient
        b = fast_autocorr(a)
        mean_b = b.sum() / len(b)
        excess = np.maximum(b - mean_b, 0)
        max_ex = excess.max()
        if max_ex < 1e-15:
            break
        w = np.exp(excess / (T * max_ex) - 1.0)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.ones(len(b)) / len(b)
        else:
            w = w / wsum

        grad = 2.0 * cross_corr_via_fft(w, a)

        # Update
        a = a - eta * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = 0.9 * (np.ones(n) / n) + 0.1 * rng.random(n)
            a = a / a.sum()
        else:
            a = a / s

        # Periodic score check
        if it % 20 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Perturbation
        if it % perturbation_interval == 0:
            noise = noise_scale * rng.standard_normal(n)
            a = np.maximum(a + noise, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def spectral_perturb(a, freq, amp, n_total):
    """Add a sinusoidal perturbation in the time domain."""
    n = len(a)
    i = np.arange(n)
    pert = amp * np.cos(2.0 * np.pi * freq * i / n)
    a_new = a + pert
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def lse_anneal(a0, time_limit, n_stages, T_start, T_end, seed, verbose):
    """Multi-stage LSE annealing with adaptive iterations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        # Allocate more iterations at low T (fine-tuning)
        stage_frac = si / max(1, n_stages - 1)
        base_iters = 30 + int(70 * stage_frac)
        time_per_stage = remaining / max(1, n_stages - si)
        max_iters = max(20, min(300, int(base_iters * time_per_stage / (time_limit / n_stages))))

        try:
            a_new = lse_lbfgs_stage(a, float(stage_T), max_iters)
            sc = fast_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                # Try from best instead of current
                pass
        except Exception:
            pass

        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si+1}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, seed, verbose):
    """Coordinate descent: move mass from peak contributors to low areas."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(8, n)]
        improved = False

        for k_idx in top_k:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            if contrib.sum() < 1e-15:
                continue

            src = int(idx[int(np.argmax(contrib))])

            # Find target: positions with lowest a, away from peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:4]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]
            sorted_cands = candidates[np.argsort(a[candidates])[:max(3, len(candidates) // 5)]]
            if len(sorted_cands) == 0:
                continue
            dst = int(sorted_cands[int(rng.integers(len(sorted_cands)))])

            for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random swap
            if rng.integers(2) == 0:
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i1] * frac
                    a_trial = a.copy()
                    a_trial[i1] -= amt
                    a_trial[i2] += amt
                    if a_trial[i1] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 300:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"      PeakShave ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.ones(n) / n
    elif variant == 1:
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        xx = np.linspace(-1, 1, n)
        a = np.exp(-3 * xx ** 2)
    elif variant == 3:
        a = np.exp(-25 * (x - 0.2) ** 2) + np.exp(-25 * (x - 0.8) ** 2)
    elif variant == 4:
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        a = 1 + 0.3 * np.cos(4 * np.pi * x)
    elif variant == 6:
        xx = np.linspace(-1, 1, n)
        a = np.sinc(xx) ** 2
    elif variant == 7:
        a = 1 + 0.2 * np.cos(6 * np.pi * x) + 0.1 * np.cos(10 * np.pi * x)
    elif variant == 8:
        a = abs(x - 0.5) + 0.3
    elif variant == 9:
        a = 1 + 0.15 * np.cos(3 * 2 * np.pi * x) - 0.1 * np.cos(5 * 2 * np.pi * x)
    elif variant == 10:
        # Multi-scale Gaussian mixture
        a = np.zeros(n)
        for _ in range(6):
            c = rng.uniform(0.1, 0.9)
            w = rng.uniform(0.05, 0.2)
            amp = rng.uniform(0.5, 1.5)
            a += amp * np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0)
    else:
        # Concentrated edges with slight center
        a = np.exp(-8 * (x - 0) ** 2) + np.exp(-8 * (x - 1) ** 2) + 0.3

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Call official evaluate_sequence."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization with length extension.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1740.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    n_parent = len(best_seq) if best_seq is not None else 1600
    if best_seq is None:
        best_seq = diverse_init(n_parent, rng, variant=0)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"Budget: {remaining:.0f}s, starting n={n_parent}")

    # === Phase 1: Quick diverse subgradient at parent length ===
    print("\n=== Phase 1: Diverse SubGradient at n={} ===".format(n_parent))
    t_phase1 = min(remaining * 0.12, 120.0)
    if t_phase1 > 20:
        inits = [best_seq.copy(),
                 diverse_init(n_parent, rng, variant=4),
                 diverse_init(n_parent, rng, variant=7),
                 diverse_init(n_parent, rng, variant=9)]
        t_per = t_phase1 / len(inits)
        for ii, init in enumerate(inits):
            if time.monotonic() - t_global > time_budget - 100:
                break
            a_out, _ = full_dist_subgrad(init, t_per, T_start=0.1, T_end=0.005,
                                          eta0=0.03, decay_rate=0.6,
                                          perturbation_interval=2000,
                                          noise_scale=0.02, seed=100 + ii, verbose=False)
            try_update(a_out, f"subgrad_v{ii}")

    # Spectral perturbation exploration
    remaining = time_budget - (time.monotonic() - t_global)
    t_spec = min(remaining * 0.06, 60.0)
    if t_spec > 10 and best_seq is not None:
        a_cur = best_seq.copy()
        n = len(a_cur)
        t0_spec = time.monotonic()
        while time.monotonic() - t0_spec < t_spec:
            freq = rng.integers(1, n // 4)
            amp = rng.uniform(0.001, 0.005)
            a_trial = spectral_perturb(a_cur, freq, amp, n)
            sc = fast_score(a_trial)
            if sc < fast_score(a_cur):
                a_cur = a_trial
        try_update(a_cur, "spectral_perturb")

    # === Phase 2: Slow LSE annealing at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: LSE Annealing at n={} ===".format(n_parent))
    t_phase2 = min(remaining * 0.28, 400.0)
    if t_phase2 > 30 and best_seq is not None:
        a_out, _ = lse_anneal(best_seq, t_phase2, n_stages=25,
                              T_start=300.0, T_end=0.01, seed=7, verbose=True)
        try_update(a_out, f"lse_slow_n{n_parent}")

    # === Phase 3: More subgradient with lower T ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Fine SubGradient at n={} ===".format(n_parent))
    t_phase3 = min(remaining * 0.15, 180.0)
    if t_phase3 > 20 and best_seq is not None:
        a_out, _ = full_dist_subgrad(best_seq, t_phase3, T_start=0.02, T_end=0.001,
                                      eta0=0.008, decay_rate=0.7,
                                      perturbation_interval=5000,
                                      noise_scale=0.005, seed=200, verbose=True)
        try_update(a_out, f"subgrad_fine_n{n_parent}")

    # === Phase 4: Length extension ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_ext1 = int(n_parent * 1.3)
    n_ext2 = int(n_parent * 1.5)
    n_ext3 = int(n_parent * 1.8)
    lengths = [n_ext1, n_ext2, n_ext3]

    for li, n_ext in enumerate(lengths):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("\n=== Phase 4.{}: Length extension to n={} ===".format(li + 1, n_ext))

        init_ext = interpolate_to_length(best_seq, n_ext)

        # LSE at new length
        t_lse = min(remaining * 0.35, 250.0)
        if t_lse > 30:
            a_out, _ = lse_anneal(init_ext, t_lse, n_stages=20,
                                  T_start=100.0, T_end=0.05, seed=300 + li, verbose=True)
            try_update(a_out, f"lse_n{n_ext}")

        # Subgradient at new length
        remaining = time_budget - (time.monotonic() - t_global)
        t_sg = min(remaining * 0.2, 100.0)
        if t_sg > 20 and best_seq is not None:
            a_out, _ = full_dist_subgrad(best_seq, t_sg, T_start=0.05, T_end=0.003,
                                          eta0=0.02, decay_rate=0.6,
                                          perturbation_interval=3000,
                                          noise_scale=0.01, seed=400 + li, verbose=False)
            try_update(a_out, f"subgrad_n{n_ext}")

    # === Phase 5: Final LSE at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print("\n=== Phase 5: Final LSE at n={} ===".format(n_final))
    t_phase5 = min(remaining * 0.2, 200.0)
    if t_phase5 > 20:
        a_out, _ = lse_anneal(best_seq, t_phase5, n_stages=15,
                              T_start=50.0, T_end=0.005, seed=500, verbose=True)
        try_update(a_out, f"lse_final_n{n_final}")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 6: Peak Shaving ===")
    t_phase6 = min(remaining * 0.25, 120.0)
    if t_phase6 > 10:
        a_out, _ = peak_shave(best_seq, t_phase6, seed=600, verbose=True)
        try_update(a_out, f"peakshave_n{n_final}")

    # === Phase 7: Final micro-LSE ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Phase 7: Micro LSE ===")
        t_phase7 = min(remaining * 0.5, 60.0)
        if t_phase7 > 10:
            a_out, _ = lse_anneal(best_seq, t_phase7, n_stages=8,
                                  T_start=10.0, T_end=0.001, seed=700, verbose=False)
            try_update(a_out, "micro_lse")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
