def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def score_and_peakinfo(a):
    """Return score and peak information."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    peak_idx = int(np.argmax(b))
    return 2.0 * n * max_b, max_b, peak_idx, b


def evaluate_safe(a):

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, M_target=None, solver=None):
    """
    Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0.
    If M_target is given, also try: max sum(b) s.t. T(a)@b <= M_target, b>=0.
    Returns (b_val, M_value, success_flag).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)

    if M_target is not None:
        # Constraint tightening: max sum(b) s.t. T(b) <= M_target, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        T = build_conv_matrix(a_fixed)
        constraints = [T @ b_var <= M_target]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    else:
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = build_conv_matrix(a_fixed)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, False

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, False

    if M_target is not None:
        b_val = b_val / s
        # Compute actual max
        T = build_conv_matrix(a_fixed)
        M_actual = float((T @ b_val).max())
        return b_val, M_actual, True
    else:
        b_val = b_val / s
        M_val = float(b_var.value.max()) if hasattr(b_var, 'value') else float(M_val)
        try:
            T = build_conv_matrix(a_fixed)
            M_val = float((T @ b_val).max())
        except Exception:
            M_val = 1.0
        return b_val, M_val, True


def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Adaptive restart mechanism
    restart_threshold = 5000
    stale_iters = 0
    last_improvement = best_score
    improve_count = 0

    # Nesterov state
    y = a.copy()
    it_offset = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp, success = solve_lp_best_response(y, solver=solver)
        if b_new is None:
            stale_iters += 1
            if stale_iters > 100:
                # Restart
                a = best_a.copy()
                y = a.copy()
                it_offset = 0
                stale_iters = 0
            continue

        ephem = it + it_offset
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * ephem * ephem)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stale_iters = 0
            improve_count += 1
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stale_iters = 0
            improve_count += 1
        else:
            stale_iters += 1

        # Adaptive restart: if stuck for too long, reset from best
        if stale_iters > restart_threshold:
            a = best_a.copy()
            y = a.copy()
            it_offset = 0
            stale_iters = 0
            # Small perturbation to escape
            if improve_count > 0 and rng_restart is not None:
                pert = 0.01 * np.ones(n)
                a = a + pert * (rng_restart.random(n) - 0.5)
                a = np.maximum(a, 0)
                a = a / a.sum()
                y = a.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f} stale={stale_iters}")

    return best_a, best_score


rng_restart = None


def anti_diagonal_repulsion(a, beta, n_peaks=3):
    """
    Goosens-inspired peak reduction.
    For each top peak lag k, the sensitivity of (a*a)[k] w.r.t. a_j is a_{k-j}.
    We reduce a_j proportionally to its contribution to the peak.
    """
    n = len(a)
    b = fast_convolve(a)
    # Find top peaks
    top_idx = np.argsort(b)[::-1][:n_peaks]
    top_idx = top_idx[top_idx > 0]  # Exclude lag 0 (a[0]^2) if possible
    if len(top_idx) == 0:
        top_idx = np.array([n - 1])

    # Compute "peak contribution" vector
    contribution = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        # Sensitivity: how much each a_j contributes to peak at lag k
        contrib_k = a[k - idx]
        contribution[idx] += contrib_k * (b[k] / b.max())

    # Apply repulsion: reduce a where it contributes to peaks
    weight = np.exp(-beta * contribution / (contribution.max() + 1e-15))
    a_new = a * weight
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def constraint_tightening_lp(a, solver=None, n_attempts=5):
    """
    Try to find b with max(a*b) < max(a*a) using constraint tightening.
    Returns improved a or None.
    """
    n = len(a)
    b_current = fast_convolve(a)
    current_max = b_current.max()

    for attempt in range(n_attempts):
        # Try to push below current max by a small margin
        margin = 0.001 * (attempt + 1)
        M_target = current_max * (1.0 - margin)

        b_new, M_actual, success = solve_lp_best_response(a, M_target=M_target, solver=solver)
        if not success or b_new is None:
            continue

        # Check if this b actually has lower peak with itself
        sc_b = score_normalized(b_new)
        if sc_b < score_normalized(a):
            # Blend
            for t in [0.3, 0.5, 0.7]:
                a_blend = (1 - t) * a + t * b_new
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / a_blend.sum()
                sc_blend = score_normalized(a_blend)
                if sc_blend < score_normalized(a):
                    return a_blend, sc_blend

    return None, None


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    # Use only the top portion for efficiency
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    total = exp_Tb.sum()
    if total < 1e-300:
        return 0.0, np.zeros(n)
    log_sum_exp = max_Tb + np.log(total)
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient
    w = exp_Tb / total
    grad_a = np.zeros(n)
    # Only process significant weights
    thresh = 1e-15
    active = np.where(w > thresh)[0]
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_refine(a0, time_limit, T_start=50.0, T_min=0.2, n_stages=20, iters_per_stage=40):
    """LogSumExp L-BFGS with adaptive temperature."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    T = T_start
    stagnant = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_lse(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_lse(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                stagnant = 0
            else:
                stagnant += 1

            if stagnant >= 2:
                T = max(T * 1.3, T_min * 5)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def dense_subgradient_polish(a0, time_limit, n_steps=200000):
    """Dense subgradient with Polyak step and averaging for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks (top 10%)
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[::-1][:30]
            active = active[top_idx]

        # Compute averaged subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Project to simplex tangent
        S = np.dot(g, a)
        grad_z = a * (g - S)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-18:
            break

        # Polyak step
        lower_bound = 1.5  # Conservative lower bound
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.3)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Line search if needed
        if sc_new > cur_score * 1.005:
            for shrink in [0.5, 0.25, 0.1]:
                z_trial = z - eta * shrink * grad_z
                a_trial = softmax_from_z(z_trial)
                if score_normalized(a_trial) < cur_score:
                    eta *= shrink
                    z_new = z - eta * grad_z
                    a_new = softmax_from_z(z_new)
                    sc_new = score_normalized(a_new)
                    break
            else:
                # Accept anyway with tiny step
                eta *= 0.01
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        # Running average
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 3000:
            avg_count = 3000
            avg_a = (avg_a * 3000 + a) / 3001

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def symmetric_constraint(a):
    """Enforce palindromic symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a[i]
        a_sym[n - 1 - i] = a[i]
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.02)
    elif choice == 8:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Beta-like distribution
        x = np.linspace(0.001, 0.999, n)
        a = x ** 2 * (1 - x) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def anti_diagonal_polish(a, time_limit, n_iter=50000):
    """
    Iterative anti-diagonal repulsion with adaptive beta.
    This is a Goosens-inspired approach for peak reduction.
    """
    n = len(a)
    a_curr = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    beta = 0.5

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Apply repulsion
        a_new = anti_diagonal_repulsion(a_curr, beta, n_peaks=5)
        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a_curr = a_new.copy()
            # Reduce beta as we make progress
            beta *= 0.995
        else:
            # Try smaller beta
            beta *= 0.5
            a_new = anti_diagonal_repulsion(a_curr, beta, n_peaks=5)
            sc_new = score_normalized(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a_curr = a_new.copy()
            else:
                # No improvement, try adaptive beta
                b = fast_convolve(a_curr)
                top_k = b.max()
                ratio = (b.max() - np.median(b)) / (top_k + 1e-15)
                beta = max(0.05, min(2.0, ratio * 0.5))

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AntiDiag iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} beta={beta:.4f}")

    return best_a, best_score


def multi_peak_subgradient(a0, time_limit, n_steps=100000):
    """Subgradient that tracks multiple peaks simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Track peak lags
    peak_lags = []
    last_check = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Periodically update peak lags
        if step - last_check > 500:
            threshold = max_b * 0.97
            peak_lags = list(np.where(b >= threshold)[0])
            if len(peak_lags) == 0:
                peak_lags = [int(np.argmax(b))]
            if len(peak_lags) > 20:
                peak_lags = list(np.argsort(b)[::-1][:20])
            last_check = step

        if not peak_lags:
            peak_lags = [n - 1]

        # Compute gradient as average over tracked peaks
        g = np.zeros(n)
        for k in peak_lags:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(peak_lags)

        # Project to simplex
        S = np.dot(g, a)
        g_proj = g - S
        g_norm = np.linalg.norm(g_proj)
        if g_norm < 1e-18:
            break

        # Step size: decreasing
        eta = 1.0 / (np.sqrt(step + 1) * 10.0)
        eta = min(eta, 0.1)

        a_new = a - eta * g_proj
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc_new = score_normalized(a_new)
        if sc_new < cur_score:
            a = a_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
        else:
            # Shrink step
            for factor in [0.5, 0.25, 0.1]:
                a_trial = a - eta * factor * g_proj
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                if score_normalized(a_trial) < cur_score:
                    a = a_trial
                    sc_new = score_normalized(a)
                    if sc_new < best_score:
                        best_score = sc_new
                        best_a = a.copy()
                    break

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiPeakSG step={step+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP main - 30% time ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score:.8f}) ===")
    t1 = 0.30 * (budget_s - 30)
    a_out, _ = nesterov_lp_adaptive(best_seq, t1, solver=solver)
    try_update(a_out, "nesterov_main")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: Anti-diagonal repulsion - 15% time ===
    print(f"\n=== Phase 2: Anti-diagonal Repulsion (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.15 * (budget_s - 30), remaining * 0.4)
    if t2 > 30:
        a_adv, _ = anti_diagonal_polish(best_seq, t2, n_iter=100000)
        try_update(a_adv, "anti_diag")
        # Follow up with Nesterov
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_after, _ = nesterov_lp_adaptive(a_adv, remaining * 0.4, solver=solver)
            try_update(a_after, "nesterov_after_adv")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Constraint tightening + LSE - 25% time ===
    print(f"\n=== Phase 3: Constraint Tightening + LSE (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.25 * (budget_s - 30), remaining * 0.5)
    if t3 > 30:
        # Constraint tightening
        t3a = t3 * 0.25
        improved, sc_imp = constraint_tightening_lp(best_seq, solver=solver, n_attempts=10)
        if improved is not None:
            try_update(improved, "constraint_tight")

        # LSE refinement
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse, _ = lse_refine(best_seq, remaining * 0.5, T_start=80.0, T_min=0.15, n_stages=20, iters_per_stage=35)
            try_update(a_lse, "lse_main")

        # LSE from perturbed
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            perturbed = best_seq + 0.03 * rng.normal(0, 1, len(best_seq))
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_lse2, _ = lse_refine(perturbed, remaining * 0.6, T_start=30.0, T_min=0.1, n_stages=15, iters_per_stage=30)
            try_update(a_lse2, "lse_perturbed")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Multi-peak subgradient - 15% time ===
    print(f"\n=== Phase 4: Multi-Peak Subgradient (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t4 > 20:
        a_mpsg, _ = multi_peak_subgradient(best_seq, t4, n_steps=200000)
        try_update(a_mpsg, "multi_peak_sg")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_sg2, _ = dense_subgradient_polish(best_seq, remaining * 0.5, n_steps=100000)
            try_update(a_sg2, "dense_sg")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Symmetric variant - 10% time ===
    print(f"\n=== Phase 5: Symmetric (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.10 * (budget_s - 30), remaining * 0.4)
    if t5 > 20:
        a_sym = symmetric_constraint(best_seq)
        # Nesterov on symmetric
        a_sym_out, _ = nesterov_lp_adaptive(a_sym, t5 * 0.5, solver=solver)
        try_update(a_sym_out, "sym_nesterov")

        # Anti-diagonal on symmetric result
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_sym_adv, _ = anti_diagonal_polish(a_sym_out, remaining * 0.5, n_iter=30000)
            try_update(a_sym_adv, "sym_anti_diag")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Multi-length exploration - 5% time ===
    print(f"\n=== Phase 6: Multi-Length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t6_total = min(0.05 * (budget_s - 30), remaining * 0.5)
    if t6_total > 20:
        for n_try in [n_base + 200, n_base + 400, n_base - 200]:
            remaining = global_deadline - time.monotonic()
            if remaining < 15:
                break
            t6 = min(t6_total / 3.0, remaining * 0.4)
            if t6 < 10:
                break
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_adaptive(init_try, t6 * 0.7, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # === Phase 7: Final combined polish - 5% time ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        # Quick Nesterov
        a_f1, _ = nesterov_lp_adaptive(best_seq, remaining * 0.3, solver=solver, max_iters=30000)
        try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # Quick anti-diagonal
            a_f2, _ = anti_diagonal_polish(best_seq, remaining * 0.4, n_iter=20000)
            try_update(a_f2, "final_anti_diag")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Quick LSE
            a_f3, _ = lse_refine(best_seq, remaining * 0.5, T_start=20.0, T_min=0.1, n_stages=10, iters_per_stage=25)
            try_update(a_f3, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
