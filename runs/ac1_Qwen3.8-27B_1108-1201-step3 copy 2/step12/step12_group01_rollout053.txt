"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP Nesterov (zero-sum game) with scipy HiGHS sparse LP + adaptive restart
2. POCS frequency-domain peak reduction with annealing (new: exploits |FFT(a)|^2)
3. Top-k LogSumExp gradient descent with Adam (multi-stage annealing)
4. Multi-length exploration + greedy peak shaving for final polish

import numpy as np
import time
from scipy.fft import rfft as np_rfft, irfft as np_irfft, fft as np_fft, ifft as np_ifft
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np_rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np_irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_conv_costructure(n):
    """Precompute COO structure for (2n-1) x n convolution matrix."""
    rows = []
    cols = []
    offsets = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            offsets.append(k - j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), np.array(offsets, dtype=np.int32)


def solve_lp_best_response_fast(a_fixed, rows, cols, offsets):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0 using scipy linprog HiGHS."""
    n = len(a_fixed)
    m = 2 * n - 1
    # x = [b_0,...,b_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M

    # A_ub: T @ b - M <= 0  =>  [T, -1] @ x <= 0
    vals = a_fixed[offsets]
    T = coo_matrix((vals, (rows, cols)), shape=(m, n))
    
    # Build A_ub as sparse: (m x (n+1))
    # Columns 0..n-1 are T, column n is -1
    A_ub = T.tocsr().astype(np.float64)
    # Append -I column
    neg_id = coo_matrix((-np.ones(m), (np.arange(m), np.full(m, n))), shape=(m, n + 1))
    A_ub_full = csr_matrix((np.hstack([np.asarray(A_ub.todense().ravel()), -np.ones(m)]),
                            None, shape=(m, n + 1)))
    # More efficient: build directly
    A_ub_ns = A_ub.nnz
    # Build A_ub as COO
    aub_rows = list(rows) + list(np.arange(m))
    aub_cols = list(cols) + [n] * m
    aub_vals = list(a_fixed[offsets]) + [-1.0] * m
    A_ub_mat = coo_matrix((aub_vals, (aub_rows, aub_cols)), shape=(m, n + 1)).tocsr()
    b_ub = np.zeros(m)

    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: b >= 0 (default), M free
    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c_obj, A_ub=A_ub_mat, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res is None or not res.success:
        return None, None

    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(res.x[-1])


def nesterov_lp_optimize(a0, time_limit, max_iters=50000):
    """Nesterov-accelerated LP best-response with adaptive restart."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    rows, cols, offsets = build_conv_costructure(n)

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall_count = 0
    eval_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response_fast(y, rows, cols, offsets)
        if b_star is None:
            continue
        eval_count += 1

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if t % 100 == 0:
            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall_count = 0

        # Adaptive restart
        if stall_count > 500 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            stall_count = 0

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}, evals={eval_count}")

    return best_a, best_score


def pocs_freq_peaks(a0, time_limit, n_stages=15):
    """
    POCS (Projection onto Convex Sets) in frequency domain with annealing.
    Alternates between:
      S1: {a >= 0, sum(a) = 1}
      S2: {max autocorr(a) <= M_target} (approximated via freq-domain clipping)
    """
    n = len(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Annealing: start with M_target = current peak, decrease geometrically
    current_peak = fast_autocorr(a).max()
    M_targets = np.geomspace(current_peak, current_peak * 0.7, n_stages)

    for si, M_target in enumerate(M_targets):
        if time.monotonic() - t0 > time_limit - 3:
            break
        # Inner iterations for this M_target
        inner_iters = max(50, int(min(time_limit - (time.monotonic() - t0), 300.0) * 2))
        for it in range(inner_iters):
            if time.monotonic() - t0 > time_limit - 3:
                break
            # Compute autocorrelation spectrum
            A = np_fft(a, fft_size)
            phase = np.angle(A)
            c = np_ifft(A * np.conj(A), fft_size).real[:2 * n - 1]

            # Clip peaks
            c_clipped = np.minimum(c, M_target)
            c_clipped = np.maximum(c_clipped, 0)

            # Back to frequency domain
            C_new = np_fft(c_clipped, fft_size)
            mag_sq_new = np.maximum(C_new.real, 0)
            mag_new = np.sqrt(mag_sq_new)
            A_new = mag_new * np.exp(1j * phase)

            # Back to time domain
            a_new = np_ifft(A_new, fft_size).real[:n]

            # Project onto S1: non-negative, normalized
            a = np.maximum(a_new, 0)
            sta = a.sum()
            if sta < 1e-10:
                a = best_a.copy()
            else:
                a = a / sta

            # Track best
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        elapsed = time.monotonic() - t0
        print(f"      POCS stage {si+1}/{n_stages} (M={M_target:.8f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def topk_lse_grad_obj(a, T, topk):
    """Compute top-k LSE objective and gradient w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    return lse_val, grad_a


def adam_topk_lse(a0, time_limit, T_start=80.0, T_end=0.3, n_stages=20, topk=12):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        base_lr = 3e-3 * min(1.0, 60.0 / n)
        lr = base_lr * (0.75 ** si)
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        remaining = time_limit - (time.monotonic() - t0)
        max_it = max(20, int(remaining * 2))
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break
            a = softmax_fn(z)
            obj, grad_a = topk_lse_grad_obj(a, stage_T, topk)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            if (it + 1) % 10 == 0:
                a_chk = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_chk).max()
                if score < best_score:
                    best_score = score
                    best_a = a_chk.copy()
                    if (it + 1) % 50 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def greedy_peak_shaving(a0, time_limit, seed=42):
    """Greedy peak shaving: move mass from peak-contributing positions to safe positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    st = a.sum()
    if st < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / st
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        bmax = c.max()
        threshold = bmax * 0.998
        top_peaks = np.where(c >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(c)[::-1][:5]

        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            # Find safe destinations
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08, 0.12]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(1, 4)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st_t = a_trial.sum()
            if st_t > 1e-10:
                a_trial = a_trial / st_t
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            if stall > 200:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(2 * x) + 0.1 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.5 * np.exp(-4 * (x - 0.7) ** 2)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn
