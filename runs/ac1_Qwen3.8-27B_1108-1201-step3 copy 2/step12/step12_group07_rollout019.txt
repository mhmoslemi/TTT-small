Given the current sequence `a` with sum=1, suppose the top peaks are at lags k_1, k_2, ..., k_m with values M_1 >= M_2 >= ... >= M_m.

1. For the dominant peak at k_1, the constraint is `sum_j a_j * a_{k_1-j} <= M_1`.

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    size = 2 * n - 1

def subgradient_nesterov_fast(a0, time_limit, max_steps=200000):

    last_improve = 0
    patience = 0

    for step in range(max_steps):

            break
        patience += 1
        if patience > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]

            if s_n < 1e-15:
                a_new = a.copy()

            best_a = a_new.copy()
            a_prev = a

def nesterov_lp_shake(a0, time_limit, solver=None, max_iters=2000000, topk_frac=0.05):
    """LP-based Nesterov with adaptive momentum, warm-start, and periodic shaking."""
    n = len(a0)
    a = normalize_seq(a0)
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    shake_interval = 500
    shake_perturb = 0.02
    patience_counter = 0
    topk_count = max(10, int(topk_frac * (2 * n - 1)))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        # Periodic shaking to escape local minima
        if it > 0 and it % shake_interval == 0 and t_nest > 3.0:
            rng = np.random.default_rng(it)
            y_shake = y * (1.0 + shake_perturb * (rng.random(n) - 0.5))
            y_shake = np.maximum(y_shake, 0)
            sy = y_shake.sum()
            if sy > 1e-15:
                y = y_shake / sy

        # Solve LP (use top-k for speed, full for verification)
        if it % 20 == 0:
            # Full LP every 20 iterations for accuracy
            b_new, M_lp = solve_lp_best_response(y, solver)
        else:
            # Top-k LP for speed
            b_conv = fast_convolve(y)
            topk_lags = np.argsort(b_conv)[-topk_count:]
            topk_lags = np.sort(topk_lags)
            b_new, M_lp = solve_lp_topk(y, topk_lags, solver)
            # Verify top-k solution against full constraints
            if b_new is not None:
                full_conv = build_T(y) @ b_new
                if full_conv.max() > M_lp * 1.01:
                    # Top-k solution violates other constraints, use full LP
                    b_new, M_lp = solve_lp_best_response(y, solver)

        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score * (1 - 1e-12):
            best_score = sc_new
            best_a = a_new.copy()
            patience_counter = 0
        else:
            patience_counter += 1

        # Adaptive restart
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    LP-Nesterov iter={it+1} ({elapsed:.0f}s): "
                      f"best={best_score:.10f} t={t_nest:.1f} restarts={restart_count} "
                      f"patience={patience_counter}")
                last_print = elapsed

    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z_in, T_val, n):
    """LogSumExp objective and gradient in z-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    active_k = np.where(w > 1e-15)[0]
    for k in active_k:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=80.0, T_end=0.01, n_stages=30):

            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):

            return obj

        def jac_staged(z_val):
            _, grad = lse_obj_grad(z_val, stage_T, n)
            return grad

        if si < n_stages // 3:
            iters = 60

            iters = 40
        else:
            iters = 25

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,

                         'disp': False}

            z_new = res.x
            a_new = softmax_from_z(z_new)

            z = z_new

        except Exception:

    return best_a, best_score

def directed_peak_reduction(a0, time_limit, max_iters=50000):
    """Targeted reduction of the dominant peak via coordinate-like moves."""
    n = len(a0)
    a = normalize_seq(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(456)
    step = 0.05
    last_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        k_star = int(np.argmax(b))

        # Find the element whose decrease most reduces peak at k_star
        # Contribution of a_j to b_{k_star} is a_{k_star - j} (if valid)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        valid_j = np.arange(js, je + 1)
        contributions = a[k_star - valid_j]

        # Pick the j with highest contribution
        j_star = valid_j[np.argmax(contributions)]

        # Find a donor: element that contributes least to the top peaks
        top_peaks = np.argsort(b)[-5:]
        donor_contrib = np.zeros(n)
        for k in top_peaks:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            donor_contrib[js2:je2 + 1] += a[k - np.arange(js2, je2 + 1)]

        # Pick donor with lowest contribution, excluding j_star
        donor_contrib[j_star] = np.inf
        j_donor = int(np.argmin(donor_contrib))

        # Transfer mass from j_star to j_donor
        delta = step * a[j_star]
        a_new = a.copy()
        a_new[j_star] -= delta
        a_new[j_donor] += delta

        if a_new[j_star] < 0:
            delta = a[j_star] * 0.99
            a_new = a.copy()
            a_new[j_star] -= delta
            a_new[j_donor] += delta

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        sc = score_normalized(a_new)
        if sc < best_score * (1 - 1e-12):
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            last_improve = it
            step = min(step * 1.005, 0.1)
        else:
            step *= 0.95
            if step < 1e-5:
                break

    return best_a, best_score


def peak_qp_reduction(a0, time_limit, solver=None):
    """Use QP to reduce the top peak while constraining other peaks."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = normalize_seq(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Build the Han matrix for each relevant lag
    b = fast_convolve(a)
    top5_lags = np.argsort(b)[-5:]
    max_b = b.max()

    for _ in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        b = fast_convolve(a)
        max_b = b.max()
        k_peak = int(np.argmax(b))

        # QP: minimize (a+d)' C_{k_peak} (a+d)
        # subject to: sum(d) = 0, a+d >= 0, (a+d)' C_k (a+d) <= max_b for k != k_peak
        # Simplified: minimize (a+d)' C_{k_peak} (a+d) s.t. sum(d)=0, a+d >= 0
        # and (a+d)' C_k (a+d) <= max_b * 1.001 for top 3 other peaks

        d = cp.Variable(n)
        a_new_var = a + d

        # Objective: minimize peak at k_peak = sum_j a_new[j] * a_new[k_peak-j]
        # This is bilinear, not quadratic. Use linearization around current a.
        # Linear approx: b_k_peak approx b_k_peak(a) + 2 * sum_j (a_j - a[j]) * a[k_peak-j]
        # = b_k_peak(a) + 2 * d' * a_shifted

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        grad_peak = np.zeros(n)
        grad_peak[js:je + 1] = 2.0 * a[k_peak - np.arange(js, je + 1)]

        # Constraint gradient vectors for other top peaks
        other_peaks = [k for k in top5_lags if k != k_peak][:3]
        obj = cp.sum(grad_peak * d)
        constraints = [cp.sum(d) == 0, a_new_var >= 0, cp.norm(d, 1) <= 0.1]

        for k_other in other_peaks:
            js2 = max(0, k_other - n + 1)
            je2 = min(n - 1, k_other)
            grad_other = np.zeros(n)
            grad_other[js2:je2 + 1] = 2.0 * a[k_other - np.arange(js2, je2 + 1)]
            # Linear constraint: b_k_other + grad_other' d <= max_b * 1.001
            constraints.append(cp.sum(grad_other * d) <= max_b * 0.001)

        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        d_val = np.asarray(d.value, dtype=np.float64)
        if d_val is None or not np.all(np.isfinite(d_val)):
            break

        a_candidate = a + d_val
        a_candidate = np.maximum(a_candidate, 0)
        s_c = a_candidate.sum()
        if s_c < 1e-15:
            break
        a_candidate = a_candidate / s_c

        sc = score_normalized(a_candidate)
        if sc < best_score * (1 - 1e-10):
            best_score = sc
            best_a = a_candidate.copy()
            a = a_candidate
        else:
            # Try smaller step
            a_candidate = a + 0.5 * d_val
            a_candidate = np.maximum(a_candidate, 0)
            s_c = a_candidate.sum()
            if s_c > 1e-15:
                a_candidate = a_candidate / s_c
                sc = score_normalized(a_candidate)
                if sc < best_score * (1 - 1e-10):
                    best_score = sc
                    best_a = a_candidate.copy()
                    a = a_candidate
                else:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            return

        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            elapsed = time.monotonic()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(arr)}) ***")

    # === Phase 1: Fast subgradient Nesterov (40s) ===
    print(f"\n=== Phase 1: Subgradient Nesterov at n={n_base} ===")
    t1 = min(40.0, (_global_deadline - time.monotonic()) * 0.03)
    if t1 > 5:
        a_out, _ = subgradient_nesterov_fast(best_seq, t1, max_steps=500000)
        try_update(a_out, "subgrad_p1")
    print(f"  Phase 1 done: best_score={best_score:.10f}")

    # === Phase 2: LP-based Nesterov with shaking (500s) ===

    # === Phase 3: Directed peak reduction (100s) ===
