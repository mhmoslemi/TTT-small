def fast_convolution(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def soft_max(x, tau):
    """Compute log-sum-exp based soft max."""
    x_max = np.max(x)
    if x_max > 1e30:
        return x_max
    return tau * np.log(np.sum(np.exp((x - x_max) / tau))) + x_max

def soft_max_weights(x, tau):
    """Compute softmax weights for the soft max."""
    x_max = np.max(x)
    w = np.exp((x - x_max) / tau)
    w = w / w.sum()
    return w

def objective_and_grad(a, tau, reg=1e-10):
    """
    Compute the smoothed objective and its gradient.
    f(a) = 2n * soft_max(conv(a), tau) / (sum(a))^2 + reg * sum(a^2)
    """
    n = len(a)
    a = np.maximum(a, 1e-30)  # ensure positive for numerical stability
    S = np.sum(a)
    if S < 1e-15:
        return 2.0, np.zeros(n)
    
    # Compute convolution
    c = fast_convolution(a)
    
    # Soft max and weights
    M = soft_max(c, tau)
    w = soft_max_weights(c, tau)
    
    # Objective
    f = 2.0 * n * M / (S * S) + reg * np.sum(a * a)
    
    # Gradient

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Multi-scale (multi-n) optimization

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_obj_grad(a, tau, reg):
    """
    Compute the log-sum-exp smoothed objective and gradient.
    f(a) = 2n * soft_max(conv(a), tau) / (sum(a))^2 + reg * ||a||^2
    
    Returns (f, grad).
    """
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return 2.0, np.zeros_like(a)
    
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    exp_sum = exp_c.sum()
    M = c_max + tau * np.log(exp_sum)
    w = exp_c / exp_sum
    
    f = 2.0 * n * M / (S * S) + reg * np.dot(a, a)
    
    # Gradient
    # df/da_i = (2n/S^2) * corr[i] - (4nM/S^3) * a_i + 2*reg*a_i
    # corr[i] = sum_{k} w[k] * a[k-i]  (a[j]=0 for j outside [0,n-1])
    corr = np.empty(n)
    for i in range(n):
        ks = max(0, i)
        ke = min(2 * n - 1, i + n)
        if ks <= ke:
            corr[i] = np.dot(w[ks:ke+1], a[0:ke+1-ks])
        else:
            corr[i] = 0.0
    
    grad = (2.0 * n / (S * S)) * corr - (4.0 * n * M / (S ** 3)) * a + 2.0 * reg * a
    return f, grad

class SmoothObj:
    """Wrapper class for scipy.optimize.minimize (avoids lambda)."""
    def __init__(self, tau, reg):
        self.tau = tau
        self.reg = reg
    
    def f(self, a):
        return smooth_obj_grad(a, self.tau, self.reg)[0]
    
    def g(self, a):
        return smooth_obj_grad(a, self.tau, self.reg)[1]

def run_lbgfs(a0, tau_list, reg, max_iter_per_tau, time_limit, start_time, verbose=True):
    """
    Run L-BFGS-B with a list of decreasing temperatures.
    Returns (best_a, best_score_estimate).
    """
    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-10)
    S = a.sum()
    if S > 1e-15:
        a = a / S
    
    best_a = a.copy()
    best_true = evaluate_safe(best_a)
    
    for tau in tau_list:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit:
            break
        
        obj = SmoothObj(tau, reg)
        
        result = minimize(
            fun=obj.f,
            x0=a.copy(),
            jac=obj.g,
            method='L-BFGS-B',
            bounds=[(1e-10, None)] * n,
            options={
                'maxiter': max_iter_per_tau,
                'ftol': 1e-16,
                'gtol': 1e-10,
                'maxfun': max_iter_per_tau * 10,
                'iprint': -1,
            }
        )
        
        a_new = np.maximum(result.x, 1e-10)
        S_new = a_new.sum()
        if S_new > 1e-15:
            a_new = a_new / S_new
        
        score_new = evaluate_safe(a_new)
        if score_new < best_true:
            best_true = score_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose:
            elapsed = time.monotonic() - start_time
            print(f"    tau={tau:.6f}: score={score_new:.6f}, best={best_true:.6f} (t={elapsed:.0f}s)")
    
    return best_a, best_true

def coord_descent_polish(a0, time_budget, start_time, verbose=True):
    """
    Peak-directed coordinate descent.
    Try to reduce the maximum convolution value by moving mass.
    """
    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-15:
        a = a / S
    
    best_a = a.copy()
    best_score = evaluate_safe(a)
    rng = np.random.default_rng(2024)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - start_time < time_budget:
        c = fast_convolve(a)
        max_k = int(np.argmax(c))
        
        # Find hot pairs for the peak lag
        ks = max(0, max_k - n + 1)
        ke = min(n - 1, max_k)
        pairs = np.arange(ks, ke + 1)
        vals = a[pairs] * a[max_k - pairs]
        
        if vals.max() < 1e-20:
            break
        
        # Try moving mass from the hottest position
        hot_idx = pairs[np.argmax(vals)]
        
        # Find cold destination (positions not in any top conv lag)
        top_lags = np.argsort(c)[::-1][:min(20, len(c))]
        hot_positions = np.zeros(n, dtype=bool)
        for k in top_lags:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            hot_positions[s:e+1] = True
        
        cold_positions = np.where(~hot_positions)[0]
        if len(cold_positions) == 0:
            cold_positions = np.arange(n)
        
        # Try moving to the best cold position
        best_move_score = best_score
        best_move = None
        
        # Sample candidates
        candidates = cold_positions[rng.permutation(len(cold_positions))[:min(200, len(cold_positions))]]
        
        for dst in candidates:
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[hot_idx] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[hot_idx] -= amt
                a_trial[dst] += amt
                if a_trial[hot_idx] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                St = a_trial.sum()
                if St < 1e-15:
                    continue
                a_trial = a_trial / St
                sc = evaluate_safe(a_trial)
                if sc < best_move_score:
                    best_move_score = sc
                    best_move = a_trial.copy()
        
        if best_move is not None:
            a = best_move
            if best_move_score < best_score:
                best_score = best_move_score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                St = a.sum()
                if St > 1e-15:
                    a = a / St
                    sc = evaluate_safe(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                    stall += 1
        
        if stall > max_stall:
            break
    
    if verbose:
        print(f"    Coord polish: score={best_score:.6f}")
    
    return best_a, best_score

def interpolate_to_new_length(a, new_n):
    """Interpolate a sequence to a new length, preserving the normalized shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    S = a.sum()
    if S < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / S
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    Sn = a_new.sum()
    if Sn < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / Sn

def make_gaussian_init(n, sigma_frac=0.3):
    """Create a Gaussian-shaped initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a / a.sum()
    return a

def make_cosine_init(n, power=2.0):
    """Create a cosine-shaped initialization."""
    x = np.arange(n)
    a = np.cos(np.pi * (x - (n-1)/2.0) / (n-1))
    a = np.abs(a) ** power
    a[np.abs(a) < 1e-10] = 0
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_asymmetric_init(n, bias=0.3):
    """Create an asymmetric initialization (helps reduce conv peak)."""
    x = np.arange(n) / (n - 1)
    # Skewed Gaussian
    center = 0.5 + bias
    sigma = 0.25
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a / a.sum()
    return a

def make_tophat_init(n, width_frac=0.6):
    """Create a trapezoidal initialization."""
    x = np.arange(n) / (n - 1)
    w = width_frac
    a = np.zeros(n)
    margin = (1.0 - w) / 2.0
    ramp = 0.05
    for i in range(n):
        if x[i] < margin - ramp or x[i] > 1.0 - margin + ramp:
            a[i] = 0.0
        elif x[i] < margin:
            a[i] = (x[i] - (margin - ramp)) / ramp
        elif x[i] > 1.0 - margin:
            a[i] = ((1.0 - margin + ramp) - x[i]) / ramp
        else:
            a[i] = 1.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

The problem: min max_k a^T M_k a / (1^T a)^2 subject to a >= 0

This can be written as: min t subject to a^T M_k a <= t * (1^T a)^2 for all k, a >= 0, 1^T a = 1

Each constraint a^T M_k a <= t * (1^T a)^2 is a quadratic constraint. Since M_k is a permutation matrix (or sub-permutation), a^T M_k a = sum_{i+j=k} a[i]*a[j].

This is a QCQP (quadratically constrained quadratic program). These are NP-hard in general, but for this specific structure, we might be able to solve them efficiently.

Actually, for fixed t, the feasibility check is:
Does there exist a >= 0, 1^T a = 1, such that sum_{i+j=k} a[i]*a[j] <= t for all k?

This is a system of quadratic inequalities. It's not convex (the feasible set is not convex in general).

However, we can use a bisection approach:
- Binary search on t
- For each t, check feasibility (which is the hard part)

The feasibility check is itself a non-convex problem. But we can use a adaptive approach:
- Start from a feasible point (the current best)
