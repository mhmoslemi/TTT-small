    # Sort x in decreasing order: x_(1) >= x_(2) >= ... >= x_(n)

def project_onto_simplex(x, S):
    """Euclidean projection onto {y >= 0, sum(y) = S}."""
    n = len(x)
    if S <= 0:
        return np.zeros(n)
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    theta_candidates = (cumsum - S) / np.arange(1, n + 1)
    valid = sorted_x > theta_candidates
    if not np.any(valid):
        rho = 1
    else:
        rho = np.max(np.where(valid)[0]) + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(x - theta, 0.0)


def build_T_subset(a, k_indices):
    """Build constraint matrix rows for given lag indices."""
    n = len(a)
    size = len(k_indices)
    T = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a[k - lo:k - lo + (hi - lo)]
    return T


def lp_solve(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_arr = np.asarray(active_set, dtype=int)
    T = build_T_subset(a_fixed, k_arr)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def check_violations(a_fixed, b, M, active_set, n):
    """Find lag indices where (a_fixed*b)[k] > M but k not in active_set."""
    size = 2 * n - 1
    all_k = np.arange(size)
    conv = np.array([

               b[max(0, k - n + 1):min(n, k + 1)][::-1] if False else
               b[np.clip(k - np.arange(max(0, k - n + 1), min(n, k + 1)), 0, n - 1)])
        for k in all_k
    ])

    viol_indices = not_active[viol_mask[not_active]]

def softmax_gradient_optimize(a0, time_limit, verbose=False):
    """
    Soft-max smoothed gradient descent with continuation on beta.
    Minimize f_beta(a) = (1/beta) * log(sum_k exp(beta * (a*a)[k]))
    with a >= 0, sum(a) = S.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Continuation schedule
    beta_start = 5.0
    beta_end = 500.0
    n_total = 5000  # target number of iterations
    beta_schedule = np.geomspace(beta_start, beta_end, n_total)

    # Nesterov state
    y = a.copy()
    y_prev = a.copy()
    a_curr = a.copy()
    step_base = 1.0 / (n * 0.5)

    for t in range(1, n_total + 1):
        if time.monotonic() - t0 > time_limit:
            break

        beta = beta_schedule[min(t - 1, n_total - 1)]
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Compute soft-max weights
        exp_vals = np.exp(beta * (conv_y - M_curr))
        p = exp_vals / exp_vals.sum()

        # Compute gradient via FFT
        N_fft = 4 * n
        A_f = fft(a_curr, N_fft)
        P_f = fft(p, N_fft)

        # c1[j] = sum_k p[k]*a[k-j] (cross-correlation)
        c1 = ifft(np.conj(P_f) * A_f, N_fft).real[:n]
        # c2[j] = sum_k p[k]*a[j-k] (convolution)
        c2 = ifft(P_f * A_f, N_fft).real[:n]

        grad = c1 + c2

        # Adaptive step size with backtracking
        step = step_base / beta
        max_backtrack = 8
        for bt in range(max_backtrack):
            a_trial = y - step * grad
            a_trial = project_onto_simplex(a_trial, norm_val)
            score_trial = fast_autocorr(a_trial).max()
            if score_trial <= M_curr * (1 + 1e-10) or bt == max_backtrack - 1:
                break
            step *= 0.5

        a_new = a_trial

        # Nesterov momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new
        y_new = project_onto_simplex(y_new, norm_val)

        momentum = (t - 1.0) / (t + 2.0)
        momentum = min(momentum, 0.98)
        a_curr_new = y_new + momentum * (y_new - y_prev)
        a_curr_new = project_onto_simplex(a_curr_new, norm_val)

        y_prev = y
        y = y_new
        a_curr = a_curr_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a_curr).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_curr.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftMax iter={t} ({elapsed:.0f}s): beta={beta:.1f}, best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_optimize(a0, time_limit, verbose=False):
    """
    Coordinate descent with exact 1D optimization.
    For each coordinate j, optimize a[j] given all others fixed.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve_passes = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = a.sum() - aj

            # Compute background and coefficient vectors
            # (a*a)[k] = bg[k] + 2*aj*c[k] + aj^2*delta_{k,2j}
            conv_current = fast_autocorr(a)

            # c[k] = a[k-j] (if valid, k-j != j) + a[j-k] (if valid, j-k != j)
            c = np.zeros(size)
            for k in range(size):
                v = 0.0
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    v += a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    v += a[i2]
                c[k] = v

            bg = conv_current - 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            # Optimize x = a[j]: minimize max_k (bg[k] + 2*x*c[k] + x^2*delta_{k,2j}) / (s_others + x)^2
            # Subject to x >= 0

            # Grid + bisection
            x_candidates = [0.0, aj * 0.25, aj * 0.5, aj * 0.75, aj, aj * 1.25, aj * 1.5,
                           aj * 2.0, s_others * 0.5, s_others, norm_val]
            x_candidates = [x for x in x_candidates if x >= 0 and x <= norm_val * 2]

            def obj_at_x(x):
                new_conv = bg + 2.0 * x * c
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                new_sum = s_others + x
                if new_sum < 1e-15:
                    return float('inf')
                return 2.0 * n * new_conv.max() / (new_sum * new_sum)

            current_obj = obj_at_x(aj)
            best_x = aj
            best_obj = current_obj

            for x in x_candidates:
                obj = obj_at_x(x)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            # Bisection around best_x
            if abs(best_x - aj) > 1e-10:
                lo = min(aj, best_x) * 0.5
                hi = max(aj, best_x) * 1.5
                for _ in range(12):
                    mid = (lo + hi) / 2.0
                    obj_mid = obj_at_x(mid)
                    if obj_mid < best_obj:
                        best_obj = obj_mid
                        best_x = mid
                        hi = mid
                    else:
                        lo = mid

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_val / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-14:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm_val / a.sum()

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= max_no_improve:
                break
        else:
            no_improve_passes = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient polishing at active peaks with backtracking."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    step = 0.3 / np.sqrt(n)
    patience = 0
    max_patience = 1000

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Pick random active peak
        threshold = M_curr * (1.0 - 0.05)
        candidates = np.where(conv >= threshold)[0]
        if len(candidates) == 0:
            k_star = int(np.argmax(conv))
        else:
            k_star = candidates[np.random.randint(len(candidates))]

        # Compute subgradient at k_star
        grad = np.zeros(n)
        for j in range(n):
            v = 0.0
            j1 = k_star - j
            if 0 <= j1 < n:
                v += a[j1]
            j2 = j - k_star
            if 0 <= j2 < n:
                v += a[j2]
            grad[j] = v

        # Try multiple step sizes (backtracking)
        trial_step = step
        found = False
        for _ in range(10):
            a_trial = a - trial_step * grad
            a_trial = project_onto_simplex(a_trial, norm_val)
            score_trial = fast_autocorr(a_trial).max()
            if score_trial < M_curr * (1 - 1e-10):
                a = a_trial
                best_score = min(best_score, score_trial)
                best_a = a.copy()
                patience = 0
                step = min(step * 1.02, 3.0)
                found = True
                break
            trial_step *= 0.5

        if not found:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t3, solver, verbose=True, seed=123)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Try different length (n+400) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_cur + 400
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 50:
        print(f"\n=== Phase 4: LP Nesterov n={n_alt}, budget={t4:.0f}s ===")
        a_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = a_alt * np.sqrt(2.0 * n_alt) / a_alt.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t4, solver, verbose=True, seed=456)
        try_update(a_out, f"lp_alt_n{n_alt}")

    # === Phase 5: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Subgrad polish n={n_cur}, budget={t5:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t5, verbose=True)
        try_update(a_out, f"subg_n{n_cur}")

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 15:
        print(f"\n=== Phase 6: CD n={n_cur}, budget={t6:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_optimize(a_norm, t6, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 7: Final LP + soft-max combo ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        t7a = min(remaining * 0.4, 120.0)
        t7b = min(remaining * 0.3, 80.0)
        if t7a > 30:
            print(f"\n=== Phase 7a: Final LP n={n_cur}, budget={t7a:.0f}s ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lp_nesterov_optimize(a_norm, t7a, solver, verbose=False, seed=789)
            try_update(a_out, f"lp_final_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.5, 100.0)
            print(f"\n=== Phase 7b: Final SoftMax n={n_cur}, budget={t7b:.0f}s ===")
            a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out, _ = softmax_gradient_optimize(a_norm, t7b, verbose=False)
            try_update(a_out, f"softmax_final_n{n_cur}")

    # === Phase 8: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 8: Final subgrad, remaining={remaining:.0f}s ===")
        t8 = min(remaining * 0.5, 60.0)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t8, verbose=False)
        try_update(a_out, f"subg_final_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
