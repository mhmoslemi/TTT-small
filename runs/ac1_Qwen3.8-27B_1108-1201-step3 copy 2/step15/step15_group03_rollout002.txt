    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def build_T(a, n, size):
    """Build the convolution matrix T where (T @ b)[k] = (a*b)[k]."""
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def build_T_fast(n, size):
    """Build T matrix efficiently using strided operations."""
    T = np.zeros((size, n))
    # T[k, m] = a[m] if 0 <= k-m < n, i.e., m <= k < m+n
    # So for row k, the valid columns are max(0, k-n+1) to min(n-1, k)
    # T[k, m] = a[m] for m in [max(0,k-n+1), min(n-1,k)]
    # Wait, that's not right. Let me reconsider.
    # (a*b)[k] = sum_m a[m] * b[k-m]
    # For this to be non-zero, we need 0 <= k-m < n, i.e., k-n+1 <= m <= k
    # And also 0 <= m < n
    # So m ranges from max(0, k-n+1) to min(n-1, k)
    # And T[k, m] = a[m] (coefficient of b[m]... wait no)
    
    # Actually (a*b)[k] = sum_m a[m] * b[k-m]
    # Let j = k-m, so m = k-j
    # = sum_j a[k-j] * b[j]
    # So the coefficient of b[j] in (a*b)[k] is a[k-j] (if valid)
    # T[k, j] = a[k-j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # Also need 0 <= j < n
    
    for j in range(n):
        # T[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        T[j:j+n, j] = a
    return T

def lp_best_response(a, M, solver, use_full=True):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_T_fast(n, size)
    
    g_var = cp.Variable(n, nonneg=True)
    
    if use_full:
        constraints = [T @ g_var <= M]
    else:
        # Screened: only include constraints where (a*a) is close to M
        aa = fast_conv_self(a)
        active = np.where(aa >= M * 0.85)[0]
        constraints = [T[active] @ g_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(g_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def anderson_accelerate_iterates(history_x, history_f, m, project_fn):
    """
    Anderson acceleration: find extrapolation weights.
    history_x: list of recent iterates (length m)
    history_f: list of recent G(x) values (length m)
    Returns: extrapolated point
    """
    if len(history_x) <= 1:
        x = history_x[-1]
        f = history_f[-1]
        return project_fn(f)
    
    # We have m points: x_i, f_i = G(x_i)
    # Residuals: r_i = f_i - x_i
    # Find w (sum w_i = 1) minimizing ||sum w_i * r_i||
    # This is: min_w ||R @ w|| s.t. 1^T w = 1
    # where R is the matrix with columns r_i
    
    m_actual = len(history_x)
    R = np.column_stack([history_f[i] - history_x[i] for i in range(m_actual)])
    
    # Solve: min_w ||R @ w||^2 s.t. sum(w) = 1
    # KKT: [2 R^T R, 1; 1^T, 0] [w; lambda] = [0; 1]
    n_rows = R.shape[0]
    A = np.zeros((m_actual + 1, m_actual + 1))
    A[:m_actual, :m_actual] = 2.0 * (R.T @ R)
    A[:m_actual, m_actual] = 1.0
    A[m_actual, :m_actual] = 1.0
    b_vec = np.zeros(m_actual + 1)
    b_vec[m_actual] = 1.0
    
    try:
        sol = np.linalg.solve(A, b_vec)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A, b_vec, rcond=None)[0]
    
    w = sol[:m_actual]
    
    # Extrapolate in f-space
    x_new = np.zeros_like(history_x[0])
    for i in range(m_actual):
        x_new += w[i] * history_f[i]
    
    return project_fn(x_new)

def project_normalize(x, S):
    """Project to non-negative and normalize to sum S."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(len(x)) * S / len(x)
    return x * S / s

def anderson_lp_optimize(a0, time_limit, solver, S=None, m_anderson=6, verbose=True):
    """
    Anderson-accelerated LP best-response.
    """
    n = len(a0)
    if S is None:
        S = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * S / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Anderson history
    hist_x = []
    hist_f = []
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        M = fast_conv_self(a).max()
        if M < 1e-12:
            break
        
        # Solve LP
        g = lp_best_response(a, M, solver, use_full=True)
        if g is None:
            g = lp_best_response(a, M, solver, use_full=False)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g * S / g_sum
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_f.append(g.copy())
        
        # Keep only last m_anderson in history
        max_hist = min(m_anderson, it)
        if len(hist_x) > max_hist:
            hist_x = hist_x[-max_hist:]
            hist_f = hist_f[-max_hist:]
        
        # Project function
        def proj(x):
            return project_normalize(x, S)
        
        if it <= m_anderson:
            # Simple averaging for first few iterations
            a_new = g.copy()
        else:
            a_new = anderson_accelerate_iterates(hist_x, hist_f, max_hist, proj)
        
        # Project to feasible set
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * S / s_new
        
        a = a_new
        
        # Check improvement
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 50 == 0:
            print(f"    AA-LP it={it} ({elapsed:.0f}s): M={M:.6f}, best={best_score:.8f}")
    
    return best_a, best_score

def lp_optimize_mreduction(a0, time_limit, solver, S=None, verbose=True):
    """
    LP best-response with M-reduction strategy.
    After convergence, try to reduce M and check feasibility.
    """
    n = len(a0)
    if S is None:
        S = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * S / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    # Phase A: Standard best-response until convergence
    M_prev = float('inf')
    it = 0
    while time.monotonic() - t0 < time_limit * 0.6:
        it += 1
        M = fast_conv_self(a).max()
        
        g = lp_best_response(a, M, solver, use_full=True)
        if g is None:
            g = lp_best_response(a, M, solver, use_full=False)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g * S / g_sum
        
        a = g.copy()
        
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Convergence check
        if abs(M - M_prev) < 1e-6 * M:
            break
        M_prev = M
        
        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MRed it={it} ({elapsed:.0f}s): M={M:.6f}, best={best_score:.8f}")
    
    # Phase B: M-reduction
    remaining = time_limit - (time.monotonic() - t0)
    if remaining < 10:
        return best_a, best_score
    
    print(f"    MRed: entering M-reduction phase ({remaining:.0f}s remaining)")
    
    a = best_a.copy()
    reduction_factor = 0.999
    n_reductions = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        M = fast_conv_self(a).max()
        M_target = M * reduction_factor
        
        # Try LP with reduced M
        g = lp_best_response(a, M_target, solver, use_full=True)
        
        if g is not None and g.sum() > 1e-10:
            # Feasible! Update
            g = g * S / g.sum()
            new_M = fast_conv_self(g).max()
            if new_M < best_score:
                best_score = new_M
                best_a = g.copy()
                a = g.copy()
                n_reductions += 1
                # Keep reducing
                reduction_factor = max(0.998, reduction_factor * 0.999)
            else:
                # G didn't help, try less aggressive reduction
                reduction_factor = max(0.998, (reduction_factor + 1.0) / 2.0)
        else:
            # Infeasible, increase M_target
            reduction_factor = min(1.0, reduction_factor * 1.001)
            if reduction_factor >= 1.0:
                # Try standard best-response again
                M_new = fast_conv_self(a).max()
                g = lp_best_response(a, M_new, solver, use_full=True)
                if g is not None and g.sum() > 1e-10:
                    g = g * S / g.sum()
                    new_sc = fast_conv_self(g).max()
                    if new_sc < best_score:
                        best_score = new_sc
                        best_a = g.copy()
                break
        
        if verbose and n_reductions % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MRed reductions={n_reductions} ({elapsed:.0f}s): best={best_score:.8f}, factor={reduction_factor:.5f}")
    
    return best_a, best_score

def peak_cd_optimize(a0, time_limit, rng, S=None, verbose=True):
    """
    Peak-targeted coordinate descent.
    For each coordinate pair (i,j), move mass from i to j to reduce the peak.
    Uses quadratic approximation for fast evaluation.
    """
    n = len(a0)
    if S is None:
        S = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * S / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find peaks
        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]
        
        # Compute per-coordinate contribution to peaks
        pressure = np.zeros(n)
        for kl in peak_lags[:min(30, len(peak_lags))]:
            for i in range(n):
                j = kl - i
                if 0 <= j < n:
                    pressure[i] += a[j]
        
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < S * 1e-8:
                continue
            
            # Choose transfer targets
            j_candidates = set()
            # Random
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            # Structured (related to peaks)
            for kl in peak_lags[:3]:
                j1 = kl - i
                if 0 <= j1 < n and j1 != i:
                    j_candidates.add(j1)
                j2 = i + kl
                if 0 <= j2 < n and j2 != i:
                    j_candidates.add(j2)
            # Low pressure coordinates
            low_pressure = np.argsort(pressure)[:5]
            j_candidates.update(low_pressure.tolist())
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in list(j_candidates)[:10]:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gain vector g where new_conv = conv + delta * g
                # Moving delta from i to j:
                # a[i] -> a[i] - delta, a[j] -> a[j] + delta
                # Change in (a*a)[k]:
                # = -delta * a_new[k-i] + delta * a_new[k-j] - delta * a_old[i + (k-i)] ... 
                
                # More carefully:
                # (a*a)[k] = sum_m a[m]*a[k-m]
                # After moving delta from i to j:
                # a'[i] = a[i] - delta, a'[j] = a[j] + delta
                # Delta (a*a)[k] = 
                #   sum_m a'[m]*a'[k-m] - sum_m a[m]*a[k-m]
                # = (a'[i]*a'[k-i] - a[i]*a[k-i]) + (a'[j]*a'[k-j] - a[j]*a[k-j])
                #   + cross terms if i + (k-m) = j or j + (k-m) = i
                # 
                # For small delta:
                # ≈ -delta * a[k-i] - delta * a[i] + delta * a[k-j] + delta * a[j]
                #   (from the i-term: d/d(delta)[-delta*a[k-i] + ... ])
                # Actually let me just compute it directly:
                # 
                # (a*a)[k] involves a[i] when m=i or k-m=i (i.e., m=k-i)
                # and a[j] when m=j or k-m=j (i.e., m=k-j)
                # 
                # If i != j and i != k-j and j != k-i:
                # Delta (a*a)[k] ≈ -delta*a[k-i] - delta*a[i] + delta*a[k-j] + delta*a[j]
                #   (the first two from removing delta from position i, last two from adding to j)
                # Wait, this isn't quite right either. Let me be very careful.
                # 
                # (a*a)[k] = sum_m a[m] * a[k-m]
                # 
                # Terms involving a[i]: m=i gives a[i]*a[k-i]; m=k-i gives a[k-i]*a[i]
                # So a[i] appears with coefficient 2*a[k-i] (if i != k-i) or a[i]^2 (if 2i=k)
                # 
                # Actually, let's just compute the difference directly:
                # a'[m] = a[m] - delta*(m==i) + delta*(m==j)
                # 
                # (a'*a')[k] = sum_m a'[m] * a'[k-m]
                # = sum_m (a[m] - delta*delta_mi + delta*delta_mj) * (a[k-m] - delta*delta_{k-m,i} + delta*delta_{k-m,j})
                # 
                # For small delta, to first order:
                # ≈ (a*a)[k] + delta * sum_m [-delta_mi * a[k-m] + delta_mj * a[k-m] - a[m]*delta_{k-m,i} + a[m]*delta_{k-m,j}]
                # = (a*a)[k] + delta * [-a[k-i] + a[k-j] - a[i] + a[j]]
                #   (where we treat the two terms carefully)
                # 
                # Wait, sum_m delta_mi * a[k-m] = a[k-i]
                # sum_m delta_mj * a[k-m] = a[k-j]
                # sum_m a[m] * delta_{k-m,i} = a[i] (when k-m=i, i.e., m=k-i, so a[k-i]... no)
                # 
                # Let me redo: sum_m a[m] * delta_{k-m, i}
                # delta_{k-m, i} = 1 when k-m = i, i.e., m = k-i
                # So this sum = a[k-i]
                # 
                # Similarly sum_m a[m] * delta_{k-m, j} = a[k-j]
                # 
                # So to first order:
                # Delta (a*a)[k] ≈ delta * [-a[k-i] + a[k-j] - a[k-i] + a[k-j]]
                # = delta * [-2*a[k-i] + 2*a[k-j]]
                # 
                # Hmm wait, I think I double-counted. Let me be more careful.
                # 
                # (a'*a')[k] = sum_m a'[m] * a'[k-m]
                # a'[m] = a[m] + delta*(delta_mj - delta_mi)
                # 
                # (a'*a')[k] = sum_m [a[m] + d*(δmj - δmi)] * [a[k-m] + d*(δ_{k-m,j} - δ_{k-m,i})]
                # where d = delta
                # 
                # = sum_m a[m]*a[k-m] + d * sum_m [a[m]*(δ_{k-m,j} - δ_{k-m,i}) + a[k-m]*(δmj - δmi)]
                #   + d^2 * sum_m (δmj - δmi)(δ_{k-m,j} - δ_{k-m,i})
                # 
                # First order term:
                # sum_m a[m]*(δ_{k-m,j} - δ_{k-m,i}) = a[k-j] - a[k-i]
                # sum_m a[k-m]*(δmj - δmi) = a[k-j] - a[k-i]
                # 
                # So first order: d * 2*(a[k-j] - a[k-i])
                # 
                # Second order:
                # (δmj - δmi)(δ_{k-m,j} - δ_{k-m,i})
                # This is non-zero when m=j or m=i, AND k-m=j or k-m=i
                # 
                # m=j, k-m=j → k=2j: value = (1)(1) = 1
                # m=j, k-m=i → k=j+i: value = (1)(-1) = -1
                # m=i, k-m=j → k=i+j: value = (-1)(1) = -1
                # m=i, k-m=i → k=2i: value = (-1)(-1) = 1
                # 
                # So second order = d^2 * [δ_{k,2j} - δ_{k,i+j} - δ_{k,i+j} + δ_{k,2i}]
                # = d^2 * [δ_{k,2j} - 2*δ_{k,i+j} + δ_{k,2i}]
                
                # Build the gain vector
                g_vec = np.empty(size)
                for k in range(size):
                    ki = k - i
                    kj = k - j
                    ai_ki = a[ki] if 0 <= ki < n else 0.0
                    ai_kj = a[kj] if 0 <= kj < n else 0.0
                    g_vec[k] = 2.0 * (ai_kj - ai_ki)
                
                # Add second-order corrections
                if 0 <= 2*j < size:
                    g_vec[2*j] += delta * 1.0  # will be d^2 * 1, but we're using linearized g
                # Actually for the quadratic, let's just use the linear approximation for now
                # and evaluate the new max directly
                
                # Try a few delta values
                n_samples = 15
                deltas = np.linspace(0.0, ai * 0.9, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g_vec
                    # Add second-order corrections
                    if 0 <= 2*i < size:
                        new_conv[2*i] += d * d  # (a[i]-d)^2 vs a[i]^2: -2*a[i]*d + d^2, but we already have the -2*a[i]*d in g_vec*2... 
                    # Hmm, the linear approximation might not be accurate enough.
                    # Let me just compute the correction terms properly.
                    # Actually, the linear approximation g_vec * d gives the first-order change.
                    # The second-order change is d^2 * [δ_{2j} - 2δ_{i+j} + δ_{2i}]
                    if 0 <= 2*i < size:
                        new_conv[2*i] += d * d
                    if 0 <= 2*j < size:
                        new_conv[2*j] += d * d
                    if i + j < size:
                        new_conv[i + j] -= 2.0 * d * d
                    
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a * S / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(5, n // 50)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.05 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * S / s_new
                # Also try a larger perturbation
                if no_improve >= 3:
                    n_perturb2 = max(10, n // 20)
                    idx_perturb2 = rng.choice(n, size=n_perturb2, replace=False)
                    a[idx_perturb2] *= (1.0 + 0.15 * rng.standard_normal(n_perturb2))
                    a = np.maximum(a, 0.0)
                    s_new = a.sum()
                    if s_new > 1e-15:
                        a = a * S / s_new
        else:
            no_improve = 0
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Using solver (fallback): ECOS")
    return cp.ECOS

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
