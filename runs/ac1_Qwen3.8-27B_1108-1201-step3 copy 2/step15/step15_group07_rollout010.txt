def build_conv_matrix_sparse(a):
    """Build the (2n-1) x n convolution matrix T where T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
        # So for row k, col j: value is a[k-j]
        # For col j, valid rows are k in [j, j+n-1]
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    # Build T as dense for simplicity when n is moderate
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = b_var.value
    if b_val is None:
        return None
    b_val = np.maximum(np.asarray(b_val, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def fourier_reweight(a0, time_limit, verbose=True):
    """
    Fourier-domain reweighting: iteratively suppress frequency bins
    that contribute most to the peak auto-correlation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute power spectrum
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2

        # Current auto-correlation
        conv = fast_autocorr(a)
        max_val = conv.max()
        max_idx = conv.argmax()

        # Find which frequency bins contribute to the peak
        # The auto-correlation at lag max_idx is:
        #   sum_k P[k] * exp(j*2*pi*k*max_idx/fft_size) (inappropriate for linear conv)
        # Instead, use a simpler heuristic: down-weight high-magnitude bins

        # Suppress top frequency bins (excluding DC)
        mag = np.abs(A)
        if max_val > best_score * 1.001:
            # Stuck - try perturbation
            noise = rng.normal(0, norm * 0.01, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert * norm / s_pert
            continue

        # Scale down top magnitude bins (except DC)
        n_suppress = max(3, n // 50)
        idx = np.argsort(mag)[::-1]
        # Skip DC (index 0)
        idx = idx[idx > 0]
        idx = idx[:n_suppress]

        # Gentle suppression
        for k in idx:
            A[k] *= 0.999
            A[fft_size - k] *= 0.999 if k < fft_size - k else 0.999

        # Invert
        a_new = np.real(irfft(A, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new * norm / s_new

        sc = fast_autocorr(a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FR it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def rabinowitz_window(n, alpha=3.0):
    """Generate a Rabinowitz-weighted window."""
    i = np.arange(n)
    t = 2.0 * (i - (n - 1) / 2.0) / (n - 1)
    w = 1.0 - np.abs(((1 + alpha) * t ** 4 - (2 + 2 * alpha) * t ** 2 + 1) /
                     (alpha * t ** 4 - (1 + alpha) * t ** 2 + 1) + 1e-30)
    w = 1.0 - np.abs(w)  # This gives the Rabinowitz shape
    w = np.abs(w)
    w = np.maximum(w, 1e-6)
    return w


def hamming_window(n):
    if n == 1:
        return np.array([1.0])
    return 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))


def blackman_window(n):
    if n == 1:
        return np.array([1.0])
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    m = np.arange(n)
    return a0 - a1 * np.cos(2 * np.pi * m / (n - 1)) + a2 * np.cos(4 * np.pi * m / (n - 1))

        max_val = conv.max()

        # Subgradient: for the active lag, the gradient contribution
        # is the "correlated" version of a
        active_idx = max_val  # just use the max lag
        # Compute gradient at the peak lag
        peak_k = conv.argmax()
        # The subgradient of max(conv) w.r.t. a is the a-shifted at the peak
        grad = np.zeros(n)
        # conv[k] = sum_i a[i]*a[i+k-n+1] (for linear conv, need care)
        # For the peak at position p in the (2n-1) length conv:
        # d/d(a[j]) of conv[p] = a[p-j] (if 0<=p-j<n) + a[j-(p-n+1)] (if 0<=j-(p-n+1)<n)
        # Simplified: grad[j] = a[p-j] + a[j-(p-n+1)] (where valid)
        for j in range(n):
            v1 = p = peak_k
            if 0 <= v1 - j < n:
                grad[j] += a[v1 - j]
            v2 = peak_k - (n - 1)
            if 0 <= j - v2 < n:
                grad[j] += a[j - v2]

        # Also subtract the "sum" gradient component
        # d/d(a[j]) of 1/sum_a^2 = -1/sum_a^3 * sum_a = -1/sum_a^2 (times the active part)
        # Simplified: just use the raw subgradient with step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Multiplicative Weights Update (mirror descent on simplex)
        a_new = a * (1.0 - eta * grad / (grad.max() + 1e-15))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new * norm / s_new

        sc = fast_autocorr(a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 200 == 0:
                eta *= 0.9
            if patience > 500:
                noise = rng.laplace(0, norm * 0.005, n)
                a = np.maximum(a + noise, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                patience = 0
                eta = 1.0 / n

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent: move mass between pairs of positions to reduce peak.
    Uses efficient delta computation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(27182)
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve > 100:
            break

        pass_improved = False
        perm = rng.permutation(min(n, 200))  # Only sample 200 positions per pass

        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = idx
            if a[i] < norm * 1e-10:
                continue

            # Try moving to a few nearby positions
            candidates = [(i + 1) % n, (i - 1) % n, (i + 2) % n,
                          (i - 2) % n, (i + 5) % n, (i - 5) % n,
                          (i + n // 4) % n, (i - n // 4) % n]

            for j in candidates:
                if j == i or a[i] < norm * 1e-10:
                    continue

                # Compute change in convolution from moving small mass d from i to j
                # New conv = old conv + d*(delta_ij)
                # where delta_ij[p] = a[p-i+n-1] + a[p-j+n-1] - a[p-j] - a[p-i]
                # (approximate, for linear conv)
                # Exact: the change at position p is:
                #   +a[p-i] where valid - a[p-j] where valid +a[p-i] ...
                # Let's just try a few step sizes with full recomputation (sampled)

                for frac in [0.1, 0.3, 0.5]:
                    d = a[i] * frac
                    a[i] -= d
                    a[j] += d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += d
                        a[j] -= d
                        break
                    # Don't renormalize for the test - just check max
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score * 0.9999:
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            break
                    else:
                        a[i] += d
                        a[j] -= d

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                # Perturb
                noise = rng.normal(0, norm * 0.01, n)
                a = np.maximum(a + noise, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                no_improve = 0
                if no_improve > 20:
                    break

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Tries multiple strategies and returns the best sequence found.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Default initialization: uniform
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
        n_base = 1600
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov on parent (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Fourier reweighting (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.10
    print(f"\n=== Phase 2: Fourier Reweight n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fourier_reweight(a_norm, t2, verbose=True)
        do_try_update(a_out, f"fr_n{n_best}")

    # === Phase 3: LP Nesterov on best (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LP Nesterov R2 n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # === Phase 4: Coordinate descent polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: CD polish n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t4, verbose=True)
        do_try_update(a_out, f"cd_n{n_best}")

    # === Phase 5: Try different n values (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Multi-n exploration (budget={t5:.0f}s) ===")
    if t5 > 20:
        n_candidates = [n_base, int(n_base * 1.25), int(n_base * 1.5), int(n_base * 0.75)]
        n_candidates = [x for x in n_candidates if 400 <= x <= 6000]
        n_candidates = list(set(n_candidates))
        time_per_n = t5 / max(1, len(n_candidates))
        for nc in n_candidates:
            if time.monotonic() - t_global > time_budget - 30:
                break
            if nc == n_base:
                continue
            # Initialize with window function
            w = hamming_window(nc)
            a_init = w / w.sum()
            a_norm = a_init * np.sqrt(2.0 * nc)
            t_sub = min(time_per_n, time_budget - (time.monotonic() - t_global))
            if t_sub < 5:
                break
            a_out, _ = nesterov_lp_optimize(a_norm, t_sub, solver, verbose=False)
            do_try_update(a_out, f"lp_n{nc}")

    # === Phase 6: Subgradient polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = remaining * 0.05
    print(f"\n=== Phase 6: Subgradient n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t6, verbose=False)
        do_try_update(a_out, f"sg_n{n_best}")

    # === Phase 7: Final LP polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_best = len(best_seq)
        t7 = remaining * 0.8
        print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Returning default.")
        return [1.0 / 1600] * 1600

    return [float(x) for x in best_seq]
