```python
def fast_best_response(a, M, active_set, max_iters=1000, tol=1e-6):
    """
    Find b >= 0 maximizing sum(b) s.t. (a*b)[k] <= M for k in active_set.
    Uses ADMM with POCS for the projection step.
    """
    n = len(a)
    K = active_set
    nK = len(K)
    
    # Precompute constraint vectors
    # For constraint k: row_k[j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    norms_sq = []
    for k in K:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            row[j] = a[k - j]
        rows.append(row)
        norms_sq.append(np.dot(row, row))
    
    rows = np.array(rows)  # shape (nK, n)
    norms_sq = np.array(norms_sq)
    
    # ADMM
    rho = 1.0
    x = np.ones(n) / n  # primal variable
    z = x.copy()  # dual variable (in constraint space)
    u = np.zeros(n)  # scaled dual
    
    best_x = x.copy()
    best_sum = x.sum()
    
    for it in range(max_iters):
        # x-update: minimize (rho/2)||x - z + u||^2 s.t. x >= 0
        x = np.maximum(z - u, 0)
        
        # z-update: project (x + u) onto intersection of halfspaces {row_k . z <= M}
        z = x + u
        for _pocs in range(10):  # POCS iterations
            for i in range(nK):
                val = np.dot(rows[i], z)
                if val > M:
                    if norms_sq[i] > 1e-15:
                        z -= (val - M) / norms_sq[i] * rows[i]
        
        # u-update
        u += x - z
        
        # Check convergence
        if max(rows @ z) <= M * (1 + 1e-8):
            break
        
        # Track best feasible solution
        # x is always feasible (x >= 0), check constraint satisfaction
        max_viol = (rows @ x).max() if nK > 0 else 0
        if max_viol <= M * (1 + 1e-6) and x.sum() > best_sum:
            best_sum = x.sum()
            best_x = x.copy()
    
    return best_x, best_sum
```

            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Bussgang: best={best_score:.10f}, it={it}, alpha={alpha_curr:.4f}, t={elapsed:.0f}s")
        else:
            # Try with smaller alpha
            if alpha_curr > alpha * 0.5:
                alpha_curr *= 0.8
            else:
                # Try adding small noise to escape
                noise = rng.standard_normal(n) * 0.001
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_p = a_pert.sum()
                if s_p > 1e-15:
                    a_new = a_pert / s_p
                    score = fast_autocorr(a_new).max()
                    if score < best_score:
                        best_score = score
                        best_a = a_new.copy()
                        a = a_new
                    else:
                        # Stuck, break
                        if it > 1000:
                            break

    return best_a, best_score


def smooth_gradient_optimize(a, time_limit, beta=200.0, verbose=True, seed=0):
    """
    Optimize using log-sum-exp smooth approximation of max autocorrelation.
    f_beta(a) = (1/beta) * log(sum_k exp(beta * (a*a)[k]))
    Uses accelerated projected gradient with adaptive beta.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_autocorr(a)
    size = 2 * n - 1
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()

    # Nesterov momentum
    y = a.copy()
    y_prev = a.copy()
    t_iter = 1
    step_base = 1.0 / (2.0 * n * np.max(a))
    stall = 0
    max_stall = 200

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        # Compute autocorrelation at y
        conv_y = fast_autocorr(y)
        M_y = conv_y.max()

        # Softmax weights (log-sum-exp trick)
        shifted = beta * (conv_y - M_y)
        exp_shifted = np.exp(shifted)
        p = exp_shifted / exp_shifted.sum()

        # Compute gradient: grad[j] = sum_k p[k] * (a[k-j] + a[j-k])
        # Use the top 50% of weights for efficiency
        n_active = min(size, max(20, size // 8))
        active_idx = np.argsort(p)[-n_active:]
        p_active = p[active_idx]
        p_active = p_active / p_active.sum()

        grad = np.zeros(n)
        for idx in range(n_active):
            k = active_idx[idx]
            w = p_active[idx]
            # Contribution: w * (a[k-j] + a[j-k])
            # a[k-j] valid when 0 <= k-j < n
            lo1 = max(0, k - n + 1)
            hi1 = min(n, k + 1)
            if lo1 < hi1:
                grad[lo1:hi1] += w * a[k - np.arange(lo1, hi1)]
            lo2 = max(0, k)
            hi2 = min(n, k + n)
            if lo2 < hi2:
                grad[lo2:hi2] += w * a[np.arange(lo2, hi2) - k]

        # Adaptive step size with backtracking
        step = step_base
        a_curr = a
        for _bs in range(50):
            a_trial = a_curr - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                step *= 0.5
                continue
            a_trial = a_trial / s_trial
            score_trial = fast_autocorr(a_trial).max()
            if score_trial < M_y:
                break
            step *= 0.5

        if step < 1e-15:
            break

        a_new = a_curr - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Nesterov update
        t_plus = (1.0 + math.sqrt(1.0 + 4.0 * t_iter * t_iter)) / 2.0
        beta_mom = (t_iter - 1.0) / t_plus
        beta_mom = min(beta_mom, 0.95)
        y_new = a_new + beta_mom * (a_new - y_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-10:
            y_new = y_new / s_y
        else:
            y_new = a_new.copy()

        y_prev = y
        y = y_new
        a = a_new
        t_iter += 1

        # Track best
        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-13:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SmoothGrad: best={best_score:.10f}, it={it}, t={elapsed:.0f}s")
        else:
            stall += 1

        # Also check y
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-13:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SmoothGrad(y): best={best_score:.10f}, it={it}, t={elapsed:.0f}s")

        # Increase beta over time for sharper max approximation
        if it % 500 == 0:
            beta = min(beta * 1.05, 2000.0)

    return best_a, best_score


def multi_peak_subgradient(a, time_limit, verbose=True, seed=0):
    """
    Subgradient method using weighted combination of top peaks.
    More targeted than smooth gradient - directly attacks the maximum.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    step = 0.1 / n
    patience = 0
    max_patience = 1000
    n_peaks = max(5, min(30, n // 64))

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get top peaks with exponential weighting
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-20.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.001, 1e-10))
        weights = weights / weights.sum()

        # Combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            lo1 = max(0, k - n + 1)
            hi1 = min(n, k + 1)
            if lo1 < hi1:
                grad[lo1:hi1] += w * a[k - np.arange(lo1, hi1)]
            lo2 = max(0, k)
            hi2 = min(n, k + n)
            if lo2 < hi2:
                grad[lo2:hi2] += w * a[np.arange(lo2, hi2) - k]

        # Normalized step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a_trial = a - step * grad / gnorm
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-10:
            step *= 0.5
            patience += 1
            continue
        a_trial = a_trial / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 1.0)
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MultiPeak: best={best_score:.10f}, it={it}, step={step:.6f}, t={elapsed:.0f}s")
        else:
            patience += 1
            step *= 0.97
            if step < 1e-12:
                break

        # Periodic random peak attack
        if it % 100 == 0 and len(top_idx) > 1:
            k_rand = top_idx[rng.integers(len(top_idx))]
            grad_r = np.zeros(n)
            lo1 = max(0, k_rand - n + 1)
            hi1 = min(n, k_rand + 1)
            if lo1 < hi1:
                grad_r[lo1:hi1] += a[k_rand - np.arange(lo1, hi1)]
            lo2 = max(0, k_rand)
            hi2 = min(n, k_rand + n)
            if lo2 < hi2:
                grad_r[lo2:hi2] += a[np.arange(lo2, hi2) - k_rand]

            gr_norm = np.linalg.norm(grad_r)
            if gr_norm > 1e-15:
                a_r = a - step * 0.3 * grad_r / gr_norm
                a_r = np.maximum(a_r, 0.0)
                sr = a_r.sum()
                if sr > 1e-10:
                    a_r = a_r / sr
                    sr_score = fast_autocorr(a_r).max()
                    if sr_score < best_score - 1e-13:
                        a = a_r
                        best_score = sr_score
                        best_a = a_r.copy()
                        patience = 0

    return best_a, best_score


def coordinate_descent_fast(a, time_limit, verbose=True):
    """
    Fast coordinate descent using FFT-based autocorrelation updates.
    For each coordinate j, try a few candidate values and pick the best.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        # Current autocorrelation
        conv = fast_autocorr(a)
        current_score = conv.max()

        # Process coordinates in random order for diversity
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-10:
                continue

            # Compute the "cross terms" for lag k: how much a[j] contributes
            # (a*a)[k] includes: a[j]*a[k-j] + a[k-j]*a[j] (if both valid and j != k-j)
            #                 + a[j]^2 (if k = 2j)
            # For changing a[j] from aj to x:
            # new_conv[k] = conv[k] - 2*aj*c_k + x*(2*c_k) + (x^2 - aj^2)*d_k
            # where c_k = a[k-j] (if 0<=k-j<n and k-j!=j) + a[j-k] (if 0<=j-k<n and j-k!=j)
            #       d_k = 1 if k==2j else 0

            # Efficient computation:
            # c_k = a[k-j] (valid: 0<=k-j<n, k-j != j) + a[j-k] (valid: 0<=j-k<n, j-k != j)
            # This is the "dictionary" shifted by j

            # Try a few candidate values
            candidates = [aj * 0.5, aj * 0.75, aj * 1.25, aj * 1.5, aj * 0.25, 0.0]
            # Add value that equalizes with current max
            # We want conv[k*] reduced, so try reducing a[j]
            candidates.append(aj * 0.6)
            candidates.append(aj * 0.8)

            best_x = aj
            best_obj = current_score

            for x in candidates:
                if x < 0 or x > 10.0:
                    continue
                dx = x - aj
                if abs(dx) < 1e-12:
                    continue

                # Quick estimate: the change in autocorrelation
                # new_conv[k] = conv[k] + dx * (a[k-j] + a[j-k]) + dx^2 * delta(k,2j)
                #             + dx * (a[k-j] + a[j-k]) for the "cross" part
                # More precisely:
                # Terms involving a[j]: a[j]*a[k-j] + a[j-k]*a[j] (when valid)
                # After change: x*a[k-j] + a[j-k]*x
                # Delta: (x-aj)*a[k-j] + (x-aj)*a[j-k] = dx*(a[k-j]+a[j-k])
                # Plus: if k==2j, there's an additional x^2 - aj^2 = dx*(x+aj)

                # For efficiency, just compute the max change
                # The worst lag is likely near the current peak
                # Approximate: new_max ≈ max(conv) + dx * max(|a[k*-j] + a[j-k*]|)
                # This is crude; let's just do it properly for a few key lags

                # Compute new autocorrelation efficiently
                # new_conv = conv + dx * cross_j + dx^2 * delta_2j
                # where cross_j[k] = a[k-j] (if valid) + a[j-k] (if valid), excluding the a[j]*a[j] term
                # Actually: cross_j[k] = (a[k-j] if 0<=k-j<n) + (a[j-k] if 0<=j-k<n)
                # But we need to subtract aj from each to get the "cross" (not including a[j] itself)
                # Hmm, let me just recompute for the top few lags

                # For speed, estimate the new max:
                k_star = int(np.argmax(conv))
                # Change at k_star:
                c_star = 0.0
                i1 = k_star - j
                if 0 <= i1 < n and i1 != j:
                    c_star += a[i1]
                i2 = j - k_star
                if 0 <= i2 < n and i2 != j:
                    c_star += a[i2]
                d_star = 1.0 if k_star == 2 * j else 0.0

                est_delta = dx * c_star + dx * (x + aj) * d_star
                est_new_max = max(0, conv[k_star] + 2 * dx * c_star / 2 + dx * c_star)

                # This is getting complicated; let's just try the update and check
                a[j] = x
                new_sum = a.sum()
                if new_sum < 1e-10:
                    a[j] = aj
                    continue
                # Scale to maintain sum=1
                scale = 1.0 / new_sum
                a_scaled = a * scale
                new_conv = fast_autocorr(a_scaled)
                new_max = new_conv.max()

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
                    a[j] = x
                else:
                    a[j] = aj

            if best_x != aj:
                s_new = a.sum()
                if s_new > 1e-10:
                    a = a / s_new
                    a_score = fast_autocorr(a).max()
                    if a_score < best_score - 1e-12:
                        best_score = a_score
                        best_a = a.copy()
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD: best={best_score:.10f}, pass={n_passes}, t={elapsed:.0f}s")

    return best_a, best_score


def decenter_and_recenter(a, epsilon=0.02):
    """
    Move some mass from the center to the edges to reduce center autocorrelation peak.
    """
    n = len(a)
    a_new = a.copy()
    center = n // 2
    width = n // 4

    # Reduce center mass
    center_mass = a_new[center - width:center + width].sum()
    if center_mass < 1e-10:
        return a_new

    # Spread mass to edges
    edge_idx_left = list(range(0, n, max(1, n // 20)))[:10]
    edge_idx_right = list(range(n - 1, n // 2, -max(1, n // 20)))[:10]
    all_edges = list(set(edge_idx_left + edge_idx_right))

    total_edge_mass = a_new[all_edges].sum()
    redistribution = center_mass * epsilon
    per_edge = redistribution / len(all_edges)
    center_reduction = redistribution / (2 * width)

    for i in all_edges:
        a_new[i] += per_edge
    for i in range(center - width, center + width):
        a_new[i] = max(0, a_new[i] - center_reduction)

    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def symmetrize(a):
    """Symmetrize: a = (a + reverse(a)) / 2."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym

                    a_out, _ = bussgang_equalize(a_interp, t7 * 0.5, alpha=0.03, verbose=False, seed=55)
                    try_update(a_out, f"bussgang_alt_n{n_alt}")
                    t7b = min(remaining * 0.15, 100.0)
                    if t7b > 20:
                        a_out, _ = smooth_gradient_optimize(a_out, t7b, beta=100.0, verbose=False, seed=66)
                        try_update(a_out, f"smooth_alt_n{n_alt}")
                # Also try the interpolation directly with subgradient
                t7c = min(remaining * 0.1, 80.0)
                if t7c > 10:
                    a_out, _ = multi_peak_subgradient(a_interp, t7c, verbose=False, seed=33)
                    try_update(a_out, f"multipk_alt_n{n_alt}")

    # === Phase 8: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.2, 150.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Final CD polish, n={n_cur}, budget={t8:.0f}s ===")
        a_out, _ = coordinate_descent_fast(best_seq, t8, verbose=True)
        try_update(a_out, f"cd_final_n{n_cur}")

    # === Phase 9: Final subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t9 = min(remaining * 0.3, 100.0)
    if t9 > 10:
        print(f"\n=== Phase 9: Final subgradient, n={n_cur}, budget={t9:.0f}s ===")
        a_out, _ = multi_peak_subgradient(best_seq, t9, verbose=False, seed=222)
        try_update(a_out, f"subg_final_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
