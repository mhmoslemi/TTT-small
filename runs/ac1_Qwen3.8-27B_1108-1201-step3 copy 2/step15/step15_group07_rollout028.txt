def active_set_cd(a0, time_limit, verbose=True):
    """
    Coordinate descent that focuses on reducing the active (max) lags.
    Uses the insight that we should move mass from positions that
    contribute most to the peak lags to positions that don't.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    rng = np.random.default_rng(999)
    
    # Precompute the contribution of each position to each lag
    # For position i, it contributes a[i]*a[j] to lag |i-j|
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve > 100:
            break
        
        conv = fast_autocorr(a)
        max_val = conv.max()
        # Find active lags (within 1% of max)
        threshold = max_val * 0.99
        active_mask = conv >= threshold
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            break
        
        # For each active lag, find which positions contribute most
        # and try to reduce their contribution
        
        improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-15:
                continue
            
            # Compute how much position i contributes to each active lag
            # Position i contributes to lag k via a[i]*a[k-i] and a[i]*a[i-k]
            # Try moving a small amount from i to a "better" position j
            # "Better" means j contributes less to active lags
            
            # Quick heuristic: try moving to random positions
            for _ in range(3):
                j = rng.integers(0, n)
                if j == i:
                    continue
                
                # Try a small transfer
                for frac in [0.1, 0.2, 0.3]:
                    d = a[i] * frac
                    a[i] -= d
                    a[j] += d
                    s = a.sum()
                    a_temp = a * norm / s
                    
                    new_conv = fast_autocorr(a_temp)
                    new_max = new_conv.max()
                    
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a_temp.copy()
                        conv = new_conv
                        a = a_temp
                        improved = True
                        no_improve = 0
                        break
                    else:
                        a[i] += d
                        a[j] -= d
            
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                # More aggressive perturbation
                noise = rng.normal(0, norm * 0.01, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                no_improve = 0
        
        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ASD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def nesterov_with_varying_n(a0, time_limit, solver, n_values, verbose=True):
    """Try different n values by downsampling/upscaling."""
    best_a = np.asarray(a0, dtype=np.float64).copy()
    best_score = fast_autocorr(best_a).max() / (best_a.sum()**2 / (2*len(best_a)))
    # Actually let's just track the raw max and normalize properly
    best_score_raw = None  # We'll use evaluate_safe
    
    t0 = time.monotonic()
    
    for n_target in n_values:
        if time.monotonic() - t0 > time_limit * 0.8:
            break
        
        # Create a sequence of length n_target from best_a
        if n_target == len(best_a):
            a_try = best_a.copy()
        else:
            # Interpolate
            a_try = np.interp(
                np.linspace(0, len(best_a)-1, n_target),
                np.arange(len(best_a)),
                best_a
            )
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try * np.sqrt(2.0 * n_target) / s
        
        # Optimize
        t_remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(t_remaining * 0.3, 120.0)
        if t_sub < 10:
            break
            
        a_out, sc = nesterov_lp_optimize_v2(a_try, t_sub, solver, verbose=False)
        
        # Evaluate
        a_out = np.maximum(a_out, 0)
        s_out = a_out.sum()
        if s_out > 1e-15:
            sc_eval = evaluate_safe(a_out / s_out)
            if np.isfinite(sc_eval):
                sc_norm = 2.0 * n_target * fast_autocorr(a_out).max() / s_out**2
                if sc_norm < best_score:
                    best_score = sc_norm
                    best_a = a_out.copy()
                    if verbose:
                        print(f"    MultiN n={n_target}: improved to {sc_norm:.8f}")
    
    return best_a, best_score

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def get_solver():

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def build_conv_matrix(a, lags):
    n = len(a)
    m = len(lags)
    T = np.zeros((m, n))
    for idx in range(m):
        k = int(lags[idx])
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_hi >= i_lo:
            j_lo = k - i_hi
            j_hi = k - i_lo
            T[idx, i_lo:i_hi + 1] = a[j_lo:j_hi + 1][::-1]
    return T


def solve_lp(a, M, solver, top_k=200):
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    k_actual = min(top_k, size)
    active_idx = np.argsort(conv)[-k_actual:]
    T = build_conv_matrix(a, active_idx)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    bv = b_var.value
    if bv is None:
        return None
    b_val = np.maximum(np.asarray(bv, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    patience = 0
    restart_count = 0
    momentum_t = 1.0
    rng = np.random.default_rng(54321)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        adaptive_k = min(80 + n_iters // 5, 350)
        b = solve_lp(y, M_curr, solver, top_k=adaptive_k)
        if b is None:
            stuck_count += 1
            if stuck_count > 25:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        if patience > 30 and n_iters % 15 == 0:
            noise_scale = norm * (0.005 + 0.01 * min(restart_count, 10))
            noise = rng.laplace(0, noise_scale, n)
            y = best_a + noise
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score


def smooth_adam(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = 0.008
    c_smooth = 80.0
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(777)

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()

        c_sched = min(c_smooth * (1.0 + it / 300.0), 400.0)
        shifted = conv - max_val
        exp_vals = np.exp(c_sched * shifted)
        exp_sum = np.sum(exp_vals)
        weights = exp_vals / exp_sum

        k_top = min(80, size)
        active_idx = np.argsort(weights)[-k_top:]

        grad = np.zeros(n)
        for idx in active_idx:
            k = int(idx)
            w_k = weights[idx] * 2.0
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_hi >= i_lo:
                j_lo = k - i_hi
                j_hi = k - i_lo
                grad[i_lo:i_hi + 1] += w_k * a[j_lo:j_hi + 1][::-1]

        n_iters += 1
        m_arr = beta1 * m_arr + (1 - beta1) * grad
        v_arr = beta2 * v_arr + (1 - beta2) * grad ** 2
        m_hat = m_arr / (1 - beta1 ** n_iters)
        v_hat = v_arr / (1 - beta2 ** n_iters)

        lr = base_lr / (1.0 + it / 500.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new

        new_max = fast_autocorr(a_new).max()
        if new_max < best_score - 1e-12:
            best_score = new_max
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            a = a_new
            patience += 1

        if patience > 150:
            noise = rng.laplace(0, norm * 0.03, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert * norm / s_pert
            m_arr = np.zeros(n)
            v_arr = np.zeros(n)
            patience = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, c={c_sched:.0f}")

    return best_a, best_score


def simulated_annealing(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)

    T_init = best_score * 0.05
    T_min = best_score * 0.001
    n_sweeps = max(1, int(time_limit / 0.5))
    t_current = t0

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        progress = (time.monotonic() - t0) / time_limit
        T = T_init * (T_min / T_init) ** progress

        n_moves = max(10, n // 50)
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                d = a[i] * frac
                if d < 1e-15:
                    continue
                a[i] -= d
                a[j] += d
                s = a.sum()
                if s < 1e-15:
                    a[i] += d
                    a[j] -= d
                    break
                a_trial = a * norm / s
                new_max = fast_autocorr(a_trial).max()

                delta = new_max - fast_autocorr(a).max()
                accept = False
                if delta < 0:
                    accept = True
                elif T > 1e-15:
                    if rng.random() < np.exp(-delta / T):
                        accept = True

                if accept:
                    a = a_trial
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                    break
                else:
                    a[i] += d
                    a[j] -= d
                    break

        if verbose and sweep % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA sweep={sweep} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.2e}")

    return best_a, best_score

            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [sa]: {best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Phase 4: LP Nesterov round 2 (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.20
    print(f"\n=== Phase 4: LP Nesterov R2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True)
        a_out = np.maximum(a_out, 0)
        s_out = a_out.sum()
        if s_out > 1e-15:
            a_out = a_out / s_out
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [lp2]: {best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Phase 5: Targeted Perturbation + LP (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Targeted Perturb n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_perturb_lp(a_norm, t5, solver, verbose=True)
        a_out = np.maximum(a_out, 0)
        s_out = a_out.sum()
        if s_out > 1e-15:
            a_out = a_out / s_out
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [tp]: {best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Phase 6: Multi-N exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        t6 = min(remaining * 0.4, 300.0)
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Multi-N (budget={t6:.0f}s) ===")
        a_out, _ = multi_n_search(best_seq, t6, solver, verbose=True)
        a_out = np.maximum(a_out, 0)
        s_out = a_out.sum()
        if s_out > 1e-15:
            a_out = a_out / s_out
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [multin]: {best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Phase 7: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t7 = min(remaining * 0.6, 120.0)
        print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
        if t7 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp(a_norm, t7, solver, verbose=False)
            a_out = np.maximum(a_out, 0)
            s_out = a_out.sum()
            if s_out > 1e-15:
                a_out = a_out / s_out
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score - 1e-9:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [final]: {best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
