        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si} T={T_val:.2f} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, active_fraction=0.5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    prev_y = a.copy()
    size = 2 * n - 1
    no_improve_count = 0
    last_full_check = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive active set: use fraction of constraint mass
        if t < 30:
            active = np.where(conv_y > M * 0.3)[0]
        else:
            # Use top constraints by value
            threshold = np.quantile(conv_y, 1.0 - active_fraction)
            active = np.where(conv_y >= threshold)[0]
            if len(active) < 100:
                active = np.where(conv_y > M * 0.5)[0]

        # Solve LP with active set
        b, duals = lp_solve(y, M, solver, active)
        if b is None:
            b, duals = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Full constraint check every 50 iterations
        if t - last_full_check >= 50:
            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * (1 + 1e-3):
                b_full, _ = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    g = b_full * norm / b_full.sum()
            last_full_check = t

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        # Dual-guided perturbation when stuck
        if no_improve_count in (80, 200, 400):
            if duals is not None:
                # Perturb in direction that reduces high-dual (peak) positions
                peak_weight = duals / (duals.sum() + 1e-30)
                # Create perturbation: reduce a at indices that contribute to peaks
                pert = np.zeros(n)
                fft_size_p = 1 << (size - 1).bit_length()
                Y = rfft(y, fft_size_p)
                D = rfft(peak_weight, fft_size_p)
                pert = irfft(np.conj(Y) * D, fft_size_p)[:n]
                pert = pert - pert.mean()
                scale = 0.1 * no_improve_count / 400.0
                y = y - scale * y * pert / (np.abs(pert).max() + 1e-30)
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y * norm / s_y
            else:
                n_perturb = max(20, n // 15)
                idx_perturb = np.random.default_rng(t).choice(n, size=n_perturb, replace=False)
                y[idx_perturb] *= (1.0 + 0.15 * np.random.default_rng(t + 1).standard_normal(n_perturb))
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y * norm / s_y
            no_improve_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.10f}, M={M:.6f}")

    return best_a, best_score

def spectral_pequalize(a0, time_limit, n_iters=200, verbose=False):
    """Equalize autocorrelation peaks via spectral domain adjustments."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        mean_c = conv[k:2*n-1-k].mean() if 0 < k < n else conv.mean()

        # Smooth target: reduce peaks, boost valleys
        k_smooth = max(5, n // 40)
        from scipy.ndimage import uniform_filter1d
        conv_smooth = uniform_filter1d(conv, size=k_smooth * 2 + 1)

        # Error signal: positive where peaks exceed target
        target = max_c * 0.95
        error = np.maximum(conv - target, 0.0)
        error_sum = error.sum()
        if error_sum < 1e-15:
            break

        # Gradient: how reducing a_i affects the peaks
        A = rfft(a, fft_size)
        E = rfft(error, fft_size)
        g1 = irfft(np.conj(A) * E, fft_size)[:n]
        g2 = irfft(A * E, fft_size)[:n]
        grad = g1 + g2

        # Gradient step with backtracking
        step = 0.5 / (it + 1.0)
        for trial in range(15):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < max_c:
                break
            step *= 0.5

        a = a - step * grad
        a = np.maximum(a, 0.0)
        s_new = a.sum()
        if s_new < 1e-15:
            a = a.copy()
        else:
            a = a * norm / s_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))

            peak_threshold = cur_max * 0.92
            peak_lags = np.where(conv >= peak_threshold)[0]
            for kl in peak_lags[:6]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                j_candidates.add(j1)
                j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 15
                deltas = np.linspace(0.01, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-14:
                    best_val = v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-14:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
                    conv = fast_conv_self(a)
                    cur_max = conv.max()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(10, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    else:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick LSE warm-up
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 90.0)
    print(f"\n=== Phase 1: LSE warm-up n={len(best_seq)} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=False)
        try_update(a_out, "lse_warm")

    # Phase 2: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True, active_fraction=0.4)
        try_update(a_out, f"lp_main_n{len(best_seq)}")

    # Phase 3: Spectral peak equalization
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 3: Spectral equalize n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = spectral_pequalize(best_seq, t3, n_iters=300, verbose=True)
        try_update(a_out, f"spec_n{len(best_seq)}")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, verbose=True, active_fraction=0.3)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 5: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Pair CD n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"pcd_n{len(best_seq)}")

    # Phase 6: LSE + spectral alternating
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LSE+Spec alternating n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    if remaining > 80:
        t6a = min(remaining * 0.35, 100.0)
        a_out, _ = lse_gradient_descent(best_seq, t6a, verbose=False)
        try_update(a_out, "lse_alt")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.4, 80.0)
            a_out, _ = spectral_pequalize(best_seq, t6b, n_iters=150, verbose=False)
            try_update(a_out, "spec_alt")

    # Phase 7: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 7: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t7 = min(remaining * 0.5, 180.0)
        a_out, _ = lp_nesterov_optimize(init_long, t7, solver, verbose=False, active_fraction=0.4)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and len(best_seq) == n_long:
            t7b = min(remaining * 0.5, 60.0)
            a_out2, _ = pair_cd_optimize(best_seq, t7b, rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 8: Final budget - try a different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = len(best_seq) + 800
    print(f"\n=== Phase 8: Alt length n={n_alt} (remaining={remaining:.0f}s) ===")
    if remaining > 80:
        init_alt = interpolate_seq(best_seq, n_alt)
        t8 = min(remaining * 0.6, 100.0)
        a_out, _ = lp_nesterov_optimize(init_alt, t8, solver, verbose=False, active_fraction=0.5)
        try_update(a_out, f"lp_alt_{n_alt}")

    # Final quick polish with whatever time is left
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.8, 40.0)
        a_out, _ = spectral_pequalize(best_seq, t9, n_iters=100, verbose=False)
        try_update(a_out, "final_spec")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
