def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):

    if S < 1e-15:

    max_b = fast_conv_self(a).max()

def build_toeplitz_row(a, k):
    """Return the k-th row of the Toeplitz matrix T where (T*b)_k = conv(a,b)_k."""
    n = len(a)
    size = 2 * n - 1
    row = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    for i in range(lo, hi):
        j = k - i
        if 0 <= j < n:
            row[i] = a[j]
    return row

def lp_best_response(a, M, solver, active_lags=None, tol=1e-8):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns optimal b and duals."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_lags is not None and len(active_lags) < size:
        constraints = []
        for k in active_lags:
            row = build_toeplitz_row(a, k)
            constraints.append(cp.sum(row * b_var) <= M)
    else:
        constraints = []
        for k in range(size):
            row = build_toeplitz_row(a, k)
            constraints.append(cp.sum(row * b_var) <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False, tol=tol)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)

    # Extract duals
    duals = np.zeros(size)
    try:
        if active_lags is not None and len(active_lags) < size:
            for idx, k in enumerate(active_lags):
                if idx < len(prob.constraints) and prob.constraints[idx].dual_value is not None:
                    duals[k] = abs(prob.constraints[idx].dual_value)
        else:
            for k in range(min(len(prob.constraints), size)):
                if prob.constraints[k].dual_value is not None:
                    duals[k] = abs(prob.constraints[k].dual_value)
    except Exception:
        pass

    return val, duals

def lp_best_response_batched(a, M, solver, active_lags=None):
    """More efficient LP using matrix formulation."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    # Build the constraint matrix efficiently
    if active_lags is not None and len(active_lags) < size:
        T = np.zeros((len(active_lags), n))
        for idx, k in enumerate(active_lags):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    T[idx, i] = a[j]
        constraints = [T @ b_var <= M]
    else:
        T = np.zeros((size, n))
        for k in range(size):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    T[k, i] = a[j]
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)

    duals = np.zeros(size)
    try:
        con = constraints[0]
        if con.dual_value is not None:
            dv = np.asarray(con.dual_value, dtype=np.float64)
            if active_lags is not None and len(active_lags) < size:
                for idx, k in enumerate(active_lags):
                    if idx < len(dv):
                        duals[k] = abs(dv[idx])
            else:
                duals[:len(dv)] = np.abs(dv)
    except Exception:
        pass

    return val, duals

def lse_gradient_descent(a0, time_limit, n_stages=10, verbose=False):
    """Multi-temperature LSE gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temps = np.geomspace(50.0, 0.3, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(200, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.02 * it))
            for trial in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score

def mirror_descent_lp(a0, time_limit, solver, verbose=True, use_multiplicative=True,

    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()  # Mirror descent iterate
    avg_y = a.copy()  # Polyak average
    no_improve = 0
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        n_iters += 1
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-14:
            break

        # Adaptive constraint screening
        if t < 30:
            threshold = 0.6
        elif t < 100:
            threshold = 0.75
        elif t < 500:
            threshold = 0.85
        else:
            threshold = screen_frac
        active = np.where(conv_y > M * threshold)[0]
        if len(active) < 20:
            active = np.where(conv_y > M * 0.5)[0]
        if len(active) < 5:
            active = np.where(conv_y >= M * 0.3)[0]

        # Solve LP best-response
        b, duals = lp_best_response_batched(y, M, solver, active)
        if b is None or b.sum() < 1e-15:
            # Fallback: full LP or simple approximation
            b, duals = lp_best_response_batched(y, M, solver, None)
            if b is None or b.sum() < 1e-15:
                # Last resort: put mass at highest-convocation positions
                b = np.zeros(n)
                for k in active[:10]:
                    lo = max(0, k - n + 1)
                    hi = min(n, k + 1)
                    for i in range(lo, hi):
                        j = k - i
                        if 0 <= j < n:
                            b[i] += y[j]
                bsum = b.sum()
                if bsum > 0:
                    b = b / bsum * norm
                else:
                    continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum  # Best response normalized to same sum

        # Verify constraint satisfaction (screening repair)
        if len(active) < size:

            alpha = min(alpha, 1.0)
            # Multiplicative update: y_new = y^{1-alpha} * g^alpha
            y_safe = np.maximum(y, 1e-30)
            g_safe = np.maximum(g, 1e-30)
            y_new = np.exp((1.0 - alpha) * np.log(y_safe) + alpha * np.log(g_safe))
        else:

        y_new = np.maximum(y_new, 0.0)

        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        s_avg = avg_y.sum()
        if s_avg > 1e-15:
            avg_y = avg_y * norm / s_avg

        y = y_new

        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score - 1e-12:

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair-swap coordinate descent for final polish."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Candidate j's: random + peak-driven
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            peak_lags = np.where(conv >= cur_max * 0.92)[0]
            for kl in peak_lags[:6]:
                j_candidates.add((kl - i) % n)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector for moving mass from i to j
                g = np.zeros(size)
                # Adding d to a[j]: conv increases by d*(a shifted by j) + d*(a shifted by -j)
                # Removing d from a[i]: conv decreases by d*(a shifted by i) + d*(a shifted by -i)
                # Plus second-order: d^2 at lags 2j and -2j (for the j,j term)
                # and -d^2 at lag 2i (for the i,i term)

                for offset in range(n):
                    # Contribution from a[j] + d interacting with a[offset]
                    lag1 = j + offset
                    if lag1 < size:
                        g[lag1] += a[offset]
                    lag2 = j - offset
                    if 0 <= lag2 < size:
                        g[lag2] += a[offset]
                    # Contribution from a[i] - d
                    lag3 = i + offset
                    if lag3 < size:
                        g[lag3] -= a[offset]
                    lag4 = i - offset
                    if 0 <= lag4 < size:
                        g[lag4] -= a[offset]

                # Try different deltas
                n_samples = 15
                deltas = np.linspace(0.05, 0.9, n_samples) * ai
                for d in deltas:
                    new_conv = conv + d * g
                    # Second-order corrections
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    if 2 * i < size:
                        new_conv[2 * i] -= d * d
                    v = new_conv.max()
                    if v < best_val - 1e-13:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        s2 = a.sum()
                        if s2 > 1e-15:
                            a = a * norm / s2

        if not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve == 5:
                n_perturb = max(10, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):

        a = np.maximum(a, 0.05)
    elif choice == 7:
        # W-shape
        a = np.abs(np.cos(2 * np.pi * x)) ** 1.2
        a = 0.2 + 0.8 * a
    elif choice == 8:
        # Two peaks at 1/4 and 3/4
        a = 0.5 * np.exp(-8.0 * (x - 0.25) ** 2) + 0.5 * np.exp(-8.0 * (x - 0.75) ** 2) + 0.1
    elif choice == 9:
        # Broad U-shape
        a = (x * (1 - x)) ** 0.3 + 0.5
    elif choice == 10:
        # Oscillatory with envelope
        a = (0.5 + 0.5 * np.cos(6 * np.pi * x)) * np.sin(np.pi * x) ** 0.5 + 0.05
    else:
        a = np.abs(np.sin(4 * np.pi * x)) ** 0.6 * np.sin(np.pi * x) + 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(12):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick LSE warm-up
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = min(remaining * 0.06, 90.0)
    if t0_phase > 20:
        print(f"\n=== Phase 0: LSE warm-up (budget={t0_phase:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, t0_phase, n_stages=8, verbose=False)
        try_update(a_out, "lse_warmup")

    # Phase 1: Main mirror descent with LP (multiplicative)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: Mirror Descent LP (multiplicative) n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = mirror_descent_lp(best_seq, t1, solver, verbose=True,
                                      use_multiplicative=True, screen_frac=0.85,
                                      restart_interval=120)
        try_update(a_out, f"md1_n{n1}")

    # Phase 2: LSE gradient descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 2: LSE polish n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t2, n_stages=12, verbose=False)
        try_update(a_out, f"lse_n{n2}")

    # Phase 3: Mirror descent (additive) for diversity
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.22, 300.0)
    print(f"\n=== Phase 3: Mirror Descent LP (additive) n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = mirror_descent_lp(best_seq, t3, solver, verbose=True,
                                      use_multiplicative=False, screen_frac=0.90,
                                      restart_interval=100)
        try_update(a_out, f"md2_n{n3}")

    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: Pair CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{n4}")

    # Phase 5: Final LSE + LSE
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.40, 200.0)
    print(f"\n=== Phase 5: Final LSE n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t5, n_stages=14, verbose=False)
        try_update(a_out, f"lse_final_n{n5}")

    # Phase 6: Another round of mirror descent from best
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    t6 = min(remaining * 0.40, 200.0)
    print(f"\n=== Phase 6: Mirror Descent final n={n6} (budget={t6:.0f}s) ===")
    if t6 > 30:
        a_out, _ = mirror_descent_lp(best_seq, t6, solver, verbose=False,
                                      use_multiplicative=True, screen_frac=0.92,
                                      restart_interval=80)
        try_update(a_out, f"md3_n{n6}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        # Quick PCD
        a_out, _ = pair_cd_optimize(best_seq, remaining * 0.6, rng, verbose=False)
        try_update(a_out, "pcd_final")

    # Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100 and len(best_seq) <= 2000:
        n_long = min(len(best_seq) + 400, 2400)
        print(f"\n=== Phase 7: Length extension to n={n_long} (remaining={remaining:.0f}s) ===")
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.6, 150.0)
        a_out, _ = mirror_descent_lp(init_long, t_long, solver, verbose=False,
                                      use_multiplicative=True, screen_frac=0.85,
                                      restart_interval=100)
        try_update(a_out, f"long_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and len(best_seq) == n_long:
            a_out2, _ = lse_gradient_descent(best_seq, remaining * 0.7, n_stages=8, verbose=False)
            try_update(a_out2, f"lse_long_{n_long}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
