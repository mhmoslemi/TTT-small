def log_adam_optimize(a0, time_limit, verbose=True):

    n = len(a0)
    norm = np.sqrt(2.0 * n)

    v = np.zeros(n)

    beta2 = 0.999

            break

        steps_per_stage = max(20, min(100, int(remaining * 8 / max(1, n_stages - si))))
        lr = 0.3 / np.sqrt(n)
        
        for it in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            a = np.exp(log_a)
            a = np.maximum(a, 1e-15)
            a = a * norm / a.sum()
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:

            # g = gradient of LSE(conv, T) with respect to log_a
            # d/d(log_a_i) = d/d(a_i) * a_i

            # d(conv_k)/d(a_i) = 2 * conv_cross_indicator(k, i)
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)

            w = exp_c / (exp_c.sum() + 1e-30)

            # d LSE / d a_i = sum_k w[k] * d(conv[k])/d(a_i)
            # d(conv[k])/d(a_i) = a[k-i] (if valid) + a[k+i-n+1]... 
            # Actually: conv[k] = sum_j a[j]*a[k-j], so d(conv[k])/d(a_i) = a[k-i] (if 0<=k-i<n) + a[i-k+n-1]... 
            # More precisely: d(conv[k])/d(a_i) = sum over ways a[i] appears in conv[k]
            # a[i] appears in conv[k] when j=i (giving a[k-i]) or k-j=i (giving a[k-i])

            # Actually, conv[k] = sum_j a[j]*a[k-j]. The term with a[i] is:
            # - j=i: a[i]*a[k-i] (if 0<=k-i<n)
            # - k-j=i: a[j]*a[i] where j=k-i (same as above, but this is the same term)
            # Wait, for self-convolution: conv[k] = sum_j a[j]*a[k-j]
            # d(conv[k])/d(a_i) = a[k-i] (from j=i term) + a[k-i] (from k-j=i term, i.e., j=k-i)
            # Only if both indices are valid. If k-i is in [0,n-1]:
            #   j=i gives a[k-i], and j=k-i gives a[i]...

            grad_log = grad_a * a  # chain rule: d/da * da/d(log_a) = d/da * a

            v = beta2 * v + (1.0 - beta2) * grad_log * grad_log

            v_hat = v / (1.0 - beta2 ** t_adam)

def spectral_peak_shave(a0, time_limit, verbose=True):

    Gerchberg-Saxton-inspired algorithm:

    The target spectrum is chosen to minimize time-domain peaks.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    # Target: we want |A|^2 to be as flat as possible
    # But DC bin is fixed at sum(a)^2, so we can't make it flat
    # Instead, we minimize the variance of the non-DC bins
    
    n_iters = max(100, int(time_limit * 20))
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compute target magnitude
        # We want to reduce the dynamic range
        # Use geometric mean as target (preserves energy approximately)
        # But keep DC fixed
        dc = mag[0]
        non_dc = mag[1:]
        
        # Target: reduce peaks in magnitude
        # Use a "soft-clip" approach
        target_mean = np.mean(non_dc)
        target_max = np.max(non_dc)
        
        # Soft clip: reduce magnitudes above target_mean * factor
        # This is like a "water filling" in the frequency domain
        clip_factor = 2.0  # clip at 2x the mean
        target_mag = mag.copy()
        target_mag[1:] = np.minimum(mag[1:], target_mean * clip_factor)
        
        # But we need to preserve the overall energy relationship
        # The peak of the time-domain convolution is related to the 
        # concentration of energy in the frequency domain
        
        # Apply the target magnitude
        new_A = target_mag * np.exp(1j * phase)
        a_new = irfft(new_A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        s = a_new.sum()
        if s < 1e-15:
            break

        if sc < best_score:

            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 20 == 0:
            print(f"    Spectral iter {it}: best={best_score:.8f}")
    
    return best_a, best_score
