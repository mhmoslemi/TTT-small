def fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):

    s = a.sum()

def softmax_grad_obj(theta, beta, n):
    """
    Compute the smoothed objective and its gradient w.r.t. theta.
    a = S * softmax(theta), S = sqrt(2n)
    Objective: (1/beta)*log(sum_k exp(beta * (a*a)[k]))
    """
    S = np.sqrt(2.0 * n)
    S2 = 2.0 * n
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Compute p = softmax(theta)
    theta_shift = theta - theta.max()
    exp_t = np.exp(theta_shift)
    Z = exp_t.sum()
    if Z < 1e-300:
        p = np.ones(n) / n
    else:
        p = exp_t / Z

    a = S * p

    # Compute convolution
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]

    # Smoothed max via log-sum-exp
    bc = beta * conv
    bc_shift = bc - bc.max()
    max_val = conv.max() + np.log(np.sum(np.exp(bc_shift))) / beta
    obj = max_val

    # Compute weights w_k = exp(beta*c_k) / sum(exp(beta*c_j))
    w = np.exp(bc_shift) / np.sum(np.exp(bc_shift))

    # Compute H_i = 2*S^2 * sum_k w[k] * p[k-i]
    # This is 2*S^2 * cross_correlation(w, p) at lags 0..n-1
    # cross_corr[i] = sum_k w[k] * p[k-i]
    # Using FFT: cross_corr = ifft(conj(fft(p)) * fft(w))
    corr_fft_size = 1 << (size + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, corr_fft_size)
    W = np.fft.rfft(w, corr_fft_size)
    cross_corr = np.fft.irfft(np.conj(P) * W, corr_fft_size)[:n]
    H = 2.0 * S2 * cross_corr

    # Gradient w.r.t. theta: grad_j = p_j * (H_j - dot(H, p))
    Hp = np.dot(H, p)
    grad = p * (H - Hp)

    return obj, grad

def build_active_constraints(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

def lp_max_sum(a, M, solver, active_idx=None):

    try:

    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val

    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == normalize_sum, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0, alpha_min=0.08, beta_blend=0.25):
    """Nesterov-accelerated LP iteration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 60

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set - more aggressive
        if active_set is None or t % 30 == 0:
            tol = 0.05 + 0.03 * np.exp(-t / 200.0)
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 10:
                active_set = np.where(conv_y >= M_curr * 0.75)[0]
            if len(active_set) > 0.85 * size:
                active_set = np.where(conv_y >= M_curr * 0.92)[0]

        # Max-sum LP
        b_max = lp_max_sum(y, M_curr, solver, active_set)

        # Min-max LP (blending)
        b_min = None
        if t % 4 == 0:
            b_min, _ = lp_min_max(y, solver, active_set, normalize_sum=norm)

        if b_max is None:
            if t % 15 == 0:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        if b_min is not None and b_min.sum() > 1e-10:
            g_min_norm = b_min * norm / b_min.sum()
            g = (1.0 - beta_blend) * g + beta_blend * g_min_norm

        # Check violations
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                T_check = build_active_constraints(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 1e-5))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        # Nesterov update
        alpha_t = max(2.0 / (t + 2.0), alpha_min)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.92)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 15 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best_conv={best_score:.10f}")

    return best_a, best_score


def coord_descent(a0, time_limit, rng, verbose=True):
    """Coordinate descent with golden section search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    phi = 0.6180339887498949

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            lo = 0.0
            hi = ai * 2.5 + norm * 0.05

            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            a_test1 = a.copy()
            a_test1[i] = c1
            s1 = a_test1.sum()
            f1 = fast_autocorr(a_test1 * norm / s1).max() if s1 > 1e-15 else float('inf')

            a_test2 = a.copy()
            a_test2[i] = c2
            s2 = a_test2.sum()
            f2 = fast_autocorr(a_test2 * norm / s2).max() if s2 > 1e-15 else float('inf')

            for _ in range(25):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a_test1 = a.copy()
                    a_test1[i] = c1
                    s1 = a_test1.sum()
                    f1 = fast_autocorr(a_test1 * norm / s1).max() if s1 > 1e-15 else float('inf')
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a_test2 = a.copy()
                    a_test2[i] = c2
                    s2 = a_test2.sum()
                    f2 = fast_autocorr(a_test2 * norm / s2).max() if s2 > 1e-15 else float('inf')

            optimal_val = min(f1, f2)
            a_test_opt = a.copy()
            a_test_opt[i] = c1 if f1 < f2 else c2
            s_opt = a_test_opt.sum()
            if s_opt > 1e-15:
                a_test_opt = a_test_opt * norm / s_opt
                sc = fast_autocorr(a_test_opt).max()
                if sc < best_score - 1e-13:
                    a = a_test_opt
                    best_score = sc
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best_conv={best_score:.10f}")

    return best_a, best_score


def pair_swap_cd(a0, time_limit, rng, verbose=True):
    """Pair swap coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx % n]
            j = perm[(idx + 1) % n]
            if i == j:
                continue
            ai = a[i]
            aj = a[j]
            if ai < 1e-12 * norm and aj < 1e-12 * norm:
                continue

            best_delta = 0.0
            best_val = fast_autocorr(a).max()

            max_d = max(ai, aj) * 0.5
            for frac in [0.05, 0.15, 0.3, 0.5, 0.7, 0.9]:
                d = max_d * frac
                if d <= 0 or d > ai:
                    continue
                a_try = a.copy()
                a_try[i] = ai - d
                a_try[j] = aj + d
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    val = fast_autocorr(a_try).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = d

                if d <= aj:
                    a_try2 = a.copy()
                    a_try2[i] = ai + d
                    a_try2[j] = aj - d
                    s_try2 = a_try2.sum()
                    if s_try2 > 1e-15:
                        a_try2 = a_try2 * norm / s_try2
                        val2 = fast_autocorr(a_try2).max()
                        if val2 < best_val - 1e-14:
                            best_val = val2
                            best_delta = -d

            if abs(best_delta) > 1e-14:
                a_try = a.copy()
                if best_delta > 0:
                    a_try[i] -= best_delta
                    a_try[j] += best_delta
                else:
                    a_try[i] += abs(best_delta)
                    a_try[j] -= abs(best_delta)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    sc = fast_autocorr(a_try).max()
                    if sc < best_score - 1e-14:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best_conv={best_score:.10f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_best_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        for sty in range(8):
            cand = structured_init(n_init, rng, sty)
            sc = evaluate_safe(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, delta={old_score-sc:.6f})")

    # === Phase 1: Smoothed L-BFGS with beta annealing (40% time) ===
    n_base = len(best_seq)
    print(f"\n=== Phase 1: Smoothed L-BFGS n={n_base} ===")
    t1 = min((time_budget - (time.monotonic() - t_global)) * 0.40, 600.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        beta_sched = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0, 500.0, 800.0]
        a_out, _ = smoothed_lbfgs(a_norm, t1, beta_schedule=beta_sched, verbose=True, seed=0)
        try_update(a_out, "slbfgs")

        # Second round with perturbation
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            pert = perturb_seq(best_seq, rng, scale=0.03)
            n_cur = len(pert)
            a_norm2 = pert * np.sqrt(2.0 * n_cur) / pert.sum()
            t1b = min(remaining * 0.3, 120.0)
            if t1b > 20:
                beta_sched2 = [20.0, 50.0, 100.0, 200.0, 400.0]
                a_out2, _ = smoothed_lbfgs(a_norm2, t1b, beta_schedule=beta_sched2, verbose=True, seed=42)
                try_update(a_out2, "slbfgs_pert")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 2: LP Nesterov refinement (25% time) ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True, seed=0)
        try_update(a_out, f"lp_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 3: Pair CD polish (10% time) ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Pair CD n={n_cur} ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_swap_cd(a_norm, t3, rng, verbose=True)
        try_update(a_out, f"pcd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 4: Coordinate descent (8% time) ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Coord Descent n={n_cur} ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t4, rng, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Try longer length (10% time) ===
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 5: Long length n={n_long} ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        init_long = interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=False, seed=42)
        try_update(a_out, f"lp_long_n{n_long}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Second LP Nesterov with different params (10% time) ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LP Nesterov v2 n={n_cur} ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=False, seed=777, alpha_min=0.05, beta_blend=0.15)
        try_update(a_out, f"lp2_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Final smoothed L-BFGS polish (10% time) ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final S-LBFGS n={n_cur} ===")
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        beta_sched_final = [50.0, 100.0, 200.0, 400.0, 700.0]
        a_out, _ = smoothed_lbfgs(a_norm, t7, beta_schedule=beta_sched_final, verbose=True, seed=123)
        try_update(a_out, "slbfgs_final")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final LP polish on best (remaining time) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} ===")
    t8 = min(remaining * 0.5, 150.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t8, solver, verbose=True, seed=999)
        try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: Final coordinate descent ===
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final CD n={n_best} ===")
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent(a_norm, t9, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 10: Final pair CD ===
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final PCD n={n_best} ===")
        t10 = min(remaining * 0.6, 50.0)
        if t10 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_swap_cd(a_norm, t10, rng, verbose=False)
            try_update(a_out, f"pcd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
