    A = rfft(a, fft_size)

            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    lp_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        threshold_1 = M * 0.85
        active = np.where(conv_y >= threshold_1)[0]
        if len(active) < 20:
            threshold_2 = M * 0.55
            active = np.where(conv_y >= threshold_2)[0]

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue
        lp_count += 1

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.002)[0]
            if len(violated) > 5:
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s, {lp_count} LPs): best={best_score:.8f}")

    return best_a, best_score


def spectral_peak_shaving(a0, time_limit, verbose=True):
    """New approach: work in frequency domain to flatten autocorrelation peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    no_improve = 0

    for outer in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Find top peaks
        peak_thresh = cur_max * 0.90
        peak_indices = np.where(conv >= peak_thresh)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # Compute pressure on each coordinate from peak lags
        pressure = np.zeros(n)
        for kl in peak_indices[:min(30, len(peak_indices))]:
            for i in range(n):
                idx1 = kl - i
                if 0 <= idx1 < n:
                    pressure[i] += a[idx1]
                idx2 = i - kl
                if 0 <= idx2 < n:
                    pressure[i] += a[idx2]

        # Move mass from high-pressure to low-pressure coordinates
        sorted_idx = np.argsort(pressure)
        n_move = max(3, n // 50)
        sources = sorted_idx[-n_move:]
        targets = sorted_idx[:n_move]

        moved = False
        for src in sources:
            if a[src] < norm * 1e-10:
                continue
            for tgt in targets:
                if tgt == src:
                    continue
                # Try to move a small amount
                delta = a[src] * 0.05
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[tgt] += delta
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                new_max = fast_conv_self(a_trial).max()
                if new_max < best_score - 1e-10:
                    a = a_trial
                    best_score = new_max
                    best_a = a.copy()
                    moved = True
                    break

        if not moved:
            no_improve += 1
            if no_improve >= 15:
                break
            # Perturb
            n_pert = max(5, n // 30)
            idx_pert = np.random.choice(n, size=n_pert, replace=False)
            a[idx_pert] *= (1.0 + 0.1 * np.random.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and outer % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral pass {outer} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """Direct subgradient on the max function with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()

    momentum_buf = np.zeros(n)
    beta = 0.85

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Adaptive temperature: start high, decay
        temperature = max(0.05, max_c * 0.08 / (1.0 + 0.005 * it))
        shifted = (conv - max_c) / temperature
        shifted = np.maximum(shifted, -50.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        # Normalize gradient
        grad = grad / (np.linalg.norm(grad) + 1e-15)

        # Polyak-like step
        lr = 0.5 / (1.0 + 0.002 * it)
        momentum_buf = beta * momentum_buf - lr * grad
        a_new = a + momentum_buf
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_polish(a0, time_limit, verbose=True):
    """Greedy coordinate-wise: for each pair (i,j), find best transfer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 0.0
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Try transferring to a few candidates
            candidates = [int(rng.integers(n)) for _ in range(8)]
            # Add structured candidates from peak lags
            peak_lags = np.argsort(conv)[-3:]
            for kl in peak_lags:
                candidates.append((kl - i) % n)
                candidates.append((kl + i) % n)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in candidates:
                if j == i or j < 0 or j >= n:
                    continue
                aj = a[j]

                # Gain vector for transferring from i to j
                g = np.zeros(size)
                # Adding to j: g += a shifted by j (both directions)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Subtracting from i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Try several delta values
                for frac in [0.1, 0.25, 0.4, 0.6, 0.8, 0.95]:
                    d = ai * frac
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_pert = max(5, n // 50)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.08 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and (idx + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordPolish ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(3.0 * (x - 0.5)) ** 2 + 0.1
    elif choice == 9:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / (0.15 + 0.1 * rng.random()))
    elif choice == 10:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 7 * x) * np.cos(2 * np.pi * 3 * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    a = a + 0.01 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Spectral peak shaving (new method)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: Spectral Peak Shaving (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = spectral_peak_shaving(best_seq, t3, verbose=True)
        try_update(a_out, "spectral")

    # Phase 4: Subgradient method
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Subgradient (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = subgradient_optimize(best_seq, t4, verbose=True)
        try_update(a_out, "subgrad")

    # Phase 5: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: LP Nesterov round 2 (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 6: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 6: Coordinate Polish (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = coordinate_polish(best_seq, t6, verbose=True)
        try_update(a_out, "coord1")

    # Phase 7: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t7 = min(remaining * 0.20, 180.0)
    print(f"\n=== Phase 7: Longer n={n_long} (budget={t7:.0f}s) ===")
    if t7 > 30 and remaining > 80:
        init_long = interpolate_seq(best_seq, n_long)
        t7a = min(t7 * 0.6, 100.0)
        a_out, _ = nesterov_lp_optimize(init_long, t7a, solver, verbose=False)
        try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.4, 50.0)
            a_out2, _ = coordinate_polish(init_long, t7b, verbose=False)
            try_update(a_out2, f"coord_long")

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t_half = remaining * 0.4
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 60.0), verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = coordinate_polish(best_seq, min(remaining * 0.5, 50.0), verbose=False)
            try_update(a_out, "coord_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, min(remaining * 0.6, 60.0), solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
