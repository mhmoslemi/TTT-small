def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    if S <= 0:
        return np.zeros(n)
    mu = np.sort(v)[::-1]
    css = np.cumsum(mu)
    rho_candidates = np.where(mu * np.arange(1, n + 1) > (css - S))[0]
    if len(rho_candidates) == 0:
        return np.zeros(n)
    rho = rho_candidates[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            # Focus on moving from high to low positions
            for offset in range(1, min(30, n // 2) + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i or a[j] >= a[i]:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    # Try moving 30% of mass
                    test_d = a[i] * 0.3
                    a_test = a.copy()
                    a_test[i] -= test_d
                    a_test[j] += test_d
                    s_test = a_test.sum()
                    if s_test < 1e-15:
                        continue
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < conv.max() - 1e-13:
                        a = a_test
                        conv = conv_test
                        if conv.max() < best_score - 1e-14:
                            best_score = conv.max()
                            best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and pass_num % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={pass_num} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def make_monotone_init(n, rng):
    """Create a monotonically increasing initialization."""
    base = np.arange(1, n + 1, dtype=np.float64)
    # Add some randomness
    noise = rng.normal(0, 0.2, n)
    a = base + base * noise
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a * np.sqrt(2.0 * n) / s


def make_sine_init(n, rng):
    """Create a sine-window initialization (tapers at both ends)."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) + 0.1 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a * np.sqrt(2.0 * n) / s


def make_alternating_init(n, rng):
    """Create an anti-palindromic initialization."""
    base = np.linspace(0.1, 1.0, n)
    # Make it anti-symmetric: large at one end, small at the other
    a = base + 0.1 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a * np.sqrt(2.0 * n) / s


def make_biased_init(n, rng, bias_location=0.7):
    """Create an initialization with mass concentrated at bias_location."""
    x = np.linspace(0, 1, n)
    center = bias_location
    width = 0.3
    a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    a = a + 0.05 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a * np.sqrt(2.0 * n) / s


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_best_seq, _global_best_score

    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        pass

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Try multiple initializations
    print("\n=== Phase 0: Initialization ===")
    inits = []

    if best_seq is not None:
        inits.append(('parent', best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()))
    else:
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        inits.append(('uniform', np.ones(n_base) * np.sqrt(2.0 * n_base) / n_base))

    inits.append(('monotone', make_monotone_init(n_base, rng)))
    inits.append(('sine', make_sine_init(n_base, rng)))
    inits.append(('alternating', make_alternating_init(n_base, rng)))
    inits.append(('biased_0.7', make_biased_init(n_base, rng, 0.7)))
    inits.append(('biased_0.3', make_biased_init(n_base, rng, 0.3)))
    inits.append(('biased_0.85', make_biased_init(n_base, rng, 0.85)))

    for label, a_init in inits:
        sc = evaluate_safe(a_init / a_init.sum())
        print(f"  Init {label}: score={sc:.6f}")
        update_best(a_init, f"init_{label}")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default: n={n_base}, score={best_score:.6f}")

    n_best = len(best_seq)
    print(f"Best after init: score={best_score:.6f}, n={n_best}")

    # Phase 1: Smooth FISTA main run (40% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: Smooth FISTA n={n_best} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        temps = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25,
                 0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.012, 0.008, 0.005, 0.003, 0.002, 0.001]
        a_out, _ = smooth_fista(a_norm, t1, temps, verbose=True)
        update_best(a_out, "fista_main")

    # Phase 2: LP Nesterov (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: LP Nesterov n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 30 and best_seq is not None:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True)
        update_best(a_out, "lp_nesterov")

    # Phase 3: Coordinate descent (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.10
    print(f"\n=== Phase 3: Peak CD n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15 and best_seq is not None:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_cd(a_norm, t3, rng, verbose=True)
        update_best(a_out, "peak_cd")

    # Phase 4: Diversified restarts (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.20
    print(f"\n=== Phase 4: Diversified restarts (budget={t4:.0f}s) ===")
    if t4 > 30 and best_seq is not None:
        n_r = len(best_seq)
        n_restarts = max(2, int(t4 / 80))
        t_per_restart = t4 / n_restarts

        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 30:
                break
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < t_per_restart * 0.5:
                break

            # Different perturbation strategies
            if r % 3 == 0:
                # Gaussian noise
                a_pert = best_seq * np.sqrt(2.0 * n_r) / best_seq.sum()
                noise_scale = 0.15 * a_pert.std()
                a_pert = a_pert + rng.normal(0, noise_scale, n_r)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert * np.sqrt(2.0 * n_r) / a_pert.sum()
            elif r % 3 == 1:
                # Structured perturbation: flip a random segment
                a_pert = best_seq * np.sqrt(2.0 * n_r) / best_seq.sum()
                seg_len = max(10, n_r // 10)
                start = rng.integers(0, n_r - seg_len)
                a_pert[start:start + seg_len] *= 0.5
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert * np.sqrt(2.0 * n_r) / a_pert.sum()
            else:
                # Blend with a different structure
                a_other = make_biased_init(n_r, rng, rng.uniform(0.3, 0.7))
                a_pert = 0.7 * (best_seq * np.sqrt(2.0 * n_r) / best_seq.sum()) + 0.3 * a_other
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert * np.sqrt(2.0 * n_r) / a_pert.sum()

            # Quick FISTA on perturbed
            temp_sub = [3.0, 1.5, 0.8, 0.4, 0.2, 0.1]
            a_out, _ = smooth_fista(a_pert, t_per_restart * 0.8, temp_sub, verbose=False)
            update_best(a_out, f"restart_{r}")

            # Follow up with short CD if improved
            if best_seq is not None and best_score < float('inf'):
                remaining_sub2 = min(t_per_restart * 0.2, time_budget - (time.monotonic() - t_global))
                if remaining_sub2 > 5:
                    a_norm2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
                    a_out2, _ = peak_cd(a_norm2, remaining_sub2, rng, verbose=False)
                    update_best(a_out2, f"restart_cd_{r}")

    # Phase 5: Final polish with very low temperature FISTA (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.50
    print(f"\n=== Phase 5: Final low-T FISTA n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20 and best_seq is not None:
        n_final = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_final) / best_seq.sum()
        temps_final = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        a_out, _ = smooth_fista(a_norm, t5, temps_final, verbose=True)
        update_best(a_out, "final_fista")

    # Phase 6: Try alternate lengths if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_cur = len(best_seq)
        alt_lengths = [n_cur + 200, n_cur - 200, 2048, 1280]
        for n_alt in alt_lengths:
            if n_alt < 200 or n_alt > 5000:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 30:
                break
            t_alt = min(remaining_sub * 0.3, 60.0)
            print(f"\n=== Alt length n={n_alt} (budget={t_alt:.0f}s) ===")
            init_alt = interpolate_sequence(best_seq, n_alt)
            temps_alt = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05]
            a_out, _ = smooth_fista(init_alt, t_alt, temps_alt, verbose=False)
            update_best(a_out, f"alt_n{n_alt}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        _global_best_seq = best_seq.copy()
        _global_best_score = best_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
