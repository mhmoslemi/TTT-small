    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_softmax_subgradient(a, ac, temperature=0.05):
    """Compute smoothed subgradient using softmax weights over ALL lags."""
    n = len(a)
    # Softmax weights (shifted to avoid underflow)
    ac_shifted = ac - ac.max()
    w = np.exp(ac_shifted / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = w / w_sum
    
    # Gradient: g_i = sum_k w_k * a_{k-i} (for valid indices)
    grad = np.zeros(n)
    for k in range(len(ac)):
        if w[k] < 1e-15:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += w[k] * a[j_idx]
    return grad


def compute_soft_subgradient_topk(a, ac, top_k=30, beta=5.0):
    """Compute subgradient using top-k weighted approach."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def softmax_subgradient_optimize(a0, time_limit, initial_temp=0.1, min_temp=0.01, verbose=True):
    """
    Optimize using softmax subgradient with adaptive temperature and line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temperature = initial_temp
    lr = 0.1
    no_improve_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - _global_timer_start > _GLOBAL_TIME_BUDGET:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Compute subgradient
        g = compute_softmax_subgradient(a, ac, temperature)
        
        # Project gradient onto simplex (subtract mean for sum constraint)
        g_proj = g - g.mean()
        g_norm = np.linalg.norm(g_proj)
        if g_norm < 1e-15:
            break
        
        # Mirror descent step (exponential family)
        eta = lr / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(a, 1e-15)) - eta * g_proj
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Line search: try multiple step sizes
        best_trial = a_new
        best_trial_m = fast_autocorr(a_new).max()
        
        # Try smaller steps
        for factor in [0.5, 0.25, 0.1]:
            log_a2 = np.log(np.maximum(a, 1e-15)) - eta * factor * g_proj
            max_log2 = log_a2.max()
            a2 = np.exp(log_a2 - max_log2)
            a2 = np.maximum(a2, 1e-30)
            s2 = a2.sum()
            if s2 > 1e-15:
                a2 = a2 / s2
                m2 = fast_autocorr(a2).max()
                if m2 < best_trial_m:
                    best_trial_m = m2
                    best_trial = a2
        
        a = best_trial
        M_new = fast_autocorr(a).max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Adaptive temperature: decrease over time
        if t % 100 == 0:
            temperature *= 0.95
            temperature = max(temperature, min_temp)
        
        # Restart if stuck
        if no_improve_count > 5000:
            no_improve_count = 0
            temperature = min(initial_temp, temperature * 2)
            # Small random perturbation
            rng = np.random.default_rng(t)
            a_pert = a + rng.normal(0, 0.005, n) * a
            a = norm_sum1(a_pert)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SoftMax restart at t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temperature:.4f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SoftMax t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temperature:.4f}")
    
    return best_a, best_m


def accelerated_descent_v3(a0, time_limit, verbose=True):
    """
    Improved accelerated descent with:
    - Nesterov acceleration
    - Adaptive top-k and beta
    - Periodic full-softmax subgradient
    - Backtracking to best
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 3.0
    stall = 0
    max_stall = 3000
    top_k = 40
    beta = 5.0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - _global_timer_start > _GLOBAL_TIME_BUDGET:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        
        # Alternate between top-k and full softmax
        if t % 1000 < 500:
            g = compute_soft_subgradient_topk(y, ac, top_k=top_k, beta=beta)
        else:
            g = compute_softmax_subgradient(y, ac, temperature=0.05)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameters
        if stall == max_stall:
            # Try different top_k
            top_k = min(top_k + 5, 80)
            beta = max(beta - 0.5, 2.0)
            stall = 0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    ADv3 param change at t={t} ({elapsed:.1f}s): top_k={top_k}, beta={beta:.1f}, J={J:.6f}")
        
        if stall > 2 * max_stall:
            # Random restart
            pert = rng.normal(0, 0.01, n) * best_a
            x = norm_sum1(best_a + pert)
            x_prev = x.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    ADv3 restart at t={t} ({elapsed:.1f}s)")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ADv3 t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m
