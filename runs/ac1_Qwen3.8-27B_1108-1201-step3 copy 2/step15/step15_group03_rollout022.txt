import numpy as np

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize_seq(a, norm_val=None):
    n = len(a)
    if norm_val is None:
        norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm_val / n
    return a * norm_val / s

def lse_smooth_graddescent(a0, time_limit, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(120.0, 0.05, n_stages)
    momentum = np.zeros(n)
    beta = 0.75

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(50, min(600, int(remaining * 10)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -80.0, 0.0)
            exp_c = np.exp(shifted)
            w = exp_c / (exp_c.sum() + 1e-30)
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            lr = 1.0 / (np.sqrt(n) * (1.0 + 0.005 * it))
            momentum = beta * momentum - lr * grad
            a_new = a + momentum
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm_val / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        if verbose:
            print(f"    LSE stage {si} T={T_val:.3f}: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")
    return best_a, best_score


def lp_best_response(a, M, solver, active_set):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    if active_set is not None and 0 < len(active_set) < size:
        rows = active_set
        T_rows = np.zeros((len(rows), n))
        for r, k in enumerate(rows):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            T_rows[r, lo:hi] = a[k - np.arange(lo, hi)][::-1]
        constraints = [T_rows @ b_var <= M]
    else:
        T = np.zeros((size, n))
        for k in range(size):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            T[k, lo:hi] = a[k - np.arange(lo, hi)][::-1]
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    dual_vals = None
    try:
        if active_set is not None and 0 < len(active_set) < size:
            dual_vals = T_rows.value if hasattr(T_rows, 'value') else None
    except Exception:
        pass
    return val, dual_vals


def frank_wolfe_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    restart_count = 0
    no_improve = 0
    best_no_improve_score = best_score

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive active constraint selection
        sorted_idx = np.argsort(conv_y)[::-1]
        # Use top fraction that covers 95% of max, minimum 50 constraints
        n_active = max(50, min(size, int(0.15 * size)))
        active = sorted_idx[:n_active]
        # Also include any constraint within 80% of max
        near_max = np.where(conv_y >= M * 0.80)[0]
        active = np.unique(np.concatenate([active, near_max]))
        if len(active) > size:
            active = np.arange(size)

        b, _ = lp_best_response(y, M, solver, active)
        if b is None:
            b, _ = lp_best_response(y, M, solver, None)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum

        # Verify and add violated constraints
        check = fast_conv_cross(y, g)
        violated = np.where(check > M * 1.005)[0]
        if len(violated) > 5:
            b2, _ = lp_best_response(y, M, solver, None)
            if b2 is not None:
                b = b2
                b_sum = b.sum()
                if b_sum < 1e-15:
                    continue
                g = b * norm_val / b_sum

        # Strong Wolfe line search between a and g
        # Try several step sizes and pick the best
        best_step = 2.0 / (t + 2.0)  # default Nesterov step
        best_step_val = conv_y.max()

        n_tries = 8
        for trial in range(n_tries):
            gamma = best_step * (0.5 ** trial) if trial > 0 else best_step
            trial_a = (1.0 - gamma) * a + gamma * g
            trial_a = np.maximum(trial_a, 0.0)
            ts = trial_a.sum()
            if ts < 1e-15:
                continue
            trial_a = trial_a * norm_val / ts
            trial_obj = fast_conv_self(trial_a).max()
            if trial_obj < best_step_val:
                best_step_val = trial_obj
                best_step = gamma

        # Nesterov update with the chosen step
        alpha = best_step
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / sy

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm_val / sa

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0

        # Momentum restart: if no improvement for a while, reset
        if no_improve >= 50:
            y = a.copy()
            no_improve = 0
            restart_count += 1
            if verbose and restart_count % 10 == 1:
                print(f"    FW restart {restart_count} at t={t} ({time.monotonic()-t0:.0f}s)")

        if verbose and t % 500 == 0:
            print(f"    FW Nesterov t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def exponentiated_gradient(a0, time_limit, verbose=True):
    """Mirror descent with entropy metric on the simplex."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Use lse-smoothed subgradient with exponential update
    for stage in range(8):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        temperature = max(0.05, 5.0 / (stage + 1))
        n_steps = max(30, min(300, int(remaining * 5)))
        lr = 1.0 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / temperature
            shifted = np.clip(shifted, -60.0, 0.0)
            w = np.exp(shifted)
            w = w / (w.sum() + 1e-30)

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Exponentiated gradient update
            log_a = np.log(np.maximum(a, 1e-30))
            log_a_new = log_a - lr * grad / np.sqrt(n)
            a_new = np.exp(log_a_new - log_a_new.max())
            a_new = a_new / (a_new.sum() + 1e-30)
            a_new = a_new * norm_val

            sc = fast_conv_self(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new

        if verbose:
            print(f"    ExpGrad stage {stage}: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def peak_refinement(a0, time_limit, rng, verbose=True):
    """Targeted refinement: transfer mass from peak contributors to reduce max autocorrelation."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        peak_thresh = cur_max * 0.93
        peak_lags = np.where(conv >= peak_thresh)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm_val * 1e-12:
                continue

            # Find transfer targets that reduce the peak
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                for offset in [-1, 0, 1]:
                    j_s = (kl - i + offset) % n
                    j_candidates.add(j_s)
                    j_s2 = (kl + i + offset) % n
                    j_candidates.add(j_s2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector for transfer of mass from i to j
                g = np.zeros(size)
                # Effect of adding to j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Effect of removing from i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Sample a few deltas
                max_d = ai * 0.8
                for frac in [0.1, 0.25, 0.4, 0.6, 0.8, 1.0]:
                    d = max_d * frac
                    if d < 1e-14:
                        continue
                    new_conv = conv + d * g
                    # Correct the diagonal terms
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-10:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_val / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_imp += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(5, n // 50)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_val / s_new
        else:
            no_improve = 0

        if verbose and total_imp % 200 == 0 and total_imp > 0:
            print(f"    PeakRef imp={total_imp} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 12))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.3 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x) + 0.1
    elif choice == 8:
        a = np.sinc(2.5 * (x - 0.5)) ** 2 + 0.05
    elif choice == 9:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    elif choice == 10:
        a = 0.4 + 0.4 * np.sin(np.pi * x) + 0.2 * np.cos(2 * np.pi * 3 * x)
    else:
        a = np.ones(n) * (0.5 + 0.3 * np.sin(np.pi * x))
    a = np.maximum(a, 0.0)
    a = a + 0.02 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t5 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 5: Peak Refinement n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = peak_refinement(best_seq, t5, rng, verbose=True)
        try_update(a_out, "peakref")

    # Phase 6: Try different length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        for n_try in [n_base + 300, n_base + 600]:
            if time.monotonic() - t_global > time_budget - 60:
                break
            t_sub = min(remaining * 0.3, 150.0)
            if t_sub < 30:
                continue
            print(f"\n=== Phase 6: Length {n_try} (budget={t_sub:.0f}s) ===")
            init_long = interpolate_seq(best_seq, n_try)
            a_out, _ = frank_wolfe_nesterov(init_long, t_sub, solver, verbose=False)
            try_update(a_out, f"fw_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)

    # Phase 7: Final LSE + FW polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t_lse = min(remaining * 0.25, 60.0)
        a_out, _ = lse_smooth_graddescent(best_seq, t_lse, verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_fw = min(remaining * 0.5, 100.0)
            a_out, _ = frank_wolfe_nesterov(best_seq, t_fw, solver, verbose=False)
            try_update(a_out, "fw_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = peak_refinement(best_seq, min(remaining * 0.6, 50.0), rng, verbose=False)
            try_update(a_out, "peakfinal")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
