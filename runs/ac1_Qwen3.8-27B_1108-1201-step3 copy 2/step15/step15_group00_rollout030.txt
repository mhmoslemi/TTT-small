    A = np.fft.rfft(a, fft_size)

    w = np.exp(ac_scaled)

    if w_sum < 1e-30:

import numpy as np

def fast_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax_subgrad(a, ac, top_k=60, tau=1.0):
    """
    Compute subgradient of (1/tau)*log(sum_k exp(tau*ac[k])) using top-k approximation.
    Returns gradient of shape (n,).
    """
    n = len(a)
    k_eff = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    
    # Softmax weights on top-k
    scaled = tau * top_vals
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_eff) / k_eff
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-15:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def fista_lse(a0, time_limit, tau=3.0, top_k=60, c0=2.0, verbose=True):
    """
    FISTA (accelerated proximal gradient) with log-sum-exp smoothed subgradient.
    The proximal step is projection onto {a >= 0, sum(a) = 1}.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    t_counter = 1.0
    
    while time.monotonic() - t0 < time_limit:
        # FISTA extrapolation
        beta_t = (t_counter - 1.0) / (t_counter + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        ac = fast_conv(y)
        g = softmax_subgrad(y, ac, top_k=top_k, tau=tau)
        
        # Step size (decreasing)
        eta = c0 / (np.sqrt(t_counter) + 1.0)
        
        # Gradient step
        z = y - eta * g
        z = np.maximum(z, 0.0)
        sz = z.sum()
        if sz < 1e-15:
            z = y.copy()
        else:
            z = z / sz
        
        x_prev = x
        x = z
        t_counter += 1.0
        
        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and int(t_counter) % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={int(t_counter)} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_subgrad(a, ac, top_k=50, tau=3.0):
    """
    Compute subgradient of (1/tau)*log(sum_k exp(tau*ac[k])) using top-k approximation.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    k_eff = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]

    scaled = tau * top_vals
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_eff) / k_eff
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-15:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def topk_power_subgrad(a, ac, top_k=50, beta=4.0):
    """Compute subgradient using top-k power weighting (like previous program)."""
    n = len(a)
    k_eff = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_eff) / k_eff
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-15:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def fista_adaptive(a0, time_limit, verbose=True, seed=42):
    """
    FISTA with adaptive log-sum-exp subgradient.
    Adapts top_k and tau over time for exploration->exploitation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    t_counter = 1.0
    stall = 0
    max_stall = 8000

    # Adaptive parameters
    current_top_k = 30
    current_tau = 6.0
    c0 = 3.0

    while time.monotonic() - t0 < time_limit:
        # FISTA extrapolation
        beta_t = (t_counter - 1.0) / (t_counter + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute adaptive subgradient
        ac = fast_conv(y)
        if current_tau > 1.5:
            g = softmax_subgrad(y, ac, top_k=current_top_k, tau=current_tau)
        else:
            g = topk_power_subgrad(y, ac, top_k=current_top_k, beta=4.0)

        # Step size
        eta = c0 / (np.sqrt(t_counter) + 1.0)

        # Gradient step + projection (simplex)
        z = y - eta * g
        z = np.maximum(z, 0.0)
        sz = z.sum()
        if sz < 1e-15:
            z = y.copy()
        else:
            z = z / sz

        x_prev = x
        x = z
        t_counter += 1.0

        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart on stall
        if stall > max_stall:
            perturbation = rng.normal(0, 0.02, n) * best_a
            x_restarted = best_a + perturbation
            x_restarted = np.maximum(x_restarted, 1e-12)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            # Vary parameters
            current_top_k = min(current_top_k + 15, 100)
            current_tau = max(current_tau - 1.5, 0.8)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    FISTA restart ({elapsed:.0f}s): top_k={current_top_k}, tau={current_tau:.1f}")

        # Periodically adapt parameters (exploit more as we go)
        if int(t_counter) % 10000 == 0:
            progress = min(1.0, t_counter / 100000.0)
            current_top_k = int(30 + 50 * progress)
            current_tau = max(0.8, 6.0 - 4.0 * progress)

        if verbose and int(t_counter) % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={int(t_counter)} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}, tau={current_tau:.1f}")

    return best_a, best_m


def fista_momentum(a0, time_limit, verbose=True, seed=123):
    """
    FISTA variant with Nesterov momentum in mirror geometry (exponential family).
    Uses log-space updates for better handling of small components.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    t_counter = 1.0
    stall = 0
    max_stall = 10000
    c0 = 2.5
    current_top_k = 40
    current_beta = 5.0

    while time.monotonic() - t0 < time_limit:
        beta_t = (t_counter - 1.0) / (t_counter + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        g = topk_power_subgrad(y, ac, top_k=current_top_k, beta=current_beta)

        eta = c0 / (np.sqrt(t_counter) + 1.0)

        # Mirror (exponential) update
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        t_counter += 1.0

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Restart with different structure
            perturbation = rng.normal(0, 0.015, n) * best_a
            x_restarted = best_a + perturbation
            x_restarted = np.maximum(x_restarted, 1e-12)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            current_top_k = min(current_top_k + 10, 80)
            current_beta = max(current_beta - 0.5, 2.5)

        if verbose and int(t_counter) % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA-m t={int(t_counter)} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent_v2(a0, time_limit, verbose=True, seed=77):
    """
    Improved coordinate descent with:
    - Analytical gradient-based scoring for destination selection
    - More transfer fraction options
    - Randomized restart on stall
    */
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 150
    fractions = [0.005, 0.01, 0.02, 0.04, 0.07, 0.12, 0.2, 0.35, 0.6]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(15, len(ac))]

        improved = False

        for k in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)

            # Gradient at this lag
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            pair_vals = a[idx] * a[j_idx]

            if pair_vals.max() < 1e-20:
                continue

            src = int(idx[np.argmax(pair_vals)])
            if a[src] < 1e-12:
                continue

            # Find best destination using gradient analysis
            a_at_src_for_k = a[j_idx[np.argmax(pair_vals)]]

            best_dst = -1
            best_score = float('inf')

            # Candidates: mix of random and structured
            n_cand = min(n, 400)
            candidates = rng.integers(0, n, size=n_cand)

            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue

                # Score: how much would moving mass to dst reduce the peak?
                # Change at lag k: 2*delta*(a[k-dst] - a[k-src])
                ks = k_int - src
                kd = k_int - dst
                val_s = a[ks] if 0 <= ks < n else 0.0
                val_d = a[kd] if 0 <= kd < n else 0.0
                local_improve = val_s - val_d  # positive means improvement at this lag

                if local_improve <= 0:
                    continue

                # Penalty: how much would this increase other top lags?
                penalty = 0.0
                for k2 in top_lags[:6]:
                    k2i = int(k2)
                    ks2 = k2i - src
                    kd2 = k2i - dst
                    v_s2 = a[ks2] if 0 <= ks2 < n else 0.0
                    v_d2 = a[kd2] if 0 <= kd2 < n else 0.0
                    pen = abs(v_d2 - v_s2)
                    penalty += pen

                score = penalty - local_improve
                if score < best_score:
                    best_score = score
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst
            found = False
            for frac in fractions:
                if time.monotonic() - t0 > time_limit:
                    break
                delta = a[src] * frac
                if delta < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[dst] += delta
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if found:
                # Continue trying other lags with updated a
                ac = fast_conv(a)
                M_curr = ac.max()
                top_lags = np.argsort(ac)[::-1][:min(15, len(ac))]

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(2, min(10, n // 10) + 1)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.02, 0.15)
                    delta = a[i] * frac
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord v2 ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def pairwise_swap_descent(a0, time_limit, verbose=True, seed=55):
    """
    Try pairwise mass swaps (not just transfers) to find improvements.
    A swap exchanges mass between two positions.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(10, len(ac))]

        improved = False
        n_trials = min(50, max(10, int((time_limit - (time.monotonic() - t0)) * 5)))

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a source from a peak lag
            k = int(top_lags[rng.integers(min(5, len(top_lags)))])
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            products = a[idx] * a[j_idx]

            if products.max() < 1e-20:
                continue

            src = int(idx[np.argmax(products)])
            dst = int(rng.integers(n))
            if dst == src:
                continue

            # Try swap: exchange some mass
            swap_amt = min(a[src], a[dst]) * rng.uniform(0.1, 0.5)
            if swap_amt < 1e-15:
                # Just transfer
                if a[src] > 1e-12:
                    transfer = a[src] * rng.uniform(0.05, 0.3)
                    a_trial = a.copy()
                    a_trial[src] -= transfer
                    a_trial[dst] += transfer
                    a_trial = np.maximum(a_trial, 0.0)
                    if a_trial.sum() > 1e-15:
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
            else:
                a_trial = a.copy()
                a_trial[src] -= swap_amt
                a_trial[dst] -= swap_amt  # Wait, this isn't a swap
                # Actually for a swap, we exchange: a[src] and a[dst] swap values
                # Let's just do transfer for simplicity
                a_trial = a.copy()
                # Transfer from larger to smaller
                if a[src] > a[dst]:
                    a_trial[src] -= swap_amt
                    a_trial[dst] += swap_amt
                else:
                    a_trial[dst] -= swap_amt
                    a_trial[src] += swap_amt
                a_trial = np.maximum(a_trial, 0.0)
                if a_trial.sum() > 1e-15:
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Pairwise swap ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe for final polish."""
    n = len(a0)
    if n > 3000:
        return a0, fast_conv(a0).max()

    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Solve LP: max sum(b) s.t. (y*b)[k] <= M_curr for all k, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        g_dir = b_val / b_sum

        # FISTA-style update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, method='sin2', seed=0):
    """Create various initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)

    if method == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif method == 'cos4':
        a = np.cos(np.pi * (x - 0.5)) ** 4
        a = np.maximum(a, 0)
    elif method == 'uniform':
        a = np.ones(n)
    elif method == 'exponential':
        a = np.exp(-2.0 * (x - 0.3) ** 2 / 0.1)
    elif method == 'bimodal':
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif method == 'random':
        a = rng.random(n) ** 2
    elif method == 'flat_spectrum':
        # Try to create a sequence with flat DFT magnitude
        a = np.ones(n) / n
        a = a + 0.1 * np.sin(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    else:
        a = np.sin(np.pi * x) ** 2

    return normalize(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Uses FISTA with adaptive subgradient, coordinate descent, and LP polishing
    across multiple sequence lengths to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 150.0, 1650.0)
    rng = np.random.default_rng(2024)

    def time_remaining():
        return time_budget - (time.monotonic() - t_global)

    print(f"Total time budget: {time_budget:.0f}s")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        best_seq = make_initialization(n_base, 'sin2')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    solver = get_solver()

    # === Phase 1: FISTA adaptive on parent length ===
    rem = time_remaining()
    print(f"\n=== Phase 1: FISTA adaptive n={n0} (budget {min(rem*0.30, 450):.0f}s) ===")
    t1 = min(rem * 0.30, 450.0)
    if t1 > 30:
        a_out, _ = fista_adaptive(best_seq, t1, verbose=True, seed=42)
        try_update(a_out, f"fista_n{n0}")

    # === Phase 2: Coordinate descent ===
    rem = time_remaining()
    print(f"\n=== Phase 2: Coord descent n={len(best_seq)} ===")
    t2 = min(rem * 0.12, 180.0)
    if t2 > 20:
        a_out, _ = coordinate_descent_v2(best_seq, t2, verbose=True, seed=77)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 3: FISTA momentum variant ===
    rem = time_remaining()
    print(f"\n=== Phase 3: FISTA momentum n={len(best_seq)} ===")
    t3 = min(rem * 0.20, 300.0)
    if t3 > 30:
        a_out, _ = fista_momentum(best_seq, t3, verbose=True, seed=123)
        try_update(a_out, f"fista_m_n{len(best_seq)}")

    # === Phase 4: More coordinate descent ===
    rem = time_remaining()
    print(f"\n=== Phase 4: Coord descent again n={len(best_seq)} ===")
    t4 = min(rem * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = coordinate_descent_v2(best_seq, t4, verbose=True, seed=99)
        try_update(a_out, f"coord2_n{len(best_seq)}")

    # === Phase 5: Try longer length ===
    rem = time_remaining()
    n1 = min(n0 + 600, 3200)
    if n1 > n0 and rem > 150:
        print(f"\n=== Phase 5: Scale to n={n1} + FISTA ===")
        a_long = interpolate_sequence(best_seq, n1)
        t5 = min(rem * 0.18, 250.0)
        if t5 > 30:
            a_out, _ = fista_adaptive(a_long, t5, verbose=True, seed=555)
            try_update(a_out, f"fista_n{n1}")
            # Quick coord polish on longer
            rem = time_remaining()
            if rem > 60:
                t5b = min(rem * 0.08, 100.0)
                if t5b > 15:
                    a_out2, _ = coordinate_descent_v2(a_long, t5b, verbose=False, seed=666)
                    try_update(a_out2, f"coord_n{n1}")

    # === Phase 6: Try shorter length ===
    rem = time_remaining()
    n_short = max(n0 - 400, 1600)
    if n_short < n0 and rem > 100:
        print(f"\n=== Phase 6: Try n={n_short} ===")
        a_short = interpolate_sequence(best_seq, n_short)
        t6 = min(rem * 0.12, 150.0)
        if t6 > 20:
            a_out, _ = fista_adaptive(a_short, t6, verbose=True, seed=777)
            try_update(a_out, f"fista_n{n_short}")

    # === Phase 7: LP Frank-Wolfe if n is small enough ===
    rem = time_remaining()
    n_best = len(best_seq)
    if n_best <= 2600 and rem > 60:
        print(f"\n=== Phase 7: LP Frank-Wolfe n={n_best} ===")
        t7 = min(rem * 0.18, 250.0)
        if t7 > 30:
            a_out, _ = lp_frank_wolfe(best_seq, t7, solver, verbose=True)
            try_update(a_out, f"lp_n{n_best}")

    # === Phase 8: Final coordinate descent ===
    rem = time_remaining()
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    t8 = min(rem * 0.25, 200.0)
    if t8 > 20:
        a_out, _ = coordinate_descent_v2(best_seq, t8, verbose=True, seed=1234)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Final FISTA polish ===
    rem = time_remaining()
    n_best = len(best_seq)
    if rem > 40:
        print(f"\n=== Phase 9: Final FISTA n={n_best} ===")
        t9 = min(rem * 0.40, 200.0)
        if t9 > 20:
            a_out, _ = fista_momentum(best_seq, t9, verbose=False, seed=5678)
            try_update(a_out, f"fista_final_n{n_best}")

    # === Phase 10: Ultimate coordinate polish ===
    rem = time_remaining()
    n_best = len(best_seq)
    if rem > 30:
        print(f"\n=== Phase 10: Ultimate coord n={n_best} ===")
        t10 = min(rem * 0.70, 150.0)
        if t10 > 15:
            a_out, _ = coordinate_descent_v2(best_seq, t10, verbose=True, seed=9999)
            try_update(a_out, f"coord_ultimate_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
