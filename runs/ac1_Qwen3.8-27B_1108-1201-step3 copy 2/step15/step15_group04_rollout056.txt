def get_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.ECOS
    print(f"  Fallback solver: ECOS")
    return _solver_cache


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def normalize_to_sum(a, S):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

Actually, let me reconsider. The LSE gradient is:
`grad[p] = 2 * sum_{k: 0<=k-p<n} w[k] * a[k-p]`

Let `j = k - p`: `grad[p] = 2 * sum_{j=0}^{n-1} w[p+j] * a[j]`

This is EXACTLY the definition of the cross-correlation `(w ⋆ a)[p]` in signal processing.

In numpy/scipy: `np.correlate(w, a, mode='valid')` would give this, but it's O(n^2).

For FFT: `np.correlate` in frequency domain:
`correlate(w, a) = ifft(conj(fft(a)) * fft(w))`

Actually, let me use a different approach. I can compute the gradient using the fact that it's a "sliding window" dot product, which can be done with FFT:

`g[p] = sum_{j=0}^{n-1} w[p+j] * a[j]`

This is the same as: flip `a` to get `a_flipped[i] = a[n-1-i]`, then:
`g[p] = sum_{j=0}^{n-1} w[p+j] * a_flipped[n-1-j]`

Let `k = n-1-j`: `g[p] = sum_{k=0}^{n-1} w[p+n-1-k] * a_flipped[k]`

= `sum_k w[p+n-1-k] * a_flipped[k]` = `(w * a_flipped)[p+n-1]` (standard convolution)

So `g = conv(w, flip(a))[n-1 : 2n-1]` (a slice of the full convolution).

This can be computed with FFT in O(n log n)!
