The inner problem is: `minimize max_k (p*q)_k subject to q in simplex`

Frank-Wolfe for this:
- At each iteration, compute the gradient (which is the shifted copy of p at the maximizing lag)
- Find the simplex vertex that minimizes the gradient (i.e., the coordinate with the smallest gradient value)
- Update q toward that vertex

This is O(n log n) per iteration (for the FFT to compute the max lag and gradient).

With 50 iterations of Frank-Wolfe, the inner solve is 50 * O(n log n) ≈ 50 * 1600 * 11 ≈ 880,000 ops.

An LP solver with cvxpy might take 5-50ms for this problem size, which is comparable.

The advantage of Frank-Wolfe is:
- No dependency on cvxpy
- Can be parallelized easily (though we only have 1 CPU)
- Easy to control the number of iterations
- No solver overhead

The disadvantage:
- Slower convergence (O(1/t) vs O(1/sqrt(n)) for interior point)
- May not find the exact optimum

I think for our purpose (we don't need the exact inner optimum, just a good direction), Frank-Wolfe is sufficient and faster in practice due to lower overhead.

                if j_low + k < n and j_high + k < n:
                    d2 = p[j_high + k] * frac * 0.5
                    if d2 > 0 and d2 < p[j_high + k]:
                        p_try[j_high + k] -= d2
                        p_try[j_low + k] += d2
                p_try = np.maximum(p_try, 0.0)
                s_try = p_try.sum()
                if s_try < 1e-15:
                    continue
                p_try = p_try / s_try
                sc = score_local(p_try)
                if sc < best_score - 1e-13:
                    p = p_try
                    best_score = sc
                    best_p = p.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0
        if verbose and it % 2000 == 0:
            el = time.monotonic() - t0
            print(f"    TCD it={it} ({el:.0f}s): best={best_score:.10f}")
    return best_p, best_score

def random_cd(p0, time_limit, rng, verbose=False):
    n = len(p0)
    p = np.maximum(p0, 0.0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    best_p = p.copy()
    best_score = score_local(p)
    t0 = time.monotonic()
    no_improve = 0
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        i = rng.integers(n)
        if p[i] < 1e-8:
            no_improve += 1
            if no_improve > 500:
                break
            continue
        for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
            p_try = p.copy()
            p_try[i] *= scale
            p_try = np.maximum(p_try, 0.0)
            s_try = p_try.sum()
            if s_try < 1e-15:
                continue
            p_try = p_try / s_try
            sc = score_local(p_try)
            if sc < best_score - 1e-13:
                p = p_try
                best_score = sc
                best_p = p.copy()
                no_improve = 0
                break
        else:
            no_improve += 1
            if no_improve > 500:
                break
        if verbose and it % 3000 == 0:
            el = time.monotonic() - t0
            print(f"    RCD it={it} ({el:.0f}s): best={best_score:.10f}")
    return best_p, best_score

def structured_init(n, style):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - (2.0 * x - 1.0) ** 2) ** 1.5
    elif style == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(4 * np.pi * x) + 0.08 * np.cos(8 * np.pi * x)
    elif style == 7:
        a = np.sin(np.pi * x) ** 0.5 * (1.0 + 0.1 * np.cos(6 * np.pi * x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
