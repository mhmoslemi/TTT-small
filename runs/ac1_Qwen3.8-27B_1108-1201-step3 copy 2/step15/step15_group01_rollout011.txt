def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient(a, w, n):
    """Compute gradient of T*log(sum exp(c/T)) where w = softmax(c/T).
    grad_i = 2 * sum_k w_k * a[(k-i) mod ...]
    Using cross-correlation: grad = 2 * cross_corr(w, a)
    """
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_corr[w, a][i] = sum_k w[k] * a[k-i] (with zero padding)
    # This is the same as the "reversed convolution": conv(w_rev, a)
    # Or equivalently, we can compute it as:
    # irfft(W * conj(A)) gives cross-correlation
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross

def subgradient_active(a, active_indices, n):
    """Compute subgradient using active set of peak indices.
    grad[m] = 2 * sum_{k in active} w_k * a[k-m] (if 0 <= k-m < n)
    """
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    w = np.zeros(size)
    w[active_indices] = 1.0 / len(active_indices)
    return smooth_gradient(a, w, n)

def smooth_optimize(a0, time_limit, verbose=True):
    """Smooth log-sum-exp optimization with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Temperature schedule - more aggressive cooling
    temps = [50.0, 25.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Estimate number of steps we can do
        n_steps = max(50, min(1000, int(remaining * 5)))
        
        y = a.copy()
        momentum = 0.0
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()
            
            grad = smooth_gradient(y, w, n)
            
            # Adaptive step size based on gradient norm
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (grad_norm * (1.0 + 0.1 * (it % 100)))
            
            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)
            
            # Nesterov momentum
            momentum = (1 + momentum) / 2  # increasing momentum
            alpha = (1 + momentum) / (2 + momentum)
            y_new = a_new + momentum * (a_new - y)
            y_new = project_simplex(y_new, norm)
            
            if it == 0:
                a = a_new
            else:
                a = y_new
            
            y = y_new
            
            # Check improvement
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.3f} iter={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def subgradient_method(a0, time_limit, verbose=True):
    """Subgradient method with active set and line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    step_size = 1.0 / np.sqrt(n)
    n_iter = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()
        
        # Active set: all peaks within 95% of max
        threshold = 0.95 * c_max
        active = np.where(c >= threshold)[0]
        
        grad = subgradient_active(a, active, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Line search
        best_step = step_size
        best_val = c_max
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            trial = a - best_step * frac * grad / grad_norm
            trial = project_simplex(trial, 1.0)
            val = fast_autocorr(trial).max()
            if val < best_val * (1 - 1e-6):
                best_val = val
                best_step *= frac
            else:
                break
        
        a_new = a - best_step * grad / grad_norm
        a_new = project_simplex(a_new, 1.0)
        
        if fast_autocorr(a_new).max() < best_score * (1 - 1e-8):
            a = a_new
            best_score = fast_autocorr(a).max()
            best_a = a.copy()
            no_improve = 0
        else:
            a = a_new  # still update to explore
            no_improve += 1
        
        # Decay step size
        step_size *= 0.999
        n_iter += 1
        
        if no_improve > 200:
            # Reset with perturbation
            a = a + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            step_size = 1.0 / np.sqrt(n)
            no_improve = 0
        
        if verbose and n_iter % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={n_iter} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def coord_descent(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with greedy pair selection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        # Focus on pairs that affect the peak
        peak_pos = np.argmax(conv)
        # Pairs (i, j) such that i + j == peak_pos (mod something) contribute to conv[peak_pos]
        # Actually conv[peak_pos] = sum_i a[i]*a[peak_pos-i]
        # So we want to reduce the contributions at the peak
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-10:
                continue
            
            # Try moving mass from i to nearby positions
            for offset in range(1, min(30, n // 4) + 1):
                for j in [i + offset, i - offset]:
                    if j < 0 or j >= n or j == i:
                        continue
                    if a[i] < 1e-10:
                        break
                    
                    # Quick test: move 20% of a[i] to a[j]
                    d = 0.2 * a[i]
                    a_test = a.copy()
                    a_test[i] -= d
                    a_test[j] += d
                    a_test = a_test / a_test.sum()
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < best_score * (1 - 1e-10):
                        a = a_test
                        conv = conv_test
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                        break  # Stop trying this i, move to next
                # Small step if 20% didn't work
                if a[i] > 1e-10:
                    d = 0.05 * a[i]
                    best_d = 0
                    best_val = conv.max()
                    for frac in [0.5, 0.2, 0.1]:
                        d_try = d * frac
                        j_try = (i + offset) % n
                        a_test = a.copy()
                        a_test[i] -= d_try
                        a_test[j_try] += d_try
                        a_test = a_test / a_test.sum()
                        val = fast_autocorr(a_test).max()
                        if val < best_val * (1 - 1e-10):
                            best_val = val
                            best_d = d_try
                    if best_d > 0:
                        a[i] -= best_d
                        a[(i + offset) % n] += best_d
                        a = a / a.sum()
                        conv = fast_autocorr(a)
                        if conv.max() < best_score:
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
