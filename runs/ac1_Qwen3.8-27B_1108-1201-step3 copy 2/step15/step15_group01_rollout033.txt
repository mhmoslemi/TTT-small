import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Evaluate the sequence safely."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def cross_corr_fft(w, a):
    """Compute sum_m w[k+m] * a_m for all k, using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(W * np.conj(A), fft_size)[:n]

def pnorm_obj_grad(a, p):
    """Compute p-norm of autocorrelation and its gradient."""
    c = fast_autocorr(a)
    c = np.maximum(c, 0.0)  # numerical safety
    cp = c ** p
    S = cp.sum()
    if S < 1e-30:
        return 0.0, np.zeros_like(a)
    fp = S ** (1.0 / p)
    w = c ** (p - 1)
    cross = cross_corr_fft(w, a)
    grad = (fp / S) * 2.0 * cross
    return fp, grad

def logsumexp_obj_grad(a, T):
    """Compute T*log(sum(exp(c/T))) and its gradient."""
    c = fast_autocorr(a)
    c = np.maximum(c, 0.0)
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0, np.zeros_like(a)
    exp_c = np.exp((c - c_max) / T)
    S = exp_c.sum()
    obj = T * np.log(S) + c_max
    w = exp_c / S
    cross = cross_corr_fft(w, a)
    grad = 2.0 * cross
    return obj, grad

def project_nonneg_sum(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def nesterov_smooth(a0, time_limit, verbose=True):
    """Nesterov-accelerated gradient descent on p-norm with backtracking line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Sweep over p values
    p_schedule = [4, 6, 8, 12, 16, 24, 32, 48, 64, 96, 128, 192, 256]
    
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        n_steps = max(50, min(300, int(remaining * 2)))
        
        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            fp, grad = pnorm_obj_grad(y, p)
            
            # Backtracking line search
            step = norm / (n * 0.1)
            func_val = fp
            improved = False
            for ls in range(20):
                a_trial = y - step * grad
                a_trial = project_nonneg_sum(a_trial, norm)
                fp_trial, _ = pnorm_obj_grad(a_trial, p)
                if fp_trial <= func_val - 1e-12 * step * np.linalg.norm(grad):
                    improved = True
                    break
                step *= 0.5
            
            a_new = project_nonneg_sum(a_trial, norm)
            
            # Nesterov update
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_nonneg_sum(y_new, norm)
            
            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = it / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_nonneg_sum(a_curr, norm)
            
            y = y_new
            a = a_curr
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    P-norm p={p} step={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def adam_smooth(a0, time_limit, T_schedule=None, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, verbose=True):
    """Adam optimizer on log-sum-exp approximation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if T_schedule is None:
        T_schedule = [50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        n_steps = max(50, min(500, int(remaining * 3)))
        m = np.zeros(n)  # Reset momentum for new temperature
        v = np.zeros(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            obj, grad = logsumexp_obj_grad(a, T)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(it + 1))
            v_hat = v / (1 - beta2**(it + 1))
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_nonneg_sum(a_new, norm)
            a = a_new
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam T={T:.2f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def subgrad_nesterov(a0, time_limit, verbose=True):
    """Subgradient Nesterov on max(conv(a,a)) using active set tracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(y)
        M = c.max()
        if M < 1e-12:
            break
        
        # Active set: positions where c is close to max
        threshold = M - 1e-6 * M
        active = c >= threshold
        
        # Subgradient: uniform over active set, then cross-correlate with a
        w = active.astype(np.float64)
        w = w / w.sum()
        cross = cross_corr_fft(w, y)
        grad = 2.0 * cross
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        step = norm / (n * 0.5)
        
        y_new = y - step * grad
        y_new = project_nonneg_sum(y_new, norm)
        
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_nonneg_sum(a_new, norm)
        
        y = y_new
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad Nesterov iter={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_focused(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent focused on reducing the peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 5:
            break
        
        # Find the peak position
        peak_pos = conv.argmax()
        peak_val = conv.max()
        
        # Find positions contributing most to the peak
        # c[peak_pos] = sum_i a[i] * a[peak_pos - i]
        # The main contributors are at positions i and peak_pos - i
        contributors = []
        for i in range(n):
            j = peak_pos - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)
        
        improved_this_pass = False
        
        # Try to reduce contributions from top pairs
        for contrib, i, j in contributors[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            if i == j:
                continue
            if a[i] < norm * 1e-10 or a[j] < norm * 1e-10:
                continue
            
            # Try moving mass from the larger of a[i], a[j] to a random position
            if a[i] >= a[j]:
                src = i
            else:
                src = j
            
            # Try several destination positions
            for _ in range(10):
                dst = rng.integers(0, n)
                if dst == src:
                    continue
                
                # Try moving a small fraction
                for frac in [0.1, 0.05, 0.02]:
                    delta = a[src] * frac
                    if delta < norm * 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[src] -= delta
                    a_test[dst] += delta
                    a_test = project_nonneg_sum(a_test, norm)
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < conv.max() - 1e-12:
                        a = a_test
                        conv = conv_test
                        if conv.max() < best_score:
                            best_score = conv.max()
                            best_a = a.copy()
                        improved_this_pass = True
                        break
                if improved_this_pass:
                    break
        
        if not improved_this_pass:
            no_improve += 1
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PairCD pass={t0 and int(elapsed)} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. P-norm smooth optimization with Nesterov acceleration and adaptive step
2. L-BFGS-B quasi-Newton on p-norm approximation
3. Adam optimizer with log-sum-exp (FFT-based gradients)
4. Subgradient Nesterov with active set tracking
5. Focused pair coordinate descent targeting peak contributors
6. Multi-start diversification with varied initializations

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Evaluate the sequence safely."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def cross_corr_fft(w, a):
    """Compute (w cross a)[k] = sum_m w[k+m] * a_m for all k, using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(W * np.conj(A), fft_size)[:n]

def pnorm_obj_grad(a, p, eps=1e-30):
    """Compute p-norm of autocorrelation and its gradient."""
    c = fast_autocorr(a)
    c = np.maximum(c, eps)
    cp = c ** p
    S = cp.sum()
    if S < 1e-30:
        return 0.0, np.zeros_like(a)
    fp = S ** (1.0 / p)
    w = c ** (p - 1)
    cross = cross_corr_fft(w, a)
    grad = (fp / S) * 2.0 * cross
    return fp, grad

def logsumexp_obj_grad(a, T, eps=1e-30):
    """Compute T*log(sum(exp(c/T))) and its gradient."""
    c = fast_autocorr(a)
    c = np.maximum(c, eps)
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0, np.zeros_like(a)
    exp_c = np.exp((c - c_max) / T)
    S = exp_c.sum()
    obj = T * np.log(S) + c_max
    w = exp_c / S
    cross = cross_corr_fft(w, a)
    grad = 2.0 * cross
    return obj, grad

def project_nonneg_sum(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def nesterov_pnorm(a0, time_limit, verbose=True):
    """Nesterov-accelerated gradient descent on p-norm with adaptive step sizes."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    p_schedule = [4, 6, 8, 12, 16, 24, 32, 48, 64, 96, 128, 192, 256, 384]

    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        n_steps = max(40, min(200, int(remaining * 2.5)))

        y = a.copy()
        step = norm / (n * 0.1)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            fp, grad = pnorm_obj_grad(y, p)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            # Backtracking line search (limited)
            func_val = fp
            a_trial = y - step * grad
            a_trial = project_nonneg_sum(a_trial, norm)
            fp_trial, _ = pnorm_obj_grad(a_trial, p)
            ls_count = 0
            while fp_trial > func_val - 1e-14 * step * gnorm and ls_count < 8:
                step *= 0.6
                a_trial = y - step * grad
                a_trial = project_nonneg_sum(a_trial, norm)
                fp_trial, _ = pnorm_obj_grad(a_trial, p)
                ls_count += 1

            a_new = a_trial
            if fp_trial < func_val:
                step *= 1.05

            # Nesterov update
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_nonneg_sum(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = it / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_nonneg_sum(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and it % 30 == 0:
                elapsed = time.monotonic() - t0
                print(f"    PNorm p={p} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lbfgs_pnorm(a0, time_limit, p=16, verbose=True):
    """L-BFGS-B on the scale-invariant p-norm objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    t0 = time.monotonic()

    def objective(x):
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            return 1e10
        c = fast_autocorr(x)
        c = np.maximum(c, 1e-30)
        return (c ** p).sum() ** (1.0 / p) / (sx * sx)

    def gradient(x):
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            return np.zeros(n)
        fp, grad = pnorm_obj_grad(x, p)
        # d/dx [fp / sx^2] = grad/sx^2 - 2*fp/sx^3
        return grad / (sx * sx) - 2.0 * fp / (sx ** 3)

    try:
        result = scipy_minimize(
            objective, a, method='L-BFGS-B', jac=gradient,
            bounds=[(0.0, None)] * n,
            options={'maxiter': 200, 'maxfun': 5000, 'maxls': 20,
                     'ftol': 1e-14, 'gtol': 1e-8, 'iprint': -1}
        )
        if result.success or result.fun < 1e10:
            a_result = np.maximum(result.x, 0.0)
            sr = a_result.sum()
            if sr > 1e-15:
                a_result = a_result / sr
                score = evaluate_safe(a_result)
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    L-BFGS p={p} ({elapsed:.0f}s): score={score:.8f}")
                return a_result, score
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")

    return a0 / a0.sum(), evaluate_safe(a0 / a0.sum())

def adam_lse(a0, time_limit, verbose=True):
    """Adam optimizer on log-sum-exp approximation with temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    T_schedule = [100.0, 50.0, 25.0, 15.0, 10.0, 6.0, 4.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        n_steps = max(50, min(400, int(remaining * 3)))
        m = np.zeros(n)
        v = np.zeros(n)
        lr = norm / (n * 0.5)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            obj, grad = logsumexp_obj_grad(a, T)

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_nonneg_sum(a_new, norm)
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - t0

def subgrad_nesterov(a0, time_limit, verbose=True):
    """Subgradient Nesterov on max(conv(a,a)) using active set tracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    step_base = norm / (n * 0.3)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(y)
        M = c.max()
        if M < 1e-12:
            break

        # Active set with fractional weight
        threshold = M * (1.0 - 1e-4)
        w = np.where(c >= threshold, 1.0, 0.0)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
            w_sum = 1.0
        w = w / w_sum

        cross = cross_corr_fft(w, y)
        grad = 2.0 * cross

        # Nesterov step with diminishing step size
        step = step_base / (1.0 + 0.001 * t)
        y_new = y - step * grad
        y_new = project_nonneg_sum(y_new, norm)

        alpha = 2.0 / (t + 2.0)
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_nonneg_sum(a_new, norm)

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_peak(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent focused on reducing the peak contributor."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 8:
            break
        n_passes += 1

        peak_pos = int(conv.argmax())
        improved = False

        # Find top contributors to the peak
        contribs = []
        for i in range(min(n, peak_pos + 1)):
            j = peak_pos - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(key=lambda x: -x[0])

        for contrib, i, j in contribs[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            if i == j:
                continue
            if a[i] < norm * 1e-8 or a[j] < norm * 1e-8:
                continue

            src = i if a[i] >= a[j] else j
            other = j if src == i else i

            # Try multiple destinations
            for attempt in range(20):
                if improved:
                    break
                dst = int(rng.integers(0, n))
                if dst == src or dst == other:
                    continue
                for frac in [0.15, 0.08, 0.04]:
                    delta = a[src] * frac
                    if delta < norm * 1e-8:
                        continue
                    a_test = a.copy()
                    a_test[src] -= delta
                    a_test[dst] += delta
                    a_test = project_nonneg_sum(a_test, norm)
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < conv.max() - 1e-12:
                        a = a_test
                        conv = conv_test
                        if conv.max() < best_score:
                            best_score = conv.max()
                            best_a = a.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1650.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: P-norm Nesterov (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.30
    n1 = n_base
    print(f"\n=== Phase 1: PNorm Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_pnorm(a_norm, t1, verbose=True)
        try_update(a_out, f"pnorm_n{n1}")

    # === Phase 2: L-BFGS-B on p-norm (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = remaining * 0.15
    print(f"\n=== Phase 2: L-BFGS n={n2} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n2) / best_seq.sum()
        # Try a few p values
        for p_val in [16, 32]:
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 10:
                break
            a_out, sc = lbfgs_pnorm(a_norm, min(remaining_sub * 0.5, t2), p=p_val, verbose=True)
            try_update(a_out, f"lbfgs_p{p_val}")

    # === Phase 3: Adam LSE (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: Adam LSE n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n3) / best_seq.sum()
        a_out, _ = adam_lse(a_norm, t3, verbose=True)
        try_update(a_out, f"adam_n{n3}")

    # === Phase 4: Subgradient Nesterov (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: SubGrad Nesterov n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n4) / best_seq.sum()
        a_out, _ = subgrad_nesterov(a_norm, t4, verbose=True)
        try_update(a_out, f"subgrad_n{n4}")

    # === Phase 5: Pair CD (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Pair CD n={n5} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n5) / best_seq.sum()
        a_out, _ = pair_cd_peak(a_norm, t5, rng, verbose=True)
        try_update(a_out, f"pcd_n{n5}")

    # === Phase 6: Diversified restarts (7% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.07
    print(f"\n=== Phase 6: Diversified restarts (budget={t6:.0f}s) ===")
    if t6 > 30 and best_seq is not None:
        n6 = len(best_seq)
        # Try a few perturbed restarts
        for restart_idx in range(3):
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 15:
                break
            t_sub = min(remaining_sub * 0.4, 120.0)
            if t_sub < 15:
                break
            a_pert = best_seq * np.sqrt(2.0 * n6) / best_seq.sum()
            noise_scale = 0.03 + 0.02 * restart_idx
            noise = rng.normal(0, noise_scale * a_pert.std(), n6)
            a_pert = np.maximum(a_pert + noise, 0)
            a_pert = a_pert * np.sqrt(2.0 * n6) / a_pert.sum()
            a_out, _ = nesterov_pnorm(a_pert, t_sub, verbose=False)
            try_update(a_out, f"restart{restart_idx}")

    # === Phase 7: Try alternate lengths (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_base_final = len(best_seq)
        n_alts = [n_base_final + 200, n_base_final - 200, n_base_final + 400]
        for n_alt in n_alts:
            if n_alt < 200:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 30:
                break
            t_alt = min(remaining_sub * 0.4, 100.0)
            if t_alt < 15:
                break
            print(f"\n=== Alt length n={n_alt} (budget={t_alt:.0f}s) ===")
            init_alt = interpolate_seq(best_seq, n_alt)
            a_out, _ = nesterov_pnorm(init_alt, t_alt, verbose=False)
            try_update(a_out, f"alt_n{n_alt}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
