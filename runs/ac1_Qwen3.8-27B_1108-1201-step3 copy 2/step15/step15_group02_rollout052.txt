import numpy as np

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CBC, cp.GLPK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using LP solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 25
    temps = np.geomspace(100.0, 0.02, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(150, int(remaining * 8)))
        lr = 0.4 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -60.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]

            aw = irfft(A * W, fft_size)
            g2 = aw[:n]

            grad = 2.0 * (g1 + g2)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score

def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and 0 < len(active_set) < size:
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[idx, i] = a[k - i]
        constraints = [T @ b_var <= M]
    else:
        T = np.zeros((size, n))
        for m_idx in range(n):
            T[m_idx:m_idx + n, m_idx] = a
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    prev_solution = None

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-14:
            break

        # Adaptive threshold for active set
        if t < 10:
            threshold = 0.5
        elif t < 50:
            threshold = 0.7
        elif t < 200:
            threshold = 0.8
        else:
            threshold = 0.85

        active_set = np.where(conv_y >= M * threshold)[0]

        # Ensure we have enough constraints
        if len(active_set) < 20:
            active_set = np.where(conv_y >= M * 0.4)[0]
        # Cap the number of constraints for speed
        if len(active_set) > 800:
            top = np.argsort(-conv_y)[:800]
            active_set = top

        b = lp_best_response(y, M, solver, active_set)

        if b is None:
            b = lp_best_response(y, M, solver, None)
            if b is None:
                continue

        # Verify constraints and add violated ones (lazy constraint generation)
        if active_set is not None and len(active_set) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.005)[0]
            if len(violated) > 0:
                active_set = np.union1d(active_set, violated)
                if len(active_set) <= 1500:
                    b2 = lp_best_response(y, M, solver, active_set)
                    if b2 is not None:
                        b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update with slightly modified step size for faster convergence
        alpha = 2.0 / (t + 1.0 + 1.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, |active|={len(active_set)}")

    return best_a, best_score

def pressure_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Find pressure positions (where convolution is high)
        pressure = np.maximum(conv - cur_max * 0.9, 0)
        if pressure.sum() < 1e-15:
            pressure = np.maximum(conv - cur_max * 0.7, 0)

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Choose j based on pressure at relevant lags
            j_candidates = set()
            # Random candidates
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            # Pressure-guided candidates
            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            for kl in peak_lags[:15]:
                j1 = int((kl - i + n) % n)
                j2 = int((kl + i) % n)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)

            # Weight by pressure
            weighted_j = []
            for j in j_candidates:
                if j == i:
                    continue
                # Compute pressure contribution at lag involving i and j
                lag = (i + j) % (2 * n - 1)
                w = conv[lag] if lag < size else conv[2*n-2-lag]
                weighted_j.append((j, w))

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j, w in weighted_j[:12]:
                aj = a[j]

                # Compute gradient of conv when moving mass from i to j
                g = np.zeros(size)

                # Remove mass from i
                lo = max(0, i)
                hi = min(size, i + n)
                if hi > lo:
                    g[lo:hi] -= a[0:hi - lo]
                lo2 = max(0, i - n + 1)
                hi2 = min(size, i + 1)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[i - np.arange(lo2, hi2)]

                # Add mass to j
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[0:hi3 - lo3]
                lo4 = max(0, j - n + 1)
                hi4 = min(size, j + 1)
                if hi4 > lo4:
                    g[lo4:hi4] += a[j - np.arange(lo4, hi4)]

                d_max = ai * 0.7
                n_samples = 20
                deltas = np.linspace(0.02, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                # Perturbation
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                scale = 0.1 + 0.05 * no_improve
                a[idx_p] *= (1.0 + scale * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
            if no_improve >= 15:
                break
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym * (a.sum() / s)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        a = np.exp(-40.0 * (x - c1) ** 2) + np.exp(-40.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * x) + 0.2 * np.sin(2 * np.pi * 5 * x)
    elif choice == 8:
        a = np.exp(-3.0 * x) * 0.5 + 0.5
    else:
        a = 0.3 + 0.7 * np.cos(np.pi * (x - 0.5)) ** 2

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

    if remaining > 80:

        lengths_to_try = [n_base + 400, n_base + 800, n_base - 400, n_base + 1200]
        for n_try in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            if n_try < 200 or n_try > 5000:
                continue
            n_try = int(n_try)
            t_try = min(remaining * 0.30, 50.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try, 25.0), verbose=False)
            check_and_update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out, _ = pressure_cd_optimize(a_out, min(remaining * 0.4, 20.0), rng, verbose=False)
                check_and_update(a_out, f"ml_pcd_n{n_try}")

    # Phase 8: Fresh random starts (diversification)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 8: Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            n_fresh = max(800, n_base + int(rng.integers(-300, 500)))
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.25, 20.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_and_update(a_out, f"fresh{attempt}_n{n_fresh}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_out, _ = pressure_cd_optimize(a_out, min(remaining * 0.35, 12.0), rng, verbose=False)
                check_and_update(a_out, f"fresh{attempt}_pcd_n{n_fresh}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
