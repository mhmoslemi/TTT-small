def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def exact_subgradient(a, ac, eps_frac=0.01):
    """
    Compute exact subgradient of max_k (a*a)_k.
    Uses all lags within eps_frac*M of the max as active set.
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    threshold = M * (1.0 - eps_frac)
    active = np.where(ac >= threshold)[0]
    count = len(active)
    if count == 0:
        active = np.array([np.argmax(ac)])
        count = 1

    grad = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * a[j_idx]
    return grad / count


def compute_smooth_grad(a, ac, top_k=60, beta=4.0):
    """Smooth subgradient using softmax weighting over top peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    top_vals = np.maximum(top_vals, 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def pairwise_local_search(a0, time_limit, verbose=True, seed=123):
    """

            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib[idx] += ac[k] * a[idx] * a[j_idx]

        # Try moving mass from high-contribution to low-contribution positions

        max_tries_per_pass = 200

            if a[idx_i] < 1e-12:
                continue

            src = int(idx_i)
            # Candidate destinations: low contribution, not in peak neighborhoods
            in_peak = np.zeros(n, dtype=bool)
            for k in k_peaks[:8]:

                    d2 = src - k_int

def freq_domain_smooth(a, alpha=0.1, time_limit=5.0):
    """Apply gentle frequency-domain smoothing to reduce high-freq autocorrelation peaks."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    # Gentle low-pass: blend with smoothed version
    A_smooth = A.copy()
    freqs = np.fft.rfftfreq(fft_size)
    # Gaussian low-pass
    sigma_f = alpha / (2.0 * np.sqrt(np.log(2.0)))
    if sigma_f > 0:
        lowpass = np.exp(-0.5 * (freqs / sigma_f) ** 2)
        A_blend = (1.0 - alpha) * A + alpha * A * lowpass
    else:
        A_blend = A
    a_new = np.fft.irfft(A_blend, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)

def exact_subgradient(a, ac, eps_frac=0.01):
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    threshold = M * (1.0 - eps_frac)
    active = np.where(ac >= threshold)[0]
    count = len(active)
    if count == 0:
        active = np.array([np.argmax(ac)])
        count = 1
    grad = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * a[j_idx]
    return grad / count
