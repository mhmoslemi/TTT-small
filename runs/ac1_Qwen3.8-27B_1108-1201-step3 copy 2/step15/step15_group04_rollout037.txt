        # Line search for optimal step size
        best_alpha = 0.5
        best_obj = M
        alpha_candidates = [0.3, 0.5, 0.7, 0.85, 0.95, 1.0]
        for alpha in alpha_candidates:
            y_trial = (1.0 - alpha) * y + alpha * g
            y_trial = np.maximum(y_trial, 0.0)
            st = y_trial.sum()
            if st < 1e-15:
                continue
            y_trial = y_trial / st
            obj = fast_conv_self(y_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g
        y_new = np.maximum(y_new, 0.0)
        st = y_new.sum()
        if st < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / st

        # Momentum (Nesterov-style, damped)
        if t > 5:
            beta = 0.9 * (t - 1) / (t + 2)
            y_new = y_new + beta * (y_new - y)
            y_new = np.maximum(y_new, 0.0)
            st = y_new.sum()
            if st > 1e-15:
                y_new = y_new / st
            else:
                y_new = (1.0 - best_alpha) * y + best_alpha * g
                y_new = np.maximum(y_new, 0.0) / y_new.sum()

        prev_y = y.copy()
        y = y_new

        sc = fast_conv_self(y).max()
        score_val = 2.0 * n * sc
        if score_val < best_score - 1e-12:
            best_score = score_val
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic perturbation to escape local optima
        if no_improve == 80:
            n_pert = max(10, n // 25)
            idx_p = np.random.default_rng(t * 7 + 3).choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.15 * np.random.default_rng(t * 11 + 1).standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            st = y.sum()
            if st > 1e-15:
                y = y / st
            no_improve = 0

        # Stronger perturbation
        if no_improve == 200:
            n_pert = max(20, n // 10)
            idx_p = np.random.default_rng(t * 13 + 5).choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.3 * np.random.default_rng(t * 17 + 2).standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            st = y.sum()
            if st > 1e-15:
                y = y / st
            no_improve = 0

        # Running average
        avg_y += y
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_y / n_avg
            st = avg_check.sum()
            if st > 1e-15:
                avg_check = avg_check / st
                sc_avg = 2.0 * n * fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-MD t={t} ({elapsed:.0f}s): score={best_score:.8f}, no_imp={no_improve}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """Projected subgradient with averaging for fine polishing."""
    n = len(a0)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Iteratively identify active lags and do subgradient steps
    active_lags = np.array([np.argmax(fast_conv_self(a))])
    step_sizes = np.geomspace(0.1, 0.001, 50)

    for si, eta in enumerate(step_sizes):
        if time.monotonic() - t0 > time_limit:
            break
        n_steps = max(10, min(100, int((time_limit - (time.monotonic() - t0)) * 2)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            C = fast_conv_self(a)
            max_C = C.max()

            # Update active set
            eps = max_C * 0.05
            active_lags = np.where(C >= max_C - eps)[0]

            # Compute subgradient (average over active lags)
            grad = np.zeros(n)
            for m in active_lags:
                for i in range(n):
                    j1 = m - i
                    if 0 <= j1 < n:
                        grad[i] += a[j1]
                    j2 = i + m
                    if 0 <= j2 < n:
                        grad[i] += a[j2]
            grad /= len(active_lags)

            # Step
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0.0)
            st = a_new.sum()
            if st < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / st
            a = a_new

            sc = 2.0 * n * fast_conv_self(a).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent focusing on peak-contributing pairs."""
    n = len(a0)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            peak_lags = np.where(conv >= cur_max * 0.85)[0]
            for kl in peak_lags[:6]:
                j1 = kl - i
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                j2 = kl + i
                if 0 <= j2 < n:
                    j_candidates.add(j2)
                j3 = kl - i + n
                if 0 <= j3 < n:
                    j_candidates.add(j3)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                gain = np.zeros(size)
                for m in range(max(0, j), min(size, j + n)):
                    gain[m] += a[m - j]
                for m in range(max(0, j - n + 1), min(size, j + 1)):
                    gain[m] += a[m + j - (n - 1) + (n - 1) - m + m - j + j]
                # Simplify: use direct convolution update
                # conv_new = conv + delta * (a shifted) - delta * (a shifted)
                # For moving delta from i to j:
                # conv_new[m] = conv[m] + delta*(a[m-j] + a[m+j-(n-1)+(n-1)]) - delta*(...)
                # Actually let's just compute the change directly
                # Change in conv at lag m: delta * (a_contrib_from_j - a_contrib_from_i)
                # This is complex, so use sampling

                n_samples = 15
                for d in np.linspace(0.02, 0.8, n_samples) * ai:
                    # Approximate new conv
                    a_mod = a.copy()
                    a_mod[i] = ai - d
                    a_mod[j] = aj + d
                    st = a_mod.sum()
                    if st < 1e-15:
                        continue
                    # Quick check: just evaluate at peak lags
                    max_est = 0.0
                    for m in peak_lags[:4]:
                        val = 0.0
                        for ii in range(n):
                            jj = m - ii
                            if 0 <= jj < n:
                                val += a_mod[ii] * a_mod[jj]
                        max_est = max(max_est, val)
                    if max_est < best_val:
                        best_val = max_est
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                st = a.sum()
                if st < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a / st
                    new_max = fast_conv_self(a).max()
                    sc = 2.0 * n * new_max
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_pert = max(10, n // 25)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
        else:
            no_improve = 0

    return best_a, best_score


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Using solver (fallback): SCS")
    return cp.SCS


def structured_init(n, rng):
    choice = int(rng.integers(0, 12))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 1:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 2:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3 * np.exp(-4.0 * (x - 0.25) ** 2) + 0.3 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(10 * np.pi * x) + 0.1 * np.cos(18 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 5:
        a = (np.sin(np.pi * x) + 0.3 * np.sin(2 * np.pi * x) + 0.1) ** 2
        a = np.maximum(a, 0.0)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x + 1.0) + 0.1 * np.cos(8 * np.pi * x + 0.5)
        a = np.maximum(a, 0.05)
    elif choice == 7:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.05)
    elif choice == 8:
        a = np.exp(-3.0 * (x - 0.3) ** 2) + np.exp(-3.0 * (x - 0.7) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.5) ** 2)
    elif choice == 9:
        a = 0.3 + 0.4 * np.sin(np.pi * 2 * x) ** 2 + 0.3 * np.cos(np.pi * 4 * x) ** 2
        a = np.maximum(a, 0.05)
    elif choice == 10:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.cos(2 * np.pi * 11 * x + 0.7)
        a = np.maximum(a, 0.05)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
