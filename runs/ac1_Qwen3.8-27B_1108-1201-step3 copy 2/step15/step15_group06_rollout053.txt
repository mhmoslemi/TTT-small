        return float('inf')


def water_fill(x, S):
    n = len(x)
    if n == 0:
        return x
    x_flat = np.asarray(x, dtype=np.float64).copy()
    hi = x_flat.max()
    lo = x_flat.min() - S
    for _ in range(50):
        tau = 0.5 * (lo + hi)
        s = np.sum(np.maximum(x_flat - tau, 0.0))
        if s > S:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    a = np.maximum(x_flat - tau, 0.0)
    s = a.sum()
    if s > 1e-30:
        a = a * S / s
    return a


def compute_multipeak_grad(a, conv, top_k=5, temperature=0.05):
    n = len(a)
    size = 2 * n - 1
    if top_k >= size:
        w = np.ones(size) / size
    else:
        sorted_idx = np.argsort(conv)[::-1]
        top_idx = sorted_idx[:top_k]
        top_vals = conv[top_idx]
        max_val = top_vals[0]
        if max_val < 1e-30:
            w = np.zeros(size)
            w[top_idx[0]] = 1.0
            return _grad_from_weights(a, w, n)
        exp_vals = np.exp(-(top_vals - max_val) / (temperature * max_val + 1e-30))
        weights = exp_vals / exp_vals.sum()
        w = np.zeros(size)
        w[top_idx] = weights
    return _grad_from_weights(a, w, n)


def _grad_from_weights(a, w, n):
    a_flipped = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    Af = rfft(a_flipped, fft_size)
    Wf = rfft(w, fft_size)
    conv_result = irfft(Af * Wf, fft_size)
    return 2.0 * conv_result[n - 1:2 * n - 1]


def subgradient_nesterov(a0, time_limit, seed=0, verbose=True, top_k=5, temp=0.05,
                         lp_every=200, lp_solver=None, n_max_iter=2000000):
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    prev_a = a.copy()
    t0 = time.monotonic()
    stall = 0
    stall_thresh = 500

    for t in range(1, n_max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M = conv_y.max()

        if t % lp_every == 0 and lp_solver is not None:
            b = _lp_best_response(y, M, lp_solver, conv_y, n)
            if b is not None:
                b = b * S / b.sum()
                y = 0.7 * y + 0.3 * b
                y = np.maximum(y, 0.0)
                y = y * S / y.sum()
                continue

        g = compute_multipeak_grad(y, conv_y, top_k=top_k, temperature=temp)
        g = g - g.mean()

        eta = 0.5 / (1.0 + 0.002 * t)
        x = y - eta * g
        a_new = water_fill(x, S)

        momentum = min(t / (t + 2.0), 0.9)
        if t > 1:
            y_new = a_new + momentum * (a_new - prev_a)
            y = water_fill(y_new, S)
        else:
            y = a_new.copy()

        prev_a = a_new
        a = a_new

        if t % 10 == 0:
            sc = fast_autocorr(y).max()
            if sc < best_score:
                best_score = sc
                best_a = y.copy()
                stall = 0
            else:
                stall += 1

            sc_a = fast_autocorr(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
                stall = 0

        if stall > stall_thresh:
            y = best_a.copy()
            a = best_a.copy()
            prev_a = best_a.copy()
            stall = 0
            if verbose and t < n_max_iter:
                elapsed = time.monotonic() - t0
                print(f"      SG restart at t={t} ({elapsed:.0f}s): best={best_score:.10f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG t={t} ({elapsed:.0f}s): M_y={M:.8f}, best={best_score:.10f}")

    return best_a, best_score


def _lp_best_response(y, M, solver, conv_y, n, active_frac=0.15):
    size = 2 * n - 1
    threshold = M * (1.0 - active_frac)
    active = np.where(conv_y >= threshold)[0]
    if len(active) < 10:
        active = np.where(conv_y >= M * 0.6)[0]
    if len(active) > 0.8 * size:
        active = np.where(conv_y >= M * 0.95)[0]

    T = np.zeros((len(active), n))
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[row, j] = y[k - j]

    b = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-10:
        return None
    return val


def kaczmarz_peak_reduce(a0, time_limit, seed=0, verbose=True, n_iter=50000):
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        peak_lag = int(np.argmax(conv))

        if M < best_score * (1 - 1e-10):
            best_score = M
            best_a = a.copy()

        grad_pk = np.zeros(n)
        lo = max(0, peak_lag - n + 1)
        hi = min(n, peak_lag + 1)
        for j in range(lo, hi):
            grad_pk[j] = 2.0 * a[peak_lag - j]

        eta = 0.3 / (1.0 + 0.001 * it)
        x = a - eta * (grad_pk - grad_pk.mean())
        a_new = water_fill(x, S)

        sc = fast_autocorr(a_new).max()
        if sc < M + 1e-12:
            a = a_new
        else:
            a = a_new * 0.9 + a * 0.1
            a = water_fill(a, S)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      KZ it={it} ({elapsed:.0f}s): M={M:.8f}, best={best_score:.10f}")

    return best_a, best_score


def coord_descent_fast(a0, time_limit, seed=0, verbose=True):
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    phi = 0.6180339887498949

    for _pass in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < S * 1e-10:
                continue

            lo = 0.0
            hi = ai * 2.5 + S / n

            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            at1 = a.copy()
            at1[i] = c1
            at1 = at1 * S / at1.sum()
            f1 = fast_autocorr(at1).max()

            at2 = a.copy()
            at2[i] = c2
            at2 = at2 * S / at2.sum()
            f2 = fast_autocorr(at2).max()

            for _ in range(25):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    at1 = a.copy()
                    at1[i] = c1
                    at1 = at1 * S / at1.sum()
                    f1 = fast_autocorr(at1).max()
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    at2 = a.copy()
                    at2[i] = c2
                    at2 = at2 * S / at2.sum()
                    f2 = fast_autocorr(at2).max()

            opt_val = min(f1, f2)
            if opt_val < best_score - 1e-14:
                a[i] = c1 if f1 < f2 else c2
                a = a * S / a.sum()
                cv = fast_autocorr(a)
                best_score = cv.max()
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and _pass % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD pass={_pass} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_cd(a0, time_limit, seed=0, verbose=True):
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for _pass in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            ai, aj = a[i], a[j]
            if ai < S * 1e-10 and aj < S * 1e-10:
                continue

            base = fast_autocorr(a).max()
            best_d = 0.0
            best_v = base

            for direction in [1, -1]:
                src = ai if direction == 1 else aj
                if src < 1e-15:
                    continue
                for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                    d = src * frac
                    if d <= 0 or d > src:
                        continue
                    at = a.copy()
                    if direction == 1:
                        at[i] = ai - d
                        at[j] = aj + d
                    else:
                        at[i] = ai + d
                        at[j] = aj - d
                    st = at.sum()
                    if st < 1e-15:
                        continue
                    at = at * S / st
                    v = fast_autocorr(at).max()
                    if v < best_v - 1e-14:
                        best_v = v
                        best_d = d if direction == 1 else -d

            if abs(best_d) > 1e-15:
                at = a.copy()
                if best_d > 0:
                    at[i] -= best_d
                    at[j] += best_d
                else:
                    at[i] += abs(best_d)
                    at[j] -= abs(best_d)
                st = at.sum()
                if st > 1e-15:
                    at = at * S / st
                    v = fast_autocorr(at).max()
                    if v < best_score - 1e-14:
                        a = at
                        best_score = v
                        best_a = a.copy()
                        pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and _pass % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PCD pass={_pass} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif style == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(5 * np.pi * x) + 0.1 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 7:
        a = np.sin(np.pi * x) ** 0.8 * (1.0 + 0.1 * np.cos(9 * np.pi * x))
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
