import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, w):
    """Compute gradient of T*log(sum exp(conv(a,a)/T)) w.r.t. a.
    w is the softmax weights: w[k] = exp(conv[k]/T) / sum(exp(conv/T))
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    c1 = irfft(np.conj(W) * A, fft_size)[:n]
    c2 = irfft(np.conj(A) * W, fft_size)[:n]
    return c1 + c2

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if v.sum() <= S and (v <= 1e-15).all():
        return v.copy()
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho_candidates = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho_candidates) == 0:
        rho = n - 1
    else:
        rho = rho_candidates[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * S / s

def make_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    S = np.sqrt(2.0 * n)
    
    # Uniform
    inits.append(np.ones(n) * S / n)

    x = np.linspace(0, 1, n)
    cos_bump = np.cos(np.pi * (x - 0.5)) ** 2
    cos_bump = np.maximum(cos_bump, 0)
    inits.append(normalize_sum(cos_bump, S))
    
    # Gaussian bump
    x = np.linspace(0, 1, n)
    gauss = np.exp(-20 * (x - 0.5) ** 2)
    inits.append(normalize_sum(gauss, S))
    
    # Two bumps

    two_bump = np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)
    inits.append(normalize_sum(two_bump, S))
    
    # Random smooth
    smooth = np.cumsum(rng.normal(0, 1, n))
    smooth = smooth - smooth.min()
    inits.append(normalize_sum(smooth, S))

    alt = np.ones(n)
    alt[1::2] = 0.5
    inits.append(normalize_sum(alt, S))
    
    # Triangular

    inits.append(normalize_sum(tri.astype(float), S))
    
    # Inverse triangular
    inv_tri = np.maximum(tri - n/2, 0) + 0.1
    inits.append(normalize_sum(inv_tri.astype(float), S))
    
    return inits

        if remaining < 0.5:

        # Estimate iterations per second (rough)
        # For n=1600: fft_size=4096, each iter has ~4 FFTs of size 4096
        # ~4 * 4096 * log(4096) * 2 ~ 330K flops per iter, say ~0.05ms
        # But Python overhead dominates, so maybe ~0.5-2ms per iter

                if mc < best_maxconv - 1e-12:
                    best_maxconv = mc
                    best_a = a_new.copy()
                    stalling_count = 0
                    if verbose and label:
                        elapsed = time.monotonic() - t0
                        print(f"    [{label}] T={T:.4f} iter={it} ({elapsed:.0f}s): maxconv={best_maxconv:.8f}")
                else:
                    stalling_count += 1
            
            if stalling_count > 10:
                break
        
        # Update a to current best
        a = best_a.copy()
    
    # Final check on y
    mc_y = fast_autocorr(y).max()
    if mc_y < best_maxconv:
        best_maxconv = mc_y
        best_a = y.copy()
    
    return best_a, best_maxconv

def grad_guided_cd(a0, time_limit, rng, verbose=True, label=""):
    """Gradient-guided pair coordinate descent."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_sum(a, S)
    
    conv = fast_autocorr(a)
    best_maxconv = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    n_moves = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 50:
            break
        
        # Compute gradient for guidance
        conv_curr = fast_autocorr(a)
        T = 0.01  # Very peaked softmax to focus on the max
        conv_shifted = conv_curr - conv_curr.max()
        exp_vals = np.exp(conv_shifted / T)
        w = exp_vals / exp_vals.sum()
        grad = compute_smooth_grad(a, w)
        
        # Find best pair (i, j) to move mass from i to j
        # We want grad[j] - grad[i] < 0 (i.e., grad[i] > grad[j])
        # among positions where a[i] > 0
        active = a > S * 1e-8
        if active.sum() < 2:
            break
        
        # Sample candidate pairs
        n_samples = min(50, n)
        i_candidates = rng.choice(np.where(active)[0], size=min(n_samples, active.sum()), replace=False)
        
        improved = False
        for i in i_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < S * 1e-8:
                continue
            
            # Find best j to move to
            j_candidates = rng.permutation(n)[:20]
            for j in j_candidates:
                if j == i or a[j] < S * 1e-12 and a[i] < S * 1e-12:
                    continue
                
                # Try moving a fraction of a[i] to a[j]
                frac = 0.1
                delta = a[i] * frac
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = normalize_sum(np.maximum(a_test, 0), S)
                
                conv_test = fast_autocorr(a_test)
                if conv_test.max() < best_maxconv - 1e-12:
                    a = a_test
                    conv = conv_test
                    if conv_test.max() < best_maxconv:
                        best_maxconv = conv_test.max()
                        best_a = a.copy()
                        no_improve = 0
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        n_moves += 1
        
        if verbose and n_moves % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{label}] GD-CD moves={n_moves} ({elapsed:.0f}s): maxconv={best_maxconv:.8f}")
    
    return best_a, best_maxconv

def ld_cd(a0, time_limit, rng, verbose=True, label=""):
    """Line-search pair coordinate descent."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_sum(a, S)
    
    conv = fast_autocorr(a)
    best_maxconv = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        no_improve += 1
        if no_improve > 5:
            break
        
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < S * 1e-8:
                continue
            
            # Try moving to nearby positions
            for offset in range(1, 30):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    
                    # Quick check: binary search for best delta
                    lo, hi = 0.0, a[i] * 0.5
                    best_delta = 0.0
                    best_val = conv.max()
                    
                    for _ in range(5):
                        mid = (lo + hi) / 2
                        if mid <= 0:
                            lo = mid
                            continue
                        a_t = a.copy()
                        a_t[i] -= mid
                        a_t[j] += mid
                        if a_t[i] < 0:
                            hi = mid
                            continue
                        a_t = normalize_sum(a_t, S)
                        conv_t = fast_autocorr(a_t)
                        if conv_t.max() < best_val - 1e-12:
                            best_val = conv_t.max()
                            best_delta = mid
                            lo = mid
                        else:
                            hi = mid
                    
                    if best_delta > 0:
                        a[i] -= best_delta
                        a[j] += best_delta
                        a = normalize_sum(np.maximum(a, 0), S)
                        conv = fast_autocorr(a)
                        if conv.max() < best_maxconv - 1e-12:
                            best_maxconv = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            break
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{label}] LD-CD pass={n_passes} ({elapsed:.0f}s): maxconv={best_maxconv:.8f}")
    
    return best_a, best_maxconv

def binning_coarse_fine(a0, n_fine, time_limit, rng, verbose=True):
    """Coarse-to-fine optimization using binning."""
    n_coarse = max(100, n_fine // 8)
    S_coarse = np.sqrt(2.0 * n_coarse)
    S_fine = np.sqrt(2.0 * n_fine)
    
    # Bin the fine sequence into coarse
    a = np.asarray(a0, dtype=np.float64)
    a = normalize_sum(a, S_fine)
    
    # Create coarse by summing bins
    bins = np.array_split(np.arange(len(a)), n_coarse)
    b = np.array([a[bnd].sum() for bnd in bins])
    b = normalize_sum(b, S_coarse)
    
    # Optimize coarse
    t_half = time_limit * 0.5
    b_opt, _ = smooth_optimize(b, t_half, verbose=False, label="coarse")
    
    # Interpolate to fine
    x_old = np.linspace(0, 1, n_coarse, endpoint=False)
    x_new = np.linspace(0, 1, n_fine, endpoint=False)
    a_fine = np.interp(x_new, x_old, b_opt)
    a_fine = np.maximum(a_fine, 0)
    a_fine = normalize_sum(a_fine, S_fine)
    
    # Fine optimization with remaining time
    t_remaining = time_limit - (time.monotonic() - (time.monotonic() - time_limit))
    # Just use remaining
    return a_fine
