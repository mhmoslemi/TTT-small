        b, duals = lp_solve(y, M, solver, active)
        if b is None or b.sum() < 1e-15:
            b, duals = lp_solve(y, M, solver, None)
        if b is None or b.sum() < 1e-15:
            continue

        b_sum = b.sum()
        g = b * norm / b_sum

        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * (1 + 5e-4))[0]
            if len(violated) > 5:
                b_full, _ = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    g = b * norm / b.sum()

        alpha = 2.0 / (t + 2.0)
        if use_multiplicative and t > 50:
            y_safe = np.maximum(y, 1e-10)
            g_safe = np.maximum(g, 1e-10)
            log_g = np.log(g_safe)
            log_y = np.log(y_safe)
            y_new = np.exp((1.0 - alpha) * log_y + alpha * log_g)
        else:
            y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        if no_improve_count == perturb_interval:
            n_perturb = max(20, n // 15)
            idx_perturb = rng_local.choice(n, size=n_perturb, replace=False)
            scale = 1.0 + 0.3 * rng_local.standard_normal(n_perturb)
            y[idx_perturb] *= scale
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            a = y.copy()
            no_improve_count = 0
            perturb_interval = min(perturb_interval + 50, 500)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")

    return best_a, best_score
