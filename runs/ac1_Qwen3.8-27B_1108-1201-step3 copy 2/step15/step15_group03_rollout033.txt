import cvxpy as cp
import time
from scipy.fft import rfft, irfft

# Fast convolution
def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

# LP best response
def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def fast_conv_self(a):

    size = 2 * n - 1

    A = rfft(a, fft_size)

def build_T(a):
    """Build the convolution design matrix T where (T @ b)[k] = (a*b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None):
    """Solve LP: max sum(b) s.t. T_a @ b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_solve_sparse(a, M, solver, active=None):

        rows = []
        cols = []
        vals = []
        for k in active:
            # Row k of T: T[k, i] = a[i] if 0 <= k-i < n
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[i])
        import scipy.sparse as sp
        T_sub = sp.csr_matrix((vals, (rows, cols)), shape=(len(active), n))
        constraints = [T_sub @ b_var <= M]

        constraints = [T @ b_var <= M]

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent for the max autocorrelation."""
    n = len(a0)

    size = 2 * n - 1

    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        n_steps = max(50, min(500, int(remaining * 8)))

        for it in range(n_steps):

                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Subtract mean to maintain sum(a) = 1 (projection onto sum=1 hyperplane)
            grad = grad - np.mean(grad)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = 1.0 / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.02 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:

            else:

            sc = fast_conv_self(a).max()
            if sc < best_score:

                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best_conv={best_score:.8f}")

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def normalize_a(a, target_sum=1.0):
    a = np.maximum(a, 0.0)

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(120.0, 0.08, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(40, min(400, int(remaining * 10)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            grad = grad - np.mean(grad)

            grad
_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = 1.0 / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    no_improve = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        threshold_1 = M * 0.88
        active = np.where(conv_y >= threshold_1)[0]
        if len(active) < 25:
            threshold_2 = M * 0.55
            active = np.where(conv_y >= threshold_2)[0]

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum

        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.005)[0]
            if len(violated) > 5:
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    g = b / b.sum()

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0

        if no_improve >= 80:
            rng_p = np.random.default_rng(t)
            n_perturb = max(15, n // 15)
            idx_perturb = rng_p.choice(n, size=n_perturb, replace=False)
            a[idx_perturb] *= (1.0 + 0.12 * rng_p.standard_normal(n_perturb))
            a = normalize_a(a, 1.0)
            y = a.copy()
            no_improve = 0

        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def qp_polish(a0, time_limit, top_k=40, verbose=True):
    """
    Linearized QP polish: freeze T at current a, solve QP to minimize
    weighted sum of squares of top-K convolution values.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    for outer in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Select top-K lags
        top_idx = np.argsort(conv)[::-1][:top_k]
        # Weight: emphasize lags near the peak
        weights = conv[top_idx] / conv[top_idx].max()
        weights = weights ** 2

        # Build the QP: minimize sum_k w_k * (T[:, k] @ a_new)^2 + lam * ||a_new - a||^2
        # s.t. a_new >= 0, sum(a_new) = 1
        # This is: minimize a_new^T (T_k^T W T_k) a_new - 2*(T_k^T W T_k a)^T a_new + const
        #         + lam * a_new^T a_new - 2*lam*a^T a_new + const

        T = build_T(a)
        T_top = T[top_idx, :]  # (top_k, n)

        # Q = T_top^T * diag(weights) * T_top + lam * I
        lam = 0.1
        W_sqrt = np.sqrt(weights)
        T_weighted = T_top * W_sqrt[:, None]  # (top_k, n)
        Q = T_weighted.T @ T_weighted + lam * np.eye(n)
        # Linear term: -2 * (T_weighted^T @ (T_weighted @ a) + lam * a)
        c = -(T_weighted.T @ (T_weighted @ a) + lam * a)

        # Solve QP: min 0.5 * a^T Q a + c^T a s.t. a >= 0, sum(a) = 1
        a_var = cp.Variable(n)
        objective = cp.quad_form(a_var, cp.psd_wrap(Q)) + c @ a_var
        constraints = [a_var >= 0, cp.sum(a_var) == 1]
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        new_max = fast_conv_self(a_new).max()
        if new_max < best_score - 1e-12:
            best_score = new_max
            best_a = a_new.copy()
            a = a_new
        else:
            # Try damped update
            t_damp = 0.5
            a_damped = (1 - t_damp) * a + t_damp * a_new
            a_damped = normalize_a(a_damped, 1.0)
            new_max_d = fast_conv_self(a_damped).max()
            if new_max_d < best_score - 1e-12:
                best_score = new_max_d
                best_a = a_damped.copy()
                a = a_damped
            else:
                break

        if verbose and outer % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QP iter={outer} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-14:
                continue

            peak_threshold = cur_max * 0.93
            peak_lags = np.where(conv >= peak_threshold)[0]

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j_s = (kl - i + n) % n
                j_candidates.add(j_s)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:20]:
                if j == i:
                    continue
                aj = a[j]
                if aj < 1e-14 and ai < 1e-14:
                    continue

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 30
                deltas = np.linspace(0.0, ai * 0.98, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize_a(a, 1.0)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_imp += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = normalize_a(a, 1.0)
        else:
            no_improve = 0

        if verbose and total_imp % 300 == 0 and total_imp > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_imp} ({elapsed:.0f}s): best={best_score:.8f}")

def mirror_descent_optimize(a0, time_limit, verbose=True):
    """Mirror descent with entropic regularization for the smoothed max."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_iters = int(time_limit * 20)  # roughly 20 iterations per second target
    n_iters = min(n_iters, 50000)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        # Adaptive temperature
        T = max(0.05, max_c * 0.3 / (1.0 + 0.005 * it))
        shifted = (conv - max_c) / T
        shifted = np.maximum(shifted, -80.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        # Mirror descent step: a_new = a * exp(-eta * grad / sum(a))
        eta = 1.0 / (n ** 0.5 * (1.0 + 0.01 * it))
        log_ratio = -eta * grad / a
        log_ratio = np.clip(log_ratio, -5.0, 5.0)
        a_new = a * np.exp(log_ratio)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def gen_hann(n, rng):
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 0.0)
    a += 0.005 * rng.random(n)
    return a / a.sum()


def gen_gaussian(n, rng):
    x = np.linspace(0.0, 1.0, n)
    sigma = 0.2 + 0.15 * rng.random()
    center = 0.5
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a += 0.005 * rng.random(n)
    return a / a.sum()


def gen_parabolic(n, rng):
    x = np.linspace(0.0, 1.0, n)
    a = 4.0 * x * (1.0 - x)
    a += 0.005 * rng.random(n)
    return a / a.sum()


def gen_binary_like(n, rng):
    a = rng.random(n)
    a = (a > 0.5).astype(float)
    a += 0.1 * rng.random(n)
    return a / a.sum()


def gen_raised_cosine_power(n, rng):
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + 2.0 * rng.random()
    a = (0.5 * (1.0 + np.cos(2.0 * np.pi * x))) ** p
    a += 0.005 * rng.random(n)
    return a / a.sum()


def gen_bimodal(n, rng):
    x = np.linspace(0.0, 1.0, n)
    c1 = 0.25 + 0.1 * rng.random()
    c2 = 0.65 + 0.1 * rng.random()
    a = np.exp(-8.0 * (x - c1) ** 2) + 0.8 * np.exp(-8.0 * (x - c2) ** 2)
    a += 0.005 * rng.random(n)
    return a / a.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update_seq(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-initialization
    print("\n=== Phase 0: Multi-initialization ===")
    if best_seq is None:
        n_base = 1600
        best_seq = gen_hann(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Try a few initializations at the same length
    inits = [
        (gen_hann, "hann"),
        (gen_gaussian, "gauss"),
        (gen_parabolic, "para"),
        (gen_raised_cosine_power, "rcp"),
        (gen_bimodal, "bimodal"),
    ]
    for gen_fn, label in inits:
        if time.monotonic() - t_global > time_budget * 0.05:
            break
        a_try = gen_fn(n_base, rng)
        sc = eval_fn(a_try)
        print(f"  Init {label}: score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_try.copy()
            print(f"  -> New best init: {label}")

    # Phase 1: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update_seq(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update_seq(a_out, "lp1")

    # Phase 3: QP Polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: QP Polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = qp_polish(best_seq, t3, top_k=50, verbose=True)
        try_update_seq(a_out, "qp1")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov r2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        try_update_seq(a_out, "lp2")

    # Phase 5: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: Mirror Descent n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = mirror_descent_optimize(best_seq, t5, verbose=True)
        try_update_seq(a_out, "mirror")

    # Phase 6: QP Polish round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 6: QP Polish r2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = qp_polish(best_seq, t6, top_k=60, verbose=True)
        try_update_seq(a_out, "qp2")

    # Phase 7: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 7: Pair CD n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t7, rng, verbose=True)
        try_update_seq(a_out, "pcd1")

    # Phase 8: Try different length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n_base + 200
        print(f"\n=== Phase 8: Alt length n={n_alt} (remaining={remaining:.0f}s) ===")
        init_alt = interpolate_seq(best_seq, n_alt)
        t8a = min(remaining * 0.4, 120.0)
        if t8a > 20:
            a_out, _ = nesterov_lp_optimize(init_alt, t8a, solver, verbose=False)
            try_update_seq(a_out, f"lp_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            n_alt2 = n_base - 200
            if n_alt2 > 400:
                print(f"\n=== Phase 8b: Alt length n={n_alt2} ===")
                init_alt2 = interpolate_seq(best_seq, n_alt2)
                t8b = min(remaining * 0.5, 100.0)
                if t8b > 20:
                    a_out, _ = nesterov_lp_optimize(init_alt2, t8b, solver, verbose=False)
                    try_update_seq(a_out, f"lp_n{n_alt2}")

    # Phase 9: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final polish n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t9a = min(remaining * 0.4, 60.0)
        if t9a > 10:
            a_out, _ = lse_gradient_descent(best_seq, t9a, verbose=False)
            try_update_seq(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t9b = min(remaining * 0.5, 40.0)
            if t9b > 8:
                a_out, _ = pair_cd_optimize(best_seq, t9b, rng, verbose=False)
                try_update_seq(a_out, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
