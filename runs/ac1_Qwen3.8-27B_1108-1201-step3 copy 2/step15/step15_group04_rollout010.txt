def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    # Build constraint matrix
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    return val

            break
        n_steps = max(20, min(500, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            # LSE weights
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient via FFT: grad = 2 * cross_correlation(w, a)
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # w * a in convolution sense
            g = irfft(W * np.conj(A), fft_size)[:size]
            # Actually, gradient of sum_k w[k] * conv[k] wrt a[i]
            # = 2 * sum_k w[k] * (a[k-i] + a[i-k]) appropriately
            # This is 2 * (w cross-correlated with a)
            # cross_corr(w, a)[i] = sum_k w[k] * a[k-i]
            # We can compute this as ifft(fft(w) * conj(fft(a)))
            g1 = irfft(W * np.conj(A), fft_size)[:n]  # sum_k w[k]*a[k-i] (circular)
            g2 = irfft(np.conj(W) * A, fft_size)[:n]  # hmm, need to be careful
            
            # Actually for the linear convolution gradient:
            # d(conv[k])/da[i] involves a[k-i] and a[i-k]
            # sum_k w[k] * a[k-i] is the cross-correlation (w ⋆ a)[i]
            # sum_k w[k] * a[i-k] is (w * a reversed)[i] = (reversed_w ⋆ a)[i]
            
            # Let me use a direct approach for correctness
            # grad[i] = 2 * sum_k w[k] * a[k-i] (where index is valid)
            #           + sum_k w[k] * a[i-k] (where index is valid, for the diagonal case)
            
            # Simpler: grad = 2 * ifft(FFT(w) * conj(FFT(a)))[:n] for circular
            # But we need linear. Let me just use the full approach.
            
            # For linear convolution conv[k] = sum_j a[j]*a[k-j]:
            # d(conv[k])/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
            #                    - a[i] (if 2i=k, to avoid double counting)
            
            # This is complex to compute via FFT for linear case.
            # Let me use a different formulation:
            # The gradient of the LSE objective is:
            # grad[i] = sum_k w[k] * d(conv[k])/da[i]
            #         = sum_k w[k] * (a[k-i] * 1{0<=k-i<n} + a[i-k] * 1{0<=i-k<n} - a[i] * 1{2i=k})
            
            # The first term is: sum_k w[k] * a[k-i] = (w ⋆ a)_linear[i]
            # The second term is: sum_k w[k] * a[i-k] = (w_reversed ⋆ a)_linear[i] ... 
            # Actually sum_k w[k] * a[i-k] = sum_j w[i+j] * a[j] ... no
            
            # Let me just use the fact that:
            # sum_k w[k] * a[k-i] for k-i in [0,n) = sum_{j=0}^{n-1} w[i+j] * a[j] (for i+j < size)
            # This is a "sliding dot product" which can be computed as a convolution.
            
            # Actually, the cleanest way: 
            # Let W[k] = w[k] for k in [0, size), 0 otherwise
            # Let A[j] = a[j] for j in [0, n), 0 otherwise
            # Then sum_k w[k] * a[k-i] = (W * A_flipped)[-i] where A_flipped[j] = A[n-1-j]
            # Hmm, this is getting complicated. Let me just use the FFT approach:
            
            # For the linear convolution gradient, the cleanest formulation is:
            # grad = 2 * IFFT(FFT(w_padded) * conj(FFT(a_padded)))[:n]
            # where both are padded to fft_size, and we take the first n elements.
            # But this gives the CIRCULAR cross-correlation, not linear.
            
            # For our purposes (large n, the boundary effects are small), let's use the circular approximation.
            # This is a valid approximation when n is large relative to the "effective support" of w.
            
            grad = 2.0 * irfft(W * np.conj(A), fft_size)[:n]
            
            # Project and normalize
            grad = np.maximum(grad, 0.0)  # Only move in valid direction (for multiplicative update)
            # Actually for additive: we can have negative gradient components
            
            # Use adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.02 * it)
                
                # Line search
                for trial in range(10):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= max_c + 0.1 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score

def adam_adaptive_optimize(a0, time_limit, verbose=True):
    """Adam-based gradient descent with peak tracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.1
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Temperature schedule for LSE
    T_start = 20.0
    T_end = 0.5
    n_steps_total = 2000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, t / n_steps_total)
        T_val = T_start * (T_end / T_start) ** progress
        
        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # LSE weights
        shifted = (conv - max_c) / T_val
        shifted = np.maximum(shifted, -300.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        
        # Gradient via FFT
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad = 2.0 * irfft(W * np.conj(A), fft_size)[:n]
        
        # Also add a small component to push towards reducing the overall peak
        # by using the top-k peak locations
        top_k = min(5, size)
        peak_idx = np.argsort(conv)[-top_k:]
        peak_weight = np.zeros(size)
        peak_weight[peak_idx] = 1.0 / top_k
        
        P = rfft(peak_weight, fft_size)
        grad_peak = 2.0 * irfft(P * np.conj(A), fft_size)[:n]
        grad = 0.7 * grad + 0.3 * grad_peak
        
        # Adam update
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        sc = fast_conv_self(a).max()
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Decay learning rate
        lr *= 0.9995
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): best={best_score:.8f}")
        
        if best_score < norm * 0.75:  # Target: objective < 1.5 means max_conv < 0.75 * norm (when sum=norm)
            # Actually objective = 2n*max_conv/sum². If sum=sqrt(2n), objective = 2n*max_conv/(2n) = max_conv
            # So we want max_conv < 1.5027
            if best_score < 1.5027:
                print(f"    *** Achieved target! score={best_score:.6f}")
                break

    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    no_improve_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening - more aggressive
        if t < 30:
            threshold = 0.4
        elif t < 100:
            threshold = 0.6
        elif t < 300:
            threshold = 0.75
        else:
            threshold = 0.85
        
        active = np.where(conv_y > M * threshold)[0]
        if len(active) < 20:
            active = np.where(conv_y > M * 0.3)[0]

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov / momentum update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        # Periodic perturbation
        if no_improve_count == 80:
            rng_p = np.random.default_rng(t)
            n_perturb = max(n // 30, 10)
            idx_perturb = rng_p.choice(n, size=n_perturb, replace=False)
            y[idx_perturb] *= (1.0 + 0.3 * rng_p.standard_normal(n_perturb))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

        if best_score < 1.5027:
            print(f"    *** LP achieved target! score={best_score:.6f}")
            break

    return best_a, best_score

def multiplicative_ascent(a0, time_limit, verbose=True):
    """Multiplicative weight update (exponential gradient) for the dual problem."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # We solve the dual: minimize max_k conv(a,a)[k] s.t. sum(a)=norm, a>=0
    # Multiplicative update: a[i] *= exp(-eta * d(max_conv)/da[i])
    # = exp(-eta * 2 * sum_k w[k] * a[k-i])
    
    eta = 0.5
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        
        # Soft weights (temperature annealing)
        T_val = max(0.1, 5.0 / (1 + t * 0.01))
        shifted = (conv - M) / T_val
        shifted = np.maximum(shifted, -300.0)
        w = np.exp(shifted)
        w = w / w.sum()
        
        # Gradient
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad = 2.0 * irfft(W * np.conj(A), fft_size)[:n]
        
        # Multiplicative update
        eta_t = eta / (1 + 0.05 * t)
        a_new = a * np.exp(-eta_t * grad)
        a_new = np.maximum(a_new, 1e-15)  # Keep strictly positive for multiplicative
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        sc = fast_conv_self(a).max()
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MulAscent t={t} ({elapsed:.0f}s): best={best_score:.8f}")
        
        if best_score < 1.5027:
            break

    return best_a, best_score

def evaluate_local(a):
    """Compute the objective locally."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def structured_init(n, rng):
    """Generate diverse initializations, emphasizing asymmetry."""
    choice = rng.integers(0, 14)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        # Exponential decay (highly asymmetric)
        a = np.exp(-3.0 * x)
    elif choice == 1:
        # Power law
        a = (1.0 - x) ** 2.0 + 0.01
    elif choice == 2:
        # Sine raised (mildly asymmetric with offset)
        a = np.sin(np.pi * (x + 0.1)) ** 2.0 + 0.01
    elif choice == 3:
        # Two peaks (asymmetric)
        a = np.exp(-8.0 * (x - 0.2) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        # Cosine with phase shift
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * (x - 0.1))
        a = np.maximum(a, 0.01)
    elif choice == 5:
        # Linear ramp
        a = x + 0.05
    elif choice == 6:
        # Inverse linear
        a = 1.0 - 0.7 * x + 0.05
    elif choice == 7:
        # Gaussian + noise
        a = np.exp(-2.0 * (x - 0.3) ** 2)
        a += 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0.0)
    elif choice == 8:
        # Sawtooth
        a = (x * 7.0) % 1.0
        a = a ** 1.5 + 0.05
    elif choice == 9:
        # Beta-distribution shape
        a = x ** 2.0 * (1.0 - x) ** 0.5 + 0.01
    elif choice == 10:
        # Multi-frequency
        a = 0.3 + 0.3 * np.cos(4 * np.pi * x - 0.5) + 0.2 * np.cos(9 * np.pi * x + 1.0)
        a = np.maximum(a, 0.01)
    elif choice == 11:
        # Step-like
        a = np.where(x < 0.4, 1.0, 0.3)
        a += 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0.0)
    elif choice == 12:
        # Logistic
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.3)))
    else:
        # Random smooth
        a = np.abs(rng.standard_normal(n))
        a = np.convolve(a, np.ones(20) / 20, mode='same')
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def peak_targeting(a0, time_limit, verbose=True):
    """Directly target reduction of the current peak by redistributing mass."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]
        
        # Find which elements contribute most to the peak
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i]
        # Contribution of a[i] is a[peak_idx - i] (if valid)
        
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = a[j]
        
        # We want to reduce the peak by moving mass from high-contributing
        # elements to low-contributing ones
        # But we also need to consider the effect on other lags
        
        # Simple approach: reduce the element with highest "peak leverage"
        # and add to a random low-value element
        
        # Find top contributors to peak
        top_contrib = np.argsort(contrib)[-10:]
        # Find low-value elements
        low_val = np.argsort(a)[:20]
        
        improved = False
        for _ in range(50):
            i = top_contrib[rng.integers(len(top_contrib))]
            j = low_val[rng.integers(len(low_val))]
            if i == j:
                continue
            
            # Try moving a small fraction of a[i] to a[j]
            delta = 0.05 * a[i]
            if delta < 1e-10:
                continue
            
            a[i] -= delta
            a[j] += delta
            
            if a[i] < 0:
                a[i] = 0
                a[j] -= a[i]  # shouldn't happen
                continue
            
            s_new = a.sum()
            if s_new < 1e-15:
                a[i] += delta
                a[j] -= delta
                continue
            
            a = a * norm / s_new
            new_conv = fast_conv_self(a)
            new_peak = new_conv.max()
            
            if new_peak < best_score - 1e-10:
                best_score = new_peak
                best_a = a.copy()
                improved = True
            else:
                # Revert
                a[j] -= delta
                a[i] += delta
                a = a * norm / a.sum()
        
        if not improved and rng.random() < 0.1:
            # Add some noise
            a += 0.05 * norm * rng.standard_normal(n) / n
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
        
        if verbose:
            elapsed = time.monotonic() - t0
            if elapsed % 10 < 0.1:  # Print occasionally
                print(f"    PeakTarget ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(10):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 1: LSE GD (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=False)
        try_update(a_out, "lse1")

    # Phase 2: Adam optimizer
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 2: Adam (budget={t2:.0f}s) ===")
    if t2 > 20:
