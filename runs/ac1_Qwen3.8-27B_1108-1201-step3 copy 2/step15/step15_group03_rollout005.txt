"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an adaptive multi-method approach:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver_name = None

def get_solver():
    global _solver_name
    if _solver_name is not None:
        return _solver_name
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_name = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver_name = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver_name


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * target_sum / s


def lse_momentum_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with Nesterov momentum and adaptive temperature."""
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, target)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Adaptive temperature schedule: start high, decrease geometrically
    n_stages = 15
    temps = np.geomspace(100.0, 0.05, n_stages)

    y = a.copy()  # Nesterov lookahead
    prev_a = a.copy()

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_budget_remaining(t0, time_limit)
        if remaining < 5:
            break

        n_steps = max(40, min(400, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(y)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = np.maximum(shifted, -80.0)

            if w_sum < 1e-30:

                w[np.argmax(conv)] = 1.0
            else:
                w = w / w_sum

            Y = rfft(y, fft_size)

            g2 = irfft(Y * W, fft_size)[:n]

            # Nesterov momentum update with adaptive step

            if grad_norm < 1e-20:

def time_budget_remaining(t0, limit):
    return max(0.0, limit - (time.monotonic() - t0))


def lp_best_response(a, M, active_indices=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    solver = get_solver()
    n = len(a)
    size = 2 * n - 1

    # Build constraint matrix (sparse would be ideal but dense is OK for n~2000)
    if active_indices is not None and len(active_indices) < size:
        T = np.zeros((len(active_indices), n))
        for row, k in enumerate(active_indices):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    T[row, i] = a[j]
    else:
        T = np.zeros((size, n))
        for i in range(n):
            T[i:i+n, i] = a

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

            break

        # Adaptive active set: include constraints that are "close" to binding
        # Use a tighter threshold for later iterations
        frac = max(0.85, 0.95 - 0.001 * t)
        active = np.where(conv_y >= frac * M)[0]
        if len(active) < 20:
            active = np.where(conv_y >= 0.7 * M)[0]

        b = lp_best_response(y, M, active)
        if b is None:
            b = lp_best_response(y, M, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = normalize(b, target)

        # Verify dropped constraints (only check occasionally for speed)
        if len(active) < size and t % 10 == 0:
            check = fast_conv_cross(y, g)
            violated = np.where(check > M * 1.01)[0]
            if len(violated) > 5:
                b_full = lp_best_response(y, M, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    g = normalize(b_full, target)

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        prev_y = y.copy()
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, target)

        momentum = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, target)

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_targeting(a0, time_limit, research_time_fn=None, verbose=True):
    """Direct subgradient on the peak set with heavy-ball momentum."""
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, target)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    momentum = np.zeros(n)
    beta = 0.7

    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        # Identify peak set (top K peaks)
        k_peaks = min(5, size)
        peak_idx = np.argsort(conv)[-k_peaks:]
        indicator = np.zeros(size)
        indicator[peak_idx] = 1.0 / k_peaks

        # Compute subgradient via FFT
        I = rfft(indicator, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * I, fft_size)[:n]
        g2 = irfft(A * I, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        # Step size with adaptive decay
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-20:
            break
        lr = target / (grad_norm * np.sqrt(1.0 + 0.005 * it))

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, target)
        a = a_new

        # Adapt beta: increase if oscillating, decrease if smooth
        if it > 10:
            # Simple heuristic: decrease beta slowly
            beta = max(0.3, 0.7 - 0.0001 * it)

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakTarget it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Targeted pair coordinate descent: transfer mass between coordinates to reduce peaks."""
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, target)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Find peak lags
        peak_threshold = cur_max * 0.90
        peak_lags = np.where(conv >= peak_threshold)[0]

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < target * 1e-11:
                continue

            # Candidate targets: random + structured (based on peak lags)
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                if 0 <= kl - i < n and kl - i != i:
                    j_candidates.add(kl - i)
                if 0 <= i + kl < n and i + kl != i:
                    j_candidates.add(i + kl)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                aj = a[j]
                if aj < 1e-15 and ai < 1e-15:
                    continue

                # Compute change in convolution when moving delta from i to j
                # (a*a)[k] changes by: +delta*(a[k-j] + a[j-k]) - delta*(a[k-i] + a[i-k])
                # Plus diagonal corrections at k=2i and k=2j
                g = np.zeros(size)
                # Contribution from j (adding)
                for offset in range(-n + 1, n):
                    k = j + offset
                    if 0 <= k < size and 0 <= offset < n:
                        g[k] += a[offset]
                    if 0 <= k < size and 0 <= -offset < n:
                        g[k] += a[-offset]
                # Contribution from i (removing)
                for offset in range(-n + 1, n):
                    k = i + offset
                    if 0 <= k < size and 0 <= offset < n:
                        g[k] -= a[offset]
                    if 0 <= k < size and 0 <= -offset < n:
                        g[k] -= a[-offset]

                # Try several delta values
                n_samples = 15
                max_d = min(ai, target * 0.1) * 0.95
                deltas = np.linspace(0.0, max_d, n_samples)

                for d in deltas:
                    if d < 1e-13:
                        continue
                    new_conv = conv + d * g
                    # Diagonal corrections
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-14:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize(a, target)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                # Perturb to escape local minimum
                n_pert = max(10, n // 25)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                a = normalize(a, target)
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def uniform_interp_seq(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-method optimization with adaptive scheduling:
    - LSE momentum descent for quick progress
    - LP Nesterov for strong local optimization
    - Peak targeting for fine equalization
    - Pair CD for final polish
    - Length exploration for global improvements
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = eval_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LSE Momentum Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 200.0)
    print(f"\n=== Phase 1: LSE Momentum (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_momentum_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 2: LP Nesterov (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, verbose=True)
        try_update(a_out, "lp1")

    # === Phase 3: Peak Targeting ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Peak Targeting (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = peak_targeting(best_seq, t3, verbose=True)
        try_update(a_out, "peak_target")

    # === Phase 4: LP Nesterov Round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov R2 (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, verbose=True)
        try_update(a_out, "lp2")

    # === Phase 5: Length Exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(best_seq)

    # Try a few different lengths
    length_candidates = []
    for delta in [200, 400, -100, 600]:
        nl = n_current + delta
        if 800 <= nl <= 4000:
            length_candidates.append(nl)

    if length_candidates and remaining > 60:
        print(f"\n=== Phase 5: Length Exploration (budget={remaining:.0f}s) ===")
        t_per_len = min(remaining * 0.6 / len(length_candidates), 120.0)
        for nl in length_candidates[:3]:
            if time.monotonic() - t_global > time_budget - 30:
                break
            print(f"  Trying n={nl}...")
            init_long = uniform_interp_seq(best_seq, nl)
            t_lp = min(t_per_len * 0.6, 80.0)
            if t_lp > 10:
                a_out, _ = nesterov_lp_optimize(init_long, t_lp, verbose=False)
                try_update(a_out, f"lp_n{nl}")

            # Quick LSE on the longer sequence
            remaining_inner = time_per_len - (time.monotonic() - t_global)
            # Can't use remaining_inner properly, let's just check budget
            t_lse = min(remaining * 0.05, 30.0)
            if t_lse > 10:
                a_out2, _ = lse_momentum_descent(best_seq, t_lse, verbose=False)
                try_update(a_out2, f"lse_n{nl}")

    # === Phase 6: Final Pair CD Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.4, 150.0)
    print(f"\n=== Phase 6: Pair CD Polish (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=True)
        try_update(a_out, "pcd1")

    # === Phase 7: Final Quick LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.5, 100.0)
    print(f"\n=== Phase 7: Final LP (budget={t7:.0f}s) ===")
    if t7 > 10:
        a_out, _ = nesterov_lp_optimize(best_seq, t7, verbose=False)
        try_update(a_out, "lp_final")

    # === Phase 8: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.6, 80.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Final CD (budget={t8:.0f}s) ===")
        a_out, _ = pair_cd_optimize(best_seq, t8, rng, verbose=False)
        try_update(a_out, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_a = np.maximum(best_seq, 0.0)
        final_a = final_a / final_a.sum()
        final_score = eval_safe(final_a)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(final_a)}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in final_a]
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
