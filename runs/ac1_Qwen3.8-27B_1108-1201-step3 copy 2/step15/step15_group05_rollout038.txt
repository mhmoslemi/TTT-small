import numpy as np
import cvxpy as cp

import scipy.sparse as sp
from scipy.fft import rfft, irfft

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_corr(a, b):
    """Compute cross-correlation (a*b)[k] = sum_j a[k-j]*b[j] via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_sparse_matrix(a, k_indices, n):
    """Build sparse constraint matrix for cross-correlation constraints."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return sp.csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))

        return float('inf')

def lp_best_response(a_fixed, M, active_set, solver, n):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_set, b >= 0."""
    k_indices = np.asarray(active_set)
    T = build_sparse_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def verify_and_expand(a_fixed, b, M, active_set, solver, n):
    """Check for constraint violations and expand active set."""
    size = 2 * n - 1
    all_k = np.arange(size)
    complement = np.setdiff1d(all_k, active_set, assume_unique=False)
    if len(complement) == 0:
        return active_set, b
    # Use FFT to check violations
    cross = fast_cross_corr(a_fixed, b)
    viol = cross[complement] > M * (1 + 1e-6)
    if np.any(viol):
        viol_idx = complement[viol]
        new_active = np.union1d(active_set, viol_idx)
        if len(new_active) > 0.9 * size:
            new_active = all_k
        b_new, _ = lp_best_response(a_fixed, M, new_active, solver, n)
        if b_new is not None:
            return new_active, b_new
    return active_set, b

def nesterov_lp(a0, time_limit, solver, seed=0, verbose=False):
    """Nesterov-accelerated LP best response."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 50
    n_iters = 0
    last_duals = None

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set selection
        if active_set is None or t % 30 == 0:
            # Tighter tolerance over time
            tol = max(0.005, 0.03 * np.exp(-t / 500.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            # Always include top peaks
            top_k = np.argsort(conv_y)[-max(5, n // 100):]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Limit active set size for speed
            if len(active_set) > max(200, n // 2):
                # Keep only the most critical ones
                active_set = np.argsort(conv_y)[::-1][:max(200, n // 2)]
                active_set = np.unique(np.concatenate([active_set, top_k]))

        b, duals = lp_best_response(y, M_curr, active_set, solver, n)

        if b is None:
            active_set = np.arange(2 * n - 1)
            b, duals = lp_best_response(y, M_curr, active_set, solver, n)
            if b is None:
                continue

        active_set, b = verify_and_expand(y, b, M_curr, active_set, solver, n)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        last_duals = duals

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best (check both y and a)
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score

def gradient_refine(a0, time_limit, verbose=False):
    """
    Gradient-based refinement: minimize max_k (a*a)[k] using subgradient method.
    Uses the active peak's gradient to update a.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.1 / np.sqrt(n)
    patience = 0
    max_patience = 5000

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Use top few peaks for combined gradient
        n_peaks = min(5, n // 200 + 1)
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Compute combined gradient (vectorized)
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            # grad[j] += w * (a[k-j] if 0<=k-j<n else 0) + w * (a[j-k] if 0<=j-k<n else 0)
            # Vectorized:
            j_range = np.arange(n)
            i1 = k - j_range
            mask1 = (i1 >= 0) & (i1 < n)
            grad[mask1] += w * a[i1[mask1]]
            i2 = j_range - k
            mask2 = (i2 >= 0) & (i2 < n)
            grad[mask2] += w * a[i2[mask2]]

        # Line search
        best_step = step
        best_obj = float('inf')
        for scale in [1.0, 0.5, 0.25, 2.0, 4.0]:
            s = step * scale
            a_trial = a - s * grad / max(grad.max(), 1e-10)
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial * norm / s_trial
            obj = fast_autocorr(a_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_step = s

        a_trial = a - best_step * grad / max(grad.max(), 1e-10)
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        if fast_autocorr(a_trial).max() < best_score - 1e-12:
            a = a_trial
            best_score = fast_autocorr(a).max()
            best_a = a.copy()
            patience = 0
            step = min(step * 1.05, 5.0)
        else:
            patience += 1
            step *= 0.95
            if step < 1e-12:
                break

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Grad t={t} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np
import cvxpy as cp
import time
import scipy.sparse as sp
from scipy.fft import rfft, irfft


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Using solver: ECOS (fallback)")
    return cp.ECOS


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_structure(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        j_range = np.arange(lo, hi)
        rows.extend([k] * len(j_range))
        cols.extend(j_range.tolist())
    return np.array(rows), np.array(cols)


def make_sparse_conv_matrix(a, rows, cols):
    vals = a[rows - cols]
    n = len(a)
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_solve_full(a_fixed, M, rows, cols, solver, n):
    T = make_sparse_conv_matrix(a_fixed, rows, cols)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dv = constraints[0].dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def lp_solve_active(a_fixed, M, active_set, rows, cols, solver, n):
    k_arr = np.asarray(active_set)
    in_active = np.zeros(2 * n - 1, dtype=bool)
    in_active[k_arr] = True
    mask = in_active[rows]
    sub_rows = np.arange(len(k_arr))[in_active[rows] & np.arange(2*n-1)[rows] >= 0]
    # Simpler approach: build sub-matrix directly
    sub_rows_list = []
    sub_cols_list = []
    sub_vals_list = []
    for idx in range(len(rows)):
        k = rows[idx]
        if in_active[k]:
            sub_rows_list.append(k)
            sub_cols_list.append(cols[idx])
            sub_vals_list.append(a_fixed[k - cols[idx]])
    T_sub = sp.csr_matrix((sub_vals_list, (sub_rows_list, sub_cols_list)), shape=(2*n-1, n))
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dv = constraints[0].dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def verify_violations(a_fixed, b, M, n):
    cross = fast_cross_corr(a_fixed, b)
    return np.where(cross > M * (1 + 1e-6))[0]


def nesterov_lp_opt(a0, time_limit, solver, seed=0, verbose=False):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    # Pre-compute matrix structure
    rows, cols = build_conv_matrix_structure(n)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    stall_threshold = 60
    n_iters = 0

    # Ergodic average
    erg_a = np.zeros(n)
    erg_count = 0
    erg_best_score = best_score
    erg_best_a = best_a.copy()

    # Use active set approach: start with top 40%, expand over time
    active_fraction = 0.4
    active_set = None

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Update active set periodically
        if active_set is None or t % 40 == 0:
            # Gradually expand the active set
            active_fraction = min(1.0, 0.3 + 0.7 * (1.0 - np.exp(-t / 300.0)))
            n_active = max(100, int(active_fraction * (2 * n - 1)))
            active_set = np.argsort(conv_y)[::-1][:n_active]
            # Always include the very top
            top_5 = np.argsort(conv_y)[-5:]
            active_set = np.unique(np.concatenate([active_set, top_5]))

        # Solve LP
        if len(active_set) >= (2 * n - 1):
            b, duals = lp_solve_full(y, M_curr, rows, cols, solver, n)
        else:
            b, duals = lp_solve_active(y, M_curr, active_set, rows, cols, solver, n)

        if b is None:
            # Fallback: use full set
            b, duals = lp_solve_full(y, M_curr, rows, cols, solver, n)
            if b is None:
                continue

        # Check for violations outside active set
        if len(active_set) < (2 * n - 1):
            viol = verify_violations(y, b, M_curr, n)
            not_in_active = np.setdiff1d(viol, active_set, assume_unique=False)
            if len(not_in_active) > 0:
                new_active = np.unique(np.concatenate([active_set, not_in_active]))
                if len(new_active) > 0.85 * (2 * n - 1):
                    new_active = np.arange(2 * n - 1)
                b2, duals = lp_solve_full(y, M_curr, rows, cols, solver, n)
                if b2 is not None:
                    b = b2
                    active_set = new_active

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t <= 3:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.96)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best (individual)
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average (check every 10 iterations)
        erg_a += a
        erg_count += 1
        if erg_count % 10 == 0:
            erg_check = erg_a / erg_count
            s_erg = erg_check.sum()
            if s_erg > 1e-15:
                erg_check = erg_check * norm / s_erg
                score_erg = fast_autocorr(erg_check).max()
                if score_erg < erg_best_score:
                    erg_best_score = score_erg
                    erg_best_a = erg_check.copy()
                if score_erg < best_score:
                    best_score = score_erg
                    best_a = erg_check.copy()
                    stall_count = 0

        # Stall restart
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, erg={erg_best_score:.10f}")

    # Compare best individual vs ergodic
    if erg_best_score < best_score:
        best_a = erg_best_a.copy()
        best_score = erg_best_score

    return best_a, best_score


def gradient_polish(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.5 / n ** 0.4
    patience = 0
    max_patience = 3000
    j_range = np.arange(n)

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Use top peaks for gradient
        n_peaks = max(3, min(8, n // 300))
        top_idx = np.argsort(conv)[-n_peaks:]
        peaks = conv[top_idx]
        weights = np.exp(-15.0 * (M_curr - peaks) / max(M_curr * 0.005, 1e-10))
        weights = weights / weights.sum()

        # Vectorized gradient computation
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            i1 = k - j_range
            mask1 = (i1 >= 0) & (i1 < n)
            grad[mask1] += w * a[i1[mask1]]
            i2 = j_range - k
            mask2 = (i2 >= 0) & (i2 < n)
            grad[mask2] += w * a[i2[mask2]]

        gmax = grad.max()
        if gmax < 1e-15:
            break
        grad_norm = grad / gmax

        # Try multiple step scales
        best_obj = float('inf')
        best_scale = 1.0
        for scale in [0.25, 0.5, 1.0, 2.0, 4.0]:
            s = step * scale
            a_trial = a - s * grad_norm
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial * norm / s_trial
            obj = fast_autocorr(a_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_scale = scale

        s_final = step * best_scale
        a_trial = a - s_final * grad_norm
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        if best_obj < best_score - 1e-13:
            a = a_trial
            best_score = best_obj
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.03, 10.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Grad t={t} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a_interp = interpolate_sequence(best_seq, target_n)
        a_norm = a_interp * np.sqrt(2.0 * target_n) / a_interp.sum()
        a_out, _ = nesterov_lp_opt(a_norm, t2, solver, verbose=True, seed=500 + target_n)
        try_update(a_out, f"lp_n{target_n}")

    # === Phase 3: Gradient polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Gradient n={n_cur}, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = gradient_polish(a_norm, t3, verbose=True)
        try_update(a_out, f"grad_n{n_cur}")

    # === Phase 4: LP round 2 on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.30, 350.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP round2 n={n_cur}, budget={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_opt(a_norm, t4, solver, verbose=True, seed=777)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Final gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.35, 120.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Final grad n={n_cur}, budget={t5:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = gradient_polish(a_norm, t5, verbose=True)
        try_update(a_out, f"gradfin_n{n_cur}")

    # === Phase 6: One more LP if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: Final LP n={n_cur}, budget={remaining-30:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_opt(a_norm, remaining - 30, solver, verbose=False, seed=1234)
        try_update(a_out, f"lpfin_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
