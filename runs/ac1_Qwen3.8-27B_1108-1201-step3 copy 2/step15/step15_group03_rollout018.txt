def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def normalize(a, target_sum):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a * target_sum / s


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(50.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(20, min(200, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = np.fft.rfft(w, fft_size)
            A = np.fft.rfft(a, fft_size)

            g1 = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
            g2 = np.fft.irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.05 * it))

            a_new = a - step * grad
            a_new = normalize(a_new, norm)
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score

            row[lo:hi] = a[k - np.arange(lo, hi)]
        T_rows.append(row)
    
    T_active = np.array(T_rows)
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_active @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    return val, prob


def lp_with_constraint_generation(a, M, solver, max_constraints=200, time_per_lp=5.0):
    """LP best-response with constraint generation."""
    n = len(a)
    size = 2 * n - 1
    
    # Start with top constraints
    # We need to estimate which constraints are tight
    # Use a random q to estimate
    q_init = np.ones(n) / n
    conv_init = fast_conv_cross(a, q_init)
    
    # Use the top K lagged values
    K = min(50, size)
    top_idx = np.argsort(conv_init)[::-1][:K]
    
    # Iterative constraint generation
    q = None
    for cg_iter in range(20):
        q, prob = lp_best_response(a, M, solver, top_idx)
        if q is None:
            # Try all constraints
            q, prob = lp_best_response(a, M, solver, None)
            if q is None:
                return None
            break
        
        # Check for violations
        conv_q = fast_conv_cross(a, q)
        violations = np.where(conv_q > M * 1.001)[0]
        
        if len(violations) == 0:
            break
        
        # Add violated constraints to the active set
        # Keep the most violated ones
        all_idx = np.unique(np.concatenate([top_idx, violations]))
        # Sort by violation amount and keep top max_constraints
        conv_all = conv_q[all_idx]
        if len(all_idx) > max_constraints:
            keep = np.argsort(conv_all)[::-1][:max_constraints]
            all_idx = all_idx[keep]
        top_idx = all_idx
    
    return q


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    
    # Adaptive parameters
    momentum = 0.0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 10:
            break  # Leave time for final eval
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Constraint generation LP
        q = lp_with_constraint_generation(y, M, solver, max_constraints=150)
        if q is None:
            continue
        
        q_sum = q.sum()
        if q_sum < 1e-15:
            continue
        g = q * norm / q_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """Polish using the exact subgradient structure."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.1
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        max_c = conv.max()
        k_star = np.argmax(conv)
        
        # Subgradient: 2 * flip(a, k_star/2)
        # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n else 0
        grad = np.zeros(n)
        valid = (k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n)
        grad[valid] = 2.0 * a[k_star - np.arange(n)[valid]]
        
        # Also include other near-max positions for stability
        threshold = max_c * 0.95
        near_max = np.where(conv >= threshold)[0]
        for k2 in near_max[:5]:
            if k2 == k_star:
                continue
            valid2 = (k2 - np.arange(n) >= 0) & (k2 - np.arange(n) < n)
            sub = np.zeros(n)
            sub[valid2] = 2.0 * a[k2 - np.arange(n)[valid2]]
            grad += 0.3 * sub
        
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        lr = 0.5 / (1.0 + 0.01 * it)
        momentum = beta * momentum - lr * grad / gn
        a_new = a + momentum
        a_new = normalize(a_new, norm)
        a = a_new
        
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def pair_cd_polish(a0, time_limit, rng, verbose=True):
    """Coordinate descent polish with exact 1D minimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_a = a.copy()
        
        perm = rng.permutation(n)
        pass_improved = False
        n_improve = 0
        
        # Focus on the most "pressured" coordinates
        # Those that contribute most to the peaks
        peak_lags = np.where(conv >= cur_max * 0.9)[0]
        
        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            i = idx
            ai = a[i]
            if ai < norm * 1e-8:
                continue
            
            # Find candidate j's
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j_s = (kl - i) % n
                if 0 <= kl - i < n:
                    j_candidates.add(kl - i)
            j_candidates.discard(i)
            
            best_val = cur_max
            best_delta = 0.0
            best_j = -1
            
            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]
                if aj < 1e-15 and ai < 1e-15:
                    continue
                
                # Transfer delta from i to j
                # New conv = old_conv + delta * (gain_vector)
                # gain_vector[k] = a[k-j] - a[k-i] (for valid indices)
                # Plus corrections at 2i and 2j
                
                # Compute gain vector
                # (a_new * a_new)[k] - (a * a)[k] where a_new[i] = ai-d, a_new[j] = aj+d
                # = -d*(a[k-i] + a[k-i]) + d*d (if 2i=k) 
                #   + d*(a[k-j] + a[k-j]) + d*d (if 2j=k)
                # Wait, let me be more careful.
                # (a_new * a_new)[k] = (a*a)[k] - 2d*a[k-i] + d^2*delta_{2i,k}
                #                              + 2d*a[k-j] + d^2*delta_{2j,k}
                # where a[x] = 0 if x out of range.
                # Hmm, this is for the case where we change a[i] by -d and a[j] by +d.
                # (a_new)[i] = ai - d, (a_new)[j] = aj + d, others same.
                # (a_new*a_new)[k] = (a*a)[k] 
                #   + (-d)*a[k-i] + (-d)*a[k-i]  [from cross terms with i]
                #   + d*a[k-j] + d*a[k-j]  [from cross terms with j]
                #   + d^2 * (delta_{k,2i} - delta_{k,2i} - delta_{k,2j} + delta_{k,2j})
                # Wait, I need to be more careful.
                
                # (a_new * a_new)[k] = sum_{p,q: p+q=k} a_new[p] * a_new[q]
                # = (a*a)[k] + sum of new cross terms
                # The change is:
                # -d*a[k-i] (from p=i, q=k-i) - d*a[k-i] (from p=k-i, q=i) [if k-i != i]
                # +d*a[k-j] + d*a[k-j]
                # + d^2 if 2i=k (the i*i term changes from ai^2 to (ai-d)^2, diff = -2ai*d + d^2)
                # + d^2 if 2j=k (the j*j term changes from aj^2 to (aj+d)^2, diff = 2aj*d + d^2)
                # - 2*ai*d if 2i=k (already counted in the -d*a[k-i] terms? Let me check)
                
                # Actually, let me just compute the gain directly:
                # gain[k] = (a_new*a_new)[k] - (a*a)[k]
                # = sum_p sum_q: (a_new[p]-a[p])*(a_new[q]-a[q]) + a_new[p]*(a_new[q]-a[q]) + a_new[q]*(a_new[p]-a[p])
                # Since a_new[p] - a[p] = -d if p=i, +d if p=j, 0 otherwise:
                # = (-d)(-d)*delta_{p,i}*delta_{q,i} + ... this is getting messy.
                
                # Let me use the formula:
                # (a_new*a_new)[k] - (a*a)[k]
                # = 2d * (a_at(k-j) - a_at(k-i)) + d^2 * (delta_{2j,k} - delta_{2i,k})
                # where a_at(x) = a[x] if 0<=x<n else 0
                # But wait, this is only correct when i != j. Let me verify:
                # (a_new)[i] = ai - d, (a_new)[j] = aj + d
                # (a_new*a_new)[k] = (a*a)[k]
                #   - d * (a_new[k-i] + a_new[k-i] - a[k-i] - a[k-i] + correction)
                # This is getting complicated. Let me just use the direct formula:
                
                # delta_a[i] = -d, delta_a[j] = +d, delta_a[other] = 0
                # (a_new*a_new)[k] - (a*a)[k]
                # = sum_{p+q=k} (a[p]+da[p])*(a[q]+da[q]) - a[p]*a[q]
                # = sum_{p+q=k} a[p]*da[q] + da[p]*a[q] + da[p]*da[q]
                # = da[i]*sum_{q=k-i} a[q] + sum_{p: k-p=i} a[p]*da[k-p] ... 
                # = sum_{p+q=k} da[p]*a[q] + sum_{p+q=k} a[p]*da[q] + sum_{p+q=k} da[p]*da[q]
                # = da[i]*a[k-i] + da[j]*a[k-j] + a[i]*da[k-i] + a[j]*da[k-j] + da[i]*da[k-i] + da[j]*da[k-j]
                # = -d*a[k-i] + d*a[k-j] - a[i]*d*delta_{k-i,i} + a[j]*d*delta_{k-j,j}
                #   + (-d)*(-d)*delta_{k-i,i} + d*d*delta_{k-j,j}
                # Hmm, this is still messy. Let me just use:
                # gain[k] = -d*(a[k-i] if valid else 0) - d*(a[k-i] if valid else 0)  [from the -d at position i]
                #           + d*(a[k-j] if valid else 0) + d*(a[k-j] if valid else 0)  [from the +d at position j]
                #           + d^2 * (1 if 2j==k else 0) - d^2 * (1 if 2i==k else 0)
                #         + 2*(-d)*ai * (1 if 2i==k else 0)  [no wait, this is included in the above]
                
                # OK I think the cleanest way is:
                # Let e_i be the unit vector at position i, e_j at position j.
                # a_new = a - d*e_i + d*e_j
                # (a_new*a_new)[k] = (a*a)[k] - 2d*(a*e_i)[k] + 2d*(a*e_j)[k] + d^2*(e_i*e_i)[k] - 2d^2*(e_i*e_j)[k] + d^2*(e_j*e_j)[k]
                # (a*e_i)[k] = a[k-i] (if valid)
                # (a*e_j)[k] = a[k-j] (if valid)
                # (e_i*e_i)[k] = 1 if k=2i else 0
                # (e_j*e_j)[k] = 1 if k=2j else 0
                # (e_i*e_j)[k] = 1 if k=i+j else 0
                
                # So: gain[k] = -2d*a_at(k-i) + 2d*a_at(k-j) + d^2*(delta_{k,2i} - 2*delta_{k,i+j} + delta_{k,2j})
                
                # This is a quadratic in d! gain[k] = A[k]*d + B[k]*d^2
                # where A[k] = -2*a_at(k-i) + 2*a_at(k-j)
                # and B[k] = delta_{k,2i} - 2*delta_{k,i+j} + delta_{k,2j}
                
                # We want to find d that minimizes max_k (conv[k] + A[k]*d + B[k]*d^2)
                # subject to 0 <= d <= ai (so that a_new[i] >= 0)
                
                # This is a 1D minimax quadratic problem. We can solve it by:
                # 1. Binary search on the target value T
                # 2. For each T, check if there exists d with conv[k] + A[k]*d + B[k]*d^2 <= T for all k
                #    (This is checking if the intersection of constraints is nonempty)
                
                # For efficiency, let's just sample d values and pick the best.
                n_samples = 30
                d_max = ai * 0.95
                d_vals = np.linspace(0, d_max, n_samples)
                
                # Precompute A vector
                A_vec = np.zeros(size)
                # a_at(k-i) for k=0..size-1
                for k_range in [(k-i, a_at_val) for k, a_at_val in []]:  # placeholder
                    pass
                
                # More efficient: compute A_vec directly
                # A[k] = 2*(a_at(k-j) - a_at(k-i))
                # a_at(k-i) is nonzero when 0 <= k-i < n, i.e., i <= k < i+n
                # a_at(k-j) is nonzero when 0 <= k-j < n, i.e., j <= k < j+n
                
                # Build A_vec
                lo_i = i
                hi_i = i + n
                lo_j = j
                hi_j = j + n
                
                A_vec = np.zeros(size)
                # -2*a_at(k-i): nonzero for k in [i, i+n-1]
                A_vec[lo_i:hi_i] += -2.0 * a[:hi_i - lo_i]  # a[k-i] for k=i..i+n-1 is a[0]..a[n-1]
                # +2*a_at(k-j): nonzero for k in [j, j+n-1]
                A_vec[lo_j:hi_j] += 2.0 * a[:hi_j - lo_j]
                
                # B is sparse: only at k=2i, k=i+j, k=2j
                # B[2i] += 1, B[i+j] -= 2, B[2j] += 1
                B_nonzero = {}
                if 2 * i < size:
                    B_nonzero[2 * i] = B_nonzero.get(2 * i, 0) + 1.0
                if i + j < size:
                    B_nonzero[i + j] = B_nonzero.get(i + j, 0) - 2.0
                if 2 * j < size:
                    B_nonzero[2 * j] = B_nonzero.get(2 * j, 0) + 1.0
                
                # Evaluate at sample points
                for d in d_vals:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * A_vec
                    for k_b, b_val in B_nonzero.items():
                        new_conv[k_b] += b_val * d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a * norm / s_new
                n_improve += 1
                pass_improved = True
        
        if not pass_improve:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                # Perturb to escape local minimum
                n_perturb = max(10, n // 50)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and n_improve > 0 and n_improve % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
