            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the score."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def norm_to(a, target_sum):
    """Normalize a to target_sum, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    return a * target_sum / s


def build_T(a, k_indices, n):
    """Build constraint matrix: T[idx, j] = a[k - j] if 0 <= k-j < n else 0."""
    m = len(k_indices)
    T = np.zeros((m, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            T[idx, lo:hi] = a[k - lo: k - lo + (hi - lo)]
    return T


def lp_solve(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    T = build_T(a_fixed, active_set, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dv = constraints[0].dual_value
        if dv is not None:
            duals = np.maximum(np.asarray(dv, dtype=np.float64), 0.0)
    except Exception:
        pass
    return b_val, duals


def check_violations(a_fixed, b, M, active_set, n):
    """Check complement constraints and return expanded active set if violations found."""
    size = 2 * n - 1
    active_arr = np.asarray(active_set)
    in_active = np.zeros(size, dtype=bool)
    valid = (active_arr >= 0) & (active_arr < size)
    in_active[active_arr[valid]] = True
    comp = np.where(~in_active)[0]
    if len(comp) == 0:
        return active_set
    T_c = build_T(a_fixed, comp, n)
    vals = T_c @ b
    viol = np.where(vals > M * (1 + 1e-7))[0]
    if len(viol) > 0:
        new_active = np.unique(np.concatenate([active_arr[valid], comp[viol]]))
        return new_active
    return active_set


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with adaptive active set."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_thresh = 40
    n_iters = 0
    erg_a = np.zeros(n)
    erg_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if active_set is None or t % 40 == 0:
            frac = min(1.0, t / 400.0)
            tol = 0.12 * (1.0 - frac) + 0.03 * frac
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            n_top = max(5, n // 80)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > 4 * n:
                vals = conv_y[active_set]
                keep = np.argsort(vals)[-4 * n:]
                active_set = active_set[keep]

        b, duals = lp_solve(y, M_curr, active_set, solver)
        if b is None:
            if t % 5 == 0:
                active_set = np.arange(2 * n - 1)
                b, duals = lp_solve(y, M_curr, active_set, solver)
            if b is None:
                continue

        active_set = check_violations(y, b, M_curr, active_set, n)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 1.0 / (t + 1.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t <= 1:
            a_new = y_new.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.99)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        erg_a += a
        erg_count += 1
        if erg_count % 15 == 0:
            erg = erg_a / erg_count
            s_erg = erg.sum()
            if s_erg > 1e-15:
                erg = erg * norm_val / s_erg
                sc = fast_conv(erg).max()
                if sc < best_score:
                    best_score = sc
                    best_a = erg.copy()
                    stall_count = 0

        if stall_count > stall_thresh:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                print(f"    LP restart iter={t} ({time.monotonic()-t0:.0f}s) score={best_score:.10f}")

        if verbose and n_iters % 1000 == 0:
            print(f"    LP iter={n_iters} ({time.monotonic()-t0:.0f}s) best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """Projected subgradient with multi-peak weighting and backtracking."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    step = 1.0 / max(1.0, n ** 0.4)
    patience = 0
    max_pat = 800

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit or patience > max_pat:
            break

        conv = fast_conv(a)
        M_curr = conv.max()
        # Multi-peak subgradient
        n_peaks = max(3, min(8, n // 200))
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.05, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            js = np.arange(lo, hi)
            grad[js] += w * a[k - js]
            # Also the symmetric part
            js2 = max(0, k) - np.arange(max(0, k), min(n, k + 1))
            # Actually: for lag k, grad[j] += a[k-j] (if valid) + a[j-k] (if valid)
            # Let's redo this properly
            grad = np.zeros(n)  # Reset and redo
            for kk in range(n):
                v1 = a[k - kk] if 0 <= k - kk < n else 0.0
                v2 = a[kk - k] if 0 <= kk - k < n else 0.0
                grad[kk] = v1 + v2
            # Accumulate with weight
            break  # Only do one peak at a time for efficiency in inner loop
        
        # Actually let's be more careful and efficient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            # grad[j] += w * (a[k-j] if 0<=k-j<n else 0) + w * (a[j-k] if 0<=j-k<n else 0)
            lo1 = max(0, k - n + 1)
            hi1 = min(n, k + 1)
            for j in range(lo1, hi1):
                grad[j] += w * a[k - j]
            lo2 = max(0, k)
            hi2 = min(n, k + n)
            for j in range(lo2, hi2):
                grad[j] += w * a[j - k]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm_val / s_trial

        sc = fast_conv(a_trial).max()
        if sc < best_score - 1e-12:
            a = a_trial
            best_score = sc
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 3.0)
        else:
            patience += 1
            step *= 0.96
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            print(f"    Subgrad t={t} ({time.monotonic()-t0:.0f}s) score={best_score:.10f} step={step:.6f}")

    return best_a, best_score


def coord_descent_fast(a0, time_limit, verbose=False):
    """Fast coordinate descent exploiting convolution structure."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 3:
        improved = False
        conv = fast_conv(a)

        # Process coordinates in order of potential impact
        # Coordinates at the center affect more conv values
        order = list(range(n))
        np.random.shuffle(order)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-10:
                continue

            # Compute effect of changing a[j]
            # delta_conv[k] = delta * (a[k-j] + a[j-k]) + delta^2 * [k==2j]
            c = np.zeros(size)
            lo1 = max(0, j - n + 1)
            hi1 = min(size, j + 1)
            c[lo1:hi1] += a[max(0, j - (lo1 - j)): min(n, j - lo1 + (hi1 - lo1))]
            # Simpler: for each k, c[k] = a[k-j] (if valid) + a[j-k] (if valid), excluding a[j] itself
            for k in range(size):
                v = 0.0
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    v += a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    v += a[i2]
                c[k] = v

            bg = conv - 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            s_others = a.sum() - aj
            # Try several candidate values
            candidates = [0.0, aj * 0.3, aj * 0.6, aj, aj * 1.4, aj * 2.0, s_others * 0.5]
            candidates = [x for x in candidates if 0 <= x <= norm_val * 2]

            best_x = aj
            best_obj = 2.0 * n * conv.max() / (a.sum() ** 2)

            for x in candidates:
                new_sum = s_others + x
                if new_sum < 1e-10:
                    continue
                new_conv = bg + 2.0 * x * c
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                nm = new_conv.max()
                if nm < 0:
                    nm = 0.0
                obj = 2.0 * n * nm / (new_sum * new_sum)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_x = x

            if best_x != aj and abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-10:
                    a = a * norm_val / s_new
                    new_sc = fast_conv(a).max()
                    if new_sc < best_score - 1e-13:
                        best_score = new_sc
                        best_a = a.copy()
                        improved = True
                        # Update conv for remaining coordinates
                        conv = fast_conv(a)
                    else:
                        a[j] = aj
                        a = a * norm_val / a.sum()

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and no_improve <= 3:
            print(f"    CD pass (no_improve={no_improve}) ({time.monotonic()-t0:.0f}s) score={best_score:.10f}")

    return best_a, best_score


def symmetrize(a):
    """Symmetrize: a_sym = (a + a[::-1]) / 2."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def perturb_seq(a, rng, scale=0.03):
    """Multiplicative perturbation."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    a_p = a * noise
    s = a_p.sum()
    if s < 1e-15:
        return a.copy()
    return a_p / s


def spectral_perturb(a, rng, n_modes=5):
    """Perturb in frequency domain."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    a_norm = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a_norm, fft_size)
    # Add small random phase perturbations
    freqs = np.arange(len(A))
    mask = freqs < n_modes
    phase = rng.standard_normal(len(A)) * 0.1
    A[mask] *= np.exp(1j * phase[mask])
    A[~mask] *= (1.0 + rng.standard_normal(len(A) - n_modes).reshape(len(A) - n_modes,))
    # Actually let's be simpler: multiply by (1 + small_noise)
    A2 = A * (1.0 + 0.05 * rng.standard_normal(len(A)))
    A2 = np.maximum(A2, 0)
    a_new = irfft(A2, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate():
    """Main search: LP Nesterov + subgradient + CD with adaptive time allocation."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    # Quick symmetrization check
    a_sym = symmetrize(best_seq)
    s_sym = a_sym.sum()
    if s_sym > 1e-15:
        a_sym_n = a_sym * np.sqrt(2.0 * len(a_sym)) / s_sym
        sc_sym = evaluate_safe(a_sym_n)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym_n / a_sym_n.sum()
            print(f"  Symmetrization improved: {sc_sym:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.38, 600.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.12, 120.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Subgrad n={n_cur}, budget={t2:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t2, verbose=True)
        try_update(a_out, f"subg1_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.22, 300.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True, seed=777)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Perturbation + LP restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4_total = min(remaining * 0.15, 150.0)
    if t4_total > 40:
        print(f"\n=== Phase 4: Perturbation restarts, budget={t4_total:.0f}s ===")
        n_restarts = 3
        t4_each = t4_total / n_restarts
        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t4 = min(t4_each, remaining - 30)
            if t4 < 20:
                break
            if r == 0:
                pert = perturb_seq(best_seq, rng, scale=0.02)
            elif r == 1:
                pert = perturb_seq(best_seq, rng, scale=0.05)
            else:
                pert = spectral_perturb(best_seq, rng)
            a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
            a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=False, seed=500 + r * 10)
            try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 15:
        print(f"\n=== Phase 5: CD n={n_cur}, budget={t5:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent_fast(a_norm, t5, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        t6 = min(remaining * 0.7, 200.0)
        print(f"\n=== Phase 6: Final LP n={n_cur}, budget={t6:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True, seed=9999)
        try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 7: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_cur = len(best_seq)
        t7 = min(remaining * 0.6, 60.0)
        if t7 > 10:
            print(f"\n=== Phase 7: Final subgrad, budget={t7:.0f}s ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = subgradient_polish(a_norm, t7, verbose=False)
            try_update(a_out, f"subg_final_n{n_cur}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: fallback")
        return [1.0]

    return [float(x) for x in best_seq]
