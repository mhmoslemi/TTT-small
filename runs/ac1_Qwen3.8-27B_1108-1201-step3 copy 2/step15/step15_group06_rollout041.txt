import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr(x, y, L):
    """Compute linear cross-correlation (x ⊛ y)[j] = sum_k x[k] * y[k-j] via FFT."""
    x_full = np.zeros(L)
    y_full = np.zeros(L)
    x_full[:len(x)] = x
    y_full[:len(y)] = y
    result = ifft(fft(x_full) * np.conj(fft(y_full)))
    return result.real[:len(y)]

        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))

def structured_init(n, style=0):
    """Generate structured initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-10.0 * (x - 0.25) ** 2) + 0.7 * np.exp(-10.0 * (x - 0.75) ** 2)
    elif style == 4:
        # Asymmetric: ramp up then slow decay
        a = np.exp(-3.0 * x) * (1.0 + 0.5 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0.0)
    elif style == 5:
        # Multi-scale
        a = 0.3 + 0.5 * np.sin(np.pi * x) ** 2 + 0.2 * np.cos(8 * np.pi * x) * np.sin(np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 6:
        # Random-ish smooth
        a = 1.0 + 0.3 * np.cos(3 * np.pi * x) + 0.2 * np.sin(7 * np.pi * x)
        a = np.maximum(a, 0.1)
    else:
        a = (1.0 - x) ** 0.5 * (0.5 + 0.5 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def perturb_seq(a, rng, scale=0.05):
    """Multiplicative perturbation."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed subgradient descent with multiplicative (mirror/KL) updates
2. Log-sum-exp smoothing for stable, differentiable gradient
3. Nesterov acceleration with adaptive momentum
4. Coarse-to-fine tau annealing
5. Multi-phase search: multiple restarts, length adaptation, final polish
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft


def fast_autocorr(a):
    """Compute linear autocorrelation a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def smoothed_gradient(w, a, L):
    """
    Compute 2 * cross_corr(w, a) where cross_corr[j] = sum_k w[k] * a[k-j].
    w has length 2n-1, a has length n, L >= 3n-2.
    """
    w_full = np.zeros(L)
    a_full = np.zeros(L)
    w_full[:len(w)] = w
    a_full[:len(a)] = a
    result = ifft(fft(w_full) * np.conj(fft(a_full)))
    return 2.0 * result.real[:len(a)]


def evaluate_local(a):
    """Compute the scale-invariant score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def evaluate_safe(a):
    """Call the official evaluator safely."""
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def mirror_descent_optimize(a0, time_limit, tau_start, tau_end, eta0, L,
                            nesterov=True, seed=0, verbose=False, patience=300):
    """
    Smoothed subgradient descent with multiplicative (mirror) updates.
    
    Minimizes max_k (a*a)[k] subject to sum(a)=1, a>=0,
    using the smoothed surrogate (1/tau)*log(sum_k exp(tau*(a*a)[k])).
    
    Parameters:
    - a0: initial sequence (normalized to sum=1)
    - time_limit: seconds
    - tau_start, tau_end: smoothing parameter (annealed from start to end)
    - eta0: initial step size (decays as 1/sqrt(t))
    - L: FFT working size (>= 3n-2)
    - nesterov: use Nesterov momentum
    - patience: restart after this many non-improving iterations
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = 2.0 * n * conv.max()

    t0 = time.monotonic()
    stall = 0

    # Nesterov state
    if nesterov:
        y = a.copy()
        a_prev = a.copy()

    # Pre-allocate buffers for FFT
    w_buf = np.zeros(L)
    a_buf = np.zeros(L)

    for t in range(1, 10000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Anneal tau
        frac = min(elapsed / time_limit, 1.0)
        tau_t = tau_start * (tau_end / tau_start) ** frac

        # Evaluate at Nesterov lookahead point
        eval_pt = y if nesterov else a

        # Compute autocorrelation at eval point
        conv_eval = fast_autocorr(eval_pt)
        max_conv = conv_eval.max()

        # Softmax weights (numerically stable)
        shifted = tau_t * (conv_eval - max_conv)
        # Clip to avoid underflow in exp
        shifted = np.maximum(shifted, -50.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv_eval))
            w[np.argmax(conv_eval)] = 1.0
        else:
            w = w / w_sum

        # Compute smoothed gradient: 2 * cross_corr(w, a)
        w_buf[:] = 0.0
        a_buf[:] = 0.0
        w_buf[:len(w)] = w
        a_buf[:n] = a
        grad_f = ifft(fft(w_buf) * np.conj(fft(a_buf)))
        grad = 2.0 * grad_f.real[:n]

        # Step size (Polyak diminishing)
        eta_t = eta0 / np.sqrt(t + 1.0)

        # Multiplicative (mirror) update
        log_update = -eta_t * grad
        # Clip to prevent extreme values
        log_update = np.clip(log_update, -30.0, 30.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Nesterov momentum update
        if nesterov:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = a_new + momentum * (a_new - a_prev)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-300:
                y_new = a_new.copy()
            else:
                y_new = y_new / s_y
            y = y_new
            a_prev = a.copy()

        a = a_new

        # Track best
        conv_a = fast_autocorr(a)
        score = 2.0 * n * conv_a.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Also check Nesterov point periodically
        if nesterov and t % 100 == 0:
            sc_y = 2.0 * n * fast_autocorr(y).max()
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
                stall = 0

        if verbose and t % 2000 == 0:
            print(f"      MD t={t} ({elapsed:.0f}s): score={best_score:.10f}, conv_max={conv_a.max():.8f}, tau={tau_t:.1f}")

        # Restart if stalled
        if stall > patience:
            if nesterov:
                a = best_a.copy()
                y = best_a.copy()
                a_prev = best_a.copy()
            else:
                a = best_a.copy()
            stall = 0

    return best_a, best_score


def structured_init(n, style=0):
    """Generate structured initial sequences (normalized to sum=1)."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-10.0 * (x - 0.25) ** 2) + 0.7 * np.exp(-10.0 * (x - 0.75) ** 2)
    elif style == 4:
        a = np.exp(-3.0 * x) * (1.0 + 0.5 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0.0)
    elif style == 5:
        a = 0.3 + 0.5 * np.sin(np.pi * x) ** 2 + 0.2 * np.cos(8 * np.pi * x) * np.sin(np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 6:
        a = 1.0 + 0.3 * np.cos(3 * np.pi * x) + 0.2 * np.sin(7 * np.pi * x)
        a = np.maximum(a, 0.1)
    elif style == 7:
        # Asymmetric triangular
        a = np.where(x < 0.4, x / 0.4, (1.0 - x) / 0.6) ** 1.5
    else:
        a = (1.0 - x) ** 0.5 * (0.5 + 0.5 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_seq(a, rng, scale=0.05):
    """Multiplicative perturbation (preserves non-negativity)."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length (normalized to sum=1)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def coordinate_refine(a0, time_limit, rng, n_inner=50):
    """
    Simple coordinate descent refinement: try adjusting each coordinate
    in a random order, keeping the best improvement.
    Works on the normalized (sum=1) representation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv = fast_autocorr(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passed = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            # Try scaling a[i] by various factors
            for scale in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                a_try = a.copy()
                a_try[i] = a[i] * scale
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                cv = fast_autocorr(a_try)
                sc = 2.0 * n * cv.max()
                if sc < best_score * (1.0 - 1e-10):
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    passed = True
                    break

        if not passed:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0

    return best_a, best_score
