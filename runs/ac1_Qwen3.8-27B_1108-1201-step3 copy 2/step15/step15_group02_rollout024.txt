Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-softmax Adam gradient descent (fast initial convergence)
2. Full/active-set LP best-response with Nesterov acceleration (main engine)

Strategy: allocate most time to LP iterations which provide the strongest

import numpy as np
import cvxpy as cp
import time

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(x)), [np.ones(39) @ x <= 40.0])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")

def conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    return 2.0 * n * conv_self(a).max() / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def normalize_a(a, norm_val):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * norm_val / len(a)
    return a * norm_val / s

def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm_val)
    best_a = a.copy()
    best_score = conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0
    n_stages = 25
    temps = np.geomspace(50.0, 0.05, n_stages)
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(30, min(150, int(remaining * 8)))
        lr = 0.4 / np.sqrt(n)
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            conv = conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = np.maximum((conv - max_c) / T_val, -60.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = np.fft.rfft(w, fft_size)
            A = np.fft.rfft(a, fft_size)
            g1 = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
            g2 = np.fft.irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            a = np.maximum(a - lr * m_hat / (np.sqrt(v_hat) + eps), 0.0)
            a = normalize_a(a, norm_val)
            sc = conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        if verbose and si % 5 == 0:
            print(f"    LSE-Adam [{si}/{n_stages}] T={T_val:.3f}: M={best_score:.8f}")
    return best_a, best_score

def build_constraint_row(a, k):
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    row = np.zeros(n)
    if hi > lo:
        row[lo:hi] = a[k - np.arange(lo, hi)]
    return row

def lp_best_response(a, M, solver, active_idx):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = np.array([build_constraint_row(a, k) for k in active_idx])
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm_val)
    best_a = a.copy()
    best_score = conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    no_improve_count = 0
    perturbation_count = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        elapsed = time.monotonic() - t0
        
        conv_y = conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive active set: top fraction of peaks
        n_active = min(size, max(100, int(0.45 * size)))
        sorted_idx = np.argsort(conv_y)[::-1]
        active_idx = sorted_idx[:n_active]
        
        b = lp_best_response(y, M, solver, active_idx)
        if b is None:
            active_idx = sorted_idx[:min(size, 200)]
            b = lp_best_response(y, M, solver, active_idx)
        if b is None:
            continue
        
        # Verify and fix violations
        if n_active < size:
            check = conv_cross(y, b)
            violated = np.where(check > M * 1.0001)[0]
            if len(violated) > 0:
                new_active = np.unique(np.concatenate([active_idx, violated]))
                b = lp_best_response(y, M, solver, new_active)
                if b is None:
                    b = lp_best_response(y, M, solver, sorted_idx)
                    if b is None:
                        continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        
        # Nesterov momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        
        y = y_new
        a = a_new
        
        sc_y = conv_self(y).max()
        sc_a = conv_self(a).max()
        cur_best = min(sc_y, sc_a)
        
        if cur_best < best_score - 1e-10:
            best_score = cur_best
            best_a = y.copy() if sc_y < sc_a else a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Periodic perturbation to escape plateaus
        if no_improve_count >= 80 and perturbation_count < 20:
            no_improve_count = 0
            perturbation_count += 1
            rng_p = np.random.default_rng(t * 7919)
            n_pert = max(20, n // 40)
            idx_p = rng_p.choice(n, size=n_pert, replace=False)
            a[idx_p] *= (1.0 + 0.08 * rng_p.standard_normal(n_pert))
            a = normalize_a(a, norm_val)
            y = a.copy()
        
        if verbose and t % 15 == 0:
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): M={best_score:.8f}, "
                  f"cur={cur_best:.8f}, effort={b_sum:.4f}")
    
    return best_a, best_score

            M = conv.max()
            if M < 1e-15:
                break

            w_sum = w.sum()
            if w_sum < 1e-15:

            else:
                w = w / w_sum
            W = np.fft.rfft(w, fft_size)

            pressure = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
            pressure += np.fft.irfft(A * W, fft_size)[:n]
            pressure = np.maximum(pressure, 0.0)

    return best_a, best_score
