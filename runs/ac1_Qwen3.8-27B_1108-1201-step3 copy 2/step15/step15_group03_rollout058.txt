"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s} ({time.monotonic()-t0:.2f}s)")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize_a(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * target_sum / s


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    max_c = fast_conv_self(a).max()
    return 2.0 * n * max_c / (s * s)


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(60.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(50, min(500, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = np.maximum((conv - max_c) / T_val, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * 1.0 / (1.0 + 0.03 * it)
            a_new = np.maximum(a - step * grad, 0.0)
            a = normalize_a(a_new, norm)

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def build_T_active(a, active_indices):
    """Build the constraint matrix for active indices only."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(active_indices), n))
    for row, k in enumerate(active_indices):
        # T[row, :] such that T[row, :] @ b = (a*b)[k]
        # (a*b)[k] = sum_i a[k-i] * b[i] for valid i
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                T[row, i] = a[idx]
    return T


def lp_solve_efficient(a, M, solver, active_indices=None):
    """Solve the LP: max sum(b) s.t. (a*b)[k] <= M for k in active, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_indices is not None and 0 < len(active_indices) < size:
        # Build constraint (a*b)[k] <= M for k in active_indices
        T = build_T_active(a, active_indices)
        constraints = [T @ b_var <= M]
    else:
        T = build_T_active(a, np.arange(size))
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_nesterov_improved(a0, time_limit, solver, verbose=True):
    """LP Nesterov with adaptive constraints and line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()

    # Adaptive parameters
    initial_threshold = 0.88
    final_threshold = 0.70
    n_iterations_max = 200000

    for t in range(1, n_iterations_max):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive threshold
        frac = min(1.0, t / 5000.0)
        threshold = initial_threshold + (final_threshold - initial_threshold) * frac

        active = np.where(conv_y >= M * threshold)[0]
        if len(active) < 20:
            threshold2 = M * 0.5
            active = np.where(conv_y >= threshold2)[0]

        b = lp_solve_efficient(y, M, solver, active)
        if b is None:
            b = lp_solve_efficient(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.002)[0]
            if len(violated) > 5:
                all_active = np.union1d(active, violated)
                b_full = lp_solve_efficient(y, M, solver, all_active)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        # Line search on mix parameter
        alpha = 2.0 / (t + 2.0)
        momentum = (t - 1.0) / (t + 2.0)

        best_alpha = alpha
        best_val = fast_conv_self(normalize_a((1 - alpha) * y + alpha * g, norm)).max()

        # Try a few other alpha values
        for alpha_try in [alpha * 0.5, alpha * 1.5, 0.1, 0.05]:
            alpha_try = min(max(alpha_try, 0.01), 0.95)
            candidate = normalize_a((1 - alpha_try) * y + alpha_try * g, norm)
            val = fast_conv_self(candidate).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha_try

        # Nesterov update
        y_new = normalize_a((1.0 - best_alpha) * y + best_alpha * g, norm)
        a_new = y_new + momentum * (y_new - y)
        a_new = normalize_a(a_new, norm)

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_lbfgs(a0, time_limit, verbose=True):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    def objective_and_grad(x_log):
        """Objective and gradient with log-parameterization to ensure non-negativity."""
        x = np.exp(x_log - x_log.max())
        x = x / x.sum()
        x = x * norm
        conv = fast_conv_self(x)
        max_c = conv.max()
        # Use log-sum-exp with temperature
        tau = max(0.1, max_c * 0.03)
        shifted = (conv - max_c) / tau
        w = np.exp(np.maximum(shifted, -50.0))
        w = w / w.sum()
        val = max_c + tau * np.log(w.sum())  # This is the smoothed max
        # Actually, logsumexp(conv/tau) * tau = max_c + tau * log(sum(exp((conv-max_c)/tau)))
        val = max_c + tau * np.log(np.exp(np.maximum((conv - max_c) / tau, -50.0)).sum())

        # Gradient
        W = rfft(w, fft_size)
        A = rfft(x, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad_x = 2.0 * (g1 + g2)
        # Chain rule for log parameterization
        # dx/dx_log = x * (1 - x/sum(x)) = x * (1 - x/norm) since sum(x) = norm
        # Actually dx/dx_log[i] = x[i] - sum(x)*exp(x_log[i])/sum(exp(x_log)) = x[i] - x[i] = 0? No...
        # x[i] = exp(x_log[i]) * norm / sum(exp(x_log))
        # dx[i]/dx_log[j] = delta_{ij} * x[i] - x[i]*x[j]/norm
        # grad_x_log[j] = sum_i grad_x[i] * (delta_{ij}*x[i] - x[i]*x[j]/norm)
        # = x[j]*grad_x[j] - x[j]*sum_i grad_x[i]*x[i]/norm
        sum_grad_x = np.sum(grad_x * x) / norm
        grad = x * (grad_x - sum_grad_x)
        return val, grad

    x_log = np.zeros(n)
    # Set initial log values from a
    a_safe = np.maximum(a, 1e-10)
    x_log = np.log(a_safe)

    result = minimize(objective_and_grad, x_log, method='L-BFGS-B',
                      options={'maxiter': 200, 'gtol': 1e-10, 'maxfun': 500})

    if result.success or result.fun < best_score:
        x_final = np.exp(result.x - result.x.max())
        x_final = normalize_a(x_final, norm)
        sc = fast_conv_self(x_final).max()
        if sc < best_score:
            best_score = sc
            best_a = x_final.copy()

    if verbose:
        print(f"    L-BFGS: best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Efficient pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            peak_threshold = cur_max * 0.93
            peak_lags = np.where(conv >= peak_threshold)[0]

            # Candidate targets
            j_candidates = []
            for _ in range(8):
                j_candidates.append(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j_candidates.append((kl - i) % n)
                j_candidates.append((kl + i) % n)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in set(j_candidates):
                if j == i or j >= n:
                    continue
                aj = a[j]

                # Compute the change in convolution
                g = np.zeros(size)
                # Adding d to a[j]: conv changes by d * (a shifted to j)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Subtracting d from a[i]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Try several delta values
                n_samples = 15
                for d in np.linspace(0.01 * ai, 0.5 * ai, n_samples):
                    new_conv = conv + d * g
                    # Handle the quadratic terms at 2i and 2j
                    if 2 * i < size:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved multi-phase approach:
1. LSE gradient descent warmup with adaptive temperature
2. LP Nesterov with adaptive constraint threshold, line search, and dual-guided screening
3. Smoothed subgradient with Polyak step size and momentum
4. LP Nesterov round 2 on best solution
5. Efficient pair coordinate descent polish
6. Multi-restart exploration with targeted perturbations
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None
_solver_time = {}

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s} ({time.monotonic()-t0:.3f}s)")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize_a(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * target_sum / s


def compute_max_conv(a):
    return fast_conv_self(a).max()


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * compute_max_conv(a) / (s * s))


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    n_stages = 16
    temps = np.geomspace(50.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(40, min(400, int(remaining * 10)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = np.maximum((conv - max_c) / T_val, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            # Adaptive step size with backtracking-like decay
            base_step = norm / (T_val * np.sqrt(n))
            step = base_step * 1.0 / (1.0 + 0.02 * it)
            a_new = np.maximum(a - step * grad, 0.0)
            a = normalize_a(a_new, norm)

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score


def build_constraint_matrix(a, active_indices):
    n = len(a)
    T = np.zeros((len(active_indices), n))
    for row, k in enumerate(active_indices):
        # (a*b)[k] = sum_i a[k-i]*b[i] for valid i
        # Valid i: max(0, k-n+1) <= i <= min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            T[row, i_start:i_end+1] = a[k - i_end:k - i_start + 1][::-1]
    return T


def lp_solve(a, M, solver, active_indices=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_indices is not None and 0 < len(active_indices) < size:
        T = build_constraint_matrix(a, active_indices)
        constraints = [T @ b_var <= M]
    else:
        T = build_constraint_matrix(a, np.arange(size))
        constraints = [T @ b_var <= M]

    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lp_nesterov(a0, time_limit, solver, verbose=True, threshold_start=0.85, threshold_end=0.65):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    best_lp_time = 1.0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive threshold: start loose, tighten over time
        frac = min(1.0, t / 3000.0)
        threshold = threshold_start + (threshold_end - threshold_start) * frac

        active = np.where(conv_y >= M * threshold)[0]
        if len(active) < 15:
            active = np.where(conv_y >= M * 0.4)[0]

        # Solve LP
        t_lp = time.monotonic()
        b = lp_solve(y, M, solver, active)
        lp_time = time.monotonic() - t_lp

        if b is None:
            b = lp_solve(y, M, solver, None)
            if b is None:
                continue
            lp_time = time.monotonic() - t_lp

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Check dropped constraints
        if active is not None and len(active) < size:
            check = fast_conv_cross(y, g)
            violated = check > M * 1.001
            n_viol = violated.sum()
            if n_viol > 3:
                all_active = np.where(conv_y >= M * threshold)[0]
                all_active = np.union1d(all_active, np.where(violated)[0])
                b2 = lp_solve(y, M, solver, all_active)
                if b2 is not None and b2.sum() > 1e-15:
                    b = b2
                    g = b * norm / b.sum()

        # Nesterov update with line search on mix parameter
        alpha_nest = 2.0 / (t + 2.0)
        momentum = (t - 1.0) / (t + 2.0)

        # Try several alpha values via line search
        alphas_to_try = [alpha_nest]
        if t > 10:
            alphas_to_try.extend([alpha_nest * 0.5, alpha_nest * 2.0])
        alphas_to_try = [min(max(al, 0.01), 0.9) for al in alphas_to_try]

        best_alpha = alphas_to_try[0]
        best_candidate = None
        best_candidate_score = float('inf')

        for al in alphas_to_try:
            candidate = normalize_a((1.0 - al) * y + al * g, norm)
            sc = fast_conv_self(candidate).max()
            if sc < best_candidate_score:
                best_candidate_score = sc
                best_candidate = candidate
                best_alpha = al

        y_new = best_candidate
        a_new = normalize_a(y_new + momentum * (y_new - y), norm)

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Ne t={t} ({elapsed:.0f}s, lp={lp_time:.2f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """Momentum subgradient with adaptive temperature and Polyak step."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    momentum = np.zeros(n)
    beta = 0.85
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv_self(a)
        max_c = conv.max()

        # Adaptive temperature: decrease over time
        tau = max(0.05, max_c * 0.08 / (1.0 + 0.005 * it))
        shifted = np.maximum((conv - max_c) / tau, -60.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        # Polyak step size
        gap = max_c - norm * norm / n  # lower bound is norm^2/n = 2
        lr = (gap + 0.1) / (np.dot(grad, grad) + 1e-10)
        lr = min(lr, 5.0 / np.sqrt(n))

        momentum = beta * momentum - lr * grad
        a_new = normalize_a(a + momentum, norm)
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            peak_lags = np.where(conv >= cur_max * 0.92)[0]

            j_cands = list(set([int(rng.integers(n)) for _ in range(6)]))
            for kl in peak_lags[:5]:
                j_cands.append((kl - i) % n)
                j_cands.append((i + kl) % n if i + kl < n else (i - kl + n) % n)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_cands[:18]:
                if j == i or j >= n or j < 0:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                for d in np.linspace(0.02 * ai, 0.6 * ai, 10):
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize_a(a, norm)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_imp += 1
                cur_max = new_max

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_p = max(8, n // 60)
                idx_p = rng.choice(n, size=n_p, replace=False)
                a[idx_p] *= (1.0 + 0.08 * rng.standard_normal(n_p))
                a = normalize_a(a, norm)
        else:
            no_improve = 0

        if verbose and total_imp % 300 == 0 and total_imp > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_imp} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):

    print(f"\n=== Phase 6: Perturbation exploration (budget={remaining:.0f}s) ===")
    if remaining > 100:
        n_perturbations = 3
        t_per_perturb = remaining / (n_perturbations * 2 + 1)

        for pi in range(n_perturbations):
            if time.monotonic() - t_global > time_budget - 60:
                break

            # Create perturbation
            a_pert = best_seq.copy()
            if pi == 0:
                # Anti-peak: reduce coordinates that contribute to the peak
                conv = fast_conv_self(a_pert)
                peak_lags = np.where(conv >= conv.max() * 0.95)[0]
                pressure = np.zeros(len(a_pert))
                for kl in peak_lags[:20]:
                    for i in range(len(a_pert)):
                        if 0 <= kl - i < len(a_pert):
                            pressure[i] += a_pert[kl - i]
                # Reduce high-pressure coords, boost low-pressure
                top_p = np.argsort(pressure)[-len(a_pert)//4:]
                bot_p = np.argsort(pressure)[:len(a_pert)//4:]
                a_pert[top_p] *= 0.7
                a_pert[bot_p] *= 1.4
            elif pi == 1:
                # Smooth perturbation
                kernel = np.ones(5) / 5.0
                a_pert = np.convolve(a_pert, kernel, mode='same')
                a_pert += 0.05 * rng.standard_normal(len(a_pert))
            else:
                # Random structured perturbation
                n_spikes = max(5, len(a_pert) // 100)
                spike_idx = rng.choice(len(a_pert), size=n_spikes, replace=False)
                a_pert[spike_idx] *= (1.0 + 0.5 * rng.random(n_spikes))
                a_pert += 0.02 * rng.standard_normal(len(a_pert))

            a_pert = normalize_a(a_pert, np.sqrt(2.0 * len(a_pert)))

            # Short LP optimization on perturbed
            t_short = min(t_per_pert, 120.0)
            if t_short > 20:
                a_out, _ = lp_nesterov(a_pert, t_short, solver, verbose=False,
                                       threshold_start=0.80, threshold_end=0.60)
                try_update(a_out, f"perturb{pi}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t_final = min(remaining * 0.5, 80.0)
        a_out, _ = pair_cd(best_seq, t_final, rng, verbose=False)
        try_update(a_out, "final_pcd")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = lse_gradient_descent(best_seq, min(remaining * 0.6, 40.0), verbose=False)
            try_update(a_out, "final_lse")

    # Final evaluation
    if best_seq is not None:
        final_score = compute_score(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
