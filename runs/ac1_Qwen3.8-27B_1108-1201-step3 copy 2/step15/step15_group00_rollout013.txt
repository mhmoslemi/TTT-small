   def eval_quick(a):
       """Quick evaluation without calling evaluate_sequence (for internal use)."""
       a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
       s = a.sum()
       if s < 0.01:
           return float('inf')
       n = len(a)
       conv = fast_conv(a)
       return float(2.0 * n * conv.max() / (s * s))

   def compute_logsumexp_weights(conv, temperature):
       """Compute softmax weights over convolution values."""
       # conv has 2n-1 values
       # Softmax: p[k] = exp(conv[k] / T) / sum(exp(conv[j] / T))
       scaled = conv / (temperature + 1e-30)
       max_scaled = scaled.max()
       exp_scaled = np.exp(scaled - max_scaled)
       p = exp_scaled / exp_scaled.sum()
       return p

   def compute_grad_lse(x, p):
       """
       Compute gradient of logsumexp smooth approximation of max(conv).
       g[i] = 2 * sum_k p[k] * x[k-i] (where x[m]=0 for m not in [0,n))
            = 2 * sum_{m=0}^{n-1} p[m+i] * x[m]
       Using: np.correlate(p, x, mode='valid')
       """
       n = len(x)
       # p has length 2n-1, x has length n
       # correlate(p, x, 'valid')[i] = sum_{m=0}^{n-1} p[i+m] * x[m]
       q = np.correlate(p, x, mode='valid')
       return 2.0 * q

   def mirror_descent_step(x, g, eta):
       """Multiplicative weights update: x_new[i] = x[i] * exp(-eta * g[i]) / Z"""
       log_x = np.log(x + 1e-30) - eta * g
       max_log = log_x.max()
       x_new = np.exp(log_x - max_log)
       x_new = np.maximum(x_new, 1e-30)
       x_new = x_new / x_new.sum()
       return x_new

   def lse_smoothed_mirror_descent(x0, time_limit, verbose=True, seed=42):
       """
       Mirror descent with log-sum-exp smoothing and Nesterov acceleration.
       Adaptive temperature schedule.
       """
       n = len(x0)
       x = norm_sum1(x0)
       best_x = x.copy()
       best_obj = fast_conv(x).max()
       t0 = time.monotonic()
       rng = np.random.default_rng(seed)

       # Nesterov state
       y = x.copy()
       y_prev = x.copy()
       
       # Temperature schedule: start high, decay
       conv0 = fast_conv(x)
       M0 = conv0.max()
       T_start = M0 * 5.0  # Start with broad smoothing
       T_end = M0 * 0.1  # End with sharp (near-exact) subgradient

       # Step size
       eta0 = 1.0
       stall_count = 0
       max_stall = 3000
       iter_count = 0

       for t in range(1, 2000000):
           if time.monotonic() - t0 > time_limit:
               break

           iter_count += 1
           
           # Nesterov momentum
           beta_t = 1.0 / (1.0 + np.sqrt(4.0 * t))  # or (t-1)/(t+2)
           # Actually, for mirror descent with Nesterov: beta_t = (1 - sqrt(t))/(1 + sqrt(t+1))
           # Let's use the standard: beta_t = (t - 1) / (t + 2)
           beta_t = (max(t - 1, 0)) / (t + 2.0)
           y = x + beta_t * (x - y_prev)
           y = np.maximum(y, 1e-15)
           y = y / y.sum()
           y_prev = x

           # Compute convolution and soft gradient
           conv_y = fast_conv(y)
           M_y = conv_y.max()
           
           # Adaptive temperature: geometric decay
           # T = T_start * (T_end/T_start)^t  (geometric)
           # Or T = T_start * exp(-lambda * t)
           # Let's use: T_t = M_y * max(0.1, 5.0 / (1 + 0.001*t))
           frac = 5.0 / (1.0 + 0.001 * t)
           frac = max(0.05, min(5.0, frac))
           T = M_y * frac + 1e-30
           
           p = compute_logsumexp_weights(conv_y, T)
           g = compute_grad_lse(y, p)
           g = np.maximum(g, 0.0)  # Gradient should be non-negative for convex part

           # Step size: eta = c / (sqrt(t) + 1)
           c = 2.0
           eta = c / (np.sqrt(t) + 1.0)
           
           x_new = mirror_descent_step(y, g, eta)

           # Check improvement
           M_new = fast_conv(x_new).max()
           if M_new < best_obj - 1e-15:
               best_obj = M_new
               best_x = x_new.copy()
               stall_count = 0
           else:
               stall_count += 1

           x = x_new

           # Periodic restart if stalled
           if stall_count > max_stall:
               stall_count = 0
               max_stall = min(max_stall * 1.5, 10000)
               # Perturb from best
               noise = rng.normal(0, 0.005, n)
               x = best_x * (1.0 + noise)
               x = np.maximum(x, 1e-10)
               x = x / x.sum()
               y = x.copy()
               y_prev = x.copy()
               if verbose:
                   elapsed = time.monotonic() - t0
                   print(f"    LSE-AMD restart at t={t} ({elapsed:.1f}s), obj={2*n*best_obj:.6f}")

           if verbose and t % 5000 == 0:
               elapsed = time.monotonic() - t0
               obj = 2.0 * n * best_obj
               print(f"    LSE-AMD t={t} ({elapsed:.1f}s): obj={obj:.6f}, T_frac={frac:.3f}")

       return best_x, best_obj

   def lp_frank_wolfe(x0, time_limit, solver=None, verbose=True):
       """
       Nesterov-accelerated Frank-Wolfe with LP subproblems.
       """
       if solver is None:
           solver = get_lp_solver()
       n = len(x0)
       x = norm_sum1(x0)
       best_x = x.copy()
       best_obj = fast_conv(x).max()
       t0 = time.monotonic()

       y = x.copy()
       x_mom = x.copy()
       iter_count = 0

       for t in range(1, 5000):
           if time.monotonic() - t0 > time_limit:
               break

           # Nesterov step
           beta_t = (max(t - 1, 0)) / (t + 2.0)
           y = x + beta_t * (x - x_mom)
           y = np.maximum(y, 1e-15)
           y = y / y.sum()
           x_mom = x

           # Compute current peak
           conv_y = fast_conv(y)
           M_curr = conv_y.max()
           if M_curr < 1e-14:
               break

           # Solve LP: max sum(b) s.t. b >= 0, (y*b)[k] <= M_curr for all k
           b_var = cp.Variable(n, nonneg=True)
           
           # Build constraint matrix: T[k, j] = y[j] for valid (k, j) pairs
           # T has shape (2n-1, n), T[k,j] = y[j] if 0 <= k-j < n else 0
           rows = []
           cols = []
           vals = []
           for j in range(n):
               if y[j] > 1e-15:
                   rows.extend(range(j, j + n))
                   cols.extend([j] * n)
                   vals.extend([y[j]] * n)
           
           T = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
           
           constraints = [T @ b_var <= M_curr]
           prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
           
           try:
               prob.solve(solver=solver, verbose=False)
           except Exception:
               continue
           
           if prob.status not in ('optimal', 'optimal_inaccurate'):
               continue
           
           b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
           b_sum = b_val.sum()
           if b_sum < 1e-15:
               continue
           v = b_val / b_sum  # Frank-Wolfe vertex

           # Line search or fixed step
           alpha = 2.0 / (t + 2.0)
           x_new = (1.0 - alpha) * x + alpha * v
           x_new = np.maximum(x_new, 0.0)
           sx = x_new.sum()
           if sx < 1e-15:
               x_new = x.copy()
           else:
               x_new = x_new / sx

           # Update
           M_xnew = fast_conv(x_new).max()
           if M_xnew < best_obj:
               best_obj = M_xnew
               best_x = x_new.copy()

           x = x_new
           iter_count += 1

           if verbose and t % 3 == 0:
               elapsed = time.monotonic() - t0
               obj = 2.0 * n * best_obj
