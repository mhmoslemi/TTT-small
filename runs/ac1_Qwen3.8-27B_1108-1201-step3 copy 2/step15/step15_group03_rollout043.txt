"""
Searches for non-negative step-function heights minimizing
    2*n*max(conv(a,a))/(sum(a))^2
by alternating:
  - a dual "slack" weight q over convolution lags,
  - cheap gradient-style updates of a with exact line search on the true max,
  - LP best-response moves when available,
  - smoothing/refinement with log-sum-exp weights,
  - adaptive restarts and exploration of nearby sequence lengths.
"""

import numpy as np
import time

try:
    from scipy.fft import rfft, irfft
except Exception:
    from numpy.fft import rfft, irfft

try:
    from scipy.optimize import linprog
    _HAVE_LINPROG = True
except Exception:
    _HAVE_LINPROG = False


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_sum(a, target=1.0):
    a = np.array(a, dtype=np.float64, copy=True)
    if a.size == 0:
        return np.array([1.0 / max(1, target)] if target > 0 else [0.0], dtype=np.float64)
    a = np.maximum(a, 0.0)
    floor = 1e-15 / max(1, a.size)
    a = np.maximum(a, floor)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.shape, target / a.size, dtype=np.float64)
    a *= target / s
    return a


def safe_normalize(v, fallback=None):
    v = np.asarray(v, dtype=np.float64)
    s = float(v.sum())
    if np.isfinite(s) and s > 1e-300:
        return v / s
    if fallback is None:
        return np.full(v.shape, 1.0 / max(1, v.size), dtype=np.float64)
    fb = np.array(fallback, dtype=np.float64, copy=True)
    fs = float(fb.sum())
    if np.isfinite(fs) and fs > 1e-300:
        return fb / fs
    return np.full(v.shape, 1.0 / max(1, v.size), dtype=np.float64)


def perturb_sequence(a, rng, sigma):
    n = a.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    x = a * np.exp(sigma * rng.standard_normal(n))
    x = np.maximum(x, 1e-14 / n)
    return normalize_sum(x, 1.0)


def self_conv(a, N):
    n = a.size
    A = rfft(a, N)
    c = irfft(A * A, N)
    return c[:2 * n - 1]


def cross_conv(a, b, N):
    size = a.size + b.size - 1
    A = rfft(a, N)
    B = rfft(b, N)
    c = irfft(A * B, N)
    return c[:size]


def conv_max_fast(a, N):
    return float(self_conv(a, N).max())


def hankel_product(a, q, N):
    """
    Returns h_i = sum_l q_l a_{l-i}, i=0..n-1, where out-of-range a_t are 0.
    This is half the gradient of sum_l q_l (a*a)_l.
    """
    n = a.size
    Af = rfft(a[::-1], N)
    Q = rfft(q, N)
    c = irfft(Af * Q, N)
    return c[n - 1:n - 1 + n].copy()


def top_k_indices(values, K):
    size = len(values)
    if K <= 0:
        return np.array([], dtype=int)
    if K >= size:
        return np.arange(size, dtype=int)
    if K == 1:
        return np.array([int(np.argmax(values))], dtype=int)
    idx = np.argpartition(values, -K)[-K:]
    return idx


def active_indices(c, m, frac):
    size = c.size
    if size == 0:
        return np.array([], dtype=int)
    if not np.isfinite(m) or m <= 0.0:
        return top_k_indices(c, size)
    frac = min(1.0, max(0.0, float(frac)))
    tol = 0.20 - 0.185 * frac
    mask = c >= m * (1.0 - tol)
    nmask = int(np.count_nonzero(mask))
    minK = min(size, max(5, size // 40))
    if nmask < minK:
        return top_k_indices(c, minK)
    return np.nonzero(mask)[0]


def game_q_update(a, q_prev, active_idx, N, size, frac, substeps=1):
    """
    Update dual lag weights q for current a, focusing on active lags.
    Returns (q, h) where h corresponds to the final q.
    """
    if active_idx.size == 0:
        active_idx = np.arange(size, dtype=int)

    active_mask = np.zeros(size, dtype=bool)
    active_mask[active_idx] = True
    active_uniform = np.zeros(size, dtype=np.float64)
    active_uniform[active_idx] = 1.0 / active_idx.size

    q = np.asarray(q_prev, dtype=np.float64).copy()
    s_active = float(q[active_idx].sum())
    if np.isfinite(s_active) and s_active > 1e-8:
        q_active = q.copy()
        q_active[~active_mask] = 0.0
        q_active = safe_normalize(q_active, active_uniform)
    else:
        q_active = active_uniform.copy()

    base = 0.5 * q_active + 0.5 * active_uniform
    q_work = base.copy()

    frac = min(1.0, max(0.0, float(frac)))
    for _ in range(max(1, int(substeps))):
        h_ref = hankel_product(a, q_work, N)
        range_h = float(h_ref.max() - h_ref.min())
        Tp = max(1e-15, 0.25 * range_h)
        p = np.exp(-(h_ref - h_ref.min()) / Tp)
        p = safe_normalize(p, a)

        cross = cross_conv(a, p, N)
        max_cross = float(cross.max())
        Tq = max(1e-15, (0.50 - 0.35 * frac) * max_cross)
        add = cross / Tq
        add = np.clip(add, 0.0, 10.0)

        q_new = q_work * np.exp(add)
        q_new = safe_normalize(q_new, q_work)
        q_new[~active_mask] = 0.0
        q_new = safe_normalize(q_new, active_uniform)

        q_work = 0.65 * q_new + 0.25 * q_active + 0.10 * active_uniform
        q_work = safe_normalize(q_work, active_uniform)

    h_last = hankel_product(a, q_work, N)
    return q_work, h_last


def line_search_update(a, direction, N, cur_m, step_rel=0.1, use_mult=True):
    """
    Try additive and multiplicative steps along a centered direction.
    Returns (candidate, new_m), or (None, cur_m).
    """
    n = a.size
    dmax = float(np.max(np.abs(direction)))
    if not np.isfinite(dmax) or dmax <= 1e-300:
        return None, cur_m

    best_a = None
    best_m = float(cur_m)
    flat_tol = 1e-9

    target = max(1e-18, step_rel / n)
    scale = target / dmax

    for f in (1.0, 0.5, 0.25, 0.125):
        a_new = a + scale * f * direction
        a_new = normalize_sum(a_new, 1.0)
        m_new = conv_max_fast(a_new, N)
        if np.isfinite(m_new):
            if m_new < best_m - 1e-12 * max(1.0, best_m):
                best_m = m_new
                best_a = a_new
                break
            if best_a is None and m_new <= cur_m * (1.0 + flat_tol) and m_new <= best_m:
                best_a = a_new
                best_m = m_new

    if best_a is not None:
        return best_a, best_m

    if use_mult:
        for eta in (0.08, 0.04, 0.02, 0.01):
            e = eta * direction / dmax
            e = np.clip(e, -8.0, 8.0)
            a_new = a * np.exp(e)
            a_new = normalize_sum(a_new, 1.0)
            m_new = conv_max_fast(a_new, N)
            if np.isfinite(m_new):
                if m_new < best_m - 1e-12 * max(1.0, best_m):
                    return a_new, m_new
                if best_a is None and m_new <= cur_m * (1.0 + flat_tol) and m_new <= best_m:
                    best_a = a_new
                    best_m = m_new
        if best_a is not None:
            return best_a, best_m

    return None, cur_m


def internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    N = next_pow2(2 * n - 1)
    return float(2.0 * n * conv_max_fast(a, N))


def lse_smooth_optimize(a0, time_limit, rng, verbose=False):
    a = normalize_sum(a0, 1.0)
    n = a.size
    size = 2 * n - 1
    N = next_pow2(size)
    c = self_conv(a, N)
    m = float(c.max())
    best_score = float(2.0 * n * m)
    best_a = a.copy()

    if time_limit <= 0.0:
        return best_a, best_score

    v = np.zeros(n, dtype=np.float64)
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(1e-12, time_limit))

        c = self_conv(a, N)
        m = float(c.max())
        score = float(2.0 * n * m)
        if np.isfinite(score) and score < best_score - 1e-13 * max(1.0, best_score):
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            a = perturb_sequence(a, rng, 0.05)
            v = np.zeros(n, dtype=np.float64)
            no_improve = 0
            continue

        T_rel = max(0.002, (0.002) ** frac)
        T = max(1e-300, m * T_rel)
        shift = (c - m) / T
        shift = np.maximum(shift, -50.0)
        w = np.exp(shift)
        w = safe_normalize(w)

        h = hankel_product(a, w, N)
        d = h - h.mean()
        v = 0.80 * v + d
        direction = -v
        direction -= direction.mean()

        step_rel = 0.25 - 0.15 * frac
        cand, m_new = line_search_update(a, direction, N, m, step_rel, True)
        if cand is not None:
            a = cand
            score = float(2.0 * n * m_new)
            if np.isfinite(score) and score < best_score - 1e-13 * max(1.0, best_score):
                best_score = score
                best_a = a.copy()
                no_improve = 0
            v *= 0.85
        else:
            v *= 0.5

        if verbose and it % 250 == 0:
            print(f"    LSE it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    final_score = float(2.0 * n * conv_max_fast(a, N))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def active_game_optimize(a0, time_limit, rng, verbose=False):
    a = normalize_sum(a0, 1.0)
    n = a.size
    size = 2 * n - 1
    N = next_pow2(size)

    c = self_conv(a, N)
    m = float(c.max())
    best_score = float(2.0 * n * m)
    best_a = a.copy()

    if time_limit <= 0.0:
        return best_a, best_score

    T_init = max(1e-300, 0.25 * m)
    w0 = np.exp(np.maximum((c - m) / T_init, -50.0))
    q = safe_normalize(w0)

    v = np.zeros(n, dtype=np.float64)
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    substeps = 2 if n < 3500 else 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(1e-12, time_limit))

        c = self_conv(a, N)
        m = float(c.max())
        score = float(2.0 * n * m)
        if np.isfinite(score) and score < best_score - 1e-13 * max(1.0, best_score):
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 35:
            a = perturb_sequence(a, rng, 0.06 if n > 2500 else 0.07)
            c = self_conv(a, N)
            m = float(c.max())
            score = float(2.0 * n * m)
            if np.isfinite(score) and score < best_score - 1e-13 * max(1.0, best_score):
                best_score = score
                best_a = a.copy()
            T_init = max(1e-300, 0.25 * m)
            w0 = np.exp(np.maximum((c - m) / T_init, -50.0))
            q = safe_normalize(w0)
            v = np.zeros(n, dtype=np.float64)
            no_improve = 0
            continue

        active_idx = active_indices(c, m, frac)
        q, h = game_q_update(a, q, active_idx, N, size, frac, substeps)

        d = h - h.mean()
        dmax = float(np.max(np.abs(d)))
        if np.isfinite(dmax) and dmax > 1e-300:
            v = 0.70 * v + d

        direction = -v
        direction -= direction.mean()
        step_rel = 0.25 - 0.15 * frac
        cand, m_new = line_search_update(a, direction, N, m, step_rel, True)

        if cand is not None:
            a = cand
            score = float(2.0 * n * m_new)
            if np.isfinite(score) and score < best_score - 1e-13 * max(1.0, best_score):
                best_score = score
                best_a = a.copy()
                no_improve = 0
            v *= 0.85
        else:
            v *= 0.5

        if verbose and it % 250 == 0:
            print(f"    ActiveGame it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    final_score = float(2.0 * n * conv_max_fast(a, N))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def lp_best_response(a, active_idx, current_m):
    if not _HAVE_LINPROG:
        return None

    n = len(a)
    K = len(active_idx)
    if K == 0:
        return None

    A_ub = np.zeros((K, n + 1), dtype=np.float64)
    for r, lag in enumerate(active_idx):
        lo = max(0, int(lag) - n + 1)
        hi = min(n, int(lag) + 1)
        if hi > lo:
            idx = np.arange(lo, hi, dtype=int)
            A_ub[r, lo:hi] = a[lag - idx]

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0

    cvec = np.zeros(n + 1, dtype=np.float64)
    cvec[n] = 1.0

    upper_t = max(1e-300, current_m * 1.01 + 1e-12)
    bounds = [(0.0, 1.0)] * n + [(0.0, upper_t)]

    try:
        res = linprog(
            cvec,
            A_ub=A_ub,
            b_ub=np.zeros(K, dtype=np.float64),
            A_eq=A_eq,
            b_eq=np.array([1.0], dtype=np.float64),
            bounds=bounds,
            method="highs",
            options={"time_limit": 1.0, "presolve": True},
        )
    except Exception:
        try:
            res = linprog(
                cvec,
                A_ub=A_ub,
                b_ub=np.zeros(K, dtype=np.float64),
                A_eq=A_eq,
                b_eq=np.array([1.0], dtype=np.float64),
                bounds=bounds,
            )
        except Exception:
            return None

    if not getattr(res, "success", False) or res.x is None:
        return None

    b = np.asarray(res.x[:n], dtype=np.float64)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    b /= s
    return b


def lp_polish(a0, time_limit, rng, verbose=False):
    a = normalize_sum(a0, 1.0)
    n = a.size
    size = 2 * n - 1
    N = next_pow2(size)

    t0 = time.monotonic()
    c = self_conv(a, N)
    m = float(c.max())
    best_score = float(2.0 * n * m)
    best_a = a.copy()

    if time_limit <= 10.0 or not _HAVE_LINPROG:
        return best_a, best_score

    K = 80 if n < 2500 else (50 if n < 5000 else 30)
    max_iters = 300 if n < 2000 else (160 if n < 4000 else 80)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        c = self_conv(a, N)
        m = float(c.max())
        score = float(2.0 * n * m)
        if np.isfinite(score) and score < best_score - 1e-13 * max(1.0, best_score):
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 12:
            a = perturb_sequence(a, rng, 0.03)
            no_improve = 0
            continue

        active_idx = top_k_indices(c, K)
        b = lp_best_response(a, active_idx, m)
        if b is None:
            no_improve += 1
            continue

        accepted = False
        for alpha in (0.5, 0.3, 0.15, 0.075, 0.04):
            cand = normalize_sum((1.0 - alpha) * a + alpha * b, 1.0)
            m_new = conv_max_fast(cand, N)
            s_new = float(2.0 * n * m_new)
            if np.isfinite(s_new) and s_new < score - 1e-12 * max(1.0, score):
                a = cand
                score = s_new
                accepted = True
                break

        if accepted:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP polish it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    final_score = float(2.0 * n * conv_max_fast(a, N))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def structured_init(n, rng):
    x = np.linspace(0.0, 1.0, n)
    r = float(rng.random())
    if r < 0.10:
        a = np.ones(n)
    elif r < 0.20:
        a = 1.0 + 0.35 * np.cos(2.0 * np.pi * 3.0 * x)
    elif r < 0.30:
        a = 1.0 + 0.30 * np.sin(4.0 * np.pi * x)
    elif r < 0.40:
        a = np.abs(np.sin(np.pi * x)) ** 0.8
    elif r < 0.50:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif r < 0.60:
        a = 0.5 + 0.3 * np.cos(2.0 * np.pi * 5.0 * x)
    elif r < 0.70:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.2
    elif r < 0.80:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.6
    elif r < 0.90:
        a = np.exp(-2.0 * x) + 0.3 + 0.2 * np.sin(3.0 * np.pi * x)
    else:
        a = np.exp(-8.0 * (x - 0.5) ** 2) + 0.1
    a = np.maximum(a, 0.0)
    a = a + 0.02 * rng.random(n)
    return normalize_sum(a, 1.0)


def interpolate_seq(f, new_n):
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = f.size
    new_n = int(new_n)
    if old_n == 0 or new_n <= 0:
        return np.ones(max(1, new_n), dtype=np.float64)
    if old_n == new_n:
        return normalize_sum(f, 1.0)

    s = float(f.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(new_n, dtype=np.float64)

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f / s)
    g = np.maximum(g, 0.0)
    return normalize_sum(g, 1.0)


def update_best(a, score, best_a, best_score, label, t_start):
    if a is None or score is None or not np.isfinite(score):
        return best_a, best_score
    if best_a is None:
        print(f"  base best {label}: score={float(score):.8f} n={a.size}")
        return np.array(a, dtype=np.float64, copy=True), float(score)
    if score < best_score - 1e-12 * max(1.0, best_score):
        elapsed = time.monotonic() - t_start if t_start is not None else 0.0
        print(f"  *** NEW BEST {label}: score={float(score):.8f} n={a.size} t={elapsed:.0f}s")
        return np.array(a, dtype=np.float64, copy=True), float(score)
    return best_a, best_score


def improve_sequence(a0, time_limit, rng, prefix=""):
    t_start = time.monotonic()
    a = normalize_sum(a0, 1.0)
    score = internal_score(a)
    best_a = a.copy()
    best_score = score

    if time_limit <= 10.0:
        return best_a, best_score

    # Main active-game work.
    t1 = max(10.0, min(0.55 * time_limit, time_limit - 5.0))
    a1, s1 = active_game_optimize(a, t1, rng, verbose=False)
    if np.isfinite(s1) and s1 < best_score - 1e-12 * max(1.0, best_score):
        best_a, best_score = a1, s1

    elapsed = time.monotonic() - t_start
    rem = time_limit - elapsed

    # LSE smoothing/refinement.
    if rem > 30.0:
        t2 = min(rem * 0.30, 180.0)
        a2, s2 = lse_smooth_optimize(best_a, t2, rng, verbose=False)
        if np.isfinite(s2) and s2 < best_score - 1e-12 * max(1.0, best_score):
            best_a, best_score = a2, s2
        elapsed = time.monotonic() - t_start
        rem = time_limit - elapsed

    # Another active-game pass.
    if rem > 20.0:
        t3 = min(rem * 0.35, 300.0)
        a3, s3 = active_game_optimize(best_a, t3, rng, verbose=False)
        if np.isfinite(s3) and s3 < best_score - 1e-12 * max(1.0, best_score):
            best_a, best_score = a3, s3
        elapsed = time.monotonic() - t_start
        rem = time_limit - elapsed

    # LP polish if time remains.
    if rem > 30.0:
        t4 = min(rem * 0.75, 400.0)
        a4, s4 = lp_polish(best_a, t4, rng, verbose=False)
        if np.isfinite(s4) and s4 < best_score - 1e-12 * max(1.0, best_score):
            best_a, best_score = a4, s4
        elapsed = time.monotonic() - t_start
        rem = time_limit - elapsed

    # Small final active-game polish.
    if rem > 15.0:
        t5 = min(rem * 0.9, 120.0)
        a5, s5 = active_game_optimize(best_a, t5, rng, verbose=False)
        if np.isfinite(s5) and s5 < best_score - 1e-12 * max(1.0, best_score):
            best_a, best_score = a5, s5

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    try:
        total_budget = min(float(budget_s) - 45.0, 1755.0)
    except Exception:
        total_budget = 1700.0
    total_budget = max(60.0, total_budget)

    best_a = None
    best_score = float("inf")
    initial_a = None

    # Try to load the supplied parent construction.
    try:
        parent = height_sequence_1
        p = np.asarray(parent, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and float(p.sum()) > 1e-6:
            best_a = normalize_sum(p, 1.0)
            best_score = internal_score(best_a)
            initial_a = best_a.copy()
            print(f"Parent loaded: n={best_a.size}, internal score={best_score:.8f}")
    except Exception:
        best_a = None

    if best_a is None:
        base_n = 1600
        best_a = structured_init(base_n, rng)
        best_score = internal_score(best_a)
        initial_a = best_a.copy()
        print(f"Default structured init: n={best_a.size}, internal score={best_score:.8f}")

    # Warm up the current best length.
    first_budget = min(120.0, 0.08 * total_budget)
    if first_budget > 30.0:
        a1, s1 = improve_sequence(best_a, first_budget, rng, "base")
        best_a, best_score = update_best(a1, s1, best_a, best_score, "base", t0)

    base_n = int(best_a.size)

    # Candidate lengths to explore. Longer sequences often have more slack,
    # but very long ones are slower and may get over-interpolated.
    optional_lengths = []
    for mult in (1.25, 1.5, 2.0, 2.5):
        L = int(round(base_n * mult))
        if 200 <= L <= 6000 and abs(L - base_n) >= max(50, base_n // 50):
            if L not in optional_lengths:
                optional_lengths.append(L)
    if base_n > 1200:
        L = int(round(base_n * 0.8))
        if 200 <= L <= 6000 and L not in optional_lengths:
            optional_lengths.append(L)
    optional_lengths = optional_lengths[:4]

    # Explore nearby/longer lengths while protecting final time.
    if len(optional_lengths) > 0:
        elapsed = time.monotonic() - t0
        reserve_final = max(250.0, 0.35 * total_budget)
        explore_budget = total_budget - elapsed - reserve_final
        if explore_budget > 60.0:
            per = min(220.0, max(45.0, explore_budget / len(optional_lengths)))
            for L in optional_lengths:
                elapsed = time.monotonic() - t0
                if total_budget - elapsed < 180.0:
                    break
                b = min(per, total_budget - elapsed - 160.0)
                if b < 35.0:
                    break
                print(f"Trying length {L} with budget {b:.0f}s")
                init_a = interpolate_seq(best_a, L)
                aL, sL = improve_sequence(init_a, b, rng, f"len{L}")
                best_a, best_score = update_best(aL, sL, best_a, best_score, f"len{L}", t0)

    # Final main optimization on the current best.
    elapsed = time.monotonic() - t0
    final_time = total_budget - elapsed
    if final_time > 60.0:
        main_final = min(final_time * 0.75, 1200.0)
        print(f"Final main optimization: budget={main_final:.0f}s, best so far={best_score:.8f}")
        aF, sF = improve_sequence(best_a, main_final, rng, "final")
        best_a, best_score = update_best(aF, sF, best_a, best_score, "final", t0)

        elapsed = time.monotonic() - t0
        rem = total_budget - elapsed
        if rem > 35.0:
            aP, sP = lp_polish(best_a, max(20.0, rem * 0.9 - 5.0), rng, verbose=True)
            best_a, best_score = update_best(aP, sP, best_a, best_score, "final-lp", t0)

    if best_a is None:
        return [1.0]

    out = [max(0.0, min(1000.0, float(x))) for x in np.asarray(best_a, dtype=np.float64)]
    if len(out) == 0 or sum(out) <= 1e-12:
        out = [1.0]

    # Safety net: if the official evaluator disagrees and parent is better, return parent.
    try:
        final_eval = float(evaluate_sequence(out))
        init_eval = float("inf")
        if initial_a is not None:
            init_list = [max(0.0, min(1000.0, float(x))) for x in np.asarray(initial_a, dtype=np.float64)]
            if len(init_list) > 0 and sum(init_list) > 1e-12:
                init_eval = float(evaluate_sequence(init_list))
        if (not np.isfinite(final_eval)) or (np.isfinite(init_eval) and final_eval > init_eval + 1e-8):
            if initial_a is not None:
                out = [max(0.0, min(1000.0, float(x))) for x in np.asarray(initial_a, dtype=np.float64)]
                if len(out) > 0:
                    final_eval = float(evaluate_sequence(out))
            else:
                out = [1.0, 1.0, 1.0]
        print(f"Final official score: {final_eval:.8f}")
    except Exception:
        pass

    print(f"Final returned: n={len(out)}, elapsed={time.monotonic() - t0:.0f}s")
    return out
