            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_rows(a, indices):
    """Build rows of the convolution matrix T for given constraint indices."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        if k >= n:
            # (a*b)_k = sum_i a[i] * b[i + k - n + 1] ... actually let me just use the definition
            # (a*b)_k = sum_j a[j] * b[k-j] where valid indices
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                T[idx, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
        else:
            j_lo = 0
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                T[idx, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return T


def lp_solve_fast(a, M, solver, active_indices):
    """Solve the best-response LP with given active constraints."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_rows(a, active_indices)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    return val


def lp_solve_full(a, M, solver):
    """Solve the full best-response LP with all constraints."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((size, n))
    for k in range(size):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            T[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    return val


def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return float(2.0 * n * max_b / (S * S))


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_seq(a, target_norm):
    """Normalize sequence to have the given sum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_norm / len(a)
    return a * target_norm / s


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, use_multiplicative=False, screening_threshold=0.85):
    """
    Iterated LP best-response with Nesterov acceleration and full running average.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    # Full running average
    avg_a = np.zeros(n)
    n_avg = 0
    no_improve_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        if t < 30:
            thresh = 0.5
        elif t < 100:
            thresh = screening_threshold - 0.1
        elif t < 300:
            thresh = screening_threshold
        else:
            thresh = min(0.95, screening_threshold + 0.05)

        active = np.where(conv_y >= M * thresh)[0]
        if len(active) < 20:
            active = np.where(conv_y >= M * 0.4)[0]
        if len(active) > size:
            active = np.arange(size)

        # Solve LP
        b = lp_solve_fast(y, M, solver, active)
        if b is None:
            b = lp_solve_full(y, M, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints
        if len(active) < size:
            check = fast_conv_cross(y, g)
            if check.max() > M * 1.01:
                b_full = lp_solve_full(y, M, solver)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    g = b * norm / b.sum()

        # Update
        alpha = 2.0 / (t + 2.0)
        if use_multiplicative and t > 50:
            ratio = g / np.maximum(y, 1e-30)
            log_ratio = np.log(np.clip(ratio, 1e-15, 1e15))
            y_new = y * np.exp(alpha * log_ratio)
        else:
            y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new

        # Nesterov extrapolation for tracking iterate
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - a)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        a = a_new

        # Track running average
        avg_a += a
        n_avg += 1

        # Check scores
        improved = False
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        # Check average more frequently (every 20 steps)
        if n_avg % 20 == 0 and n_avg >= 50:
            avg_check = avg_a / n_avg
            sc_avg = fast_conv_self(avg_check).max()
            if sc_avg < best_score - 1e-10:
                best_score = sc_avg
                best_a = avg_check.copy()
                improved = True

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        # Perturbation to escape stagnation
        if no_improve_count == 150:
            n_pert = max(5, n // 30)
            idx_p = np.random.default_rng(t).choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.3 * np.random.default_rng(t + 7).standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final check of average
    if n_avg > 0:
        avg_final = avg_a / n_avg
        sc_final = fast_conv_self(avg_final).max()
        if sc_final < best_score:
            best_score = sc_final
            best_a = avg_final.copy()

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=False):
    """LSE smoothed gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(80.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)
            for trial in range(12):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.03 * T_val:
                    break
                step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a = a.copy()
            else:
                a = a_new * norm / s_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with peak-based candidate selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))

            peak_lags = np.where(conv >= cur_max * 0.92)[0]
            for kl in peak_lags[:10]:
                j1 = int(kl - i)
                j2 = int(kl + i)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute change in convolution when moving delta from i to j
                # (a*a)_k changes by: delta * (a[k-i] + a[i-k] - a[k-j] - a[j-k])
                # plus second-order terms at 2i and 2j
                g = np.zeros(size)
                # +delta at j: adds delta * a[k-j] for valid k
                for k in range(max(0, j), min(size, j + n)):
                    g[k] += a[k - j]
                # Also the symmetric part: a[j - (k-j)] = a[2j-k] for k < j
                for k in range(max(0, 2 * j - n + 1), min(j + 1, size)):
                    g[k] += a[2 * j - k] if 2 * j - k < n else 0.0

                # -delta at i: subtracts delta * a[k-i] for valid k
                for k in range(max(0, i), min(size, i + n)):
                    g[k] -= a[k - i]
                for k in range(max(0, 2 * i - n + 1), min(i + 1, size)):
                    g[k] -= a[2 * i - k] if 2 * i - k < n else 0.0

                n_samples = 15
                deltas = np.linspace(0.02, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 0.01:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / max(a.sum(), 1e-15)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_pert = max(10, n // 25)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def dual_gradient_refine(a0, time_limit, solver, verbose=False):
    """Use LP dual variables for gradient refinement."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(100):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break

        active = np.where(conv >= M * 0.8)[0]
        if len(active) < 20:
            active = np.arange(size)

        # Solve LP and get dual
        b_var = cp.Variable(n, nonneg=True)
        T = build_T_rows(a, active)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        # Get dual values
        duals = None
        try:
            con = constraints[0]
            if con.dual_value is not None:
                full_duals = np.zeros(size)
                full_duals[active] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
        except Exception:
            pass

        if duals is None or duals.sum() < 1e-15:
            # Fallback: weight by constraint tightness
            duals = np.zeros(size)
            tight = conv >= M * 0.95
            duals[tight] = 1.0
            if duals.sum() < 1e-15:
                duals[np.argmax(conv)] = 1.0

        # Compute gradient from duals
        A = rfft(a, fft_size)
        D = rfft(duals, fft_size)
        g1 = irfft(np.conj(A) * D, fft_size)[:n]
        g2 = irfft(A * D, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Line search
        step = 0.05
        for trial in range(15):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < M - 1e-10:
                a = a_new
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                break
            step *= 0.5

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    return f_interp * np.sqrt(2.0 * new_n) / ss


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        # Broader window
        a = np.cos(np.pi * (x - 0.5)) ** 0.8
        a = np.maximum(a, 0.01)
    else:
        # Multi-scale oscillation
        a = 0.5 + 0.2 * np.cos(6 * np.pi * x) + 0.15 * np.cos(17 * np.pi * x) + 0.1 * np.cos(31 * np.pi * x)
        a = np.maximum(a, 0.02)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_fn(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(10):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick LSE warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = min(remaining * 0.05, 60.0)
    print(f"\n=== Phase 0: LSE warmup (budget={t0_phase:.0f}s) ===")
    if t0_phase > 15:
        a_out, _ = lse_gradient_descent(best_seq, t0_phase, verbose=False)
        try_update(a_out, "lse_warmup")

    # Phase 1: Main LP Nesterov (additive)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: LP Nesterov (additive) n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t1, solver, verbose=True, use_multiplicative=False, screening_threshold=0.85)
        try_update(a_out, f"lp1_n{n1}")

    # Phase 2: LP Nesterov (multiplicative)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LP Nesterov (multiplicative) n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True, use_multiplicative=True, screening_threshold=0.88)
        try_update(a_out, f"lp2_n{n2}")

    # Phase 3: Dual gradient refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 3: Dual Gradient n={n3} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = dual_gradient_refine(best_seq, t3, solver, verbose=False)
        try_update(a_out, f"dualgrad_n{n3}")

    # Phase 4: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.18, 200.0)
    print(f"\n=== Phase 4: Pair CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{n4}")

    # Phase 5: LP round 3 (tighter screening)
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.18, 200.0)
    print(f"\n=== Phase 5: LP Nesterov (tight) n={n5} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t5, solver, verbose=True, use_multiplicative=False, screening_threshold=0.92)
        try_update(a_out, f"lp3_tight_n{n5}")

    # Phase 6: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 300
    print(f"\n=== Phase 6: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.45, 180.0)
        a_out, _ = lp_nesterov_optimize(init_long, t_long, solver, verbose=False, use_multiplicative=False, screening_threshold=0.85)
        try_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_long:
            t_long_cd = min(remaining * 0.5, 60.0)
            a_out2, _ = pair_cd_optimize(best_seq, t_long_cd, rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 7: Final polish with remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        t_f1 = min(remaining * 0.3, 50.0)
        a_out, _ = lse_gradient_descent(best_seq, t_f1, verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_f2 = min(remaining * 0.4, 60.0)
            a_out, _ = pair_cd_optimize(best_seq, t_f2, rng, verbose=False)
            try_update(a_out, "pcd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_f3 = min(remaining * 0.5, 40.0)
            a_out, _ = dual_gradient_refine(best_seq, t_f3, solver, verbose=False)
            try_update(a_out, "dual_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_f4 = min(remaining * 0.7, 30.0)
            a_out, _ = lp_nesterov_optimize(best_seq, t_f4, solver, verbose=False, use_multiplicative=True, screening_threshold=0.9)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure we return well before the deadline
    return [float(x) for x in best_seq]
