def compute_soft_subgradient(a, conv, beta):
    n = len(a)
    size = 2 * n - 1
    # Soft weights: w_k = exp(beta * (c_k - max_c)) / sum exp
    max_c = conv.max()
    scaled = beta * (conv - max_c)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 0)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(size)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # Compute SG[i] = sum_k w[k] * 2*a[k-i] * I[0<=k-i<n]
    SG = np.zeros(n)
    threshold = 1e-12
    for k in range(size):
        if w[k] < threshold:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_hi >= i_lo:
            # a[k-i] for i in [i_lo, i_hi] = a[k-i_lo], a[k-i_lo-1], ..., a[k-i_hi]
            # = a[k-i_lo : k-i_hi-1 : -1]
            SG[i_lo:i_hi + 1] += w[k] * a[k - i_lo:k - i_hi - 1:-1]
    SG *= 2.0
    return SG


def mirror_descent_nesterov(a0, time_limit, beta_init=50.0, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv_max = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()  # Nesterov momentum variable
    t_mom = 1.0
    eta = 1.0 / n**0.5
    patience = 0
    n_iters = 0
    rng = np.random.default_rng(7777)

    while time.monotonic() - t0 < time_limit:
        batch_end = time.monotonic() + 0.05  # Process in ~50ms batches
        
        while time.monotonic() < batch_end and time.monotonic() - t0 < time_limit:
            conv_y = fast_autocorr(y)
            M = conv_y.max()
            if M < 1e-15:
                break

            # Compute soft subgradient
            SG = compute_soft_subgradient(y, conv_y, beta_init)
            sg_norm = np.linalg.norm(SG)
            if sg_norm < 1e-15:
                break
            SG = SG / sg_norm

            # Mirror descent update (exponential family)
            log_y = np.log(y + 1e-30)
            y_new = np.exp(log_y - eta * SG)
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_new

            # Nesterov update
            alpha = 2.0 / (t_mom + 2.0)
            y = (1.0 - alpha) * y + alpha * y_new
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = y_new.copy()
            else:
                y = y * norm / s_y

            if t_mom > 1.0:
                momentum = (t_mom - 1.0) / (t_mom + 2.0)
                a = y + momentum * (y - y_new)
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a < 1e-15:
                    a = y.copy()
                else:
                    a = a * norm / s_a
            else:
                a = y.copy()

            t_mom += 1.0
            n_iters += 1

            # Track best
            conv_a = fast_autocorr(a)
            cur_max = conv_a.max()
            if cur_max < best_conv_max - 1e-13:
                best_conv_max = cur_max
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            # Adaptive eta decay
            if patience % 200 == 0 and patience > 0:
                eta *= 0.95

            # Restart if stuck
            if patience > 500:
                y = best_a.copy()
                a = best_a.copy()
                t_mom = 1.0
                eta = 1.0 / n**0.5
                patience = 0
                # Add small perturbation
                noise = rng.normal(0, norm * 0.01, n)
                y = np.maximum(y + noise, 0)
                s = y.sum()
                if s > 1e-15:
                    y = y * norm / s
                    a = y.copy()

    return best_a, best_conv_max


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    try:
        import cvxpy as cp
        b_var = cp.Variable(n, nonneg=True)
        # Build constraint: conv(a, b)[k] <= M for all k
        # conv(a,b)[k] = sum_j a[k-j]*b[j] for valid j
        # This is T @ b where T[k,j] = a[k-j] if 0<=k-j<n else 0
        # Build sparse-ish
        constraints = []
        for k in range(size):
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_hi >= i_lo:
                row = np.zeros(n)
                row[i_lo:i_hi + 1] = a[k - i_lo:k - i_hi - 1:-1]
                constraints.append(row @ b_var <= M)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            if b_val.sum() > 1e-15:
                return b_val
    except Exception:
        pass
    return None

        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)

            continue

        patience = 0
        b = b * norm / b.sum()
        n_iters += 1

        alpha = 2.0 / (t_mom + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)

        if s_y < 1e-15:

        else:
            y_new = y_new * norm / s_y

        if t_mom > 1.0:
            momentum = (t_mom - 1.0) / (t_mom + 2.0)
            a = y_new + momentum * (y_new - y)
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a < 1e-15:
                a = y_new.copy()
            else:
                a = a * norm / s_a
        else:
            a = y_new.copy()

        y = y_new
        t_mom += 1.0

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-13:
            best_score = score_a
            best_a = a.copy()

        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-13:
            best_score = score_y
            best_a = y.copy()

        if (n_iters % 50 == 0) and patience > 30:
            y = best_a.copy()
            a = best_a.copy()
            t_mom = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                print(f"    LP restart {restart_count} at iter={n_iters}: best={best_score:.8f}")

        if verbose and n_iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

def coordinate_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(4242)
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit and no_improve < 100:
        n_passes += 1
        conv = fast_autocorr(a)
        max_val = conv.max()
        # Find top peaks
        top_k = min(10, size)
        active = np.argsort(conv)[-top_k:]
        improved = False

        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue
            # Try moving mass from i to j
            for offset in [1, -1, 2, -2, 3, -3, 5, -5, 10, -10, 25, -25]:
                j = (i + offset) % n
                if j == i:
                    continue
                for frac in [0.3, 0.5, 0.7]:
                    d = a[i] * frac
                    a[i] -= d
                    a[j] += d
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        no_improve = 0
                        break
                    else:
                        a[i] += d
                        a[j] -= d

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                # Perturb
                noise = rng.normal(0, norm * 0.003, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s = a_pert.sum()
                if s > 1e-15:
                    a = a_pert * norm / s
                no_improve = 0

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_shaping(a, strength=0.3):
    """Flatten the autocorrelation by modifying the power spectrum."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a_norm = a / s

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_norm, fft_size)
    power = np.abs(A)**2

    # Compress the dynamic range of the power spectrum
    # This makes the autocorrelation flatter
    p_min = power[power > 1e-10].min() if np.any(power > 1e-10) else 1e-10
    p_max = power.max()
    if p_max < 1e-10:
        return a

    # Geometric interpolation between p_min and p_max
    log_power = np.log(power + 1e-30)
    lp_min = np.log(p_min + 1e-30)
    log_power_clipped = np.clip(log_power, lp_min, None)

    # Blend: new_power = power^(1-strength) * (flat)^strength
    # Flat = p_min everywhere (white noise spectrum)
    # In log domain: new_log = (1-strength)*log_power + strength*lp_min
    new_log_power = (1.0 - strength) * log_power_clipped + strength * lp_min
    new_power = np.exp(new_log_power)

    # Recover with original phases
    phase = A / (np.abs(A) + 1e-30)
    new_A = np.sqrt(new_power) * phase
    new_a = irfft(new_A, fft_size)[:n]

    # Project to non-negative
    new_a = np.maximum(new_a, 0.0)
    s_new = new_a.sum()
    if s_new < 1e-15:
        return a
    return new_a * s / s_new


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to use parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default uniform init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    solver = get_solver()

    # Phase 1: Mirror Descent Nesterov (40% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: Mirror Descent + Nesterov (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = mirror_descent_nesterov(a_norm, t1, beta_init=100.0, verbose=True)
        update_best(a_out, "md_nesterov")

    # Phase 2: LP Nesterov (25% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: LP Nesterov n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        update_best(a_out, "lp_nesterov")

    # Phase 3: Spectral shaping + MD refinement (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Spectral Shaping + MD n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_curr = best_seq.copy()
        rng = np.random.default_rng(99)
        t3_start = time.monotonic()
        n_spec_iters = 0
        while time.monotonic() - t3_start < t3:
            # Apply spectral shaping
            a_shaped = spectral_shaping(a_curr, strength=0.2 + 0.3 * rng.random())
            # Quick MD refinement
            t_sub = min(10.0, (time_budget - (time.monotonic() - t_global)) * 0.3)
            if t_sub < 2:
                break
            a_norm = a_shaped * np.sqrt(2.0 * n_best) / a_shaped.sum()
            a_out, _ = mirror_descent_nesterov(a_norm, t_sub, beta_init=80.0, verbose=False)
            update_best(a_out, f"spec_md_{n_spec_iters}")
            if a_out is not None:
                a_curr = a_out
            n_spec_iters += 1

    # Phase 4: LP Nesterov round 2 (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: LP Nesterov R2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        update_best(a_out, "lp_r2")

    # Phase 5: Coordinate Descent polishing (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: CD Polish n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t5, verbose=True)
        update_best(a_out, "cd_polish")

    # Phase 6: Final LP + CD (last 5%)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t6a = min(remaining * 0.5, 60.0)
        t6b = remaining - t6a - 5
        if t6a > 10:
            print(f"\n=== Phase 6a: Final LP (budget={t6a:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t6a, solver, verbose=False)
            update_best(a_out, "lp_final")
        if t6b > 10:
            print(f"\n=== Phase 6b: Final CD (budget={t6b:.0f}s) ===")
            n_best = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coordinate_descent(a_norm, t6b, verbose=False)
            update_best(a_out, "cd_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence, returning uniform!")
        return [1.0]

    return [float(x) for x in best_seq]
