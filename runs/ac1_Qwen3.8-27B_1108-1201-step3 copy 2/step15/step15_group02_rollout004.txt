def lp_best_response_v2(a, M, solver, active_set=None):
    """LP to find best response b that maximizes sum(b) subject to conv(a,b) <= M."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and 0 < len(active_set) < size:
        # Build constraint matrix for active set
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[idx, i] = a[k - i]
        constraints = [T @ b_var <= M]
    else:
        # Full constraint matrix
        T = np.zeros((size, n))
        for m in range(n):
            T[m:m + n, m] = a
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 25
    temps = np.geomspace(60.0, 0.01, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(25, min(400, int(remaining * 15)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and 0 < len(active_set) < size:
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[idx, i] = a[k - i]
        constraints = [T @ b_var <= M]
    else:
        T = np.zeros((size, n))
        for m in range(n):
            T[m:m + n, m] = a
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        frac = 0.55 + 0.25 * np.sin(t * 0.03)
        threshold = M * frac
        active_set = np.where(conv_y > threshold)[0]
        if len(active_set) < 5:
            active_set = np.where(conv_y > M * 0.35)[0]
        if len(active_set) > 400:
            active_set = np.where(conv_y > M * 0.85)[0]
            if len(active_set) < 5:
                active_set = None

        b = lp_best_response(y, M, solver, active_set)
        if b is None and active_set is not None:
            b = lp_best_response(y, M, solver, None)
        if b is None:
            continue

        if active_set is not None and len(active_set) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.002)[0]
            if len(violated) > 0:
                active_set = np.union1d(active_set, violated)
                b2 = lp_best_response(y, M, solver, active_set)
                if b2 is not None:
                    b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = compute_max_conv(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = compute_max_conv(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 25 == 0:
            print(f"    LP-Nest t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            peak_lags = np.where(conv >= cur_max * 0.88)[0]
            for kl in peak_lags[:8]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
            low_vals = np.argsort(a)[:4]
            for lv in low_vals:
                j_candidates.add(int(lv))
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue

                g = np.zeros(size)
                lo = max(0, i)
                hi = min(size, i + n)
                if hi > lo:
                    g[lo:hi] -= a[0:hi - lo]
                lo2 = max(0, i - n + 1)
                hi2 = min(size, i + 1)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[i - np.arange(lo2, hi2)]
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[0:hi3 - lo3]
                lo4 = max(0, j - n + 1)
                hi4 = min(size, j + 1)
                if hi4 > lo4:
                    g[lo4:hi4] += a[j - np.arange(lo4, hi4)]

                d_max = min(ai, norm * 0.25)
                deltas = np.linspace(0.01, d_max, 12)

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] += d * d
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                a = normalize(a, norm)
                new_max = compute_max_conv(a)
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = np.maximum(a, 0.0)
                    a = normalize(a, norm)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            n_perturb = max(5, n // 60)
            idx_p = rng.choice(n, size=n_perturb, replace=False)
            a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
            a = np.maximum(a, 0.0)
            a = normalize(a, norm)
        else:
            no_improve = 0

        if verbose and int((time.monotonic() - t0) // 30) != int((time.monotonic() - t0 - 1) // 30):
            print(f"    PCD ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 14))
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = int(rng.integers(1, 6))
        f2 = int(rng.integers(6, 20))
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-5.0 * x) + 0.4
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = int(rng.integers(3, 10))
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    elif choice == 11:
        # Edge-concentrated (U-shaped)
        a = (x - 0.5) ** 2 + 0.3
    elif choice == 12:
        # Asymmetric multi-scale
        a = 0.4 + 0.3 * np.sin(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x + 1.0)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
