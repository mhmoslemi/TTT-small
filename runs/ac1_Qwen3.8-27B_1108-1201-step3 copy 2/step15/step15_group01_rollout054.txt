def build_slice(a, indices):
    """Build convolution matrix rows for given indices.
    T[idx, j] = a[j] if 0 <= idx-k < len(a) properly... 
    Actually: conv(a,b)[k] = sum_j a[j]*b[k-j]
    So (T @ b)[k] = sum_j T[k,j] * b[j] = sum_j a[j] * b[k-j]
    Hence T[k,j] = a[j] when 0 <= k-j < len(b) = n.
    i.e., k-n+1 <= j <= k, and 0 <= j < n.
    """
    n = len(a)
    T = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            T[idx, j_start:j_end + 1] = a[j_start:j_end + 1]
    return T


def lp_best_response(a, M, solver, constraint_indices, verbose=False):

            # We want c[i] = sum_j Y[j] * W[i+j] for j=0..n-1, i=0..n-1
            # This is: c = cross_correlation(Y, W) at non-negative lags
            # = (flip(W) * Y)[len(W)-1 - (0..n-1)]... 
            # 
            # Easiest: use scipy.signal.correlate or just compute via FFT
            # c[i] = sum_j Y[j] * W[i+j]
            # Let's pad Y to length 2n-1 (zero at the end)
            # Then c[i] = sum_{j=0}^{n-1} Y[j] * W[i+j]
            # This is the correlation of Y (length n) with W (length 2n-1) at lag i
            # = (Y * flip(W))[i + (2n-1) - 1]... I'm overcomplicating this.
            #
            # Let me just use: c[i] = sum_j Y[j] * W[i+j]
            # Pad W to length at least i + n for each i. Since i < n and W has length 2n-1,
            # we need W[i+j] for i+j < 2n-1, which is i < 2n-1-j, always true for i < n, j >= 0.
            #
            # Use FFT: create a signal of length 2n-1 (or next power of 2):
            # Let Z = W (length 2n-1)
            # Let Y_ext = Y padded to length 2n-1
            # Then c[i] = sum_j Y_ext[j] * Z[i+j]
            # This is the "cross-correlation" at lag i, which equals:
            # conv(Y_ext_reversed, Z)[i + n - 1] where Y_ext_reversed = Y[::-1] padded
            #
            # OK I'll just use the formula directly with a small loop for the gradient since
            # n is at most ~2560 and we're doing this ~300 times per temperature.
            # Actually that's O(n^2) per step which is too slow for n=2000.
            #
            # Better approach: use the fact that grad[i] = 2 * sum_j y[j] * w[i+j]
            # Let's define v = w (length 2n-1)
            # grad[i] = 2 * (y circulant-convolved with v)[i] where we only sum over valid j
            #
            # Actually the cleanest way: 
            # grad[i] = 2 * sum_{j=0}^{n-1} y[j] * w[i+j]
            # Let's reverse y: y_rev[j] = y[n-1-j]
            # Then grad[i] = 2 * sum_j y_rev[n-1-j] * w[i+j]
            # = 2 * conv(y_rev, w)[i + n - 1]  (checking: conv(y_rev, w)[k] = sum_j y_rev[j] * w[k-j])
            # conv(y_rev, w)[i + n - 1] = sum_j y_rev[j] * w[i + n - 1 - j]
            # Let j' = n-1-j, so j = n-1-j':
            # = sum_{j'} y_rev[n-1-j'] * w[i + n - 1 - (n-1-j')]
            # = sum_{j'} y[j'] * w[i + j']
            # ✓
            
            y_rev = y[::-1]
            fft_size2 = 1 << (2 * n - 1 - 1).bit_length()
            YR = rfft(y_rev, fft_size2)
            W = rfft(w, fft_size2)
            conv_yr_w = irfft(YR * W, fft_size2)[:2 * n - 1]
            grad = 2.0 * conv_yr_w[n - 1:n - 1 + n]
            
            # Projected gradient step
            step = norm / (n * T * (1.0 + 0.1 * it / n_steps))
            step = min(step, norm / (n * 0.3))
            
            a_new = y - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * norm / s_new
            
            # Nesterov
            alpha_n = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha_n) * y + alpha_n * a_new
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new * norm / s_y
            
            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = np.maximum(a_curr, 0.0)
                s_a = a_curr.sum()
                if s_a > 1e-15:
                    a_curr = a_curr * norm / s_a
            
            y = y_new
            a = a_curr
            
            score = fast_conv_self(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a.copy(), best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=False):
    """Pair coordinate descent with optimal step size search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a.copy(), float('inf')
    a = a * norm / s0
    
    conv = fast_conv_self(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    window = min(60, n // 10)
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue
            
            # Try moving mass from i to nearby positions
            offsets = list(range(1, window + 1))
            rng.shuffle(offsets)
            
            for offset in offsets[:15]:  # Try only 15 random offsets
                for sign in [1, -1]:
                    j = (i + sign * offset) % n
                    if j == i or a[i] < norm * 1e-12:
                        continue
                    
                    # Simple: try moving 10%, 25%, 50% of a[i] to j
                    for frac in [0.1, 0.25, 0.5]:
                        d = a[i] * frac
                        a_test = a.copy()
                        a_test[i] -= d
                        a_test[j] += d
                        s_test = a_test.sum()
                        if s_test < 1e-15:
                            continue
                        a_test = a_test * norm / s_test
                        conv_test = fast_conv_self(a_test)
                        if conv_test.max() < best_score - 1e-12:
                            a = a_test
                            conv = conv_test
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            break
                    else:
                        continue
                    break
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a.copy(), best_score


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving the objective."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def make_initial_sequence(n, rng, method='uniform'):
    """Create an initial sequence using various methods."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'random':
        a = rng.random(n) + 0.1
        return a / a.sum()
    elif method == 'bump':
        x = np.linspace(0, 1, n)
        center = rng.random() * 0.6 + 0.2
        width = rng.random() * 0.15 + 0.05
        a = np.exp(-((x - center) / width) ** 2)
        a += 0.01
        return a / a.sum()
    elif method == 'two_bumps':
        x = np.linspace(0, 1, n)
        c1 = rng.random() * 0.3 + 0.1
        c2 = rng.random() * 0.3 + 0.6
        a = np.exp(-((x - c1) / 0.08) ** 2) + np.exp(-((x - c2) / 0.08) ** 2)
        a += 0.01
        return a / a.sum()
    elif method == 'ramp':
        x = np.linspace(0, 1, n)
        a = x + 0.01
        return a / a.sum()
    elif method == 'symmetric_ramp':
        x = np.linspace(0, 0.5, n)
        a = np.concatenate([x, x[::-1]])
        a += 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n
