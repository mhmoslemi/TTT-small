_global_solver = None

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_softmax_subgradient(a, ac, temperature):
    """Compute full soft-max subgradient using all autocorrelation lags."""
    n = len(a)
    log_ac = np.log(np.maximum(ac, 1e-30) / temperature)
    log_ac -= log_ac.max()
    weights = np.exp(log_ac)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        # Fallback: uniform over top 50
        top_idx = np.argsort(ac)[::-1][:min(50, len(ac))]
        weights = np.zeros_like(ac)
        weights[top_idx] = 1.0
        w_sum = weights.sum()
    weights /= w_sum

    grad = np.zeros(n)
    for k in range(len(ac)):
        w = weights[k]
        if w < 1e-12:
            continue
        k_int = k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w * a[j_idx]
    return grad

def compute_topk_subgradient(a, ac, top_k=60, beta=5.0):
    """Compute subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def nesterov_mirror_descent(a0, time_limit, initial_temp=0.01, final_temp=0.001,
                            c0=3.0, verbose=True):
    """Nesterov accelerated mirror descent with adaptive soft-max subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 3000
    running_avg = a.copy()
    avg_count = 1

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature
        frac = min(1.0, t / 5000.0)
        temp = initial_temp * (1 - frac) + final_temp * frac
        temp = max(temp, 1e-5)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        # Use full softmax for large temp, top-k for small temp
        if temp > 0.005:
            g = compute_softmax_subgradient(y, ac, temp)
        else:
            g = compute_topk_subgradient(y, ac, top_k=80, beta=4.0)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Periodic restarting
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(1000, max_stall // 2)
            # Perturb and restart
            pert = rng.normal(0, 0.005, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(1.0, gamma * 0.5)

        # Running average check (essentially the ergodic iterate)
        running_avg += x
        avg_count += 1
        if avg_count % 500 == 0:
            avg = running_avg / avg_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.5f}, stall={stall_count}")

    return best_a, best_m

def lp_frank_wolfe_focused(a0, time_limit, solver, max_constraints=200, verbose=True):
    """LP Frank-Wolfe with focused constraint set for speed."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Select constraints: all lags within 0.01 of max
        threshold = M_curr * 0.98
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            # Keep the most active ones
            active_lags = np.argsort(ac)[::-1][:max_constraints]

        # Build constraint matrix (only for active lags)
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            n_act = len(active_lags)
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, active={n_act}")

    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with bisection line search."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 5

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        sweep_improved = False

        # Randomize coordinate order each sweep
        rng.shuffle(coord_order)

        for i in coord_order:
            if time.monotonic() - t0 > time_limit:
                break

            # Determine valid range for a[i]
            a_i = a[i]
            if a_i < 1e-15:
                continue

            # Try moving a small fraction from i to another coordinate
            # Use bisection: find best delta
            # Try a few candidate moves
            best_delta = 0.0
            best_trial_m = best_m

            # Candidates: move to positions that are "far" from peak lags
            # Quick heuristic: try moving to random positions
            n_tries = min(8, n)
            candidates = rng.integers(0, n, size=n_tries)

            for dst in candidates:
                if dst == i:
                    continue
                # Try small fractions
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[i] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[dst] += amt
                    if a_trial[i] < -1e-15:

                    a_trial = np.maximum(a_trial, 0.0)

                    if s_t < 1e-15:

                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m - 1e-16:

            if best_delta != 0.0:

                a[i] -= amt

                if s_t > 1e-15:

    # === Phase 3: NMD polish after LP (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: NMD polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent(a_in, t3, initial_temp=0.005, final_temp=0.001,
                                           c0=2.0, verbose=False)
        try_update(a_out, f"NMD_polish_{len(best_seq)}")

    # === Phase 4: Coordinate descent (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_polish(a_in, t4, verbose=True, rng_seed=789)
        try_update(a_out, f"CD_{len(best_seq)}")

    # === Phase 5: Paired swap polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.05, 60.0)
    print(f"\n=== Phase 5: PairSwap n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = paired_swap_polish(a_in, t5, verbose=True, rng_seed=321)
        try_update(a_out, f"PS_{len(best_seq)}")

    # === Phase 6: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_alt = n0 + 400
        print(f"\n=== Phase 6: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_in = norm_sum1(a_long)
        t6 = min(remaining * 0.4, 200.0)
        if t6 > 30:
            a_out, _ = nesterov_mirror_descent(a_in, t6, initial_temp=0.01, final_temp=0.002,
                                               c0=3.0, verbose=True)
            try_update(a_out, f"NMD_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) <= 3000:
            t6b = min(remaining * 0.5, 150.0)
            if t6b > 30:
                a_out, _ = lp_frank_wolfe_focused(a_in, t6b, solver, max_constraints=200, verbose=False)
                try_update(a_out, f"LP_n{n_alt}")

    # === Phase 7: Final polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
        a_in = norm_sum1(best_seq)

        # Quick LP
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 20:
            a_out, _ = lp_frank_wolfe_focused(a_in, t7, solver, max_constraints=150, verbose=False)
            try_update(a_out, f"LP_final_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.6, 60.0)
            if t7b > 10:
                a_out, _ = coordinate_descent_polish(a_in, t7b, verbose=False, rng_seed=555)
                try_update(a_out, f"CD_final_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
