def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size, n=size)


def build_conv_matrix(a):
    """Build the cross-correlation matrix T where (T @ b) = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def lp_best_response(a, M, solver, tol=1e-9):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False, tol=tol)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    # Get dual variables if available
    dual_vars = None
    try:
        if constraints[0].dual_value is not None:
            dual_vars = np.asarray(constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    return b_val, dual_vars

def simplex_project(v, target_sum):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    # Standard simplex projection algorithm
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n+1) > (cv - target_sum))[0][-1]
    tau = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0)

def subgradient_polish(a0, time_limit, top_k=20, verbose=True):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:

    best_a = a.copy()

    n_iters = 0
    patience = 0
    eta = 0.01  # initial step size
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Active set: top-k lags
        k_act = min(top_k, size)
        active = np.argsort(conv)[-k_act:]
        
        # Subgradient: for each active lag k, the gradient w.r.t. a_i is 2*a[(k-i) % ... ]
        # Actually: d/dc_i [(c*c)[k]] = 2*c[k-i] if 0 <= k-i < n
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_hi >= i_lo:
                # grad[i] += 2 * a[k - i] for i in [i_lo, i_hi]
                # k - i ranges from k - i_hi to k - i_lo
                # = k - min(k, n-1) to k - max(0, k-n+1)
                # For i in [i_lo, i_hi]: k-i in [k-i_hi, k-i_lo]
                # k-i_hi = k - min(k, n-1) >= 0
                # k-i_lo = k - max(0, k-n+1) <= n-1
                indices = np.arange(i_lo, i_hi + 1)
                grad[indices] += 2.0 * a[k - indices]
        
        grad /= k_act  # average over active set
        
        # Project gradient onto the tangent space of the simplex (sum = 0 direction)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        
        # Try several step sizes
        best_new_a = None
        best_new_M = M
        for factor in [0.1, 0.3, 0.5, 1.0, 2.0]:
            eta_try = eta * factor
            step = eta_try * grad_centered / grad_norm
            a_try = a - step
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            new_M = fast_autocorr(a_try).max()
            if new_M < best_new_M - 1e-14:
                best_new_M = new_M
                best_new_a = a_try
                eta = eta_try  # remember good step size
        
        if best_new_a is not None:
            a = best_new_a
            n_iters += 1
            if best_new_M < best_score - 1e-13:
                best_score = best_new_M
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            eta *= 0.5  # reduce step size
        
        if patience > 100:
            # Reset
            a = best_a.copy()
            eta = 0.01 / np.sqrt(n)
            patience = 0
            # Add small noise
            rng = np.random.default_rng(42 + n_iters)
            noise = rng.normal(0, norm * 0.001, n)
            a = np.maximum(a + noise, 0)
            a = a * norm / a.sum()
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score

def fast_autocorr(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size, n=size)


def build_sparse_T(a):
    """Build sparse cross-correlation matrix T where (T @ b)[k] = sum_j a[j]*b[k-j]."""
    n = len(a)

    col_idx = []
    row_idx = []
    data = []
    for j in range(n):
        lo = j
        hi = j + n
        col_idx.extend([j] * (hi - lo))
        row_idx.extend(range(lo, hi))
        data.extend([a[j]] * (hi - lo))
    return csr_matrix((data, (row_idx, col_idx)), shape=(size, n))
