"""
Minimizes 2*n*max_b/(sum_a)^2 using sparse Newton method exploiting 
the banded Hessian structure of autocorrelation peaks, combined with 
multiplicative updates and coordinate descent for polishing.
"""

import numpy as np
import time
import scipy.sparse as sp
import scipy.sparse.linalg as spl

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_hessian_sparse(n, k, lam):
    """Build H + lam*I for peak lag k, as a sparse matrix."""
    rows = []
    cols = []
    vals = []
    
    # Diagonal: lam
    for i in range(n):
        rows.append(i)
        cols.append(i)
        vals.append(lam)
    
    # Off-diagonal for peak k
    if k == 0:
        # (a*a)_0 = sum a_i^2, Hessian is 2*I
        for i in range(n):
            rows.append(i)
            cols.append(i)
            vals.append(2.0)
    else:
        for i in range(n - k):
            rows.append(i)
            cols.append(i + k)
            vals.append(1.0)
            rows.append(i + k)
            cols.append(i)
            vals.append(1.0)
    
    H = sp.coo_matrix((vals, (rows, cols)), shape=(n, n)).tocsr()
    return H

def newton_step(a, k, lam=1.0):
    """Compute Newton direction for peak lag k."""
    n = len(a)
    
    # Gradient for peak k
    g = np.zeros(n)
    if k == 0:
        g = 2.0 * a
    else:
        g[:n-k] += a[k:]
        g[k:] += a[:n-k]
    
    # Build sparse Hessian
    H = build_hessian_sparse(n, k, lam)
    
    # Solve (H) d = -g
    try:
        d = spl.spsolve(H, -g)
    except Exception:
        # Fallback to gradient descent direction
        d = -g
        d = d - d.mean()
        return d

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    if s < 1e-15:

                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    M_t = fast_autocorr(a_trial).max()
                    if M_t < best_M:
                        best_M = M_t
                        best_dst = dst
                        best_frac = frac
                        best_src = src
            
            if best_dst >= 0 and best_M < M_curr:
                a[best_src] -= a[best_src] * best_frac
                a[best_dst] += a[best_src + a[best_src] * best_frac - a[best_src] * best_frac]  # this is wrong
                # Let me redo
                amt = a[best_src] * best_frac
                a[best_src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                improved = True
                stall = 0
                break
        
        # Strategy 2: Split-and-merge (move mass from one position to two cold positions)
        if not improved:
            k_main = int(top_k[0])
            if k_main > 0:
                pairs = a[:n - k_main] * a[k_main:]
                src = int(np.argmax(pairs))
                cold_idx = np.argsort(a)[:min(50, n)]
                for dst1 in cold_idx[:10]:
                    for dst2 in cold_idx[:10]:
                        if dst1 == src or dst2 == src:
                            continue
                        if dst1 == dst2:
                            continue
                        for frac in [0.005, 0.01, 0.02]:
                            amt = a[src] * frac
                            half = amt / 2
                            a_trial = a.copy()
                            a_trial[src] -= amt
                            a_trial[dst1] += half
                            a_trial[dst2] += half
                            a_trial = np.maximum(a_trial, 0.0)
                            s_t = a_trial.sum()
                            if s_t < 1e-15:
                                continue
                            a_trial = a_trial / s_t
                            M_t = fast_autocorr(a_trial).max()
                            if M_t < best_m:
                                a = a_trial
                                best_m = M_t
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            i3 = int(rng.integers(n))
            if a[i1] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i1] * frac
                a[i1] -= amt
                a[i2] += amt / 2
                a[i3] += amt / 2
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new > best_m * 1.01:
                    a = best_a.copy()
        
        if verbose and (stall == 0 or stall % 50 == 0):
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.1f}s): score={2*n*best_m:.6f}, stall={stall}")
    
    return best_a, best_m

def spectral_repair(a0, time_limit, verbose=False):
    """Try to flatten |DFT(a)|^2 to reduce autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        
        # Identify "hot" frequency bins
        hot_bins = np.where(P > 1.5 * P_mean)[0]
        if len(hot_bins) == 0:
            break
        
        # Compute "pressure" in time domain from hot bins
        # The contribution of bin k to a_i is |A_k| * cos(2*pi*k*i/n + phase_k)
        # We want to reduce the amplitude at hot bins
        
        # Simple approach: compute the "hot component" of a
        hot_component = np.zeros(n)
        for k in hot_bins[:min(50, len(hot_bins))]:
            if k >= len(A):
                continue
            phase = np.angle(A[k])
            if k == 0 or (n % 2 == 0 and k == n // 2):
                hot_component += np.real(A[k]) * np.ones(n) / n
            else:
                hot_component += 2.0 * np.real(A[k] * np.exp(-2j * np.pi * k * np.arange(n) / n)) / n
        
        # Push mass away from high hot_component regions
        hc_norm = hot_component / (hot_component.max() - hot_component.min() + 1e-15)
        hc_norm = (hc_norm - hc_norm.min()) / (hc_norm.max() - hc_norm.min() + 1e-15)
        
        # Multiplicative update: reduce where hot_component is high
        alpha = 0.01
        a_new = a * np.exp(-alpha * hc_norm)
        a_new = np.maximum(a_new, 1e-20)
        a_new = a_new / a_new.sum()
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            a_new2 = a * np.exp(-alpha * 0.3 * hc_norm)
            a_new2 = np.maximum(a_new2, 1e-20)
            a_new2 = a_new2 / a_new2.sum()
            M_new2 = fast_autocorr(a_new2).max()
            if M_new2 < best_m:
                best_m = M_new2
                best_a = a_new2.copy()
                a = a_new2
            else:
                break
        
        alpha *= 0.99
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Multiplicative updates (exact subgradient) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MU exact n={n0} ({remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multiplicative_optimize(a_in, t1, top_k=50, beta=8.0,
                                           use_exact=True, temp_init=0.5,
                                           temp_decay=0.9995, verbose=True)
        try_update(a_out, f"mu_exact_n{n0}")
    
    # === Phase 2: Multiplicative updates (smooth subgradient, different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MU smooth n={len(best_seq)} ({remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multiplicative_optimize(a_in, t2, top_k=60, beta=5.0,
                                           use_exact=False, temp_init=0.3,
                                           temp_decay=0.999, verbose=True)
        try_update(a_out, f"mu_smooth_n{len(best_seq)}")
    
    # === Phase 3: SciPy L-BFGS-B on smoothed objective ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: scipy LSE n={n_best} ({remaining:.0f}s) ===")
    if n_best <= 2500 and remaining > 60:
        t3 = min(remaining * 0.20, 300.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = scipy_lse_optimize(a_in, t3, beta_lse=100.0, max_iter=3000, verbose=True)
        try_update(a_out, f"scipy_lse_n{n_best}")
    
    # === Phase 4: Newton local refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Newton refine n={n_best} ({remaining:.0f}s) ===")
    if n_best <= 3000 and remaining > 30:
        t4 = min(remaining * 0.15, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = newton_local_refine(a_in, t4, verbose=True)
        try_update(a_out, f"newton_n{n_best}")
    
    # === Phase 5: Coordinate descent multi-step ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: CD multi n={n_best} ({remaining:.0f}s) ===")
    if remaining > 30:
        t5 = min(remaining * 0.15, 250.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_multi(a_in, t5, verbose=True, rng_seed=123)
        try_update(a_out, f"cd_n{n_best}")
    
    # === Phase 6: Spectral repair ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Spectral n={n_best} ({remaining:.0f}s) ===")
    if remaining > 30:
        t6 = min(remaining * 0.10, 150.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_repair(a_in, t6, verbose=True)
        try_update(a_out, f"spectral_n{n_best}")
    
    # === Phase 7: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 7: Length exploration ({remaining:.0f}s) ===")
        for n_try in [n_best + 200, n_best - 400, n_best + 400, n_best - 800]:
            if remaining < 40:
                break
            if n_try < 500 or n_try > 3000:
                continue
            a_long = interpolate_sequence(best_seq, n_try)
            t7 = min(remaining * 0.15, 120.0)
            if t7 > 20:
                a_out, _ = multiplicative_optimize(a_long, t7, top_k=40, beta=7.0,
                                                   use_exact=True, temp_init=0.3,
                                                   temp_decay=0.999, verbose=False)
                try_update(a_out, f"mu_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final CD n={n_best} ({remaining:.0f}s) ===")
    if remaining > 20:
        t8 = min(remaining * 0.4, 150.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_multi(a_in, t8, verbose=True, rng_seed=777)
        try_update(a_out, f"cd_final_n{n_best}")
    
    # === Phase 9: One more MU round with tighter params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final MU n={n_best} ({remaining:.0f}s) ===")
        t9 = min(remaining * 0.5, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = multiplicative_optimize(a_in, t9, top_k=80, beta=4.0,
                                           use_exact=True, temp_init=0.1,
                                           temp_decay=0.9995, verbose=True)
        try_update(a_out, f"mu_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
