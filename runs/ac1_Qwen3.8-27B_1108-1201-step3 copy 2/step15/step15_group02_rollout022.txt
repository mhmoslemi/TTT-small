def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):

def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 25
    temps = np.geomspace(100.0, 0.02, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(300, int(remaining * 12)))
        lr = 0.4 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / (exp_c.sum() + 1e-30)

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]

            grad = 2.0 * (ca + aw)
            grad_mean = grad.mean()
            grad = grad - grad_mean

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score

def build_active_scores(a, top_k, frac=0.85):
    conv = fast_conv_self(a)
    M = conv.max()
    threshold = max(M * frac, M - 0.01 * M)
    indices = np.where(conv >= threshold)[0]
    if len(indices) > top_k:
        order = np.argsort(conv[indices])[::-1][:top_k]
        indices = indices[order]
    return indices, M

def lp_best_response(a, M, active_set, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0:
        num_constraints = len(active_set)
        A_mat = np.zeros((num_constraints, n))
        for idx, k in enumerate(active_set):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    A_mat[idx, j] = a[i]
        constraints = [A_mat @ b_var <= M]
    else:
        size = 2 * n - 1
        A_mat = np.zeros((size, n))
        for k in range(size):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    A_mat[k, j] = a[i]
        constraints = [A_mat @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    max_iterations = 20000
    top_k = min(200, size // 4)
    lp_interval = max(1, n // 400)

    for t in range(1, max_iterations + 1):
        if time.monotonic() - t0 > time_limit:
            break

        if t % lp_interval == 0 or t == 1:
            conv_y = fast_conv_self(y)
            M = conv_y.max()
            if M < 1e-12:
                break

            active_set, _ = build_active_scores(y, top_k, frac=0.7)
            b = lp_best_response(y, M, active_set, solver)

            if b is None:
                b = lp_best_response(y, M, None, solver)

            if b is None:
                continue

            if active_set is not None and len(active_set) < size:
                check = fast_conv_cross(y, b)
                violated = np.where(check > M * 1.0005)[0]
                if len(violated) > 0:
                    active_set = np.union1d(active_set, violated)
                    b2 = lp_best_response(y, M, active_set, solver)
                    if b2 is not None:
                        b = b2

            b_sum = b.sum()
            if b_sum < 1e-15:
                continue
            g = b * norm / b_sum
        else:
            g = y.copy()

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        if t % 10 == 0:
            sc_y = fast_conv_self(y).max()
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()

            sc_a = fast_conv_self(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()

            if verbose and t % (lp_interval * 10) == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            peak_lags = np.where(conv >= cur_max * 0.8)[0]

            j_candidates = set()
            for _ in range(10):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:15]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
                j_candidates.add(int((kl - 2 * i + n) % n))
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)

                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]

                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]

                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]

                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                d_max = ai * 0.9
                n_samples = 30
                deltas = np.linspace(0.005, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                n_perturb = max(3, n // 60)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                scale = 1.0 + 0.2 * rng.standard_normal(n_perturb)
                a[idx_p] *= scale
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 8:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements % 300 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def spectral_refine(a0, time_limit, rng, verbose=True):
    """Refine in frequency domain to reduce convolution peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2 - 1).bit_length()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    for it in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        power = A.real ** 2 + A.imag ** 2
        power_max_idx = np.argmax(power[:fft_size // 2 + 1])

        # Reduce peak in frequency domain
        reduction = 0.01
        for freq in range(max(0, power_max_idx - 5), min(fft_size // 2 + 1, power_max_idx + 6)):
            if freq == power_max_idx:
                power[freq] *= (1 - reduction)
            else:
                power[freq] *= (1 - reduction * 0.5)

        # Reconstruct: we want a new a' >= 0 with DFT magnitude ~ sqrt(power)
        # Use iterative projection: alternate between magnitude constraint and non-negativity
        for inner in range(50):
            # Get current phase from a
            A_new = rfft(a, fft_size)
            # Set magnitude to sqrt(power)
            mag = np.sqrt(power)
            phase = np.angle(A_new)
            A_target = mag * np.exp(1j * phase)
            # Inverse FFT
            a_new = irfft(A_target, fft_size)[:n]
            # Project to non-negative
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm / s_new
            a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 50 == 0:
            print(f"    Spectral it={it}: best={best_score:.8f}")

    return best_a, best_score

        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    elif choice == 11:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 12:
        # U-shaped: high at edges
        a = 0.3 + 0.7 * (4 * x * (1 - x))
        a = 1.0 / (a + 0.1)
    else:
        # Multi-scale oscillation
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 11 * x) + 0.15 * np.sin(2 * np.pi * 23 * x) + 0.1 * np.sin(2 * np.pi * 41 * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Quick LSE-Adam to make initial progress
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{n_base}")

    # Phase 2: Vegas multiplicative updates
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Vegas n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = vegas_optimize(best_seq, t2, verbose=True)
        check_and_update(a_out, f"vegas_n{len(best_seq)}")

    # Phase 3: LP Nesterov (main workhorse - allocate most time here)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t3, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(best_seq)}")

    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Pair CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(best_seq)}")

    # Phase 5: LP round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 5: LP round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t5, solver, verbose=True)
        check_and_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 6: Spectral refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 6: Spectral n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = spectral_refine(best_seq, t6, rng, verbose=True)
        check_and_update(a_out, f"spectral_n{len(best_seq)}")

    # Phase 7: Final CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 7: Final PCD n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t7, rng, verbose=True)
        check_and_update(a_out, f"pcd_final_n{len(best_seq)}")

    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 8: Multi-length exploration (remaining={remaining:.0f}s) ===")
        lengths_to_try = [n_base + 300, n_base + 600, n_base - 200, n_base + 900]
        tried = set()
        for n_try in lengths_to_try:
            if remaining < 45:
                break
            if n_try < 400 or n_try > 4000 or n_try in tried:
                continue
            tried.add(n_try)
            remaining = time_budget - (time.monotonic() - t_global)
            t_try = min(remaining * 0.30, 80.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            if t_try < 15:
                continue
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try * 0.4, 30.0), verbose=False)
            check_and_update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out2, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 30.0), rng, verbose=False)
                check_and_update(a_out2, f"ml_pcd_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)

    # Phase 9: Symmetrization and final checks
    a_sym = symmetrize(best_seq)
    check_and_update(a_sym, f"sym_n{len(best_seq)}")

    # Try a slightly different LSE temperature schedule
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t9 = min(remaining * 0.5, 60.0)
        print(f"\n=== Phase 9: Final LSE (budget={t9:.0f}s) ===")
        a_out, _ = lse_adam_optimize(best_seq, t9, verbose=False)
        check_and_update(a_out, f"lse_final_n{len(best_seq)}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out2, _ = pair_cd_optimize(a_out, min(remaining * 0.7, 30.0), rng, verbose=False)
            check_and_update(a_out2, f"pcd_last_n{len(best_seq)}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
