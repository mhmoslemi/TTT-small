def cross_corr_fast(p, a):
    """Compute c[m] = sum_j p[m+j] * a[j] for m=0..n-1 using FFT.
    p has length 2n-1, a has length n."""
    n = len(a)
    n_p = len(p)
    size = n_p + n - 1
    fft_size = 1 << (size - 1).bit_length()
    b = p[::-1]
    conv_full = irfft(rfft(b, fft_size) * rfft(a, fft_size), fft_size)[:size]
    return conv_full[n_p - 1: n_p - 1 + n]


def smooth_objective(a, T):
    """Compute T * log(sum exp(conv/T)) and its gradient."""
    conv = fast_autocorr(a)
    c_max = conv.max()
    log_p = (conv - c_max) / T
    exp_p = np.exp(log_p)
    log_sum = c_max / T + np.log(exp_p.sum())
    F = T * log_sum
    p = exp_p / exp_p.sum()
    g = 2.0 * cross_corr_fast(p, a)
    return F, g


def mirror_descent_step(a, g, eta):
    """Perform one mirror descent step: a_new = a * exp(-eta*g) / Z."""
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * g
    log_Z = log_a_new.max() + np.log(np.exp(log_a_new - log_a_new.max()).sum())
    return np.exp(log_a_new - log_Z)


def backtracking_mirror(a, T, max_iter=100):
    """Mirror descent with backtracking line search on smooth objective."""
    F_curr, g = smooth_objective(a, T)
    log_a = np.log(np.maximum(a, 1e-300))
    eta = 5.0
    for _ in range(max_iter):

        F_new, _ = smooth_objective(a_new, T)
        if F_new <= F_curr - 1e-4 * eta * np.dot(g, a):
            return a_new, eta, F_new
        eta *= 0.5
        if eta < 1e-12:

    return a, eta, F_curr

def lp_best_response(a, M, solver, active_only=None):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1

    if active_only is not None:
        rows = active_only
    else:
        rows = range(size)

    # Build constraint matrix: (T @ b)[j] = (a*b)[j]
    # T[j, i] = a[j-i] if 0 <= j-i < n, else 0
    T = np.zeros((len(rows), n))
    for idx, j in enumerate(rows):
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        if lo <= hi:
            T[idx, lo:hi+1] = a[j-hi:j-lo+1]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_nesterov_cycles(a_init, time_limit, solver, n_cycles=5, verbose=True):
    """LP best-response with Nesterov, run in cycles with restarts."""
    n = len(a_init)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a * norm / a.sum()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()

    for cycle in range(n_cycles):
        if time.monotonic() - t0 > time_limit:
            break
        cycle_budget = (time_limit - (time.monotonic() - t0)) / (n_cycles - cycle)
        cycle_budget = min(cycle_budget, 300.0)

        y = best_a.copy()
        t_cycle = time.monotonic()
        n_iters = 0

        for t in range(1, 100000):
            if time.monotonic() - t_cycle > cycle_budget:
                break

            conv_y = fast_autocorr(y)
            M_curr = conv_y.max()
            if M_curr < 1e-12:
                break

            # Constraint generation
            threshold = 0.8 * M_curr
            active = np.where(conv_y >= threshold)[0]
            if len(active) < 5:
                active = np.where(conv_y >= 0.5 * M_curr)[0]
            if len(active) < 3:
                active = np.arange(len(conv_y))

            b = lp_best_response(y, M_curr, solver, active_only=active)
            if b is None:
                b = lp_best_response(y, M_curr, solver, active_only=None)
                if b is None:
                    break

            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            g = b * norm / b_sum
            n_iters += 1

            # Nesterov update
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            if t == 1:
                a_new = g.copy()
            else:
                momentum = (t - 1.0) / (t + 2.0)
                a_new = y_new + momentum * (y_new - y)
                a_new = np.maximum(a_new, 0.0)
                s_a = a_new.sum()
                if s_a < 1e-15:
                    a_new = y_new.copy()
                else:
                    a_new = a_new * norm / s_a

            y = y_new
            a = a_new

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LP cycle {cycle} ({elapsed:.0f}s, {n_iters} iters): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a_init, time_limit, rng, verbose=True):
    """Coordinate descent: for each coordinate, find optimal value via 1D optimization."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-15:
                continue

            # Optimize a[i] with others fixed
            # conv(a,a)[k] as function of a[i]:
            # For k = i: a[i]^2 appears (from a[i]*a[i] when i+j=k, i.e., j=i, k=2i)
            # Actually let's just do a 1D line search
            total_others = a.sum() - a[i]
            if total_others < 1e-15:
                continue

            def obj(val):
                if val < 0:
                    return np.inf
                a_new = a.copy()
                a_new[i] = val
                s = a_new.sum()
                if s < 1e-15:
                    return np.inf
                a_norm = a_new / s
                conv = fast_autocorr(a_norm)
                return conv.max()

            # Try a few candidate values
            best_val = a[i]
            best_OBJ = obj(best_val)

            # Try decreasing and increasing
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                for direction in [-1, 1]:
                    candidate = a[i] * (1 + direction * frac)
                    if candidate < 0:
                        continue
                    val_OBJ = obj(candidate)
                    if val_OBJ < best_OBJ - 1e-12:
                        best_OBJ = val_OBJ
                        best_val = candidate

            if best_val != a[i]:
                a[i] = best_val
                a = a / a.sum()
                score = fast_autocorr(a).max()
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd(a_init, time_limit, rng, window=50, verbose=True):

        for idx in perm:
            if time.monotonic() - t0 > time_limit:

            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def make_parametric_init(n, shape_type, rng):
    """Create parametric initializations of various shapes."""
    x = np.linspace(0, 1, n, endpoint=False)

    if shape_type == 0:
        # Uniform
        a = np.ones(n)
    elif shape_type == 1:
        # Cosine bump (high in middle)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif shape_type == 2:
        # U-shape (high at ends)
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif shape_type == 3:
        # Gaussian
        sigma = rng.uniform(0.1, 0.3)
        center = rng.uniform(0.3, 0.7)
        a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    elif shape_type == 4:
        # Double peak
        sigma = rng.uniform(0.05, 0.15)
        c1 = rng.uniform(0.2, 0.4)
        c2 = 1.0 - c1
        a = np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2))
    elif shape_type == 5:
        # Power law decay from center
        alpha = rng.uniform(0.5, 2.0)
        d = np.abs(x - 0.5) * 2
        a = (1 - d) ** alpha
        a = np.maximum(a, 0)
    elif shape_type == 6:
        # Comb / periodic
        freq = rng.integers(2, 8)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    elif shape_type == 7:
        # Ramp
        direction = rng.integers(0, 2)
        if direction == 0:
            a = x
        else:
            a = 1 - x
        a = a + 0.1 * np.ones(n)  # avoid zeros
    elif shape_type == 8:
        # Triangular (high at center, zero at edges)
        a = 1 - np.abs(2 * x - 1)
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(5):
            center = rng.uniform(0, 1)
            width = rng.uniform(0.05, 0.3)
            amp = rng.uniform(0.5, 2.0)
            a += amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_n = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
            parent_n = len(best_seq)
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Try parametric initializations (30s) ===
    print("\n=== Phase 0: Parametric initialization ===")
    t0 = time.monotonic()
    n_base = parent_n if best_seq is not None else 1600

    for st in range(9):
        if time.monotonic() - t0 > 30:
            break
        a_init = make_parametric_init(n_base, st, rng)
        sc = evaluate_safe(a_init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_init.copy()
            print(f"  Shape {st}: score={sc:.6f}")

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: score={best_score:.6f}")

    # === Phase 1: Smooth mirror descent (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t1 = remaining * 0.40
    n_use = len(best_seq)
    print(f"\n=== Phase 1: Mirror descent n={n_use} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_use) / best_seq.sum()
        a_out, _ = smooth_optimize_mirror(a_norm, t1, verbose=True)
        do_try_update(a_out, f"mirror_n{n_use}")

    # === Phase 2: LP Nesterov (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t2 = remaining * 0.25
    n_use = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_use} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_use) / best_seq.sum()
        a_out, _ = lp_nesterov_cycles(a_norm, t2, solver, n_cycles=4, verbose=True)
        do_try_update(a_out, f"lp_n{n_use}")

    # === Phase 3: Mirror descent round 2 (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t3 = remaining * 0.15
    n_use = len(best_seq)
    print(f"\n=== Phase 3: Mirror descent R2 n={n_use} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_use) / best_seq.sum()
        a_out, _ = smooth_optimize_mirror(a_norm, t3, verbose=True)
        do_try_update(a_out, f"mirror2_n{n_use}")

    # === Phase 4: Pair CD (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t4 = remaining * 0.10
    n_use = len(best_seq)
    print(f"\n=== Phase 4: Pair CD n={n_use} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_use) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t4, rng, window=60, verbose=True)
        do_try_update(a_out, f"paircd_n{n_use}")

    # === Phase 5: Coordinate descent (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t5 = remaining * 0.05
    n_use = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_use} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = coordinate_descent(best_seq, t5, rng, verbose=True)
        do_try_update(a_out, f"cd_n{n_use}")

    # === Phase 6: Diversified restart (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t6 = remaining * 0.05
    n_use = len(best_seq)
    print(f"\n=== Phase 6: Diversified restart n={n_use} (budget={t6:.0f}s) ===")
    if t6 > 30:
        a_pert = best_seq.copy().astype(np.float64)
        noise = rng.normal(0, 0.05, n_use)
        a_pert = a_pert + noise * a_pert.mean()
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        t6a = t6 * 0.6
        a_norm = a_pert * np.sqrt(2.0 * n_use)
        a_out, _ = smooth_optimize_mirror(a_norm, t6a, verbose=False)
        do_try_update(a_out, f"restart_n{n_use}")
        remaining6 = time_budget - (time.monotonic() - t_global_start)
        if remaining6 > 15:
            a_norm2 = best_seq * np.sqrt(2.0 * n_use) / best_seq.sum()
            a_out2, _ = pair_cd(a_norm2, min(remaining6, 60.0), rng, window=80, verbose=False)
            do_try_update(a_out2, f"restart_pcd_n{n_use}")

    # === Phase 7: Final polish with LP (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        n_use = len(best_seq)
        print(f"\n=== Phase 7: Final LP polish n={n_use} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_use) / best_seq.sum()
        a_out, _ = lp_nesterov_cycles(a_norm, remaining * 0.7, solver, n_cycles=3, verbose=True)
        do_try_update(a_out, f"final_lp_n{n_use}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
