def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_cols(a, indices):
    """Build only the rows of T specified by indices."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(indices), n))
    for row_idx, m in enumerate(indices):
        for col in range(n):
            lo = max(0, m - n + 1)
            hi = min(m + 1, n)
            if lo <= col < hi:
                T[row_idx, col] = a[col]
    return T


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        T = build_T_cols(a, active)
        constraints = [T @ b_var <= M]
    else:
        T = build_T(a)
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            raw_duals = np.asarray(con.dual_value, dtype=np.float64)
            if active is not None and 0 < len(active) < size:
                full_duals = np.zeros(size)
                full_duals[active] = raw_duals
                duals = full_duals
            else:
                duals = raw_duals
    except Exception:
        pass
    return val, duals

        D = rfft(duals, fft_size)
        g1 = irfft(np.conj(A) * D, fft_size)[:n]
        g2 = irfft(A * D, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = 0.5
        found_step = False
        for trial in range(25):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < M - 1e-10:
                a = a_new
                found_step = True
                break
            step *= 0.5

        if found_step:
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, verbose=True):
    """Log-domain mirror descent using KL divergence as Bregman distance."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10).copy()
    s = a.sum()
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Work in log domain
    log_a = np.log(a)
    prev_score = best_score

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        a = np.exp(log_a)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        a = a * norm / s

        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break

        # Subgradient of max(conv) w.r.t. a
        # The max is achieved at some lag k*, and the gradient is:
        # grad_i = sum over active k of (a[k*-i] + a[i-k*]) (with appropriate bounds)
        # For simplicity, use LSE approximation
        T_val = max(0.1, M * 0.05)
        shifted = (conv - M) / T_val
        shifted = np.maximum(shifted, -200.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2) / (norm * norm)  # normalize gradient for log domain

        # Mirror descent step in log domain
        eta = 1.0 / np.sqrt(t + 1.0)
        log_a_new = log_a - eta * grad
        log_a_new = np.minimum(log_a_new, np.log(1000.0))
        log_a_new = np.maximum(log_a_new, np.log(1e-10))
        log_a = log_a_new

        sc = fast_conv_self(np.exp(log_a)).max()
        if sc < best_score:
            best_score = sc
            best_a = np.exp(log_a).copy()
            best_a = best_a / best_a.sum() * norm

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imps = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        peak_threshold = cur_max * 0.88
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:10]:
                # The convolution at lag kl involves pairs (i, kl+i) and (kl-i, i)
                j1 = kl + i
                j2 = kl - i
                if 0 <= j1 < n and j1 != i:
                    j_candidates.add(j1)
                if 0 <= j2 < n and j2 != i:
                    j_candidates.add(j2)
                # Also center-lag pairs
                c1 = (size - 1) // 2 - i
                c2 = (size - 1) // 2 + i
                if 0 <= c1 < n and c1 != i:
                    j_candidates.add(c1)
                if 0 <= c2 < n and c2 != i:
                    j_candidates.add(c2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:15]:
                if j == i:
                    continue
                aj = a[j]
                if aj < 0:
                    continue

                # Compute the change vector
                g = np.zeros(size)
                # Change at position i: a[i] -> a[i] - d
                # Change at position j: a[j] -> a[j] + d
                # Effect on conv[k]:
                # For the removed part (from i):
                lo = max(0, i)
                hi = min(size, i + n)
                if hi > lo:
                    g[lo:hi] -= a[0:hi - lo]
                lo2 = max(0, i - n + 1)
                hi2 = min(size, i + 1)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[i - np.arange(lo2, hi2)]
                # For the added part (to j):
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[0:hi3 - lo3]
                lo4 = max(0, j - n + 1)
                hi4 = min(size, j + 1)
                if hi4 > lo4:
                    g[lo4:hi4] += a[j - np.arange(lo4, hi4)]

                # Binary search for optimal delta
                lo_d = 0.0
                hi_d = 0.98 * ai
                best_d = 0.0
                best_v = cur_max
                for _ in range(12):
                    mid_d = (lo_d + hi_d) / 2.0
                    new_conv = conv + mid_d * g
                    # Quadratic corrections at self-lags
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * mid_d + mid_d * mid_d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * mid_d + mid_d * mid_d
                    if i != j and (i - j) >= 0 and (i - j) < size:
                        new_conv[i - j] = conv[i - j] - ai * mid_d + mid_d * aj + mid_d * mid_d
                    if j - i >= 0 and (j - i) < size:
                        new_conv[j - i] = conv[j - i] - ai * mid_d + mid_d * aj + mid_d * mid_d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = mid_d
                    if v < mid_v if False else (v <= cur_max * 0.999):
                        lo_d = mid_d
                    else:
                        hi_d = mid_d

                if best_d > 1e-6 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 1e-6:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_imps += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(15, n // 25)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_imps % 500 == 0 and total_imps > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imps={total_imps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Symmetric bump with ripple
        a = np.exp(-8.0 * (x - 0.5) ** 2) * (1.0 + 0.2 * np.cos(30 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 7:
        # Multi-scale
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x) + 0.1 * np.cos(23 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 8:
        # W-shaped
        a = 0.3 + 0.5 * np.abs(1.0 - 2.0 * x) + 0.2 * np.sin(20 * np.pi * x)
        a = np.maximum(a, 0.02)
    elif choice == 9:
        # Random smooth
        freqs = rng.integers(1, 20, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        a = 1.0
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        # Power law decay from center
        a = 1.0 / (1.0 + 10.0 * np.abs(x - 0.5))
    else:
        # Gaussian mixture
        centers = rng.uniform(0.1, 0.9, size=3)
        widths = rng.uniform(0.05, 0.2, size=3)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-0.5 * ((x - c) / w) ** 2)
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick LSE warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t_init = min(remaining * 0.06, 80.0)
    if t_init > 15:
        print(f"=== Phase 0: LSE warmup ({t_init:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, t_init, verbose=False)
        try_update(a_out, "lse_warmup")

    # Phase 1: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True, use_adaptive=True)
        try_update(a_out, f"lp1_n{n1}")

    # Phase 2: LSE refinement with lower temperatures
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: LSE refine n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t2, verbose=True)
        try_update(a_out, "lse_refine")

    # Phase 3: LP Nesterov round 2 (different parameters)
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.22, 350.0)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True, use_adaptive=False)
        try_update(a_out, f"lp2_n{n3}")

    # Phase 4: Mirror descent exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: Mirror Descent n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = mirror_descent_optimize(best_seq, t4, verbose=True)
        try_update(a_out, f"mirror_n{n4}")

    # Phase 5: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 5: Pair CD n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"pcd_n{n5}")

    # Phase 6: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_long = max(n_base + 400, 2000)
        print(f"\n=== Phase 6: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.4, 180.0)
        a_out, _ = nesterov_lp_optimize(init_long, t_long, solver, verbose=False, use_adaptive=True)
        try_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_long:
            t_cd = min(remaining * 0.5, 80.0)
            a_out2, _ = pair_cd_optimize(best_seq, t_cd, rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 7: Final alternating polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 50:
        # LSE
        t_f1 = min(remaining * 0.35, 100.0)
        if t_f1 > 15:
            a_out, _ = lse_gradient_descent(best_seq, t_f1, verbose=False)
            try_update(a_out, "lse_final1")
        remaining = time_budget - (time.monotonic() - t_global)
        # LP
        if remaining > 40:
            t_f2 = min(remaining * 0.35, 120.0)
            if t_f2 > 20:
                a_out, _ = nesterov_lp_optimize(best_seq, t_f2, solver, verbose=False, use_adaptive=True)
                try_update(a_out, "lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        # CD
        if remaining > 30:
            t_f3 = min(remaining * 0.6, 80.0)
            if t_f3 > 15:
                a_out, _ = pair_cd_optimize(best_seq, t_f3, rng, verbose=False)
                try_update(a_out, "pcd_final")

    # Phase 8: Dual gradient last chance
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t_dg = min(remaining * 0.7, 50.0)
        print(f"\n=== Phase 8: Dual gradient (t={t_dg:.0f}s) ===")
        a_out, _ = dual_gradient_refine(best_seq, t_dg, solver, verbose=False)
        try_update(a_out, "dualgrad_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
