```python
import numpy as np
from scipy.optimize import minimize as scipy_minimize
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_max_and_grad(ac, beta):
    """Compute smooth max and its gradient w.r.t. ac."""
    max_ac = ac.max()
    shifted = beta * (ac - max_ac)
    # Handle overflow
    exp_shifted = np.exp(shifted - shifted.max())
    p = exp_shifted / exp_shifted.sum()
    smooth_max = max_ac + np.log(exp_shifted.sum()) / beta
    return smooth_max, p

def compute_smooth_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    ac = fast_autocorr(a)
    smooth_max, p = smooth_max_and_grad(ac, beta)
    
    # Compute (p ⊛ a)[m] = sum_k p_k * a_{k-m} * I(0<=k-m<n)
    # This is the cross-correlation of p and a
    # (p ⊛ a)[m] = sum_k p[k] * a[k-m] where 0<=k-m<n and 0<=k<len(p)
    # = sum_{j=0}^{n-1} p[j+m] * a[j] (where j+m < len(p))
    # This is like: for each m, sum the element-wise product of p[m:m+n] and a
    
    # Using FFT: (p ⊛ a) is the cross-correlation
    # cross_correlation(p, a)[m] = sum_j p[j+m] * a[j]
    # = convolution of reversed p with a, shifted
    # Actually, let's use: sum_j p[j+m]*a[j] = (flip(p) * a)[n-1+m] ... let me think
    # 
    # conv(flip(p), a)[k] = sum_i flip(p)[i] * a[k-i] = sum_i p[len(p)-1-i] * a[k-i]
    # Let j = len(p)-1-i, so i = len(p)-1-j:
    # = sum_j p[j] * a[k-len(p)+1+j]
    # For this to match sum_j p[j+m]*a[j], we need k-len(p)+1+j = j+m... that doesn't work directly.
    
    # Let me just use the direct formula with FFT:
    # (p ⊛ a)[m] = sum_{j=0}^{n-1} p[j+m] * a[j]
    # Let q = p (length 2n-1), and we want for m=0,...,n-1: sum_{j=0}^{n-1} q[j+m] * a[j]
    # This is the cross-correlation of q and a, evaluated at lags 0 to n-1.
    # cross_corr(q, a)[m] = sum_j q[j+m] * a[j]
    # = conv(flip(a), q)[n-1+m] ... let me verify
    # conv(flip(a), q)[k] = sum_i flip(a)[i] * q[k-i] = sum_i a[n-1-i] * q[k-i]
    # Let j = n-1-i: = sum_j a[j] * q[k-n+1+j]
    # For k = n-1+m: = sum_j a[j] * q[m+j] = (p ⊛ a)[m] ✓
    
    fft_size = 1 << (3*n - 2).bit_length()  # need size >= (2n-1) + n - 1 = 3n-2
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(np.flip(a), fft_size)
    cross = np.fft.irfft(P * A, fft_size)
    # (p ⊛ a)[m] = cross[n-1+m] for m = 0, ..., n-1
    cross_corr = cross[n-1:n-1+n]
    
    # gradient = 2 * cross_corr - p[2*arange(n)] * a (where 2*m < len(p))
    two_m = 2 * np.arange(n)
    p_2m = np.zeros(n)
    valid = two_m < len(p)
    p_2m[valid] = p[two_m[valid]]
    
    grad = 2 * cross_corr - p_2m * a
    return smooth_max, grad

def smooth_objective(x, beta, n):
    """Objective in log-parameterization."""
    a = np.exp(x - x.max())
    a = a / a.sum()
    ac = fast_autocorr(a)
    max_ac = ac.max()
    if max_ac * beta > 500:
        return 2.0 * n * max_ac
    shifted = beta * (ac - max_ac)
    smooth_max = max_ac + np.log(np.exp(shifted).sum()) / beta
    return 2.0 * n * smooth_max

def smooth_gradient(x, beta, n):
    """Gradient of smooth objective in log-parameterization."""
    a = np.exp(x - x.max())
    a = a / a.sum()
    smooth_val, grad_a = compute_smooth_grad(a, beta)
    # Project to tangent space: gradient w.r.t. x = a * (grad_a - a @ grad_a)
    grad_x = a * (grad_a - np.dot(a, grad_a))
    return 2.0 * n * grad_x

def lbfqs_smooth(a0, beta_start=20, beta_end=200, time_limit=60, verbose=True):
    """Optimize using smooth max + L-BFGS."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    x0 = np.log(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    
    # Continuation on beta
    n_steps = 5
    for step in range(n_steps):
        beta = beta_start * (beta_end / beta_start) ** (step / max(n_steps-1, 1))
        if verbose:
            print(f"    L-BFGS beta={beta:.0f}")
        
        def func(x):
            return smooth_objective(x, beta, n)
        
        def grad(x):
            return smooth_gradient(x, beta, n)
        
        try:
            result = scipy_minimize(func, x0, jac=grad, method='L-BFGS-B',
                                   options={'maxiter': 200, 'gtol': 1e-8, 'maxls': 50})
            x0 = result.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error: {e}")
            break
        
        # Update
        a_new = np.exp(x0 - x0.max())
        a_new = a_new / a_new.sum()
        val = 2.0 * n * fast_autocorr(a_new).max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_val
```

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=40, beta=6.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def accelerated_mirror_descent(a0, time_limit, top_k=50, beta=5.0, c0=4.0, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 5000
    current_top_k = top_k
    current_beta = beta

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=current_top_k, beta=current_beta)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        x_prev = x
        x = a_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.01, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            current_top_k = min(current_top_k + 10, 80)
            current_beta = max(current_beta - 1.0, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    AMD restart at t={t} ({elapsed:.1f}s)")
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def smooth_objective_and_grad(x, beta, n):
    a = np.exp(x - x.max())
    a = a / a.sum()
    ac = fast_autocorr(a)
    max_ac = ac.max()
    if max_ac * beta > 500:
        val = 2.0 * n * max_ac
        top_idx = np.argsort(ac)[::-1][:20]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 20.0
        w = w / w.sum()
        grad_a = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad_a[idx] += 2.0 * wi * a[j_idx]
        grad_x = a * (grad_a - np.dot(a, grad_a))
        return 2.0 * n * val, 2.0 * n * grad_x
    shifted = beta * (ac - max_ac)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    smooth_max = max_ac + np.log(exp_shifted.sum()) / beta

    # Compute cross-correlation (p ⊛ a)[m] = sum_j p[j+m] * a[j]
    fft_size = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_flip = np.fft.rfft(np.flip(a), fft_size)
    cross = np.fft.irfft(P * A_flip, fft_size)
    cross_corr = cross[n - 1:n - 1 + n]

    two_m = 2 * np.arange(n)
    p_2m = np.zeros(n)
    valid = two_m < len(p)
    p_2m[valid] = p[two_m[valid]]

    grad_a = 2.0 * cross_corr - p_2m * a
    grad_x = a * (grad_a - np.dot(a, grad_a))
    return 2.0 * n * smooth_max, 2.0 * n * grad_x


def lbfqs_smooth_optimize(a0, time_limit, verbose=True, rng_seed=42):
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    x0 = np.log(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    rng = np.random.default_rng(rng_seed)

    betas = [15, 30, 60, 120, 200, 400]
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break

        def func(x):
            v, _ = smooth_objective_and_grad(x, beta, n)
            return v

        def grad(x):
            _, g = smooth_objective_and_grad(x, beta, n)
            return g

        if verbose:
            print(f"    L-BFGS beta={beta}")
        for attempt in range(3):
            if time.monotonic() - t0 > time_limit:
                break
            try:
                result = scipy_minimize(func, x0, jac=grad, method='L-BFGS-B',
                                        options={'maxiter': 300, 'gtol': 1e-10, 'maxls': 40,
                                                 'iprint': -1})
                x0 = result.x
            except Exception as e:
                if verbose:
                    print(f"    L-BFGS error: {e}")
                break
            a_new = np.exp(x0 - x0.max())
            a_new = a_new / a_new.sum()
            val = 2.0 * n * fast_autocorr(a_new).max()
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
                if verbose:
                    print(f"      L-BFGS beta={beta} attempt={attempt}: J={best_val:.6f}")
            else:
                break

    return best_a, best_val


def lp_frank_wolfe_minimax(a0, time_limit, solver, verbose=True, max_iters=None):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    iter_count = 0
    n_iters_no_improve = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if max_iters is not None and iter_count >= max_iters:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()

        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        ac_sq = M_curr + 1e-15
        constraints = [
            T @ b_var <= t_var * ac_sq,
            cp.sum(b_var) == 1,
            t_var >= 0
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        b_val = b_val / b_sum
        t_opt = t_var.value

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_val
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m - 1e-14:
            best_m = my
            best_a = y.copy()
            n_iters_no_improve = 0
        else:
            n_iters_no_improve += 1

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-MM iter={t} ({elapsed:.1f}s): J={J:.6f}, t_opt={t_opt:.4f}")

        if n_iters_no_improve > 30:
            break

    return best_a, best_m

def quadratic_coord_descent(a0, time_limit, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.where(rng.random(n) < 0.5)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:min(len(candidates), 200)]]

            best_dst = -1
            best_new_m = best_m
            for dst in chosen:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        break
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_dst = dst
                        best_frac = frac

            if best_dst >= 0 and best_new_m < best_m - 1e-15:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                a = a / s_t
                best_m = best_new_m
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    return best_a, best_m

    print(f"\n=== Phase 4: L-BFGS polish n={len(best_seq)} ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfqs_smooth_optimize(a_in, t4, verbose=True, rng_seed=77)
        try_update(a_out, f"lbfqs2_n{len(best_seq)}")

    # Phase 5: Standard LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2500 and remaining > 60:
        print(f"\n=== Phase 5: Standard LP n={n_best} ===")
        t5 = min(remaining * 0.15, 150.0)
        if t5 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_standard(a_in, t5, solver, verbose=True)
            try_update(a_out, f"lpstd_n{n_best}")

    # Phase 6: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord n={n_best} ===")
    if remaining > 30:
        t6 = min(remaining * 0.15, 120.0)
        if t6 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = quadratic_coord_descent(a_in, t6, verbose=True, rng_seed=42)
            try_update(a_out, f"coord_n{n_best}")

    # Phase 7: Scale to different n
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        for scale_factor in [1.2, 0.8]:
            n_new = int(len(best_seq) * scale_factor)
            n_new = max(500, min(4000, n_new))
            if n_new == len(best_seq):
                continue
            if time.monotonic() - t_global > time_budget - 60:
                break
            print(f"\n=== Phase 7: Scale to n={n_new} ===")
            a_long = interpolate_sequence(best_seq, n_new)
            t7 = min(remaining * 0.2, 120.0)
            if t7 > 20:
                a_out, m1 = lbfqs_smooth_optimize(a_long, t7 * 0.6, verbose=True, rng_seed=99)
                try_update(a_out, f"lbfqs_n{n_new}")
                if time.monotonic() - t_global < time_budget - 30:
                    t7b = t7 * 0.4
                    a_out2, _ = accelerated_mirror_descent(a_long, t7b, top_k=40, beta=5.0, c0=3.0, verbose=False)
                    try_update(a_out2, f"amd_n{n_new}")

    # Phase 8: Final L-BFGS polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final L-BFGS n={n_best} ===")
    if remaining > 30:
        t8 = min(remaining * 0.7, 200.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lbfqs_smooth_optimize(a_in, t8, verbose=True, rng_seed=2024)
            try_update(a_out, f"lbfqs_final_n{n_best}")

    # Phase 9: Final coordinate
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.8, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = quadratic_coord_descent(a_in, t9, verbose=False, rng_seed=555)
            try_update(a_out, f"coord_final_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    result = [float(max(0.0, x)) for x in best_seq]
    return result
