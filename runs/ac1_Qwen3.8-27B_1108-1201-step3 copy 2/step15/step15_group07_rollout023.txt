"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent with FFT-based gradient (softmax-smoothed max)
2. Exact subgradient mirror descent (top-k active lags)
3. LP-based Frank-Wolfe refinement
4. Multi-scale search with multiple n values and restarts
Key insight: The self-convolution (not autocorrelation) is the target.
The smoothed gradient g_i = sum_k softmax(tau*c)_k * a[k-i] is computed via FFT.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

_global_solver = None

            prob.solve(solver=s)

        except Exception:

def fast_conv(a):
    """Compute full convolution conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def smoothed_gradient(a, tau, fft_size):
    """
    Compute the gradient of the log-sum-exp smoothed max of conv(a,a).
    g[i] = sum_k w[k] * a[k-i] where w = softmax(tau * conv(a,a))
    Computed via FFT: g = ifft(fft(w) * conj(fft(a)))
    """
    n = len(a)
    c = fast_conv(a)
    # Softmax with numerical stability
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w = w / w.sum()
    # Cross-correlation via FFT
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    g_full = irfft(W * np.conj(A), N)
    return g_full[:n]


def exact_subgradient(a, active_lags):
    """
    Compute exact subgradient of max_k c_k w.r.t. a_i.
    g_i = (1/|K|) * sum_{k in K} a[k-i]
    """
    n = len(a)
    size = 2 * n - 1
    g = np.zeros(n)
    for k in active_lags:
        k = int(k)
        if k < 0 or k >= size:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_hi < i_lo:
            continue
        indices = np.arange(i_lo, i_hi + 1)
        j_indices = k - indices
        g[i_lo:i_hi + 1] += a[j_indices]
    return g / len(active_lags)


def mirror_descent_smoothed(a_init, time_limit, top_k=15, verbose=True):
    """Mirror descent with smoothed and exact subgradients."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (3 * n - 2).bit_length()
    best_a = a.copy()
    c = fast_conv(a)
    best_M = c.max()
    eta = 1.0
    tau = 50.0  # smoothing parameter
    t0 = time.monotonic()
    patience = 0
    best_patience = 0
    rng = np.random.default_rng(42)
    n_iters = 0
    restart_count = 0

    for it in range(100000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(a)
        M = c.max()

        if M < best_M - 1e-12:
            if M < best_M * (1 - 1e-8):
                best_M = M
                best_a = a.copy()
                patience = 0
                best_patience = 0
            else:
                best_M = M
                best_a = a.copy()
                patience = 0
            best_patience = 0
        else:
            patience += 1
            best_patience += 1

        # Compute gradient (mix of smoothed and exact)
        if M > best_M * 1.001:
            # Far from best: use smoothed gradient
            g = smoothed_gradient(a, tau, fft_size)
        else:
            # Near best: use exact subgradient
            k_actual = min(top_k, 2 * n - 1)
            active = np.argsort(c)[-k_actual:]
            g = exact_subgradient(a, active)

        g_max = np.abs(g).max()
        if g_max < 1e-15:
            break

        # Line search with multiple step sizes
        log_a = np.log(a + 1e-30)
        improved = False
        for factor in [0.01, 0.03, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
            eta_try = eta * factor / g_max
            if eta_try < 1e-8:
                continue
            log_a_try = log_a - eta_try * g
            a_try = np.exp(log_a_try - log_a_try.max())
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = fast_conv(a_try)
            M_try = c_try.max()
            if M_try < M - 1e-14:
                a = a_try
                c = c_try
                M = M_try
                eta = eta_try * g_max
                improved = True
                break

        if not improved:
            eta *= 0.5
            # Also adjust tau
            tau = max(10.0, tau * 0.95)

        n_iters += 1

        # Restart logic
        if best_patience > 500 and n_iters > 100:
            restart_count += 1
            if restart_count <= 10:
                if verbose and restart_count % 3 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    MD restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.10f}")
                # Perturb from best
                noise = rng.laplace(0, 0.02 * n / n, n)
                a_new = best_a * np.exp(noise)
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
                else:
                    a = best_a.copy()
                eta = 1.0
                tau = 50.0
                patience = 0
                best_patience = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_M
            print(f"    MD iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.10f}, score={score:.6f}, restarts={restart_count}")

    final_score = 2.0 * n * best_M
    return best_a, final_score

def build_conv_matrix(a):
    """Build the convolution matrix T where (T @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, verbose=False):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M, cp.sum(b_var) <= 1.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def frank_wolfe_lp(a_init, time_limit, solver, verbose=True):
    """Frank-Wolfe style LP optimization."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stuck = 0
    momentum_t = 1.0
    y = a.copy()

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_conv(y).max()
        if M_curr < 1e-15:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck += 1
            if stuck > 20:
                break
            continue
        stuck = 0

        b = b / b.sum()
        n_iters += 1

        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if momentum_t < 2.0:
            a_new = b.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        for candidate in [y, a]:
            M_cand = fast_conv(candidate).max()
            if M_cand < best_M - 1e-13:
                best_M = M_cand
                best_a = candidate.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.10f}")

    return best_a, 2.0 * n * best_M


def coordinate_descent_polish(a_init, time_limit, top_lags=8, verbose=True):
    """Targeted coordinate descent: move mass from high-contribution positions."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    c = fast_conv(a)
    best_a = a.copy()
    best_M = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        n_passes += 1
        c = fast_conv(a)
        M = c.max()
        active = np.argsort(c)[-top_lags:]
        g = exact_subgradient(a, active)

        # Find positions with highest gradient (most contributing to peak)
        perm = np.argsort(-g)[:n // 2]

        improved_this_pass = False
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            # Try moving mass to low-gradient positions
            low_positions = np.argsort(g)[:n // 4]
            for j in low_positions:
                if j == i or a[j] > a[i]:
                    continue
                # Try a few transfer amounts
                for frac in [0.1, 0.3, 0.5]:
                    d = a[i] * frac
                    a[i] -= d
                    a[j] += d
                    M_try = fast_conv(a).max()
                    if M_try < M - 1e-14:
                        M = M_try
                        c = fast_conv(a)
                        if M < best_M - 1e-14:
                            best_M = M
                            best_a = a.copy()
                        improved_this_pass = True
                        break
                    else:
                        a[i] += d
                        a[j] -= d

        if not improved_this_pass:
            no_improve += 1
            if no_improve >= 10:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best_M={best_M:.10f}")

    return best_a, 2.0 * n * best_M


def structured_init(n, rng):
    """Create a structured initial sequence."""
    a = np.zeros(n)
    # Use a smooth bump-like function as starting point
    center = n // 2
    width = n // 4
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    # Add some periodic perturbation
    period = max(2, n // 50)
    a += 0.1 * np.sin(2 * np.pi * x / period)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a
