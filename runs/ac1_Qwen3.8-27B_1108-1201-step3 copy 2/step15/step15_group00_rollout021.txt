    #
    #   Wait, case (b): k=2i, the term is a[i]*a[i]. Its derivative is 2*a[i] = 2*a[k-i]. ✓
    #   Case (a): k≠2i. Two terms: a[i]*a[k-i] (from j=i) and a[k-i]*a[i] (from j=k-i). Each has derivative a[k-i] w.r.t. a[i]. Total: 2*a[k-i]. ✓
    #
    #   GREAT! So the formula is simply:
    #   d c[k]/d a[i] = 2 * a[k-i] if 0 <= k-i < n, else 0.
    #   
    #   Let me verify once more: n=2, c = [a0^2, 2a0a1, a1^2]
    #   d c[0]/d a[0] = 2*a[0-0] = 2*a0. ✓
    #   d c[1]/d a[0] = 2*a[1-0] = 2*a1. ✓
    #   d c[1]/d a[1] = 2*a[0] = 2*a0. ✓
    #   d c[2]/d a[1] = 2*a[1] = 2*a1. ✓
    #
    #   Perfect! So the gradient of the logsumexp objective w.r.t. a[i] is:
    #   grad[i] = (2/T) * sum_k softmax[k] * a[k-i] * [0<=k-i<n]
    #
    #   This can be computed efficiently! For fixed i:
    #   grad[i] = (2/T) * sum_{k: 0<=k-i<n} softmax[k] * a[k-i]
    #            = (2/T) * sum_{j: 0<=j<n} softmax[i+j] * a[j]   (substituting j=k-i)
    #            = (2/T) * sum_{j=0}^{n-1} softmax[i+j] * a[j]
    #
    #   But we need i+j < 2n-1 (the length of the autocorrelation), so j < 2n-1-i.
    #   Since j < n always, we need i+j < 2n-1, i.e., j < 2n-1-i. For i >= 0, 2n-1-i >= n (when i <= n-1), so all j in [0,n-1] are valid.
    #   Wait: i is in [0, n-1], so i+j is in [0, 2n-2] (when j=n-1, i+j = 2n-2 which is valid since autocorr has 2n-1 elements indexed 0 to 2n-2). ✓
    #
    #   So: grad[i] = (2/T) * sum_{j=0}^{n-1} softmax[i+j] * a[j]
    #
    #   This is a cross-correlation between softmax and a! Can be computed via FFT!
    #
    #   Specifically, if we define s = softmax (length 2n-1) and a (length n), then:
    #   grad[i] = (2/T) * (s * a)[i] where * denotes correlation (not convolution)
    #   Actually it's: sum_j s[i+j] * a[j] which is the cross-correlation of s and a at lag i.
    #
    #   Cross-correlation can be computed as: corr(i) = sum_j s[i+j] * a[j]
    #   = sum_j conj(s[i+j]) * a[j] (since s is real)
    #   This is the same as the convolution of s with reversed a, evaluated at specific points.
    #
    #   Alternatively: let s_reversed = s[::-1] (length 2n-1, reversed).
    #   Then (s_reversed * a)[i + (n-1)] = sum_j s_reversed[i+n-1-j] * a[j] = sum_j s[2n-2-(i+n-1-j)] * a[j] = sum_j s[n-1-i+j] * a[j]
    #   Hmm that's not quite right.
    #
    #   Let me use a different approach. Define:
    #   S = s (length 2n-1, zero-padded to FFT size)
    #   A = a (length n, zero-padded)
    #   
    #   I want: g[i] = sum_{j=0}^{n-1} s[i+j] * a[j] for i = 0, ..., n-1
    #   
    #   This is: g[i] = sum_j s[i+j] a[j]
    #   
    #   Let's define s_flipped[t] = s[2n-2-t] for t = 0,...,2n-2.
    #   Then g[i] = sum_j s[i+j] a[j] = sum_j s_flipped[2n-2-i-j] a[j]
    #   = (s_flipped * a)[2n-2-i]  (convolution)
    #   
    #   So I can compute the full convolution of s_flipped and a via FFT, and then extract the needed values.
    #
    #   Actually, even simpler: this is just a cross-correlation.
    #   g = np.correlate(s, a, mode='valid')  # This gives sum_j s[i+j]*a[j] for i=0,...,len(s)-len(a)
    #   But len(s) = 2n-1, len(a) = n, so 'valid' gives n values. 
    #   np.correlate(s, a, 'valid')[i] = sum_j s[i+j]*a[j] for j=0..n-1. ✓
    #
    #   But np.correlate might be slow for large arrays. Let me use FFT instead.
    #
    #   Actually for n=2400, np.correlate should be fine (O(n^2) = ~5.7M operations).
    #   But FFT would be O(n log n) which is better. Let me use FFT.
    #
    #   To compute g[i] = sum_j s[i+j]*a[j]:
    #   Let A = FFT(a padded to size N)
    #   Let S = FFT(s padded to size N)
    #   Then the convolution (a * s_flipped) can be computed, but let me think more carefully.
    #
    #   Actually, the simplest FFT approach:
    #   g[i] = sum_{j=0}^{n-1} s[i+j] * a[j]
    #   Let's reverse s: sr[k] = s[2n-2-k] for k=0,...,2n-2
    #   Then g[i] = sum_j s[i+j] a[j] = sum_j sr[2n-2-i-j] a[j]
    #   Let h = convolution of sr and a (full, length 2n-1+n-1 = 3n-2)
    #   h[m] = sum_j sr[m-j] a[j]
    #   g[i] = h[2n-2-i]
    #
    #   So: compute FFT of sr (padded) and FFT of a (padded), multiply, inverse FFT, extract h[2n-2-i] for i=0,...,n-1.
    #
    #   This is O(n log n). 

    return f_val, softmax, ac

def compute_logsumexp_grad(a, T=1.0):
    """Compute gradient of T*logsumexp(autocorr/T) w.r.t. a."""
    ac = fast_autocorr(a)
    n = len(a)
    
    # Numerically stable logsumexp
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    softmax = exp_vals / sum_exp
    
    # grad[i] = (2/T) * sum_j softmax[i+j] * a[j] for j=0..n-1
    # Use FFT: reverse softmax, convolve with a
    sr = softmax[::-1]  # length 2n-1
    fft_size = 1 << (2 * n - 1).bit_length()
    SR = np.fft.rfft(sr, fft_size)
    A = np.fft.rfft(a, fft_size)
    h = np.fft.irfft(SR * A, fft_size)
    # g[i] = (2/T) * h[2n-2-i]
    grad = (2.0 / T) * h[2*n-2 : 2*n-2-n : -1]  # h[2n-2], h[2n-3], ..., h[n-1]
    # Wait: h[2n-2-i] for i=0 gives h[2n-2], for i=1 gives h[2n-3], ..., for i=n-1 gives h[n-1]
    # So grad[i] = (2/T) * h[2n-2-i]
    # In numpy: indices from 2n-2 down to n-1, reversed
    grad = (2.0 / T) * h[n-1 : 2*n-1][::-1]
    
    return grad

def logsumexp_amo(a0, time_limit, T_start=0.5, T_end=0.05, n_anneal_steps=50, verbose=True, c0=3.0):
    """
    Accelerated Mirror Descent with logsumexp smooth max.
    Uses multiplicative (mirror) updates in log-space.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    
    total_steps = 5000
    
    for t in range(1, total_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = min(1.0, t / n_anneal_steps)
        T = T_start + (T_end - T_start) * progress
        T = max(T, T_end)
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient of logsumexp objective
        g = compute_logsumexp_grad(y, T)
        
        # Mirror descent step (multiplicative update in log space)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        score_new = 2.0 * n * M_new
        if score_new < best_score:
            best_score = score_new
            best_a = x.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSA t={t} ({elapsed:.1f}s): score={best_score:.6f}, T={T:.3f}")
    
    return best_a, best_score

def logsumexp_agd(a0, time_limit, T_start=0.3, T_end=0.03, verbose=True, c0=2.0):
    """
    Nesterov Accelerated Gradient Descent with logsumexp smooth max.
    Uses Euclidean (additive) projected gradient instead of mirror descent.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = min(1.0, t / 200.0)
        T = max(T_start + (T_end - T_start) * progress, T_end)
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient
        g = compute_logsumexp_grad(y, T)
        
        # Gradient step
        eta = gamma / (t + 1.0)  # Or use 1/sqrt(t)
        a_new = y - eta * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        score_new = 2.0 * n * M_new
        if score_new < best_score:
            best_score = score_new
            best_a = x.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSA-AGD t={t} ({elapsed:.1f}s): score={best_score:.6f}, T={T:.4f}")
    
    return best_a, best_score

def peak_shaving_descent(a0, time_limit, verbose=True, rng_seed=42):
    """
    Targeted peak-shaving: identify top autocorrelation peaks and 
    perform multiplicative updates that specifically reduce them.
    More aggressive than soft subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    for round_num in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Find top peaks
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        
        # For each peak, compute the "responsibility" of each a[i]
        # and reduce a[i] proportionally to its responsibility for the peaks
        grad = np.zeros(n)
        for k in top_k:
            k_int = int(k)
            # Contribution of a[i] to ac[k]: 2*a[k-i] if 0<=k-i<n
            j = k_int - np.arange(n)
            valid = (j >= 0) & (j < n)
            contrib = np.zeros(n)
            contrib[valid] = a[j[valid]]
            # Weight by peak height (stronger for taller peaks)
            weight = ac[k] ** 2
            grad += weight * contrib
        
        # Normalized gradient
        g_norm = grad / (grad.sum() + 1e-30)
        
        # Multiplicative update: reduce positions that contribute to peaks
        # a_new[i] = a[i] * (1 - step * g_norm[i])
        step = 0.1
        a_new = a * (1.0 - step * g_norm)
        a_new = np.maximum(a_new, 1e-15)
        
        # Also add some mass back to low-contribution positions
        low_mask = g_norm < np.median(g_norm)
        if low_mask.any():
            total_reduction = (a - a_new).sum()
            a_new[low_mask] += total_reduction * (a[low_mask] / (a[low_mask].sum() + 1e-30))
        
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < M * (1 - 1e-10):
            a = a_new
            score_new = 2.0 * n * M_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
        
        if verbose and round_num % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS round={round_num} ({elapsed:.1f}s): score={best_score:.6f}")
    
    return best_a, best_score

def lp_projection(a0, time_limit, verbose=True, max_iters=100):
    """
    LP-based Frank-Wolfe: maximize sum(b) subject to (a*b)[k] <= M for all k, b >= 0.
    Then move toward b.
    """
    n = len(a0)
    solver = None
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_mom = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        b_var = cp.Variable(n, nonneg=True)
        # Build the convolution matrix T: T[k, j] = y[k-j] if 0<=k-j<n
        # (y*b)[k] = sum_j y[k-j] * b[j] for 0<=k-j<n
        T = np.zeros((2*n-1, n))
        for j in range(n):
            T[j:j+n, j] = y
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_mom = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_mom = y_new + beta_t * (y_new - y)
            x_mom = np.maximum(x_mom, 0.0)
            sx = x_mom.sum()
            if sx < 1e-15:
                x_mom = y_new.copy()
            else:
                x_mom = x_mom / sx
        
        y = y_new
        
        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            sc = 2.0 * n * mc
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
        
        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.1f}s): score={best_score:.6f}")
    
    return best_a, best_score

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent: for each position, try moving mass to other positions
    and accept if it reduces the max autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(10, len(ac))]
        
        improved = False
        # Try moving mass from high-contributing positions to low-contributing ones
        n_trials = 50
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a source from top-contributing positions
            k = int(top_peaks[rng.integers(len(top_peaks))])
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Pick a random destination
            dst = int(rng.integers(n))
            if dst == src:
                continue
            
            # Try moving a fraction of mass
            for frac in [0.01, 0.03, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < M_curr * (1 - 1e-12):
                    a = a_trial
                    M_curr = M_trial
                    score_new = 2.0 * n * M_curr
                    if score_new < best_score:
                        best_score = score_new
                        best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_score / (2.0 * n):
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord ({elapsed:.1f}s): score={best_score:.6f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np

def fast_autocorr(a):
    """Compute full linear autocorrelation (self-convolution) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    try:

def compute_logsumexp_grad(a, T=1.0):
    """
    Compute gradient of T*log(sum_k exp(c[k]/T)) where c = a*a (full convolution).
    
    grad[i] = 2 * sum_{j=0}^{n-1} softmax[i+j] * a[j]
    
    where softmax = exp(c/T) / sum(exp(c/T)) computed in a numerically stable way.

    """
    ac = fast_autocorr(a)
    n = len(a)
    
    # Numerically stable softmax
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    if sum_exp < 1e-300:
        return np.zeros(n)
    softmax = exp_vals / sum_exp

    sr = softmax[::-1]  # length 2n-1
    conv_len = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()

    AV = np.fft.rfft(a, fft_size)
    h = np.fft.irfft(SR * AV, fft_size)[:conv_len]

    # i=0 -> h[2n-2], i=n-1 -> h[n-1]
    grad = 2.0 * h[n - 1:2 * n - 1][::-1]
    
    return grad
