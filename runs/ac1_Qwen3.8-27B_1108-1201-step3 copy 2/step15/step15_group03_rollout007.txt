import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_conv_self(a):
    """Compute full linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Compute full linear cross-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

def solve_lp(a, M, active_indices=None):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M on active indices, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_indices is not None and len(active_indices) < size:
        T = build_T(a)
        constraints = [T[active_indices, :] @ b_var <= M]
    else:
        T = build_T(a)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:

            # LSE gradient

            shifted = np.maximum(shifted, -100.0)

                a = a * norm / s_new

                if new_max < best_score - 1e-12:

def spectral_peak_reduction(a0, time_limit, rng, verbose=True):
    """Reduce peaks by modifying the frequency domain representation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find top peaks
        n_peaks = min(20, size // 10)
        peak_idx = np.argsort(conv)[-n_peaks:][::-1]
        peak_vals = conv[peak_idx]
        if peak_vals[0] < 1e-12:
            break
        
        # Compute FFT of a
        A = rfft(a, fft_size)

        # We modify A to reduce the contribution to these peaks
        modified_A = A.copy()
        for pi, k in enumerate(peak_idx):
            weight = peak_vals[pi] / peak_vals[0]
            # Compute the gradient direction in frequency domain
            # The contribution of freq m to conv[k] is A[m]*conj(A[m-k]) + conj(A[m])*A[m-k]
            # Simplified: reduce |A[m]| where it contributes most to peak k
            m_range = np.arange(len(A))
            kernel = np.cos(2 * np.pi * m_range * k / fft_size)
            # Reduce A where kernel is positive (contributing to peak)
            reduction = 0.02 * weight * kernel
            modified_A = modified_A * (1.0 - reduction)
        
        a_new = irfft(modified_A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller reduction
            modified_A = A.copy()
            for pi, k in enumerate(peak_idx[:5]):
                weight = peak_vals[pi] / peak_vals[0]
                m_range = np.arange(len(A))
                kernel = np.cos(2 * np.pi * m_range * k / fft_size)
                reduction = 0.005 * weight * kernel
                modified_A = modified_A * (1.0 - reduction)
            a_new = irfft(modified_A, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                # Perturb and continue
                a += 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                a = a * norm / s_new
        
        if verbose and it % 100 == 0:
            print(f"    Spectral it={it}: best={best_score:.8f}")
    
    return best_a, best_score
