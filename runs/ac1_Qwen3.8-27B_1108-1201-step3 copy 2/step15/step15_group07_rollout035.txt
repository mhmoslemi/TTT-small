def lp_best_response_trust(a, M, trust_r, eps_frac, solver):
    """
    Solve LP: max 1'b - (regularization via eps lower bound)
    s.t. T(a) @ b <= M * 1
         b >= eps_frac * sum(b)  (anti-sparsity)

        T @ b_var <= M,
        b_var >= eps_frac * sum_b,
        cp.norm(b_var - a, 1) <= trust_r

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def compute_sensitivity(a):
    """Compute sens[i, l] = a[i+l] + a[i-l] for all i (positions) and l (lags)."""
    n = len(a)
    size = 2 * n - 1
    sens = np.zeros((n, size))
    for i in range(n):
        # a[i+l] for l = 0, ..., n-1-i
        if n - 1 - i >= 0:
            sens[i, 0:n - i] += a[i:]
        # a[i-l] for l = 0, ..., i
        if i >= 0:
            sens[i, 0:i + 1] += a[i::-1]
    return sens


def lp_solve(a, M, solver, use_trust, trust_r, eps_frac):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    sum_b = cp.sum(b_var)
    constraints = [T @ b_var <= M]
    if eps_frac > 0:
        constraints.append(b_var >= eps_frac * sum_b)
    if use_trust and trust_r is not None and trust_r > 0:
        constraints.append(cp.norm(b_var - a, 1) <= trust_r)
    prob = cp.Problem(cp.Maximize(sum_b), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def sensitivity_cd(a0, time_limit, verbose=True):

    Coordinate descent using precomputed sensitivity vectors.
    For transfer of d from i to j: conv_new = conv + d*(sens[j]-sens[i]) + O(d^2)
    Uses first-order check for speed, exact verification on acceptance.
    """

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    sens = compute_sensitivity(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    # Precompute candidate offsets
    max_off = min(n // 2, 200)
    offsets = np.arange(1, max_off + 1)

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve > 80:
            break

        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(perm[idx])
            if a[i] < norm * 1e-11:
                continue

            best_new_max = best_score
            best_j = -1
            best_d = 0.0

            # Try candidate offsets (subset for speed)
            n_try = min(len(offsets), 40)
            trial_offsets = offsets[:n_try]

            for off in trial_offsets:
                j1 = (i + int(off)) % n
                j2 = (i - int(off)) % n
                for j in [j1, j2]:
                    if j == i:
                        continue
                    # First-order check
                    delta = sens[j] - sens[i]
                    # Try a few fractions
                    for frac in [0.3, 0.6, 0.9]:
                        d = a[i] * frac
                        if d < 1e-15:
                            continue
                        # Approximate new max
                        new_conv_approx = conv + d * delta
                        approx_max = new_conv_approx.max()
                        if approx_max < best_new_max - 1e-12:
                            # Verify with exact computation
                            a[i] -= d
                            a[j] += d
                            exact_conv = fast_autocorr(a)
                            exact_max = exact_conv.max()
                            if exact_max < best_new_max:
                                best_new_max = exact_max
                                best_j = j
                                best_d = d
                                best_conv = exact_conv.copy()
                                pass_improved = True
                            else:
                                a[j] -= d
                                a[i] += d

            # Apply best move found for this i
            if best_j >= 0 and best_d > 0 and best_new_max < best_score:
                a[i] -= best_d
                a[best_j] += best_d
                conv = best_conv
                best_score = best_new_max
                best_a = a.copy()
                # Update sensitivity (only changed rows)
                sens[i] = 0.0
                if n - 1 - i >= 0:
                    sens[i, 0:n - i] += a[i:]
                sens[i, 0:i + 1] += a[i::-1]
                sens[best_j] = 0.0
                if n - 1 - best_j >= 0:
                    sens[best_j, 0:n - best_j] += a[best_j:]
                sens[best_j, 0:best_j + 1] += a[best_j::-1]
                no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                # Small perturbation to escape
                noise = rng.normal(0, norm * 0.003, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                    conv = fast_autocorr(a)
                    if conv.max() < best_score:
                        best_score = conv.max()
                        best_a = a.copy()
                    sens = compute_sensitivity(a)
                    no_improve = 0

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score

    print(f"\n=== Phase 2: Sensitivity CD n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = sensitivity_cd(a_norm, t2, verbose=True)
        try_update(a_out, "cd1")

    # === Phase 3: LP round 2 from CD result (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: LP R2 n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = enhanced_lp(a_norm, t3, solver, verbose=True)
        try_update(a_out, "lp2")

    # === Phase 4: CD round 2 (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: CD R2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = sensitivity_cd(a_norm, t4, verbose=True)
        try_update(a_out, "cd2")

    # === Phase 5: Final LP polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.05
    print(f"\n=== Phase 5: Final LP n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = enhanced_lp(a_norm, t5, solver, verbose=False)
        try_update(a_out, "lp3")

    # === Phase 6: Final CD polish (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_best = len(best_seq)
        t6 = max(remaining - 30, 10)
        print(f"\n=== Phase 6: Final CD n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = sensitivity_cd(a_norm, t6, verbose=False)
        try_update(a_out, "cd3")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
