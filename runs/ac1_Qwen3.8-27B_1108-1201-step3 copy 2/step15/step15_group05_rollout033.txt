    """Compute linear autocorrelation (full convolution of a with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    result = irfft(A * A, fft_size)[:size]
    return result


def evaluate_safe(a):
    """Safely evaluate the score."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        rho = np.array([0])
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def get_subgradient(x, k):
    """Compute gradient of f_k(x) = (x*x)[k] at point x."""
    n = len(x)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = k - j
        if 0 <= i1 < n:
            val += x[i1]
        i2 = j - k
        if 0 <= i2 < n:
            val += x[i2]
        grad[j] = val
    return grad


def get_subgradient_fast(x, k):
    """Efficient gradient computation using the structure of the convolution."""
    n = len(x)
    grad = np.zeros(n)
    # (x*x)[k] = sum_i x[i] * x[k-i]
    # d/dx[j] = x[k-j] (if 0<=k-j<n) + x[j-k] (if 0<=j-k<n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    for j in range(lo, hi):
        grad[j] += x[k - j]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    for j in range(lo2, hi2):
        if j - k != k - j:  # avoid double counting when j-k == k-j, i.e., j=k (only if k<n)
            grad[j] += x[j - k]
        else:
            # j-k == k-j means j = k (only valid if 0 <= k < n)
            pass
    # Actually let me redo this more carefully
    # d/dx[j] of sum_i x[i]*x[k-i]:
    # Term where i = j: x[j]*x[k-j] -> derivative w.r.t x[j] gives x[k-j] (if 0<=k-j<n)
    # Term where k-i = j, i.e., i = k-j: x[k-j]*x[j] -> derivative w.r.t x[j] gives x[k-j] (if 0<=k-j<n and k-j != j)
    # If k-j == j (i.e., 2j = k), we get 2*x[j] from a single term
    grad2 = np.zeros(n)
    for j in range(n):
        i = k - j
        if 0 <= i < n:
            if i == j:
                grad2[j] = 2.0 * x[j]
            else:
                grad2[j] = 2.0 * x[i]
    return grad2


def nesterov_projected_gradient(x0, time_limit, verbose=False):
    """
    Nesterov-accelerated projected gradient descent on the simplex.
    Minimizes f(x) = max_k (x*x)[k] over x >= 0, sum(x) = 1.
    """
    n = len(x0)
    x = np.asarray(x0, dtype=np.float64).copy()
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) / n
    else:
        x = x / s

    best_x = x.copy()
    best_score = fast_autocorr(x).max()
    t0 = time.monotonic()

    # Nesterov momentum
    momentum = 0.0
    x_prev = x.copy()
    y = x.copy()

    # Step size with backtracking
    step = 0.5 / n
    max_step = 2.0 / np.sqrt(n)
    min_step = 1e-6

    # Ergodic average
    erg = np.zeros(n)
    erg_count = 0

    # Tracking
    patience = 0
    max_patience = 5000
    n_iters = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        # Compute convolution and find max
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        k_star = int(np.argmax(conv_y))

        # Subgradient at y
        grad = get_subgradient_fast(y, k_star)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        trial_step = step
        improved = False
        for _bt in range(20):
            x_trial = y - trial_step * grad
            x_trial = project_simplex(x_trial)
            conv_trial = fast_autocorr(x_trial)
            score_trial = conv_trial.max()
            if score_trial < M_curr - 1e-14 * max(M_curr, 1.0):
                improved = True
                break
            trial_step *= 0.5
            if trial_step < min_step:
                break

        if not improved:
            # Try multiple subgradients
            conv_y = fast_autocorr(y)
            n_peaks = min(5, n // 100 + 1)
            top_idx = np.argsort(conv_y)[-n_peaks:]
            found = False
            for ki in top_idx:
                grad_ki = get_subgradient_fast(y, ki)
                gn = np.linalg.norm(grad_ki)
                if gn < 1e-15:
                    continue
                grad_ki = grad_ki / gn
                x_trial = y - trial_step * gn * grad_ki
                x_trial = project_simplex(x_trial)
                score_trial = fast_autocorr(x_trial).max()
                if score_trial < M_curr - 1e-14:
                    grad = grad_ki * gn
                    improved = True
                    found = True
                    break
            if not found:
                patience += 1
                step *= 0.95
                if step < min_step:
                    break
                # Still update with small step
                x_trial = y - step * 0.1 * grad
                x_trial = project_simplex(x_trial)
        else:
            step = min(trial_step * 1.1, max_step)
            patience = 0

        # Nesterov momentum update
        x_new = x_trial.copy()
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / s_new

        momentum = (t - 1.0) / (t + 2.0)
        momentum = min(momentum, 0.95)
        y_new = x_new + momentum * (x_new - x_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new / s_y

        x_prev = x
        x = x_new
        y = y_new
        n_iters += 1

        # Ergodic average
        erg += x
        erg_count += 1

        # Track best
        score_x = fast_autocorr(x).max()
        if score_x < best_score:
            best_score = score_x
            best_x = x.copy()

        # Check ergodic average periodically
        if erg_count >= 50 and erg_count % 100 == 0:
            erg_check = erg / erg_count
            s_erg = erg_check.sum()
            if s_erg > 1e-15:
                erg_check = erg_check / s_erg
                score_erg = fast_autocorr(erg_check).max()
                if score_erg < best_score:
                    best_score = score_erg
                    best_x = erg_check.copy()

        if verbose and n_iters % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PG iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_x, best_score


def build_conv_matrix_rows(a, k_indices, n):
    """Build rows of the convolution matrix for given lag indices."""
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a[k - j]
    return mat


def lp_best_response(a_fixed, M, k_indices, solver):
    """
    Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in k_indices, b >= 0.
    """
    n = len(a_fixed)
    k_arr = np.asarray(k_indices)
    T = build_conv_matrix_rows(a_fixed, k_arr, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_iterate(a0, time_limit, solver, verbose=False, seed=0):
    """
    Nesterov-accelerated LP best-response iteration.
    Uses adaptive active set (top fraction of lags).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    n_iters = 0

    # Ergodic average
    erg = np.zeros(n)
    erg_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set: use top fraction of lags
        # Start with top 10%, expand to top 50% over time
        frac = min(0.5, 0.1 + 0.4 * t / 500.0)
        n_active = max(10, int(frac * (2 * n - 1)))
        # Always include the top few
        top_k = np.argsort(conv_y)[-n_active:]
        k_indices = np.unique(top_k)

        b = lp_best_response(y, M_curr, k_indices, solver)
        if b is None:
            # Try with all constraints if active set fails
            if t % 20 == 0:
                k_indices = np.arange(2 * n - 1)
                b = lp_best_response(y, M_curr, k_indices, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.97)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        erg += a
        erg_count += 1
        if erg_count % 20 == 0:
            erg_check = erg / erg_count
            s_erg = erg_check.sum()
            if s_erg > 1e-15:
                erg_check = erg_check * norm / s_erg
                score_erg = fast_autocorr(erg_check).max()
                if score_erg < best_score:
                    best_score = score_erg
                    best_a = erg_check.copy()
                    stall_count = 0

        # Stall detection and restart
        if stall_count > 60:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, try to decrease the max autocorrelation.
    Uses a smart sampling of candidate values.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_rounds = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        if no_improve_rounds >= max_no_improve:
            break

        improved = False
        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Process coordinates in order of "importance" (those at peaks)
        # Get the top active lags
        n_active_lags = max(5, n // 100)
        active_lags = np.argsort(conv)[-n_active_lags:]

        # For each coordinate, compute its effect on the active lags
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-10:
                continue

            # Find which active lags this coordinate affects
            # a[j] affects (a*a)[k] for k where j is in the valid range
            # Valid: max(0, k-n+1) <= j <= min(n-1, k), i.e., k in [j, j+n-1]
            affected_lags = []
            for k in active_lags:
                if j >= max(0, k - n + 1) and j <= min(n - 1, k):
                    # Contribution of a[j] to (a*a)[k]:
                    # From term i=j: a[j]*a[k-j] (if 0<=k-j<n and k-j != j)
                    # From term k-i=j, i=k-j: a[k-j]*a[j] (if 0<=k-j<n and k-j != j)
                    # If k-j == j (i.e., k=2j): contribution is 2*a[j]*a[j]... wait
                    # (a*a)[k] includes a[j]*a[k-j] and a[k-j]*a[j] unless j == k-j
                    # If j == k-j (k = 2j): only one term a[j]^2
                    # So the "linear" part in a[j] is: a[j] * (a[k-j] + a[j-k]) where each is 0 if invalid
                    # But if k-j == j, it's a[j]^2 (quadratic)
                    i1 = k - j
                    i2 = j - k
                    linear_part = 0.0
                    quad_part = False
                    if 0 <= i1 < n:
                        if i1 == j:
                            quad_part = True
                        else:
                            linear_part += a[i1]
                    if 0 <= i2 < n:
                        if i2 == j:
                            pass  # already counted
                        else:
                            linear_part += a[i2]

                    if linear_part > 0 or quad_part:
                        affected_lags.append((k, linear_part, quad_part))

            if not affected_lags:
                continue

            # Try a few candidate values
            s_others = a.sum() - aj
            candidates = [aj * 0.5, aj * 0.7, aj * 0.85, aj * 1.0,
                          aj * 1.15, aj * 1.3, aj * 1.5]
            # Add values based on the gradient direction
            total_linear = sum(lp for _, lp, _ in affected_lags)
            if total_linear > 0:
                candidates.append(aj - 0.1 * aj)
                candidates.append(aj + 0.1 * aj)

            candidates = [max(0.0, min(x, norm)) for x in candidates]
            candidates.append(0.0)

            best_x = aj
            best_local_score = 2.0 * n * M_curr / (a.sum() ** 2)

            for cx in candidates:
                if abs(cx - aj) < 1e-12:
                    continue
                new_sum = s_others + cx
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                # Estimate new max autocorrelation at affected lags
                # (a*a)[k] changes by: cx^2*scale^2 - aj^2*scale_old^2 + scale^2*(2*cx - 2*aj)*linear_part ...
                # This is getting complicated. Let's just do a quick FFT for the worst case.
                pass

            # Simpler approach: just try scaling the coordinate
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3]:
                new_aj = aj * factor
                a[j] = new_aj
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = aj
                    continue
                a_scaled = a * norm / s_new
                new_conv = fast_autocorr(a_scaled)
                new_score = 2.0 * n * new_conv.max() / norm ** 2
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a_scaled.copy()
                    a = a_scaled.copy()
                    improved = True
                    break
            if not improved:
                a[j] = aj
                a = a * norm / a.sum()

        if improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD round={no_improve_rounds} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_and_optimize(a_best, rng, time_per_run, n_runs, solver, verbose=False):
    """Try multiple perturbed versions of the best sequence."""
    n = len(a_best)
    norm = np.sqrt(2.0 * n)
    best_score = fast_autocorr(a_best).max()
    best_result = a_best.copy()

    for r in range(n_runs):
        if time_per_run <= 10:
            break
        # Multiplicative perturbation
        scale = 0.01 + 0.02 * r
        noise = 1.0 + scale * rng.standard_normal(n)
        noise = np.maximum(noise, 0.0)
        pert = a_best * noise
        s = pert.sum()
        if s < 1e-15:
            continue
        pert = pert * norm / s

        # Short projected gradient
        pg_result, pg_score = nesterov_projected_gradient(pert, time_per_run, verbose=False)
        if pg_score < best_score:
            best_score = pg_score
            best_result = pg_result.copy()
            if verbose:
                print(f"    Perturb {r}: improved to {best_score:.10f}")

    return best_result, best_score


def propose_candidate():
    """
    Main search function. Multi-phase optimization:
    1. Projected gradient (fast, many iterations)
    2. LP Nesterov (fewer but more impactful iterations)
    3. Coordinate descent polish
    4. Perturbation restarts
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0  # Leave 50s safety margin

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
            n_base = len(parent)
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        # Default initialization: raised cosine
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.maximum(0.0, 0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 1.5
        best_seq = np.maximum(best_seq, 1e-6)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_cur = len(best_seq)
    print(f"Starting optimization: n={n_cur}, initial score={best_score:.8f}")

    # === Phase 1: Projected Gradient (fast, ~300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Projected Gradient, budget={t1:.0f}s ===")
        x_init = best_seq / best_seq.sum()
        a_out, s_out = nesterov_projected_gradient(x_init, t1, verbose=True)
        if np.isfinite(s_out):
            try_update(a_out, f"pg1_n{n_cur}")

    # === Phase 2: LP Nesterov (main, ~600s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP Nesterov n={n_cur}, budget={t2:.0f}s ===")
        norm = np.sqrt(2.0 * n_cur)
        a_init = best_seq * norm / best_seq.sum()
        a_out, s_out = nesterov_lp_iterate(a_init, t2, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 3: Projected Gradient refinement (~200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: PG refinement n={n_cur}, budget={t3:.0f}s ===")
        x_init = best_seq / best_seq.sum()
        a_out, s_out = nesterov_projected_gradient(x_init, t3, verbose=True)
        if np.isfinite(s_out):
            try_update(a_out, f"pg2_n{n_cur}")

    # === Phase 4: LP Nesterov round 2 (~300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.25, 300.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP round 2 n={n_cur}, budget={t4:.0f}s ===")
        norm = np.sqrt(2.0 * n_cur)
        a_init = best_seq * norm / best_seq.sum()
        a_out, s_out = nesterov_lp_iterate(a_init, t4, solver, verbose=True, seed=123)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Coordinate descent polish (~100s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Coord Descent n={n_cur}, budget={t5:.0f}s ===")
        norm = np.sqrt(2.0 * n_cur)
        a_init = best_seq * norm / best_seq.sum()
        a_out, s_out = coordinate_descent_polish(a_init, t5, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 6: Perturbation restarts (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: Perturbation restarts n={n_cur}, budget={remaining:.0f}s ===")
        n_restarts = 3
        t_each = min(remaining / n_restarts, 80.0)
        for r in range(n_restarts):
            if time_budget - (time.monotonic() - t_global) < 30:
                break
            scale = 0.02 + 0.015 * r
            noise = 1.0 + scale * rng.standard_normal(n_cur)
            noise = np.maximum(noise, 0.0)
            pert = best_seq * noise
            s = pert.sum()
            if s < 1e-15:
                continue
            x_pert = pert / s
            a_out, s_out = nesterov_projected_gradient(x_pert, t_each, verbose=False)
            if np.isfinite(s_out):
                try_update(a_out, f"pert{r}_n{n_cur}")

            # Also try LP from perturbed point
            if time_budget - (time.monotonic() - t_global) > 30:
                norm = np.sqrt(2.0 * n_cur)
                a_pert = pert * norm / s
                a_out2, _ = nesterov_lp_iterate(a_pert, min(t_each, 40.0), solver, verbose=False, seed=500 + r)
                try_update(a_out2, f"lp_pert{r}_n{n_cur}")

    # === Phase 7: Final short PG polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_cur = len(best_seq)
        t7 = min(remaining * 0.8, 30.0)
        x_init = best_seq / best_seq.sum()
        a_out, s_out = nesterov_projected_gradient(x_init, t7, verbose=False)
        if np.isfinite(s_out):
            try_update(a_out, f"pg_final_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        best_seq = np.ones(1600) / 1600
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
