def fft_conv(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fft_grad(a, w):
    """Compute gradient: grad[i] = 2 * sum_j a[j] * w[i+j] for i=0..n-1.
    Uses cross-correlation via FFT.
    """
    n = len(a)
    m = len(w)
    # Need linear convolution of a_rev (length n) with w (length m)
    # Result has length n+m-1, we want elements [n-1, n, ..., 2n-2]
    total_len = n + m - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = a[::-1]
    A = rfft(a_rev, fft_size)
    W = rfft(w, ftp_size) if False else rfft(w, fft_size)
    conv_full = irfft(A * W, fft_size)
    return 2.0 * conv_full[n - 1: 2 * n - 1]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_optimize(a0, time_limit, verbose=True):
    """Minimize max(conv(a,a)) using smooth log-sum-exp gradient with Nesterov."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fft_conv(a).max()
    t0 = time.time()

    temps = [50.0, 25.0, 10.0, 5.0, 2.5, 1.5, 1.0, 0.6, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]

    for T in temps:
        if time.time() - t0 > time_limit:
            break
        remaining = time_limit - (time.time() - t0)
        if remaining < 0.5:
            break
        n_steps = max(20, min(800, int(remaining * 4)))

        y = a.copy()
        for it in range(n_steps):
            if time.time() - t0 > time_limit:
                break

            c = fft_conv(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = fft_grad(y, w)

            # Adaptive step size with backtracking
            step = norm / (n * T)
            step = min(step, norm / (n * 0.3))

            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            # Nesterov momentum
            alpha_mom = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha_mom) * y + alpha_mom * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it + 1.0) / (it + 3.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fft_conv(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fft_conv(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        if verbose:
            elapsed = time.time() - t0
            print(f"    Smooth T={T:.3f} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def build_conv_matrix(a):
    """Build dense convolution matrix T where (T@b)[k] = (a*b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T

def lp_best_response(a, M, solver, active_constraints=None):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)

    if active_constraints is not None and len(active_constraints) < size:
        T_use = T[active_constraints, :]
    else:
        T_use = T

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration, adaptive constraints, and restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fft_conv(a).max()

    y = a.copy()
    t0 = time.time()
    n_iters = 0
    stagnation = 0
    best_at_stagnation = best_score

    for t in range(1, 10000000):
        if time.time() - t0 > time_limit:
            break

        M_curr = fft_conv(y).max()
        if M_curr < 1e-14:
            break

        # Adaptive constraint selection
        conv_y = fft_conv(y)
        # Use constraints that are > 80% of max (tighten over time)
        frac = min(0.95, 0.75 + 0.0001 * n_iters)
        threshold = frac * M_curr
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 5:
            threshold = 0.5 * M_curr
            active = np.where(conv_y >= threshold)[0]
        if len(active) < 3:
            active = np.arange(len(conv_y))

        b = lp_best_response(y, M_curr, solver, active_only=active)
        if b is None:
            b = lp_best_response(y, M_curr, solver)
            if b is None:
                stagnation += 1
                if stagnation > 50:
                    break
                continue

        stagnation = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update with periodic restart
        if t % 100 == 1:
            # Restart momentum
            y = a.copy()
            a_new = g.copy()
        else:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            momentum = (t) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fft_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            best_at_stagnation = best_score

        score_a = fft_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            best_at_stagnation = best_score

        # Periodic average check
        if n_iters % 50 == 0:
            avg_score = fft_conv(a).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = a.copy()

        if verbose and n_iters % 100 == 0:

def coordinate_descent(a0, time_limit, rng, window=30, verbose=True):

    n_passes = 0

                    if conv_t.max() < best_score - 1e-13:

                        conv = conv_t

                        best_a = a.copy()
                        pass_improved = True

                    else:

                        d2 = a[i] * 0.05

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.time() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        # Use power law initialization
        best_seq = power_law_init(n_base, 0.35) / power_law_init(n_base, 0.35).sum()
        best_score = evaluate_safe(best_seq)
        print(f"Init (power law a=0.35): n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Smooth optimization on parent (20% time) ===
    remaining = time_budget - (time.time() - t_start)
    t1 = remaining * 0.20
    print(f"\n=== Phase 1: Smooth opt n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t1, verbose=True)
        try_update(a_out, f"smooth_n{n_base}")

    # === Phase 2: LP Nesterov on best (45% time) ===
    remaining = time_budget - (time.time() - t_start)
    n_c = len(best_seq)
    t2 = remaining * 0.45
    print(f"\n=== Phase 2: LP Nesterov n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_c}")

    # === Phase 3: Coordinate descent (10% time) ===
    remaining = time_budget - (time.time() - t_start)
    n_c = len(best_seq)
    t3 = remaining * 0.10
    print(f"\n=== Phase 3: Coord descent n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t3, rng, window=60, verbose=True)
        try_update(a_out, f"cd_n{n_c}")

    # === Phase 4: LP Nesterov round 2 (10% time) ===
    remaining = time_budget - (time.time() - t_start)
    n_c = len(best_seq)
