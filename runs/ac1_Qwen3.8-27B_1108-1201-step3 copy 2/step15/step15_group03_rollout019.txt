"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def normalize_a(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s

def lse_gradient_descent(a0, time_limit, n_stages=10, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    temps = np.geomspace(50.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        n_steps = max(20, min(200, int(remaining * 8)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.02 * it))
            
            a_new = a - step * grad
            a_new = normalize_a(a_new, norm)
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")
    
    return best_a, best_score

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None, warm_start=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        if warm_start is not None:
            b_var.value = warm_start
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    # Try to get dual values
    dual_val = None
    try:
        if active is not None and 0 < len(active) < size:
            dual_val = np.abs(np.asarray(constraints[0].dual_value, dtype=np.float64))
        else:
            dual_val = np.abs(np.asarray(constraints[0].dual_value, dtype=np.float64))
    except:
        dual_val = None
    return val, dual_val

def accelerated_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    x_prev = a.copy()
    warm_start = None
    
    # Adaptive parameters
    screen_ratio = 0.75  # Start with top 75% of constraints
    full_lp_interval = 30  # Do full LP every 30 iterations
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive constraint screening
        if t % full_lp_interval == 0:
            active = None  # Full LP
        else:
            # Tighten screening as we converge
            progress = min(1.0, t / 500.0)
            threshold_ratio = screen_ratio - 0.2 * progress  # From 0.75 to 0.55
            active = np.where(conv_y >= M * threshold_ratio)[0]
            if len(active) < 20:
                active = None
        
        b, dual = lp_solve(y, M, solver, active, warm_start)
        if b is None:
            b, dual = lp_solve(y, M, solver, None)
            if b is None:
                continue
        warm_start = b
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Verify if we dropped constraints
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * 1.01:
                b_full, _ = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = normalize_a(y_new, norm)
        
        beta = (t - 1.0) / (t + 2.0)
        a_new = y_new + beta * (y_new - x_prev)
        a_new = normalize_a(a_new, norm)
        
        x_prev = y_new
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): M={M:.6f}, best={best_score:.8f}")
    
    return best_a, best_score

def adam_smooth_optimize(a0, time_limit, verbose=True):
    """Adam optimizer on smoothed max-convolution objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    # Temperature schedule
    T_start, T_end = 20.0, 0.05
    n_iters_target = 500
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / n_iters_target)
        T = T_start * (T_end / T_start) ** progress
        
        conv = fast_conv_self(a)
        max_c = conv.max()
        
        shifted = (conv - max_c) / T
        shifted = np.maximum(shifted, -60.0)
        w = np.exp(shifted)
        w = w / w.sum()
        
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        
        lr = 0.5 * T / (np.sqrt(np.maximum(v_hat, eps)).mean() + eps)
        a_new = a - lr * m_hat
        a_new = normalize_a(a_new, norm)
        a = a_new
        
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam it={it} ({elapsed:.0f}s): T={T:.3f}, best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):

    if old_n == new_n:

    s = f.sum()
    if s < 1e-15:

    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)

    f_interp = np.interp(new_x, old_x, f_norm)

    return f_interp / ss

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)

            break

        n_top = min(15, size)
        top_lags = np.argsort(conv)[-n_top:]

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Compute shift vectors for i
            shift_i = np.zeros(size)
            lo = max(0, i)
            hi = min(size, i + n)
            if hi > lo:
                shift_i[lo:hi] = a[:hi - lo]

            # Candidate j values
            j_set = set()
            for _ in range(6):
                j_set.add(int(rng.integers(n)))
            for kl in top_lags[:8]:
                kl_int = int(kl)
                j1 = (kl_int - i + n) % n
                j2 = (kl_int + i) % n
                if j1 != i:
                    j_set.add(j1)
                if j2 != i:
                    j_set.add(j2)
            j_set.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_set)[:15]:
                if j == i:
                    continue
                aj = a[j]

                shift_j = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    shift_j[lo:hi] = a[:hi - lo]

                linear_change = 2.0 * (shift_j - shift_i)

                max_d = ai * 0.95
                if max_d < 1e-12:
                    continue
                n_samples = 20
                deltas = np.linspace(0.0, max_d, n_samples)

                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * linear_change
                    if 2 * i < size:
                        new_conv[2 * i] += (-2.0 * ai + d) * d
                    if 2 * j < size:
                        new_conv[2 * j] += (2.0 * aj + d) * d
                    if i + j < size:
                        new_conv[i + j] -= 2.0 * d * d

                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize_a(a, norm)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                n_perturb = max(5, n // 60)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = normalize_a(a, norm)
        else:
            no_improve = 0

        if verbose and (idx + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave pass ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def adam_smooth_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8

    T_start, T_end = 15.0, 0.03
    n_iters_target = 400

    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / n_iters_target)
        T = T_start * (T_end / T_start) ** progress

        conv = fast_conv_self(a)
        max_c = conv.max()

        shifted = (conv - max_c) / T
        shifted = np.maximum(shifted, -60.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = 0.8 * T / (np.sqrt(np.mean(np.maximum(v_hat, 0.0))) + eps)
        lr = min(lr, 5.0 / np.sqrt(n))
        a_new = a - lr * m_hat
        a_new = normalize_a(a_new, norm)
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam it={it} ({elapsed:.0f}s): T={T:.3f}, best={best_score:.8f}")

    return best_a, best_score


def pair_cd_polish(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_steps = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        top_lags = np.argsort(conv)[-min(20, size):]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            shift_i = np.zeros(size)
            lo = max(0, i)
            hi = min(size, i + n)
            if hi > lo:
                shift_i[lo:hi] = a[:hi - lo]

            j_set = set()
            for _ in range(5):
                j_set.add(int(rng.integers(n)))
            for kl in top_lags[:10]:
                kl_int = int(kl)
                for di in [-1, 1]:
                    jj = (kl_int + di * i) % (2 * n)
                    if 0 <= jj < n and jj != i:
                        j_set.add(jj)
                    jj2 = (kl_int - di * i + n) % n
                    if jj2 != i:
                        j_set.add(jj2)
            j_set.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_set)[:12]:
                if j == i:
                    continue
                aj = a[j]

                shift_j = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    shift_j[lo:hi] = a[:hi - lo]

                linear_change = 2.0 * (shift_j - shift_i)
                max_d = ai * 0.95
                if max_d < 1e-12:
                    continue
                n_samples = 18
                deltas = np.linspace(0.0, max_d, n_samples)

                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * linear_change
                    if 2 * i < size:
                        new_conv[2 * i] += (-2.0 * ai + d) * d
                    if 2 * j < size:
                        new_conv[2 * j] += (2.0 * aj + d) * d
                    if i + j < size:
                        new_conv[i + j] -= 2.0 * d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize_a(a, norm)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_steps += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(4, n // 80)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.06 * rng.standard_normal(n_perturb))
                a = normalize_a(a, norm)
        else:
            no_improve = 0

        if verbose and total_steps > 0 and total_steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD steps={total_steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    else:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x) * np.cos(np.pi * x)
    a = np.maximum(a, 0.0) + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: Accelerated LP (main workhorse, first round)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = accelerated_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: Peak Shaving n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_shaving(best_seq, t3, rng, verbose=True)
        try_update(a_out, "peak_shave1")

    # Phase 4: Adam smooth optimization
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 130.0)
    print(f"\n=== Phase 4: Adam Smooth n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = adam_smooth_optimize(best_seq, t4, verbose=True)
        try_update(a_out, "adam1")

    # Phase 5: LP Nesterov round 2 (with improved state)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 5: LP Nesterov r2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = accelerated_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 6: Peak shaving round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 6: Peak Shaving r2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = peak_shaving(best_seq, t6, rng, verbose=True)
        try_update(a_out, "peak_shave2")

    # Phase 7: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 7: Pair CD n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 10:
        a_out, _ = pair_cd_polish(best_seq, t7, rng, verbose=True)
        try_update(a_out, "pcd1")

    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Multi-length exploration (budget={remaining:.0f}s) ===")
    if remaining > 60:
        # Try a slightly longer length
        n_long = n_base + 300
        init_long = interpolate_seq(best_seq, n_long)
        t8a = min(remaining * 0.35, 120.0)
        if t8a > 20:
            a_out, _ = accelerated_lp_optimize(init_long, t8a, solver, verbose=False)
            try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            # Try a slightly shorter length (sometimes helps)
            n_short = max(1200, n_base - 200)
            init_short = interpolate_seq(best_seq, n_short)
            t8b = min(remaining * 0.30, 100.0)
            if t8b > 15:
                a_out, _ = accelerated_lp_optimize(init_short, t8b, solver, verbose=False)
                try_update(a_out, f"lp_short{n_short}")

    # Phase 9: Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t9a = min(remaining * 0.4, 60.0)
        if t9a > 10:
            a_out, _ = lse_gradient_descent(best_seq, t9a, n_stages=6, verbose=False)
            try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t9b = min(remaining * 0.5, 50.0)
            if t9b > 8:
                a_out, _ = pair_cd_polish(best_seq, t9b, rng, verbose=False)
                try_update(a_out, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
