import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def get_solver():

    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:

            prob.solve(solver=s)

def fast_conv_self(a):
    """Compute self-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Compute cross-convolution a*b using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def ift_flatten(a0, time_limit, verbose=True):
    """
    Iterative Fourier Transform (IFT) to flatten the power spectrum.
    This reduces the peaks in the autocorrelation by making the spectrum more uniform.
    """
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Multi-stage alpha schedule: start gentle, end aggressive
    n_stages = 15
    alphas = np.linspace(0.95, 0.4, n_stages)
    
    for si, alpha in enumerate(alphas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_iters = max(30, min(300, int(remaining * 4)))
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            A = np.fft.fft(a, N)
            amp = np.abs(A)
            amp_flat = amp ** alpha
            A_new = amp_flat * np.exp(1j * np.angle(A))
            a_new = np.fft.ifft(A_new).real[:n]
            a = np.maximum(a_new, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = best_a.copy()
                break
            a = a / s_new
            
            sc = fast_conv_self(a).max()
            if sc < best_max:
                best_max = sc
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0

    temps = np.geomspace(100.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        n_steps = max(50, min(500, int(remaining * 8)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # Soft-max weights on the autocorrelation
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)

            w = exp_c / exp_c.sum()

            g1 = irfft(np.conj(A) * W, fft_size)[:n]

            if s_new < 1e-15:

            a = a_new

def build_T(a):
    """Build the Toeplitz constraint matrix for the LP."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    """Solve the LP: max sum(g) s.t. T@g <= M, g >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active is not None and 0 < len(active) < size:
        T = build_T(a)
        constraints = [T[active, :] @ b_var <= M]
    else:
        T = build_T(a)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. IFT (Iterative Fourier Transform) spectral flattening for global shape
2. LSE (Log-Sum-Exp) smoothed gradient descent with adaptive temperature
3. LP Nesterov acceleration with adaptive constraint screening
4. Peak-targeted coordinate descent for local refinement
5. Multi-length exploration with interpolation between scales

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def ift_flatten(a0, time_limit, verbose=True):
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    n_stages = 18
    alphas = np.linspace(0.95, 0.3, n_stages)
    for si, alpha in enumerate(alphas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_iters = max(50, min(500, int(remaining * 5)))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            A = np.fft.fft(a, N)
            amp = np.abs(A)
            amp_flat = amp ** alpha
            phase = np.exp(1j * np.angle(A))
            A_new = amp_flat * phase
            a_new = np.fft.ifft(A_new).real[:n]
            a = np.maximum(a_new, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = best_a.copy()
                break
            a = a / s_new
            sc = fast_conv_self(a).max()
            if sc < best_max:
                best_max = sc
                best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    IFT stage {si} alpha={alpha:.2f} ({elapsed:.0f}s): max={best_max:.8f}")

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def make_init(n, rng, strategy=0):
    x = np.linspace(0.0, 1.0, n)
    if strategy == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif strategy == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif strategy == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    elif strategy == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif strategy == 4:
        a = np.sinc(2.0 * (x - 0.5)) ** 2 + 0.05
    elif strategy == 5:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.0
    elif strategy == 6:
        a = 0.3 + 0.5 * np.sin(np.pi * x) ** 0.8 + 0.2 * np.cos(4 * np.pi * x) ** 2
    elif strategy == 7:
        a = np.exp(-4.0 * (x - 0.5) ** 2) * (1.0 + 0.3 * np.sin(6 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        a = np.sin(np.pi * x)
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
