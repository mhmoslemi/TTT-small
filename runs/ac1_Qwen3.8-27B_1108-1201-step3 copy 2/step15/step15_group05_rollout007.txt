def normalize_seq(a, target_sum=None):
    """Normalize sequence to non-negative with given sum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

def subgradient_at_peak(a, k_star, n):
    """Compute subgradient of (a*a)[k_star] w.r.t. a.
    Result: g[j] = 2 * a[k_star - j] if 0 <= k_star - j < n, else 0.
    """
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    for j in range(lo, hi):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    return g


def multi_peak_subgradient(a, conv, n, top_k=5, temperature=0.1):
    """Compute weighted combination of subgradients at top peaks."""
    M = conv.max()
    if M < 1e-15:
        return np.zeros(n)
    # Get top peaks
    top = np.argsort(conv)[-top_k:]
    # Weight by exponential of (conv[k] / M)
    weights = np.exp(-temperature * (M - conv[top]) / max(M * 0.01, 1e-10))
    weights = weights / weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, top):
        g += w * subgradient_at_peak(a, k, n)
    return g


def projected_subgradient_nesterov(a0, time_limit, verbose=False, seed=42):

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    step = 1.0 / np.sqrt(n)
    patience = 0
    max_patience = 200
    iters = 0

    # Track previous subgradient for BB step size
    g_curr = None
    g_prev = None
    x_curr = None
    x_prev = None

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        M_curr = conv_y.max()

        # Compute multi-peak subgradient

        g = multi_peak_subgradient(y, conv_y, n, top_k=top_k, temperature=0.15)

        # Barzilai-Borwein step size
        if g_curr is not None and x_curr is not None:
            s_val = x_curr - x_prev
            y_val = g - g_curr
            ss = np.dot(s_val, s_val)
            sy = np.dot(s_val, y_val)
            if ss > 1e-15:
                if sy > 1e-15:
                    step_bb = ss / sy
                else:
                    step_bb = ss / max(abs(sy), 1e-15) * 0.5
                step_bb = np.clip(step_bb, 0.01 / n, 5.0 / np.sqrt(n))
                # Smooth
                step = 0.7 * step + 0.3 * step_bb
        g_prev = g_curr
        g_curr = g.copy()
        x_prev = x_curr
        x_curr = y.copy()

        # Nesterov momentum
        alpha = 2.0 / (t + 2.0)
        y_proposed = (1.0 - alpha) * y + alpha * y_prev
        y_proposed = np.maximum(y_proposed, 0.0)
        s_y = y_proposed.sum()
        if s_y < 1e-15:
            y_proposed = y.copy()
        else:
            y_proposed = y_proposed / s_y

        # Gradient step
        a_new = y_proposed - step * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        y_prev = y
        y = a_new

            best_a = y.copy()
            patience = 0
        else:

                score_sym = fast_autocorr(a_sym).max()
                if score_sym < best_score:

                    y = a_sym.copy()
                    y_prev = a_sym.copy()

def lp_best_response(a_fixed, M, active_set):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    import cvxpy as cp
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    # Build constraint matrix using FFT-friendly approach
    # (a_fixed * b)[k] = sum_j b[j] * a_fixed[k-j] for valid j
    # For each k in active_set, the row is a shifted version of a_fixed
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        mat[idx, lo:hi] = a_fixed[k - hi + 1:k - lo + 1]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    solver = get_solver()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        if verbose and passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={passes} ({elapsed:.0f}s): best_score={best_score:.10f}")

    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def structured_init(n, rng):
    """Create a structured initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    # Combination of cosines to create a "Barker-like" shape
    a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x) + 0.1 * np.sin(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def propose_candidate():
    """
    Main search function. Uses hybrid Nesterov subgradient + LP + CD
    to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    t_global = time.monotonic()
    time_budget = 1750.0

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, np.random.default_rng(0))
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    # Try symmetrization
    a_sym = 0.5 * (best_seq + best_seq[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s_sym = a_sym.sum()
    if s_sym > 1e-15:
        a_sym = a_sym / s_sym
        sc_sym = evaluate_safe(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.copy()
            print(f"  Symmetrization improved to: {sc_sym:.8f}")

    # Try a few different lengths
    lengths_to_try = sorted(set([len(best_seq), max(1000, len(best_seq) - 200), min(2400, len(best_seq) + 200)]))
    print(f"Lengths to try: {lengths_to_try}")

    candidates = {}
    for n_try in lengths_to_try:
        if n_try == len(best_seq):
            candidates[n_try] = best_seq.copy()
        elif parent is not None:
            candidates[n_try] = interpolate_to_length(parent, n_try)
        else:
            candidates[n_try] = structured_init(n_try, np.random.default_rng(n_try))

    # Evaluate candidates
    for n_try, a_try in candidates.items():
        sc = evaluate_safe(a_try)
        print(f"  Candidate n={n_try}: score={sc:.8f}")

    # Main optimization loop over lengths
    # Allocate time: primary length gets most time
    primary_n = len(best_seq)

    for n_try in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break

        a_cur = candidates[n_try].copy()
        if n_try == primary_n:
            time_fraction = 0.5
        else:
            time_fraction = 0.25

        t_alloc = remaining * time_fraction
        if t_alloc < 30:
            t_alloc = remaining * 0.3

        print(f"\n=== Optimizing n={n_try}, budget={t_alloc:.0f}s ===")

        # Phase A: Nesterov subgradient (bulk of time)
        t_sub = t_alloc * 0.5
        if t_sub > 20:
            a_out, _ = projected_subgradient_nesterov(a_cur, t_sub, verbose=False, seed=n_try)
            try_update(a_out, f"psg_n{n_try}")
            if best_seq is not None and len(best_seq) == n_try:
                a_cur = best_seq.copy()
            else:
                a_cur = a_out

        # Phase B: LP polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_lp = min(t_alloc * 0.3, remaining * 0.4)
        if t_lp > 20:
            a_out, _ = lp_polish(a_cur, t_lp, verbose=True)
            try_update(a_out, f"lp_n{n_try}")
            if best_seq is not None and len(best_seq) == n_try:
                a_cur = best_seq.copy()
            else:
                a_cur = a_out

        # Phase C: Coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        t_cd = min(t_alloc * 0.2, remaining * 0.3)
        if t_cd > 10:
            a_out, _ = coordinate_descent_fast(a_cur, t_cd, seed=n_try)
            try_update(a_out, f"cd_n{n_try}")

    # Final: multi-restart with perturbations on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Final restarts, remaining={remaining:.0f}s ===")
    if remaining > 60 and best_seq is not None:
        n_best = len(best_seq)
        rng_final = np.random.default_rng(12345)
        n_restarts = min(3, int(remaining // 60))
        t_each = (remaining - 30) / max(n_restarts, 1)

        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_r = min(t_each, remaining - 20)
            if t_r < 15:
                break

            # Perturb
            scale = 0.01 + 0.005 * r
            a_pert = best_seq * (1.0 + scale * rng_final.standard_normal(n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()

            a_out, _ = projected_subgradient_nesterov(a_pert, t_r * 0.6, verbose=False, seed=1000 + r)
            try_update(a_out, f"restart{r}_n{n_best}")

            # Quick CD polish
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20 and best_seq is not None:
                t_cp = min(remaining * 0.3, 30.0)
                a_out, _ = coordinate_descent_fast(best_seq, t_cp, seed=2000 + r)
                try_update(a_out, f"cd_restart{r}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
