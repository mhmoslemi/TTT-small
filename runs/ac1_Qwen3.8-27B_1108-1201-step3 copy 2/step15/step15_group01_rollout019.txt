def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        # Adaptive window based on spread of peaks
        peak_thresh = 0.9 * conv.max()
        n_peaks = (conv >= peak_thresh).sum()
        window = max(10, min(80, int(n / max(1, n_peaks) * 2)))
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue
            
            for offset in range(1, window + 1):
                j1 = (i + offset) % n
                j2 = (i - offset) % n
                for j in [j1, j2]:
                    if j == i or a[i] < norm * 1e-12:
                        continue
                    
                    # Quick test with fraction of a[i]
                    for frac in [0.5, 0.25]:
                        d = a[i] * frac
                        a_test = a.copy()
                        a_test[i] -= d
                        a_test[j] += d
                        a_test = normalize_sum(a_test, norm)
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < conv.max() - 1e-13:
                            a = a_test
                            conv = conv_test
                            if conv.max() < best_score:
                                best_score = conv.max()
                                best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            break
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} win={window} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def spectral_init(n, rng, mode='random_phase'):
    """Generate a diverse initial sequence using frequency domain design."""
    if mode == 'random_phase':
        phases = rng.uniform(0, 2 * np.pi, n)
        A = np.ones(n) + 0.5 * np.exp(1j * phases)
        a = np.abs(ifft(A)) ** 2
    elif mode == 'chirp':
        t = np.arange(n)
        A = np.exp(1j * np.pi * t * t / n)
        a = np.abs(ifft(A)) ** 2
    elif mode == 'multi_sine':
        n_freqs = rng.integers(3, 8)
        freqs = rng.integers(1, n // 2, n_freqs)
        phases = rng.uniform(0, 2 * np.pi, n_freqs)
        A = np.ones(n)
        for f, p in zip(freqs, phases):
            A[f] += np.exp(1j * p)
            A[n - f] += np.exp(-1j * p)
        a = np.abs(ifft(A)) ** 2
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function implementing accelerated mirror descent, LP, and pair CD."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    print(f"\nStarting with n={n_base}, score={best_score:.6f}")
    
    # === Phase 1: Mirror Descent with Acceleration (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    n1 = n_base
    print(f"\n=== Phase 1: Mirror Descent (Accel) n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        norm1 = np.sqrt(2.0 * n1)
        a_init = best_seq.copy()
        a_init = normalize_sum(a_init, norm1)
        a_out, _ = mirror_descent_accelerated(a_init, t1, verbose=True, init_temp=8.0, final_temp=0.2)
        do_try_update(a_out, f"md_accel_n{n1}")
    
    # === Phase 2: LP Best-Response (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: LP Nesterov n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 30:
        norm_c = np.sqrt(2.0 * n_c)
        a_init = normalize_sum(best_seq, norm_c)
        a_out, _ = nesterov_lp(a_init, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_c}")
    
    # === Phase 3: Pair CD (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: Pair CD n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 15:
        norm_c = np.sqrt(2.0 * n_c)
        a_init = normalize_sum(best_seq, norm_c)
        a_out, _ = pair_cd_adaptive(a_init, t3, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_c}")
    
    # === Phase 4: Mirror Descent round 2 (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: MD round 2 n={n_c} (budget={t4:.0f}s) ===")
    if t4 > 20:
        norm_c = np.sqrt(2.0 * n_c)
        a_init = normalize_sum(best_seq, norm_c)
        a_out, _ = mirror_descent_accelerated(a_init, t4, verbose=False, init_temp=3.0, final_temp=0.1)
        do_try_update(a_out, f"md2_n{n_c}")
    
    # === Phase 5: Multi-restart with spectral inits (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Multi-restart (budget={t5:.0f}s) ===")
    if t5 > 40:
        n_restarts = 3
        t_per_restart = t5 / n_restarts
        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget:
                break
            if t_per_restart < 15:
                break
            n_r = n_base
            modes = ['random_phase', 'multi_sine', 'chirp']
            mode = modes[r % len(modes)]
            a_r = spectral_init(n_r, rng, mode=mode)
            norm_r = np.sqrt(2.0 * n_r)
            a_r = normalize_sum(a_r, norm_r)
            print(f"  Restart {r}: mode={mode}")
            a_out, _ = mirror_descent_accelerated(a_r, t_per_restart * 0.6, verbose=False, init_temp=5.0, final_temp=0.3)
            do_try_update(a_out, f"restart_{mode}")
            if best_seq is not None and len(best_seq) == n_r:
                remaining_r = time_budget - (time.monotonic() - t_global)
                if remaining_r > 10:
                    a_out2, _ = pair_cd_adaptive(normalize_sum(best_seq, norm_r), min(remaining_r * 0.3, 30), rng, verbose=False)
                    do_try_update(a_out2, f"restart_{mode}_pcd")
    
    # === Phase 6: Final polishing (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        n_c = len(best_seq)
        t6 = min(remaining * 0.5, 60.0)
        print(f"\n=== Phase 6: Final polish n={n_c} (budget={t6:.0f}s) ===")
        norm_c = np.sqrt(2.0 * n_c)
        a_init = normalize_sum(best_seq, norm_c)
        a_out, _ = pair_cd_adaptive(a_init, t6, rng, verbose=False)
        do_try_update(a_out, "final_pcd")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15 and best_seq is not None:
            norm_c = np.sqrt(2.0 * len(best_seq))
            a_out, _ = mirror_descent_accelerated(normalize_sum(best_seq, norm_c), min(remaining * 0.5, 30), verbose=False, init_temp=2.0, final_temp=0.05)
            do_try_update(a_out, "final_md")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
