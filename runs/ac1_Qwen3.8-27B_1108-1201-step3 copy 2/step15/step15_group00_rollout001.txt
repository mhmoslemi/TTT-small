    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def log_domain_subgradient(a, ac, top_k=1, beta=10.0):
    """
    Compute subgradient of max_k C[k] (or soft version) with respect to x = log(a).
    For the exact subgradient at k*, the gradient is a[i]*a[k*-i].
    For the soft version, we use weighted sum over top-k lags.
    """
    n = len(a)
    if top_k <= 1:
        # Exact subgradient: use the single maximum lag
        k_star = int(np.argmax(ac))
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = a[i] * a[j]
        return g
    else:
        # Soft subgradient: weighted sum over top-k lags
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += wi * a[idx] * a[j_idx]
        return g

def log_mirror_descent(a0, time_limit, top_k=1, beta=10.0, c0=1.0, 
                       verbose=True, nesterov=True, restart_interval=0):
    """
    Log-domain mirror descent (equivalent to multiplicative updates in a-space).
    Minimizes max_k C[k] subject to sum(a) = 1, a >= 0.
    """
    n = len(a0)
    a = norm_sum1(a0)
    x = np.log(np.maximum(a, 1e-30))  # log domain
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if nesterov:
        x_prev = x.copy()
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if nesterov and t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        
        # Convert to a-space for computation
        y_shifted = y - y.max()
        a_y = np.exp(y_shifted)
        a_y = a_y / a_y.sum()
        
        # Compute autocorrelation
        ac = fast_autocorr(a_y)
        M_curr = ac.max()
        
        # Compute subgradient (in log-domain)
        g = log_domain_subgradient(a_y, ac, top_k=top_k, beta=beta)
        
        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Update in log-domain
        x_new = y - eta * g
        
        # Project: ensure sum(exp(x_new)) = 1
        # This means x_new should be shifted so that sum(exp(x_new)) = 1
        x_new_shifted = x_new - x_new.max()
        S = np.exp(x_new_shifted).sum()
        x_new = x_new - np.log(S)
        
        if nesterov:
            x_prev = x
        x = x_new
        
        # Evaluate in a-space
        x_shifted = x - x.max()
        a_curr = np.exp(x_shifted)
        a_curr = a_curr / a_curr.sum()
        
        M_new = fast_autocorr(a_curr).max()

    return best_a, best_m
