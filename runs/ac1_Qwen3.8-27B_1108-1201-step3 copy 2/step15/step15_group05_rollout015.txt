        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        diff = y_new - y
        err_vectors.append(diff)
        if len(err_vectors) > 20:
            err_vectors.pop(0)
        if len(err_vectors) >= 5:
            err_vec_ema = np.mean(err_vectors, axis=0)
        if err_vec_ema is not None and t % 10 == 0:
            y_ema = y + 0.5 * err_vec_ema
            y_ema = np.maximum(y_ema, 0.0)
            s_ema = y_ema.sum()
            if s_ema > 1e-15:
                y_ema = y_ema * norm_val / s_ema
                score_ema = fast_autocorr(y_ema).max()
                if score_ema < best_score:
                    best_score = score_ema
                    best_a = y_ema.copy()
                    stall_count = 0

        if stall_count > 60:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            last_active_set = None
            use_full_set = False
            stall_count = 0
            err_vectors = []
            err_vec_ema = None
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=False):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n * norm_val
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    step = 2.0 / np.sqrt(n)
    patience = 0
    max_patience = 300
    prev_grad = None
    prev_a = None
    n_passed = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        n_peaks = max(3, min(8, n // 150))
        top_indices = np.argsort(conv)[-n_peaks:]
        max_val = conv[top_indices[0]]
        weights = np.exp(-8.0 * (max_val - conv[top_indices]) / max(max_val * 0.02, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                grad[j] += w * 2.0 * a[k - j]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad_unit = grad / grad_norm

        if prev_grad is not None and prev_a is not None:
            y_bb = grad_unit - prev_grad
            s_bb = a - prev_a
            denom = np.dot(s_bb, y_bb)
            if abs(denom) > 1e-15:
                step_bb = np.dot(s_bb, s_bb) / denom
                step_bb = np.clip(step_bb, 0.1 * step, 5.0 * step)
                step = 0.7 * step + 0.3 * step_bb

        prev_grad = grad_unit.copy()
        prev_a = a.copy()

        a_trial = project_simplex(a - step * grad_unit, norm_val)
        score_trial = fast_autocorr(a_trial).max()

        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            n_passed = 0
            step = min(step * 1.01, 3.0)
        else:
            patience += 1
            n_passed += 1
            step *= 0.97
            if step < 1e-12:
                break
            half_trial = project_simplex(a - 0.5 * step * grad_unit, norm_val)
            score_half = fast_autocorr(half_trial).max()
            if score_half < best_score - 1e-14:
                a = half_trial
                best_score = score_half
                best_a = half_trial.copy()
                patience = 0
                n_passed = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def fast_coordinate_descent(a0, time_limit, rng, verbose=False):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n * norm_val
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        order = rng.permutation(n)
        conv = fast_autocorr(a)
        current_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            lo = max(0, 2 * j - n + 1)
            hi = min(size - 1, 2 * j)
            c = conv[lo:hi + 1]
            d = a[(2 * j - lo): hi + 1 - (2 * j - lo)]
            vals = c - 2.0 * aj * d + aj * aj
            idx_max = np.argmax(vals)
            k_star = lo + idx_max
            peak_others = vals[idx_max]
            d_other = d[idx_max]

            s_others = a.sum() - aj
            if s_others < 1e-15:
                continue

            if d_other > 1e-15 and k_star != 2 * j:
                x_zero = peak_others
                base = s_others * s_others
                for x in [aj * 0.5, aj * 1.5]:
                    new_val = x * x + 2.0 * x * d_other + peak_others
                    s_new = s_others + x
                    obj = new_val / (s_new * s_new)
                    if obj < 1.0 / (4.0 * n) * 2:
                        a[j] = x
                        s_new2 = a.sum()
                        a = a * norm_val / s_new2
                        new_conv = fast_autocorr(a)
                        new_sc = new_conv.max()
                        if new_sc < best_score - 1e-13:
                            best_score = new_sc
                            best_a = a.copy()
                            improved = True
                        else:
                            a[j] = aj
                            s_new3 = a.sum()
                            a = a * norm_val / s_new3
                        break

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pairwise_swaps(a0, time_limit, rng, verbose=False):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    patience = 0
    max_patience = 200

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        ai, aj = a[i], a[j]
        a[i], a[j] = aj, ai
        s_new = a.sum()
        a = a * norm_val / s_new
        score = fast_autocorr(a).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            a[i], a[j] = ai, aj
            s_back = a.sum()
            a = a * norm_val / s_back
            patience += 1

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Swap t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def spectral_init(n, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x + 0.5)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
