Given current `a` and bound `M = max(a*a)`:

        v_hat = v / (1.0 - beta2 ** t_adam)

        lr = lr_base / (1.0 + 0.1 * t_adam)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_a(a_new, norm_val)
        a = a_new
        T_curr = max(T_curr * T_decay, min_T)

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and t_adam % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE t={t_adam} T={T_curr:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_solve(a, M, active_constraints, solver):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    constraints = []
    for k in active_constraints:
        # (a*b)[k] = sum_{i} a[k-i] * b[i] for valid i
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            coeffs = a[k - np.arange(lo, hi + 1)]
            constraints.append(coeffs @ b_var[lo:hi+1] <= M)

    if not constraints:
        return None

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lp_best_response_full(a, M, solver, max_iter=50):
    """LP best-response with full constraint generation."""
    n = len(a)
    size = 2 * n - 1

    # Start with constraints at current peaks
    conv_a = fast_conv_self(a)
    peak_idx = np.where(conv_a >= M * 0.95)[0]
    if len(peak_idx) < 3:
        peak_idx = np.where(conv_a >= M * 0.8)[0]
    if len(peak_idx) < 1:
        peak_idx = np.array([np.argmax(conv_a)])

    active = list(peak_idx)
    b = None

    for it in range(max_iter):
        b = lp_solve(a, M, active, solver)
        if b is None:
            # Fallback: use all constraints
            all_idx = np.arange(size)
            b = lp_solve(a, M, all_idx, solver)
            break

        # Check for violated constraints
        conv_ab = fast_conv_cross(a, b)
        violated = np.where(conv_ab > M * 1.001)[0]

        if len(violated) == 0:
            break

        # Add violated constraints
        active = list(set(active) | set(violated.tolist()))
        if len(active) > size:
            active = list(np.arange(size))

    if b is None:
        # Fallback: uniform
        b = np.ones(n) / n
    return b


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive step size."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm_val / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()  # Nesterov lookahead
    no_improve = 0
    stagnation_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # LP best response with constraint generation
        b = lp_best_response_full(y, M, solver, max_iter=30)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum

        # Adaptive step size: try Nesterov step, then backtrack
        alpha_base = 2.0 / (t + 2.0)
        # Scale up for early iterations
        if t < 10:
            alpha_base = min(alpha_base * 2.0, 0.5)

        # Compute Nesterov momentum
        if t == 1:
            a_new = g.copy()
        else:
            y_new = (1.0 - alpha_base) * y + alpha_base * g
            y_new = np.maximum(y_new, 0.0)
            y_new = normalize_a(y_new, norm_val)
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_a(a_new, norm_val)
            y = y_new

        sc_new = fast_conv_self(a_new).max()

        if sc_new >= best_score * (1.0 + 1e-6):
            # No improvement - try smaller step
            for retry_alpha in [alpha_base * 0.5, alpha_base * 0.25, alpha_base * 0.1]:
                if t == 1:
                    a_try = g * retry_alpha + a * (1.0 - retry_alpha)
                else:
                    y_try = (1.0 - retry_alpha) * y + retry_alpha * g
                    y_try = np.maximum(y_try, 0.0)
                    y_try = normalize_a(y_try, norm_val)
                    a_try = y_try.copy()
                a_try = np.maximum(a_try, 0.0)
                a_try = normalize_a(a_try, norm_val)
                sc_try = fast_conv_self(a_try).max()
                if sc_try < best_score:
                    a_new = a_try
                    sc_new = sc_try
                    y = y_try.copy()
                    break
            else:
                # No improvement at any step size
                a_new = a.copy()
                sc_new = best_score
                no_improve += 1
                stagnation_count += 1

                if stagnation_count > 20 and stagnation_count % 10 == 0:
                    # Add small random perturbation
                    n_perturb = max(3, n // 100)
                    idx_p = np.random.default_rng(t).choice(n, size=n_perturb, replace=False)
                    a[idx_p] *= (1.0 + 0.1 * np.random.default_rng(t + 1).standard_normal(n_perturb))
                    a = np.maximum(a, 0.0)
                    a = normalize_a(a, norm_val)
                    y = a.copy()

                if no_improve > 200:
                    break
        else:
            no_improve = 0
            stagnation_count = 0
            a_best_improve = sc_new
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, no_imp={no_improve}")

    return best_a, best_score


def peak_directed_cd(a0, time_limit, verbose=True):
    """
    Peak-directed coordinate descent: identifies pairs that contribute most
    to the peak and redistributes mass to reduce it.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    rng = np.random.default_rng(12345)
    total_improvements = 0
    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        avg = conv.sum() / size

        # Find peaks (above average)
        peak_lags = np.where(conv > avg)[0]
        if len(peak_lags) == 0:
            break

        # For each peak lag, find which indices contribute most
        # (a*a)[k] = sum_i a[i]*a[k-i], contribution of pair (i, k-i) is a[i]*a[k-i]
        # To reduce peak at lag k, reduce a[i] or a[k-i] for large a[i]*a[k-i]

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm_val * 1e-8:
                continue

            # Find which peak lags are affected by a[i]
            # a[i] contributes to (a*a)[k] for k in [i, i+n-1] (as a[i]*a[k-i])
            # and for k in [0, i] (as a[k-i+i]=a[k]*a[i], i.e., as the second factor)
            affected_lags_lo = max(0, i - n + 1)
            affected_lags_hi = min(size - 1, i + n - 1)

            # Find peaks among affected lags
            mask = (np.arange(size) >= affected_lags_lo) & (np.arange(size) <= affected_lags_hi)
            affected_convs = conv[mask]
            peak_positions = np.where(affected_convs > avg)[0] + affected_lags_lo

            if len(peak_positions) == 0:
                continue

            # Try moving mass from i to various j
            # Best j: one that reduces peak lags the most
            best_j = -1
            best_new_max = cur_max
            best_delta = 0.0

            # Candidate j's: those that DON'T contribute to peak lags
            # j contributes to (a*a)[k] for k in [j, j+n-1] and [0, j]
            # We want j such that [j, j+n-1] intersects peak_lags minimally
            n_candidates = min(30, n)
            j_candidates = set()
            for _ in range(n_candidates):
                j_candidates.add(int(rng.integers(n)))
            # Also add strategic candidates
            for pk in peak_lags[:5]:
                # j that's far from pk (so a[j] doesn't add to lag pk)
                j_far = (pk + n // 2) % n
                j_candidates.add(j_far)
            j_candidates.discard(i)

            for j in list(j_candidates)[:20]:
                if j == i:
                    continue
                aj = a[j]

                # Compute the change in convolution if we move delta from i to j
                # delta_a = a with a[i]-=delta, a[j]+=delta
                # Change in (a*a)[k]:
                # = delta * (a[k-i] - a[k-j]) + delta * (a[i-k] ... ) 
                # More precisely:
                # new (a*a)[k] = old (a*a)[k] + delta*(a[k-j] - a[k-i]) + delta*(a[j-k+... ]) 
                # Let's compute the gradient effect:
                # d(a*a)[k]/da[i] = 2*a[k-i] (if valid) [from our earlier derivation, factor of 2]
                # Wait, d(a*a)[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n
                # But that's for the gradient of sum_k w[k]*(a*a)[k]
                # For a single k: d(a*a)[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[k-i] (from symmetric term)
                # Actually (a*a)[k] = sum_j a[j]*a[k-j]
                # d/da[i] = a[k-i] (when j=i) + a[k-i] (when k-j=i, i.e., j=k-i)
                # = 2*a[k-i] if 0<=k-i<n, else 0
                # But wait, when k=2i, j=i and k-j=i are the SAME term, so it's a[i]^2, d/da[i] = 2*a[i]. Correct.

                # So moving delta from i to j changes:
                # (a*a)[k] by: -delta*2*a[k-i] (removing from i) + delta*2*a[k-j] (adding to j)
                # Plus second-order: +delta^2 for lag 2i (removing) and +delta^2 for lag 2j (adding)
                # And cross terms: the interaction between the modified positions

                # For small delta, the first-order change is:
                # delta_k = 2*delta*(a[k-j] - a[k-i]) [where a[m]=0 for m outside [0,n-1]]

                # We want to find delta that minimizes max(conv + delta_k)
                # For simplicity, just check a few delta values

                # Precompute the change vector
                # change[k] = 2*delta*(a_valid[k-j] - a_valid[k-i])
                # where a_valid[m] = a[m] if 0<=m<n else 0

                lo_i = max(0, i)
                hi_i = min(size, i + n)
                lo_j = max(0, j)
                hi_j = min(size, j + n)

                # change[k] for k in [0, size-1]:
                # = 2*delta * (a[k-j] if 0<=k-j<n else 0 - a[k-i] if 0<=k-i<n else 0)

                n_delta_samples = 15
                max_frac = 0.9
                for d_frac in np.linspace(0.02, max_frac, n_delta_samples):
                    delta = ai * d_frac
                    if delta > ai:
                        continue

                    # Compute new max (approximate, first order)
                    # conv_new[k] = conv[k] + 2*delta*(a[k-j]_valid - a[k-i]_valid)
                    # + second order at lags 2i and 2j

                    # Efficient: only check around peaks
                    check_lags = peak_lags[:15]
                    # Also check lags 2i and 2j (second order effects)
                    extra = [2*i % size, 2*j % size]
                    for e in extra:
                        if 0 <= e < size and e not in check_lags:
                            check_lags = np.append(check_lags, e)

                    max_new = 0.0
                    for k in check_lags:
                        term_i = a[k - i] if 0 <= k - i < n else 0.0
                        term_j = a[k - j] if 0 <= k - j < n else 0.0
                        new_val = conv[k] + 2.0 * delta * (term_j - term_i)
                        # Second order at diagonals
                        if k == 2 * i and 2 * i < size:
                            new_val = conv[k] - 2.0 * ai * delta + delta * delta + 2.0 * delta * (term_j - 2.0 * ai + delta)
                        elif k == 2 * j and 2 * j < size:
                            new_val = conv[k] + 2.0 * aj * delta + delta * delta
                        if new_val > max_new:
                            max_new = new_val

                    # Also check the overall max (conservative)
                    if max_new < best_new_max:
                        best_new_max = max_new
                        best_j = j
                        best_delta = delta

            if best_j >= 0 and best_delta > 0:
                # Verify and apply
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                a = normalize_a(a, norm_val)
                new_max = fast_conv_self(a).max()

                if new_max < best_score * (1.0 - 1e-8):
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1
                else:
                    # Revert
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = normalize_a(a, norm_val)

        if not pass_improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 8:
                # Perturb to escape
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                a = normalize_a(a, norm_val)
                no_improve_rounds = 0
                if total_improvements == 0 and no_improve_rounds > 0:
                    pass
        else:
            no_improve_rounds = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = int(rng.integers(1, 5))
        f2 = int(rng.integers(5, 15))
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-5.0 * x) + 0.4
    elif choice == 7:
        n_peaks = int(rng.integers(3, 8))
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 8:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    else:
        a = np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * 2 * x))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a


def vegas_multiplicative(a0, time_limit, verbose=True):
    """Vegas/multiplicative update method."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 20
    etas = np.geomspace(0.08, 0.001, n_stages)

    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(60, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break

            # Pressure: how much each coordinate contributes to peaks
            w = np.maximum(conv - M * 0.85, 0.0)
            ww = w / (w.sum() + 1e-15)

            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            pressure = irfft(np.conj(A) * W, fft_size)[:n]

            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-12)
            a = a * norm_val / a.sum()

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 5 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.5f}: best={best_score:.8f}")

    return best_a, best_score
