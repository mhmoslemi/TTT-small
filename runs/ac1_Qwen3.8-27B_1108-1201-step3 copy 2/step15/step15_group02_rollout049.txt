def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def eval_safe(a):

def normalize_to_sum(a, target_sum):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s

def get_solver():
    for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def lp_best_response(a, M, active_set=None):
    """Solve LP: max sum(b) s.t. (a*b)(k) <= M for k in active_set, b >= 0"""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and 0 < len(active_set) < size:
        # Build constraint matrix A @ b <= M for k in active_set
        rows = []
        for k in active_set:
            # (a*b)(k) = sum_i a[i] * b[k-i] where k-i is valid
            # b[j] contributes to (a*b)(k) when j = k-i, i.e., i = k-j
            # So coefficient of b[j] in row k is a[k-j] if 0 <= k-j < n
            row = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    row[j] = a[idx]
            rows.append(row)
        T = np.array(rows)
        constraints = [T @ b_var <= M]
    else:
        # Full constraint matrix (expensive for large n)
        T = np.zeros((size, n))
        for m in range(n):
            T[m:m+n, m] = a
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=SOLVER, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def lp_optimize(a0, time_limit, verbose=True):
    """LP-based optimization with Nesterov acceleration and adaptive active sets."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive active set: start with top constraints, expand if needed
        # Use a threshold that tightens over time
        frac = 0.5 + 0.45 * min(1.0, t / 200.0)  # goes from 0.5 to 0.95
        active_set = np.where(conv_y > M * frac)[0]
        
        if len(active_set) > 800:
            active_set = np.where(conv_y > M * 0.92)[0]
        if len(active_set) < 5:
            active_set = np.where(conv_y > M * 0.5)[0]
        
        b = lp_best_response(y, M, active_set)
        
        if b is None:
            b = lp_best_response(y, M, None)
        
        if b is None:
            continue
        
        # Verify and fix with constraint generation
        if active_set is not None and len(active_set) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.01)[0]
            if len(violated) > 0:
                active_set = np.union1d(active_set, violated)
                if len(active_set) < size:
                    b2 = lp_best_response(y, M, active_set)
                    if b2 is not None:
                        b = b2
                else:
                    b2 = lp_best_response(y, M, None)
                    if b2 is not None:
                        b = b2
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): M={M:.6f}, best={best_score:.8f}")
    
    return best_a, best_score

def lse_adam(a0, time_limit, verbose=False):
    """LSE-smoothed max with Adam optimizer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_count = 0
    
    n_stages = 15
    temps = np.geomspace(50.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(100, int(remaining * 8)))
        lr = 0.3 / np.sqrt(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_count += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -60.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_count)
            v_hat = v / (1.0 - beta2 ** t_count)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=False):
    """Pair coordinate descent for local polish."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find j that reduces max convolution when moving mass from i to j
            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            
            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)
            
            best_v = cur_max
            best_delta = 0.0
            best_j = -1
            
            for j in list(j_candidates)[:15]:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute delta in convolution from moving mass d from i to j
                g = np.zeros(size)
                
                # Adding a[j] + d at position j
                # Self-conv change: +d*(a shifted by j) + d*(a shifted by (size-1-j)) + 2*aj*d at appropriate positions
                # Simplified: compute the effect
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi-lo]
                
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                
                # Removing d from position i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3-lo3]
                
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                d_max = ai * 0.7
                if d_max < 1e-15:
                    continue
                deltas = np.linspace(0.02, d_max, 15)
                
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*d + d*d
                    if 2 * j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*d + d*d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a * norm / a.sum()
                new_max = fast_conv_self(a).max()
                if new_max < best_score:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                # Perturb
                n_p = max(3, n // 100)
                idx_p = rng.choice(n, size=n_p, replace=False)
                a[idx_p] *= np.exp(0.1 * rng.standard_normal(n_p))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                if no_improve >= 15:
                    break
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse initial structures."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2*np.pi*f1*x) + 0.2 * np.cos(2*np.pi*f2*x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30*(x-c1)**2) + np.exp(-30*(x-c2)**2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0*x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.08)
            a += np.exp(-(x-c)**2 / (2*width**2))
    elif choice == 7:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 8:
        a = 1.0 + 0.3*np.sin(2*np.pi*3*x) + 0.2*np.sin(2*np.pi*7*x)
    else:
        a = 0.5 + 0.3*np.cos(2*np.pi*x) + 0.2*np.cos(4*np.pi*x)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 7:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 8:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    elif choice == 9:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 10:
        a = np.exp(-8.0 * x) + 0.3
    else:
        a = 0.4 + 0.4 * np.sin(np.pi * x) + 0.2 * np.sin(2 * np.pi * x)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.008 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a.copy()
    return a_sym / s
