def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), size)
    B = rfft(np.asarray(b, dtype=np.float64), size)
    return irfft(A * B, size)[:n]


def _fast_autocorr(a):
    """Compute convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _evaluate_local(a):
    """Evaluate the objective locally (fast version)."""

    if s < 0.01:
        return float('inf')
    conv = _fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def _build_conv_matrix(a, indices):
    """Build the constraint matrix T where T[i,j] gives contribution of b[j] to lag indices[i]."""
    n = len(a)
    size = 2 * n - 1
    K = len(indices)
    T = np.zeros((K, n))
    for idx, k in enumerate(indices):
        k = int(k)
        # (a*b)[k] = sum_j a[k-j] * b[j] for valid j

        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            # T[idx, j] = a[k-j]
            for j in range(j_lo, j_hi + 1):
                T[idx, j] = a[k - j]
    return T

def _nesterov_lp_optimize(a0, time_limit, solver, K=80, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t_start = time.monotonic()
    
    y = a.copy()
    momentum_t = 1.0
    n_iters = 0
    patience = 0
    restart_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t_start > time_limit:
            break
        
        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = _lp_best_response(y, M_curr, K=K, solver=solver)
        if b is None:
            patience += 5
            if patience > 50:
                # Reset
                y = best_a.copy()
                momentum_t = 1.0
                patience = 0
                restart_count += 1
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        g = b * norm / b_sum
        n_iters += 1
        stuck_reset = False
        
        # Nesterov update
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            stuck_reset = True
        else:
            y_new = y_new * norm / s_y
        
        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        momentum_t += 1.0
        
        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        
        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 100 and n_iters % 50 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t_start
                print(f"    NLP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")
    
    return best_a, best_score


def _subgradient_optimize(a0, time_limit, top_k=30, verbose=True):
    """Subgradient method with backtracking line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_obj = _evaluate_local(a)
    t_start = time.monotonic()
    
    eta = 0.1
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(2024)
    
    size = 2 * n - 1
    
    for it in range(1000000):
        if time.monotonic() - t_start > time_limit:
            break
        
        conv = _fast_autocorr(a)
        max_val = conv.max()
        
        # Get active lags
        k_actual = min(top_k, size)
        active_idx = np.argsort(conv)[-k_actual:]
        
        # Compute subgradient (average of gradients of active functions)
        grad = np.zeros(n)
        for k in active_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_hi >= j_lo:
                grad[j_lo:j_hi+1] += a[k - j_hi:k - j_lo+1][::-1]
        grad *= 2.0 / len(active_idx)
        
        # Projected gradient step on simplex (or just non-negative + normalize)
        # Try multiple step sizes
        best_step = 0
        best_new_obj = best_obj
        
        for factor in [0.3, 0.5, 0.7, 1.0]:
            step = eta * factor
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = _evaluate_local(a_try)
            if obj_try < best_new_obj:
                best_new_obj = obj_try
                best_step = step
        
        if best_step > 0:
            a = a - best_step * grad
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
        
        n_iters += 1
        
        obj = _evaluate_local(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                eta *= 0.95
        
        if patience > 200:
            # Perturb
            noise = rng.laplace(0, 0.01, n)
            a_pert = a + noise / n
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert / s_pert
            patience = 0
            eta = 0.1
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    SG iter={n_iters} ({elapsed:.0f}s): best={best_obj:.8f}, eta={eta:.4f}")
    
    return best_a, best_obj


def _coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with line search, focusing on active lags."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_obj = _evaluate_local(a)
    t_start = time.monotonic()
    rng = np.random.default_rng(42)
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t_start < time_limit:
        conv = _fast_autocorr(a)
        max_val = conv.max()
        
        # Get active lags and their contributing indices
        k_actual = min(20, size)
        active_idx = np.argsort(conv)[-k_actual:]
        
        # Find which coordinates are most "important" (appear in active lags)
        importance = np.zeros(n)
        for k in active_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            importance[j_lo:j_hi+1] += conv[k]
        
        # Process coordinates in order of importance
        order = np.argsort(-importance)
        
        improved = False
        for pass_count in range(3):
            for i in order:
                if time.monotonic() - t_start > time_limit:
                    break
                if a[i] < 1e-12:
                    continue
                
                # Try moving mass from i to a few nearby positions
                for off in [1, -1, 2, -2, 3, -3]:
                    j = (i + off) % n
                    if j == i:
                        continue
                    for frac in [0.1, 0.3, 0.5, 0.7]:
                        d = a[i] * frac
                        if d < 1e-15:
                            continue
                        a[i] -= d
                        a[j] += d
                        obj = _evaluate_local(a)
                        if obj < best_obj - 1e-12:
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                        else:
                            a[i] += d
                            a[j] -= d
                        if obj < best_obj + 1e-9:  # keep if close
                            # actually revert if not improving
                            if obj >= best_obj:
                                a[i] += d
                                a[j] -= d
                            else:
                                # already applied
                                pass
                        # Actually let me simplify: only keep if strictly better
                        # (already handled above)
        
        if not improved:
            no_improve += 1
            if no_improve >= 10:
                # Perturb
                noise = rng.normal(0, 0.002, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert / s_pert
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and (no_improve % 20 == 0 or time.monotonic() - t_start > time_limit * 0.5):
            elapsed = time.monotonic() - t_start
            print(f"    CD ({elapsed:.0f}s): best={best_obj:.8f}, no_improve={no_improve}")
    
    return best_a, best_obj


def _make_raised_cosine(n):
    """Create a raised cosine (sin^2) shaped sequence."""
    x = np.arange(n)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.4):
    """Create a Gaussian-shaped sequence."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _evaluate_safe(a):
    """Safely evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
