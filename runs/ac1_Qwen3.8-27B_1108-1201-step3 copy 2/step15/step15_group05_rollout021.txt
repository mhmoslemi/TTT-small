Let me also think about whether there's a better way to compute the gradient.

For the subgradient at the maximizing lag k*:
grad[j] = 2 * a[k*-j] (if 0 <= k*-j < n), else 0

This is just a shifted and possibly reversed copy of a! So computing the gradient is O(n).

But for the softmax-smoothed version, the gradient is:
grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)

This is a weighted sum of shifts of a, which is a convolution. Can be done in O(n log n) with FFT.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed (softmax) Nesterov gradient descent - fast main optimizer
2. Subgradient polishing at active peaks
3. LP best-response for final accuracy
4. Multi-length exploration with diverse initializations
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

_time_start = None

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max() / (s * s)

def normalize_target(a, target_sum):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * target_sum / s

def proj_simplex_nonneg(x, target_sum):
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.full(len(x), target_sum / len(x))
    return x * target_sum / s

def smoothed_gradient(a, conv, T):
    """
    Compute gradient of softmax-smoothed max of conv w.r.t. a.
    grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j in [0, n-1]
    Using FFT for efficiency.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Softmax weights
    conv_max = conv.max()
    exp_conv = np.exp(T * (conv - conv_max))
    w = exp_conv / exp_conv.sum()
    
    # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * w[j+i]
    # This is a cross-correlation computable via FFT
    # Let's define: for j in [0, n-1], grad[j]/2 = sum_{i=0}^{n-1} a[i] * w[j+i]
    # = sum_{i=0}^{n-1} a[i] * w[j+i]
    
    # Using convolution: define w_rev = w[::-1] (length 2n-1)
    # Then sum_i a[i] * w[j+i] = sum_i a[i] * w_rev[2n-2-j-i]
    # = (a * w_rev)[2n-2-j]  (linear convolution, index 2n-2-j)
    
    w_rev = w[::-1]
    conv_len = n + (2*n - 1) - 1  # 3n - 2
    fft_sz = 1 << (conv_len - 1).bit_length()
    A_f = rfft(a, fft_sz)
    WR_f = rfft(w_rev, fft_sz)
    conv_res = irfft(A_f * WR_f, fft_sz)[:conv_len]
    
    # We need elements at positions 2n-2-j for j=0,...,n-1
    # i.e., positions 2n-2, 2n-3, ..., n-1
    corr = conv_res[n-1:2*n-1][::-1]  # positions n-1, n, ..., 2n-2, reversed to get 2n-2, ..., n-1
    
    # Wait, let me be more careful.
    # (a * w_rev)[m] = sum_i a[i] * w_rev[m-i] = sum_i a[i] * w[2n-2-(m-i)] = sum_i a[i] * w[2n-2-m+i]
    # We want: sum_i a[i] * w[j+i] for i=0,...,n-1
    # Set 2n-2-m+i = j+i => 2n-2-m = j => m = 2n-2-j
    # So (a * w_rev)[2n-2-j] = sum_i a[i] * w[j+i] ✓
    
    # So corr[j] = conv_res[2n-2-j] for j=0,...,n-1
    # That's indices 2n-2, 2n-3, ..., n-1
    corr = conv_res[n-1:2*n-1][::-1]
    
    return 2.0 * corr, w

def nesterov_smoothed(a0, time_limit, verbose=False, seed=0):
    """
    Nesterov-accelerated smoothed gradient descent.
    Minimizes 2n*max(conv(a,a))/sum(a)^2 by fixing sum(a)=sqrt(2n).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.full(n, target_sum / n)
    else:
        a = a * target_sum / s0
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    T_schedule = [(0, 200), (200, 50), (500, 20), (1000, 10), (2000, 5), (4000, 2), (6000, 1)]
    T = 200.0
    step = 2.0 / np.sqrt(n)
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Update T
        for (thresh, t_val) in T_schedule:
            if it > thresh:
                T = t_val
        T = max(T, 1.0)
        
        # Compute conv and gradient
        conv = fast_autocorr(y)
        grad, w = smoothed_gradient(y, conv, T)
        
        # Projected gradient step
        a_trial = y - step * grad / np.sqrt(n)
        a_trial = proj_simplex_nonneg(a_trial, target_sum)
        
        score_trial = compute_score(a_trial)
        
        if score_trial < best_score - 1e-12:
            new_a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            stall = 0
            step = min(step * 1.02, 10.0 / np.sqrt(n))
        else:
            new_a = y
            stall += 1
            step *= 0.95
            if step < 1e-12:
                step = 2.0 / np.sqrt(n)
                # Perturb to escape
                new_a = best_a + 0.01 * rng.standard_normal(n)
                new_a = proj_simplex_nonneg(new_a, target_sum)
        
        # Nesterov
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * new_a
        y_new = proj_simplex_nonneg(y_new, target_sum)
        
        if it <= 1:
            a_mom = new_a
        else:
            beta = (it - 1.0) / (it + 2.0)
            beta = min(beta, 0.93)
            a_mom = y_new + beta * (y_new - y_prev)
            a_mom = proj_simplex_nonneg(a_mom, target_sum)
        
        y_prev = y
        y = y_new
        a = a_mom
        
        # Track best from momentum
        score_m = compute_score(a)
        if score_m < best_score:
            best_score = score_m
            best_a = a.copy()
            stall = 0
        
        # Ergodic average check
        if it % 50 == 0:
            # (Skip for now to save time)
            pass
        
        # Restart
        if stall > 200:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall = 0
            step = 2.0 / np.sqrt(n)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS iter={it} ({elapsed:.0f}s): score={best_score:.10f}, T={T:.1f}")
    
    return best_a, best_score

def subgradient_smooth(a0, time_limit, verbose=False, seed=0):
    """
    Subgradient method using top peaks with adaptive weights.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        a = np.full(n, target_sum / n)
    else:
        a = a * target_sum / s0
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 1.0 / np.sqrt(n)
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Get top peaks (adaptive number)
        n_peaks = max(5, min(20, n // 80))
        # Use exponential weighting
        top_idx = np.argsort(conv)[-n_peaks:]
        range_vals = conv[top_idx] - conv.min()
        max_range = range_vals.max()
        if max_range < 1e-12:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = np.exp(-3.0 * (range_vals.max() - range_vals) / max_range)
            weights /= weights.sum()
        
        # Combined gradient
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_idx):
            # grad_contribution[j] = 2 * a[k-j] if 0 <= k-j < n
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            # a[k-j] for j in [lo, hi)
            # k-j ranges from k-lo to k-(hi-1)
            # Let's just do: for j in range(n): if 0<=k-j<n: grad[j] += 2*w*a[k-j]
            valid = np.arange(n)
            idx = k - valid
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * w_k * a[idx[mask]]
        
        # Projected step
        a_trial = a - step * grad / np.sqrt(n)
        a_trial = proj_simplex_nonneg(a_trial, target_sum)
        
        score_trial = compute_score(a_trial)
        if score_trial < best_score - 1e-12:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            stall = 0
            step = min(step * 1.01, 5.0 / np.sqrt(n))
        else:
            stall += 1
            step *= 0.98
            if step < 1e-12:
                break
        
        if stall > 500:
            a = best_a.copy()
            step = 1.0 / np.sqrt(n)
            stall = 0
        
        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it} ({elapsed:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score
