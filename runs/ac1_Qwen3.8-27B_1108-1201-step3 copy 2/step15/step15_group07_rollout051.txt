def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix T where (T@b)[k] = sum_j a[k-j]*b[j]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. T@a_cross(b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val, None

def spectral_flattening(a0, time_limit, verbose=True):
    """
    Gerchberg-Saxton style spectral flattening for non-negative sequences.
    Iteratively:
    1. Compute FFT of current sequence
    2. Compress the power spectrum (make it flatter)
    3. Inverse FFT
    4. Clip to non-negative
    5. Normalize
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    patience = 0

    # FFT size
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute FFT
        A = rfft(a, fft_size)

        # Power spectrum
        power = np.abs(A) ** 2

        # Compress the spectrum
        # Strategy: raise to a power < 1 to compress dynamic range
        # Or: subtract mean and add a smaller mean
        p_max = power.max()
        p_mean = power.mean()
        
        # Use a soft compression: new_power = p_mean + (p_max - p_mean) * (power / p_max)^gamma
        for gamma in [0.3, 0.5, 0.7]:
            if power.max() < 1e-30:
                break
            compressed = p_mean + (p_max - p_mean) * (power / power.max()) ** gamma
            
            # Inverse FFT (keeping the phase of A)
            phase = A / (np.abs(A) + 1e-30)
            new_A = np.sqrt(compressed) * phase
            new_a = irfft(new_A, fft_size)[:n]
            new_a = np.maximum(new_a, 0)
            new_a = new_a * norm / max(new_a.sum(), 1e-15)
            
            new_score = fast_autocorr(new_a).max()
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = new_a.copy()
                a = new_a
                patience = 0
                break
        else:
            patience += 1

        # Alternative: direct spectral shaping
        if patience % 20 == 0:
            # Try to make the spectrum more uniform
            target_power = np.ones_like(power) * p_mean
            # Blend
            blend = 0.3
            new_power = (1 - blend) * power + blend * target_power
            phase = A / (np.abs(A) + 1e-30)
            new_A = np.sqrt(new_power) * phase
            new_a = irfft(new_A, fft_size)[:n]
            new_a = np.maximum(new_a, 0)
            s = new_a.sum()
            if s > 1e-15:
                new_a = new_a * norm / s
                new_score = fast_autocorr(new_a).max()
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = new_a.copy()
                    a = new_a
                    patience = 0

        if patience > 200:
            # Perturb and restart
            rng = np.random.default_rng(it)
            noise = rng.normal(0, norm * 0.01, n)
            a = np.maximum(a + noise, 0)
            a = a * norm / max(a.sum(), 1e-15)
            patience = 0

        n_iters += 1
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def dual_guided_cd(a0, time_limit, solver, verbose=True):
    """
    Coordinate descent guided by LP dual variables.
    Uses the dual information to identify which lags are most constraining,
    then shifts mass to reduce those specific lags.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    rng = np.random.default_rng(555)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if n_passes > 500 and no_improve > 100:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()
        # Top active lags
        top_k = min(20, size)
        active_idx = np.argsort(conv)[-top_k:]
        active_sorted = np.sort(active_idx)

        # Get gradient information from active lags
        # grad[i] = sum over active k of (coefficient of a_i in conv[k])
        # conv[k] = sum_j a[j] * a[k-j]
        # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (if valid)
        grad = np.zeros(n)
        for k in active_idx:
            k = int(k)
            for i in range(n):
                j1 = k - i
                j2 = i - k
                if 0 <= j1 < n:
                    grad[i] += a[j1]
                if 0 <= j2 < n:
                    grad[i] += a[j2]
        
        # Try to reduce the gradient (which increases max conv) by 
        # shifting mass from high-grad positions to low-grad positions
        improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue
            if grad[i] < np.median(grad):
                continue  # Already at a "good" position

            # Find best destination: low grad, far from active lags
            candidates = np.argsort(grad)[:20]
            for j in candidates:
                j = int(j)
                if j == i or a[j] >= a[i] * 0.5:
                    continue
                # Try shifting
                for frac in [0.2, 0.4, 0.6]:
                    d = a[i] * frac
                    a[i] -= d
                    a[j] += d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += d
                        a[j] -= d
                        continue
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        no_improve = 0
                        break
                    else:
                        a[i] += d
                        a[j] -= d
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0 and no_improve > 0:
                # Periodic perturbation
                scale = norm * 0.01 * (1 + no_improve / 100.0)
                noise = rng.laplace(0, scale, n)
                a_pert = np.maximum(a + noise, 0)
                s = a_pert.sum()
                if s > 1e-15:
                    a = a_pert * norm / s
                no_improve = 0

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_scale_lp(a0, time_limit, solver, verbose=True):
    """
    Multi-scale LP: solve the problem on a coarser grid first,
    then refine on the full grid. This can escape shallow local minima.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a_full = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a_full.sum()
    if s0 < 1e-15:
        a_full = np.ones(n) * norm / n
    else:
        a_full = a_full * norm / s0

    best_a = a_full.copy()
    best_score = fast_autocorr(a_full).max()
    t0 = time.monotonic()

    # Try different "coarsenings" by averaging blocks
    block_sizes = [2, 4, 8]
    for bs in block_sizes:
        if time.monotonic() - t0 > time_limit:
            break
        if n % bs != 0:
            continue
        n_coarse = n // bs
        if n_coarse < 10:
            continue

        # Coarsen
        a_coarse = a_full.reshape(-1, bs).mean(axis=1)
        a_coarse = a_coarse * np.sqrt(2.0 * n_coarse) / a_coarse.sum()

        # Optimize coarse

        if t_sub < 10:

        a_c_out, sc_c = nesterov_lp_adaptive(a_coarse, t_sub, solver, verbose=False)
        
        # Upsample and use as initialization
        a_up = np.repeat(a_c_out, bs)
        a_up = a_up * norm / a_up.sum()
        
        # Quick refinement
        t_refine = min(30.0, (time_limit - (time.monotonic() - t0)) * 0.2)

                best_a = a_r_out.copy()
                if verbose:
                    print(f"    Multi-scale bs={bs}: improved to {best_score:.8f}")

            patience = 0

        n_iters += 1
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Spectral iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def dual_guided_cd(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t_start = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    rng = np.random.default_rng(8888)

    while time.monotonic() - t_start < time_limit:
        n_passes += 1
        if n_passes > 2000 and no_improve > 200:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()
        top_k = min(15, size)
        active_idx = np.argsort(conv)[-top_k:]

        # Compute subgradient at active lags
        grad = np.zeros(n)
        for k in active_idx:
            k_int = int(k)
            for i in range(n):
                j1 = k_int - i
                j2 = i - k_int
                if 0 <= j1 < n:
                    grad[i] += a[j1]
                if 0 <= j2 < n:
                    grad[i] += a[j2]

        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t_start > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue
            if grad[i] < np.median(grad) * 0.8:
                continue

            # Find low-gradient destinations
            low_grad_idx = np.argsort(grad)[:30]
            for j in low_grad_idx:
                j_int = int(j)
                if j_int == i:
                    continue
                for frac in [0.3, 0.5, 0.7]:
                    d = a[i] * frac
                    if d < norm * 1e-12:
                        break
                    a[i] -= d
                    a[j_int] += d
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        no_improve = 0
                        break
                    else:
                        a[i] += d
                        a[j_int] -= d
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                scale = norm * 0.008 * (1.0 + no_improve / 80.0)
                noise = rng.laplace(0, scale, n)
                a = np.maximum(a + noise, 0)
                a = a * norm / max(a.sum(), 1e-15)
                no_improve = 0

        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    DCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def targeted_perturbation_lp(a0, time_limit, solver, verbose=True):
    """
    Alternates between targeted perturbations (based on peak lags) and LP optimization.
    This helps escape local minima by making large jumps followed by refinement.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) * norm / n
    else:
        best_a = best_a * norm / s
    best_score = fast_autocorr(best_a).max()
    t_start = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(2024)

    n_cycles = 0
    while time.monotonic() - t_start < time_limit:
        n_cycles += 1
        t_remaining = time_limit - (time.monotonic() - t_start)
        if t_remaining < 30:
            break

        # Targeted perturbation: shift mass from positions contributing to peak lags
        conv = fast_autocorr(best_a)
        peak_lag = int(np.argmax(conv))

        # Find which positions contribute most to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                contrib[i] += a0[i] * a0[j] if False else best_a[i] * best_a[j]
            j2 = i - peak_lag
            if 0 <= j2 < n:
                contrib[i] += best_a[i] * best_a[j2]

        # Perturb: reduce high-contribution positions, increase others
        top_contrib = np.argsort(contrib)[-n // 4:]
        bottom_contrib = np.argsort(contrib)[:n // 4]

        a_pert = best_a.copy()
        total_reduce = 0.0
        for i in top_contrib:
            reduce_amt = a_pert[i] * 0.3
            a_pert[i] -= reduce_amt
            total_reduce += reduce_amt

        a_pert = np.maximum(a_pert, 0)
        for j in bottom_contrib:
            a_pert[j] += total_reduce / len(bottom_contrib)

        a_pert = a_pert * norm / max(a_pert.sum(), 1e-15)

        # Quick LP optimization from perturbed state
        t_lp = min(t_remaining * 0.4, 120.0)
        if t_lp > 10:
            a_out, sc = nesterov_lp_adaptive(a_pert, t_lp, solver, verbose=False)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_out.copy()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    TPL cycle={n_cycles} ({elapsed:.0f}s): improved to {best_score:.8f}")
            else:
                # Try smaller perturbation
                a_pert2 = best_a.copy()
                for i in top_contrib[:n // 8]:
                    reduce_amt = a_pert2[i] * 0.1
                    a_pert2[i] -= reduce_amt
                a_pert2 = np.maximum(a_pert2, 0)
                a_pert2 = a_pert2 * norm / max(a_pert2.sum(), 1e-15)
                t_lp2 = min(t_remaining * 0.2, 60.0)
                if t_lp2 > 10:
                    a_out2, sc2 = nesterov_lp_adaptive(a_pert2, t_lp2, solver, verbose=False)
                    if sc2 < best_score - 1e-10:
                        best_score = sc2
                        best_a = a_out2.copy()
                        if verbose:
                            elapsed = time.monotonic() - t_start
                            print(f"    TPL cycle={n_cycles} small ({elapsed:.0f}s): improved to {best_score:.8f}")

    return best_a, best_score


def try_update_global(best_freq, a_candidate, label, n_base):
    """Evaluate candidate and update if better. Returns (improved, new_best_freq, new_best_a)."""
    if a_candidate is None:
        return False, best_freq, None
    a = np.maximum(np.asarray(a_candidate, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return False, best_freq, None
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_freq - 1e-9:
        elapsed = time.monotonic()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True, sc, a.copy()
    return False, best_freq, None


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining adaptive LP Nesterov, spectral flattening,
    dual-guided CD, and targeted perturbation with LP polishing.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_freq = float('inf')
    best_a = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = p / p.sum()
            p_score = evaluate_safe(p)
            print(f"Parent: n={len(p)}, score={p_score:.6f}")
            best_a = p.copy()
            best_freq = p_score
    except Exception:
        pass

    if best_a is None:
        n_base = 1600
        best_a = np.ones(n_base) / n_base
        best_freq = evaluate_safe(best_a)
        print(f"Default init: n={n_base}, score={best_freq:.6f}")
    else:
        n_base = len(best_a)

    # === Phase 1: Adaptive LP Nesterov (38% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.38
    print(f"\n=== Phase 1: Adaptive LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_a * np.sqrt(2.0 * n_base) / best_a.sum()
        a_out, _ = nesterov_lp_adaptive(a_norm, t1, solver, verbose=True)
        _, best_freq, best_a_new = try_update_global(best_freq, a_out, "lp1", n_base)
        if best_a_new is not None:

    # === Phase 2: Spectral Flattening (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 2: Spectral Flattening n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_a * np.sqrt(2.0 * n_cur) / best_a.sum()
        a_out, _ = spectral_flattening(a_norm, t2, verbose=True)
        _, best_freq, best_a_new = try_update_global(best_freq, a_out, "spectral", n_cur)
        if best_a_new is not None:
            best_a = best_a_new

    # === Phase 3: LP Nesterov Round 2 (22% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    t3 = remaining * 0.22
    print(f"\n=== Phase 3: LP Nesterov R2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_a * np.sqrt(2.0 * n_cur) / best_a.sum()
        a_out, _ = nesterov_lp_adaptive(a_norm, t3, solver, verbose=True)

        if best_a_new is not None:
            best_a = best_a_new

    # === Phase 4: Targeted Perturbation + LP (12% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    t4 = remaining * 0.12
    print(f"\n=== Phase 4: Targeted Perturbation + LP n={n_cur} (budget={t4:.0f}s) ===")

    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 5: Dual-guided CD n={n_cur} (budget={t5:.0f}s) ===")

    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_a]
