                continue

        if duals is not None and len(duals) > n // 10:
            dual_thresh = max(1e-10, np.percentile(np.abs(duals), 30))
            keep = np.abs(duals) > dual_thresh
            if keep.sum() < max(5, n // 50):
                n_keep = max(5, n // 50)
                idx = np.argsort(np.abs(duals))[-n_keep:]
                keep = np.zeros(len(duals), dtype=bool)
                keep[idx] = True
            if keep.sum() < len(active_set):
                active_set = active_set[keep]

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t <= 2:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.99)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart iter {t} ({elapsed:.0f}s) score={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f} M={M_curr:.6f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max() / (a.sum() ** 2) * 2 * n
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_passes = 0

    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_autocorr(a)
        current_peak = conv.max()
        S = a.sum()

        order = list(range(n))
        rng.shuffle(order)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            delta_conv = np.zeros(size)
            lo1 = max(0, 0)
            for k in range(size):
                v = 0.0
                i1 = k - j
                if 0 <= i1 < n:
                    v += a[i1]
                i2 = j - k
                if 0 <= i2 < n:
                    v += a[i2]
                delta_conv[k] = v

            # conv_new[k] = conv[k] + delta * delta_conv[k] + delta^2 * delta_{k,2j}
            # We want to minimize max_k (conv[k] + delta*delta_conv[k] + delta^2*_{k,2j}) / (S + delta)^2
            # Let's do a golden section search on delta in [-aj, some_upper]

            is_2j = (2 * j < size)

            def obj(delta):
                new_S = S - aj + (aj + delta)
                new_S = S + delta
                if new_S <= 1e-15:
                    return 1e30
                new_conv = conv + delta * delta_conv
                if is_2j:
                    new_conv[2 * j] += delta * delta
                return new_conv.max() / (new_S * new_S)

            # Search in delta range [-aj, 2*aj + norm/n]
            lo_d = -aj
            hi_d = 2.0 * aj + norm / n
            # Try a few initial points
            candidates = [0.0, -aj * 0.5, aj * 0.5, -aj * 0.9, aj * 0.9,
                          0.1 * norm / n, -0.1 * a.mean()]
            candidates = [max(lo_d, min(hi_d, c)) for c in candidates]
            candidates.append(lo_d)
            candidates.append(hi_d)

            best_d = 0.0
            best_obj = obj(0.0)
            for d in candidates:
                o = obj(d)
                if o < best_obj:
                    best_obj = o
                    best_d = d

            if abs(best_d) > 1e-12:
                # Refine with bisection
                lo_b = best_d - max(0.1 * abs(best_d), 0.01 * norm / n, 1e-4)
                hi_b = best_d + max(0.1 * abs(best_d), 0.01 * norm / n, 1e-4)
                lo_b = max(lo_b, lo_d)
                hi_b = min(hi_b, hi_d)
                obj_lo = obj(lo_b)
                obj_hi = obj(hi_b)
                if best_obj < obj_lo:
                    hi_b = best_d
                elif best_obj < obj_hi:
                    lo_b = best_d
                else:
                    lo_b = min(lo_b, hi_b)
                    hi_b = max(lo_b, hi_b)
                for _ in range(12):
                    mid = (lo_b + hi_b) / 2.0
                    obj_mid = obj(mid)
                    if obj_mid < best_obj:
                        best_obj = obj_mid
                        best_d = mid
                    if obj_mid < obj((lo_b + hi_b) / 2.0):
                        pass
                    # Simple bisection
                    obj_lo2 = obj(lo_b)
                    obj_hi2 = obj(hi_b)
                    if obj_mid <= min(obj_lo2, obj_hi2):
                        if best_obj < obj_mid:
                            pass
                        best_d = mid
                        best_obj = obj_mid
                        if obj_mid <= obj_lo2:
                            hi_b = mid
                        else:
                            lo_b = mid
                    else:
                        if obj_mid > obj_lo2:
                            lo_b = mid
                        else:
                            hi_b = mid

                a[j] = aj + best_d
                if a[j] < 0:
                    a[j] = 0
                S_new = a.sum()
                if S_new > 1e-15:
                    a = a * norm / S_new
                    new_conv = fast_autocorr(a)
                    new_score = 2.0 * n * new_conv.max() / S_new**2
                    if new_score < best_score - 1e-14:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm / a.sum()

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

        if verbose and no_improve_passes == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def gradient_polish(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.3 / np.sqrt(n)
    patience = 0
    max_patience = 300

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Softmax weights
        eps = max(0.01 * M_curr, 1e-6)
        shifted = conv / eps - conv.max() / eps
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        # Compute gradient: grad[m] = 2 * sum_k weights[k] * a[k-m] for valid k-m
        grad = np.zeros(n)
        active_k = np.where(weights > 1e-10)[0]
        for k in active_k:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if hi > lo:
                grad[lo:hi] += 2.0 * weights[k] * a[k - np.arange(lo, hi)]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 2.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Grad t={t} ({elapsed:.0f}s): best={best_score:.10f} step={step:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_seq(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
