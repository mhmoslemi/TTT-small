def get_solver():

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)[:size]
    return c


def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a, active_indices=None):
    """Build the constraint matrix for the LP. T[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    size = 2 * n - 1
    if active_indices is not None:
        rows = len(active_indices)
    else:
        rows = size
        active_indices = np.arange(size)
    
    # For each active constraint k, T[k,i] = a[k-i] if 0 <= k-i < n
    T = np.zeros((rows, n))
    for row, k in enumerate(active_indices):
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            # T[k,i] = a[k-i] for i in [i_start, i_end]
            vals = a[k - i_end : k - i_start + 1][::-1]
            T[row, i_start:i_end+1] = vals
    return T


def lp_solve(a, M, solver, active_indices=None, verbose=False):
    n = len(a)
    size = 2 * n - 1
    
    if active_indices is None:
        active_indices = np.arange(size)
    
    T = build_T(a, active_indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception as e:
        if verbose:
            print(f"    LP error: {e}")
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def adam_lse_optimize(a0, time_limit, verbose=True):
    """Adam optimizer on the smoothed LSE objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    # Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 1.0 / np.sqrt(n)

    n_stages = 15
    temps = np.geomspace(50.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Adaptive number of steps per stage
        n_steps = max(50, min(200, int(remaining * 8)))
        # Decrease learning rate per stage
        stage_lr = lr * (1.0 / (1.0 + 0.3 * si))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            # LSE smoothing
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient of LSE: grad = 2 * (a * w + flip(a) * flip(w)) effectively
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))
            step = stage_lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            if (it + 1) % 5 == 0:
                sc = fast_conv_self(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if verbose:
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            print(f"    Adam-LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score

            a_new = y_new.copy()
        a_new = np.maximum(a_new, 0.0)

def spectral_gradient_step(a, n_steps, lr, time_limit, t0, verbose=False):
    """Gradient descent on the max autocorrelation using smooth approximation."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    fft_size = 1 << (2 * n - 1).bit_length()

    # Use a soft-max temperature
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        T = max(0.01, max_c * 0.02)

        shifted = (conv - max_c) / T
        shifted = np.maximum(shifted, -80.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        # Gradient step with decay
        step = lr / (1.0 + 0.01 * it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        if (it + 1) % 10 == 0:
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

        it_count += 1

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Dynamic active set
        n_active = min(size, max(n + 30, int(0.35 * size)))
        sorted_idx = np.argsort(conv_y)[::-1]
        active = sorted_idx[:n_active]
        # Always include peaks
        peaks = np.where(conv_y >= M * 0.95)[0]
        if len(peaks) > 0:
            active = np.unique(np.concatenate([active, peaks]))
        if len(active) > n + 150:
            active_vals = conv_y[active]
            top = np.argsort(active_vals)[::-1][:n + 150]
            active = active[top]

        b = lp_solve_fast(y, M, solver, active)
        if b is None:
            active_few = np.where(conv_y >= M * 0.6)[0]
            if len(active_few) < 50:
                active_few = sorted_idx[:n + 50]
            b = lp_solve_fast(y, M, solver, active_few)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify periodically
        if len(active) < size and it_count % 15 == 0:
            check = fast_conv_cross(y, g)
            if (check > M * 1.01).sum() > 30:
                active_full = sorted_idx[:min(size, n + 200)]
                b2 = lp_solve_fast(y, M, solver, active_full)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum > 0:
            a_new = y_new + momentum * (y_new - y)
        else:
            a_new = y_new.copy()
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if improved:
            stuck = 0
            momentum = min(0.92, momentum + 0.005)
        else:
            stuck += 1
            if stuck > 80:
                momentum = max(0.0, momentum - 0.1)
                stuck = 0

        if verbose and t % 80 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LPNest t={t} ({elapsed:.0f}s): best={best_score:.8f} M={M:.5f} mom={momentum:.2f}")

    return best_a, best_score


def pair_cd_refine(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_imp = False

        peak_lags = np.where(conv >= cur_max * 0.93)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            j_cands = set()
            for _ in range(6):
                j_cands.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                if j1 != i:
                    j_cands.add(j1)
                if j2 != i:
                    j_cands.add(j2)
            j_cands.discard(i)
            j_list = list(j_cands)[:12]

            for j in j_list:
                if j == i:
                    continue

                # Linear change vector: 2*(a shifted to j) - 2*(a shifted to i)
                lin = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    lin[lo:hi] += 2.0 * a[lo - j:hi - j]
                lo = max(0, i)
                hi = min(size, i + n)
                if hi > lo:
                    lin[lo:hi] -= 2.0 * a[lo - i:hi - i]

                # Quadratic: +1 at 2j, -2 at i+j, +1 at 2i
                quad = np.zeros(size)
                if 2 * j < size:
                    quad[2 * j] = 1.0
                s_ij = i + j
                if s_ij < size:
                    quad[s_ij] -= 2.0
                if 2 * i < size:
                    quad[2 * i] += 1.0

                # Search for best delta
                n_samp = 18
                d_max = ai * 0.95
                if d_max < 1e-10:
                    continue
                if d_max > 1e-6:
                    d_vals = np.geomspace(1e-8, d_max, n_samp)
                else:
                    d_vals = np.linspace(d_max * 0.01, d_max, n_samp)

                best_val = cur_max
                best_d = 0.0
                for d in d_vals:
                    nc = conv + d * lin + d * d * quad
                    v = nc.max()
                    if v < best_val:
                        best_val = v
                        best_d = d

                if best_d > 0 and best_val < cur_max - 1e-13:
                    a[i] -= best_d
                    a[j] += best_d
                    conv = fast_conv_self(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_imp = True
                        total_imp += 1

        if not pass_imp:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_pert = max(5, n // 80)
                ip = rng.choice(n, size=n_pert, replace=False)
                a[ip] *= (1.0 + 0.03 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_imp % 80 == 0 and total_imp > 0:
            elapsed = time.monotonic() - t0
            print(f"  PCD imp={total_imp} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_refine(a0, time_limit, verbose=True):
    """Fine spectral gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m_mom = np.zeros(n)
    v_mom = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-10
    lr = 0.5 / np.sqrt(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv_self(a)
        max_c = conv.max()
        T = max(0.005, max_c * 0.01 * (1.0 + 0.005 * it) ** 0.1)

        shifted = (conv - max_c) / T
        shifted = np.maximum(shifted, -100.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        m_mom = b1 * m_mom + (1 - b1) * grad
        v_mom = b2 * v_mom + (1 - b2) * grad * grad
        ml = m_mom / (1 - b1 ** it)
        vl = v_mom / (1 - b2 ** it)
        step = lr / (1.0 + 0.005 * it) * ml / (np.sqrt(vl) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / sn
        a = a_new

        if it % 5 == 0:
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Spectral it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    else:
        a = np.sinc(3.0 * (x - 0.5)) ** 2 + 0.1
    a = np.maximum(a, 0.0)
    a = a + 0.01 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Adam LSE (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: AdamLSE n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = adam_lse_optimize(best_seq.copy(), t1, verbose=True)
        try_update(a_out, "adam1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Spectral refine
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: Spectral n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = spectral_refine(best_seq, t3, verbose=True)
        try_update(a_out, "spec1")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: LP Nesterov r2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=False)
        try_update(a_out, "lp2")

    # Phase 5: Pair CD refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 5: PCD n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = pair_cd_refine(best_seq, t5, rng, verbose=True)
        try_update(a_out, "pcd1")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t6 = min(remaining * 0.20, 200.0)
    print(f"\n=== Phase 6: Long n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t6a = min(t6 * 0.6, 120.0)
        a_out, _ = nesterov_lp_optimize(init_long, t6a, solver, verbose=False)
        try_update(a_out, f"lp_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out2, _ = pair_cd_refine(best_seq, min(remaining * 0.5, 60.0), rng, verbose=False)
            try_update(a_out2, "pcd2")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t7 = min(remaining * 0.4, 80.0)
        a_out, _ = adam_lse_optimize(best_seq.copy(), t7, verbose=False)
        try_update(a_out, "adam_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_refine(best_seq, min(remaining * 0.6, 60.0), rng, verbose=False)
            try_update(a_out, "pcd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = spectral_refine(best_seq, min(remaining * 0.5, 30.0), verbose=False)
            try_update(a_out, "spec_final")

    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
