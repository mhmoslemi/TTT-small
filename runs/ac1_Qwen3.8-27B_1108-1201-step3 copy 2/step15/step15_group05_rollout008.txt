def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_matrix_full(a, n):
    """Build full (2n-1) x n constraint matrix T where T[k,j] = a[k-j] if valid."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    # T[k, j] = a[k-j] if 0 <= k-j < n
    for j in range(n):
        lo = max(0, j)
        hi = min(size, j + n)
        T[lo:hi, j] = a[:hi - lo]
    return T


def build_matrix_subset(a, n, k_indices):
    """Build constraint matrix for selected lags only."""
    k_indices = np.asarray(k_indices)
    size = len(k_indices)
    T = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[idx, j] = a[k - j]
    return T


def solve_lp(a_fixed, M, T, solver):
    """Solve LP: max sum(b) s.t. T @ b <= M, b >= 0. Returns (b, duals)."""
    n = a_fixed.shape[0]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def lp_optimize_full(a0, time_limit, solver, verbose=False, seed=0):

    rng = np.random.default_rng(seed)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    a_momentum = a.copy()
    full_T = None
    full_T_count = 0
    n_full = 2 * n - 1

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if t <= 20:
            tol = 0.3
        elif t <= 100:
            tol = 0.1
        elif t <= 500:
            tol = 0.03
        else:
            tol = 0.01

        if t % 50 == 0:
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            top_k = np.argsort(conv_y)[-max(5, n // 100):]
            active_set = np.unique(np.concatenate([active_set, top_k]))

        if len(active_set) >= n_full or (t > 200 and t % 100 == 0):
            # Use full constraint matrix
            if full_T is None:
                full_T = build_matrix_full(y, n)
            T_use = full_T
            active_use = None
        else:

            active_use = active_set

        b, duals = solve_lp(y, M_curr, T_use, solver)
        if b is None:
            # Fallback: full LP
            if full_T is None:
                full_T = build_matrix_full(y, n)
            b, duals = solve_lp(y, M_curr, full_T, solver)
            if b is None:
                continue

        if active_use is not None and t % 200 == 0:

                full_T = build_matrix_full(y, n)

                    b = b2

        if b_sum < 1e-15:
            continue

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

            a_momentum = g.copy()
        else:

            momentum = min(momentum, 0.99)
            a_momentum = y_new + momentum * (y_new - y_prev)

            if s_a < 1e-15:
                a_momentum = y_new.copy()
            else:

        y_prev = y
        y = y_new

        # Track best
        for candidate in [y, a_momentum]:

            if sc < best_conv:
                best_conv = sc
                best_a = candidate.copy()

            if s_erg > 1e-15:

                if sc < best_conv:

                a_momentum = best_a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_conv:.12f}")

def mirror_descent_polish(a0, time_limit, verbose=False, seed=0):
    """Mirror descent (multiplicative subgradient) for the autocorrelation minimization."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 1.0 / (n ** 0.6)
    patience = 0
    max_patience = 1000

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get top peaks (multiple for robustness)
        n_peaks = max(3, min(15, n // 50))
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Combined subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for j in range(n):
                v = 0.0
                j1 = k - j
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = j - k
                if 0 <= j2 < n:
                    v += a[j2]
                grad[j] += w * v

        # Normalize gradient
        gm = np.abs(grad).max()
        if gm < 1e-15:
            break
        grad = grad / gm

        # Multiplicative (mirror) update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        a_new = np.exp(log_a_new - log_a_new.max())
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-14:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            step = min(step * 1.005, 3.0)
        else:
            patience += 1
            step *= 0.995
            if step < 1e-10:
                break

        # Occasional diversification
        if t % 500 == 0 and patience > 200:
            pert = a * (1.0 + 0.01 * rng.standard_normal(n))
            pert = np.maximum(pert, 0.0)
            sp = pert.sum()
            if sp > 1e-15:
                pert = pert * norm / sp
                sc_pert = fast_autocorr(pert).max()
                if sc_pert < best_score:
                    a = pert
                    best_score = sc_pert
                    best_a = pert.copy()
                patience = 0
                step = 1.0 / (n ** 0.6)

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror t={t} ({elapsed:.0f}s): best={best_score:.12f}, step={step:.6f}")

    return best_a, best_score

                        a_shifted = np.zeros(size)
                        a_shifted[j:j+n] = a
                        c_new = c + 2.0 * delta * a_shifted
                        if 2 * j < size:
                            c_new[2 * j] += delta * delta
                        c_new = np.maximum(c_new, 0.0)
                        obj = 2.0 * n * c_new.max() / (denom * denom)
                    if obj < best_obj:
                        hi = mid
                        best_obj = obj
                        best_x = mid
                    else:
                        lo = mid

            if abs(best_x - aj) > 1e-12:
                # Update a and c
                delta = best_x - aj
                s_others = norm - aj
                denom = s_others + best_x
                if denom < 1e-10:
                    continue

                # Update autocorrelation
                a_shifted = np.zeros(size)
                a_shifted[j:j+n] = a
                c = c + 2.0 * delta * a_shifted
                if 2 * j < size:
                    c[2 * j] += delta * delta
                c = np.maximum(c, 0.0)

                a[j] = best_x
                # Renormalize
                a = a * norm / a.sum()
                c = fast_autocorr(a)  # Recompute for accuracy after renormalization

                new_score = 2.0 * n * c.max() / (norm * norm)
                if new_score < best_score - 1e-14:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                else:
                    # Revert
                    a = best_a * (a.sum() / norm) if best_a.sum() > 0 else a
                    c = fast_autocorr(a)

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.12f}")

    return best_a, best_score


def spectral_shaping(a0, time_limit, verbose=False, seed=0):
    """Iterative spectral peak reduction."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    alpha = 0.1
    patience = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 500:
            break

        # Compute spectrum
        A = rfft(a, fft_size)
        power = A.real ** 2 + A.imag ** 2
        phase = np.angle(A)

        # Compute current autocorrelation from spectrum
        c = irfft(power, fft_size)[:size]
        M_curr = c.max()
        c_mean = c.mean()

        # Create flattening filter
        # Damp the time-domain peaks by reducing corresponding frequency content
        # Use an exponential map: new_power = power * exp(-alpha * (c_normalized)^p)
        c_norm = c / M_curr
        damping = np.exp(-alpha * c_norm ** 4)
        # Keep DC component alive
        damping[0] = 1.0

        new_power = power * damping
        new_power = np.maximum(new_power, 0.0)
        if new_power.sum() < 1e-15:
            break

        # Reconstruct with original phase
        new_A = np.sqrt(new_power) * (np.cos(phase) + 1j * np.sin(phase))
        a_new = irfft(new_A, fft_size)[:n]

        # Alternative: use magnitude only (random phase) for diversity every 50th iter
        if t % 50 == 0:
            new_A_mag = np.sqrt(new_power)
            random_phase = rng.standard_normal(fft_size // 2 + 1)
            random_phase = random_phase / (np.abs(random_phase).max() + 1e-15) * np.pi
            new_A = new_A_mag * (np.cos(random_phase) + 1j * np.sin(random_phase))
            new_A[0] = np.sqrt(new_power[0])  # Keep DC real and positive
            a_new = irfft(new_A, fft_size)[:n]

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            patience += 1
            continue
        a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-14:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            alpha = min(alpha * 1.01, 1.0)
        else:
            patience += 1
            alpha *= 0.99
            if alpha < 0.001:
                break

        # Blend with current best to maintain progress
        if t % 20 == 0:
            blend = 0.3 * a + 0.7 * best_a
            blend = blend * norm / blend.sum()
            sc_blend = fast_autocorr(blend).max()
            if sc_blend < best_score:
                a = blend
                best_score = sc_blend
                best_a = blend.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral t={t} ({elapsed:.0f}s): best={best_score:.12f}, alpha={alpha:.4f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def symmetrize(a):
    """Symmetrize: a = (a + flip(a)) / 2"""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def propose_candidate():
    """
    Main search: multi-phase optimization combining LP best-response, mirror descent,
    spectral shaping, and coordinate descent across multiple sequence lengths.
    """
    rng = np.random.default_rng(42)

    time_budget = 1750.0

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    # Quick symmetrization check
    a_sym = symmetrize(best_seq)
    sc_sym = evaluate_safe(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_seq = a_sym.copy()
        print(f"  Symmetrization improved: {sc_sym:.10f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.10f}")

    # === Phase 1: Main LP optimization at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP full n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lp_optimize_full(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp_main_n{n_base}")

    # === Phase 2: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Mirror descent n={n_cur}, budget={t2:.0f}s ===")
