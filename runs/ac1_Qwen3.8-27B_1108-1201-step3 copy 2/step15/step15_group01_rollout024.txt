def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix T where (T @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T

def lp_solve(a, M, constraints_idx=None):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    if constraints_idx is not None:
        T = T[constraints_idx, :]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]

    prob.solve(solver=get_solver(), verbose=False)

            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.4f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_improved(a0, time_limit, verbose=True):
    """LP Nesterov with Polyak-Ruppert averaging and adaptive constraints."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = a.copy()
    n_avg = 1
    stuck = 0
    prev_best = best_score

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive constraint selection
        # Use constraints above 70% of max, capped at 300
        threshold = 0.7 * M_curr
        active = np.where(conv_y >= threshold)[0]
        if len(active) > 300:
            active = np.argsort(conv_y)[::-1][:300]
        if len(active) < 5:
            active = np.argsort(conv_y)[::-1][:20]

        b = lp_solve_constr(y, M_curr, active)
        if b is None:
            b = lp_solve_constr(y, M_curr, None)
            if b is None:
                stuck += 1
                if stuck > 50:
                    break
                continue

        stuck = 0
        b = np.maximum(b, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t <= 2:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Polyak-Ruppert averaging (exponential)
        if t >= 20:
            rate = 1.0 / (n_avg + 1)
            avg_a = avg_a * (1 - rate) + a * rate
            n_avg += 1
            avg_a = np.maximum(avg_a, 0)
            s_avg = avg_a.sum()
            if s_avg > 1e-15:
                avg_a = avg_a * norm / s_avg

        # Check all candidates
        for cand in [a, y, avg_a]:
            sc = fast_autocorr(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        # Early stopping check
        if t > 100 and best_score >= prev_best - 1e-12:
            if t % 500 == 0:
                stuck += 1
            if stuck > 2000:
                break
        prev_best = best_score

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_analytical(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with better step selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            # Try moving to a few nearby positions
            offsets = rng.choice(range(1, min(n, 80) + 1), size=min(10, n - 1), replace=False)
            for offset in offsets:
                for sign in [1, -1]:
                    j = (i + sign * offset) % n
                    if j == i or a[i] < norm * 1e-12:
                        continue

                    # Try several transfer fractions
                    for frac in [0.5, 0.3, 0.1, 0.7]:
                        d = a[i] * frac
                        if d <= 0:
                            continue
                        a_test = a.copy()
                        a_test[i] -= d
                        a_test[j] += d
                        if a_test[i] < -1e-12:
                            continue
                        a_test = np.maximum(a_test, 0)
                        s_test = a_test.sum()
                        if s_test < 1e-12:
                            continue
                        a_test = a_test * norm / s_test
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < best_score - 1e-10:
                            a = a_test
                            conv = conv_test
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            break

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PairCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def multi_start_smooth(a0, time_limit, rng, n_starts=3, verbose=True):
    """Multiple perturbed starts with smooth optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = a0.copy()
    best_score = fast_autocorr(a0).max()
    t0 = time.monotonic()
    t_per_start = time_limit / n_starts

    for s in range(n_starts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        t_use = min(t_per_start, remaining)
        if t_use < 5:
            break

        # Perturb
        noise_scale = 0.05 * (s + 1) * best_a.std()
        perturbed = best_a + rng.normal(0, noise_scale, n)
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed * norm / perturbed.sum()

        a_out, sc = smooth_optimize_improved(perturbed, t_use, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    MultiStart {s}: improved to {best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)

    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = smooth_optimize_improved(a_norm, t2, verbose=True)

    # === Phase 3: LP Nesterov round 2 (15% time) ===
