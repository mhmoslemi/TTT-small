"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral (L2) warm-up with FISTA
2. Smooth log-sum-exp FISTA with adaptive temperature
3. Pair coordinate descent with diverse move types
4. LP polish and multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def cross_corr_fft(x, y, n_out):
    """Compute cross-correlation sum_m x[m] * y[m-j] for j=0..n_out-1 via FFT."""
    size = len(x) + len(y) - 1
    fft_size = 1 << (size - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * np.conj(Y), fft_size)[:n_out]

def smooth_grad(a, w):
    """Gradient of T*log(sum exp(conv/T)) = 2 * cross_corr(w, a)."""
    n = len(a)
    return 2.0 * cross_corr_fft(w, a, n)

def l2_grad(a):
    """Gradient of sum conv[k]^2 = 4 * cross_corr(conv, a)."""
    n = len(a)
    conv = fast_autocorr(a)
    return 4.0 * cross_corr_fft(conv, a, n)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_fISTA(a0, T_schedule, time_limit, verbose=True):
    """
    Minimize T*log(sum exp(conv/T)) using FISTA with backtracking.
    T_schedule: list of temperatures (decreasing).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_conv_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        n_steps = max(50, min(2000, int(remaining * 5)))
        
        # FISTA initialization
        y = a.copy()
        k = 1
        L = norm / (n * T * 0.1)  # initial Lipschitz estimate
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute gradient at y
            conv_y = fast_autocorr(y)
            c_max = conv_y.max()
            exp_c = np.exp((conv_y - c_max) / T)
            w = exp_c / exp_c.sum()
            grad_y = smooth_grad(y, w)
            
            # FISTA step
            x_new = y - grad_y / L
            
            # Backtracking (simplified: just ensure non-negativity and correct sum)
            x_new = project_simplex(x_new, norm)
            
            # Check for divergence
            if not np.all(np.isfinite(x_new)):
                L *= 2
                x_new = y - grad_y / L
                x_new = project_simplex(x_new, norm)
            
            # Nesterov momentum
            k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k * k))
            y_new = x_new + ((k - 1.0) / k_new) * (x_new - y)
            y_new = project_simplex(y_new, norm)
            
            if not np.all(np.isfinite(y_new)):
                y_new = x_new.copy()
            
            k = k_new
            y = y_new
            a = x_new
            
            # Track best
            score = fast_autocorr(a).max()
            if score < best_conv_max:
                best_conv_max = score
                best_a = a.copy()
            
            score_y = fast_autocorr(y).max()
            if score_y < best_conv_max:
                best_conv_max = score_y
                best_a = y.copy()
            
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"      T={T:.4f} it={it}: best={best_conv_max:.8f} (t={elapsed:.0f}s)")
        
        # Reset FISTA state between temperatures
        a = project_simplex(a, norm)
    
    return best_a, best_conv_max

def l2_fISTA(a0, time_limit, verbose=True):
    """
    Minimize sum(conv[k]^2) using FISTA.
    This is the spectral/kurtosis warm-up.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)

            break

        k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k * k))
        y_new = x_new + ((k - 1.0) / k_new) * (x_new - y)
        y_new = project_simplex(y_new, norm)

    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=True):
    """
    Pair coordinate descent with line search and diverse moves.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    n_passes = 0
    no_improve = 0
    window = max(10, n // 20)
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            i = perm[idx]
            if a[i] < norm * 1e-8:
                continue
            
            # Pick nearby and far targets
            targets = set()
            for d in range(1, min(window, n) + 1):
                targets.add((i + d) % n)
                targets.add((i - d) % n)
            # Add a few far targets
            for _ in range(3):
                targets.add(rng.integers(0, n))
            
            for j in targets:
                if j == i or a[i] < norm * 1e-8:
                    continue
                
                # Try moving mass from i to j
                lo, hi = 0.0, a[i] * 0.99
                best_d = 0.0
                best_val = conv.max()
                
                for _ in range(6):
                    mid = (lo + hi) / 2
                    if mid < norm * 1e-10:
                        lo = mid
                        continue
                    a_t = a.copy()
                    a_t[i] -= mid
                    a_t[j] += mid
                    a_t = a_t * norm / a_t.sum()
                    conv_t = fast_autocorr(a_t)
                    if conv_t.max() < best_val - 1e-10:
                        best_val = conv_t.max()
                        best_d = mid
                        lo = mid
                    else:
                        hi = mid
                
                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    if conv.max() < best_score - 1e-10:
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PC pass={n_passes} (t={elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss
