def fast_conv(a):
    """Full convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):

    s = a.sum()
    if s < 0.01:

    c = fast_conv(a)
    return float(2.0 * len(a) * c.max() / (s * s))

    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * (target if target else 1.0) / len(a)
    if target is not None:
        return a * target / s
    return a / s


def build_T(a, indices):
    """Build constraint matrix rows for given autocorrelation indices."""
    n = len(a)
    m = len(indices)
    T = np.zeros((m, n))
    for row, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[row, j] = a[k - j]
    return T


def solve_lp_maxsum(a, M, active_idx, solver):
    """Solve max sum(b) s.t. a*b <= M, b >= 0."""
    n = len(a)
    T = build_T(a, active_idx)
    b = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(b.value, 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def solve_lp_minmax(a, active_idx, solver, target_sum):
    """Solve min t s.t. a*b <= t, sum(b) = target_sum, b >= 0."""
    n = len(a)
    T = build_T(a, active_idx)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    prob = cp.Problem(cp.Minimize(t), [T @ b <= t, cp.sum(b) == target_sum, b >= 0])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optotal_inaccurate'):
        return None, None
    val = np.maximum(b.value, 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def lp_nesterov_optimize(a0, time_limit, solver, seed=42, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive switches."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    y = a.copy()
    
    best_a = a.copy()
    best_peak = fast_conv(a).max()
    
    t0 = time.monotonic()
    stall = 0
    active_set = None
    use_minmax = 0
    
    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv(y)
        M = conv_y.max()
        if M < 1e-14:
            break
        
        # Update active set
        if active_set is None or it % 30 == 0:
            # Use peaks that are close to maximum
            frac = 0.05 + 0.03 * np.exp(-it / 200.0)
            thresh = M * (1.0 - frac)
            active_set = np.where(conv_y >= thresh)[0]
            if len(active_set) < 5:
                active_set = np.where(conv_y >= M * 0.6)[0]
            if len(active_set) > int(0.7 * size):
                active_set = np.where(conv_y >= M * 0.85)[0]
        
        # Alternate between max-sum and min-max LP
        use_minmax = (it % 4 == 0)
        
        if use_minmax:
            b_raw, _ = solve_lp_minmax(y, active_set, solver, norm)
        else:
            b_raw = solve_lp_maxsum(y, M, active_set, solver)
        
        if b_raw is None:
            # Retry with full active set
            if it % 10 == 0:
                active_set = None
                b_raw = solve_lp_maxsum(y, M, np.arange(size), solver)
            if b_raw is None:
                continue
        
        g = normalize(b_raw, norm)
        
        # Check and fix constraint violations
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                viol = fast_conv(y)[:0]  # placeholder
                # Quick check: compute a*g at inactive positions
                conv_ig = fast_conv(y)  # reuse
                # Actually need a*g not a*a, let's compute properly
                # For efficiency, skip this check most of the time
                pass
        
        # Nesterov momentum update
        alpha = max(2.0 / (it + 2.0), 0.05)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)
        
        # Momentum extrapolation
        beta = min((it - 1.0) / (it + 2.0), 0.9)
        a_new = y_new + beta * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, norm)
        
        prev_y = y
        y = y_new
        a = a_new
        
        # Track best
        peak_y = fast_conv(y).max()
        if peak_y < best_peak:
            best_peak = peak_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        peak_a = fast_conv(a).max()
        if peak_a < best_peak:
            best_peak = peak_a
            best_a = a.copy()
            stall = 0
        
        # Restart on stall
        if stall > 60:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall = 0
        
        if verbose and it % 100 == 0:
            print(f"    LPN it={it} ({time.monotonic()-t0:.0f}s): peak={best_peak:.10f}")
    
    return best_a, best_peak


def lse_gradient_step(a, conv_a, T):
    """Compute log-sum-exp smoothed gradient of peak autocorrelation.
    
    Returns gradient vector (unnormalized, before projection).
    """
    n = len(a)
    size = 2 * n - 1
    # Softmax weights
    shifted = conv_a - conv_a.max()
    weights = np.exp(T * shifted)
    weights /= weights.sum()
    
    # Gradient: (2/T) * sum_k w_k * P_k * a
    # P_k * a has entries (P_k a)[i] = a[k-i] if 0 <= k-i < n
    # This is equivalent to cross-correlation
    # grad[i] = (2/T) * sum_k w_k * a[k-i] (when valid)
    # = (2/T) * (w_reversed * a)[i] in appropriate convolution sense
    
    # Compute via FFT: grad[i] = (2/T) * sum_{k} w[k] * a[k-i]
    # This is the cross-correlation of w and a
    # corr(w, a)[i] = sum_k w[k] * a[k-i]
    # = sum_j w[j] * a[j-i] (substituting k=j)
    # This is convolution of w with reversed a, evaluated at specific points
    
    # More directly: grad[i] = (2/T) * sum_{k=max(i,0)}^{min(i+n-1, 2n-2)} w[k] * a[k-i]
    # Let l = k-i, then l ranges from max(0,-i) to min(n-1, 2n-2-i)
    # Since i is in [0,n-1]: l from 0 to min(n-1, 2n-2-i)
    # So grad[i] = (2/T) * sum_{l=0}^{min(n-1, 2n-2-i)} w[i+l] * a[l]
    
    # This is the cross-correlation of w[0:2n-1] with a[0:n-1]
    # Specifically: grad * T/2 = corr(w, a) where corr is defined as
    # (corr(w,a))[i] = sum_l w[i+l] * a[l]
    
    # We can compute this via FFT:
    # Let W = w padded to size 4n, A = a padded
    # Then corr = irfft(conj(rfft(W)) * rfft(A)) ... 
    # Actually, let's just do it directly with a small convolution
    
    # corr[i] = sum_{l=0}^{n-1} w[i+l] * a[l] (with w out of range = 0)
    # This is the cross-correlation which = conv(reversed_w, a) at appropriate shifts
    # Or: corr = conv(w, reversed_a) ... 
    
    # Simplest: use the fact that this is a matrix-vector product
    # K[i,l] = w[i+l] for valid i+l
    # grad = (2/T) * K @ a
    
    # For efficiency with n up to ~3000, direct computation is O(n^2) which might be slow
    # Use FFT instead:
    # corr[i] = sum_l w[i+l] * a[l]
    # = (w * a_rev)[n-1+i] where a_rev[l] = a[n-1-l] and * is convolution
    # Wait: (w * a_rev)[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-m+j]
    # We want sum_l w[i+l] * a[l]. Let j = i+l, l = j-i:
    # = sum_j w[j] * a[j-i] = (w * a)[i] in the cross-correlation sense
    
    # Actually the cleanest way: 
    # grad[i] = sum_l w[i+l] * a[l] for l=0..n-1 (with w[j]=0 for j out of [0,2n-2])
    # This is the "anti-convolution" or cross-correlation
    
    # Use FFT: pad w to length >= 3n, convolve with a
    fft_len = 1 << (3 * n - 1).bit_length()
    W = np.zeros(fft_len)
    A = np.zeros(fft_len)
    W[:size] = weights
    A[:n] = a
    # corr[i] = sum_l W[i+l] * A[l]
    # = sum_j W[j] * A[j-i] where j = i+l
    # This is the cross-correlation of W and A
    # = irfft(conj(rfft(W)) * rfft(A)) ... no
    # cross_corr(W, A)[i] = sum_j W[j] * A[j-i] = (reverse(W) * A)[i + len(W) - 1]
    
    # Let's use: cross_corr = irfft(conj(rfft(W)) * rfft(A))
    # (conj(rfft(W)) * rfft(A)) gives the cross-correlation in frequency domain
    # cross_corr[m] = sum_j W[j+m] * A[j] ... 
    # Actually: if X = rfft(W), Y = rfft(A), then irfft(conj(X)*Y)[m] = sum_j W[j+m]*A[j]
    # That's exactly what we want! (with circular correlation, but we padded enough)
    
    corr = irfft(np.conj(rfft(W, fft_len)) * rfft(A, fft_len), fft_len)
    grad = (2.0 / T) * corr[:n]
    
    return grad, weights


def lse_fista(a0, time_limit, seed=0, verbose=True):
    """Log-sum-exp smoothed gradient with FISTA acceleration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    # Temperature schedule: start high, anneal down
    T_start = 5.0
    T_end = 200.0
    total_iters_estimate = int(time_limit * 3)  # rough estimate of iterations possible
    
    x = a.copy()
    y_var = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    k = 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Anneal temperature
        progress = min((time.monotonic() - t0) / time_limit, 1.0)
        T = T_start + (T_end - T_start) * progress ** 0.7
        
        # Compute smoothed gradient
        conv_y = fast_conv(y_var)
        grad, weights = lse_gradient_step(y_var, conv_y, T)
        
        # Lipschitz estimate: L ~ 4/T (conservative)
        L = 4.0 / T + 0.1
        step = 1.0 / L
        
        # FISTA update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        x_new = y_var - step * grad
        x_new = np.maximum(x_new, 0.0)
        x_new = normalize(x_new, norm)
        
        y_new = x_new + ((k - 1.0) / t_k) * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)
        
        x = x_new
        y_var = y_new
        k += 1
        
        # Evaluate and track best
        sc = fast_conv(y_var).max()
        if sc < best_score:
            best_score = sc
            best_a = y_var.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Reset momentum
            y_var = x.copy()
            no_improve = 0
        
        if verbose and k % 500 == 0:
            print(f"    LSEF k={k} ({time.monotonic()-t0:.0f}s): T={T:.1f}, peak={best_score:.10f}")
    
    return best_a, best_score


def subgrad_fista(a0, time_limit, seed=0, verbose=True):
    """Subgradient method on the max autocorrelation peak with FISTA."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    x = a.copy()
    y_var = a.copy()
    best_a = a.copy()
    best_peak = fast_conv(a).max()
    t0 = time.monotonic()
    k = 1
    
    while time.monotonic() - t0 < time_limit:
        conv_y = fast_conv(y_var)
        M = conv_y.max()
        
        # Find active peaks (top few)
        top_k = min(10, len(conv_y))
        active = np.argsort(conv_y)[-top_k:]
        
        # Subgradient: average of gradients at active peaks
        # grad[i] for peak k: 2*a[k-i] (when valid)
        grad = np.zeros(n)
        for kk in active:
            for i in range(n):
                j = kk - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= top_k
        
        # Adaptive step size (Polyak-like)
        # We want to minimize M, current value is M
        # Lower bound is at least ||a||^2 / n (rough)
        lb = (a @ a) / n
        step = (M - lb) / (grad @ grad + 1e-10)
        step = min(step, 1.0 / M * 0.5)
        
        # FISTA
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        x_new = y_var - step * grad
        x_new = np.maximum(x_new, 0.0)
        x_new = normalize(x_new, norm)
        
        y_new = x_new + ((k - 1.0) / t_k) * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)
        
        x = x_new
        y_var = y_new
        k += 1
        
        peak = fast_conv(y_var).max()
        if peak < best_peak:
            best_peak = peak
            best_a = y_var.copy()
        
        if verbose and k % 500 == 0:
            print(f"    SGF k={k} ({time.monotonic()-t0:.0f}s): peak={best_peak:.10f}")
    
    return best_a, best_peak


def coord_descent_fast(a0, time_limit, seed=0, verbose=True):
    """Fast coordinate descent using analytic gradient structure."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_conv = fast_conv(a)
    best_peak = best_conv.max()
    t0 = time.monotonic()
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try scaling a[i] by various factors
            ai = a[i]
            if ai < 1e-10:
                # Try small positive values
                candidates = [norm * 0.001, norm * 0.005]
            else:
                candidates = [ai * f for f in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]]
                candidates.append(0.0)
            
            best_val = best_peak
            best_cand = ai
            
            for cv in candidates:
                a_try = a.copy()
                a_try[i] = cv
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try * norm / s_try
                pk = fast_conv(a_try).max()
                if pk < best_val - 1e-12:
                    best_val = pk
                    best_cand = cv
            
            if best_cand != ai:
                a[i] = best_cand
                a = a * norm / a.sum()
                new_conv = fast_conv(a)
                new_peak = new_conv.max()
                if new_peak < best_peak - 1e-13:
                    best_peak = new_peak
                    best_a = a.copy()
                    best_conv = new_conv
                    improved = True
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 8:
                break
        else:
            no_improve_passes = 0
        
        if verbose:
            print(f"    CD ({time.monotonic()-t0:.0f}s): peak={best_peak:.10f}")
    
    return best_a, best_peak


def structured_init(n, style, rng=None):
    """Generate structured initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.35 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-10.0 * (x - 0.25) ** 2) + 0.7 * np.exp(-10.0 * (x - 0.75) ** 2)
    elif style == 4:
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif style == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif style == 6:
        a = 1.0 + 0.3 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 7:
        # Raised cosine with different parameters
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb(a, rng, scale):
    """Perturb sequence with multiplicative noise."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

        n_cur = len(best_seq)
        scale = 0.02 + 0.015 * r
        pert = perturb(best_seq, rng, scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        # Alternate between LP and LSE
        if r % 2 == 0:
            a_out, _ = lp_nesterov_optimize(a_norm, t5, solver, seed=100 + r, verbose=False)
        else:
            a_out, _ = lse_fista(a_norm, t5, seed=200 + r, verbose=False)
        try_update(a_out, f"restart{r}_n{n_cur}")
    
    remaining_budget = time_budget - (time.monotonic() - t_start)
    
    # === Phase 6: Different lengths (15% time) ===
    print(f"\n[Phase 6] Multi-length exploration")
    t6_total = min(remaining_budget * 0.15, 200.0)
    lengths_to_try = []
    base_n = len(best_seq)
    for factor in [1.25, 1.5, 0.8, 1.1]:
        new_n = int(base_n * factor)
        if new_n >= 500 and new_n <= 5000 and new_n != base_n:
            lengths_to_try.append(new_n)
    
    for new_n in lengths_to_try[:3]:
        remaining_budget = time_budget - (time.monotonic() - t_start)
        if remaining_budget < 80:
            break
        t6 = min(t6_total / max(len(lengths_to_try[:3]), 1), remaining_budget - 40)
        if t6 < 20:
            break
        init_long = interpolate_to(best_seq, new_n)
        a_norm = init_long * np.sqrt(2.0 * new_n) / init_long.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t6, solver, seed=500 + new_n, verbose=False)
        try_update(a_out, f"len_{new_n}")
    
    remaining_budget = time_budget - (time.monotonic() - t_start)
    
    # === Phase 7: Final LP polish (all remaining time with buffer) ===
    print(f"\n[Phase 7] Final polish n={len(best_seq)}")
    t7 = max(remaining_budget * 0.5, 60.0)
    t7 = min(t7, remaining_budget - 50)
    if t7 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t7, solver, seed=999, verbose=True)
        try_update(a_out, f"final_lp_n{n_cur}")
    
    remaining_budget = time_budget - (time.monotonic() - t_start)
    
    # === Phase 8: Final LSE + CD ===
    if remaining_budget > 60:
        print(f"\n[Phase 8] Final LSE + CD")
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        t8a = min(remaining_budget * 0.4, remaining_budget - 30)
        if t8a > 15:
            a_out, _ = lse_fista(a_norm, t8a, seed=777, verbose=False)
            try_update(a_out, "final_lse")
        
        remaining_budget = time_budget - (time.monotonic() - t_start)
        if remaining_budget > 30:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = coord_descent_fast(a_norm, remaining_budget - 20, seed=888, verbose=False)
            try_update(a_out, "final_cd")
    
    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
