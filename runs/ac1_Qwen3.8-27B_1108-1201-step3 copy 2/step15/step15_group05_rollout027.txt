"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. SMAC (smooth max) accelerated proximal gradient (FISTA) - main workhorse
2. LP best-response with Nesterov acceleration
3. Exact coordinate descent with incremental autocorrelation updates
4. Multi-iteration refinement with perturbations
Key improvements: SMAC-FISTA avoids expensive LP solves, coordinate descent

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to(a, target_sum):
    """Normalize a to havetarget_sum sum, clipping negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s


def project_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Binary search for theta
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (css[-1] - c))[0][-1]
    theta = (css[rho] - c) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smac_gradient(a, conv, eps, n):
    """Compute gradient of SMAC objective at point a.
    Returns grad[j] = sum_k w_k * (a[k-j] + a[j-k])
    where w_k = exp(eps*conv[k]) / sum(exp(eps*conv[m]))
    """
    # Compute weights (shift to avoid overflow)
    log_weights = eps * conv - eps * conv.max()
    w = np.exp(log_weights)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    # Compute term2[j] = sum_k w[k] * a[j-k] = (a*w)[j]
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv_aw = irfft(A * W, fft_size)[:2 * n - 1]
    term2 = conv_aw[:n]

    # Compute term1[j] = sum_k w[k] * a[k-j]
    a_rev = a[::-1]
    A_rev = rfft(a_rev, fft_size)
    conv_w_ar = irfft(W * A_rev, fft_size)[:2 * n - 1]
    term1 = conv_w_ar[n - 1:2 * n - 1]

    return term1 + term2


def smac_value(conv, eps):
    """Compute SMAC value: (1/eps)*log(sum(exp(eps*conv)))."""
    max_c = conv.max()
    if max_c < -1e300:
        return 0.0
    shifted = eps * (conv - max_c)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (1.0 / eps) * (np.log(np.sum(np.exp(shifted))) + eps * max_c)

def lp_best_response_v2(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a_fixed[k - j]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def coordinate_descent_v2(a0, time_limit, verbose=False):

                break

                new_conv = conv + delta

                    best_new = new_val

            if abs(best_new - old) > 1e-12:

                    new_sum = target_sum - old + mid
                    if new_sum < 1e-15:

                    obj_mid = new_conv.max() * (target_sum / new_sum) ** 2
                    obj_lo = 0
                    delta_lo = np.zeros(size)
                    lo_d2 = j
                    hi_d2 = min(j + n, size)
                    if hi_d2 > lo_d2:

                        hi_b = mid
                best_new = (lo_b + hi_b) / 2.0

def perturb_sequence(a, rng, scale=0.03):
    """Add multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 6: Perturbation + FISTA restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Perturb+FISTA, remaining={remaining:.0f}s ===")
    n_restarts = 3
    t6_total = min(remaining * 0.20, 180.0)
    t6_each = t6_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t6 = min(t6_each, remaining - 20)
        if t6 < 20:
            break
        pert = perturb_sequence(best_seq, rng, scale=0.02 + 0.01 * r)
        a_norm = normalize_to(pert, np.sqrt(2.0 * n_cur))
        a_out, sc_out = fista_smac(a_norm, t6, verbose=False, seed=100 + r * 7)
        try_update(a_out, f"pert{r}_fista_n{n_cur}")

    # === Phase 7: Final Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final CD n={n_cur} ===")
        t7 = min(remaining * 0.35, 120.0)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, sc_out = coordinate_descent_v2(a_norm, t7, verbose=True)
        try_update(a_out, f"cd_final_n{n_cur}")

    # === Phase 8: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 8: Try n=2000 ===")
        n_alt = 2000
        a_interp = interpolate_sequence(best_seq, n_alt)
        a_norm = normalize_to(a_interp, np.sqrt(2.0 * n_alt))
        t8 = min(remaining * 0.4, 80.0)
        if t8 > 30:
            a_out, sc_out = fista_smac(a_norm, t8, verbose=False, seed=999)
            try_update(a_out, f"fista_n{n_alt}")

    # === Final polish: quick CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, sc_out = coordinate_descent_v2(a_norm, min(remaining * 0.5, 30.0), verbose=False)
        try_update(a_out, "cd_polish")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        fallback = np.ones(1600) / 1600
        best_score = evaluate_safe(fallback)
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
