def build_constraint_matrix(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_best_response(a, M, solver, active_idx):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = build_constraint_matrix(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_min_max(a, solver, active_idx, target_sum):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_constraint_matrix(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == target_sum, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')

def block_cd(a0, time_limit, rng, verbose=False, block_size=8):
    """Block coordinate descent: optimize block_size coordinates at a time."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    n_blocks = n // block_size
    if n_blocks == 0:
        return a, best_score

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n_blocks)

        for b_idx in perm:
            if time.monotonic() - t0 > time_limit:
                break

            start = b_idx * block_size
            end = min(start + block_size, n)
            coords = np.arange(start, end)

            # Simple: try random redistribution within block
            block_vals = a[coords].copy()
            block_sum = block_vals.sum()
            if block_sum < 1e-15:
                continue

            # Try proportional perturbation
            for trial in range(5):
                noise = 1.0 + 0.3 * rng.standard_normal(block_size)
                noise = np.maximum(noise, 0.1)
                new_block = block_vals * noise
                new_block = new_block * block_sum / new_block.sum()

                a_try = a.copy()
                a_try[coords] = new_block
                sc = fast_autocorr(a_try).max()
                if sc < best_score - 1e-13:
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    pass_improved = True
                    block_vals = a[coords].copy()
                    break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    BCD ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)

    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * x * 5.0)
        a = np.maximum(a, 0.0)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = nesterov_lp(a_norm, t9, solver, verbose=False, seed=456,
                                   alpha_floor=0.02, momentum_cap=0.99,
                                   active_tol_start=0.01, active_tol_min=0.002)
            try_update(a_out, "lp_fine")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
