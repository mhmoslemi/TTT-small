def smoothed_max_mirror_descent(a0, time_limit, betas=None, verbose=True):
    """Mirror descent with smoothed max objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    
    if betas is None:
        betas = [10, 20, 50, 100, 200, 500]
    
    for beta in betas:
        eta = 1.0 / (np.sqrt(beta) + 1)
        for t in range(1, 100000):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov step
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
            
            # Compute gradient of smoothed max
            grad, conv_val = smoothed_max_grad(y, beta)
            
            # Mirror descent step (multiplicative update)
            eta_t = eta / (np.sqrt(t) + 1)
            log_y = np.log(y)
            log_new = log_y - eta_t * grad
            log_new -= log_new.max()  # numerical stability
            a_new = np.exp(log_new)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            
            x_prev = x
            x = a_new
            
            # Check improvement
            M = fast_conv(x).max()
            if M < best_obj:
                best_obj = M
                best_a = x.copy()
            
            if verbose and t % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"  SMD beta={beta} t={t} ({elapsed:.1f}s): max_conv={best_obj:.8f}")
    
    return best_a, best_obj

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax_weights(conv_val, beta):
    shifted = conv_val - conv_val.max()
    exp_vals = np.exp(beta * shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        return np.ones(len(conv_val)) / len(conv_val)
    return exp_vals / s


def smoothed_max_gradient(a, beta):
    """Compute exact gradient of (1/beta)*log(sum exp(beta*conv(a,a)(k)))."""
    n = len(a)
    conv_val = fast_conv(a)
    p = softmax_weights(conv_val, beta)
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
    # Using FFT: g[i] = conv(a, p_rev)[2n-2-i]
    p_rev = p[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_rev, fft_size)
    conv_result = np.fft.irfft(A * P, fft_size)[:3 * n - 2]
    g = conv_result[n - 1:2 * n - 1][::-1]
    grad = 2.0 * g
    return grad, conv_val, p


def smoothed_max_mirror_descent(a0, time_limit, verbose=True, seed_betas=None):
    """Mirror descent with smoothed max objective and adaptive beta."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()

    if seed_betas is None:
        seed_betas = [5, 10, 20, 40, 80, 150, 300, 600]

    x = a.copy()
    x_prev = a.copy()
    global_t = 0

    for beta in seed_betas:
        if time.monotonic() - t0 > time_limit:
            break
        beta_start = time.monotonic()
        gamma = 3.0 / (1.0 + 0.1 * beta)
        running_avg = x.copy()
        avg_count = 1

        for t in range(1, 200000):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - beta_start > time_limit * 0.3:
                break

            global_t += 1
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

            grad, conv_val, p = smoothed_max_gradient(y, beta)

            eta = gamma / (np.sqrt(t) + 1.0)
            log_y = np.log(y)
            log_new = log_y - eta * grad
            log_new -= log_new.max()
            a_new = np.exp(log_new)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new

            x_prev = x
            x = a_new

            M_new = fast_conv(x).max()
            if M_new < best_obj:
                best_obj = M_new
                best_a = x.copy()

            running_avg += x
            avg_count += 1
            if avg_count % 500 == 0:
                avg = running_avg / avg_count
                avg = np.maximum(avg, 0.0)
                s_avg = avg.sum()
                if s_avg > 1e-15:
                    avg = avg / s_avg
                    M_avg = fast_conv(avg).max()
                    if M_avg < best_obj:
                        best_obj = M_avg
                        best_a = avg.copy()

            if verbose and t % 5000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_obj
                print(f"    SMD beta={beta} t={t} ({elapsed:.1f}s): J={J:.6f}")

        # Update x for next beta
        x = best_a.copy()
        x_prev = best_a.copy()
        running_avg = best_a.copy()
        avg_count = 1

    return best_a, best_obj


def build_constraint_matrix(a):
    """Build the constraint matrix T where T[k,j] = a[k-j] if valid."""
    n = len(a)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            T[k, j] = a[k - j]
    return T
