        improved = False

                    f1 = fast_conv(a1).max()

                    f2 = fast_conv(a2).max()

                no_imp = 0

        if not improved:
            no_imp += 1
            if no_imp >= 4:
                break

        if verbose and passes % 3 == 0:
            print(f"    CD pass={passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=False):
    """Pair swap coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    best_score = fast_conv(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_imp = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            ai, aj = a[i], a[j]
            if ai < 1e-14 * norm and aj < 1e-14 * norm:
                continue

            cur_score = fast_conv(a).max()
            best_d = 0.0
            best_s = cur_score

            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                d = min(ai * frac, ai)
                at = a.copy(); at[i] = ai - d; at[j] = aj + d
                at = at * norm / at.sum()
                s = fast_conv(at).max()
                if s < best_s - 1e-13:
                    best_s = s; best_d = d

                d2 = min(aj * frac, aj)
                at = a.copy(); at[i] = ai + d2; at[j] = aj - d2
                at = at * norm / at.sum()
                s = fast_conv(at).max()
                if s < best_s - 1e-13:
                    best_s = s; best_d = -d2

            if abs(best_d) > 1e-14:
                at = a.copy()
                if best_d > 0:
                    at[i] -= best_d; at[j] += best_d
                else:
                    at[i] += abs(best_d); at[j] -= abs(best_d)
                at = at * norm / at.sum()
                sc = fast_conv(at).max()
                if sc < best_score - 1e-13:
                    a = at; best_score = sc; best_a = a.copy()
                    improved = True; no_imp = 0

        if not improved:
            no_imp += 1
            if no_imp >= 5:
                break

        if verbose and passes % 5 == 0:
            print(f"    PCD pass={passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def step_init(n, alpha=0.3, orient=1):
    """Step function initialization."""
    x = np.linspace(0, 1, n)
    if orient == 1:
        a = np.where(x < 0.5, 1 + alpha, 1 - alpha)
    else:
        a = np.where(x < 0.5, 1 - alpha, 1 + alpha)
    return a / a.sum()

def ramp_init(n, alpha=0.3):
    """Linear ramp initialization."""
    x = np.linspace(0, 1, n)
    a = 1 + alpha * (2 * x - 1)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def cos_step_init(n, alpha=0.3, phase=0.0):
    """Cosine step initialization."""
    x = np.linspace(0, 1, n)
    a = 1 + alpha * np.cos(np.pi * (x + phase))
    a = np.maximum(a, 0.01)
    return a / a.sum()

def bump_init(n, alpha=0.2, center=0.3):
    """Gaussian bump initialization."""
    x = np.linspace(0, 1, n)
    a = 1 + alpha * np.exp(-20 * (x - center) ** 2)
    return a / a.sum()

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def perturb(a, rng, scale=0.03):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

def propose_candidate():
    """Main search function: finds best sequence minimizing 2n*max(conv)/(sum)^2."""
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    time_budget = 1740.0
    solver = get_best_solver()

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = 1600
    if best_seq is None:
        # Try multiple initializations
        for init_func, name in [(step_init, 'step'), (ramp_init, 'ramp'),
                                 (cos_step_init, 'cos'), (bump_init, 'bump')]:
            for alpha in [0.1, 0.2, 0.3, 0.4]:
                try:
                    cand = init_func(n_base, alpha=alpha)
                    sc = eval_safe(cand)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand
                except Exception:
                    pass
        if best_seq is None:
            best_seq = np.ones(n_base) / n_base
            best_score = eval_safe(best_seq)
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            delta = best_score - sc
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s, d={delta:.6f})")

    def remaining():
        return time_budget - (time.monotonic() - t_start)

    # === Phase 1: Try different initialization shapes at current n ===
    n_cur = len(best_seq)
    remaining_t = remaining()
    if remaining_t > 200:
        print(f"\n=== Phase 1: Shape exploration n={n_cur} ===")
        shapes = [
            ("step1", step_init, {'alpha': 0.25, 'orient': 1}),
            ("step2", step_init, {'alpha': 0.35, 'orient': -1}),
            ("ramp1", ramp_init, {'alpha': 0.2}),
            ("ramp2", ramp_init, {'alpha': 0.35}),
            ("cos1", cos_step_init, {'alpha': 0.25, 'phase': 0.1}),
            ("cos2", cos_step_init, {'alpha': 0.3, 'phase': 0.3}),
        ]
        for name, func, kw in shapes:
            if remaining_t < 100:
                break
            cand = func(n_cur, **kw)
            # Blend with parent
            for w in [0.3, 0.5, 0.7]:
                blended = w * best_seq + (1 - w) * cand
                blended = blended / blended.sum()
                sc = eval_safe(blended)
                if sc < best_score:
                    best_score = sc
                    best_seq = blended.copy()

    # === Phase 2: Main LP Nesterov ===
    remaining_t = remaining()
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} ({remaining_t:.0f}s available) ===")
    t2 = min(remaining_t * 0.40, 500.0)
    if t2 > 60:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, seed=0, verbose=True)
        try_update(a_out, f"lp_n{n_cur}")

    # === Phase 3: Pair CD polish ===
    remaining_t = remaining()
    if remaining_t > 100:
        print(f"\n=== Phase 3: Pair CD n={len(best_seq)} ===")
        t3 = min(remaining_t * 0.12, 150.0)
        if t3 > 30:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = pair_cd(a_norm, t3, rng, verbose=True)
            try_update(a_out, f"pcd_n{n_cur}")

    # === Phase 4: Perturbation + LP restart ===
    remaining_t = remaining()
    if remaining_t > 200:
        print(f"\n=== Phase 4: Perturbation restarts ===")
        n_restarts = 3
        t_each = min(remaining_t * 0.15 / n_restarts, 150.0)
        for r in range(n_restarts):
            if remaining() < 100:
                break
            n_cur = len(best_seq)
            scale = 0.02 + 0.015 * r
            pert = perturb(best_seq, rng, scale)
            a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
            t_r = min(t_each, remaining() - 80)
            if t_r < 40:
                break
            a_out, _ = nesterov_lp(a_norm, t_r, solver, seed=200 + r * 13, verbose=False)
            try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 5: Try different lengths ===
    remaining_t = remaining()
    if remaining_t > 150:
        print(f"\n=== Phase 5: Multi-length exploration ===")
        n_cur = len(best_seq)
        for n_try in [n_cur + 200, n_cur - 200, n_cur + 400]:
            if remaining() < 100:
                break
            if n_try < 500:
                continue
            init_try = interp_seq(best_seq, n_try)
            a_norm = init_try * np.sqrt(2.0 * n_try) / init_try.sum()
            t_try = min(remaining() * 0.12, 120.0)
            if t_try < 40:
                break
            a_out, _ = nesterov_lp(a_norm, t_try, solver, seed=500 + n_try, verbose=False)
            try_update(a_out, f"lp_n{n_try}")

    # === Phase 6: Coordinate descent polish ===
    remaining_t = remaining()
    if remaining_t > 60:
        print(f"\n=== Phase 6: Coord Descent n={len(best_seq)} ===")
        t6 = min(remaining_t * 0.15, 120.0)
        if t6 > 20:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = coord_descent(a_norm, t6, rng, verbose=False)
            try_update(a_out, f"cd_n{n_cur}")

    # === Phase 7: Final LP polish on best ===
    remaining_t = remaining()
    if remaining_t > 80:
        print(f"\n=== Phase 7: Final LP n={len(best_seq)} ===")
        t7 = min(remaining_t * 0.50, remaining_t - 40)
        if t7 > 50:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / a_norm if False else best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = nesterov_lp(a_norm, t7, solver, seed=9999, verbose=True)
            try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final pair CD ===
    remaining_t = remaining()
    if remaining_t > 40:
        print(f"\n=== Phase 8: Final PCD n={len(best_seq)} ===")
        t8 = min(remaining_t * 0.7, 80.0)
        if t8 > 15:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = pair_cd(a_norm, t8, rng, verbose=False)
            try_update(a_out, f"pcd_final_n{n_cur}")

    # === Phase 9: One more short LP if time ===
    remaining_t = remaining()
    if remaining_t > 50:
        print(f"\n=== Phase 9: Final short LP ===")
        t9 = remaining_t - 20
        if t9 > 30:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = nesterov_lp(a_norm, t9, solver, seed=77777, verbose=False)
            try_update(a_out, "lp_last")

    # Final
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
