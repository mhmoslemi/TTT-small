def fast_autocorr(a):

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

def subgradient_step(a, conv, active_lags, lags_weights, eta):
    """
    Compute multi-peak subgradient step.
    grad of conv[k] w.r.t. a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

def fienup_peak_reduction(a0, time_limit, verbose=False, seed=0):
    """Fienup/IRAM peak reduction algorithm."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_to(a0, target_sum)
    
    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_max = best_conv.max()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Target: start at current peak, try to reduce
    M_target = best_max
    t0 = time.monotonic()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        A = rfft(a, fft_size)
        power = A * A  # This is |A|^2 for the convolution
        
        # Compute autocorrelation
        conv = irfft(power, fft_size)[:size]
        
        # Peak reduction: clip the spectrum to have max impulse <= M_target
        # In time domain (autocorrelation domain), clip peaks
        conv_clipped = np.minimum(conv, M_target)
        
        # Go back to frequency domain
        power_new = rfft(conv_clipped, fft_size)
        
        # Take square root (magnitude), keep phase from old A
        # power_new should be real and non-negative for a valid PSD
        # But it might not be, so we take abs
        power_new = np.maximum(power_new, 0)
        mag_new = np.sqrt(power_new)
        phase_old = np.angle(A)
        A_new = mag_new * np.exp(1j * phase_old)
        
        # Back to time domain
        a_new = irfft(A_new, fft_size)[:n]
        
        # Project: non-negative + fixed sum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to(a_new, target_sum)
        
        # Check
        conv_check = fast_autocorr(a_new)
        max_check = conv_check.max()
        
        if max_check < best_max - 1e-12:
            best_max = max_check
            best_a = a_new.copy()
            # Try to lower the target
            M_target = max_check * 0.99
            a = a_new
        else:

            M_target = min(M_target, max_check)
            a = a_new

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def perturb_seq(a, rng, scale=0.03):
    """Perturb a sequence with multiplicative noise."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

def lp_max_sum(a, M, active_idx=None, solver=None):
    """LP: maximize sum(b) s.t. (a*b)(k) <= M for k in active_idx, b >= 0."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    if active_idx is None:
        active_idx = np.arange(size)
    
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    
    b = cp.Variable(n, nonneg=True)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver or cp.SCS, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(b.value, 0.0)
    if val is None or val.sum() < 1e-15:
        return None
    return val

"""
Hybrid optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines multi-peak subgradient descent with Nesterov acceleration,
Fienup/IRAM Fourier-domain peak reduction, LP best-response,
and coordinate descent polishing. Explores multiple lengths.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation of objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def eval_score(a):
    """Full evaluation using evaluate_sequence."""
    a_list = [float(max(0.0, min(1000.0, x))) for x in np.asarray(a, dtype=np.float64)]
    return evaluate_sequence(a_list)


def normalize_to(a, target_sum):
    """Normalize a to have the target sum, keeping non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def get_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")

        except Exception:

    print("  LP solver: SCS (fallback)")

def subgradient_step(y, active_lags, lags_weights):
    """
    Compute multi-peak subgradient.
    grad of conv[k] w.r.t. a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(y)
    grad = np.zeros(n)
    for k, w in zip(active_lags, lags_weights):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = k - np.arange(lo, hi + 1)
            grad[lo:hi + 1] += 2.0 * w * y[idx]
    return grad


def nesterov_subgrad(a0, time_limit, verbose=False, seed=0, tol_start=0.06):
    """Nesterov-accelerated multi-peak subgradient descent."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_to(a0, target_sum)
    best_a = a.copy()
    best_max = fast_autocorr(a).max()

    y = a.copy()
    t = 1
    t0 = time.monotonic()
    stall_count = 0
    stall_reset = 120
    erg_avg = np.zeros(n)
    n_erg = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-15:
            break

        # Active set: lags near the peak
        tol = tol_start + 0.02 * np.exp(-t / 300.0)
        threshold = M * (1.0 - tol)
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 3:
            active = np.where(conv_y >= M * 0.85)[0]
        if len(active) > 60:
            # Keep only the most active
            top_k = np.argsort(conv_y[active])[-40:]
            active = active[top_k]

        # Weights proportional to activity
        active_vals = conv_y[active]
        weights = (active_vals - threshold + 1e-8)
        weights = weights / weights.sum()

        # Adaptive step size
        eta_t = 2.0 / (t + 2.0)
        grad_norm_approx = np.sqrt(2.0) * target_sum / np.sqrt(len(active) + 1)
        eta = eta_t * target_sum / (n * grad_norm_approx + 1e-10)

        # Subgradient
        grad = subgradient_step(y, active, weights)

        # Projected gradient step
        y_new = y - eta * grad
        y_new = normalize_to(y_new, target_sum)

        # Nesterov momentum
        if t == 1:
            x_new = y_new.copy()
        else:
            beta = min((t - 1.0) / (t + 2.0), 0.92)
            x_new = y_new + beta * (y_new - y)
            x_new = normalize_to(x_new, target_sum)

        y = y_new
        a = x_new

        # Track best
        max_a = fast_autocorr(a).max()
        max_y = conv_y_new = fast_autocorr(y).max()
        if max_a < best_max - 1e-13:
            best_max = max_a
            best_a = a.copy()
            stall_count = 0
        elif max_y < best_max - 1e-13:
            best_max = max_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Ergodic average
        erg_avg += a
        n_erg += 1
        if n_erg % 50 == 0:
            avg = normalize_to(erg_avg / n_erg, target_sum)
            sc_avg = fast_autocorr(avg).max()
            if sc_avg < best_max - 1e-13:
                best_max = sc_avg
                best_a = avg.copy()
                stall_count = 0

        # Restart
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            erg_avg = np.zeros(n)
            n_erg = 0
            stall_count = 0

        if verbose and t % 500 == 0:
            print(f"    SubGrad t={t}: best={best_max:.10f}, eta={eta:.2e}")

    return best_a, best_max


def fienup_peak_reduction(a0, time_limit, verbose=False, seed=0):
    """Fienup/IRAM Fourier-domain peak reduction."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_to(a0, target_sum)

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    M_target = best_max

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    t0 = time.monotonic()
    stall = 0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        power = A * A
        conv = irfft(power, fft_size)[:size]

        # Clip peaks in autocorrelation domain
        conv_clipped = np.minimum(conv, M_target)
        power_new = rfft(conv_clipped, fft_size)
        power_new = np.maximum(power_new.real, 0.0)

        # New magnitude, old phase
        mag_new = np.sqrt(power_new)
        phase_old = np.angle(A)
        A_new = mag_new * np.cos(phase_old) + 1j * mag_new * np.sin(phase_old)

        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize_to(a_new, target_sum)

        max_new = fast_autocorr(a_new).max()

        if max_new < best_max - 1e-13:
            best_max = max_new
            best_a = a_new.copy()
            M_target = max_new * 0.995
            a = a_new
            stall = 0
        else:
            a = a_new
            M_target = min(M_target, max_new * 1.001)
            stall += 1
            if stall > 200:
                a = best_a.copy()
                M_target = best_max * 1.005
                stall = 0

        if verbose and it % 1000 == 0:
            print(f"    Fienup it={it}: best={best_max:.10f}")

    return best_a, best_max

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent with multiplier search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_to(a0, target_sum)

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 25
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-14 * target_sum:
                continue

            best_local = best_max
            best_mult = 1.0
            for mult in [0.3, 0.5, 0.7, 0.85, 0.92, 1.08, 1.15, 1.3, 1.6, 2.0]:
                a_try = a.copy()
                a_try[i] = ai * mult
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try * target_sum / s
                cv = fast_autocorr(a_try)
                if cv.max() < best_local - 1e-14:
                    best_local = cv.max()
                    best_mult = mult

            if best_mult != 1.0:
                a[i] = ai * best_mult
                a = a * target_sum / a.sum()
                best_max = best_local
                best_a = a.copy()
                improved = True
                no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and pass_count % 10 == 0:
            print(f"    CD pass={pass_count}: best={best_max:.10f}")

    return best_a, best_max


def pair_cd(a0, time_limit, verbose=False, seed=0):
    """Pair coordinate descent: exchange mass between pairs."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_to(a0, target_sum)

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 20

    while time.monotonic() - t0 < time_limit:
        improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx % n]
            j = perm[(idx + 1) % n]
            if i == j:
                continue
            ai, aj = a[i], a[j]
            if ai < 1e-14 * target_sum and aj < 1e-14 * target_sum:
                continue

            best_local = best_max
            best_i, best_j = ai, aj
            for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                # i -> j
                d = ai * frac
                if d > 0:
                    a_try = a.copy()
                    a_try[i] = ai - d
                    a_try[j] = aj + d
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try = a_try * target_sum / s
                        cv = fast_autocorr(a_try)
                        if cv.max() < best_local - 1e-14:
                            best_local = cv.max()
                            best_i, best_j = ai - d, aj + d
                # j -> i
                d2 = aj * frac
                if d2 > 0:
                    a_try = a.copy()
                    a_try[j] = aj - d2
                    a_try[i] = ai + d2
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try = a_try * target_sum / s
                        cv = fast_autocorr(a_try)
                        if cv.max() < best_local - 1e-14:
                            best_local = cv.max()
                            best_i, best_j = ai + d2, aj - d2

            if best_i != ai or best_j != aj:
                a[i] = best_i
                a[j] = best_j
                a = a * target_sum / a.sum()
                best_max = best_local
                best_a = a.copy()
                improved = True
                no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose:
            pass

    return best_a, best_max


def structured_init(n, style=0):
    """Create structured initializations."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.03):
    """Perturb a sequence with multiplicative noise."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def propose_candidate():
    """Main search function: hybrid optimization for low-peak autocorrelation."""
    solver = get_solver()
    t_global = time.monotonic()
    time_budget = 1740.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_score = p_score
            best_seq = parent.copy()
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        for sty in range(7):
            cand = structured_init(n_init, sty)
            sc = eval_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            el = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={el:.0f}s, delta={old - sc:.6f})")

    # Phase 1: Subgradient (25% of budget)
    remaining = time_budget - (time.monotonic() - t_global)
    alloc = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: SubGradient n={len(best_seq)}, budget={alloc:.0f}s ===")
    if alloc > 30:
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = nesterov_subgrad(a_norm, alloc, verbose=True, seed=0)
        try_update(a_out, "subgrad1")

    # Phase 2: Fienup (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    alloc = min(remaining * 0.10, 200.0)
    print(f"\n=== Phase 2: Fienup n={len(best_seq)}, budget={alloc:.0f}s ===")
    if alloc > 30:
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = fienup_peak_reduction(a_norm, alloc, verbose=True, seed=1)
        try_update(a_out, "fienup1")

    # Phase 3: LP Nesterov (30%)
    remaining = time_budget - (time.monotonic() - t_global)
    alloc = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)}, budget={alloc:.0f}s ===")
    if alloc > 60:
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = nesterov_lp(a_norm, alloc, solver=solver, verbose=True, seed=0)
        try_update(a_out, "lp1")

    # Phase 4: Perturb + SubGrad
    remaining = time_budget - (time.monotonic() - t_global)
    alloc = min(remaining * 0.10, 150.0)
    n_restarts = 2
    if alloc > 40:
        per_restart = alloc / n_restarts
        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t_r = min(per_restart, remaining - 40)
            if t_r < 20:
                break
            print(f"\n=== Phase 4.{r+1}: Perturb+SubGrad ===")
            n_cur = len(best_seq)
            pert = perturb_seq(best_seq, rng, scale=0.02 + 0.02 * r)
            a_norm = normalize_to(pert, np.sqrt(2.0 * n_cur))
            a_out, _ = nesterov_subgrad(a_norm, t_r, verbose=False, seed=100 + r * 13)
            try_update(a_out, f"pert_subgrad{r}")

    # Phase 5: Different length (longer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_new = len(best_seq) + 200
    alloc = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: LP n={n_new}, budget={alloc:.0f}s ===")
    if alloc > 40:
        init_long = interpolate_seq(best_seq, n_new)
        a_norm = normalize_to(init_long, np.sqrt(2.0 * n_new))
        a_out, _ = nesterov_lp(a_norm, alloc, solver=solver, verbose=True, seed=77)
        try_update(a_out, f"lp_long{n_new}")

    # Phase 6: Different length (shorter)
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(1000, len(best_seq) - 200)
    alloc = min(remaining * 0.08, 100.0)
    if n_short != len(best_seq) and alloc > 30:
        print(f"\n=== Phase 6: LP n={n_short}, budget={alloc:.0f}s ===")
        init_short = interpolate_seq(best_seq, n_short)
        a_norm = normalize_to(init_short, np.sqrt(2.0 * n_short))
        a_out, _ = nesterov_lp(a_norm, alloc, solver=solver, verbose=False, seed=33)
        try_update(a_out, f"lp_short{n_short}")

    # Phase 7: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    alloc = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 7: CD n={len(best_seq)}, budget={alloc:.0f}s ===")
    if alloc > 20:
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = coordinate_descent(a_norm, alloc, verbose=True, seed=5)
        try_update(a_out, "cd1")

    # Phase 8: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    alloc = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 8: PairCD n={len(best_seq)}, budget={alloc:.0f}s ===")
    if alloc > 15:
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = pair_cd(a_norm, alloc, verbose=False, seed=8)
        try_update(a_out, "pcd1")

    # Phase 9: Final Fienup on best
    remaining = time_budget - (time.monotonic() - t_global)
    alloc = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 9: Fienup final n={len(best_seq)}, budget={alloc:.0f}s ===")
    if alloc > 30:
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = fienup_peak_reduction(a_norm, alloc, verbose=False, seed=99)
        try_update(a_out, "fienup_final")

    # Phase 10: Final CD
    remaining = time_budget - (time.monotonic() - t_global)
    alloc = min(remaining * 0.15, 100.0)
    print(f"\n=== Phase 10: CD final n={len(best_seq)}, budget={alloc:.0f}s ===")
    if alloc > 15:
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = coordinate_descent(a_norm, alloc, verbose=False, seed=123)
        try_update(a_out, "cd_final")

    # Final check
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
        return [float(x) for x in best_seq]
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
