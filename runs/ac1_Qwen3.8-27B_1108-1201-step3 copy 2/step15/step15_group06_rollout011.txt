def smooth_obj_and_grad(a, tau):
    """Compute smoothed objective value and gradient using log-sum-exp."""
    n = len(a)
    conv = fast_autocorr(a)

    c_max = conv.max()
    tc = tau * (conv - c_max)

    F = c_max + np.log(total) / tau

def build_active_constraints(a, active_idx):
    """Build constraint matrix for LP: T @ b <= M for active lags."""
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_direction(a, M, solver, active_idx):
    """Solve max-sum LP for Frank-Wolfe direction."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = build_active_constraints(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val
