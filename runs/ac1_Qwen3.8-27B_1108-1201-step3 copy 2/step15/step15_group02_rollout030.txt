def lse_grad_and_weights(a, T, fft_size):
    """Compute LSE-smoothed max gradient and weights for conv(a,a)."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_conv_self(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n), np.ones(size) / size, max_c
    shifted = (conv - max_c) / T

    w = exp_c / exp_c.sum()

    W = np.fft.rfft(w, fft_size)
    grad = 2.0 * np.fft.irfft(W * np.conj(A), fft_size)[:n]
    return grad, w, max_c


def adam_lse_optimize(a0, time_limit, T_start=40.0, T_end=0.05, verbose=True):
    """Adam optimizer on LSE-smoothed max self-convolution."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * S / s
    best_a = a.copy()
    best_val = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 35
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(15, min(150, int(remaining * 6.0)))
        lr = 0.4 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            grad, w, max_c = lse_grad_and_weights(a, T_val, fft_size)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()

                a_new = a.copy()

            a = a_new
            val = fast_conv_self(a).max()
            if val < best_val:
