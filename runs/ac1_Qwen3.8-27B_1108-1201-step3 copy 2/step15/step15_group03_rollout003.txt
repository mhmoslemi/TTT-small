        conv_a = fast_conv_self(a)
        active = np.where(conv_a >= M * 0.85)[0]
        if len(active) < 20:
            active = np.where(conv_a >= M * 0.5)[0]
        constraints = [T[active, :] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    return val, prob


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(500, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best_conv_max={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_all_constraints=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    conv_counter = 0

    best_score_ema = best_score
    stall_count = 0
    last_improve_iter = 0

    max_iters = 100000
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        b = lp_best_response(y, M, solver, use_all_constraints)
        if b is None:
            b = lp_best_response(y, M, solver, False)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            last_improve_iter = t
            stall_count = 0
        else:
            stall_count += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            last_improve_iter = t
            stall_count = 0

        # Momentum restart if stalled
        if stall_count % 150 == 0 and stall_count > 0:
            y = a.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP Nesterov t={t} restart ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall_count}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """1D golden-section search on each coordinate for final polish."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    phi = (np.sqrt(5.0) - 1.0) / 2.0

    def objective_1d(i, val):
        a[i] = val
        s_new = a.sum()
        if s_new < 1e-15:
            return 1e10
        a_save = a.copy()
        a_norm = a_save * norm / s_new
        conv = fast_conv_self(a_norm)
        a[i] = a_save[i]
        return conv.max()

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-12:
                continue
            # Search in [0, ai * 3]
            lo = 0.0
            hi = ai * 3.0
            # Golden section
            c = hi - phi * (hi - lo)
            d = lo + phi * (hi - lo)
            fc = objective_1d(i, c)
            fd = objective_1d(i, d)
            for _ in range(30):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - phi * (hi - lo)
                    fc = objective_1d(i, c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + phi * (hi - lo)
                    fd = objective_1d(i, d)
            best_val = (lo + hi) / 2.0
            old_ai = a[i]
            a[i] = best_val
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
            new_score = fast_conv_self(a).max()
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                improved = True
            else:
                a[i] = old_ai
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new

        if verbose and passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {passes} ({elapsed:.0f}s): best={best_score:.8f}")
        if not improved:
            break

    return best_a, best_score


def pair_transfer_optimize(a0, time_limit, rng, verbose=True):
    """Transfer mass between pairs of coordinates to reduce peak autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Find a few target coordinates
            targets = []
            for _ in range(8):
                j = int(rng.integers(n))
                if j != i:
                    targets.append(j)
            peak_lags = np.where(conv >= cur_max * 0.95)[0]
            for kl in peak_lags[:3]:
                j1 = (kl - i + n) % n
                j2 = (kl + i) % n
                if j1 != i and j1 < n:
                    targets.append(j1)
                if j2 != i and j2 < n:
                    targets.append(j2)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in targets[:12]:
                if j == i or j >= n:
                    continue
                aj = a[j]
                if aj < 1e-15 and ai < 1e-15:
                    continue

                # Compute the effect of transferring delta from i to j
                # conv_new = conv + delta * (effect)
                # The cross terms change
                g = np.zeros(size)
                # Adding to j: conv contribution of a_j interacting with all a_i
                for k in range(n):
                    if k < j:
                        g[k + j] += a[k] * 2  # cross terms
                    elif k > j:
                        g[k + j] += a[k]
                    else:
                        g[2 * j] += 0  # self term handled separately
                # Subtracting from i
                for k in range(n):
                    if k < i:
                        g[k + i] -= a[k] * 2
                    elif k > i:
                        g[k + i] -= a[k]
                    else:
                        g[2 * i] -= 0

                # Simpler: just try a few delta values and evaluate
                n_samples = 12
                deltas = np.linspace(0.0, ai * 0.8, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    a[i] = ai - d
                    a[j] = aj + d
                    s_new = a.sum()
                    if s_new < 1e-15:
                        a[i] = ai
                        a[j] = aj
                        continue
                    a_norm = a * norm / s_new
                    new_max = fast_conv_self(a_norm).max()
                    a[i] = ai
                    a[j] = aj
                    if new_max < best_val:
                        best_val = new_max
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(10, n // 25)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and (idx + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairTransfer pass ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 3 * x)
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.0
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5 + 0.15 * np.sin(4 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.25 * np.sin(3 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(2.5 * (x - 0.5)) ** 2 + 0.15
    elif choice == 9:
        a = 1.0 / (1.0 + np.exp(-8 * (x - 0.5)) + np.exp(8 * (x - 0.5)))
    elif choice == 10:
        a = np.cos(np.pi * x / 2.0) ** 1.5 * np.sin(np.pi * x) ** 0.5 + 0.1
    else:
        a = np.exp(-3.0 * (x - 0.4) ** 2 / (0.05 + 0.1 * rng.random()))
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_eval_score(a):
    """Compute the evaluation score directly (same formula as evaluate_sequence)."""
    n = len(a)
    a = np.maximum(a, 0.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    conv = fast_conv_self(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a ** 2))


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_best_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = compute_eval_score(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if not parent_loaded:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = compute_eval_score(best_seq)
        print(f"New init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick LSE warmup ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    print(f"\n=== Phase 1: LSE warmup n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gradient_descent(best_seq.copy(), t1, verbose=False)
        try_update(a_out, "lse_init")

    # === Phase 2: Main LP + Nesterov (full constraints) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 2: LP Nesterov full-constraints n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t2, solver, verbose=True,
                                        use_all_constraints=True)
        try_update(a_out, "lp_full1")

    # === Phase 3: Pair transfer polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Pair Transfer n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_transfer_optimize(best_seq.copy(), t3, rng, verbose=True)
        try_update(a_out, "pair1")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t4, solver, verbose=True,
                                        use_all_constraints=True)
        try_update(a_out, "lp_full2")

    # === Phase 5: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 300
    t5 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 5: Longer length n={n_long} (budget={t5:.0f}s) ===")
    if t5 > 40 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t5a = min(t5 * 0.5, 100.0)
        a_out, _ = nesterov_lp_optimize(init_long, t5a, solver, verbose=False,
                                        use_all_constraints=True)
        try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.3, 60.0)
            a_out2, _ = lse_gradient_descent(best_seq.copy(), t5b, verbose=False)
            try_update(a_out2, "lse_mid")

    # === Phase 6: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.25, 150.0)
    print(f"\n=== Phase 6: CD Polish n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = coordinate_descent_polish(best_seq.copy(), t6, verbose=True)
        try_update(a_out, "cd_final")

    # === Phase 7: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final LP n={len(best_seq)} (budget={remaining:.0f}s) ===")
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), remaining * 0.7, solver,
                                        verbose=False, use_all_constraints=True)
        try_update(a_out, "lp_final")

    # Final validation using evaluate_sequence
    if best_seq is not None:
        final_list = [float(x) for x in best_seq]
        try:
            final_score = evaluate_sequence(final_list)
            if np.isfinite(final_score) and final_score < best_score:
                best_score = final_score
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
