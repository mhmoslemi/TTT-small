def fast_autocorr(a):

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_conv(a, b):
    """Full linear convolution of a and b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    return fast_conv(a, a)


def smooth_grad(a, w):
    """Compute gradient of T*log(sum exp(c_k/T)) where w_k = exp(c_k/T)/sum(exp(c_j/T)).
    grad_i = 2 * sum_m a[m] * w[m+i] for i in [0, n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    c = fast_conv(a_rev, w)
    # (a_rev * w)[i + n - 1] = sum_m a[m] * w[m + i]
    return 2.0 * c[n - 1:2 * n - 1]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_solver():
    """Find the best available LP solver."""
    for s in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def build_T_rows(a, indices):
    """Build rows of the convolution matrix T where (T @ b)[k] = (a*b)[k]."""
    n = len(a)
    rows = len(indices)
    T = np.zeros((rows, n))
    for row, k in enumerate(indices):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            T[row, j] = a[k - j]
    return T


def lp_solve(a, M, solver, indices):
    """Solve max sum(b) s.t. (a*b)[k] <= M for k in indices, b >= 0."""
    n = len(a)
    T = build_T_rows(a, indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def smooth_optimize(a0, time_limit, verbose=False):
    """Minimize smooth approximation with adaptive step size and Nesterov."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Temperature schedule: start high, decrease
    temps = [50.0, 25.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001, 0.0005]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Adaptive number of steps based on remaining time
        n_steps = max(20, min(300, int(remaining * 2)))

        y = a.copy()
        step_count = 0

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = smooth_grad(y, w)

            # Backtracking line search
            base_step = 1.0 / (n * T * 0.5)
            step = base_step
            improved = False
            for _bt in range(10):
                a_new = y - step * grad
                a_new = project_simplex(a_new, norm)
                score_new = fast_autocorr(a_new).max()
                if score_new < c_max - 1e-12 or step < 1e-10:
                    improved = True
                    break
                step *= 0.5

            if not improved:
                a_new = project_simplex(y - step * grad, norm)

            # Nesterov update
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr
            step_count += 1

            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and step_count % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.4f} step={step_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp(a0, time_limit, solver, verbose=False):
    """LP best-response with Nesterov acceleration and adaptive constraints."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    last_improve = 0
    rng_local = np.random.default_rng(0)

    size = 2 * n - 1
    # Adaptive number of constraints
    base_n_cons = min(300, size)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Select constraints: top by value + some random
        frac = min(1.0, n_iters / 100.0)
        n_cons = int(base_n_cons * (0.3 + 0.7 * frac))
        top_idx = np.argsort(conv_y)[::-1][:n_cons]
        n_rand = max(3, n_cons // 10)
        rand_idx = rng_local.integers(0, size, n_rand)
        all_idx = np.unique(np.concatenate([top_idx, rand_idx]))

        b = lp_solve(y, M_curr, solver, all_idx)

        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            b = lp_solve(y, M_curr, solver, np.arange(size))
            if b is None:
                continue
        else:
            stuck_count = 0

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = n_iters

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            avg_check = avg_check * norm / avg_check.sum()
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if n_iters - last_improve > 300 and n_iters > 200:
            break

    return best_a, best_score


def coord_descent(a0, time_limit, rng, verbose=False):
    """Gradient-guided pair coordinate descent with line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Compute gradient of smooth approximation to guide search
        T_guide = 0.05
        c = fast_autocorr(a)
        c_max = c.max()
        exp_c = np.exp((c - c_max) / T_guide)
        w = exp_c / exp_c.sum()
        grad = smooth_grad(a, w)

        # Sort indices by |gradient| (most important first)
        order = np.argsort(np.abs(grad))[::-1]

        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            i = idx
            if a[i] < norm * 1e-9:
                continue

            # Try moving a small amount from i to nearby j
            # Focus on j that reduces the gradient at i
            for offset in [1, 2, 3, 5, 8, 13, 21, 34]:
                j1 = (i + offset) % n
                j2 = (i - offset) % n
                for j in [j1, j2]:
                    if j == i:
                        continue
                    # Try a few step sizes
                    for frac in [0.1, 0.3, 0.5]:
                        d = a[i] * frac
                        if d < norm * 1e-10:
                            continue
                        a_test = a.copy()
                        a_test[i] -= d
                        a_test[j] += d
                        a_test = a_test * norm / a_test.sum()
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < best_score - 1e-12:
                            a = a_test.copy()
                            conv = conv_test
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve_passes = 0
                            break

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 2:
                break
        else:
            no_improve_passes = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_descent_fine(a0, time_limit, rng, verbose=False):
    """Very fine-grained coordinate descent for final polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_moves = 0
    stuck = 0

    perm = rng.permutation(n)

    while time.monotonic() - t0 < time_limit:
        i = perm[n_moves % n]
        if a[i] < norm * 1e-10:
            n_moves += 1
            continue

        # Try all nearby j with small perturbations
        improved = False
        n_nonzero = np.count_nonzero(a > norm * 1e-8)

        for offset in range(1, min(20, n) + 1):
            for j in [(i + offset) % n, (i - offset) % n]:
                if j == i:
                    continue

                # Binary search for best transfer amount
                lo = 0.0
                hi = a[i] * 0.8
                best_d = 0.0
                best_val = conv.max()

                for _ in range(6):
                    mid = (lo + hi) / 2.0
                    if mid < norm * 1e-10:
                        lo = mid
                        continue
                    a_t = a.copy()
                    a_t[i] -= mid
                    a_t[j] += mid
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        hi = mid
                        continue

                    conv_t = fast_autocorr(a_t)

                        best_d = mid
                        lo = mid
                    else:
                        hi = mid

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    if conv.max() < best_score - 1e-12:
                        best_score = conv.max()
                        best_a = a.copy()

        if not improved:
            stuck += 1

        else:
            stuck = 0

        n_moves += 1
        if verbose and n_moves % 200 == 0:

    return best_a, best_score


def make_init_shape(n, shape_type, rng):
    """Create initial sequence of length n with given shape."""
    x = np.linspace(0, 1, n, endpoint=False)
    if shape_type == 'uniform':
        a = np.ones(n)
    elif shape_type == 'sine':
        a = np.maximum(np.sin(np.pi * x), 0.0) + 0.01
    elif shape_type == 'cosine_bell':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.01
    elif shape_type == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.01
    elif shape_type == 'two_bumps':
        a = np.exp(-16.0 * (x - 0.2) ** 2) + np.exp(-16.0 * (x - 0.8) ** 2) + 0.01
    elif shape_type == 'ramp':
        a = x + 0.1
    elif shape_type == 'anti_ramp':
        a = (1.0 - x) + 0.1
    elif shape_type == 'trapezoid':
        a = np.minimum(4.0 * x, 4.0 * (1.0 - x)) + 0.01
    elif shape_type == 'random':
        a = np.maximum(rng.standard_normal(n) * 0.3 + 1.0, 0.05)
    elif shape_type == 'comb':
        a = np.where((np.arange(n) % 4 < 2), 1.0, 0.3)
    else:
        a = np.ones(n)
    return np.maximum(a, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss

                if time.monotonic() - t_global > time_budget - 30:
                    break
                a_init = make_init_shape(n_main, shape, rng)
                a_init = a_init * np.sqrt(2.0 * n_main) / a_init.sum()
                a_out, _ = smooth_optimize(a_init, t_smooth, verbose=False)
                try_update(a_out, f"init_{shape}_n{n_main}")

    # === Phase 2: LP Nesterov on best (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.30
    print(f"\n=== Phase 2: LP Nesterov (budget={t2:.0f}s) ===")

    if t2 > 30 and best_seq is not None:
        n_c = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n_c}")

    # === Phase 3: Smooth optimization on best (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: Smooth grad (budget={t3:.0f}s) ===")

    if t3 > 15 and best_seq is not None:
        n_c = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t3, verbose=True)
        try_update(a_out, f"smooth_n{n_c}")

    # === Phase 4: Coordinate descent (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Coord descent (budget={t4:.0f}s) ===")

    if t4 > 10 and best_seq is not None:
        n_c = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t4, rng, verbose=True)
        try_update(a_out, f"cd_n{n_c}")

    # === Phase 5: Fine coordinate descent (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Fine CD (budget={t5:.0f}s) ===")

    if t5 > 10 and best_seq is not None:
        n_c = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = coord_descent_fine(a_norm, t5, rng, verbose=True)
        try_update(a_out, f"cd_fine_n{n_c}")

    # === Phase 6: LP round 2 + smooth (7% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.07
    print(f"\n=== Phase 6: Polish LP+Smooth (budget={t6:.0f}s) ===")

    if t6 > 20 and best_seq is not None:
        n_c = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        t6a = t6 * 0.5
        a_out, _ = nesterov_lp(a_norm, t6a, solver, verbose=False)
        try_update(a_out, "polish_lp")

        remaining6 = time_budget - (time.monotonic() - t_global)
        if remaining6 > 10 and best_seq is not None:
            n_r = len(best_seq)
            a_norm_r = best_seq * np.sqrt(2.0 * n_r) / best_seq.sum()
            a_out_r, _ = smooth_optimize(a_norm_r, remaining6 * 0.6, verbose=False)
            try_update(a_out_r, "polish_smooth")

    # === Phase 7: Length exploration (3% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Length exploration (budget={remaining:.0f}s) ===")

    if remaining > 60 and best_seq is not None:
        n_base = len(best_seq)
        alt_lengths = sorted(set([n_base + 200, n_base - 200, n_base + 448, n_base - 448]),
                             key=lambda x: abs(x - n_base))
        per_len = remaining / max(1, len(alt_lengths) + 1)

        for n_alt in alt_lengths:
            if n_alt < 100 or n_alt > 5000:
                continue
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_alt = min(per_len, 120.0)
            if t_alt < 10:
                continue
            print(f"  Trying n={n_alt} (budget={t_alt:.0f}s)")
            init_alt = interpolate_sequence(best_seq * np.sqrt(2.0 * n_base) / best_seq.sum(), n_alt)
            a_out, _ = smooth_optimize(init_alt, t_alt * 0.6, verbose=False)
            try_update(a_out, f"len_{n_alt}_smooth")

            remaining_alt = min(time_budget - (time.monotonic() - t_global), t_alt * 0.4)
            if remaining_alt > 5 and best_seq is not None:
                n_b = len(best_seq)
                a_n = best_seq * np.sqrt(2.0 * n_b) / best_seq.sum()
                a_out2, _ = coord_descent_fine(a_n, remaining_alt, rng, verbose=False)
                try_update(a_out2, f"len_polish_{n_alt}")

    # === Final verification and return ===
    if best_seq is not None:
        fs = evaluate_wrap(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
