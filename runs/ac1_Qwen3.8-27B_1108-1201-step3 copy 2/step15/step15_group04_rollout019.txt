def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(200.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)
            for trial in range(12):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.05 * T_val:
                    break

            a_new = a - step * grad

            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

                    best_a = avg_check.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, no_impr={no_improve_count}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """Subgradient method on the max of convolution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()
        peak_idx = np.where(conv >= M * 0.999)[0]

        # Subgradient: 2 * T_{k*}^T a for each peak k*
        # T_{k}^T a at position j = sum of a[i] where i contributes to conv[k]
        # conv[k] = sum_j a[j] * a[k-j], so d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (for valid indices)
        # For peak at k: grad[j] = sum_{k in peaks} (a[k-j] + a[j-k]) where valid
        grad = np.zeros(n)
        for k in peak_idx[:20]:  # Limit to top peaks
            for j in range(n):
                if 0 <= k - j < n:
                    grad[j] += a[k - j]
                if 0 <= j - k < n:
                    grad[j] += a[j - k]
        grad *= 2.0

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Adaptive step size
        eta = norm / (np.sqrt(it + 1) * g_norm / np.sqrt(n))
        eta = min(eta, 0.5)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        sc = fast_conv_self(a_new).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Efficient pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    max_peaks = 12

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_threshold = cur_max * 0.92
        peak_lags = np.where(conv >= peak_threshold)[0][:max_peaks]

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Candidate j's from peak lags
            j_candidates = set()
            for kl in peak_lags:
                j1 = (kl - i) % n if n > 0 else 0
                j2 = (kl + i) % n if n > 0 else 0
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            # Add some random
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            g = np.zeros(size)
            lo = max(0, i)
            hi = min(size, i + n)
            g[lo:hi] -= a[0:hi - lo]
            lo2 = max(0, i - n + 1)
            hi2 = min(size, i + 1)
            g[lo2:hi2] -= a[i - np.arange(lo2, hi2)]

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Gain from moving mass from i to j
                g[:size] = g  # reset base
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                g_pos = np.zeros(size)
                g_pos[lo3:hi3] += a[0:hi3 - lo3]
                lo4 = max(0, j - n + 1)
                hi4 = min(size, j + 1)
                g_pos[lo4:hi4] += a[j - np.arange(lo4, hi4)]

                total_g = g_pos + g  # g is negative (removal from i)

                # Try a few delta values
                n_samples = 15
                deltas = np.linspace(0.02, 0.8, n_samples) * ai
                for d in deltas:
                    new_conv = conv + d * total_g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(5, n // 40)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD impr={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        # Asymmetric decay - key for breaking symmetry
        a = np.exp(-3.0 * x) * (1.0 + 0.5 * np.sin(5 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 10:
        # Ramp with oscillation
        a = x * (1.0 + 0.4 * np.cos(8 * np.pi * x))
        a = np.maximum(a, 0.01)
    else:
        # Random smooth
        freqs = rng.integers(2, 15, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 1.0, size=5)
        a = np.ones(n)
        for f, p, am in zip(freqs, phases, amps):
            a += am * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Quick LSE to improve init
    remaining = time_budget - (time.monotonic() - t_global)
    t_init = min(remaining * 0.06, 90.0)
    if t_init > 15:
        print(f"\n=== Init LSE (budget={t_init:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, t_init, verbose=False)
        try_update(a_out, "lse_init")

    # Phase 1: LP Nesterov main workhorse
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t1, solver, verbose=True)
        try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Subgradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Subgradient n={n2} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = subgradient_polish(best_seq, t2, verbose=True)
        try_update(a_out, f"subgrad_n{n2}")

    # Phase 3: LP Nesterov round 2 (from improved point)
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LP Nesterov R2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t3, solver, verbose=True, use_adaptive=False)
        try_update(a_out, f"lp2_n{n3}")

    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 4: Pair CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{n4}")

    # Phase 5: Alternate LSE + LP
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    print(f"\n=== Phase 5: Alternate n={n5} (remaining={remaining:.0f}s) ===")
    if remaining > 80:
        t_l = min(remaining * 0.30, 80.0)
        a_out, _ = lse_gradient_descent(best_seq, t_l, verbose=False)
        try_update(a_out, "lse_mid")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t_lp = min(remaining * 0.40, 120.0)
            a_out, _ = lp_nesterov_optimize(best_seq, t_lp, solver, verbose=False)
            try_update(a_out, "lp_mid")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        for delta in [200, 400]:
            n_try = n_base + delta
            if remaining < 80:
                break
            init_long = interpolate_seq(best_seq, n_try)
            t_long = min(remaining * 0.35, 120.0)
            a_out, _ = lp_nesterov_optimize(init_long, t_long, solver, verbose=False)
            try_update(a_out, f"lp_long_{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40 and len(best_seq) == n_try:
                t_cd = min(remaining * 0.35, 60.0)
                a_out2, _ = pair_cd_optimize(best_seq, t_cd, rng, verbose=False)
                try_update(a_out2, f"pcd_long_{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)

    # Phase 7: Try shorter length (sometimes shorter is better due to n factor)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and n_base > 800:
        n_short = max(800, n_base - 400)
        init_short = interpolate_seq(best_seq, n_short)
        t_short = min(remaining * 0.4, 80.0)
        print(f"\n=== Phase 7: Shorter length n={n_short} (budget={t_short:.0f}s) ===")
        a_out, _ = lp_nesterov_optimize(init_short, t_short, solver, verbose=False)
        try_update(a_out, f"lp_short_{n_short}")

    # Phase 8: Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t_f1 = min(remaining * 0.35, 50.0)
        a_out, _ = lp_nesterov_optimize(best_seq, t_f1, solver, verbose=False)
        try_update(a_out, "lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_f2 = min(remaining * 0.5, 40.0)
            a_out, _ = pair_cd_optimize(best_seq, t_f2, rng, verbose=False)
            try_update(a_out, "pcd_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_f3 = min(remaining * 0.6, 30.0)
            a_out, _ = subgradient_polish(best_seq, t_f3, verbose=False)
            try_update(a_out, "subgrad_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
