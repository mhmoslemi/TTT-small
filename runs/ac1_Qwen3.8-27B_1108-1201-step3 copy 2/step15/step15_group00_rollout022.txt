def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the objective (with a normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    return float(2 * n * max_b / (s * s))

    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def subgradient_at_peak(a, k):
    """Compute the subgradient of b[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    if i_start >= i_end:
        return grad
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    grad[idx] = 2.0 * a[j_idx]
    return grad

def active_set_subgradient(a, b, top_k=20, temperature=1.0):
    """Compute a soft subgradient using the top-k peaks with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(b))
    
    # Get top-k peaks
    top_idx = np.argsort(b)[::-1][:k_max]
    top_vals = b[top_idx]
    
    # Softmax weighting based on how close to the max each peak is
    max_val = top_vals[0]
    if max_val < 1e-30:
        return np.zeros(n)
    
    # Weights: peaks closer to the max get higher weight
    diffs = max_val - top_vals
    w = np.exp(-diffs / (temperature * max_val + 1e-30))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += wi * a[j_idx]
    return 2.0 * grad

        A = np.fft.rfft(a, fft_size)

        mags = np.abs(A)

        phases = np.angle(A)
        new_mags = np.ones_like(mags) * target
        new_mags[0] = mags[0]  # Keep DC component

        a_new = np.fft.irfft(A_new, fft_size)[:n]

        mags = np.abs(A)
        # DC component is mags[0] = sum(a) = 1
        # Target magnitude for non-DC
        target = mags[1:].mean()
        
        # Create modified spectrum
        phases = np.angle(A)
        A_new = np.zeros_like(A)
        A_new[0] = A[0]  # Keep DC
        for k in range(1, len(A)):
            A_new[k] = target * np.exp(1j * phases[k])
        
        # Inverse FFT
        a_new_padded = np.fft.irfft(A_new, pad_size)
        a_new = a_new_padded[:n]
        
        # Project to non-negative, sum=1
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Score
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Damped update
        a = 0.5 * a_new + 0.5 * a
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral t={t} ({elapsed:.1f}s): score={best_score:.6f}")
    
    return best_a, best_score

Let me also consider: what if we use a **different objective function** for the intermediate optimization? Instead of directly minimizing max_b, we could minimize a smooth approximation like sum(b[k]^p) for large p (which approximates max_b^p). This gives a different gradient and might escape different local minima.

For p=2: minimize sum(b[k]^2) = ||b||^2. This is the "energy" of the convolution. The gradient is easier to compute:
d/d a_i (sum_k b[k]^2) = 2 * sum_k 2*b[k] * db[k]/da_i = 4 * sum_k b[k] * a[k-i] (for valid k-i)

This is 4 * (b * a)[i] (convolution of b and a, evaluated at i). This can be computed efficiently with FFT!

So we could minimize ||conv(a,a)||^2 using gradient descent, which is a smooth objective and might be easier to optimize. Then, as a final step, switch to minimizing max_b.

The insight: minimizing ||b||^2 spreads the energy of b more evenly, which indirectly reduces the peak. It's like a "preconditioning" step.

Actually, let me reconsider the spectral flattening approach. The issue is that zero-padding changes the relationship between the DFT and the linear convolution. Let me think about this more carefully.

If a has length n and we pad it to length L (where L >= 2n-1), then:
- The circular convolution of the padded sequence (length L) equals the linear convolution (length 2n-1, embedded in L)
- The DFT of the padded sequence is A[k] = sum_{j=0}^{n-1} a[j] * exp(-2πikj/L)
- The DFT of the circular conv = A[k]^2
- The inverse DFT gives the circular conv, which equals the linear conv (for L >= 2n-1)

So if we want to modify the spectrum to make the convolution flatter, we should:
1. Pad a to length L
2. Compute A = DFT(a_padded)
3. Modify A to make A^2 have a flatter inverse DFT
4. Inverse DFT to get a_new_padded
5. Take the first n elements
6. Project to non-negative simplex

But there's a subtlety: we want the LINEAR convolution of the n-element sequence to be flat, not the circular convolution of the L-element padded sequence. These are the same if L >= 2n-1, so the padding approach works.

However, when we modify A and take the inverse DFT, we get an L-element sequence. The first n elements should be non-negative and sum to 1 for a valid solution. The remaining L-n elements should ideally be zero (but won't be after modification).

This is the key challenge: the spectral modification will generally produce non-zero values outside the first n positions, and possibly negative values within. The projection (clip to non-negative, take first n, renormalize) will destroy some of the spectral structure.

Despite this, the approach might still work as a heuristic: each iteration produces a "better" sequence in terms of convolution flatness, even if the projection is imperfect.

Let me also think about a simpler variant: instead of making |A| constant, we can make |A|^2 (which is related to the power spectrum) constant. Since the convolution is ifft(A^2), making A^2 "flat" (in the sense of having constant magnitude) would make the convolution flat.

Actually, A^2 is already the spectrum of the convolution. So making |A^2| constant is the same as making |A| constant. The phase of A^2 determines the "shape" of the convolution.

For a real sequence a, A has conjugate symmetry: A[L-k] = conj(A[k]). So A^2 also has this property. The inverse DFT of A^2 is real (as expected for the convolution of a real sequence with itself).

OK, I think the spectral approach is sound in principle, but the projection step is the main issue. Let me just implement it and see.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Accelerated Mirror Descent (AMD) with adaptive soft subgradient
2. Spectral flattening (novel approach targeting flat power spectrum)
3. Energy-based gradient descent (smooth surrogate objective)
4. Peak-shaving local search
5. Multi-initialization and length exploration

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def soft_subgradient(a, b, top_k=30, temperature=0.5):
    """Compute soft subgradient of max(b) w.r.t. a."""
    n = len(a)
    k_max = min(top_k, len(b))
    top_idx = np.argsort(b)[::-1][:k_max]
    top_vals = b[top_idx]
    
    max_val = top_vals[0]
    if max_val < 1e-30:
        return np.zeros(n)
    
    # Exponential weighting based on proximity to max
    w = np.exp(-(max_val - top_vals) / (temperature * max_val + 1e-30))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        grad[idx] += wi * a[k_int - idx]
    return 2.0 * grad

            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e
