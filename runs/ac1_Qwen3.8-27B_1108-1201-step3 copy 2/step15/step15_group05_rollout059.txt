"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response with adaptive active sets
2. FFT-accelerated subgradient polishing with BB step sizes
3. Fast incremental coordinate descent
4. Multi-length exploration
5. Symmetrization checks
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import next_fast_len
from scipy.linalg import toeplitz

# Global state
_solver = None
_best_global = None
_best_score_global = float('inf')


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    return _solver


def fast_autocorr(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_len = next_fast_len(size)
    A = np.fft.rfft(a, fft_len)
    return np.fft.irfft(A * A, fft_len)[:size]


def score_of(a):
    """Compute the score 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * len(a) * b.max() / (s * s)


def normalize(a, target_sum):
    """Normalize a to have the given sum."""
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - S))[0][-1]
    theta = (cv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, n_active=5):
    """Compute subgradient of max_k conv[k] at a."""
    n = len(a)
    size = 2 * n - 1
    # Find active set: top peaks
    top_idx = np.argsort(conv)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        # d conv[k] / d a[j] = 2 * a[k-j] if 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            grad[j] += 2.0 * a[k - j]
    grad /= n_active
    return grad

Hmm, the subgradient computation with a Python loop is O(n * n_active) which for n=1600 and n_active=10 is 16000 operations. That's fine.

```python
def compute_subgradient_fast(a, conv, n_active=5):
    n = len(a)
    size = 2 * n - 1
    top_idx = np.argsort(conv)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        # a[k-j] for j in [max(0,k-n+1), min(n,k+1))
        # This is a[k-1], a[k-2], ..., a[k-(hi-lo)]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        indices = k - np.arange(lo, hi)
        grad[lo:hi] += a[indices]
    return 2.0 * grad / n_active
```

This is O(n_active * n) but with vectorized inner loop, so much faster.

Now let me think about the LP formulation more carefully.

Given fixed y (the "virtual" sequence), we want to solve:
max sum(b) s.t. (y * b)[k] <= M for k in active_set, b >= 0

where M = max(y * y).

The constraint (y * b)[k] = sum_j y[j] * b[k-j] <= M.

The constraint matrix T has T[k, j] = y[k-j] if 0 <= k-j < n, else 0.

For the LP to be efficient, I want:
- Small active set (20-50 constraints)
- Sparse matrix (each row has at most n nonzeros, but for edge lags, fewer)

Let me use a dense matrix since the active set is small (50 * 1600 = 80000 entries is small).

Actually, for the middle lags, all n entries are non-zero. For edge lags, fewer. On average, maybe 0.75n nonzeros per row. So the matrix is mostly dense.

For cvxpy, I'll use a dense matrix. With 50 constraints and 1600 variables, ECOS should handle this in < 0.1s.

import numpy as np

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:

    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_len = next_fast_len(size)
    A = np.fft.rfft(a, fft_len)
    return np.fft.irfft(A * A, fft_len)[:size]


def score_of(a):
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * len(a) * b.max() / (s * s)


def safe_eval(a):
    try:

        sc = evaluate_sequence(a_list)

    except Exception:

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    csv = cv - S
    cond = u * np.arange(1, n + 1) > csv
    if not cond.any():
        return np.full(n, S / n)
    rho = np.nonzero(cond)[0][-1]
    theta = csv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_T(y, k_indices):
    n = len(y)
    T = np.zeros((len(k_indices), n))
    for i, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[i, j] = y[k - j]
    return T


def lp_best_response_dual(y, M, k_indices, solver):
    n = len(y)
    T = build_T(y, k_indices)
    m = len(k_indices)
    yd = cp.Variable(m, nonneg=True)
    objective = cp.Minimize(M * cp.sum(yd))
    constraints = [T.T @ yd >= 1]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if prob.value is None or prob.value < 0:
        return None
    n_eff = prob.value / M
    if n_eff < 1e-10:
        return None
    return n_eff


def lp_best_response_primal(y, M, k_indices, solver):
    n = len(y)
    T = build_T(y, k_indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    duals = None
    try:
        dv = constraints[0].dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 50
    y = a.copy()
    y_prev = a.copy()
    erg = np.zeros(n)
    erg_count = 0
    n_dp = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 40 == 0 or stall_count > stall_threshold:
            frac = max(0.01, 0.06 * np.exp(-t / 400.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - frac))[0]
            topk = np.argsort(conv_y)[-max(5, n // 100):]
            active_set = np.unique(np.concatenate([active_set, topk]))
            if len(active_set) > 200:
                active_set = topk[:min(len(topk), 80)]

        use_dual = len(active_set) <= 60
        if use_dual:
            n_eff = lp_best_response_dual(y, M_curr, active_set, solver)
            if n_eff is not None and n_eff > 1e-10:
                pass
            else:
                b_val, _ = lp_best_response_primal(y, M_curr, active_set, solver)
                if b_val is None:
                    if t % 20 == 0:
                        active_set = np.arange(2 * n - 1)
                        b_val, _ = lp_best_response_primal(y, M_curr, active_set, solver)
                    if b_val is None:
                        continue
        else:
            b_val, _ = lp_best_response_primal(y, M_curr, active_set, solver)
            if b_val is None:
                if t % 20 == 0:
                    active_set = np.argsort(conv_y)[:min(60, len(conv_y))]
                    b_val, _ = lp_best_response_primal(y, M_curr, active_set, solver)
                if b_val is None:
                    continue

        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val * norm / b_sum
        n_dp += 1

        alpha = 2.0 / (t + 2.0)
        y_new = np.maximum((1.0 - alpha) * y + alpha * g, 0.0)
        s_y = y_new.sum()
        y_new = y_new * norm / s_y if s_y > 1e-15 else y.copy()

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.97)
            a_new = np.maximum(y_new + momentum * (y_new - y_prev), 0.0)
            s_a = a_new.sum()
            a_new = a_new * norm / s_a if s_a > 1e-15 else y_new.copy()

        y_prev = y
        y = y_new
        a = a_new

        sy = fast_autocorr(y).max()
        if sy < best_score:
            best_score = sy
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        sa = fast_autocorr(a).max()
        if sa < best_score:
            best_score = sa
            best_a = a.copy()
            stall_count = 0

        erg += a
        erg_count += 1
        if erg_count % 25 == 0:
            ec = erg / erg_count
            sc_erg = ec.sum()
            ec = ec * norm / sc_erg if sc_erg > 1e-15 else ec
            se = fast_autocorr(ec).max()
            if se < best_score:
                best_score = se
                best_a = ec.copy()
                stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                print(f"    LP restart iter {t} ({time.monotonic()-t0:.0f}s) best={best_score:.10f}")

        if verbose and n_dp % 500 == 0:
            print(f"    LP iter={n_dp} ({time.monotonic()-t0:.0f}s) best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False, method='bb'):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 2.0 / np.sqrt(n)
    step_min = 1e-6
    patience = 0
    max_patience = 2000
    n_active = max(3, min(10, n // 100))

    g_prev = None
    dx_prev = None
    f_prev = None
    f_curr = best_score

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        top_idx = np.argsort(conv)[-n_active:]
        weights = np.exp(-8.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.02, 1e-10))
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                grad[j] += w * 2.0 * a[k - j]

        if method == 'bb' and g_prev is not None and dx_prev is not None:
            gy = np.dot(g_prev, dx_prev)
            dy = np.dot(dx_prev, dx_prev)
            if dy > 1e-15 and abs(gy) > 1e-15:
                step = max(step_min, min(50.0, dy / abs(gy)))

        a_trial = np.maximum(a - step * grad / np.sqrt(n), 0.0)
        s_t = a_trial.sum()
        a_trial = a_trial * norm / s_t if s_t > 1e-15 else a.copy()

        new_conv = fast_autocorr(a_trial)
        new_score = new_conv.max()

        if method == 'bb':
            dx_new = a_trial - a
            if g_prev is not None and dx_prev is not None:
                gy_new = np.dot(grad, dx_new)
                dy_new = np.dot(dx_new, dx_new)
                if dy_new > 1e-15 and abs(gy_new) > 1e-15:
                    step = max(step_min, min(50.0, dy_new / abs(gy_new)))
            g_prev = grad
            dx_prev = dx_new

        if new_score < best_score - 1e-14:
            dx_for_pat = a_trial - a
            a = a_trial
            best_score = new_score
            best_a = a_trial.copy()
            patience = 0
            f_curr = new_score
        else:
            patience += 1
            step *= 0.97
            if step < step_min:
                break

        if t % 100 == 0:
            rand_idx = np.random.randint(0, 2 * n - 1)
            g_r = np.zeros(n)
            lo = max(0, rand_idx - n + 1)
            hi = min(n, rand_idx + 1)
            for j in range(lo, hi):
                g_r[j] += 2.0 * a[rand_idx - j]
            a_r = np.maximum(a - step * 0.3 * g_r / np.sqrt(n), 0.0)
            sr = a_r.sum()
            a_r = a_r * norm / sr if sr > 1e-15 else a.copy()
            sc_r = fast_autocorr(a_r).max()
            if sc_r < best_score - 1e-14:
                a = a_r
                best_score = sc_r
                best_a = a_r.copy()
                patience = 0

        if verbose and t % 5000 == 0:
            print(f"    Subgrad t={t} ({time.monotonic()-t0:.0f}s) best={best_score:.10f} step={step:.6f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_passes = 0
    max_no_improve = 5

    for pass_num in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve_passes >= max_no_improve:
            break

        improved = False
        conv = fast_autocorr(a)
        curr_score = conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            v = a[j]
            lo = max(0, j - n + 1)
            hi = min(n, j + 1)
            c = np.zeros(size)
            for jj in range(lo, hi):
                c[k2j_idx(j, jj, n)] += a[k2j_idx2(j, jj)]

            c_vec = np.zeros(size)
            for jj in range(lo, hi):
                k_val = j + jj
                if 0 <= k_val < size:
                    c_vec[k_val] += a[jj]
                k_val2 = j - jj
                if 0 <= k_val2 < size and jj != j:
                    c_vec[k_val2] += a[jj]

            bg = conv - 2.0 * v * c_vec
            if 2 * j < size:
                bg[2 * j] -= v * v

            s_others = a.sum() - v
            candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v, v * 1.25, v * 1.5, v * 2.0]
            candidates = [x for x in candidates if x >= 0]
            if s_others > 0:
                candidates.append(s_others * 0.5)
                candidates.append(s_others)

            best_x = v
            best_obj = 2.0 * n * curr_score
            for x in candidates:
                new_conv = bg + 2.0 * x * c_vec
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                ns = s_others + x
                if ns < 1e-15:
                    continue
                obj = 2.0 * n * new_conv.max() / (ns * ns)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            if best_x != v and abs(best_x - v) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                a = a * norm / s_new
                new_sc = fast_autocorr(a).max()
                if new_sc < best_score - 1e-14:
                    best_score = new_sc
                    best_a = a.copy()
                    improved = True
                    conv = fast_autocorr(a)
                    curr_score = conv.max()
                else:
                    a[j] = v
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    curr_score = conv.max()

        if not improved:
            no_improve_passes += 1
        else:
            no_improve_passes = 0

        if verbose and pass_num % 3 == 0:
            print(f"    CD pass={pass_num} ({time.monotonic()-t0:.0f}s) best={best_score:.10f}")

    return best_a, best_score


def k2j_idx(j, jj, n):
    return j - jj


def k2j_idx2(j, jj):
    return jj


def symmetrize(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def perturb_sequence(a, rng, scale=0.02):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def make_cosine_window(n):
    x = np.linspace(0, 1, n)
    w = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    w = np.maximum(w, 1e-6)
    return w / w.sum()


def make_gaussian_window(n):
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = 0.25
    w = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    w = np.maximum(w, 1e-6)
    return w / w.sum()


def make_fejer_window(n):
    x = np.arange(n)
    center = (n - 1) / 2.0
    w = 1.0 - np.abs(x - center) / (center + 1.0)
    w = np.maximum(w, 1e-6)
    return w / w.sum()


def propose_candidate():
    global _best_seq_global, _best_score_global

    # Try windows

            print(f"  *** {name} window: {sc_w:.8f}")

    n_base = len(best_seq)
    print(f"\nStarting with n={n_base}, score={best_score:.8f}")

    # Phase 1: LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # Phase 2: Subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Subgrad BB n={n_cur}, budget={t2:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t2, verbose=True, method='bb')
        try_update(a_out, f"subg_n{n_cur}")

    # Phase 3: Coordinate Descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        print(f"\n=== Phase 3: CD n={n_cur}, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t3, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # Phase 4: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP round 2 n={n_cur}, budget={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, seed=123)
        try_update(a_out, f"lp2_n{n_cur}")

    # Phase 5: Multi-length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 5: Multi-length, remaining={remaining:.0f}s ===")
        n_cur = len(best_seq)
        target_lengths = [1200, 2000, 2400]
        t_each = min(remaining * 0.25, 150.0) / len(target_lengths)
        for tl in target_lengths:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_e = min(t_each, remaining - 20)
            if t_e < 15:
                break
            print(f"  Trying n={tl}, budget={t_e:.0f}s")
            interp = interpolate_seq(best_seq, tl)
            a_norm = interp * np.sqrt(2.0 * tl) / interp.sum()
            # Quick LP
            a_out, _ = nesterov_lp_optimize(a_norm, t_e * 0.6, solver, verbose=False, seed=tl)
            # Quick subgrad
            a_out2, _ = subgradient_polish(a_out, t_e * 0.3, verbose=False, method='bb')
            try_update(a_out, f"ml_lp_{tl}")
            try_update(a_out2, f"ml_sg_{tl}")

    # Phase 6: Perturbation restarts
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: Perturbations, remaining={remaining:.0f}s ===")
        n_restarts = 2
        t_each = min(remaining * 0.20, 100.0) / n_restarts
        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_e = min(t_each, remaining - 15)
            if t_e < 10:
                break
            pert = perturb_sequence(best_seq, rng, scale=0.015 + 0.01 * r)
            a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t_e * 0.7, solver, verbose=False, seed=500 + r)
            a_out2, _ = subgradient_polish(a_out, t_e * 0.3, verbose=False, method='bb')
            try_update(a_out, f"pert{r}_lp")
            try_update(a_out2, f"pert{r}_sg")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.30, 150.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Final polish n={n_cur}, budget={t7:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        t_sg = t7 * 0.5
        t_cd = t7 * 0.5
        a_out, _ = subgradient_polish(a_norm, t_sg, verbose=False, method='bb')
        try_update(a_out, "final_sg")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 5:
            n_cur2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_cur2) / best_seq.sum()
            a_out2, _ = coordinate_descent_fast(a_norm2, min(t_cd, remaining - 2), verbose=True)
            try_update(a_out2, "final_cd")

    # Final check
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
