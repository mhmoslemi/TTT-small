def fast_autocorr(a):
    """Compute autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the convolution matrix T where conv(a,b) = T @ b."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve the LP best-response: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    momentum_t = 1.0
    patience = 0
    restart_count = 0
    n_iters = 0
    stuck_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        # Restart if stuck
        if patience > 60 and n_iters % 30 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                print(f"    NLP restart {restart_count} at iter={n_iters}: best={best_score:.10f}")

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

        c2 = conv_w_a[:n]
        
        # Gradient wrt a
        grad_a = 2.0 * (c1 + c2)
        
        # Now we need gradient wrt x (the parameter).
        # a = norm * softmax(x) = norm * exp(x) / sum(exp(x))
        # da[m]/dx[l] = norm * (delta_ml * exp(x_m) - exp(x_m)*exp(x_l)) / (sum(exp(x)))^2
        #             = a[m] * (delta_ml - p[l])  where p = softmax(x)
        
        # So grad_x[l] = sum_m grad_a[m] * da[m]/dx[l]
        #             = sum_m grad_a[m] * a[m] * (delta_ml - p[l])
        #             = grad_a[l] * a[l] - p[l] * sum_m grad_a[m] * a[m]
        
        # Compute in O(n):
        p = a / norm  # = softmax(x)
        ga_dot_a = np.dot(grad_a, a)
        grad_x = grad_a * a - ga_dot_a * p
        
        # Adam step
        n_iters += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** n_iters)
        v_hat = v / (1 - beta2 ** n_iters)
        
        # Adaptive learning rate (decrease over time)
        lr_curr = lr / (1.0 + 0.1 * n_iters / 1000)
        x = x - lr_curr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        p_new = exp_x / exp_x.sum()
        a_new = p_new * norm
        conv_new = fast_autocorr(a_new)
        score = conv_new.max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 2000:
            # Restart x
            x = np.log(np.maximum(a_new, 1e-15))
            m = np.zeros(n)
            v = np.zeros(n)
            patience = 0
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, t_smooth={t_smooth:.2e}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=True):
    """Intensive coordinate descent using FFT-based evaluation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    n_pers = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_pers += 1
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-10:
                continue
            
            # Try moving mass from i to various j
            # Use a few candidate j values
            candidates = []
            # Nearby
            for off in [1, -1, 2, -2, 3, -3, 5, -5]:
                j = (i + off) % n
                if j != i:
                    candidates.append(j)
            # Far
            for _ in range(5):
                j = rng.integers(0, n)
                if j != i:
                    candidates.append(j)
            
            for j in candidates:
                if a[i] < norm * 1e-10:
                    break
                # Try a few fractions of mass
                for frac in [0.1, 0.3, 0.5, 0.7]:
                    d = a[i] * frac
                    if d < norm * 1e-12:
                        continue
                    a[i] -= d
                    a[j] += d
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        no_improve = 0
                        break  # break frac loop
                    else:
                        a[i] += d
                        a[j] -= d
            
            if improved:
                # Re-randomize after improvement
                break
        
        if not improved:
            no_improve += 1
            if no_improve >= 20:
                # Perturb
                scale = norm * 0.01
                noise = rng.laplace(0, scale, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s = a_pert.sum()
                if s > 1e-15:
                    a = a_pert * norm / s
                no_improve = 0
        
        if verbose and n_pers % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_pers} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score
