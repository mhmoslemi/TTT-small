def solve_lp(a, M, use_active=None):

            j_hi = min(k + 1, n)
            T[idx, j_lo:j_hi] = a[j_lo:j_hi]

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(50.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Precompute T
    T_full = build_T(a)

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening with two levels
        threshold_1 = M * 0.95
        active = np.where(conv_y >= threshold_1)[0]
        if len(active) < 20:
            threshold_2 = M * 0.70
            active = np.where(conv_y >= threshold_2)[0]
        if len(active) < 10:
            active = np.where(conv_y >= M * 0.4)[0]

        # Solve LP with active constraints
        if len(active) < size:
            b = solve_lp(y, M, active)
        else:
            b = solve_lp(y, M, None)

        if b is None:
            b = solve_lp(y, M, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints
        if len(active) < size:
            check = fast_conv_cross(y, g)
            violated = np.where(check > M * 1.0005)[0]
            if len(violated) > 2:
                b_full = solve_lp(y, M, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Rebuild T periodically since a changed
        if t % 50 == 0:
            T_full = build_T(a)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pressure_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent that targets pressure points (high autocorrelation contributors)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, fast_conv_self(a0).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Find peak lags
        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]

        # Compute pressure: how much each coordinate contributes to peaks
        pressure = np.zeros(n)
        for kl in peak_lags[:min(30, len(peak_lags))]:
            for i in range(n):
                j = kl - i
                if 0 <= j < n:
                    pressure[i] += a[j]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Candidate target indices: low pressure, high value
            j_candidates = list(rng.integers(n, size=min(15, n)))
            # Add structured candidates near peaks
            for kl in peak_lags[:5]:
                j_s = (kl - i + n) % n
                if j_s != i:
                    j_candidates.append(j_s)
                j_s2 = (i + kl) % n
                if j_s2 != i:
                    j_candidates.append(j_s2)

            best_val = cur_max
            best_delta = 0.0
            best_j = -1

            # Recompute conv once per i
            conv = fast_conv_self(a)
            cur_max = conv.max()

            for j in j_candidates:
                if j == i or j >= n:
                    continue
                aj = a[j]
                if aj < 1e-15 and ai < 1e-15:
                    continue

                # Compute gain vector for transferring d from i to j
                g = np.zeros(size)
                # Effect of increasing a[j] by d
                jlo = max(0, j)
                jhi = min(size, j + n)
                if jhi > jlo:
                    g[jlo:jhi] += a[0:jhi - jlo]
                jlo2 = max(0, j - n + 1)
                jhi2 = min(size, j + 1)
                if jhi2 > jlo2:
                    g[jlo2:jhi2] += a[j - np.arange(jlo2, jhi2)]
                # Effect of decreasing a[i] by d
                ilo = max(0, i)
                ihi = min(size, i + n)
                if ihi > ilo:
                    g[ilo:ihi] -= a[0:ihi - ilo]
                ilo2 = max(0, i - n + 1)
                ihi2 = min(size, i + 1)
                if ihi2 > ilo2:
                    g[ilo2:ihi2] -= a[i - np.arange(ilo2, ihi2)]

                # Sample deltas and find best
                n_samples = 15
                d_max = ai * 0.95
                deltas = np.linspace(0.0, d_max, n_samples + 1)[1:]
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g
                    # Fix the diagonal terms (2*i and 2*j lags)
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                # Perturb to escape
                n_perturb = max(5, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and (idx + 1) % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-4.0 * (x - 0.5) ** 2 / (0.1 + 0.15 * rng.random()))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    elif choice == 4:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 5:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 6:
        a = np.exp(-3.0 * x) * (1.0 + 0.3 * np.sin(4 * np.pi * x))
    elif choice == 7:
        a = np.sinc(4.0 * (x - 0.5)) ** 2 + 0.05
    elif choice == 8:
        # Two-lobe
        sigma = 0.1 + 0.1 * rng.random()
        a = np.exp(-(x - 0.25) ** 2 / (2 * sigma ** 2)) + np.exp(-(x - 0.75) ** 2 / (2 * sigma ** 2))
    elif choice == 9:
        # Smooth bumps
        a = 0.3 + 0.5 * np.sin(np.pi * x) ** 2 + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 10:
        # Power law taper
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** (1.0 + rng.random() * 2.0)
    else:
        # Gaussian mixture
        n_comp = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(n_comp):
            mu = rng.random()
            sig = 0.05 + 0.15 * rng.random()
            a += np.exp(-(x - mu) ** 2 / (2 * sig ** 2))

    a = np.maximum(a, 0.0)
    a = a + 0.005 * np.abs(rng.standard_normal(n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def mock_eval(a):
    """Fast local evaluation (same formula as evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    conv = fast_conv_self(a)
    max_b = conv.max()
    return float(2 * n * max_b / (sum_a ** 2))


def anneal_restart(a0, rng, n_iters=500, time_limit=60.0):
    """Simulated annealing in the space of sequences."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    T = 0.1 * best_score / n

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb
        idx = rng.integers(0, n, size=rng.integers(1, max(2, n // 100) + 1))
        a_pert = a.copy()
        a_pert[idx] *= (1.0 + T * rng.standard_normal(len(idx)))
        a_pert = np.maximum(a_pert, 0.0)
        s_p = a_pert.sum()
        if s_p < 1e-15:
            continue
        a_pert = a_pert * norm / s_p

        score_p = fast_conv_self(a_pert).max()
        delta = score_p - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_pert
            if score_p < best_score:
                best_score = score_p
                best_a = a.copy()
        # Cool
        T *= 0.995

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = mock_eval(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = mock_eval(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # If no parent, initialize
    if not parent_loaded:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = mock_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LSE gradient descent (fast warmup) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gradient_descent(best_seq.copy(), t1, verbose=True)
        try_update(a_out, "lse1")

    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp(best_seq.copy(), t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # === Phase 3: Pressure CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Pressure CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = pressure_cd(best_seq.copy(), t3, rng, verbose=True)
        try_update(a_out, "cd1")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov R2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp(best_seq.copy(), t4, solver, verbose=True)
        try_update(a_out, "lp2")

    # === Phase 5: CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: CD polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = pressure_cd(best_seq.copy(), t5, rng, verbose=False)
        try_update(a_out, "cd2")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length exploration (budget={remaining:.0f}s) ===")
    if remaining > 60:
        # Try larger length
        n_long = min(n_base + 400, 2400)
        t6a = min(remaining * 0.3, 120.0)
        if t6a > 20:
            init_long = interpolate_seq(best_seq, n_long)
            a_out, _ = nesterov_lp(init_long, t6a, solver, verbose=False)
            try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            # Try a fresh structured init at same length with LP
            t6b = min(remaining * 0.3, 100.0)
            if t6b > 20:
                fresh = structured_init(n_base, rng)
                a_out, _ = lse_gradient_descent(fresh, min(t6b * 0.3, 30), verbose=False)
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 20:
                    a_out, _ = nesterov_lp(a_out, min(remaining * 0.7, 80), solver, verbose=False)
                    try_update(a_out, "fresh_lp")

    # === Phase 7: Annealing restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Annealing (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t7 = min(remaining * 0.3, 60.0)
        a_out = anneal_restart(best_seq.copy(), rng, n_iters=3000, time_limit=t7)
        try_update(a_out, "anneal")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t8 = min(remaining * 0.5, 80.0)
        a_out, _ = nesterov_lp(best_seq.copy(), t8, solver, verbose=False)
        try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = pressure_cd(best_seq.copy(), min(remaining * 0.6, 40), rng, verbose=False)
            try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = mock_eval(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        # Verify with actual evaluate_sequence
        try:
            verify_score = eval_fn(best_seq)
            if np.isfinite(verify_score):
                print(f"Verify: mock={best_score:.6f}, actual={verify_score:.6f}")
        except Exception as e:
            print(f"Verify error: {e}")
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
