def _project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_step(a, target_sum, step_size):
    """One projected subgradient step for minimizing max_k conv(a,a)_k."""
    conv = _autocorr(a)
    k_star = np.argmax(conv)
    n = len(a)
    # Subgradient of max_k conv(a,a)_k at a is 2 * (T_{k*} @ a)
    # where (T_k @ a)_j = a[k-j] (circular-like, but linear)
    # Actually conv(a,a)_k = sum_j a[j]*a[k-j], so d/da_j = a[k-j] + a[j-k] (for the max at k)
    # But since we're at the max, the subgradient is 2*(shifted a) or just (a shifted by k) + (a shifted by -k)
    # For simplicity and since a >= 0, use: g[j] = a[k-j] + a[j+k] (with bounds)
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        idx1 = k_star - j
        if 0 <= idx1 < n:
            g[j] += a[idx1]
        idx2 = j + k_star
        if 0 <= idx2 < n:
            g[j] += a[idx2]
    # If k_star == 0, g = 2*a. If k_star > 0, it's the "folded" version.
    # Actually for autocorrelation: conv(a,a)_k = sum_j a[j]*a[j+k] (for k>=0, linear conv)
    # d/d a[j] of conv(a,a)_k = a[j+k] + (a[j-k] if j-k >= 0)
    # Let me just use the full gradient properly
    g = np.zeros(n)
    if k_star == 0:
        g = 2.0 * a
    else:
        # conv(a,a)_{k_star} = sum_j a[j] * a[j+k_star] (for j where j+k_star < n)
        # + sum_j a[j+k_star] * a[j] (same thing)
        # d/da[j] = a[j+k_star] (if j+k_star < n) + a[j-k_star] (if j-k_star >= 0)
        for j in range(n):
            if j + k_star < n:
                g[j] += a[j + k_star]
            if j - k_star >= 0:
                g[j] += a[j - k_star]
    # Projected gradient step
    a_new = a - step_size * g
    a_new = _project_simplex(a_new, target_sum)
    return a_new, k_star, g

                if j - k_star >= 0:
                    g[j] += y[j - k_star]

        # Adaptive step size
        step = norm / (np.linalg.norm(g) + 1e-10) * 0.5

        x_new = y - step * g
        x_new = _project_simplex(x_new, norm)

        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - y)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x_new.copy()
        else:
            y = y * norm / s_y
        t_k = t_new

        sc_x = float(2.0 * n * _autocorr(x_new).max())
        if sc_x < best_sc:
            best_sc = sc_x
            best_a = x_new.copy()

        sc_y = float(2.0 * n * _autocorr(y).max())
        if sc_y < best_sc:
            best_sc = sc_y
            best_a = y.copy()

    return best_a, best_sc / (2.0 * n)


def _coord_descent(a0, time_limit, seed=0):
    """Coordinate descent with golden section for best coordinate."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_sc = float(2.0 * n * _autocorr(a).max())
    t0 = time.monotonic()
    no_improve = 0
    passes = 0
    phi = 0.6180339887498949

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-10 * norm:
                continue

            lo = 0.0
            hi = ai * 2.5 + 0.05 * norm
            c1 = lo + (1.0 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            at1 = a.copy()
            at1[i] = c1
            at1 = at1 * norm / at1.sum()
            f1 = _autocorr(at1).max()

            at2 = a.copy()
            at2[i] = c2
            at2 = at2 * norm / at2.sum()
            f2 = _autocorr(at2).max()

            for _ in range(25):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1.0 - phi) * (hi - lo)
                    at1 = a.copy()
                    at1[i] = c1
                    at1 = at1 * norm / at1.sum()
                    f1 = _autocorr(at1).max()
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    at2 = a.copy()
                    at2[i] = c2
                    at2 = at2 * norm / at2.sum()
                    f2 = _autocorr(at2).max()

            best_c = c1 if f1 < f2 else c2
            f_best = min(f1, f2)
            if f_best < best_sc / (2.0 * n) - 1e-13:
                a[best_c] = 0  # placeholder
                a = a.copy()
                a[i] = best_c
                a = a * norm / a.sum()
                cur_sc = float(2.0 * n * _autocorr(a).max())
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_sc / (2.0 * n)


def _pair_swap(a0, time_limit, seed=0):
    """Pair swap coordinate descent."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_sc = float(2.0 * n * _autocorr(a).max())
    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            if i == j:
                continue
            ai, aj = a[i], a[j]
            if ai < 1e-10 * norm and aj < 1e-10 * norm:
                continue

            cur_max = _autocorr(a).max()
            best_delta = 0.0
            best_val = cur_max

            for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                d = ai * frac
                if d <= 0:
                    continue
                at = a.copy()
                at[i] = ai - d
                at[j] = aj + d
                at = at * norm / at.sum()
                v = _autocorr(at).max()
                if v < best_val - 1e-14:
                    best_val = v
                    best_delta = d

                d2 = aj * frac
                if d2 <= 0:
                    continue
                at2 = a.copy()
                at2[i] = ai + d2
                at2[j] = aj - d2
                at2 = at2 * norm / at2.sum()
                v2 = _autocorr(at2).max()
                if v2 < best_val - 1e-14:
                    best_val = v2
                    best_delta = -d2

            if abs(best_delta) > 1e-14:
                at = a.copy()
                if best_delta > 0:
                    at[i] -= best_delta
                    at[j] += best_delta
                else:
                    at[i] += abs(best_delta)
                    at[j] -= abs(best_delta)
                at = at * norm / at.sum()
                sc = float(2.0 * n * _autocorr(at).max())
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = at.copy()
                    a = at
                    improved = True
                    no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_a, best_sc / (2.0 * n)


def _structured_init(n, style):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif style == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(5 * np.pi * x) + 0.1 * np.cos(11 * np.pi * x)
    elif style == 7:
        a = np.sin(np.pi * x) ** 0.5 * (1.0 + 0.2 * np.cos(8 * np.pi * x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def _perturb(a, rng, scale=0.04):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.03)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def propose_candidate():
    rng = np.random.default_rng(31415)
    t_global = time.monotonic()
    time_budget = 1750.0
    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        for sty in range(8):
            cand = _structured_init(n_init, sty)
            sc = _eval(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = _eval(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            el = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={el:.0f}s, d={old-sc:.6f})")

    def get_time():
        return time.monotonic() - t_global

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov on parent (40% time) ===
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min((time_budget - get_time()) * 0.40, 600.0)
    if t1 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = _nesterov_lp(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: FISTA subgradient (15% time) ===
    print(f"\n=== Phase 2: FISTA n={len(best_seq)} ===")
    t2 = min((time_budget - get_time()) * 0.15, 250.0)
    if t2 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _fista_subgradient(a_norm, t2, seed=7)
        try_update(a_out, f"fista_n{n_cur}")

    # === Phase 3: LP again with different seed (15% time) ===
    print(f"\n=== Phase 3: LP restart n={len(best_seq)} ===")
    t3 = min((time_budget - get_time()) * 0.15, 250.0)
    if t3 > 60:
        n_cur = len(best_seq)
        pert = _perturb(best_seq, rng, 0.03)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = _nesterov_lp(a_norm, t3, solver, verbose=True, seed=123)
        try_update(a_out, f"lp3_n{n_cur}")

    # === Phase 4: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 4: Multi-length ===")
    t4_total = min((time_budget - get_time()) * 0.15, 200.0)
    n_candidates = [len(best_seq) + 300, len(best_seq) + 500, max(1000, len(best_seq) - 200)]
    t4_each = t4_total / max(1, len(n_candidates))

        t4 = min(t4_each, rem - 30)

            break
        print(f"  Trying n={nc}...")

        a_out, _ = _nesterov_lp(a_norm, t4, solver, verbose=False, seed=99 + nc % 100)

    # === Phase 6: Pair swap polish ===
    rem = time_budget - get_time()
    if rem > 40:
        print(f"\n=== Phase 6: Pair swap n={len(best_seq)} ===")
        t6 = min(rem * 0.3, 80.0)
        if t6 > 20:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = _pair_swap(a_norm, t6, seed=77)
            try_update(a_out, f"pswap_n{n_cur}")

    # === Phase 7: Final LP polish with remaining time ===
    rem = time_budget - get_time()
    if rem > 60:
        print(f"\n=== Phase 7: Final LP n={len(best_seq)} ===")

            n_cur = len(best_seq)

            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()

    # Final check
    if best_seq is not None:

    else:
        print("\nERROR: No valid sequence!")
        return [1.0, 1.0]

    return [float(x) for x in best_seq]
