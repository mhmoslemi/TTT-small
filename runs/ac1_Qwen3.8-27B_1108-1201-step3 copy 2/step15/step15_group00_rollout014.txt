def compute_hybrid_subgradient(a, ac, top_k=30, beta=8.0, hard_weight=0.5):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)

    # Soft weights
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    # Hard weight on the argmax
    grad_soft = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad_soft[idx] += 2.0 * wi * a[j_idx]

    # Hard subgradient at the single maximum
    k_max_pos = int(np.argmax(ac))
    i_start = max(0, k_max_pos - n + 1)
    i_end = min(n, k_max_pos + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k_max_pos - idx
    grad_hard = np.zeros(n)
    grad_hard[idx] += 2.0 * a[j_idx]

    grad = hard_weight * grad_hard + (1.0 - hard_weight) * grad_soft
    return grad

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build constraint matrix T where T[j, i] = y[j-i] if valid
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Fallback: use uniform direction
            g_dir = np.ones(n) / n
        else:
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                g_dir = np.ones(n) / n
            else:
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                b_sum = b_val.sum()
                if b_sum < 1e-15:
                    g_dir = np.ones(n) / n
                else:
                    g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:

            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:

        mx = fast_autocorr(x).max()
        if mx < best_m:

        if verbose and t % 3 == 0:

            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                pass

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(1, min(5, n))
            srcs = rng.choice(n, size=n_pert, replace=False)
            dsts = rng.choice(n, size=n_pert, replace=False)
            for si, di in zip(srcs, dsts):
                if a[si] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[si] * frac
                    a[si] -= amt
                    a[di] += amt
            a = np.maximum(a, 0.0)
            s_t = a.sum()
            if s_t > 1e-15:
                a = a / s_t
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with adaptive time allocation."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: AMD v3 (primary workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD v3 n={n0} ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_mirror_descent_v3(a_in, t1, top_k=50, beta=6.0,
                                                  c0=3.5, hard_w=0.4, verbose=True)
        try_update(a_out, f"amd3_n{n0}")

    # === Phase 2: Spectral shaping (new) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Spectral n={len(best_seq)} ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_shaping(a_in, t2, verbose=True)
        try_update(a_out, f"spectral_n{len(best_seq)}")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP n={len(best_seq)} ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{len(best_seq)}")

    # === Phase 4: AMD polish after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: AMD polish n={len(best_seq)} ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_mirror_descent_v3(a_in, t4, top_k=60, beta=5.0,
                                                  c0=3.0, hard_w=0.5, verbose=False)
        try_update(a_out, f"amd_polish_n{len(best_seq)}")

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coord n={len(best_seq)} ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish_aggressive(a_in, t5, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Scale to different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    for target_n in [n0 + 400, n0 - 200, n0 + 800]:
        if target_n < 100 or target_n > 4000:
            continue
        if remaining < 60:
            break
        print(f"\n=== Phase 6: Scale to n={target_n} ===")
        a_long = interpolate_sequence(best_seq, target_n)
        t6 = min(remaining * 0.15, 150.0)
        if t6 > 30:
            a_out, _ = accelerated_mirror_descent_v3(norm_sum1(a_long), t6,
                                                      top_k=40, beta=6.0, c0=3.0,
                                                      hard_w=0.4, verbose=False)
            try_update(a_out, f"amd_n{target_n}")
            remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Alternative initialization + AMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Alt init + AMD ===")
        x = np.linspace(0, 1, n_best)
        inits = []
        # Bimodal
        bim = np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2)
        inits.append(bim / bim.sum())
        # Centered triangle
        tri = np.maximum(1 - 2 * np.abs(x - 0.5), 0) ** 3
        inits.append(tri / max(tri.sum(), 1e-15))
        # Blend with best
        blend = 0.6 * norm_sum1(best_seq) + 0.4 * inits[0]
        inits.append(norm_sum1(blend))

        t_per = min(remaining * 0.20, 80.0)
        for idx, init in enumerate(inits[:3]):
            if time.monotonic() - t_global > time_budget - 50:
                break
            t7 = min(t_per, time_budget - 50 - (time.monotonic() - t_global))
            if t7 > 20:
                a_out, _ = accelerated_mirror_descent_v3(init, t7, top_k=40,
                                                          beta=6.0, c0=3.0,
                                                          hard_w=0.4, verbose=False)
                try_update(a_out, f"alt{idx}_amd")

    # === Phase 8: Spectral + Coord combined final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final spectral+coord n={len(best_seq)} ===")
    if remaining > 60:
        t8a = min(remaining * 0.4, 80.0)
        t8b = min(remaining * 0.4, 80.0)
        if t8a > 15:
            a_out, _ = spectral_shaping(norm_sum1(best_seq), t8a, verbose=False)
            try_update(a_out, f"spectral_final")
        if t8b > 15:
            a_out, _ = coordinate_polish_aggressive(norm_sum1(best_seq), t8b,
                                                     verbose=True, rng_seed=999)
            try_update(a_out, f"coord_final")

    # === Phase 9: Final LP if time and size permit ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2500 and remaining > 40:
        print(f"\n=== Phase 9: Final LP n={n_best} ===")
        t9 = min(remaining * 0.7, 150.0)
        if t9 > 25:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t9, solver, verbose=True)
            try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
