Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

_solver = None

def get_lp_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _solver

def fft_autocorr(a):
    n = len(a)
    out_len = 2 * n - 1
    fsize = 1 << (out_len - 1).bit_length()
    spec = np.fft.rfft(np.asarray(a, dtype=np.float64), fsize)
    return np.fft.irfft(spec * spec, fsize)[:out_len]

def safe_eval(a):

        return float(evaluate_sequence([float(x) for x in a]))

def norm_to_S(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * S / len(a)
    return a * S / s

def build_T(a, lags):
    n = len(a)
    T = np.zeros((len(lags), n))
    for i, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[i, j] = a[k - j]
    return T

def solve_lp(a, M, lags, solver):
    n = len(a)
    T = build_T(a, lags)
    b = cp.Variable(n, nonneg=True)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    try:
        duals = np.asarray(cons[0].dual_value, dtype=np.float64)
    except Exception:
        duals = None
    return b_val, duals

def lp_nesterov(a_init, time_limit, solver, verbose=True, seed=42):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / a.sum() * S

    best_a = a.copy()
    best_M = fft_autocorr(a).max()

    rng = np.random.default_rng(seed)
    y = a.copy()
    y_prev = a.copy()
    active = None
    stall = 0
    t0 = time.monotonic()

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fft_autocorr(y)
        M = conv_y.max()
        if M < 1e-14:
            break

        if active is None or t % 25 == 0:
            tol = max(0.0005, 0.03 * np.exp(-t / 400.0))
            threshold = M * (1.0 - tol)
            active = np.where(conv_y >= threshold)[0]
            top_k = np.argsort(conv_y)[-max(5, n // 80):]
            active = np.unique(np.concatenate([active, top_k]))
            if len(active) > 2 * n - 2:
                active = np.arange(2 * n - 1)

        b, duals = solve_lp(y, M, active, solver)
        if b is None:
            if t % 15 == 0:
                active = np.arange(2 * n - 1)
                b, duals = solve_lp(y, M, active, solver)
            if b is None:
                continue

        # Verify complement (sampled)
        if len(active) < 2 * n - 2:
            complement = np.setdiff1d(np.arange(2 * n - 1), active)
            sample_size = min(80, len(complement))
            check_idx = complement[rng.choice(len(complement), size=sample_size, replace=False)]
            T_check = build_T(y, check_idx)
            viol = T_check @ b
            viol_mask = viol > M * 1.0005
            if viol_mask.any():
                new_active = np.unique(np.concatenate([active, check_idx[viol_mask]]))
                active = new_active
                b, duals = solve_lp(y, M, active, solver)
                if b is None:
                    active = np.arange(2 * n - 1)
                    b, _ = solve_lp(y, M, active, solver)
                    if b is None:
                        continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        g = b / b_sum * S

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum() * S

        if t == 1:
            a_new = g.copy()
        else:
            mu = min((t - 1.0) / (t + 2.0), 0.99)
            a_new = y_new + mu * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum() * S

        y_prev = y
        y = y_new
        a = a_new

        sc_y = fft_autocorr(y).max()
        if sc_y < best_M:
            best_M = sc_y
            best_a = y.copy()
            stall = 0
        sc_a = fft_autocorr(a).max()
        if sc_a < best_M:
            best_M = sc_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 60:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active = None
            stall = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best_M={best_M:.10f}, |act|={len(active)}")

    return best_a, best_M

def coord_descent(a_init, time_limit, verbose=True):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / a.sum() * S

    best_a = a.copy()
    best_M = fft_autocorr(a).max()
    t0 = time.monotonic()

    preconv = fft_autocorr(a)
    pass_num = 0
    no_improve = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-15:
                continue

            # Compute contribution vector c[k] = a[k-j] + a[j-k] (with bounds, excluding self)
            c = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                v = 0.0
                i1 = k - j
                if 0 <= i1 < n:
                    v += a[i1]
                i2 = j - k
                if 0 <= i2 < n:
                    v += a[i2]
                c[k] = v

            # Background: conv without a[j]'s contribution
            # conv[k] = bg[k] + aj*c[k] - aj^2 * 1_{k=2j}
            # So bg[k] = conv[k] - aj*c[k] + aj^2 * 1_{k=2j}
            bg = preconv.copy()
            bg -= aj * c
            if 2 * j < 2 * n - 1:
                bg[2 * j] += aj * aj

            # Try candidate x values
            s_others = a.sum() - aj
            candidates = [0.0, aj * 0.25, aj * 0.5, aj * 0.75, aj, aj * 1.25, aj * 1.5, s_others * 0.5]
            candidates = [x for x in candidates if 0 <= x <= S]
            # Add value that equalizes top two peaks (approximate)
            top2 = np.argsort(preconv)[-2:]
            if len(top2) == 2 and preconv[top2[1]] > 0:
                # Solve: bg[k1] + (x-aj)*c[k1] + (x-aj)^2*d1 = bg[k2] + (x-aj)*c[k2] + (x-aj)^2*d2
                # where d1 = 1 if k1==2j else 0
                k1, k2 = top2[1], top2[0]
                d1 = 1.0 if k1 == 2 * j else 0.0
                d2 = 1.0 if k2 == 2 * j else 0.0
                # (x-aj)^2*(d1-d2) + (x-aj)*(c[k1]-c[k2]) + (bg[k1]-bg[k2]) = 0
                A_q = d1 - d2
                B_q = c[k1] - c[k2]
                C_q = bg[k1] - bg[k2]
                if abs(A_q) > 1e-12:
                    disc = B_q * B_q - 4 * A_q * C_q
                    if disc >= 0:
                        x_eq1 = (-B_q + np.sqrt(disc)) / (2 * A_q) + aj
                        x_eq2 = (-B_q - np.sqrt(disc)) / (2 * A_q) + aj
                        for xe in [x_eq1, x_eq2]:
                            if 0 <= xe <= S:
                                candidates.append(xe)
                elif abs(B_q) > 1e-12:
                    x_eq = -C_q / B_q + aj
                    if 0 <= x_eq <= S:
                        candidates.append(x_eq)

            best_x = aj
            best_obj = 2 * n * preconv.max() / (a.sum() ** 2)

            for x in candidates:
                delta = x - aj
                new_conv = bg + delta * c
                if 2 * j < 2 * n - 1:
                    new_conv[2 * j] += delta * delta
                new_sum = s_others + x
                if new_sum < 1e-10:
                    continue
                obj = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                new_diff = best_x - aj
                new_conv = bg + new_diff * c
                if 2 * j < 2 * n - 1:
                    new_conv[2 * j] += new_diff * new_diff
                new_sum = s_others + best_x
                if new_sum > 1e-10:
                    new_score = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                    if new_score < 2 * n * best_M / S**2 - 1e-14:
                        a[j] = best_x
                        a = a / a.sum() * S
                        preconv = fft_autocorr(a)
                        cur_M = preconv.max()
                        sc = 2 * n * cur_M / S**2
                        if sc < 2 * n * best_M / S**2:
                            best_M = cur_M
                            best_a = a.copy()
                            improved = True
                            # Bisection refinement
                            lo = min(aj, best_x)
                            hi = max(aj, best_x)
                            for _ in range(10):
                                mid = (lo + hi) / 2.0
                                d2 = mid - aj
                                nc = bg + d2 * c
                                if 2 * j < 2 * n - 1:
                                    nc[2 * j] += d2 * d2
                                ns = s_others + mid
                                if ns < 1e-10:
                                    break
                                obj_mid = 2 * n * nc.max() / (ns * ns)
                                if obj_mid < 2 * n * cur_M / S**2:
                                    lo = mid
                                else:
                                    hi = mid
                            final_x = (lo + hi) / 2.0
                            a[j] = final_x
                            a = a / a.sum() * S
                            preconv = fft_autocorr(a)
                            if preconv.max() < best_M:
                                best_M = preconv.max()
                                best_a = a.copy()
                    else:
                        a[j] = aj
                        a = a / a.sum() * S

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and pass_num % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_num} ({elapsed:.0f}s): best_M={best_M:.10f}")

    return best_a, best_M

def subgrad_polish(a_init, time_limit, verbose=True):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / a.sum() * S

    best_a = a.copy()
    best_M = fft_autocorr(a).max()
    t0 = time.monotonic()

    step = 2.0 / n
    patience = 0
    max_patience = 300
    momentum = np.zeros(n)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fft_autocorr(a)
        M_curr = conv.max()

        # Get top peaks with weights
        n_peaks = max(3, min(8, n // 200))
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.01, 1e-10))
        weights /= weights.sum()

        # Compute combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for j in range(n):
                v = 0.0
                j1 = k - j
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = j - k
                if 0 <= j2 < n:
                    v += a[j2]
                grad[j] += w * v

        # Nesterov momentum
        momentum = 0.9 * momentum - step * grad
        a_trial = a + momentum
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = a_trial / a_trial.sum() * S

        sc = fft_autocorr(a_trial).max()
        if sc < best_M - 1e-13:
            a = a_trial
            best_M = sc
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.02, 5.0)
        else:
            patience += 1
            step *= 0.95
            momentum *= 0.5
            if step < 1e-10:
                break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best_M={best_M:.10f}")

    return best_a, best_M

def smooth_sequence(a, window=5):
    """Apply a mild smoothing filter."""
    n = len(a)
    kernel = np.ones(window) / window
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s > 1e-15:
        a_smooth = a_smooth / s
    return a_smooth
