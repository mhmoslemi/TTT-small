def cross_corr(x, y):
    """Compute cross-correlation (x * y)[i] = sum_k x[k]*y[i+k] for i in range(len(x))."""
    n = len(x)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * np.conj(Y), fft_size)[:n]


def smooth_grad(a, w):

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                    if dst == src:
                        continue

                    # Try moving 20% of a[src] to dst
                    delta = a[src] * 0.2
                    if delta < norm * 1e-12:
                        continue

                    a_test = a.copy()
                    a_test[src] -= delta
                    a_test[dst] += delta
                    s_test = a_test.sum()
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    max_test = conv_test.max()

                    if max_test < best_score - 1e-10:
                        a = a_test
                        conv = conv_test
                        best_score = max_test
                        best_a = a.copy()
                        improved_this_pass = True
                        no_improve_passes = 0
                        # Update peaks
                        peak_thresh = 0.9 * conv.max()
                        peaks = set(np.where(conv >= peak_thresh)[0].tolist())
                    else:
                        # Try smaller step
                        delta2 = delta * 0.3
                        a_test2 = a.copy()
                        a_test2[src] -= delta2
                        a_test2[dst] += delta2
                        a_test2 = a_test2 * norm / a_test2.sum()
                        conv_test2 = fast_autocorr(a_test2)
                        if conv_test2.max() < conv.max() - 1e-10:
                            a = a_test2
                            conv = conv_test2
                            if conv.max() < best_score:
                                best_score = conv.max()
                                best_a = a.copy()
                            improved_this_pass = True
                            no_improve_passes = 0

        if not improved_this_pass:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD-PQ pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence using frequency mixing."""
    a = np.ones(n)
    # Add several cosine components at different frequencies
    n_freqs = min(8, n // 2)
    for k in range(1, n_freqs + 1):
        freq = k * n // (n_freqs + 1)
        if freq == 0:
            continue
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a * np.sqrt(2.0 * n) / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = np.abs(rng.standard_normal(n))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a * np.sqrt(2.0 * n) / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def fft_size_cached(a):
    n = len(a)
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()
