1. LP best-response with Nesterov acceleration (primary)
2. Subgradient descent with Nesterov (fast alternative)
3. Mirror descent with multiplicative updates
4. Symmetrization exploitation
5. Coordinate descent polishing
6. Multi-length exploration with interpolation
7. Aggressive active set management with violation-driven expansion
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver_cache = None


def get_best_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    candidates = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for s in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.SCS
    print(f"  Fallback LP solver: SCS")
    return _solver_cache


def fast_conv(a):
    """Full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation without calling the official evaluator."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_conv(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def evaluate_safe(a):

def build_active_constraints(a, active_idx):
    """Build constraint matrix T where T[row, j] = a[k-j] for active lag k."""
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_best_response(a, M, solver, active_idx=None):
    """Solve LP: max sum(b) s.t. conv(a,b)[active] <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None, None
    # Check violations on inactive constraints
    violations = None
    if active_idx is not None and len(active_idx) < size:
        inactive = np.setdiff1d(np.arange(size), active_idx)
        if len(inactive) > 0:
            T_check = build_active_constraints(a, inactive)
            viol = T_check @ val
            viol_idx = np.where(viol > M * (1.0 + 1e-9))[0]
            if len(viol_idx) > 0:
                violations = inactive[viol_idx]
    return val, violations


def compute_subgradient(a, conv, near_max_frac=0.05):
    """Compute subgradient of max(conv) at point a."""
    n = len(a)
    M = conv.max()
    if M < 1e-15:
        return np.zeros(n)
    threshold = M * (1.0 - near_max_frac)
    active_k = np.where(conv >= threshold)[0]
    g = np.zeros(n)
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                g[j] += a[idx]
    g /= len(active_k)
    return g


def nesterov_lp_subgrad(a0, time_limit, solver, verbose=True, seed=0,
                         use_symmetry=False, hybrid_mode=True):
    """
    Nesterov-accelerated iteration with LP best-response and subgradient fallback.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    if use_symmetry:
        a = 0.5 * (a + a[::-1])
        s = a.sum()
        if s > 1e-15:
            a = a * norm / s

    best_a = a.copy()
    best_conv = fast_conv(a)
    best_score = best_conv.max()

    y = a.copy()
    prev_y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 60
    lp_fail_count = 0
    eta_base = 1.0 / n

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv(y)
        M_curr = conv_y.max()
        if M_curr < 1e-15:
            break

        # Adaptive active set
        if active_set is None or t % 30 == 0:
            tol = 0.02 + 0.03 * np.exp(-t / 200.0)
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 5:
                active_set = np.where(conv_y >= M_curr * 0.85)[0]
            # Cap active set size for speed
            if len(active_set) > 0.3 * size:
                # Keep top 30%
                idx_sorted = np.argsort(conv_y)[::-1]
                active_set = idx_sorted[:int(0.3 * size)]

        # Get direction: LP or subgradient
        g_dir = None
        use_lp = (hybrid_mode and lp_fail_count < 5) or not hybrid_mode

        if use_lp:
            b_max, violations = lp_best_response(y, M_curr, solver, active_set)
            if b_max is not None:
                if violations is not None:
                    active_set = np.union1d(active_set, violations)
                    # Re-solve with expanded active set (only if small expansion)
                    if len(active_set) < 0.5 * size:
                        b_max, violations2 = lp_best_response(y, M_curr, solver, active_set)
                        if b_max is not None:
                            b_max = b_max
                        else:
                            b_max = None
                if b_max is not None:
                    b_sum = b_max.sum()
                    g_dir = b_max * norm / b_sum
                    lp_fail_count = 0
                else:
                    lp_fail_count += 1
            else:
                lp_fail_count += 1
                if lp_fail_count % 3 == 0:
                    active_set = None

        if g_dir is None:
            # Subgradient fallback
            g_sub = compute_subgradient(y, conv_y, near_max_frac=0.03)
            g_norm = np.linalg.norm(g_sub)
            if g_norm < 1e-15:
                g_dir = y.copy()
            else:
                # Mirror descent style: move opposite to subgradient
                scale = norm / g_norm
                g_dir = y - eta_base * scale * g_sub
                g_dir = np.maximum(g_dir, 0.0)
                s_dir = g_dir.sum()
                if s_dir < 1e-15:
                    g_dir = y.copy()
                else:
                    g_dir = g_dir * norm / s_dir

        if use_symmetry:
            g_dir = 0.5 * (g_dir + g_dir[::-1])
            s = g_dir.sum()
            if s > 1e-15:
                g_dir = g_dir * norm / s

        # Nesterov update
        # Use adaptive step size
        alpha_t = 2.0 / (t + 2.0)
        alpha_t = max(alpha_t, 0.01)

        y_new = (1.0 - alpha_t) * y + alpha_t * g_dir
        y_new = np.maximum(y_new, 0.0)
        if use_symmetry:
            y_new = 0.5 * (y_new + y_new[::-1])
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum extrapolation
        if t == 1:
            a_new = g_dir.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.90)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            if use_symmetry:
                a_new = 0.5 * (a_new + a_new[::-1])
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_conv(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_conv(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_conv(avg_check).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0
            lp_fail_count = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}, active={len(active_set) if active_set is not None else 'all'}")

    return best_a, best_score


def coord_descent(a0, time_limit, rng, verbose=True):
    """Coordinate descent with golden section search on individual elements."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_conv(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    phi = 0.6180339887498949

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-10 * norm:
                # Try turning on a small value
                a_try = a.copy()
                a_try[i] = norm * 0.001
                s_try = a_try.sum()
                a_try = a_try * norm / s_try
                sc = fast_conv(a_try).max()
                if sc < best_score - 1e-12:
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                continue

            # Golden section search
            lo = 0.0
            hi = ai * 2.5 + norm * 0.05

            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            a_t1 = a.copy()
            a_t1[i] = c1
            a_t1 = a_t1 * norm / a_t1.sum()
            f1 = fast_conv(a_t1).max()

            a_t2 = a.copy()
            a_t2[i] = c2
            a_t2 = a_t2 * norm / a_t2.sum()
            f2 = fast_conv(a_t2).max()

            for _ in range(20):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a_t1 = a.copy()
                    a_t1[i] = c1
                    a_t1 = a_t1 * norm / a_t1.sum()
                    f1 = fast_conv(a_t1).max()
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a_t2 = a.copy()
                    a_t2[i] = c2
                    a_t2 = a_t2 * norm / a_t2.sum()
                    f2 = fast_conv(a_t2).max()

            best_c = c1 if f1 < f2 else c2
            best_f = min(f1, f2)

            if best_f < best_score - 1e-12:
                a[i] = best_c
                a = a * norm / a.sum()
                conv = fast_conv(a)
                best_score = conv.max()
                best_a = a.copy()
                pass_improved = True
                no_improve = 0

        if not pass_improve:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent: move mass between pairs."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_conv(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            ai, aj = a[i], a[j]
            if ai < 1e-12 * norm and aj < 1e-12 * norm:
                continue

            best_delta = 0.0
            best_val = best_score

            # Try several fractions
            max_d = max(ai, aj, norm * 0.01)
            for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9]:
                d = max_d * frac
                # i -> j
                if d <= ai:
                    a_try = a.copy()
                    a_try[i] = ai - d
                    a_try[j] = aj + d
                    s_try = a_try.sum()
                    if s_try > 1e-15:
                        a_try = a_try * norm / s_try
                        val = fast_conv(a_try).max()
                        if val < best_val - 1e-13:
                            best_val = val
                            best_delta = d
                # j -> i
                if d <= aj:
                    a_try = a.copy()
                    a_try[i] = ai + d
                    a_try[j] = aj - d
                    s_try = a_try.sum()
                    if s_try > 1e-15:
                        a_try = a_try * norm / s_try
                        val = fast_conv(a_try).max()
                        if val < best_val - 1e-13:
                            best_val = val
                            best_delta = -d

            if abs(best_delta) > 1e-14:
                a_try = a.copy()
                if best_delta > 0:
                    a_try[i] -= best_delta
                    a_try[j] += best_delta
                else:
                    a_try[i] += abs(best_delta)
                    a_try[j] -= abs(best_delta)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    sc = fast_conv(a_try).max()
                    if sc < best_score - 1e-13:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, verbose=True, seed=0,
                          use_symmetry=False, near_max_frac=0.03):
    """Pure subgradient Nesterov method (no LP)."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    if use_symmetry:
        a = 0.5 * (a + a[::-1])
        s = a.sum()
        if s > 1e-15:
            a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv(y)
        M_curr = conv_y.max()

        # Subgradient
        g = compute_subgradient(y, conv_y, near_max_frac)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Adaptive step size
        # Use diminishing step: eta_t = c / sqrt(t)
        eta = norm / (g_norm * np.sqrt(t + 1.0))
        eta = min(eta, 0.5 / n)

        y_new = y - eta * g
        y_new = np.maximum(y_new, 0.0)
        if use_symmetry:
            y_new = 0.5 * (y_new + y_new[::-1])
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov extrapolation
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.85)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            if use_symmetry:
                a_new = 0.5 * (a_new + a_new[::-1])
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_conv(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_conv(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_conv(avg_check).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        if stall_count > 100:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall_count = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    """Generate diverse structured initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    elif style == 8:
        # Power law
        a = (x + 0.01) ** (-0.5)
    elif style == 9:
        # Inverse power law
        a = (1.0 - x + 0.01) ** (-0.5)
    elif style == 10:
        # Three peaks
        a = np.exp(-20 * (x - 0.2) ** 2) + 0.7 * np.exp(-20 * (x - 0.5) ** 2) + np.exp(-20 * (x - 0.8) ** 2)
    elif style == 11:
        # U-shaped
        a = (x - 0.5) ** 2 + 0.1
    elif style == 12:
        # Random smooth
        coeffs = rng.standard_normal(8)
        a = 1.0
        for k, c in enumerate(coeffs):
            a += c / (k + 1) * np.cos((k + 1) * np.pi * x)
        a = np.maximum(a, 0.0)
    else:
        # Gaussian centered
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    """Perturb sequence with multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def propose_candidate():
    """Main optimization function."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_best_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        for sty in range(13):
            cand = structured_init(n_init, rng, sty)
            sc = evaluate_safe(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, delta={old_score-sc:.6f})")

    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)

    # === Phase 1: LP Nesterov on parent (30% time) ===
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_subgrad(a_norm, t1, solver, verbose=True, seed=0,
                                        use_symmetry=True, hybrid_mode=True)
        try_update(a_out, f"lp_sym_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 2: Subgradient Nesterov (15% time) ===
    print(f"\n=== Phase 2: Subgrad Nesterov n={len(best_seq)} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 40:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_nesterov(a_norm, t2, verbose=True, seed=42,
                                         use_symmetry=True, near_max_frac=0.02)
        try_update(a_out, f"subgrad_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 3: Asymmetric LP (15% time) ===
    print(f"\n=== Phase 3: LP Asymmetric n={len(best_seq)} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 40:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_subgrad(a_norm, t3, solver, verbose=True, seed=123,
                                        use_symmetry=False, hybrid_mode=True)
        try_update(a_out, f"lp_asym_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 4: Pair CD polish (10% time) ===
    print(f"\n=== Phase 4: Pair CD n={len(best_seq)} ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Coordinate descent (10% time) ===
    print(f"\n=== Phase 5: Coord Descent n={len(best_seq)} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t5, rng, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Perturbation + LP restarts (10% time) ===
    print(f"\n=== Phase 6: Perturbation restarts ===")
    t6_total = min(remaining * 0.10, 150.0)
    n_restarts = 2
    t6_each = t6_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t6 = min(t6_each, remaining - 30)
        if t6 < 30:
            break
        n_cur = len(best_seq)
        scale = 0.03 + 0.02 * r
        pert = perturb_seq(best_seq, rng, scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        use_sym = (r % 2 == 0)
        a_out, _ = nesterov_lp_subgrad(a_norm, t6, solver, verbose=False,
                                        seed=200 + r * 11, use_symmetry=use_sym,
                                        hybrid_mode=True)
        try_update(a_out, f"pert{r}_sym{use_sym}_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 7: Multi-length exploration ===")
    t7 = min(remaining * 0.10, 150.0)
    if t7 > 60:
        # Try n+200
        n_long = len(best_seq) + 200
        init_long = interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        t7a = t7 * 0.5
        a_out, _ = nesterov_lp_subgrad(a_norm, t7a, solver, verbose=True,
                                        seed=300, use_symmetry=True, hybrid_mode=True)
        try_update(a_out, f"lp_long_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            # Try n-200
            n_short = max(1000, len(best_seq) - 200)
            if n_short != len(best_seq):
                init_short = interpolate_seq(best_seq, n_short)
                a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
                t7b = min(remaining * 0.3, 80.0)
                a_out, _ = nesterov_lp_subgrad(a_norm, t7b, solver, verbose=False,
                                                seed=311, use_symmetry=True, hybrid_mode=True)
                try_update(a_out, f"lp_short_n{n_short}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final LP polish (20% of remaining) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} ===")
    t8 = max(remaining * 0.35, 60.0)
    t8 = min(t8, remaining - 20)
    if t8 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_subgrad(a_norm, t8, solver, verbose=True, seed=999,
                                        use_symmetry=True, hybrid_mode=True)
        try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: Final CD polish ===
    n_best = len(best_seq)
    if remaining > 50:
        print(f"\n=== Phase 9: Final CD n={n_best} ===")
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent(a_norm, t9, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 10: Final pair CD ===
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 10: Final PCD n={n_best} ===")
        t10 = min(remaining * 0.7, 80.0)
        if t10 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_cd(a_norm, t10, rng, verbose=False)
            try_update(a_out, f"pcd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
