import numpy as np

from scipy.fft import rfft, irfft

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):

def normalize(a, norm):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * norm / len(a)
    return a * norm / s

def build_T_sparsity(a, indices):
    """Build sparse rows of convolution matrix for given lag indices."""
    n = len(a)
    rows = np.zeros((len(indices), n))
    for ri, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            rows[ri, lo:hi] = a[k - np.arange(lo, hi)]
    return rows

def solve_lp_constraint_gen(a, M, solver, max_rounds=8, verbose=False):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b>=0
    Using constraint generation (cutting plane)."""
    n = len(a)
    size = 2 * n - 1
    
    # Initial active set: lags where a self-conv is high
    conv_a = fast_conv_self(a)
    threshold = M * 0.6
    active = np.where(conv_a >= threshold)[0]
    if len(active) < 20:
        active = np.where(conv_a >= M * 0.3)[0]
    if len(active) < 10:
        active = np.arange(size)
    
    b = None
    duals = None
    
    for round_idx in range(max_rounds):
        T_act = build_T_sparsity(a, active)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_act @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return b, duals
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return b, duals
        
        b = np.asarray(b_var.value, dtype=np.float64)
        b = np.maximum(b, 0.0)
        
        if b.sum() < 1e-15:
            return b, None
        
        # Get duals
        duals = None
        try:
            con = constraints[0]
            if con.dual_value is not None:
                d = np.asarray(con.dual_value, dtype=np.float64)
                duals = np.zeros(size)
                duals[active] = np.maximum(d, 0.0)
        except Exception:
            pass
        
        # Check for violated constraints
        if b is not None and b.sum() > 1e-15:
            cross = fast_conv_cross(a, b)
            violated = np.where(cross > M * (1.0 + 1e-6))[0]
            violated_not_active = violated[~np.isin(violated, active)]
            
            if len(violated_not_active) == 0:
                break  # No violations, solution is feasible
            
            # Add worst violated constraints
            n_add = min(len(violated_not_active), 200)
            worst = violated_not_active[np.argsort(cross[violated_not_active])[-n_add:]]
            active = np.unique(np.concatenate([active, worst]))
            
            if verbose:
                print(f"      CG round {round_idx}: added {len(worst)} constraints, total {len(active)}")
    
    return b, duals

def lse_descent(a0, time_limit, n_stages=10, verbose=True):
    """LSE gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    temps = np.geomspace(200.0, 0.05, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(400, int(remaining * 5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            # Adaptive step size
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            sc = fast_conv_self(a_new).max()
            if sc > max_c + 0.1 * T_val:
                # Backtrack
                step *= 0.5
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm / s_new
                else:
                    a_new = a.copy()
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"      LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")
    
    return best_a, best_score

def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, use_geom=False):

    n = len(a0)

    t0 = time.monotonic()
    y = a.copy()

    total_iters = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        total_iters += 1

        M = conv_y.max()
        if M < 1e-12:
            break

        b, duals = solve_lp_constraint_gen(y, M, solver, max_rounds=6, verbose=False)
        if b is None or b.sum() < 1e-15:
            continue
        
        g = normalize(b, norm)
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        if use_geom and t > 50:
            # Geometric mean update
            y_safe = np.maximum(y, 1e-30)
            g_safe = np.maximum(g, 1e-30)
            y_new = np.exp((1.0 - alpha) * np.log(y_safe) + alpha * np.log(g_safe))
        else:

        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)
        
        # Nesterov extrapolation
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        # Perturbation to escape local optima
        if no_improve == 150:
            rng_p = np.random.default_rng(t * 7 + 3)
            n_pert = max(20, n // 15)
            idx_p = rng_p.choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.3 * rng_p.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            y = normalize(y, norm)
            no_improve = 0
        
        # Average tracking

            if sc_check < best_score:
                best_score = sc_check
                best_a = a.copy()
        
        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_score / 2.0  # already normalized
            print(f"      LP t={t} ({elapsed:.0f}s): max_conv={best_score:.8f}, score={2*n*best_score/(norm*norm):.6f}")
    
    return best_a, best_score

                        best_v_for_j = new_max
                        best_d_for_j = d
                
                if best_d_for_j > 0 and best_v_for_j < best_val - 1e-12:
                    best_val = best_v_for_j
                    best_delta = best_d_for_j
                    best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize(a, norm)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = normalize(a, norm)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve >= 5:
                # Perturb
                n_pert = max(20, n // 20)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
                a = normalize(a, norm)
        else:
            no_improve = 0
        
        if verbose and no_improve % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PCD ({elapsed:.0f}s): score={2*n*best_score/(norm*norm):.6f}, no_improve={no_improve}")
    
    return best_a, best_score

def lse_descent_v2(a0, time_limit, verbose=True):
    """Improved LSE with adaptive temperature and backtracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.maximum(np.asarray(a0, dtype=np.float64), 0.0), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Adaptive temperature: start high, decrease based on progress
    T = 100.0
    min_T = 0.01
    patience = 0
    
    max_iters = int(time_limit * 10)  # rough upper bound on iterations
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 50:
            break
        
        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Softmax weights on convolution
        shifted = (conv - max_c) / T
        shifted = np.maximum(shifted, -300.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        
        # Gradient via FFT
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        aw = irfft(A * W, fft_size)
        g2 = aw[:n]
        grad = 2.0 * (g1 + g2)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Line search
        step = norm / (T * np.sqrt(n))
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new
        
        sc_new = fast_conv_self(a_new).max()
        
        if sc_new < max_c:
            a = a_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            # Cool down
            T = max(min_T, T * 0.995)
        else:
            # Backtrack
            for trial in range(10):
                step *= 0.5
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm / s_new
                    sc_new = fast_conv_self(a_new).max()
                    if sc_new < max_c:
                        a = a_new
                        if sc_new < best_score:
                            best_score = sc_new
                            best_a = a.copy()
                        patience = 0
                        break
            else:
                patience += 2
                T = max(min_T, T * 0.9)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSEv2 it={it} T={T:.3f} ({elapsed:.0f}s): score={2*n*best_score/(norm*norm):.6f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25)**2) + 0.8 * np.exp(-6.0 * (x - 0.75)**2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Edge-concentrated
        a = np.exp(-8.0 * x) + np.exp(-8.0 * (1.0 - x))
    elif choice == 7:
        # Multi-peak
        centers = rng.uniform(0.1, 0.9, size=4)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-50.0 * (x - c)**2)
    elif choice == 8:
        # Periodic
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.cos(2 * np.pi * 13 * x)
        a = np.maximum(a, 0.01)
    elif choice == 9:
        # Serrated
        a = np.abs(np.sin(4 * np.pi * x)) ** 0.6
    elif choice == 10:
        # Inverted U with noise
        a = np.sin(np.pi * x) * (1.0 + 0.2 * rng.standard_normal(n))
        a = np.maximum(a, 0.0)
    else:
        # Polynomial
        a = 1.0 - 0.5 * (2*x - 1)**4
        a = np.maximum(a, 0.01)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def get_solver():
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    print("  Using LP solver (fallback): ECOS")
    return cp.ECOS

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 0: Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # Phase 1: Quick LSE warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t_init = min(remaining * 0.06, 100.0)
    print(f"\n=== Phase 1: LSE warmup n={len(best_seq)} (budget={t_init:.0f}s) ===")
    if t_init > 15:
        a_out, _ = lse_descent(best_seq, t_init, verbose=False)
        try_update(a_out, "lse_init")
    
    # Phase 2: Main LP Nesterov optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True, use_geom=False)
        try_update(a_out, f"lp1_n{n2}")
    
    # Phase 3: LSE v2 refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: LSEv2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = lse_descent_v2(best_seq, t3, verbose=True)
        try_update(a_out, f"lsev2_n{n3}")
    
    # Phase 4: LP with geometric mixing
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 4: LP Geometric n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, verbose=True, use_geom=True)
        try_update(a_out, f"lp_geo_n{n4}")
    
    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    target_lengths = sorted(set([
        max(400, int(n_curr * 0.75)),
        n_curr,
        int(n_curr * 1.25),
        int(n_curr * 1.5),
        int(n_curr * 1.75),
    ]))
    
    print(f"\n=== Phase 5: Multi-length exploration (remaining={remaining:.0f}s) ===")
    for target_n in target_lengths:
        if time.monotonic() - t_global > time_budget - 150:
            break
        if target_n == n_curr and best_score == eval_fn(best_seq):
            continue  # Already optimized at this length
        
        per_length_time = min((time_budget - (time.monotonic() - t_global) - 100) / max(1, len(target_lengths)), 200.0)
        if per_length_time < 20:
            break
        
        init_long = interpolate_seq(best_seq, target_n)
        sc_init = eval_fn(init_long)
        print(f"  Trying n={target_n} (init score={sc_init:.6f}, budget={per_length_time:.0f}s)")
        
        # Quick LP at new length
        a_out, _ = lp_nesterov_optimize(init_long, per_length_time * 0.7, solver, verbose=False, use_geom=False)
        try_update(a_out, f"lp_len_{target_n}")
        
        # Quick LSE polish
        remaining_here = time_budget - (time.monotonic() - t_global)
        if remaining_here > 40:
            a_out2, _ = lse_descent_v2(best_seq, min(remaining_here * 0.3, 60.0), verbose=False)
            try_update(a_out2, f"lse_len_{len(best_seq)}")
    
    # Phase 6: Peak-targeting coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    t6 = min(remaining * 0.3, 250.0)
    print(f"\n=== Phase 6: Peak CD n={n6} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = peak_target_cd(best_seq, t6, rng, verbose=True)
        try_update(a_out, f"pcd_n{n6}")
    
    # Phase 7: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n7 = len(best_seq)
    t7 = min(remaining * 0.4, 150.0)
    print(f"\n=== Phase 7: Final LP n={n7} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = lp_nesterov_optimize(best_seq, t7, solver, verbose=False, use_geom=False)
        try_update(a_out, f"lp_final_n{n7}")
    
    # Phase 8: Final LSE
    remaining = time_budget - (time.monotonic() - t_global)
    n8 = len(best_seq)
    t8 = min(remaining * 0.6, 80.0)
    print(f"\n=== Phase 8: Final LSE n={n8} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_out, _ = lse_descent_v2(best_seq, t8, verbose=False)
        try_update(a_out, f"lse_final_n{n8}")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 1600
    
    return [float(x) for x in best_seq]
