import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize_to(a, target_sum):
    """Normalize a to have the target sum."""
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s

def lp_cutting_plane(a0, time_limit, verbose=True):
    """
    Cutting plane method: iteratively solve LP with linearized constraints.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    
    # Active constraints (cuts) to maintain
    active_cuts = []  # List of (C_k_a0, rhs) pairs
    
    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cur = a.copy()
        conv_cur = fast_autocorr(a_cur)
        M_cur = conv_cur.max()
        
        # Select top constraints for cuts
        top_k = min(50, 2*n-1)
        top_idx = np.argsort(conv_cur)[::-1][:top_k]
        
        # Build new cuts from current point
        new_cuts = []
        for k in top_idx:
            # C_k * a_cur gives the gradient direction
            # (C_k a_cur)[i] = a_cur[k-i] if 0 <= k-i < n
            gradient = np.zeros(n)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                gradient[lo:hi+1] = a_cur[k-hi:k-lo+1][::-1]
            # Constraint: a^T (2*gradient) <= M + conv_cur[k]
            # i.e., 2 * a @ gradient <= M + conv_cur[k]
            # But we also need M >= conv_cur[k] (since M >= max_k conv[k])
            # Actually, add constraint: 2 * a @ gradient <= M + conv_cur[k]
            # And M >= conv_cur[k] is implied if we're minimizing M.
            new_cuts.append((2.0 * gradient, conv_cur[k]))
        
        active_cuts = new_cuts  # Replace with new cuts (or accumulate)
        
        # Solve LP: min M s.t. 2*a@grad_i <= M + rhs_i for all i, sum(a)=norm, a>=0
        # Rearranged: 2*a@grad_i - M <= rhs_i
        import cvxpy as cp
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == norm, M_var >= 0]
        for grad, rhs in active_cuts:
            constraints.append(2.0 * a_var @ grad - M_var <= rhs)
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        M_new = float(M_var.value)
        a_new = normalize_to(a_new, norm)
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr_weights(weights, a):
    """Compute cross_correlation(weights, a)[:len(a)] where weights has length 2*len(a)-1."""
    n = len(a)
    fft_size = 1 << (2 * n - 1 + n - 1).bit_length()  # need at least (2n-1) + n - 1 = 3n-2
    W = rfft(weights, fft_size)
    A = rfft(a, fft_size)
    return irfft(W * np.conj(A), fft_size)[:n]

def evaluate_safe(a):

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize_to(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s

def smooth_optimize(a0, time_limit, verbose=True):

    for Ti in range(n_temps):
        if time.monotonic() - t0 > time_limit:
            break
        T = np.exp(log_temps[Ti])
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Number of steps for this temperature
        n_steps = max(50, int(remaining * 1.5 * (1.0 - Ti / n_temps)))
        n_steps = min(n_steps, 5000)
        
        y = a.copy()
        prev_a = a.copy()
        prev_grad = None
        bb_step = norm / (n * 0.5)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute softmax weights
            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()
            
            # Gradient
            grad = 2.0 * cross_corr_weights(w, y)
            
            # Barzilai-Borwein step size
            if prev_grad is not None:
                s_vec = y - prev_a
                y_vec = grad - prev_grad
                s_sq = np.dot(s_vec, s_vec)
                y_sq = np.dot(y_vec, y_vec)
                if s_sq > 1e-20 and y_sq > 1e-20:
                    sy = np.dot(s_vec, y_vec)
                    if abs(sy) > 1e-20:

                        gamma2 = sy / y_sq if sy > 0 else bb_step

            y_new = (1.0 - alpha) * y + alpha * (y - bb_step * grad)

            if it == 0:

            else:

                a_new = y_new + momentum * (y_new - y)
                a_new = project_simplex(a_new, norm)
            
            prev_a = y.copy()
            prev_grad = grad.copy()
            y = y_new
            a = a_new
            
            # Track best
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stuck_count = 0
            else:
                stuck_count += 1
            
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stuck_count = 0
            
            total_iters += 1
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.4f} it={it} ({elapsed:.0f}s): best={best_score:.8f}")
