For (1), the cost of optimization scales roughly as O(n) per iteration (due to FFT), so going from 2400 to 3600 should be about 1.5x slower per iteration. We can afford this if we allocate time properly.

For (2), the LP-based approach is the most precise. We should invest more time in it.

For (3), I'm not sure there's a better shape than what the optimizer finds, but we can try different random initializations.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))² using adaptive mirror descent,
cutting-plane LP Frank-Wolfe, and multi-scale coordinate descent.
"""

import numpy as np
import cvxpy as cp
import time

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.CBC, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def soft_subgrad(a, ac, top_k, beta):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 3000
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        bt = (t - 1.0) / (t + 1.5)
        y = x + bt * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = soft_subgrad(y, ac, cur_top_k, cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

            stall = 0
            pert = rng.normal(0, 0.015, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.5, 2.0)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def lp_fw_step(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Use cutting plane: start with active constraints only."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix sparsely
    # (a * b)[k] = sum_j a[|k-j|] * b[j] for valid j... 
    # Actually (a*b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j] * b[j]
    
    # For efficiency, only use constraints where the current a gives values near M
    ac_self = fast_autocorr(a)
    threshold = 0.5 * M
    active_k = np.where(ac_self >= threshold)[0]
    
    # But we need constraints on a*b, not a*a. Let's use all constraints
    # for safety, but limit to a subset for speed
    if len(active_k) > 200:
        # Sample 200 active constraints
        idx = np.random.default_rng(0).choice(len(active_k), 200, replace=False)
        active_k = active_k[idx]
    
    constraints = [cp.sum(b_var) == 1]  # normalize to make LP bounded
    
    # Actually, let's not normalize - we want the max sum
    # The LP is: max sum(b) s.t. T b <= M, b >= 0
    # This is unbounded if there's a direction where T b = 0 and sum(b) > 0
    # But since a >= 0 and a has positive entries, T is "positive" in some sense
    # Actually, if a has a zero, there might be directions with T b = 0
    # Let's add b <= 1 as a safety bound
    
    constraints = []
    for k in active_k:
        k_int = int(k)
        # (a*b)[k] = sum_j a[k-j]*b[j] for valid j
        js = max(0, k_int - n + 1)
        je = min(n, k_int + 1)
        # row k of T has a[k-j] at position j for j in [js, je)
        row = np.zeros(n)
        for j in range(js, je):
            row[j] = a[k_int - j]
        constraints.append(row @ b_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=_lp_solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def fw_optimize(a0, time_limit, verbose=False, max_iters=100):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_mom = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        direction = lp_fw_step(y, M_curr)
        if direction is None:
            break
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_mom = y_new.copy()
        else:
            bt = (t - 1.0) / (t + 2.0)
            x_mom = y_new + bt * (y_new - y)
            x_mom = np.maximum(x_mom, 0.0)
            sx = x_mom.sum()
            if sx < 1e-15:
                x_mom = y_new.copy()
            else:
                x_mom = x_mom / sx
        
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()
        
        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))² using:
1. Adaptive mirror descent with temperature scheduling and restarts

import numpy as np

import time

_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.CBC, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                print("  LP solver: " + str(s))
                return s
        except Exception:
            continue
    _lp_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _lp_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def soft_subgrad(a, ac, top_k, beta):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 4000
    cur_top_k = top_k
    cur_beta = float(beta)

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        bt = (t - 1.0) / (t + 1.5)
        y = x + bt * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = soft_subgrad(y, ac, cur_top_k, cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.02, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            cur_top_k = min(cur_top_k + 15, 120)
            cur_beta = max(cur_beta - 0.7, 2.0)

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    AMD t=%d (%.1fs): J=%.6f top_k=%d beta=%.1f" % (t, elapsed, J, cur_top_k, cur_beta))

    return best_a, best_m


def lp_direction(a, M, max_constraints=300):
    n = len(a)
    ac_self = fast_autocorr(a)
    # Select active constraints (where a*a is large)
    threshold = 0.4 * M
    active_k = np.where(ac_self >= threshold)[0]
    if len(active_k) > max_constraints:
        rng_local = np.random.default_rng(0)
        idx = rng_local.choice(len(active_k), max_constraints, replace=False)
        active_k = active_k[idx]
    # Always include the max
    k_max = np.argmax(ac_self)
    if k_max not in active_k:
        active_k = np.append(active_k, k_max)

    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_k:
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n, k_int + 1)
        row = np.zeros(n)
        for j in range(js, je):
            row[j] = a[k_int - j]
        constraints.append(row @ b_var <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=_lp_solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def fw_optimize(a0, time_limit, verbose=False, max_iters=150):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()
    failed = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        t_lp_start = time.monotonic()
        direction = lp_direction(y, M_curr, max_constraints=250)
        t_lp_end = time.monotonic()

        if direction is None:
            failed += 1
            if failed > 5:
                break
            continue
        failed = 0

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_mom = y_new.copy()
        else:
            bt = (t - 1.0) / (t + 2.0)
            x_mom = y_new + bt * (y_new - y)
            x_mom = np.maximum(x_mom, 0.0)
            sx = x_mom.sum()
            if sx < 1e-15:
                x_mom = y_new.copy()
            else:
                x_mom = x_mom / sx

        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()

        if verbose and t % 2 == 0:
            elapsed = time.monotonic() - t0
            lp_time = t_lp_end - t_lp_start
            J = 2.0 * n * best_m
            print("    FW iter=%d (%.1fs, lp=%.2fs): J=%.6f" % (t, elapsed, lp_time, J))

    return best_a, best_m


def coord_descent(a0, time_limit, verbose=False, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    iteration = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iteration += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(25, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Build in-peak mask
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:12]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            n_cand = min(300, len(candidates))
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_score = float('inf')

            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:12]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    sens += add
                rem_sens = 0.0
                for k2 in top_k[:12]:
                    k2i = int(k2)
                    d1 = k2i - src
                    d2 = src - k2i
                    rem = 0.0
                    if 0 <= d1 < n:
                        rem += a[d1]
                    if 0 <= d2 < n:
                        rem += a[d2]
                    rem_sens += rem

                net = sens - rem_sens
                if net < best_score:
                    best_score = net
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.005, 0.008, 0.012, 0.02, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                if a_t[src] < -1e-15:
                    continue
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                M_t = fast_autocorr(a_t).max()
                if M_t < best_m - 1e-15:
                    a = a_t
                    best_m = M_t
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                # Try smaller source (not just the max contributor)
                src2 = idx[np.argsort(contrib)[::-1][min(3, len(idx)-1)]]
                if src2 != src:
                    found2 = False
                    for frac2 in [0.003, 0.01, 0.03, 0.05]:
                        amt2 = a[src2] * frac2
                        if amt2 < 1e-18:
                            continue
                        a_t = a.copy()
                        a_t[src2] -= amt2
                        a_t[dst] += amt2
                        if a_t[src2] < -1e-15:
                            continue
                        a_t = np.maximum(a_t, 0.0)
                        s_t = a_t.sum()
                        if s_t < 1e-15:
                            continue
                        a_t = a_t / s_t
                        M_t = fast_autocorr(a_t).max()
                        if M_t < best_m - 1e-15:
                            a = a_t
                            best_m = M_t
                            best_a = a.copy()
                            improved = True
                            found2 = True
                            stall = 0
                            break
                    if found2:
                        found = True

        if not improved:
            stall += 1
            # Random perturbation to escape local optima
            n_pert = min(50, n // 10)
            for _p in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord iter=%d (%.1fs): J=%.6f stall=%d" % (iteration, elapsed, J, stall))

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print("    Coord done (%.1fs): J=%.6f" % (elapsed, J))
    return best_a, best_m

                best_score = sc
                best_seq = np.maximum(a_out, 0.0)
                best_seq = best_seq / best_seq.sum()
                print("  *** NEW BEST [fw_final]: %.6f (n=%d)" % (sc, len(best_seq)))

    # Phase 8: Last coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.70, 100.0)
        if t8 > 10:
            print("\n=== Phase 8: Last Coord n=%d ===" % len(best_seq))
            a_out, _ = coord_descent(best_seq, t8, verbose=True, seed=999)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0.0)
                best_seq = best_seq / best_seq.sum()
                print("  *** NEW BEST [coord_final]: %.6f (n=%d)" % (sc, len(best_seq)))

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("\nFINAL: score=%.6f, n=%d, time=%.0fs" % (best_score, len(best_seq), time.monotonic() - t_global))
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
