def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on threshold."""
    n = len(v)
    if n == 0:
        return v
    # Use the sorted approach
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_objective_and_grad(a, T):
    """
    Compute the smooth objective and gradient of T*log(sum(exp(c_j/T)))
    where c_j = conv(a,a)_j.
    Returns (smooth_obj, grad) where grad is w.r.t. a.
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    smooth_obj = T * (np.log(exp_c.sum()) + c_max / T)
    
    # grad_i = 2 * sum_j w_j * a_{j-i} = 2 * (w * a)_i (cross correlation)
    # Using FFT: cross_corr(w, a)[i] = sum_j w_j * a_{j-i}
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross correlation: sum_j w[j] * conj(a[j-i]) but since real: sum_j w[j] * a[j-i]
    # This is the same as convolution of flipped w with a
    # Actually: (w * a)[k] = sum_j w[j] * a[k-j]
    # We want: sum_j w[j] * a[j-i] = (a * w_flipped)[i] where w_flipped[j] = w[n-1-j]
    # Or equivalently: it's the cross-correlation
    # Let's use: ifft(conj(W) * A) gives cross-correlation
    cross = irfft(np.conj(W) * A, fft_size)[:n]
    grad = 2.0 * cross
    return smooth_obj, grad

def lp_best_response(a, M, solver, active_only=None):
    """Solve max sum(b) s.t. conv(a,b) <= M componentwise, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    
    if active_only is not None:
        T_use = T[active_only, :]
    else:
        T_use = T
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):

def coordinate_descent_multi(a0, time_limit, rng, verbose=True):
    """
    Multi-point coordinate descent: try moving mass from one position to multiple others simultaneously.
    Also includes single-pair moves with line search.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        return a, float('inf')

    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Strategy 1: Single pair moves with line search (nearby positions)
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-8:
                continue
            
            # Try moving to nearby positions
            for offset in rng.choice(range(1, min(100, n//2)+1), size=min(20, n//2), replace=False):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i or a[i] < norm * 1e-8:
                        continue
                    
                    # Try a few step sizes
                    for frac in [0.5, 0.7, 0.9]:
                        d = a[i] * frac
                        if d < 1e-15:
                            continue

                        a_test[j] += d

                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < best_score - 1e-12:
                            a = a_test.copy()
                            conv = conv_test
                            best_score = conv.max()
                            best_a = a.copy()
                            improved = True
                            break
                    # If improved, move to next i
                    if improved:
                        break
                
                if improved:
                    break

        if not improved:
            # Find the peak of the convolution and identify contributing terms
            peak_k = np.argmax(conv)
            # The peak conv[peak_k] = sum_i a[i]*a[peak_k-i]
            # Identify the largest contributing pair
            contributions = []
            for i in range(min(n, peak_k+1)):
                j = peak_k - i
                if j < n:
                    contributions.append((a[i] * a[j], i, j))
            contributions.sort(reverse=True)
            
            # Try moving mass from the largest contributor's source to a random other position
            if contributions:
                _, si, sj = contributions[0]
                if a[si] > norm * 1e-6:
                    # Move a fraction from si to a random position
                    for _ in range(5):
                        target = rng.integers(0, n)
                        if target == si or target == sj:

                            a_test[si] -= d
                            a_test[target] += d

                            a_test = a_test * norm / a_test.sum()
                            conv_test = fast_autocorr(a_test)
                            if conv_test.max() < best_score - 1e-12:
                                a = a_test.copy()
                                conv = conv_test
                                best_score = conv.max()
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
        
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
        
        if verbose and no_improve % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={no_improve} ({elapsed:.0f}s): best={best_score:.8f}")

def spectral_flatten(a0, time_limit, verbose=True):
    """
    Frequency-domain approach: identify frequency bins that contribute most to
    the time-domain peak and try to reduce them.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    fft_size = 1 << (2*n - 2).bit_length()
    
    for it in range(1000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute DFT of a (zero-padded)
        A = rfft(a, fft_size)
        power = np.abs(A)**2
        
        # The time-domain convolution is the inverse DFT of power
        # We want to find which frequency bins contribute to the peak
        peak_k = np.argmax(conv)
        
        # Contribution of frequency m to time bin k: power[m] * exp(2*pi*i*m*k/N) / N
        # The dominant contribution to the peak comes from frequencies where
        # the phase aligns. We want to reduce the power at those frequencies.
        
        # Simple heuristic: find the frequency with maximum power and try to
        # reduce it by adjusting a in the time domain
        m_max = np.argmax(power[1:]) + 1  # Skip DC
        
        # The time-domain effect of frequency m is a cosine wave: cos(2*pi*m*k/N)
        # To reduce the contribution at the peak, we want to decrease a where
        # the cosine is positive at the peak location
        m = m_max
        N = fft_size
        phase = 2.0 * np.pi * m * np.arange(n) / N
        cos_phase = np.cos(phase)
        
        # Move mass from high-cosine positions to low-cosine positions
        # (this reduces the contribution of this frequency to the signal)
        if cos_phase.std() > 1e-10:
            # Scale factor for the adjustment
            adj = 0.01 * cos_phase
            a_test = a - adj * a.mean() / max(a.std(), 1e-10)
            a_test = np.maximum(a_test, 0)
            s_test = a_test.sum()
            if s_test > 1e-15:
                a_test = a_test * norm / s_test
                conv_test = fast_autocorr(a_test)
                if conv_test.max() < best_score - 1e-12:
                    a = a_test.copy()
                    conv = conv_test
                    best_score = conv.max()
                    best_a = a.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Spectral iter={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss

    best_combined = None

                conv = fast_autocorr(combined)

def _fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _eval(a):
    try:

def _project(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smooth_obj_grad(a, T):
    """Compute smooth approximation of max(conv) and its gradient."""
    c = _fast_conv(a)
    c_max = c.max()
    shift = c - c_max
    max_shift = shift.max()
    exps = np.exp((shift - max_shift) / T)
    w = exps / exps.sum()
    smooth_val = T * (np.log(exps.sum()) + (c_max + max_shift) / T)
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(np.conj(W) * A, fft_size)[:n]
    grad = 2.0 * cross
    return smooth_val, grad


def _smooth_optimize(a0, time_limit, verbose=False):
    """accelerated smooth gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_score = _fast_conv(a).max()
    t0 = time.monotonic()

    # Adaptive temperature: start high, decrease as we get closer to the max
    T_current = 10.0
    T_min = 0.003
    it_total = 0

    while time.monotonic() - t0 < time_limit:
        c = _fast_conv(a)
        c_max = c.max()

        c_above = c[c > 0.5 * c_max]

            spread = np.std(c_above)
            T_target = max(T_min, spread * 0.3)
        else:
            T_target = T_min

        T = max(T_current, T_min)

        _, grad = _smooth_obj_grad(a, T)

        if g_norm < 1e-14:
            break

        # Adaptive step: larger when gradient is small

        a_new = _project(a_new, norm)

        # Nesterov momentum

        y = (1.0 - alpha) * a + alpha * a_new
        y = _project(y, norm)
        if it_total > 0:

        it_total += 1

            best_a = a.copy()

def _lp_br(a, M, solver, active=None):
    """LP best response: max sum(b) s.t. T@b <= M (active rows), b >= 0."""
    n = len(a)
    size = 2 * n - 1
    # Build constraint matrix rows for active indices
    if active is not None:
        rows = active
    else:
        rows = np.arange(size)
    m = len(rows)
    # T is m x n, T[k, i] = a[rows[k] - i] if 0 <= rows[k]-i < n
    T = np.zeros((m, n))
    for col_i in range(n):
        row_idx = rows - col_i
        valid = (row_idx >= 0) & (row_idx < n)
        T[valid, col_i] = a[row_idx[valid]]
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    bv = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if bv.sum() < 1e-15:
        return None
    return bv

def _spectral_refine(a0, time_limit, verbose=False):
    """Refine by reducing spectral concentration at dominant frequencies."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = _fast_conv(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Find dominant non-DC frequency
        if len(power) > 1:
            m_dominant = np.argmax(power[1:]) + 1
        else:
            break

        # Compute the time-domain "fingerprint" of this frequency
        # cos(2*pi*m*k/N) tells us where this frequency adds constructively
        m = m_dominant
        ks = np.arange(n)
        cos_ph = np.cos(2.0 * np.pi * m * ks / fft_size)
        sin_ph = np.sin(2.0 * np.pi * m * ks / fft_size)

        # Try to reduce the amplitude of this frequency by adjusting a
        # in the direction opposite to the frequency's pattern
        # a_new = a - epsilon * (a * cos_ph normalized)
        eps = 0.005
        adj = eps * a * cos_ph
        a_t = a - adj
        a_t = np.maximum(a_t, 0)
        st = a_t.sum()
        if st > 1e-10:
            a_t = a_t * norm / st
            cv_t = _fast_conv(a_t)
            if cv_t.max() < best_score - 1e-12:
                a = a_t.copy()
                conv = cv_t
                best_score = conv.max()
                best_a = a.copy()

        # Also try sine component
        adj2 = eps * a * sin_ph
        a_t2 = a - adj2
        a_t2 = np.maximum(a_t2, 0)
        st2 = a_t2.sum()
        if st2 > 1e-10:
            a_t2 = a_t2 * norm / st2

            if cv_t2.max() < best_score - 1e-12:

        if verbose and it % 500 == 0:

            print(f"  Spectral it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _interp(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    fn = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return fi * np.sqrt(2.0 * new_n) / ss

    if t5 > 10:
        a_n = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        out, _ = _spectral_refine(a_n, t5, verbose=True)
        try_update(out, f"spec_n{n_cur}")

    # === Phase 6: Diversified restarts (5%) ===
    rem = T_BUDGET - (time.monotonic() - t_start)
    t6 = rem * 0.05
    n_cur = len(best_seq)
    print(f"\n=== P6: Restarts n={n_cur} ({t6:.0f}s) ===")
    if t6 > 15:
        n_r = min(4, max(1, int(t6 / 12)))
        for r in range(n_r):
            sub_t = t6 / (n_r + 1)
            if time.monotonic() - t_start > T_BUDGET - 5:
                break
            # Structured perturbation: mix with a different shape
            if r == 0:
                # Add a triangular bump
                bump = np.abs(1 - 2 * np.linspace(0, 1, n_cur))
                a_p = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                a_p = 0.9 * a_p + 0.1 * bump * np.sqrt(2.0 * n_cur)
            elif r == 1:
                # Add a comb structure
                a_p = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum().copy()
                comb = np.zeros(n_cur)
                comb[::3] = 1.0
                a_p = 0.9 * a_p + 0.1 * comb * np.sqrt(2.0 * n_cur) / np.sqrt(comb.sum())
            else:
                # Random noise
                a_p = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                noise = rng.normal(0, 0.08 * a_p.std(), n_cur)
                a_p = a_p + noise
            a_p = np.maximum(a_p, 0)
            a_p = a_p * np.sqrt(2.0 * n_cur) / a_p.sum()
            out, _ = _smooth_optimize(a_p, sub_t, verbose=False)
            try_update(out, f"restart_{r}")

    # === Phase 7: Alternate lengths (remaining) ===
    rem = T_BUDGET - (time.monotonic() - t_start)
    if rem > 20 and best_seq is not None:
        n_base = len(best_seq)
        for df in [+200, -200, +400]:
            n_alt = n_base + df
            if n_alt < 400 or n_alt > 4000:
                continue
            rem_sub = T_BUDGET - (time.monotonic() - t_start)
            if rem_sub < 15:
                break
            t_a = min(rem_sub * 0.35, 80.0)
            print(f"\n=== Alt n={n_alt} ({t_a:.0f}s) ===")
            init_a = _interp(best_seq, n_alt)
            out, _ = _smooth_optimize(init_a, t_a * 0.6, verbose=False)
            try_update(out, f"alt_n{n_alt}")
            # Quick CD on the result
            rem_sub2 = T_BUDGET - (time.monotonic() - t_start)
            if rem_sub2 > 10 and out is not None:
                a_cd = np.maximum(out, 0) * np.sqrt(2.0 * n_alt) / np.maximum(out, 0).sum()
                out2, _ = _targeted_cd(a_cd, min(rem_sub2 * 0.3, 30.0), rng, verbose=False)
                try_update(out2, f"alt_cd_n{n_alt}")

    # === Final: Combine solutions ===
    if len(solutions) >= 2:
        print(f"\n=== Combining {len(solutions)} solutions ===")
        # Only combine same-length solutions
        by_len = {}
        for s in solutions:
            L = len(s)
            if L not in by_len:
                by_len[L] = []
            by_len[L].append(s)
        for L, sols in by_len.items():
            if len(sols) < 2:
                continue
            for i in range(len(sols)):
                for j in range(i + 1, len(sols)):
                    for alpha in [0.3, 0.4, 0.5, 0.6, 0.7]:
                        comb = alpha * sols[i] + (1 - alpha) * sols[j]
                        comb = np.maximum(comb, 0)
                        sc = comb.sum()
                        if sc > 1e-10:
                            comb = comb / sc
                            cv = _fast_conv(comb)
                            if cv.max() < best_score / (2 * L) * (eval_score_ratio(L)):
                                # Actually just evaluate directly
                                pass
                    # Just try the 50/50 mix
                    comb = 0.5 * sols[i] + 0.5 * sols[j]
                    comb = np.maximum(comb, 0)
                    sc = comb.sum()
                    if sc > 1e-10:
                        try_update(comb, f"comb_{i}_{j}")

    # Final check
    if best_seq is not None:
        fs = _eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]


def eval_score_ratio(n):
    """Helper: 2*n/(2*n-1) lower bound ratio."""
    return 2.0 * n / (2.0 * n - 1)
