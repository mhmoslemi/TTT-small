def diverse_init(n, rng, seed_offset=0):
    x = np.linspace(0.0, 1.0, n)
    choice = (int(rng.integers(0, 16)) + seed_offset) % 16
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.2 * np.cos(16 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        a = x ** 0.5 * (1 - x) ** 0.5
    elif choice == 11:
        a = np.exp(-3.0 * (x - 0.5) ** 2) * (1.0 + 0.3 * np.cos(12 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 12:
        a = 1.0 / (1.0 + 8.0 * x * (1.0 - x))
    elif choice == 13:
        a = np.power(np.sin(np.pi * x), 0.8 + 0.4 * float(rng.integers(0, 3)))
    elif choice == 14:
        freq = float(rng.integers(2, 8))
        a = 0.6 + 0.4 * np.cos(2 * np.pi * freq * x)
        a = np.maximum(a, 0.1)
    else:
        a = (1.0 - x) ** 0.3 + x ** 0.3
    a = np.maximum(a, 0.0)
    a = symmetrize(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    f_interp = symmetrize(f_interp)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_palindrome=True,
                         screening_schedule='adaptive', momentum_restart=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    if use_palindrome:
        a = symmetrize(a)
        a = normalize_seq(a, norm)

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    prev_b = None
    prev_duals = None
    no_improve_count = 0
    func_increases = 0
    prev_y_score = best_score

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening using dual info
        if prev_duals is not None and t > 10:
            dual_mag = np.abs(prev_duals)
            dual_threshold = np.sort(dual_mag)[::-1][min(100, size - 1)]
            active = np.where(dual_mag > max(dual_threshold * 0.1, 1e-10))[0]
            # Also include the top convolution lags
            conv_threshold = M * 0.6
            conv_active = np.where(conv_y > conv_threshold)[0]
            active = np.unique(np.concatenate([active, conv_active]))
            if len(active) < 50:
                active = np.where(conv_y > M * 0.3)[0]
        elif t < 30:
            active = np.where(conv_y > M * 0.4)[0]
            if len(active) < 30:
                active = None
        elif t < 100:
            active = np.where(conv_y > M * 0.65)[0]
            if len(active) < 30:
                active = np.where(conv_y > M * 0.4)[0]
        else:
            active = np.where(conv_y > M * 0.8)[0]
            if len(active) < 30:
                active = np.where(conv_y > M * 0.5)[0]

        b, duals = lp_solve(y, M, solver, active, warm_start=prev_b)
        if b is None or b.sum() < 1e-15:
            b, duals = lp_solve(y, M, solver, None, warm_start=prev_b)
        if b is None or b.sum() < 1e-15:
            continue

        # Verify screened solution
        if active is not None and len(active) < size:
            check = fast_conv_cross(y, b)
            if check.max() > M * 1.01:
                b_full, _ = lp_solve(y, M, solver, None, warm_start=b)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    duals = None

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        if use_palindrome:
            g = symmetrize(g)
            g = np.maximum(g, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g * norm / gs

        prev_b = b
        prev_duals = duals

        # Nesterov update with restart
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        if use_palindrome:
            y_new = symmetrize(y_new)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        sc_y_new = fast_conv_self(y_new).max()

        # Nesterov restart: if function value increased significantly, reset momentum
        if momentum_restart and sc_y_new > prev_y_score * 1.05:
            y_new = y.copy()  # Reset to previous
            func_increases += 1
            if func_increases % 10 == 0:
                # Perturb to escape
                n_perturb = max(5, n // 30)
                idx_perturb = np.random.default_rng(t).choice(n, size=n_perturb, replace=False)
                y_new[idx_perturb] *= (1.0 + 0.3 * np.random.default_rng(t + 7).standard_normal(n_perturb))
                y_new = np.maximum(y_new, 0.0)
                if use_palindrome:
                    y_new = symmetrize(y_new)
                    y_new = np.maximum(y_new, 0.0)
                s_new = y_new.sum()
                if s_new > 1e-15:
                    y_new = y_new * norm / s_new

        y = y_new
        prev_y_score = sc_y_new

        # Evaluate and track best
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        # Periodic perturbation
        if no_improve_count == 80:
            n_perturb = max(10, n // 25)
            idx_perturb = np.random.default_rng(t * 3 + 1).choice(n, size=n_perturb, replace=False)
            y[idx_perturb] *= (1.0 + 0.25 * np.random.default_rng(t * 3 + 2).standard_normal(n_perturb))
            y = np.maximum(y, 0.0)
            if use_palindrome:
                y = symmetrize(y)
                y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve_count = 0
            prev_b = None

        if no_improve_count == 200:
            # Stronger perturbation
            n_perturb = max(20, n // 15)
            idx_perturb = np.random.default_rng(t * 5 + 3).choice(n, size=n_perturb, replace=False)
            y[idx_perturb] *= (1.0 + 0.5 * np.random.default_rng(t * 5 + 4).standard_normal(n_perturb))
            y = np.maximum(y, 0.0)
            if use_palindrome:
                y = symmetrize(y)
                y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve_count = 0
            prev_b = None

        # Averaging
        if t % 200 == 0:
            avg_a = symmetrize((y + best_a) / 2.0) if use_palindrome else (y + best_a) / 2.0
            avg_a = np.maximum(avg_a, 0.0)
            s_avg = avg_a.sum()
            if s_avg > 1e-15:
                avg_a = avg_a * norm / s_avg
                sc_avg = fast_conv_self(avg_a).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True, use_palindrome=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    if use_palindrome:
        a = symmetrize(a)
        a = normalize_seq(a, norm)

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(50.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            if use_palindrome:
                grad = symmetrize(grad)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n) * (1.0 + 0.03 * it))

            # Line search
            for trial in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                if use_palindrome:
                    a_new = symmetrize(a_new)
                    a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            if use_palindrome:
                a_new = symmetrize(a_new)
                a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.10f}")

    return best_a, best_score

def subgradient_cd(a0, time_limit, rng, verbose=True, use_palindrome=True):
    """Coordinate descent using subgradient of max convolution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    if use_palindrome:
        a = symmetrize(a)
        a = normalize_seq(a, norm)

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        # Identify peak lags
        peak_threshold = cur_max * 0.95
        peak_lags = np.where(conv >= peak_threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(conv)])

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # For each peak lag, find which pair (i, j) contributes most
            best_j = -1
            best_new_max = cur_max
            best_delta = 0.0

            for kl in peak_lags[:5]:
                # The convolution at lag kl involves a_i * a_{kl-i} (if valid)
                j = kl - i
                if j < 0 or j >= n or j == i:
                    j2 = kl - i + n  # wrap
                    if 0 <= j2 < n and j2 != i:
                        j = j2
                    else:
                        continue

                aj = a[j]
                # Moving delta from i to j changes conv[2i] and conv[2j] and cross terms
                # Simplified: try a few delta values
                n_samples = 12
                deltas = np.linspace(0.05, 0.5, n_samples) * ai
                for d in deltas:
                    # Approximate new max
                    new_ai = ai - d
                    new_aj = aj + d
                    # Update affected convolution values
                    delta_conv = np.zeros(size)
                    # Contribution change at positions involving i
                    for pos in range(max(0, i - n + 1), min(size, i + n)):
                        # old contribution: a[i] * a[pos - i] (if valid)
                        old_j = pos - i
                        if 0 <= old_j < n and old_j != i:
                            delta_conv[pos] += d * a[old_j]
                        elif old_j == i:
                            delta_conv[pos] += d * d  # (ai-d)^2 - ai^2 = -2*ai*d + d^2
                    for pos in range(max(0, j - n + 1), min(size, j + n)):
                        old_j = pos - j
                        if 0 <= old_j < n and old_j != j:
                            delta_conv[pos] -= d * a[old_j]
                        elif old_j == j:
                            delta_conv[pos] -= d * d
                    # Self terms
                    if 2 * i < size:
                        delta_conv[2 * i] = -2 * ai * d + d * d
                    if 2 * j < size:
                        delta_conv[2 * j] = 2 * aj * d + d * d

                    new_conv = conv + delta_conv
                    new_max = new_conv.max()
                    if new_max < best_new_max - 1e-12:
                        best_new_max = new_max
                        best_j = j
                        best_delta = d

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                if use_palindrome:
                    a[n - 1 - i] -= best_delta
                    a[n - 1 - best_j] += best_delta
                a = np.maximum(a, 0.0)
                if use_palindrome:
                    a = symmetrize(a)
                s_new = a.sum()
                if s_new < 1e-15:
                    a = best_a.copy()
                    continue
                a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                    if use_palindrome:
                        a[n - 1 - i] += best_delta
                        a[n - 1 - best_j] -= best_delta
                    a = np.maximum(a, 0.0)
                    if use_palindrome:
                        a = symmetrize(a)
                    a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve == 5:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                if use_palindrome:
                    a = symmetrize(a)
                    a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and (idx // n) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def dual_refine(a0, time_limit, solver, verbose=True, use_palindrome=True):
    """Use LP dual variables for gradient-guided refinement."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    if use_palindrome:
        a = symmetrize(a)
        a = normalize_seq(a, norm)

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(100):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break

        active = np.where(conv > M * 0.7)[0]
        if len(active) < 20:
            active = np.where(conv > M * 0.4)[0]
        b, duals = lp_solve(a, M, solver, active)

        if duals is None or b is None:
            b2, duals2 = lp_solve(a, M, solver, None)
            if duals2 is not None:
                duals = duals2
                b = b2
            else:
                duals = np.zeros(size)
                duals[np.argmax(conv)] = 1.0

        A = rfft(a, fft_size)
        D = rfft(duals, fft_size)
        g1 = irfft(np.conj(A) * D, fft_size)[:n]
        g2 = irfft(A * D, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        if use_palindrome:
            grad = symmetrize(grad)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Line search
        step = 0.1
        improved_this_iter = False
        for trial in range(15):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            if use_palindrome:
                a_new = symmetrize(a_new)
                a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < M - 1e-10:
                a = a_new
                improved_this_iter = True
                break
            step *= 0.5

        if improved_this_iter:
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualRefine it={it} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Small-n multi-start to find good shape
    if best_seq is None:
        n_small = 200
        print(f"\n=== Phase 1: Small-n ({n_small}) multi-start ===")
        best_small = None
        best_small_score = float('inf')
        t_phase1 = min(60.0, time_budget * 0.04)
        n_starts = 8
        per_start_time = t_phase1 / n_starts
        for si in range(n_starts):
            if time.monotonic() - t_global > time_budget - 120:
                break
            init = diverse_init(n_small, rng, seed_offset=si)
            a_out, sc_out = nesterov_lp_optimize(init, per_start_time, solver, verbose=False,
                                                   use_palindrome=True)
            sc = fast_conv_self(a_out).max()
            if sc < best_small_score:
                best_small_score = sc
                best_small = a_out.copy()
        if best_small is not None:
            # Interpolate to base length
            n_base = 1600
            best_seq = interpolate_seq(best_small, n_base)
            best_score = eval_fn(best_seq)
            print(f"  Small-n best: {best_small_score:.8f}, interpolated to n={n_base}: {best_score:.8f}")
    else:
        n_base = len(best_seq)
        print(f"  Starting from parent: n={n_base}, score={best_score:.8f}")

    # Phase 2: LSE gradient descent for quick improvement
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 80.0)
    if t2 > 10:
        print(f"\n=== Phase 2: LSE n={len(best_seq)} (budget={t2:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, t2, verbose=False, use_palindrome=True)
        try_update(a_out, "lse_init")

    # Phase 3: Main LP Nesterov optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 3: LP Nesterov n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True,
                                         use_palindrome=True, momentum_restart=True)
        try_update(a_out, f"lp_main_n{n3}")

    # Phase 4: Dual refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Dual Refine n={n4} (budget={t4:.0f}s) ===")
        a_out, _ = dual_refine(best_seq, t4, solver, verbose=True, use_palindrome=True)
        try_update(a_out, f"dual_n{n4}")

    # Phase 5: LP Nesterov round 2 with different settings
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.22, 350.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP Nesterov R2 n={n5} (budget={t5:.0f}s) ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True,
                                         use_palindrome=True, momentum_restart=True)
        try_update(a_out, f"lp_r2_n{n5}")

    # Phase 6: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    t6 = min(remaining * 0.18, 250.0)
    if t6 > 15:
        print(f"\n=== Phase 6: CD Polish n={n6} (budget={t6:.0f}s) ===")
        a_out, _ = subgradient_cd(best_seq, t6, rng, verbose=True, use_palindrome=True)
        try_update(a_out, f"cd_n{n6}")

    # Phase 7: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_ext = len(best_seq) + 400
        print(f"\n=== Phase 7: Length extension to n={n_ext} (remaining={remaining:.0f}s) ===")
        init_ext = interpolate_seq(best_seq, n_ext)
        t_ext = min(remaining * 0.4, 150.0)
        a_out, _ = nesterov_lp_optimize(init_ext, t_ext, solver, verbose=False,
                                         use_palindrome=True)
        try_update(a_out, f"lp_ext_n{n_ext}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_ext:
            t_ext2 = min(remaining * 0.4, 80.0)
            a_out, _ = subgradient_cd(best_seq, t_ext2, rng, verbose=False, use_palindrome=True)
            try_update(a_out, f"cd_ext_n{n_ext}")

    # Phase 8: Final quick polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    if remaining > 50:
        print(f"\n=== Phase 8: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
        t_f1 = min(remaining * 0.35, 50.0)
        a_out, _ = lse_gradient_descent(best_seq, t_f1, verbose=False, use_palindrome=True)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_f2 = min(remaining * 0.4, 50.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t_f2, solver, verbose=False,
                                             use_palindrome=True)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_f3 = min(remaining * 0.5, 30.0)
            a_out, _ = subgradient_cd(best_seq, t_f3, rng, verbose=False, use_palindrome=True)
            try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
