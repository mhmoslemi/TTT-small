from scipy.fft import rfft, irfft

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def solve_lp_small(a, M, selected_lags, solver):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in selected_lags, b >= 0"""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraints: (a*b)_k = sum_i a_{k-i} * b_i for valid i
    # For lag k: (a*b)_k = sum_{i} a[max(0,k-n+1):min(n,k+1)] * b[...]
    # More precisely: (a*b)_k = sum_{i=0}^{n-1} a_{k-i} * b_i where a_j = 0 if j < 0 or j >= n
    
    constraints = []
    for k in selected_lags:
        # (a*b)_k = sum_{i=0}^{n-1} a[k-i] * b_i
        # a[k-i] is nonzero when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # b indices: lo to hi-1
        # a indices: k-lo to k-hi+1 (reversed)
        a_seg = a[k - hi + 1:k - lo + 1][::-1]
        constraints.append(a_seg @ b_var[lo:hi] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def solve_lp_full(a, M, solver):
    """Solve LP with all constraints (slower but exact)"""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M, b_var >= 0])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def normalize(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * target_sum / s

def lp_best_response(n, time_limit, solver, K=40, verbose=True, use_full_lp_every=50):
    """
    Main LP best-response optimization with:
    - Adaptive constraint screening (top-K lags)
    - Heavy-ball momentum with adaptive decay
    - Periodic full LP for accuracy
    - Peak deflation
    """
    norm = np.sqrt(2.0 * n)
    best_a = None
    best_score = float('inf')
    t0 = time.monotonic()
    
    # Initialize
    a = np.ones(n) / n * norm
    
    y = a.copy()  # momentum variable
    velocity = np.zeros(n)
    
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv_self(a)
        M = conv.max()
        
        if M < best_score - 1e-10:
            best_score = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Select top-K constraints
        if it == 1 or it % use_full_lp_every == 0:
            # Use all constraints for accuracy
            b = solve_lp_full(a, M, solver)
        else:
            # Use top-K constraints for speed
            K_actual = min(K, 2 * n - 1)
            sorted_idx = np.argsort(conv)[::-1]
            selected = sorted_idx[:K_actual]
            b = solve_lp_small(a, M, selected, solver)
            if b is None:
                b = solve_lp_full(a, M, solver)
        
        if b is None or b.sum() < 1e-15:
            if no_improve > 100:
                # Perturb and continue
                a *= (1 + 0.1 * np.random.standard_normal(n))
                a = np.maximum(a, 0)
                a = normalize(a, norm)
            continue
        
        g = normalize(b, norm)
        
        # Heavy-ball momentum update
        # Adaptive momentum: start low, increase over time
        mu = min(0.95, 0.3 + 0.6 * (it / (it + 200)))
        alpha = 2.0 / (it + 2.0)  # Nesterov step size
        
        # Fessler-Birge style update
        y_new = (1 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0)
        y_new = normalize(y_new, norm)
        
        # Extrapolation
        a_new = y_new + mu * (y_new - y)
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Peak deflation every 100 iterations
        if it % 100 == 0 and it > 200:
            conv2 = fast_conv_self(a)
            peak_lag = np.argmax(conv2)
            # Gradient of c_{peak_lag} w.r.t. a
            n_local = len(a)
            grad_deflate = np.zeros(n_local)
            for i in range(n_local):
                # c_k = sum_j a_{j+k} * a_j
                # dc_k/da_i = a_{i-k} + a_{i+k} (with boundary)
                j1 = i - peak_lag
                j2 = i + peak_lag
                if 0 <= j1 < n_local:
                    grad_deflate[i] += a[j1]
                if 0 <= j2 < n_local:
                    grad_deflate[i] += a[j2]
            # Step against the gradient
            step = 0.05
            a_deflated = a - step * grad_deflate / (np.linalg.norm(grad_deflate) + 1e-10)
            a_deflated = np.maximum(a_deflated, 0)
            a_deflated = normalize(a_deflated, norm)
            sc_deflated = fast_conv_self(a_deflated).max()
            if sc_deflated < fast_conv_self(a).max():
                a = a_deflated
                y = a.copy()
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it} ({elapsed:.0f}s): M={M:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

                a = a * norm / s
                v = fast_conv_self(a).max()
                if v < best_v:
                    best_v = v
                    best_d = d
                a[i] = ai
                a[j] = aj

                a[i] -= best_d
                a[j] += best_d
                a = np.maximum(a, 0)
                a = normalize(a, norm)
                new_max = fast_conv_self(a).max()

                    best_score = new_max
                    best_a = a.copy()

        else:

def lse_optimize(a0, time_limit, verbose=True):
    """Log-sum-exp gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for T_val in [50.0, 10.0, 2.0, 0.5, 0.1, 0.05]:
        if time.monotonic() - t0 > time_limit:
            break
        n_steps = 100
        step_size = 0.05
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = g1 + g2
            
            a_new = a - step_size * grad
            a_new = np.maximum(a_new, 0)
            a_new = normalize(a_new, norm)
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE T={T_val}: best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 8)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    else:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x) + 0.3 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP best-response with adaptive constraint screening (top-K lags)
2. Nesterov acceleration with extrapolation and adaptive momentum
3. Peak deflation via subgradient at maximum autocorrelation lag
4. LSE (log-sum-exp) smoothing gradient descent
5. Multi-length search, interpolation, and coordinate descent polish

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Using solver (fallback): ECOS")
    return cp.ECOS


def build_T_sub(a, selected_lags):
    """Build sub-Toeplitz matrix for selected lag positions."""
    n = len(a)
    K = len(selected_lags)
    T = np.zeros((K, n))
    for idx, k in enumerate(selected_lags):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            T[idx, i] = a[k - i]
    return T


def solve_lp_sub(a, M, selected_lags, solver):
    """Solve LP: max sum(b) s.t. T_sub @ b <= M, b >= 0."""
    n = len(a)
    T = build_T_sub(a, selected_lags)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def solve_lp_full(a, M, solver):
    """Solve LP with all constraints."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_nesterov_optimize(a0, time_limit, solver, K=30, verbose=True,
                         full_lp_every=80, use_extrapolation=True):
    """
    LP best-response with Nesterov acceleration.
    Uses top-K constraint screening for speed, periodic full LP for accuracy.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)

    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()

    y = a.copy()
    prev_y = a.copy()
    no_improve = 0
    it = 0
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Select constraints
        if it == 1 or it % full_lp_every == 0:
            b = solve_lp_full(y, M, solver)
            if b is None:
                sorted_idx = np.argsort(conv_y)[::-1]
                selected = sorted_idx[:min(K, size)]
                b = solve_lp_sub(y, M, selected, solver)
        else:
            sorted_idx = np.argsort(conv_y)[::-1]
            # Use top K lags, but at least those > 0.7*M
            K_actual = min(K, size)
            selected = sorted_idx[:K_actual]
            b = solve_lp_sub(y, M, selected, solver)
            if b is None:
                b = solve_lp_full(y, M, solver)

        if b is None or b.sum() < 1e-15:
            if no_improve > 200:
                y *= (1 + 0.05 * np.random.default_rng(it).standard_normal(n))
                y = np.maximum(y, 0)
                y = normalize_to(y, norm)
            no_improve += 1
            continue

        g = normalize_to(b, norm)

        # Nesterov update
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to(y_new, norm)

        if use_extrapolation and it > 1:
            mu = (it - 1.0) / (it + 2.0)
            a_new = y_new + mu * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, norm)
        else:
            a_new = y_new.copy()

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Evaluate
        M_y = conv_y.max()
        M_a = fast_conv_self(a).max()
        cur_best = min(M_y, M_a)

        if cur_best < best_M - 1e-12:
            if M_y < M_a:
                best_a = y.copy()
            else:
                best_a = a.copy()
            best_M = cur_best
            no_improve = 0
        else:
            no_improve += 1

        # Perturbation to escape stagnation
        if no_improve == 150:
            n_pert = max(10, n // 30)
            rng_p = np.random.default_rng(it * 7 + 3)
            idx_p = rng_p.choice(n, size=n_p, replace=False)
            y[idx_p] *= (1.0 + 0.15 * rng_p.standard_normal(n_p))
            y = np.maximum(y, 0.0)
            y = normalize_to(y, norm)
            prev_y = y.copy()
            no_improve = 0
        elif no_improve == 300:
            # Stronger perturbation
            n_pert = max(20, n // 15)
            rng_p = np.random.default_rng(it * 13 + 7)
            idx_p = rng_p.choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.3 * rng_p.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            y = normalize_to(y, norm)
            prev_y = y.copy()
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    LP-Neo it={it} ({elapsed:.0f}s): M={M:.8f}, best={best_M:.8f}")

    return best_a, best_M


def lse_gradient_descent(a0, time_limit, verbose=True):
    """Log-sum-exp smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)
    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temperatures = [50.0, 20.0, 5.0, 1.0, 0.3, 0.1, 0.05, 0.02]

    for T_val in temperatures:
        if time.monotonic() - t0 > time_limit:
            break
        n_steps = 150
        step_size = 1.0 / (T_val * np.sqrt(n))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = g1 + g2

            # Backtracking line search
            step = step_size
            for trial in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = normalize_to(a_new, norm)
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = normalize_to(a_new, norm)

            sc = fast_conv_self(a).max()
            if sc < best_M:
                best_M = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE T={T_val:.2f}: best={best_M:.8f}")

    return best_a, best_M


def peak_deflation(a0, time_limit, verbose=True):
    """Targeted deflation of peak autocorrelation lags."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)
    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_conv_self(a)
        M = conv.max()
        # Identify all lags near the peak
        threshold = M * 0.95
        peak_lags = np.where(conv >= threshold)[0]
        if len(peak_lags) == 0:
            break

        # Compute subgradient: average of gradients at peak lags
        grad = np.zeros(n)
        for k in peak_lags:
            for i in range(n):
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    grad[i] += a[j1]
                if 0 <= j2 < n:
                    grad[i] += a[j2]
        grad /= len(peak_lags)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Step against gradient
        step = 0.1
        for trial in range(15):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = normalize_to(a_new, norm)
            sc = fast_conv_self(a_new).max()
            if sc < M:
                break
            step *= 0.5

        a_new = a - step * grad / grad_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = normalize_to(a_new, norm)
            sc = fast_conv_self(a).max()
            if sc < best_M:
                best_M = sc
                best_a = a.copy()

        if verbose and it % 50 == 0:
            print(f"    Deflate it={it}: M={best_M:.8f}")

    return best_a, best_M


def coordinate_descent_polish(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with structured candidate selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)
    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()
    no_improve = 0
    total_improv = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_M = conv.max()
        size = 2 * n - 1
        pass_improved = False

        # Get peak lags for structured candidates
        peak_threshold = cur_M * 0.85
        peak_lags = np.where(conv >= peak_threshold)[0]

        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Generate candidate j's
            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:10]:
                j_candidates.add((kl - i) % n)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_M
            best_j = -1

            for j in j_candidates:
                aj = a[j]
                # Compute change vector for the convolution
                delta = np.zeros(size)
                for i2 in range(n):
                    aii = a[max(0, i2)] if i2 < n else 0
                # Efficient: delta[k] = a[k-j]+a[k+j] - a[k-i]-a[k+i] (approx for cross terms)
                # Simpler: just try a few transfer amounts
                for frac in [0.1, 0.2, 0.4, 0.6, 0.8]:
                    d = frac * ai
                    a[i] = ai - d
                    a[j] = aj + d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] = ai
                        a[j] = aj
                        continue
                    a = normalize_to(a, norm)
                    v = fast_conv_self(a).max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j
                    a[i] = ai
                    a[j] = aj
                    a = normalize_to(a, norm)

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0)
                a = normalize_to(a, norm)
                new_M = fast_conv_self(a).max()
                if new_M < best_M - 1e-13:
                    best_M = new_M
                    best_a = a.copy()
                    pass_improved = True
                    total_improv += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 20:
                break
            if no_improve == 10:
                n_pert = max(5, n // 40)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0)
                a = normalize_to(a, norm)
        else:
            no_improve = 0

        if verbose and total_improv % 200 == 0 and total_improv > 0:
            print(f"    CD improv={total_improv}: M={best_M:.8f}")

    return best_a, best_M


def structured_init(n, rng):
    x = np.linspace(0.0, 1.0, n)

        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7

    elif choice == 7:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(10):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick LSE warm-start
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    print(f"\n=== Phase 1: LSE warm-start (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=False)
        try_update(a_out, "lse_init")

    # Phase 2: Main LP Nesterov with K=20 (fast)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 2: LP Nesterov K=20 n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, K=20, verbose=True,
                                        full_lp_every=60, use_extrapolation=True)
        try_update(a_out, f"lp_k20_n{n2}")

    # Phase 3: Peak deflation
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 3: Peak Deflation (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = peak_deflation(best_seq, t3, verbose=True)
        try_update(a_out, "deflate")

    # Phase 4: LP Nesterov K=50 (more accurate)
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: LP Nesterov K=50 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, K=50, verbose=True,
                                        full_lp_every=100, use_extrapolation=True)
        try_update(a_out, f"lp_k50_n{n4}")

    # Phase 5: LSE refinement at low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: LSE refine (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t5, verbose=True)
        try_update(a_out, "lse_refine")

    # Phase 6: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    t6 = min(remaining * 0.12, 120.0)
    print(f"\n=== Phase 6: Coord Descent n={n6} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = coordinate_descent_polish(best_seq, t6, rng, verbose=True)
        try_update(a_out, f"cd_n{n6}")

    # Phase 7: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 7: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 150:
        init_long = interpolate_seq(best_seq, n_long)
        t7 = min(remaining * 0.35, 250.0)
        a_out, _ = lp_nesterov_optimize(init_long, t7, solver, K=30, verbose=False,
                                        full_lp_every=80, use_extrapolation=True)
        try_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7b = min(remaining * 0.3, 80.0)
            a_out2, _ = peak_deflation(best_seq, t7b, verbose=False)
            try_update(a_out2, f"deflate_long_{n_long}")

    # Phase 8: Final LP polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n8 = len(best_seq)
    t8 = min(remaining * 0.5, 120.0)
    print(f"\n=== Phase 8: Final LP polish n={n8} (remaining={remaining:.0f}s) ===")
    if t8 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t8, solver, K=40, verbose=True,
                                        full_lp_every=50, use_extrapolation=True)
        try_update(a_out, "lp_final")

    # Phase 9: Final deflation
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.5, 60.0)
    if t9 > 10:
        a_out, _ = peak_deflation(best_seq, t9, verbose=False)
        try_update(a_out, "deflate_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
