a_new = (1-α) * a + α * b (additive, Nesterov style)

a_new = a^{1-α} * b^α (multiplicative)

a_new[i] = a[i] * (b[i] + ε) / (a[i] + ε) ^ α * normalization

        nv_alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - nv_alpha) * y + nv_alpha * b_star
        y_new = normalize_a(y_new, norm)
        momentum = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        x_new = y_new + momentum * (y_new - y)
        x_new = normalize_a(x_new, norm)

        # Pick the better of multiplicative and Nesterov
        obj_mult = compute_obj(a_new)
        obj_nest = compute_obj(x_new)

        if obj_mult <= obj_nest:
            y = a_new
            a = a_new
            cur_obj = obj_mult
        else:
            y = x_new
            a = x_new
            cur_obj = obj_nest

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Perturbation to escape local optima
        if no_improve == 80:
            rng_p = np.random.default_rng(t * 7919)
            n_pert = max(5, n // 25)
            idx_pert = rng_p.choice(n, size=n_pert, replace=False)
            y[idx_pert] *= (1.0 + 0.15 * rng_p.standard_normal(n_pert))
            y = normalize_a(y, norm)
            no_improve = 0

        total_iters += 1
        if verbose and total_iters % 200 == 0:
            print(f"    MirrorLP t={t} ({elapsed:.0f}s): obj={best_obj:.8f}, iters={total_iters}")

    return best_a, best_obj

def lse_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_obj = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(80.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(200, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n) * (1.0 + 0.03 * it))
                for trial in range(12):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= max_c + 0.1 * T_val:
                        break
                    step *= 0.5

            a = a - step * grad
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = (a + 1e-10)
                s_new = a.sum()
            a = a * norm / s_new

            sc = fast_conv_self(a).max()
            if sc < best_obj:
                best_obj = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best_peak={best_obj:.8f}")

    return best_a, best_obj

def local_search(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_obj = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0

    # Precompute peak lags for structured candidate selection
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Get peak lags for structured candidates
        peak_lags = np.where(conv >= 0.92 * cur_max)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(perm[idx])
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:10]:
                j1 = int(kl - i)
                j2 = int(kl + i)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            # Also try symmetric partner
            j_candidates.add(n - 1 - i)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector for transfer from i to j
                # Delta conv = +delta * g where g[k] = 2*a[k-j] - 2*a[k-i] (approx, ignoring delta^2)
                g = np.zeros(size)
                # +delta at position j: adds delta * 2*a shifted
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += 2.0 * a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += 2.0 * a[j - np.arange(lo2, hi2)]
                # -delta at position i: subtracts
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= 2.0 * a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= 2.0 * a[i - np.arange(lo4, hi4)]

                # Find optimal delta via sampling
                n_samples = 15
                deltas = np.linspace(0.005, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    # Correct diagonal terms
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = int(j)

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize_a(a, norm)
                new_max = fast_conv_self(a).max()
                if new_max < best_obj - 1e-12:
                    best_obj = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_imp += 1
                else:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = normalize_a(a, norm)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_pert = max(8, n // 30)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = normalize_a(a, norm)
        else:
            no_improve = 0

        if verbose and total_imp % 100 == 0 and total_imp > 0:
            print(f"    LocalSearch imp={total_imp} ({time.monotonic()-t0:.0f}s): obj={best_obj:.8f}")

    return best_a, best_obj

def make_init(n, rng):
    choice = int(rng.integers(0, 12))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        # Edge-biased (important: asymmetric!)
        a = np.exp(-3.0 * x) + 0.3 * np.exp(-3.0 * (1.0 - x))
    elif choice == 6:
        # Strongly asymmetric ramp
        a = np.exp(-2.0 * x) + 0.1
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        # Asymmetric with oscillation
        a = np.exp(-1.5 * x) * (1.0 + 0.4 * np.cos(15 * np.pi * x))
        a = np.maximum(a, 0.02)
    elif choice == 10:
        # Bi-modal asymmetric
        a = 0.6 * np.exp(-8.0 * (x - 0.2) ** 2) + 0.4 * np.exp(-8.0 * (x - 0.8) ** 2)
    else:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Starting. Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_parent = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            n_parent = len(best_seq)
            print(f"Parent loaded: n={n_parent}, score={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # --- Phase 0: Quick LSE to improve parent ---
    remaining = time_budget - (time.monotonic() - t_start)
    if best_seq is None:
        n0 = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(6):
            cand = make_init(n0, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n=1600, score={best_score:.6f}")

    if remaining > 30:
        t0 = min(remaining * 0.06, 90.0)
        print(f"\n=== Phase 0: Quick LSE (budget={t0:.0f}s) n={len(best_seq)} ===")
        a_out, _ = lse_descent(best_seq, t0, verbose=False)
        try_update(a_out, "lse_init")

    # --- Phase 1: Mirror LP descent (main workhorse) ---
    remaining = time_budget - (time.monotonic() - t_start)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: Mirror LP n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = mirror_lp_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"mirror_lp_n{n1}")

    # --- Phase 2: LSE descent ---
    remaining = time_budget - (time.monotonic() - t_start)
    n2 = len(best_seq)
    t2 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 2: LSE n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = lse_descent(best_seq, t2, verbose=False)
        try_update(a_out, f"lse_n{n2}")

    # --- Phase 3: Mirror LP round 2 ---
    remaining = time_budget - (time.monotonic() - t_start)
    n3 = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Mirror LP round 2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = mirror_lp_descent(best_seq, t3, verbose=True)
        try_update(a_out, f"mirror_lp2_n{n3}")

    # --- Phase 4: Local search polish ---
    remaining = time_budget - (time.monotonic() - t_start)
    n4 = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: Local Search n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = local_search(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"local_n{n4}")

    # --- Phase 5: Another mirror LP + local search ---
    remaining = time_budget - (time.monotonic() - t_start)
    n5 = len(best_seq)
    if remaining > 60:
        t5a = min(remaining * 0.4, 150.0)
        t5b = min(remaining * 0.3, 100.0)
        print(f"\n=== Phase 5: Mirror+Local n={n5} (budget={t5a+t5b:.0f}s) ===")
        if t5a > 20:
            a_out, _ = mirror_lp_descent(best_seq, t5a, verbose=False)
            try_update(a_out, "mirror3")
            remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 30:
            t5b = min(remaining * 0.5, 100.0)
            a_out2, _ = local_search(best_seq, t5b, rng, verbose=False)
            try_update(a_out2, "local2")

    # --- Phase 6: Try different length ---
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 100 and n_parent > 0:
        n_alt = n_parent + 200
        print(f"\n=== Phase 6: Length extension n={n_alt} (remaining={remaining:.0f}s) ===")
        init_alt = interpolate_seq(best_seq, n_alt)
        t6 = min(remaining * 0.4, 200.0)
        if t6 > 30:
            a_out, _ = mirror_lp_descent(init_alt, t6, verbose=False)
            try_update(a_out, f"mirror_long_{n_alt}")
            remaining = time_budget - (time.monotonic() - t_start)
            if remaining > 30 and len(best_seq) == n_alt:
                t6b = min(remaining * 0.4, 80.0)
                a_out2, _ = local_search(best_seq, t6b, rng, verbose=False)
                try_update(a_out2, f"local_long_{n_alt}")

    # --- Phase 7: Final polish with local search ---
    remaining = time_budget - (time.monotonic() - t_start)
    nf = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={nf} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        tf = min(remaining * 0.7, 120.0)
        a_out, _ = local_search(best_seq, tf, rng, verbose=False)
        try_update(a_out, "final_local")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
