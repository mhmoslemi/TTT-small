            a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new

    if verbose:
        print(f"  Local search done: best={best_score:.8f}")
    return best_a, best_score

def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = int(rng.integers(1, 6))
        f2 = int(rng.integers(6, 20))
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-5.0 * x) + 0.4
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = int(rng.integers(3, 10))
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    else:
        a = np.sinc(x - 0.5) ** 2 + 0.1

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.005 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a.copy()
    return a_sym / s

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LSE-Adam (quick initial improvement) - 12%
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 200.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} ({t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_update(a_out, f"lse1_n{n_base}")

    # Phase 2: LP Nesterov (main workhorse) - 45%
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} ({t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        check_update(a_out, f"lp1_n{len(best_seq)}")

    # Phase 3: Vegas pressure - 10%
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 3: Vegas n={len(best_seq)} ({t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = vegas_optimize(best_seq, t3, verbose=True)
        check_update(a_out, f"vegas_n{len(best_seq)}")

    # Phase 4: LP Nesterov round 2 from improved point - 15%
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: LP round 2 n={len(best_seq)} ({t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, verbose=True)
        check_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 5: Local search polish - 10%
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Local search n={len(best_seq)} ({t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = local_search_polish(best_seq, t5, rng, verbose=True)
        check_update(a_out, f"local_n{len(best_seq)}")

    # Phase 6: Multi-n exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 6: Multi-n exploration (remaining={remaining:.0f}s) ===")
        lengths_to_try = [n_base - 200, n_base + 200, n_base - 400, n_base + 400, n_base + 600]
        for n_try in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            if n_try < 400 or n_try > 4000:
                continue
            t_try = min(remaining * 0.4, 80.0)
            print(f"  Trying n={n_try} ({t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            if t_try > 20:
                a_out, _ = lse_adam_optimize(init_try, min(t_try * 0.5, 30.0), verbose=False)
                check_update(a_out, f"lse_ml_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out, _ = lp_nesterov_optimize(a_out, min(remaining * 0.4, 40.0), solver, verbose=False)
                check_update(a_out, f"lp_ml_n{n_try}")

    # Phase 7: Final LP polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7 = min(remaining * 0.6, 120.0)
        print(f"\n=== Phase 7: Final LP polish ({t7:.0f}s) ===")
        a_out, _ = lp_nesterov_optimize(best_seq, t7, solver, verbose=False)
        check_update(a_out, f"lp_final_n{len(best_seq)}")

    # Try symmetrization
    a_sym = symmetrize(best_seq)
    check_update(a_sym, f"sym_n{len(best_seq)}")

    # Final: fresh random starts if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 8: Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            n_fresh = max(800, n_base + int(rng.integers(-400, 400)))
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.4, 25.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_update(a_out, f"fresh{attempt}_n{n_fresh}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = lp_nesterov_optimize(a_out, min(remaining * 0.4, 20.0), solver, verbose=False)
                check_update(a_out, f"fresh{attempt}_lp_n{n_fresh}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
