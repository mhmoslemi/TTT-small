   from scipy.optimize import minimize
   import time

   def fast_autocorr(a):
       n = len(a)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       A = rfft(np.asarray(a, dtype=np.float64), fft_size)
       return irfft(A * A, fft_size)[:size]

       """Returns (objective, gradient) of log-sum-exp smooth approximation."""
       n = len(a)
       c = fast_autocorr(a)
       c_max = c.max()
       exp_c = np.exp((c - c_max) / T)
       w = exp_c / exp_c.sum()
       obj = c_max / T + np.log(exp_c.sum())  # log(sum exp(c/T))
       # Gradient: df/da[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       W = rfft(w, fft_size)
       A = rfft(a, fft_size)
       cross = irfft(W * np.conj(A), fft_size)[:n]
       grad = 2.0 * cross
       return obj, grad

               exp_c = np.exp((c - c_max) / T)

               size = 2 * n - 1

               W = rfft(w, fft_size)
               A = rfft(zn, fft_size)
               cross = irfft(W * np.conj(A), fft_size)[:n]
               # Chain rule: d/dz of f(z/sum(z))
               # = (1/sum(z)) * grad_f(zn) - (grad_f(zn) . zn / sum(z)^2) * sum(z)
               # Simplifies to: (1/sz) * (grad_f(zn) - mean(grad_f(zn)) * sz / n) ...

               a = np.maximum(res.x, 0.0)
               s = a.sum()
               if s > 1e-15:
                   a = a / s
               score = fast_autocorr(a).max()
               if score < best_score:
                   best_score = score
                   best_a = a.copy()
           except Exception as e:

                   print(f"  L-BFGS-B error at T={T}: {e}")

           if verbose:
               elapsed = time.monotonic() - t0
               print(f"    L-BFGS T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

       return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with FFT-based smooth log-sum-exp gradients and adaptive temperature annealing
2. Multiplicative weight (mirror descent) with momentum and adaptive step size
3. Diverse multi-start: uniform, random Dirichlet, structured (sin/cos/exp), parent-based
4. Subgradient descent with averaging for non-convex problem
5. Pair coordinate descent with peak-targeting
6. Length interpolation and exploration
Key: L-BFGS-B handles the non-convex landscape better than projected gradient;
multiplicative weights maintain non-negativity naturally; diverse starts + annealing
navigate the complex energy landscape.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_cross(w, p):
    """Compute cross-correlation: result[i] = sum_k w[k]*p[k-i] for i=0..n-1."""
    n = len(p)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    P = rfft(p, fft_size)
    return irfft(W * np.conj(P), fft_size)[:n]


def smooth_objective(z, T):
    """Smooth log-sum-exp objective with implicit normalization."""
    n = len(z)
    z = np.maximum(z, 0.0)
    sz = z.sum()
    if sz < 1e-15:
        return 1e10
    p = z / sz
    c = fast_autocorr(p)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    return c_max / T + np.log(exp_c.sum())


def smooth_gradient(z, T):
    """Gradient of smooth log-sum-exp objective w.r.t. z (before normalization)."""
    n = len(z)
    z = np.maximum(z, 0.0)
    sz = z.sum()
    if sz < 1e-15:
        return np.zeros(n)
    p = z / sz
    c = fast_autocorr(p)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    cross = compute_cross(w, p)
    gg = 2.0 * cross
    # Chain rule through normalization p = z/sz
    return (gg - p * gg.sum()) / sz


def evaluate_safe(a):
    """Safely evaluate the sequence."""

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0.0)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lgbfgs_anneal(a0, temps, max_iter_per_T, time_limit, t_start, verbose=True):
    """L-BFGS-B with temperature annealing schedule."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    bounds = [(0.0, None)] * n

    for Ti, T in enumerate(temps):
        if time.monotonic() - t_start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 3:
            break

        # Wrap with args for scipy
        try:
            res = minimize(
                lambda z: smooth_objective(z, T),
                a,
                jac=lambda z: smooth_gradient(z, T),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': max_iter_per_T, 'gtol': 1e-5,
                         'ftol': 1e-12, 'iprint': 0, 'maxfun': max_iter_per_T * 3}
            )
            a_new = np.maximum(res.x, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                score = fast_autocorr(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

        if verbose:
            elapsed = time.monotonic() - t_start
            print(f"    L-BFGS T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multiplicative_descent(a0, time_limit, t_start, verbose=True):
    """Multiplicative weight (mirror descent) with momentum and adaptive step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    momentum = np.zeros(n)
    beta = 0.9

    # Temperature schedule
    temps = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]

    for T in temps:
        if time.monotonic() - t_start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 2:
            break
        n_steps = max(50, min(500, int(remaining * 2)))
        lr = 1.0 / (np.sqrt(n) + 1.0)

        for it in range(n_steps):
            if time.monotonic() - t_start > time_limit:
                break

            c = fast_autocorr(a)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()
            cross = compute_cross(w, a)
            grad = 2.0 * cross

            # Momentum update
            momentum = beta * momentum + (1.0 - beta) * grad
            lr_adapt = lr / (1.0 + 0.1 * it)

            # Multiplicative update
            a_new = a * np.exp(-lr_adapt * momentum)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            a = a_new
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def subgradient_average(a0, time_limit, t_start, verbose=True):
    """Subgradient descent with running average (works for non-convex too heuristically)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    avg_a = np.zeros(n)
    count = 0
    lr_base = 1.0 / np.sqrt(n)
    temps = [3.0, 2.0, 1.0, 0.5, 0.3, 0.1, 0.05, 0.02]

    for T in temps:
        if time.monotonic() - t_start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 2)))

        for it in range(n_steps):
            if time.monotonic() - t_start > time_limit:
                break

            c = fast_autocorr(a)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()
            cross = compute_cross(w, a)
            grad = 2.0 * cross

            lr = lr_base / (1.0 + 0.05 * it)
            a_new = a - lr * grad
            a_new = project_simplex(a_new, 1.0)
            a = a_new

            avg_a += a
            count += 1

            if count % 10 == 0:
                avg_p = avg_a / count
                avg_p = project_simplex(avg_p, 1.0)
                score_avg = fast_autocorr(avg_p).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_p.copy()

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def pair_cd_refine(a0, time_limit, t_start, rng, window=60, verbose=True):
    """Pair coordinate descent: move mass between pairs to reduce peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    no_improve = 0

    while time.monotonic() - t_start < time_limit:
        if no_improve > 5:
            break
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t_start > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-12:
                continue

            # Find best partner to move mass to
            best_delta = 0.0
            best_j = -1
            best_val = conv.max()

            for offset in range(1, min(window, n) + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    # Try moving 20% of a[i] to j
                    d = a[i] * 0.2
                    a_t = a.copy()
                    a_t[i] -= d
                    a_t[j] += d
                    a_t = a_t / a_t.sum()
                    ct = fast_autocorr(a_t)
                    if ct.max() < best_val - 1e-12:
                        best_val = ct.max()
                        best_delta = d
                        best_j = j

            if best_j >= 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_autocorr(a)
                if conv.max() < best_score:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True

        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (no_improve > 0):
            elapsed = time.monotonic() - t_start
            print(f"    PairCD ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score


def single_cd_refine(a0, time_limit, t_start, verbose=True):
    """Single coordinate descent: adjust one element at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    no_improve = 0

    while time.monotonic() - t_start < time_limit and no_improve < 3:
        pass_improved = False

        for i in range(n):
            if time.monotonic() - t_start > time_limit:
                break
            if a[i] < 1e-12:
                continue
            # Try reducing a[i] by various amounts and spreading to neighbors
            for frac in [0.1, 0.3, 0.5]:
                d = a[i] * frac
                a_t = a.copy()
                a_t[i] -= d
                # Spread to neighbors
                neighbors = [j for j in [(i-1)%n, (i+1)%n, (i-2)%n, (i+2)%n] if j != i]
                if not neighbors:
                    continue
                a_t[neighbors] += d / len(neighbors)
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                ct = fast_autocorr(a_t)
                if ct.max() < best_score - 1e-12:
                    a = a_t
                    conv = ct
                    best_score = ct.max()
                    best_a = a.copy()
                    pass_improved = True
                    break

        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n, preserving integral."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_structured_starts(n, rng):
    """Create diverse structured initializations."""
    starts = []
    # Uniform
    starts.append(np.ones(n) / n)
    # Sine
    x = np.linspace(0, 1, n, endpoint=False)
    f = np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * x), 0.01)
    starts.append(f / f.sum())
    # Cosine
    f = np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * x), 0.01)
    starts.append(f / f.sum())
    # U-shape (high at ends)
    f = np.maximum(1.0 + 2.0 * (2 * x - 1)**2, 0.01)
    starts.append(f / f.sum())
    # Inverted U (high in middle)
    f = np.maximum(2.0 - 1.5 * (2 * x - 1)**2, 0.01)
    starts.append(f / f.sum())
    # Exponential
    f = np.exp(-3.0 * x)
    starts.append(f / f.sum())
    # Double exponential
    f = np.exp(-3.0 * x) + np.exp(-3.0 * (1 - x))
    starts.append(f / f.sum())
    # Random Dirichlet
    f = rng.dirichlet(np.ones(n) * 0.5)
    starts.append(f)
    # Another random
    f = rng.dirichlet(np.ones(n) * 2.0)
    starts.append(f)
    # Sparse random
    f = rng.random(n)
    f[rng.random(n) < 0.5] = 0
    if f.sum() < 1e-10:
        f = np.ones(n) / n
    starts.append(f / f.sum())
    # High frequency oscillation
    f = np.maximum(1.0 + 0.8 * np.sin(20 * np.pi * x), 0.01)
    starts.append(f / f.sum())
    return starts

        t6_start = time.monotonic()
        a_out, _ = pair_cd_refine(best_seq, t6, t6_start, rng, window=100, verbose=True)
        try_update(a_out, "final_pcd")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 7: Length Exploration (budget={remaining:.0f}s) ===")
        n_current = len(best_seq)
        alt_lengths = [n_current + 200, n_current - 200, n_current + 500, n_current - 500,
                       n_current + 1000, 2000, 800]
        alt_lengths = [nl for nl in alt_lengths if 200 <= nl <= 5000 and nl != n_current]
        # Deduplicate
        seen = set()
        unique_lengths = []
        for nl in alt_lengths:
            if nl not in seen:
                seen.add(nl)
                unique_lengths.append(nl)

        for nl in unique_lengths[:4]:
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 30:
                break
            print(f"  Trying n={nl}...")
            init_alt = interpolate_sequence(best_seq, nl)
            t_alt = min(remaining_sub * 0.3, 120.0)
            t_alt_start = time.monotonic()
            a_out, _ = lgbfgs_anneal(init_alt, temps_short, 200, t_alt_start + t_alt, t_alt_start, verbose=False)
            try_update(a_out, f"alt_n{nl}")

            # Quick pair CD
            remaining_sub2 = time_budget - (time.monotonic() - t_global)
            if remaining_sub2 > 20:
                a_out2, _ = pair_cd_refine(a_out, min(remaining_sub2 * 0.2, 30.0),
                                            time.monotonic(), rng, window=50, verbose=False)
                try_update(a_out2, f"alt_pcd_n{nl}")

    # === Final: one more L-BFGS-B polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final Polish (budget={remaining:.0f}s) ===")
        t_final = remaining * 0.8
        t_final_start = time.monotonic()
        temps_final = [1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
        a_out, _ = lgbfgs_anneal(best_seq, temps_final, 500, t_final_start + t_final, t_final_start, verbose=True)
        try_update(a_out, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
