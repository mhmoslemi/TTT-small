    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A_f = rfft(a, fft_size)
    B_f = rfft(b, fft_size)
    conv_ab = irfft(A_f * B_f, fft_size)[:size]

    all_k = np.arange(size)
    comp = np.setdiff1d(all_k, np.asarray(active_set))
    if len(comp) > 0:
        viol = conv_ab[comp] > M * (1.0 + tol_expand)
        if viol.any():
            new_active = np.union1d(np.asarray(active_set), comp[viol])
            if len(new_active) > 0.9 * size:
                new_active = all_k
            b, duals = solve_lp(a, M, new_active, solver)
            if b is None:
                return None, active_set, None
            active_set = new_active

    # Line search: minimize score of normalized blend
    s_a = a.sum()
    s_b = b.sum()
    if s_b < 1e-15:
        return None, active_set, None
    b_scaled = b * (s_a / s_b)  # Same sum as a

    # Golden section search on t in [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    def eval_blend(t):
        trial = (1.0 - t) * a + t * b_scaled
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            return float('inf')
        return conv_max(trial) / (s * s)

    f_c = eval_blend(c)
    f_d = eval_blend(d)

    for _ in range(25):
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - gr * (hi - lo)
            f_c = eval_blend(c)
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + gr * (hi - lo)
            f_d = eval_blend(d)

    t_opt = (lo + hi) / 2.0
    new_a = (1.0 - t_opt) * a + t_opt * b_scaled
    new_a = np.maximum(new_a, 0.0)
    s_new = new_a.sum()
    if s_new < 1e-15:
        return None, active_set, None
    new_a = new_a / s_new

    return new_a, active_set, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, seed=42):
    """
    Main LP best-response optimizer with line search.
    a0 should have sum 1.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    active_set = select_active_set(fast_conv(a), fast_conv(a).max(), n, frac=0.1, min_count=3)
    warm_b = None
    stall = 0
    iter_count = 0
    frac_schedule = [0.1, 0.05, 0.03, 0.01, 0.005]
    frac_idx = 0

    while time.monotonic() - t0 < time_limit:
        # Adapt active set every 30 iterations
        if iter_count % 30 == 0:
            conv_a = fast_conv(a)
            M = conv_a.max()
            frac = frac_schedule[min(frac_idx, len(frac_schedule) - 1)]
            # Tighten over time
            effective_frac = max(frac * np.exp(-iter_count / 500.0), 0.001)
            active_set = select_active_set(conv_a, M, n, frac=effective_frac, min_count=3)
            if iter_count % 100 == 0 and iter_count > 0 and frac_idx < len(frac_schedule) - 1:
                frac_idx += 1

        new_a, active_set, duals = lp_iter_with_ls(a, active_set, solver, n, warm_b=warm_b)
        if new_a is None:
            # Restart active set
            conv_a = fast_conv(a)
            active_set = select_active_set(conv_a, conv_a.max(), n, frac=0.15, min_count=5)
            warm_b = None
            continue

        a = new_a
        warm_b = None  # a changed, so warm start is less useful
        iter_count += 1

        cur_score = score_fast(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iter_count} ({elapsed:.0f}s): best={best_score:.10f}, stall={stall}")

        if stall > 60:
            # Restart
            a = best_a.copy()
            active_set = select_active_set(fast_conv(a), fast_conv(a).max(), n, frac=0.2, min_count=5)
            warm_b = None
            stall = 0
            if verbose:
                print(f"    Restart at iter {iter_count}, score={best_score:.10f}")

    return best_a, best_score


def spectral_perturb(a, rng, n_peaks=5):
    """
    Perturb in spectral domain to reduce autocorrelation peaks.
    Modifies the phase of the spectrum to spread out energy.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    # Get autocorrelation
    conv = fast_conv(a)
    M = conv.max()
    top_lags = np.argsort(conv)[-n_peaks:]

    # For each top lag, try to reduce its contribution
    # R(k) = (1/N) sum_f |A(f)|^2 * exp(2*pi*i*f*k/N)
    # We can modulate |A(f)| to reduce concentration at specific lags
    mag = np.abs(A)
    phase = np.angle(A)

    # Add small random phase perturbation
    phase += rng.standard_normal(len(phase)) * 0.01

    # Slightly flatten the magnitude spectrum
    alpha = 0.05
    mag_target = mag ** (1 - alpha) * (mag.mean() ** alpha)
    mag = (1 - alpha) * mag + alpha * mag_target

    A_new = mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_descent_fast(a0, time_limit, verbose=False):
    """
    Fast coordinate descent with incremental autocorrelation update.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv = fast_conv(a)
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved_any = False

        # Process coordinates in random order
        order = np.arange(n)
        np.random.shuffle(order)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-15:
                # Try setting it to a small positive value
                candidates = [aj, 1e-4, 1e-3]
            else:
                candidates = [aj * 0.5, aj * 0.75, aj * 1.25, aj * 1.5, 0.0]

            best_x = aj
            best_local = conv_max(a)

            # Precompute the "column" contributions
            # Changing a[j] to x changes conv[k] by:
            #   (x - aj) * (a[k-j] + a[j-k]) + (x^2 - aj^2) * [k == 2j]
            # For k where 0 <= k-j < n: contribution from a[k-j]
            # For k where 0 <= j-k < n: contribution from a[j-k]

            for x in candidates:
                if x < 0 or abs(x - aj) < 1e-12:
                    continue
                # Compute new conv at affected lags
                # Only lags k where |k - j| < n are affected
                k_lo = max(0, j - n + 1)
                k_hi = min(size - 1, j + n - 1)
                delta = x - aj
                new_conv_max = conv_max(a)  # current max (unchanged lags)
                # Check if the max is in the affected range
                # For simplicity, just compute the affected lags
                affected_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    val = conv[k]
                    i1 = k - j
                    i2 = j - k
                    contrib = 0.0
                    if 0 <= i1 < n:
                        contrib += a[i1] if i1 != j else aj
                    if 0 <= i2 < n:
                        contrib += a[i2] if i2 != j else aj
                    # New value: conv[k] - delta*(contrib_at_j) + ... 
                    # Actually: new_conv[k] = conv[k] - aj*(a[k==j?1:0] stuff) + x*(stuff)
                    # This is getting complicated. Let me just do it directly.
                    pass
                # Too complex for incremental, just evaluate directly for few candidates
                trial = a.copy()
                trial[j] = x
                st = trial.sum()
                if st < 1e-15:
                    continue
                trial = trial / st
                trial_max = conv_max(trial)
                if trial_max < best_local - 1e-13:
                    best_local = trial_max
                    best_x = x
                    a = trial
                    conv = fast_conv(a)

            if best_x != aj:
                improved_any = True
                cur_score = score_fast(a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()

        if not improved_any:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """
    Subgradient method: minimize max_k (a*a)[k] on the simplex.
    Uses multiple active peaks with exponential weighting.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    step = 2.0 / n
    patience = 0
    max_patience = 300
    rng = np.random.default_rng(123)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_conv(a)
        M = conv.max()

        # Weighted combination of top peak gradients
        n_peaks = max(3, min(8, n // 200))
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M - conv[top_idx]) / max(M * 0.05, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for j in range(n):
                v = 0.0
                i1 = k - j
                if 0 <= i1 < n:
                    v += a[i1]
                i2 = j - k
                if 0 <= i2 < n:
                    v += a[i2]
                grad[j] += w * v

        # Subgradient step with projection onto simplex
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = score_fast(a_trial)
        if sc < best_score - 1e-13:
            a = a_trial
            best_score = sc
            best_a = a.copy()
            patience = 0
            step = min(step * 1.01, 5.0 / n)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10 / n:
                break

        # Random lag exploration every 20 iters
        if t % 20 == 0:
            threshold = M * 0.9
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 1:
                k_r = candidates[rng.integers(len(candidates))]
                grad_r = np.zeros(n)
                for j in range(n):
                    v = 0.0
                    i1 = k_r - j
                    if 0 <= i1 < n:
                        v += a[i1]
                    i2 = j - k_r
                    if 0 <= i2 < n:
                        v += a[i2]
                    grad_r[j] = v
                a_tr2 = a - 0.5 * step * grad_r
                a_tr2 = np.maximum(a_tr2, 0.0)
                st2 = a_tr2.sum()
                if st2 > 1e-15:
                    a_tr2 = a_tr2 / st2
                    sc2 = score_fast(a_tr2)
                    if sc2 < best_score - 1e-13:
                        a = a_tr2
                        best_score = sc2
                        best_a = a.copy()
                        patience = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def make_initialization(n, method='parent'):
    """Create an initial sequence of length n."""
    if method == 'parent':
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            p = np.clip(p, 0, 1000)
            if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
                return p / p.sum()
        except Exception:
            pass
    # Cosine window
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def propose_candidate():
    """
    Main search function.
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_n = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_n = len(p)
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={parent_n}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = make_initialization(n_base, 'cosine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Main LP optimization (longest phase) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 750.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP best-response n={n_base}, budget={t1:.0f}s ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: LP with different seed (different active set trajectory) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 40:
        print(f"\n=== Phase 2: LP round 2 n={n_cur}, budget={t2:.0f}s ===")
        # Perturb slightly before restarting
        a_pert = best_seq * (1 + 0.01 * rng.standard_normal(n_cur))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_out, _ = nesterov_lp_optimize(a_pert, t2, solver, verbose=False, seed=777)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 3: Spectral perturbation + LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Spectral perturbation n={n_cur}, budget={t3:.0f}s ===")
        a_spec = spectral_perturb(best_seq, rng, n_peaks=5)
        sc_spec = evaluate_safe(a_spec)
        if sc_spec < best_score:
            best_score = sc_spec
            best_seq = a_spec.copy()
            print(f"  Spectral init: {sc_spec:.8f}")
        a_out, _ = nesterov_lp_optimize(a_spec, t3, solver, verbose=False, seed=314)
        try_update(a_out, f"spec_n{n_cur}")

    # === Phase 4: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Subgradient n={n_cur}, budget={t4:.0f}s ===")
        a_out, _ = subgradient_polish(best_seq, t4, verbose=True)
        try_update(a_out, f"subg_n{n_cur}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 5: Multi-length, remaining={remaining:.0f}s ===")
        n_cur = len(best_seq)
        length_trials = [n_cur - 100, n_cur + 100, n_cur + 200]
        length_trials = [l for l in length_trials if l >= 100]
        t_per = min(60.0, (remaining - 30) / max(len(length_trials), 1))
        for nl in length_trials:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t5 = min(t_per, remaining - 20)
            if t5 < 15:
                break
            a_interp = interpolate_seq(best_seq, nl)
            a_out, _ = nesterov_lp_optimize(a_interp, t5, solver, verbose=False, seed=nl)
            try_update(a_out, f"len{nl}")

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.3, 150.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Final LP n={n_cur}, budget={t6:.0f}s ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t6, solver, verbose=True, seed=999)
        try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 7: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        t7 = min(remaining * 0.4, 60.0)
        print(f"\n=== Phase 7: Final subgrad n={n_cur} ===")
        a_out, _ = subgradient_polish(best_seq, t7, verbose=False)
        try_update(a_out, f"subg_final_n{n_cur}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
