            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Sliding window ergodic average
        window_buf -= window_buf[window_ptr % window_size] if window_filled >= window_size else 0
        # Simplified: just accumulate
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}, active={len(active_set)}")

    return best_a, best_score


def nesterov_lp_v2(a0, time_limit, solver, verbose=True, seed=0):
    """Improved Nesterov LP with peak-reduction interleaving and better restarts."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0

    phase = 0  # 0: normal, 1: peak reduction, 2: restart

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if active_set is None or t % 25 == 0:
            if phase == 0:
                tol = 0.025 + 0.015 * np.exp(-t / 800.0)
            else:
                tol = 0.015
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 12:
                active_set = np.where(conv_y >= M_curr * 0.75)[0]
            if len(active_set) > 0.8 * size:
                active_set = np.where(conv_y >= M_curr * 0.93)[0]

        if phase == 0:
            # Standard Nesterov LP
            b_max = lp_max_sum(y, M_curr, solver, active_set)
            b_min = None
            if t % 3 == 0:
                b_min, _ = lp_min_max(y, solver, active_set, normalize_sum=norm)

            if b_max is None:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
                if b_max is None:
                    continue

            b_sum = b_max.sum()
            g = b_max * norm / b_sum
            if b_min is not None and b_min.sum() > 1e-10:
                g_min_norm = b_min * norm / b_min.sum()
                g = 0.7 * g + 0.3 * g_min_norm

        elif phase == 1:
            # Peak reduction: use only top 15 peaks
            sorted_idx = np.argsort(conv_y)[::-1]
            top_peaks = sorted_idx[:15]
            b_peak = peak_reduction_lp(y, 15, solver, M_target=M_curr)
            if b_peak is not None:
                b_sum_p = b_peak.sum()
                g = b_peak * norm / b_sum_p
            else:
                b_max = lp_max_sum(y, M_curr, solver, active_set)
                if b_max is None:
                    continue
                g = b_max * norm / b_max.sum()
            phase = 0  # Alternate back

        else:
            # Restart phase
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            avg_a = np.zeros(n)
            n_avg = 0
            phase = 0

        # Check constraint violations
        if len(active_set) < size and t % 15 == 0:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                T_check = build_active_constraints(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 5e-5))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        # Nesterov update
        alpha_t = max(2.5 / (t + 2.5), 0.04)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.94)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 25 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Phase management
        if t % 50 == 0 and phase == 0:
            phase = 1  # Switch to peak reduction every 50 steps
        if stall_count > 100:
            phase = 2  # Restart

        if verbose and t % 400 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP2 t={t} ({elapsed:.0f}s): best={best_score:.10f}, phase={phase}, stall={stall_count}")

    return best_a, best_score


def coord_descent(a0, time_limit, rng, verbose=True):
    """Coordinate descent with golden section on individual coordinates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    phi = 0.6180339887498949

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-14 * norm:
                continue

            lo = 0.0
            hi = ai * 2.5 + norm * 0.05
            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            a_t1 = a.copy()
            a_t1[i] = c1
            s_t1 = a_t1.sum()
            if s_t1 < 1e-15:
                continue
            a_t1 = a_t1 * norm / s_t1
            f1 = fast_autocorr(a_t1).max()

            a_t2 = a.copy()
            a_t2[i] = c2
            s_t2 = a_t2.sum()
            if s_t2 < 1e-15:
                continue
            a_t2 = a_t2 * norm / s_t2
            f2 = fast_autocorr(a_t2).max()

            for _ in range(25):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a_t1 = a.copy()
                    a_t1[i] = c1
                    s_t1 = a_t1.sum()
                    if s_t1 < 1e-15:
                        break
                    a_t1 = a_t1 * norm / s_t1
                    f1 = fast_autocorr(a_t1).max()
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a_t2 = a.copy()
                    a_t2[i] = c2
                    s_t2 = a_t2.sum()
                    if s_t2 < 1e-15:
                        break
                    a_t2 = a_t2 * norm / s_t2
                    f2 = fast_autocorr(a_t2).max()

            opt_val = min(f1, f2)
            if opt_val < best_score - 1e-13:
                a = a_t1 if f1 < f2 else a_t2
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                    best_score = fast_autocorr(a).max()
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_swap_cd(a0, time_limit, rng, verbose=True):
    """Pair swap coordinate descent for finer adjustments."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    fracs = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx % n]
            j = perm[(idx + 1) % n]
            if i == j:
                continue
            ai = a[i]
            aj = a[j]
            if ai < 1e-14 * norm and aj < 1e-14 * norm:
                continue

            best_val = fast_autocorr(a).max()
            best_a_try = None

            for frac in fracs:
                d = ai * frac
                if d <= 0 or d > ai * 0.99:
                    continue
                a_try = a.copy()
                a_try[i] = ai - d
                a_try[j] = aj + d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                val = fast_autocorr(a_try).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_a_try = a_try.copy()

                d2 = aj * frac
                if d2 <= 0 or d2 > aj * 0.99:
                    continue
                a_try2 = a.copy()
                a_try2[i] = ai + d2
                a_try2[j] = aj - d2
                s_try2 = a_try2.sum()
                if s_try2 < 1e-15:
                    continue
                a_try2 = a_try2 * norm / s_try2
                val2 = fast_autocorr(a_try2).max()
                if val2 < best_val - 1e-14:
                    best_val = val2
                    best_a_try = a_try2.copy()

            if best_a_try is not None and best_val < best_score - 1e-14:
                a = best_a_try
                best_score = best_val
                best_a = a.copy()
                pass_improved = True
                no_improve = 0

        if not pass_improve:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.04):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif style == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(5 * np.pi * x) + 0.1 * np.cos(10 * np.pi * x)
        a = np.maximum(a, 0.0)
    else:
        a = np.sin(np.pi * x) ** 0.5 * (1.0 + 0.12 * np.cos(9 * np.pi * x))
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_v2(a_norm, t8, solver, verbose=True, seed=999)
            try_update(a_out, f"nlp_final_n{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: Final CD polish ===
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 9: Final CD n={n_best} ===")
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent(a_norm, t9, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 10: Final pair swap ===
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 10: Final PCD n={n_best} ===")
        t10 = min(remaining * 0.5, 80.0)
        if t10 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_swap_cd(a_norm, t10, rng, verbose=False)
            try_update(a_out, f"pcd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
