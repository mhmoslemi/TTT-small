def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_best_response(a, M, solver, active_set):
    """Solve max sum(b) s.t. conv(a,b)[j] <= M for j in active_set, b >= 0."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    k = len(active_set)
    
    # Build constraint matrix for active constraints
    # (a*b)[j] = sum_i a[i] * b[j-i] for valid i, j-i
    # For a given j, the coefficient of b[i] is a[j-i] if 0 <= j-i < n, else 0
    T = np.zeros((k, n))
    for idx, j in enumerate(active_set):
        # For b[i], coefficient is a[j-i] if 0 <= j-i < n
        i_min = max(0, j - (n - 1))
        i_max = min(n - 1, j)
        if i_max >= i_min:
            T[idx, i_min:i_max+1] = a[j - i_max : j - i_min : -1]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive constraint generation."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    t = 0
    stuck = 0
    
    # Adaptive constraint threshold
    base_threshold = 0.90
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-10:
            break
        
        # Adaptive number of constraints
        progress = min(1.0, t / 500.0)
        threshold = base_threshold - 0.35 * progress
        active = np.where(conv_y >= threshold * M_curr)[0]
        if len(active) < 3:
            active = np.argsort(conv_y)[-20:]
        
        b = lp_best_response(y, M_curr, solver, active)
        if b is None:
            # Try with all constraints
            active = np.arange(2 * n - 1)
            b = lp_best_response(y, M_curr, solver, active)
            if b is None:
                stuck += 1
                if stuck > 50:
                    break
                continue
        
        stuck = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * S / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * S / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * S / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_score:.8f}, n_cons={len(active)}")
    
    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with adaptive window and line search."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * S / s
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        # Focus on positions that contribute to the peak
        peak_k = np.argmax(conv)
        # Positions that contribute to conv[peak_k]: a[i]*a[peak_k-i] for i in [0,n)
        # So positions i and peak_k-i are "peak contributors"
        peak_contributors = set()
        for i in range(n):
            if 0 <= peak_k - i < n:
                peak_contributors.add(i)
                peak_contributors.add(peak_k - i)
        
        # Prioritize peak contributors
        contributor_list = [i for i in perm if i in peak_contributors]
        other_list = [i for i in perm if i not in peak_contributors]
        ordered = contributor_list + other_list
        
        for i in ordered:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < S * 1e-10:
                continue
            
            # Try moving to nearby positions
            window = min(20, n // 4)
            for d in range(1, window + 1):
                for j in [(i + d) % n, (i - d) % n]:
                    if j == i:
                        continue
                    # Try a few amounts
                    for frac in [0.5, 0.3, 0.1]:
                        delta = a[i] * frac
                        a_test = a.copy()
                        a_test[i] -= delta
                        a_test[j] += delta
                        s_test = a_test.sum()
                        a_test = a_test * S / s_test
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < best_score - 1e-10:
                            a = a_test.copy()
                            conv = conv_test
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            break
                    if no_improve == 0 and pass_improved:
                        # Continue with same i, try next offset
                        continue
            # (continue to next i)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def make_chirp(n, rng):
    """Create a chirp-like initialization."""
    t = np.arange(n) / n
    freq = 2.0 * t + 0.5  # Linear chirp
    phase = 2 * np.pi * np.cumsum(freq) - np.pi
    a = np.abs(np.cos(phase)) + 0.1
    # Add some randomness
    a = a * (1.0 + 0.3 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

def get_solver():
    import cvxpy as cp
    for s in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS
