```python
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except:
        return float('inf')

def mirror_descent(a0, time_limit, n_peaks=5, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 1.0 / n  # initial step size
    patience = 0
    max_patience = 200
    stall_count = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Get top peaks
        top_k = min(n_peaks, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]
        # Weight by proximity to max
        weights = np.exp(-3.0 * (M - conv[peak_indices]) / max(M * 0.05, 1e-10))
        weights = weights / weights.sum()
        
        # Combined subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            # (a*a)[k] = sum_j a[j]*a[k-j]
            # d/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            # But when j = k-j, we only count once
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                i1 = k - j
                if 0 <= i1 < n:
                    grad[j] += w * a[i1]
                if j != i1:
                    i2 = j - k
                    if 0 <= i2 < n:
                        grad[j] += w * a[i2]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new * norm / s_new
        
        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-12:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            stall_count = 0
            eta = min(eta * 1.01, 5.0 / n)
        else:
            patience += 1
            eta *= 0.97
            stall_count += 1
            if stall_count > 50:
                # Restart from best
                a = best_a.copy()
                eta = 1.0 / n
                stall_count = 0
                patience = 0
        
        if verbose and t % 1000 == 0:
            print(f"  MD t={t}: score={best_score:.10f}, eta={eta:.6f}")
    
    return best_a, best_score
```

1. The term where j = x: a[x] * a[k-x] (if 0 <= k-x < n)
2. The term where k-j = x, i.e., j = k-x: a[k-x] * a[x] (if 0 <= k-x < n)

If x ≠ k-x (i.e., 2x ≠ k), these are two different terms in the sum, and the derivative is:
a[k-x] + a[k-x] = 2*a[k-x]

If x = k-x (i.e., 2x = k), they're the same term: a[x]*a[x] = a[x]^2, and the derivative is 2*a[x].

So in BOTH cases, g[x] = 2*a[k-x] (if 0 <= k-x < n), and g[x] = 0 otherwise!

Wait, but that gives g[0] = 2*a[2] for k=2, x=0, which matches!

And g[2] = 2*a[0] for k=2, x=2, which matches!
And g[3] = 0 for k=2, x=3 (since k-x = -1, out of bounds), which matches!

So the formula is simply: **g[x] = 2*a[k-x] if 0 <= k-x < n, else 0.**

```python
def subgrad_at_peak(a, k):
    n = len(a)
    g = np.zeros(n)
    x_lo = max(0, k - n + 1)
    x_hi = min(n - 1, k)
    if x_lo <= x_hi:
        xs = np.arange(x_lo, x_hi + 1)
        g[xs] = 2.0 * a[k - xs]
    return g
```
