    S = a.sum()
    if S < 1e-15:
        return lse_val / (S*S), np.zeros(n)
    w_rev = w[::-1]
    N = 1 << (3 * n - 1).bit_length()
    W = rfft(w_rev, N)
    A = rfft(a, N)
    conv_full = irfft(W * A, N)[:3*n - 2]
    cross = conv_full[n-1:2*n-1][::-1]
    grad = 2.0 * (S * cross - lse_val) / (S ** 3)
    score = lse_val / (S * S)
    return score, grad

def lse_lbfgs_optimize(a0, time_limit, t_start=5.0, t_end=50.0, n_anneal=8, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = float('inf')
    t0 = time.monotonic()
    
    temps = np.geomspace(t_start, t_end, n_anneal)
    
    for ti, t_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def objective_and_grad(x, t_val=t_val):
            x = np.maximum(x, 0.0)
            s_x = x.sum()
            if s_x < 1e-15:
            return 1e10, np.zeros_like(x)
            x_norm = x / s_x
            c = fast_conv_self(x_norm)
            lse_val, w = compute_lse_and_weights(c, t_val)
            w_rev = w[::-1]
            N_fft = 1 << (3 * len(x_norm) - 1).bit_length()
            W = rfft(w_rev, N_fft)
            A = rfft(x_norm, N_fft)
            conv_full = irfft(W * A, N_fft)[:3*len(x_norm) - 2]
            cross = conv_full[len(x_norm)-1:2*len(x_norm)-1][::-1]
            score = lse_val
            grad = 2.0 * (cross - lse_val / s_x) / s_x
            return score, grad
        
        max_iter = max(20, int(remaining * 5))
        try:
            res = minimize(objective_and_grad, a, method='L-BFGS-B',
                          jac=True, options={'maxiter': max_iter, 'ftol': 1e-10,
                                             'gtol': 1e-8, 'maxls': 50,
                                             'maxfun': max_iter * 3})
            if res.x is not None:
                a_new = np.maximum(res.x, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                    c = fast_conv_self(a_new)
                    lse_v, _ = compute_lse_and_weights(c, t_val)
                    sc = lse_v
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                    a = a_new
        except Exception as e:
            if verbose:
                print(f"      L-BFGS-B error at t={t_val:.1f}: {e}")
        
        if verbose and (ti % 2 == 0 or ti == n_anneal - 1):
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS t={t_val:.1f} ({elapsed:.0f}s): best_lse={best_score:.8f}")
    
    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    n_stages = 20
    etas = np.geomspace(0.05, 0.002, n_stages)
    
    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(10, min(40, int(remaining * 3)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            A = rfft(a, fft_size)
            conv = irfft(A * A, fft_size)[:2*n-1]
            M = conv.max()
            if M < 1e-15:
                break
            w = np.maximum(conv - M * 0.85, 0.0)
            ww = w / (w.sum() + 1e-15)
            W = rfft(ww, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and si % 5 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.5f}: best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-10:
                continue
            
            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
                j_candidates.add(min(n-1, max(0, kl - i)))
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                d_max = ai * 0.7
                n_samples = 15
                deltas = np.linspace(0.02, d_max, n_samples)
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*d + d*d
                    if 2 * j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*d + d*d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                n_perturb = max(5, n // 100)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                if no_improve >= 8:
                    break
        else:
            no_improve = 0
        
        if verbose and total_improvements % 300 == 0 and total_improvements > 0:
            print(f"    PCD improv={total_improvements} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 16)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 8)
        f2 = rng.integers(8, 25)
        a = 0.5 + 0.3 * np.cos(2*np.pi*f1*x) + 0.2 * np.cos(2*np.pi*f2*x)
    elif choice == 3:
        c1 = rng.uniform(0.1, 0.4)
        c2 = rng.uniform(0.6, 0.9)
        a = np.exp(-40.0*(x-c1)**2) + np.exp(-40.0*(x-c2)**2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0*x - 1.0)) ** 4
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.3
    elif choice == 6:
        a = np.exp(-8.0 * x) + 0.3
    elif choice == 7:
        a = 0.5 + 0.3*np.sin(6*np.pi*x)*np.cos(3*np.pi*x)
    elif choice == 8:
        n_peaks = rng.integers(4, 12)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.01, 0.08)
            a += np.exp(-(x-c)**2 / (2*width**2))
    elif choice == 9:
        a = np.sinc(x - 0.5)**2 + 0.15
    elif choice == 10:
        a = 1.0 + 0.3*np.sin(2*np.pi*5*x) + 0.2*np.sin(2*np.pi*11*x)
    elif choice == 11:
        a = 0.5 + 0.3*np.cos(2*np.pi*x) + 0.2*np.cos(4*np.pi*x)
    elif choice == 12:
        p = rng.uniform(0.5, 3.0)
        a = np.abs(np.sin(np.pi * x * p)) ** (1.0/p)
    elif choice == 13:
        a = np.exp(-3.0 * (x - 0.5)**2 / 0.1) * (1.0 + 0.3*np.cos(20*np.pi*x))
    elif choice == 14:
        a = 1.0 / (1.0 + np.exp(-10*(x - 0.5))) + 0.5 / (1.0 + np.exp(10*(x - 0.5)))
    else:
        a = np.abs(np.cos(np.pi * x * 1.5)) ** 0.7 + 0.2 * np.sin(2*np.pi*x)**2
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
