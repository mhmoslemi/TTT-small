_solver_cache = None

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))

def build_active_constraints(a, active_idx):
    n = len(a)
    size = 2 * n - 1
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_max_sum(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_min_max(a, solver, active_idx=None, normalize_sum=1.0):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == normalize_sum, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0, alpha_min=0.08, beta_blend=0.25):
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 60

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 40 == 0:
            tol = 0.02 + 0.02 * np.exp(-t / 200.0)
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 6:
                active_set = np.where(conv_y >= M_curr * 0.6)[0]
            if len(active_set) > 0.85 * size:
                active_set = np.where(conv_y >= M_curr * 0.88)[0]

        b_max = lp_max_sum(y, M_curr, solver, active_set)

        b_min = None
        if t % 4 == 0:
            b_min, _ = lp_min_max(y, solver, active_set, normalize_sum=norm)

        if b_max is None:
            if t % 15 == 0:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        if b_min is not None and b_min.sum() > 1e-10:
            g_min_norm = b_min * norm / b_min.sum()
            g = (1.0 - beta_blend) * g + beta_blend * g_min_norm

        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                T_check = build_active_constraints(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 1e-6))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        alpha_t = max(2.0 / (t + 2.0), alpha_min)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.92)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 15 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0

        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def fast_cd_optimize(a0, time_limit, rng, verbose=True):
    """
    Fast coordinate descent with O(n) incremental C updates.
    Key insight: changing a[i] by d only affects C[k] for k in [i, i+n-1].
    We track C incrementally and recompute from scratch periodically.
    """
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    C = fast_autocorr(a)
    best_score = C.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve_passes = 0
    n_passes = 0

    # Wide range factors for exploration
    wide_factors = np.array([0.1, 0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0])
    # Medium range
    med_factors = np.array([0.5, 0.75, 0.9, 0.95, 1.05, 1.1, 1.25, 1.5])
    # Fine range
    fine_factors = np.array([0.92, 0.96, 0.99, 1.01, 1.04, 1.08])

    # Recompute interval: every K passes, recompute C from scratch
    recompute_interval = 50

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Choose factor range based on convergence
        if n_passes < 50:
            factors = wide_factors
        elif n_passes < 200:
            factors = med_factors
        else:
            factors = fine_factors

        # Precompute prefix and suffix max of C
        prefix_max = np.maximum.accumulate(C)
        suffix_max = np.maximum.accumulate(C[::-1])[::-1]

        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-14:
                continue

            # Find unaffected max
            M_unaff = 0.0
            if i > 0:
                M_unaff = max(M_unaff, prefix_max[i - 1])
            if i + n < size:
                M_unaff = max(M_unaff, suffix_max[i + n])

            best_factor = 1.0
            best_val = C.max()

            for f in factors:
                if f == 1.0:
                    continue
                d = ai * (f - 1.0)
                # Affected lags: C[i:i+n] + 2*d*a (linear approx)
                # Note: we use current a (before update), so a[i] is still ai
                affected_max = (C[i:i+n] + 2.0 * d * a).max()
                new_max = max(affected_max, M_unaff)
                if new_max < best_val:
                    best_val = new_max
                    best_factor = f

            # Also try zeroing out
            if ai > 1e-10:
                d_zero = -ai
                affected_max_zero = (C[i:i+n] + 2.0 * d_zero * a).max()
                new_max_zero = max(affected_max_zero, M_unaff)
                if new_max_zero < best_val:
                    best_val = new_max_zero
                    best_factor = 0.0

            if best_factor != 1.0:
                old_ai = a[i]
                d = old_ai * (best_factor - 1.0)
                a[i] = old_ai * best_factor
                # Update C incrementally
                # C[i:i+n] += 2*d*a_old[k-i] for k in [i, i+n-1]
                # Using new a: C[i:i+n] += 2*d*a, then fix diagonal
                C[i:i+n] += 2.0 * d * a
                # Fix diagonal: we need to subtract the overcount
                # At k=2i: we added 2*d*a[i]_new = 2*d*old_ai*best_factor
                # Should have added 2*old_ai*d + d^2
                # Difference: 2*old_ai*d + d^2 - 2*d*old_ai*best_factor = d^2 - 2*d*old_ai*(best_factor-1) = d^2 - 2*d^2 = -d^2
                # So we subtract d^2
                C[2 * i] -= d * d
                # Clamp to non-negative (numerical)
                if C[2 * i] < 0:
                    C[2 * i] = 0.0

                pass_improved = True
                if best_val < best_score:
                    best_score = best_val
                    best_a = a.copy()
                    no_improve_passes = 0
                else:
                    no_improve_passes += 1

        # End of pass
        if not pass_improved:
            no_improve_passes += 10  # penalize no-update passes more
        else:
            if no_improve_passes > 5 and not pass_improved:
                no_improve_passes += 5

        # Periodic recompute to correct drift
        if n_passes % recompute_interval == 0:
            C = fast_autocorr(a)
            actual_max
