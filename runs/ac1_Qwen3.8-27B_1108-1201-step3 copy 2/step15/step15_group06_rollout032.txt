def build_conv_matrix(a, active_idx):
    """Build the convolution constraint matrix T where T[k,j] = a[k-j] for valid (k,j)."""
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_max_sum(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_conv_matrix(a, np.arange(size))
    else:
        T = build_conv_matrix(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_min_max(a, solver, active_idx=None, normalize_sum=1.0):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    if active_idx is None or len(active_idx) >= size:
        T = build_conv_matrix(a, np.arange(size))
    else:
        T = build_conv_matrix(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == normalize_sum, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0):
    """Nesterov-accelerated LP iteration with improved active set management."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 100

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 30 == 0:
            # Progressive tightening
            tol = 0.05 * np.exp(-t / 500.0) + 0.01
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 5:
                active_set = np.where(conv_y >= M_curr * 0.8)[0]
            if len(active_set) > 0.8 * size:
                # Too many constraints, tighten
                active_set = np.where(conv_y >= M_curr * 0.95)[0]

        # Primary: max-sum LP
        b_max = lp_max_sum(y, M_curr, solver, active_set)

        # Secondary: min-max LP (every 2 iterations)
        b_min = None
        if t % 2 == 0 and len(active_set) < size:
            b_min, _ = lp_min_max(y, solver, active_set, normalize_sum=norm)

        if b_max is None:
            if t % 10 == 0:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        # Blend with min-max if available
        if b_min is not None and b_min.sum() > 1e-10:
            g_min_norm = b_min * norm / b_min.sum()
            # Adaptive blend: more min-max when close to convergence

            g = (1.0 - blend) * g + blend * g_min_norm

        # Verify no constraint violations
        if len(active_set) < size:

                if np.any(viol_mask):
                    active_set = np.union1d(active_set, check_idx[viol_mask])

        # Nesterov update with adaptive momentum
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            # Adaptive momentum: start high, decrease
            momentum = min(0.5 + 0.4 * (1.0 - np.exp(-t / 200.0)), 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Geometric ergodic average (decay factor 0.99)
        avg_a = 0.99 * avg_a + 0.01 * a
        if n_avg % 50 == 0 and n_avg > 100:
            s_avg = avg_a.sum()

                avg_check = avg_a * norm / s_avg

                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0
        n_avg += 1

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            active_set = None
            stall_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def fast_cd(a0, time_limit, rng, verbose=True):
    """Fast gradient-based coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            # Try a few candidate values using FFT
            best_val = best_score
            best_candidate = a[i]
            
            for multiplier in [0.5, 0.75, 1.25, 1.5, 2.0]:
                a_try = a.copy()
                a_try[i] = ai * multiplier
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                c_try = fast_autocorr(a_try)
                val = c_try.max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_candidate = ai * multiplier

            if best_candidate != a[i]:
                a[i] = best_candidate
                a = a * norm / a.sum()
                conv = fast_autocorr(a)
                if conv.max() < best_score - 1e-13:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2

        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)

    if s < 1e-15:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver_cache = None


def get_best_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    candidates = [cp.CLARABEL, cp.PDLP, cp.SCS]
    for s in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.SCS
    print(f"  Fallback LP solver: SCS")
    return _solver_cache


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        sc = evaluate_sequence(a_list)
        return float(sc)

def build_conv_matrix(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_max_sum(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_conv_matrix(a, np.arange(size))
    else:
        T = build_conv_matrix(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_min_max(a, solver, active_idx=None, normalize_sum=1.0):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    if active_idx is None or len(active_idx) >= size:
        T = build_conv_matrix(a, np.arange(size))
    else:
        T = build_conv_matrix(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == normalize_sum, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 120

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set with progressive tightening
        if active_set is None or t % 40 == 0:
            tol = 0.04 * np.exp(-t / 400.0) + 0.008
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 4:
                active_set = np.where(conv_y >= M_curr * 0.85)[0]
            if len(active_set) > 0.85 * size:
                active_set = np.where(conv_y >= M_curr * 0.96)[0]

        # Primary LP: max-sum
        b_max = lp_max_sum(y, M_curr, solver, active_set)

        # Secondary LP: min-max (more frequent early on)
        b_min = None
        if t % 2 == 0:
            b_min, _ = lp_min_max(y, solver, active_set, normalize_sum=norm)

        if b_max is None:
            if t % 15 == 0:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        # Blend with min-max: adaptive weight increasing over time
        if b_min is not None and b_min.sum() > 1e-10:
            g_min_norm = b_min * norm / b_min.sum()
            blend = min(0.15 + 0.35 * (1.0 - np.exp(-t / 150.0)), 0.5)
            g = (1.0 - blend) * g + blend * g_min_norm

        # Check for violations in a sample of inactive constraints
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if 0 < len(inactive) < 3 * n:
                sample_size = min(150, len(inactive))
                check_idx = inactive[:sample_size]
                T_check = build_conv_matrix(y, check_idx)
                viol = T_check @ g
                viol_mask = viol > M_curr * (1.0 + 1e-5)
                if np.any(viol_mask):
                    active_set = np.union1d(active_set, check_idx[viol_mask])

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(0.4 + 0.45 * (1.0 - np.exp(-t / 300.0)), 0.88)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-15:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-15:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Geometric ergodic average
        avg_a = 0.995 * avg_a + 0.005 * a
        if n_avg % 80 == 0 and n_avg > 200:
            s_avg = avg_a.sum()
            if s_avg > 1e-15:
                avg_check = avg_a * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0
        n_avg += 1

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0
            if verbose:
                print(f"    NLP restart at t={t}, score={best_score:.10f}")

        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    # Also check the average at the end
    s_avg = avg_a.sum()
    if s_avg > 1e-15:
        avg_final = avg_a * norm / s_avg
        sc = fast_autocorr(avg_final).max()
        if sc < best_score:
            best_score = sc
            best_a = avg_final.copy()

    return best_a, best_score


def fast_cd(a0, time_limit, rng, verbose=True):
    """Fast coordinate descent with FFT-based evaluation and multi-scale candidates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    # Multi-scale candidates: coarse and fine
    coarse_mults = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.7, 2.5]
    fine_mults = [0.9, 0.95, 1.05, 1.1]

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-10 * norm:
                continue

            best_val = best_score
            best_mult = 1.0

            # Coarse search
            for mult in coarse_mults:
                if mult == 1.0:
                    continue
                a_try = a.copy()
                a_try[i] = ai * mult
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                c_try = fast_autocorr(a_try)
                val = c_try.max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_mult = mult

            # Fine search around the best
            if best_mult != 1.0:
                for mult in fine_mults:
                    a_try = a.copy()
                    a_try[i] = ai * best_mult * mult
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try * norm / s_try
                    c_try = fast_autocorr(a_try)
                    val = c_try.max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_mult *= mult

            if best_mult != 1.0:
                a[i] = ai * best_mult
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                if conv.max() < best_score - 1e-14:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= (3 if n_passes < 20 else 6):
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        a = 0.6 + 0.4 * np.cos(2 * np.pi * x) ** 2
    elif style == 8:
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

    t6 = min(remaining * 0.30, 250.0)
    if t6 > 50:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True, seed=1234)
        try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Final fast CD ===
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 7: Final CD n={n_best} ===")
        t7 = min(remaining * 0.40, 150.0)
        if t7 > 30:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = fast_cd(a_norm, t7, rng, verbose=True)
            try_update(a_out, f"cd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Micro CD for ultra-fine polishing ===
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 8: Micro CD n={n_best} ===")
        t8 = min(remaining * 0.50, 100.0)
        if t8 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = micro_cd(a_norm, t8, rng, verbose=True)
            try_update(a_out, f"mcd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: One more LP on the best if time allows ===
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 9: Extra LP n={n_best} ===")
        t9 = min(remaining * 0.70, 150.0)
        if t9 > 40:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp(a_norm, t9, solver, verbose=False, seed=5678)
            try_update(a_out, f"lp_extra_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
