    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    a = a * norm / a.sum()

    best_score = fast_autocorr(a).max()
    t0 = time.time()

    y = a.copy()
    prev_y = a.copy()
    
    for t in range(1, 100000):
        if time.time() - t0 > time_limit:
            break
        
        # Compute autocorrelation and find peak
        conv = fast_autocorr(y)
        peak_val = conv.max()
        peak_idx = conv.argmax()
        
        # Subgradient: g[j] = y[peak_idx - j] (for valid indices)
        # peak_idx is in [0, 2n-2]
        # y[peak_idx - j] is valid when 0 <= peak_idx - j < n
        # i.e., peak_idx - n + 1 <= j <= peak_idx
        g = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        for j in range(lo, hi + 1):
            g[j] = y[peak_idx - j]
        
        # Multiple peaks (use all peaks above 0.9 * max)
        threshold = 0.9 * peak_val
        peak_positions = np.where(conv >= threshold)[0]
        if len(peak_positions) > 1:
            g = np.zeros(n)
            for pk in peak_positions:
                lo = max(0, pk - n + 1)
                hi = min(n - 1, pk)
                for j in range(lo, hi + 1):
                    g[j] += y[pk - j]
            g /= len(peak_positions)
        
        # Step size (Polyak-like)
        # gap = peak_val - lower_bound
        # lower_bound = norm^2 / (2n-1) = 2n/(2n-1) ≈ 1
        lb = 2.0 * n / (2.0 * n - 1.0)
        gap = peak_val - lb
        if gap < 1e-12:
            break
        step = gap / (np.dot(g, g) + 1e-16)
        # Cap the step
        step = min(step, norm / (5.0 * np.sqrt(n)))
        
        # Update
        x = y - step * g
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            x = y.copy()
        else:
            x = x * norm / s
        
        # Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * x
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Check
        conv_new = fast_autocorr(y_new)
        if conv_new.max() < best_score:
            best_score = conv_new.max()
            best_a = y_new.copy()
        
        conv_x = fast_autocorr(x)
        if conv_x.max() < best_score:
            best_score = conv_x.max()
            best_a = x.copy()
        
        prev_y = y
        y = y_new
        
        if verbose and t % 1000 == 0:
            print(f"  SubGrad t={t}: best={best_score:.8f}")
    
    return best_a, best_score

import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(a, b, n=None):
    if n is None:
        n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * np.conj(B), fft_size)[:n]


def project_nonneg_sum(v, S):
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.full(n, S / n)
    return v * (S / s)


def smooth_grad(a, w, n):
    return 2.0 * cross_corr(w, a, n)

            else:
                a_new = a_new * norm / s_new

            # Nesterov-like mixing
            alpha = 2.0 / (it + 2.0)
            a_curr = (1.0 - alpha) * a_curr + alpha * a_new
            a_curr = project_nonneg_sum(a_curr, norm)

def subgradient_opt(a0, time_limit, verbose=False, tag="subgrad"):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound on peak

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(y)
        peak_val = conv.max()
        if peak_val <= lb + 1e-10:
            break

        # Multi-peak subgradient
        threshold = 0.92 * peak_val
        peak_positions = np.where(conv >= threshold)[0]

        # Weighted subgradient: weight = (conv[pos] - threshold+eps) / sum
        weights = conv[peak_positions] - threshold
        weights = weights / weights.sum()

        g = np.zeros(n)
        for idx, pk in enumerate(peak_positions):
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            for j in range(lo, hi + 1):
                g[j] += weights[idx] * y[pk - j]

        # Polyak step
        gap = peak_val - lb
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break
        step = gap / g_norm_sq
        step = min(step, 2.0 * norm / (n * np.sqrt(g_norm_sq) + 1e-10))

        x = y - step * g

        # Away step: if a component is at 0, try to move it away
        at_zero = x < 0
        if at_zero.any():
            x[~at_zero] -= step * g[~at_zero]
            x[at_zero] = y[at_zero]  # keep at zero
            # Project
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-15:
                x = y.copy()
            else:
                x = x * norm / s

        x = project_nonneg_sum(x, norm)

        # Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * x
        y_new = project_nonneg_sum(y_new, norm)

        sc_y = fast_autocorr(y_new).max()
        sc_x = fast_autocorr(x).max()

        if sc_y < best_score:
            best_score = sc_y
            best_a = y_new.copy()
        if sc_x < best_score:
            best_score = sc_x
            best_a = x.copy()

        y = y_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{tag}] t={t} ({elapsed:.0f}s): best={best_score:.8f} peak={peak_val:.6f}")

    return best_a, best_score


def gradient_descent_direct(a0, time_limit, verbose=False, tag="grad_direct"):
    """Direct gradient descent on the smooth objective with line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = a * norm / s0

    t0 = time.monotonic()

    # Use a fixed moderate temperature for practical smoothness
    T = 0.5

    y = a.copy()
    step = 0.5 / n

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(y)
        c_max = c.max()
        exp_c = np.exp((c - c_max) / T)
        w = exp_c / exp_c.sum()
        grad = smooth_grad(y, w, n)

        # Line search (backtracking)
        alpha = step
        for _ in range(10):
            x = y - alpha * grad
            x = project_nonneg_sum(x, norm)
            c_x = fast_autocorr(x)
            exp_x = np.exp((c_x - c_x.max()) / T)
            f_x = T * np.log(exp_x.sum())
            f_y = T * np.log(exp_c.sum())
            if f_x <= f_y - 0.01 * alpha * np.dot(grad, grad):
                break
            alpha *= 0.5

        x = y - alpha * grad
        x = project_nonneg_sum(x, norm)

        # Nesterov
        beta = 2.0 / (t + 2.0)
        y_new = (1.0 - beta) * y + beta * x
        y_new = project_nonneg_sum(y_new, norm)

        sc_y = fast_autocorr(y_new).max()
        sc_x = fast_autocorr(x).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y_new.copy()
        if sc_x < best_score:
            best_score = sc_x
            best_a = x.copy()

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{tag}] t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            best_j = -1
            best_val = conv.max()

            # Try a handful of candidate destinations
            candidates = []
            # Positions that would reduce peak contribution
            for offset in list(range(1, min(window + 1, n))) + [n // 2, n - 1 - i if 0 <= n - 1 - i < n else 0]:
                j = (i + offset) % n
                if j != i and j not in candidates:
                    candidates.append(j)

            for j in candidates[:15]:
                # Try moving fraction of a[i] to j
                for frac in [0.3, 0.5, 0.7]:
                    d = ai * frac
                    if d < norm * 1e-10:
                        continue
                    a_t = a.copy()
                    a_t[i] -= d
                    a_t[j] += d
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t * norm / st
                    conv_t = fast_autocorr(a_t)
                    if conv_t.max() < best_val - 1e-12:
                        best_val = conv_t.max()
                        best_d = d
                        best_j = j

            if best_j >= 0 and best_d > 0:
                a[i] -= best_d
                a[best_j] += best_d
                a = a * norm / a.sum()
                conv = fast_autocorr(a)
                if conv.max() < best_score - 1e-12:
                    best_score = conv.max()
                    best_a = a.copy()
                pass_improved = True

        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and no_improve == 0 and (perm[0] == 0):  # approximate per-pass
            elapsed = time.monotonic() - t0
            print(f"    [{tag}] ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def make_random_init(n, rng, pattern="uniform"):
    if pattern == "uniform":
        a = np.ones(n)
    elif pattern == "random":
        a = rng.random(n)
    elif pattern == "bimodal":
        a = np.concatenate([np.ones(n // 2) * rng.uniform(0.5, 1.5), np.ones(n - n // 2) * rng.uniform(0.5, 1.5)])
    elif pattern == "sine_mod":
        x = np.linspace(0, 2 * np.pi, n, endpoint=False)
        a = 0.5 + 0.3 * np.sin(x) + 0.2 * np.sin(2 * x + rng.random() * np.pi)
        a = np.maximum(a, 0.01)
    elif pattern == "ramp":
        a = np.linspace(1.0, 0.3, n)
        a = a + 0.1 * rng.random(n)
        a = np.maximum(a, 0.01)
    elif pattern == "gaussian_centered":
        x = np.arange(n) - n / 2
        a = np.exp(-x ** 2 / (2 * (n / 4) ** 2))
        a = a + 0.05
    elif pattern == "step_asym":
        # Asymmetric step: large on one side, small on other
        a = np.zeros(n)
        a[:n // 3] = 1.0
        a[n // 3:2 * n // 3] = 0.4
        a[2 * n // 3:] = 0.1
        a = a + 0.05 * rng.random(n)
    elif pattern == "optimistic_asym":
        # Designed to have low center autocorrelation
        # a[i] * a[n-1-i] should be small
        a = np.zeros(n)
        center = n // 2
        for i in range(n):
            dist_from_center = abs(i - center)
            # Make a[i] large when a[n-1-i] is small
            mirror_dist = abs((n - 1 - i) - center)
            a[i] = 0.5 + 0.5 * (dist_from_center / center) if center > 0 else 0.5
        a = a + 0.1 * rng.random(n)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    return a

            return float('inf')

        if s < 0.01:
            return
