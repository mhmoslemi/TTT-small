The goal is to make b[k] as equal as possible for all k. At each step:

        if remaining < 5:

        progress = stage / n_stages
        lr = lr_base * (0.1 + 0.9 * 0.5 * (1 + math.cos(math.pi * progress)))

        sharpness = sharp_start + (sharp_end - sharp_start) * (progress ** 0.5)

            if M < 1e-15:

            if g_max > 1e-15:

            g = g - g.mean()
            
            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * g * g
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            # Multiplicative update for non-negativity
            a_new = a * np.exp(-lr * m_hat / (np.sqrt(v_hat) + eps))

            if s_new < 1e-15:

        for i in range(lo, hi):
            T[idx, i] = a[k - i]
    return T

def lp_best_response(a, M, solver, active_lags=None):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for k in active_lags, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_lags is not None and 0 < len(active_lags) < size:
        T = build_constraint_matrix(a, active_lags)
        constraints = [T @ b_var <= M]
    else:
        # Full constraint set
        T = np.zeros((size, n))
        for k in range(size):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                T[k, i] = a[k - i]
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, seed=42):
    """LP Nesterov optimization with constraint generation."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Constraint generation: use lags above threshold
        threshold = M * 0.7
        active_set = np.where(conv_y > threshold)[0]
        if len(active_set) < 5:
            active_set = np.where(conv_y > M * 0.4)[0]
        if len(active_set) > 300:
            active_set = np.where(conv_y > M * 0.85)[0]
            if len(active_set) < 3:
                active_set = None
        
        b = lp_best_response(y, M, solver, active_set)
        
        if b is None:
            b = lp_best_response(y, M, solver, None)
        
        if b is None:
            continue
        
        # Check for violated constraints (constraint generation)
        if active_set is not None and len(active_set) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.001)[0]
            if len(violated) > 0:
                active_set = np.union1d(active_set, violated[:50])
                b2 = lp_best_response(y, M, solver, active_set)
                if b2 is not None:
                    b = b2
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        sc_y = compute_score(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = compute_score(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with gradient-informed candidate selection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue
            
            # Find candidate j's based on peak lags
            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j1 = int((kl - i) % n)
                j2 = int((kl + i) % n)
                j_candidates.add(j1)
                j_candidates.add(j2)
            j_candidates.discard(i)
            
            if not j_candidates:
                continue
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gradient of conv w.r.t. moving mass from i to j
                g = np.zeros(size)
                
                # Add contribution of j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                
                # Remove contribution of i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Try different delta values
                d_max = ai * 0.7
                n_samples = 15
                if d_max > 1e-15:
                    deltas = np.linspace(0.05, d_max, n_samples)
                    for d in deltas:
                        new_conv = conv + d * g
                        # Fix the diagonal terms
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                        v = new_conv.max()
                        if v < best_v:
                            best_v = v
                            best_delta = d
                            best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                    new_max = fast_conv_self(a).max()
                    new_score = 2.0 * n * new_max
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                n_perturb = max(5, n // 100)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                if no_improve >= 8:
                    break
        else:
            no_improve = 0
        
        if verbose and total_improvements % 300 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True, seed=42):
    """Vegas-style multiplicative updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 20
    etas = np.geomspace(0.05, 0.001, n_stages)
    
    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(10, min(50, int(remaining * 3)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break
            
            # Pressure from peaks
            w = np.maximum(conv - M * 0.85, 0.0)
            ww = w / (w.sum() + 1e-15)
            
            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            
            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-14)
            a = a / a.sum()
            
            sc = fast_conv_self(a).max() * 2.0 * n
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Vegas stage {si} eta={eta_s:.5f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        # Skewed Gaussian
        c = rng.uniform(0.3, 0.7)
        w = rng.uniform(0.1, 0.3)
        a = np.exp(-(x - c) ** 2 / (2 * w ** 2))
    elif choice == 1:
        # Two peaks (asymmetric)
        c1 = rng.uniform(0.1, 0.4)
        c2 = rng.uniform(0.6, 0.9)
        w = rng.uniform(0.05, 0.15)
        a = np.exp(-(x - c1) ** 2 / (2 * w ** 2)) + 0.7 * np.exp(-(x - c2) ** 2 / (2 * w ** 2))
    elif choice == 2:
        # Polynomial phase (use cos for non-negativity)
        f1 = rng.integers(2, 8)
        f2 = rng.integers(8, 20)
        phase1 = 2 * np.pi * f1 * x
        phase2 = 2 * np.pi * f2 * x
        a = 0.5 + 0.3 * np.cos(phase1) + 0.2 * np.cos(phase2)
    elif choice == 3:
        # Power law
        p = rng.uniform(0.5, 3.0)
        a = x ** p + 0.1 * (1 - x) ** p + 0.05
    elif choice == 4:
        # Random multi-peak
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        widths = rng.uniform(0.03, 0.12, n_peaks)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-(x - c) ** 2 / (2 * w ** 2))
    elif choice == 5:
        # Sinc-like
        f = rng.uniform(2.0, 8.0)
        a = np.sinc(f * (x - 0.5)) ** 2 + 0.05
    elif choice == 6:
        # Exponential decay (asymmetric)
        rate = rng.uniform(2.0, 8.0)
        a = np.exp(-rate * x) + 0.3 * np.exp(-rate * (1 - x)) + 0.05
    elif choice == 7:
        # Modulated carrier
        f_car = rng.integers(5, 15)
        f_mod = rng.integers(1, 4)
        a = (1.0 + 0.5 * np.cos(2 * np.pi * f_mod * x)) * np.cos(2 * np.pi * f_car * x) ** 2 + 0.1
    elif choice == 8:
        # Step-like (sparse)
        n_blocks = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(n_blocks):
            start = rng.integers(0, n - n // 10)
            width = rng.integers(n // 20, n // 5)
            a[start:start + width] += 1.0
        a = a + 0.01
    else:
        # Stretched uniform
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * x)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def sparsify(a, threshold_frac=0.002):
    """Zero out small elements."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a_norm = a / s
    threshold = threshold_frac * np.max(a_norm)
    mask = a_norm > threshold
    if np.sum(mask) < n // 5:
        mask = a_norm > threshold * 0.1
    a_s = np.where(mask, a, 0.0)
    s_new = a_s.sum()
    if s_new < 1e-15:
        return a
    return a_s / s_new

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses a multi-phase strategy:
    Phase 1: Soft subgradient Adam (fast, effective for initial progress)
    Phase 2: Vegas multiplicative updates (exploits non-negativity)
    Phase 3: LP Nesterov (precise refinement)
    Phase 4: Pair coordinate descent (final polish)
    Phase 5: Multi-start exploration (escape local minima)
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = compute_score(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Soft Subgradient Adam (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 1: Subgradient Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = softmax_adam_optimize(best_seq, t1, verbose=True, seed=seed)
        check_and_update(a_out, "subgrad_keep")
    
    # === Phase 2: Vegas multiplicative updates ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Vegas n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = vegas_optimize(best_seq, t2, verbose=True, seed=seed + 1)
        check_and_update(a_out, "vegas_n{}".format(len(best_seq)))
    
    # === Phase 3: Subgradient Adam round 2 (with different hyperparams) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Subgradient Adam R2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        # Slightly perturb before second round
        a_perturbed = best_seq.copy()
        n_pert = max(5, len(a_perturbed) // 100)
        idx_p = rng.choice(len(a_perturbed), size=n_pert, replace=False)
        a_perturbed[idx_p] *= (1.0 + 0.05 * rng.standard_normal(n_pert))
        a_perturbed = np.maximum(a_perturbed, 0.0)
        a_perturbed = a_perturbed / a_perturbed.sum()
        a_out, _ = softmax_adam_optimize(a_perturbed, t3, verbose=False, seed=seed + 2)
        check_and_update(a_out, "subgrad_r2_n{}".format(len(best_seq)))
    
    # === Phase 4: LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True, seed=seed + 3)
        check_and_update(a_out, "lp_n{}".format(len(best_seq)))
    
    # === Phase 5: Pair Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 5: Pair CD n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        check_and_update(a_out, "pcd_n{}".format(len(best_seq)))
    
    # === Phase 6: Sparsification + final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Sparsify + Final polish ===")
    a_spar = sparsify(best_seq, 0.003)
    check_and_update(a_spar, "sparse_n{}".format(len(best_seq)))
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t6 = min(remaining * 0.4, 60.0)
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=False)
        check_and_update(a_out, "pcd_final_n{}".format(len(best_seq)))
    
    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 7: Multi-length exploration (remaining={remaining:.0f}s) ===")
        lengths_to_try = [n_base + 200, n_base + 400, n_base - 100, n_base + 600]
        for n_try in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            if n_try < 100 or n_try > 4000:
                continue
            t_try = min(remaining * 0.3, 50.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = softmax_adam_optimize(init_try, min(t_try, 30.0), verbose=False, seed=seed + n_try)
            check_and_update(a_out, "ml_n{}".format(n_try))
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 15.0), rng, verbose=False)
                check_and_update(a_out, "ml_pcd_n{}".format(n_try))
    
    # === Phase 8: Fresh starts ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 8: Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(4):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            n_fresh = n_base + int(rng.integers(-100, 300))
            if n_fresh < 400:
                n_fresh = 800
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.25, 20.0)
            a_out, _ = softmax_adam_optimize(a_fresh, t_fresh, verbose=False, seed=seed + 100 + attempt)
            check_and_update(a_out, "fresh{}_n{}".format(attempt, n_fresh))
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_out, _ = pair_cd_optimize(a_out, min(remaining * 0.3, 10.0), rng, verbose=False)
                check_and_update(a_out, "fresh_pcd{}_n{}".format(attempt, n_fresh))
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0, 1.0, 1.0, 1.0]
    
    return [float(x) for x in best_seq]
