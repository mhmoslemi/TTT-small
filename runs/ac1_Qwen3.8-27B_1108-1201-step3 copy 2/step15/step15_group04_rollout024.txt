def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def normalize_seq(a, target_sum=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    if target_sum is not None:
        return a * target_sum / s
    return a / s


def lse_smoothed_grad(a, T_val):
    """Compute gradient of LSE-smoothed max of auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    conv = fast_conv_self(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n), conv, max_c
    shifted = (conv - max_c) / T_val
    shifted = np.maximum(shifted, -200.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g1 = irfft(np.conj(A) * W, fft_size)[:n]
    aw = irfft(A * W, fft_size)
    g2 = aw[:n]
    grad = 2.0 * (g1 + g2)
    return grad, conv, max_c


def lse_momentum_descent(a0, time_limit, verbose=True):
    """LSE gradient descent with Nesterov momentum and adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0)
    a = a * norm
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    v = np.zeros(n)
    y = a.copy()
    total_iters = 0

    n_phases = 8
    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        T_start = 80.0 * (0.7 ** phase)
        T_end = 0.5 * (0.6 ** phase)
        n_steps = max(40, int(remaining * 3))
        lr_base = 1.0 / (np.sqrt(n) * T_start)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            progress = it / max(1, n_steps)
            T_val = T_start * (T_end / T_start) ** progress
            lr = lr_base * (1.0 / (1.0 + 0.02 * it))

            grad_y, conv_y, max_c = lse_smoothed_grad(y, T_val)
            if max_c < 1e-15:
                break

            v = 0.9 * v + 0.1 * grad_y
            y_new = y - lr * v
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            a_new = y_new + 0.7 * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
            else:
                a_new = y_new.copy()

            y = y_new
            a = a_new
            total_iters += 1

            sc = fast_conv_self(a).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()

            sc_y = fast_conv_self(y).max()
            if sc_y < best_score - 1e-12:
                best_score = sc_y
                best_a = y.copy()

        if verbose:
            print(f"    LSE-M phase {phase} T=[{T_start:.1f}->{T_end:.1f}]: best={best_score:.8f}")

    return best_a, best_score


def block_cd_optimize(a0, time_limit, rng, verbose=True, block_sizes=(2, 3, 4)):
    """Block coordinate descent using gain vectors for fast updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0)
    a = a * norm
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        if cur_max < 1e-12:
            break

        perm = rng.permutation(n)
        pass_improved = False
        elapsed_in_pass = 0.0

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            block_size = block_sizes[0]
            if total_improvements % 500 == 0 and total_improvements > 0:
                block_size = block_sizes[1] if len(block_sizes) > 1 else 2

            # Select block partners based on peak positions
            peak_lags = np.where(conv >= cur_max * 0.85)[0]
            candidates = []
            for kl in peak_lags[:10]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                if j1 != i:
                    candidates.append(j1)
                if j2 != i:
                    candidates.append(j2)
            # Add random candidates
            for _ in range(5):
                candidates.append(int(rng.integers(n)))

            # Try pairs (i, j)
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            best_delta_j = 0.0

            tried = set()
            for j in candidates[:block_size * 3]:
                if j == i or j in tried:
                    continue
                tried.add(j)
                aj = a[j]

                # Gain vectors
                hi = np.zeros(size)
                lo_i = max(0, 2 * i - (n - 1))
                hi_i = min(size, 2 * i + 1)
                # h_i[k] = 2*a[k-i] for 0<=k-i<n
                start = max(0, i)
                end = min(size, i + n)
                hi[start:end] = 2.0 * a[0:end - start]

                hj = np.zeros(size)
                start_j = max(0, j)
                end_j = min(size, j + n)
                hj[start_j:end_j] = 2.0 * a[0:end_j - start_j]

                # Second order corrections
                # At k=2i: delta_i^2
                # At k=2j: delta_j^2
                # At k=i+j: 2*delta_i*delta_j
                k_ii = 2 * i
                k_jj = 2 * j
                k_ij = i + j
                valid_ii = k_ii < size
                valid_jj = k_jj < size
                valid_ij = k_ij < size

                # Sample deltas
                n_samples = 30
                deltas_i = np.linspace(0.0, 0.9 * ai, n_samples)
                for di in deltas_i:
                    dj = di  # transfer mass
                    new_vals = conv + di * hi - dj * hj
                    if valid_ii:
                        new_vals[k_ii] += di * di - 2 * ai * di + 2 * ai * di  # already in hi
                        # Correction: hi already accounts for linear term
                        # The quadratic term at k=2i is di^2
                        new_vals[k_ii] += di * di
                    if valid_jj:
                        new_vals[k_jj] += -dj * dj + 2 * aj * dj
                        # hj accounts for linear: 2*aj*dj at k_jj
                        # Quadratic: -dj^2 (since a_j decreases)
                        new_vals[k_jj] += -dj * dj
                    if valid_ij and i != j:
                        new_vals[k_ij] += 2 * di * dj

                    v = new_vals.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = di
                        best_j = j
                        best_delta_j = dj

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 3:
                # Perturb in frequency domain
                fft_size = 1 << (size - 1).bit_length()
                A = rfft(a, fft_size)
                magnitude = np.abs(A)
                phase = np.angle(A)
                # Perturb phase slightly
                n_pert = max(5, n // 20)
                freq_idx = rng.choice(1, fft_size // 2 + 1 - 1, size=n_pert, replace=False) + 1
                phase[freq_idx] += 0.1 * rng.standard_normal(n_pert)
                A_new = magnitude * np.exp(1j * phase)
                a_new = irfft(A_new, fft_size)[:n]
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc < cur_max * 1.05:  # Accept if not too bad
                        a = a_new
                no_improve = 0
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_nesterov_fast(a0, time_limit, solver, verbose=True):
    """LP Nesterov with aggressive screening and early termination."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0)
    a = a * norm
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    no_improve = 0
    conv_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 30:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive screening - very aggressive
        if t < 20:
            frac = 0.3
        elif t < 100:
            frac = 0.5
        elif t < 500:
            frac = 0.7
        else:
            frac = 0.85
        active = np.where(conv_y > M * frac)[0]
        if len(active) < 20:
            active = np.where(conv_y > M * 0.3)[0]

        # Build constraint matrix only for active set
        b_var = cp.Variable(n, nonneg=True)
        T_act = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            lo = max(0, m)
            hi = min(size, m + n)
            if hi > lo:
                T_act[idx, 0:hi - lo] = a[:hi - lo]
            lo2 = max(0, m - n + 1)
            hi2 = min(size, m + 1)
            if hi2 > lo2:
                T_act[idx, m - np.arange(lo2, hi2)] += a[:hi2 - lo2]

        constraints = [T_act @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b = np.asarray(b_var.value, dtype=np.float64)
        b = np.maximum(b, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum

        # Check if dropped constraints are violated
        if len(active) < size:
            check = fast_conv_cross(y, g)
            if check.max() > M * 1.01:
                # Re-solve with all constraints (expensive, so limit)
                if t < 200:
                    T_full = np.zeros((size, n))
                    for m in range(size):
                        lo = max(0, m)
                        hi = min(size, m + n)
                        if hi > lo:
                            T_full[m, 0:hi - lo] = a[:hi - lo]
                        lo2 = max(0, m - n + 1)
                        hi2 = min(size, m + 1)
                        if hi2 > lo2:
                            T_full[m, m - np.arange(lo2, hi2)] += a[:hi2 - lo2]
                    constraints_full = [T_full @ b_var <= M]
                    prob_full = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints_full)
                    try:
                        prob_full.solve(solver=solver, verbose=False)
                    except Exception:
                        pass
                    if prob_full.status in ('optimal', 'optimal_inaccurate'):
                        b = np.asarray(b_var.value, dtype=np.float64)
                        b = np.maximum(b, 0.0)
                        if b.sum() > 1e-15:
                            g = b * norm / b.sum()
                else:
                    continue

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        a_new = y_new + min(1.0, (t - 1.0) / (t + 2.0)) * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a > 1e-15:
            a_new = a_new * norm / s_a
        else:
            a_new = y_new.copy()

        y = y_new
        a = a_new

        sc = fast_conv_self(a).max()
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if no_improve == 50:
            # Perturb
            n_pert = max(10, n // 15)
            idx_p = rng_local_choice(n, n_pert, t)
            y[idx_p] *= (1.0 + 0.3 * np.random.default_rng(t * 7).standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-N t={t} ({elapsed:.0f}s): best={best_score:.8f}")

        if no_improve > 200:
            break

    return best_a, best_score


def rng_local_choice(n, k, seed):
    rng = np.random.default_rng(seed)
    return rng.choice(n, size=k, replace=False)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        # Asymmetric with oscillation
        a = (1.0 - x) * (0.5 + 0.3 * np.cos(15 * np.pi * x))
        a = np.maximum(a, 0.01)
    else:
        # Power law decay
        a = (1.0 - x) ** 0.5 * (1.0 + 0.2 * np.sin(25 * np.pi * x))
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def dual_guided_step(a, solver):
    """One step using LP dual variables for gradient direction."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    conv = fast_conv_self(a)
    M = conv.max()
    if M < 1e-12:
        return a
    active = np.where(conv > M * 0.7)[0]
    if len(active) < 10:
        active = np.where(conv > M * 0.4)[0]

    b_var = cp.Variable(n, nonneg=True)
    T_act = np.zeros((len(active), n))
    for idx, m in enumerate(active):
        lo = max(0, m)
        hi = min(size, m + n)
        if hi > lo:
            T_act[idx, 0:hi - lo] = a[:hi - lo]
        lo2 = max(0, m - n + 1)
        hi2 = min(size, m + 1)
        if hi2 > lo2:
            T_act[idx, m - np.arange(lo2, hi2)] += a[:hi2 - lo2]

    constraints = [T_act @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return a
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a

    b = np.asarray(b_var.value, dtype=np.float64)
    b = np.maximum(b, 0.0)
    if b.sum() < 1e-15:
        return a

    # Try to get duals
    duals = None
    try:
        if constraints[0].dual_value is not None:
            full_duals = np.zeros(size)
            full_duals[active] = np.asarray(constraints[0].dual_value, dtype=np.float64)
            duals = full_duals
    except Exception:
        pass

    if duals is None:
        # Fallback: use b direction
        g = b * norm / b.sum()
        a_new = 0.7 * a + 0.3 * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            return a_new * norm / s
        return a

    # Gradient from duals
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    D = rfft(duals, fft_size)
    g1 = irfft(np.conj(A) * D, fft_size)[:n]
    g2 = irfft(A * D, fft_size)[:n]
    grad = 2.0 * (g1 + g2)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return a

    # Line search
    best_a = a
    best_sc = M
    for step in [0.05, 0.1, 0.2, 0.3, 0.5, 0.1 * gn, 0.5 * gn]:
        a_new = a - step * grad / gn
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new * norm / s
        sc = fast_conv_self(a_new).max()
        if sc < best_sc:
            best_sc = sc
            best_a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    t0 = min(remaining * 0.06, 90.0)
    if t0 > 15:
        print(f"\n=== Phase 0: LSE polish (budget={t0:.0f}s) ===")
        a_out, _ = lse_momentum_descent(best_seq, t0, verbose=True)
        try_update(a_out, "lse0")

    # Phase 1: Block CD (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 1: Block CD n={len(best_seq)} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = block_cd_optimize(best_seq, t1, rng, verbose=True)
        try_update(a_out, "bcd1")

    # Phase 2: LSE momentum
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: LSE momentum n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = lse_momentum_descent(best_seq, t2, verbose=True)
        try_update(a_out, "lse2")

    # Phase 3: LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_nesterov_fast(best_seq, t3, solver, verbose=True)
        try_update(a_out, "lp3")

    # Phase 4: Block CD round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: Block CD round 2 (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = block_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, "bcd4")

    # Phase 5: Dual-guided steps
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: Dual-guided (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_curr = best_seq.copy()
        t5_start = time.monotonic()
        for _ in range(50):
            if time.monotonic() - t5_start > t5:
                break
            a_curr = dual_guided_step(a_curr, solver)
            try_update(a_curr, "dual5")

    # Phase 6: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length exploration (remaining={remaining:.0f}s) ===")
    if remaining > 80:
        n_trial = n_base + 400
        init_long = interpolate_seq(best_seq, n_trial)
        t_long = min(remaining * 0.3, 120.0)
        a_out, _ = lse_momentum_descent(init_long, t_long, verbose=False)
        try_update(a_out, f"lse_long{n_trial}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and len(best_seq) == n_trial:
            t_bcd_long = min(remaining * 0.4, 80.0)
            a_out, _ = block_cd_optimize(best_seq, t_bcd_long, rng, verbose=False)
            try_update(a_out, f"bcd_long{n_trial}")

    # Phase 7: Final rapid alternation
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final alternation (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        n_rounds = int(remaining // 30)
        for r in range(max(1, n_rounds)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            # Alternate BCD and LSE
            if r % 2 == 0:
                t_i = min(remaining * 0.4, 60.0)
                a_out, _ = block_cd_optimize(best_seq, t_i, rng, verbose=False)
                try_update(a_out, f"bcd_fin{r}")
            else:
                t_i = min(remaining * 0.4, 60.0)
                a_out, _ = lse_momentum_descent(best_seq, t_i, verbose=False)
                try_update(a_out, f"lse_fin{r}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
