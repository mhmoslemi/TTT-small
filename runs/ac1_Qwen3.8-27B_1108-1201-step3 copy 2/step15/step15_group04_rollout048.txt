def fast_conv_self(a):

def lp_solve_fast(a, M, solver, active_set=None, warm_start=None):
    """Solve the LP: max sum(b) s.t. T(a) @ b <= M, b >= 0"""
    n = len(a)
    size = 2 * n - 1
    
    # Build the constraint matrix only for active constraints
    if active_set is not None:
        rows = len(active_set)
    else:
        rows = size
        active_set = np.arange(size)
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Build Toeplitz rows
    # T[k, i] = a[|k-i|] if |k-i| < n else 0
    # Actually, for the convolution (a*b)[k] = sum_i a[j]*b[k-j]
    # For each output k, the constraint is sum_i b[i] * a_contribution[i, k] <= M
    
    # Let's build it efficiently
    # (a*b)[k] = sum_i a[i] * b[k-i] where both indices in [0, n-1]
    # So for fixed k, the coefficient of b[j] is a[k-j] if 0 <= k-j < n
    
    # We'll build the matrix row by row
    T_active = np.zeros((rows, n))
    for row_idx, k in enumerate(active_set):
        # (a*b)[k] = sum_j a[k-j] * b[j] for valid j
        # j must satisfy 0 <= j < n and 0 <= k-j < n
        # So j in [max(0, k-n+1), min(n-1, k)]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            T_active[row_idx, j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi+1)]
    
    constraints = [T_active @ b_var <= M]
    
    if warm_start is not None:
        b_var.value = warm_start
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    # Get dual variables
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            duals_iter = np.asarray(con.dual_value, dtype=np.float64)
            if len(duals_iter) == rows:
                duals = np.zeros(size)
                duals[active_set] = duals_iter
    except Exception:
        pass
    
    return val, duals

    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    # Progressive constraint tightening
    screen_fractions = [0.5, 0.6, 0.7, 0.75, 0.8, 0.85, 0.9]
    
    y = a.copy()
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max() * m_relaxation
        if M < 1e-12:
            break
        
        # Adaptive constraint screening
        screen_idx = min(len(screen_fractions) - 1, t // 50)
        threshold = M * screen_fractions[screen_idx]

        if top_k < size:
            top_indices = np.argsort(conv_y)[-top_k:]
            active = top_indices[conv_y[top_indices] > threshold * 0.8]
        else:
            active = np.where(conv_y > threshold)[0]
        
        if len(active) < 20:
            active = np.where(conv_y > M * 0.3)[0]

        if b is None:
            # Try with all constraints
            b, duals = lp_solve_fast(y, M, solver, None, warm_start=y)
        
        if b is None or b.sum() < 1e-15:
            continue
        
        g = b * norm / b.sum()
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            a_new = y_new  # Use the Nesterov point
            # Also try the averaged point
            if t % 10 == 0:
                a_avg = y_new  # Just use y_new for now
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        # Perturbation to escape plateaus
        if no_improve in [50, 100, 200, 400]:
            rng_local = np.random.default_rng(t)
            n_pert = max(10, n // 30)
            idx_p = rng_local.choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.15 * rng_local.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = no_improve % 50  # Reset partially
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPV2 t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def subgradient_refine(a0, time_limit, verbose=True):
    """Subgradient method using active constraint weights."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    eta = 0.1  # Step size
    decay = 0.995
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-15:
            break
        
        # Compute subgradient: weighted sum of gradients of active constraints
        # Use softmax weighting
        eps = 0.05 * M
        weights = np.exp((conv - M) / eps)
        weights /= weights.sum()
        
        # Gradient of (a*a)[k] wrt a[i] = a[k-i] + a[i-k] (where defined)
        # This is the cross-correlation of a with the "flipped" version at each lag
        # More efficiently: grad[i] = sum_k weights[k] * (a[k-i] + a[i-k mod stuff])
        
        # Use FFT: the gradient is 2 * (cross-correlation of weights with a)
        # Actually, for (a*a)[k] = sum_j a[j]*a[k-j], d/d(a[i]) = a[k-i] + a[i-k] (with bounds)
        # This is the "symmetric cross-correlation"
        
        # In FFT: if we define the cross-correlation as C[k] = sum_j a[j]*a[j+k]
        # then d(C[k])/d(a[i]) involves a[k-i] and a[i-k]
        
        # More precisely, using the full convolution:
        # (a*a)[k] = sum_j a[j] * a[k-j] for j where 0<=j<n and 0<=k-j<n
        # d(a*a)[k] / d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        #                        = a[k-i] (if 0<=k-i<n) + a[(i-k) mod ... ] 
        
        # The gradient vector is: grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])
        # where a[j] = 0 for j outside [0, n-1]
        
        # This can be computed as:
        # sum_k weights[k] * a[k-i] = (weights * a)[reversed] at position related to i
        # Actually it's the cross-correlation of weights and a
        
        # Let's use FFT:
        # cross_corr[i] = sum_k weights[k] * a[(k-i) mod size]  -- but we need bounded indices
        
        # Simpler: use the fact that the gradient of the max constraint is:
        # grad = 2 * (write_convolution of weights with a)
        
        # Let me compute it directly using the Toeplitz structure:
        # For each k, the contribution to grad[i] is weights[k] * (a[k-i] + a[i-k])
        # = weights[k] * a[k-i] (if 0 <= k-i < n) + weights[k] * a[i-k] (if 0 <= i-k < n)
        
        # First term: sum_k weights[k] * a[k-i] for k in [i, i+n-1] (bounded by size)
        # This is (weights convolved with a) evaluated at specific points
        
        # Actually, the simplest correct computation:
        # Let f = weights (length size = 2n-1), g = a (length n)
        # The cross-correlation (f * g)[i] = sum_k f[k] * g[k-i]
        # We want: sum_k f[k] * a[k-i] for valid k-i, plus sum_k f[k] * a[i-k] for valid i-k
        
        # The second sum is: sum_k f[k] * a[i-k] = sum_j f[i-j] * a[j] (substituting j = i-k)
        # = (f * a)[i] (cross-correlation at lag i)
        
        # The first sum: sum_k f[k] * a[k-i] = sum_j f[i+j] * a[j] (substituting j = k-i)
        # = (f * a)[-i] (cross-correlation at lag -i)
        
        # So grad[i] = (f * a)[i] + (f * a)[-i]
        # = 2 * Re{(f * a)[i]} if we think of it in complex terms
        
        # In practice: grad = 2 * (cross-correlation of weights and a, at lags 0 to n-1)
        
        # Cross-correlation via FFT:
        # cross_correlation[i] = sum_k f[k] * conj(g[k-i])  (for real: f[k] * g[k-i])
        # Using FFT: IFFT(FFT(f) * conj(FFT(g)))
        
        F = rfft(weights, fft_size)
        A = rfft(a, fft_size)
        cross_corr = irfft(F * np.conj(A), fft_size)
        # cross_corr[i] = sum_k weights[k] * a[k-i] (circular)
        # For linear: we need to be careful, but with zero-padding it should work
        
        # grad[i] = cross_corr[i] + cross_corr[n-1-i] ... hmm not quite right
        
        # Let me just do it the direct way for correctness:
        # Actually, the gradient of sum_k weights[k] * (a*a)[k] wrt a[i] is:
        # sum_k weights[k] * (d(a*a)[k]/d(a[i]))
        # d(a*a)[k]/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n, i≠k case)
        # Wait, if k-i = i-k (i.e., k=2i), then it's 2*a[0]... no that's wrong.
        
        # Let me be very careful:
        # (a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d/d(a[i]): 
        #   - a[i] appears as a[j] when j=i: contributes a[k-i] (if 0<=k-i<n)
        #   - a[i] appears as a[k-j] when k-j=i, i.e., j=k-i: contributes a[k-(k-i)] = a[i]... 
        #     wait, a[j]*a[k-j], derivative wrt a[i]:
        #     if j=i: a[k-i] (coefficient of a[i] in the product)
        #     if k-j=i, i.e., j=k-i: a[k-i] (the other factor)
        #     if i = k-i (i.e., k=2i): both terms give a[i], so total is 2*a[i]... 
        #     no wait, that's the value, not the derivative.
        
        # The derivative of a[j]*a[k-j] wrt a[i]:
        # = delta(j,i) * a[k-j] + a[j] * delta(k-j, i)
        # = (1 if j=i) * a[k-i] + a[j] * (1 if j=k-i)
        
        # So d(a*a)[k]/d(a[i]) = a[k-i] * (1 if 0<=k-i<n) + a[k-i] * (1 if 0<=k-i<n)
        # Wait no. Let me redo:
        # Sum over j of a[j]*a[k-j]. The derivative wrt a[i] is:
        # For each j: d/d(a[i]) [a[j]*a[k-j]] = delta(j,i)*a[k-j] + a[j]*delta(k-j,i)
        # = delta(j,i)*a[k-j] + a[j]*delta(j,k-i)
        
        # Sum over j: = a[k-i] (from j=i term, if 0<=k-i<n) + a[k-i] (from j=k-i term, if 0<=k-i<n)
        # = 2*a[k-i] (if 0<=k-i<n)
        # But wait, when j=i and j=k-i are the same term (i.e., k=2i), we should only count it once.
        # Actually no, in the sum sum_j a[j]*a[k-j], each j is a separate term. When j=i, we get a[i]*a[k-i].
        # The derivative of a[i]*a[k-i] wrt a[i] is a[k-i] + a[i]*delta(k-i, i)... 
        # No! The product a[j]*a[k-j] for fixed j: if j ≠ k-j, it's a product of two different variables, 
        # and the derivative wrt a[i] is:
        #   (1 if j=i)*a[k-j] + a[j]*(1 if k-j=i)
        # If j = k-j (i.e., k=2j), then it's a[j]^2, and the derivative is 2*a[j]*(1 if j=i).
        
        # So for the full sum:
        # d(a*a)[k]/d(a[i]) = sum_j [delta(j,i)*a[k-j] + a[j]*delta(j,k-i)]
        # = a[k-i] + a[k-i]  (if 0<=k-i<n and k-i≠i)
        # = 2*a[k-i]  (if 0<=k-i<n and k-i≠i, i.e., k≠2i)
        # = a[k-i] + a[k-i] = 2*a[i]  (if k-i=i, i.e., k=2i)
        # = a[k-i]  (if k-i < 0 or k-i >= n, one of the deltas doesn't fire)
        
        # Wait, I think I'm overcomplicating this. Let me just use the simple formula:
        # d(a*a)[k]/d(a[i]) = a[k-i] + a[i-k]  (where a[j]=0 for j outside [0,n-1])
        
        # This is correct because:
        # The convolution (a*a)[k] can be seen as the inner product of a and a shifted by k.
        # The gradient is the "anti-derivative" in convolution, which gives the sum of the two shifts.
        
        # So: grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])  [a[j]=0 outside [0,n-1]]
        
        # This equals 2 * sum_k weights[k] * a[k-i]  [since by symmetry, sum_k w[k]*a[i-k] = sum_k w[k]*a[k-i] 
        # when we account for the range... actually this isn't quite right in general]
        
        # Let me just compute it using the FFT approach:
        # Define: h[i] = sum_k weights[k] * a[k-i] for 0<=i<n
        # This is the "deconvolution" or "cross-correlation" 
        # h = ifft(fft(weights) * conj(fft(a)))  -- standard cross-correlation
        # But we need to handle the linear (non-circular) case with proper zero-padding.
        
        # With fft_size >= size + n - 1 = 3n - 2, the circular equals linear.
        
        F = rfft(weights, fft_size)
        A = rfft(a, fft_size)
        # cross_corr[m] = sum_p F_ext[p] * a_ext[m-p] where extended with zeros
        # = sum_p weights[p] * a[m-p] (linear cross-correlation)
        cross = irfft(F * np.conj(A), fft_size)[:size]
        # cross[m] = sum_p weights[p] * a[m-p] for valid p, m-p
        
        # grad[i] = sum_k weights[k] * a[k-i] + sum_k weights[k] * a[i-k]
        # First term: sum_k weights[k] * a[k-i] = cross[n-1-i]... 
        # Hmm, let me think about the indexing more carefully.
        
        # cross[m] = sum_p weights[p] * a[m-p]
        # For first term: sum_k weights[k] * a[k-i], let m-p = k-i, so m = k-i+p... 
        # This is getting confusing. Let me just use a direct computation for now.
        
        # Direct computation (O(n * len(active_lags))):
        # Find the significant lags (where weights are non-negligible)
        sig_lags = np.where(weights > weights.max() * 0.01)[0]
        
        grad = np.zeros(n)
        for k in sig_lags:
            w = weights[k]
            if w < 1e-15:
                continue
            # Contribution: w * (a[k-i] + a[i-k]) for i in [0, n-1]
            # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
            
            # First part: a[k-i] for i in [max(0, k-n+1), min(n-1, k)]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_hi >= i_lo:
                grad[i_lo:i_hi+1] += w * a[k - np.arange(i_lo, i_hi+1)]
            
            # Second part: a[i-k] for i in [max(0, k), min(n-1, k+n-1)]
            i_lo2 = max(0, k)
            i_hi2 = min(n - 1, k + n - 1)
            if i_hi2 >= i_lo2:
                grad[i_lo2:i_hi2+1] += w * a[np.arange(i_lo2, i_hi2+1) - k]
        
        if grad.max() < 1e-15:
            break
        
        # Negative gradient (we want to decrease the max)
        eta_current = eta * (decay ** it)
        a_new = a - eta_current * grad / grad.max()
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def spectral_shape(a0, time_limit, verbose=True):
    """Spectral peak-shaping: work in frequency domain to reduce time-domain peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Parameters
    alpha = 0.1  # Shaping strength
    no_improve = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT of current sequence
        A = rfft(a, fft_size)
        
        # Power spectrum
        P = A * np.conj(A)  # |A|^2
        
        # Current convolution = IFFT(P)
        conv = irfft(P, fft_size)[:size]
        peak = conv.max()
        
        if peak < 1e-15:
            break
        
        # Shape the convolution: reduce peaks
        # Use a "soft" clipping
        threshold = peak * (1.0 - alpha)
        conv_shaped = conv.copy()
        excess = conv_shaped - threshold
        conv_shaped[excess > 0] = threshold + excess[excess > 0] * 0.5
        
        # Ensure non-negativity
        conv_shaped = np.maximum(conv_shaped, 0.0)
        
        # Back to frequency: need to take "square root" of the spectrum
        # P_new = FFT(conv_shaped)
        # But P_new should be |A_new|^2, so A_new = sqrt(P_new) with appropriate phase
        
        P_new = rfft(conv_shaped, fft_size)
        
        # The issue: P_new is complex, and we need A_new such that |A_new|^2 = P_new
        # This requires P_new to be non-negative real (PSD)
        # For real a, P = |A|^2 is always real and non-negative
        # But conv_shaped might not have a non-negative PSD in general
        
        # Alternative: use the phase of A and modify only the magnitude
        mag_new = np.sqrt(np.maximum(P_new.real, 0))
        # Keep the phase from A
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Reduce alpha
                alpha *= 0.9
                no_improve = 0
        
        a = a_new
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.8f}, alpha={alpha:.3f}")
    
    return best_a, best_score

def lse_gradient_descent_v2(a0, time_limit, verbose=True):
    """LSE (log-sum-exp) smoothing of the max, with gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Temperature stages
    temps = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(200, int(remaining * 2)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # LSE: log(sum(exp(conv/T))) * T ≈ max(conv) for small T
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient of LSE wrt a[i] = sum_k w[k] * d(a*a)[k]/d(a[i])
            # = sum_k w[k] * (a[k-i] + a[i-k])
            # Use FFT for efficiency
            F = rfft(w, fft_size)
            A = rfft(a, fft_size)
            cross = irfft(F * np.conj(A), fft_size)[:size]
            # cross[m] = sum_p w[p] * a[m-p]
            # We want grad[i] = sum_k w[k] * a[k-i] + sum_k w[k] * a[i-k]
            # First term: sum_k w[k] * a[k-i] = cross[n-1-i] ... 
            # Actually let me just use the direct method with significant lags
            
            sig = np.where(w > w.max() * 0.001)[0]
            grad = np.zeros(n)
            for k in sig:
                wv = w[k]
                if wv < 1e-15:
                    continue
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_hi >= i_lo:
                    grad[i_lo:i_hi+1] += wv * a[k - np.arange(i_lo, i_hi+1)]
                i_lo2 = max(0, k)
                i_hi2 = min(n - 1, k + n - 1)
                if i_hi2 >= i_lo2:
                    grad[i_lo2:i_hi2+1] += wv * a[np.arange(i_lo2, i_hi2+1) - k]
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            # Step size with line search
            step = 1.0 / (grad_norm / norm + 1e-10)
            for _ in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a = a_new * norm / s_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
            
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE T={T_val:.1f} it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_v2(a0, time_limit, rng, verbose=True):
    """Improved pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find target j from peak lags
            peak_lags = np.where(conv >= cur_max * 0.85)[0]
            j_candidates = set()
            for kl in peak_lags[:10]:
                j_candidates.add((kl - i) % n)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)
            
            # Also add some random
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in list(j_candidates)[:15]:
                if j == i:
                    continue
                aj = a[j]
                
                # Gain vector for moving delta from i to j
                # New conv = old conv + delta * (contribution of a[j] increase - contribution of a[i] decrease)
                # Contribution of a[j] at position j: adds a * shifted by j
                # Contribution of a[i] at position i: adds a * shifted by i
                
                g = np.zeros(size)
                # +a shifted by j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi-lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # -a shifted by i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3-lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Find optimal delta via golden section or sampling
                # Binary search for the delta that minimizes max(conv + delta*g)
                # conv + delta*g is linear in delta (for the diagonal terms, it's quadratic)
                
                lo_d = 0.0
                hi_d = 0.95 * ai
                
                # Sample-based search
                n_samples = 15
                for _ in range(n_samples):
                    mid_d = (lo_d + hi_d) / 2
                    new_conv = conv + mid_d * g
                    # Diagonal corrections
                    if 2*i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*mid_d + mid_d*mid_d
                    if 2*j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*mid_d + mid_d*mid_d
                    v = new_conv.max()
                    if v < cur_max:
                        lo_d = mid_d
                    else:
                        hi_d = mid_d
                
                best_d = (lo_d + hi_d) / 2
                if best_d > 0.01 * ai:
                    new_conv = conv + best_d * g
                    if 2*i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*best_d + best_d*best_d
                    if 2*j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*best_d + best_d*best_d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = best_d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
