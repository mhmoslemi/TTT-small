def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE Adam with adaptive temperature schedule."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_score_fast(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0
    lr = 0.5 / np.sqrt(n)
    
    # Adaptive temperature: start high, decrease
    T_current = 50.0
    T_min = 0.05
    T_decay = 0.995  # per step
    
    while time.monotonic() - t0 < time_limit:
        t_adam += 1
        T_current = max(T_min, T_current * T_decay)
        
        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        shifted = (conv - max_c) / T_current
        shifted = np.clip(shifted, -80, 80)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        ca = irfft(np.conj(A) * W, fft_size)[:n]
        aw = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (ca + aw)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new
        a = a_new
        
        sc = compute_score_fast(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t_adam % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam t={t_adam} T={T_current:.3f} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and 0 < len(active_set) < size:
        # Build constraint matrix for active set only
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[idx, i] = a[k - i]
        constraints = [T @ b_var <= M]
    else:
        # Full constraint matrix (expensive for large n)
        T = np.zeros((size, n))
        for m in range(n):
            T[m:m + n, m] = a
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP-based Nesterov optimization with improved constraint management."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = compute_score_fast(a)
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    
    # Adaptive step size
    base_alpha = 0.1
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive constraint selection: include lags above 70% of max
        # but cap at a reasonable number
        threshold = M * 0.70
        active_set = np.where(conv_y >= threshold)[0]
        
        # If too many, use a higher threshold
        max_constraints = min(200, size)
        if len(active_set) > max_constraints:
            # Use top-k approach
            top_k = np.argsort(conv_y)[-max_constraints:]
            active_set = top_k
        
        b = lp_best_response(y, M, solver, active_set)
        
        if b is None:
            b = lp_best_response(y, M, solver, None)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov acceleration with adaptive step
        alpha = base_alpha / (1.0 + 0.01 * t)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 1.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc = compute_score_fast(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        sc_y = compute_score_fast(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.4f}")
    
    return best_a, best_score

def targeted_peak_reduction(a0, time_limit, rng, verbose=True):
    """Directly target the peak locations and reduce them via local adjustments."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_score_fast(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 2).bit_length()
    
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_lags = np.where(conv >= cur_max * 0.95)[0]
        
        improved = False
        
        for lag in peak_lags[:5]:  # Focus on the top peaks
            if time.monotonic() - t0 > time_limit:
                break
            
            # Get the pairs contributing to this lag
            # c_lag = sum_i a[i] * a[lag-i] for valid i
            lo = max(0, lag - n + 1)
            hi = min(n, lag + 1)
            
            # For each contributing pair, try reducing the product
            # by transferring mass from the larger to the smaller (spreads out the contribution)
            for i in range(lo, hi):
                j = lag - i
                if j < 0 or j >= n or j == i:
                    continue
                
                ai, aj = a[i], a[j]
                if ai < 1e-10 or aj < 1e-10:
                    continue
                
                # Try moving a small amount from the larger to a random other position
                src = i if ai > aj else j
                dst_candidates = rng.choice(n, size=min(10, n), replace=False)
                dst_candidates = dst_candidates[dst_candidates != src]
                
                delta = min(ai, aj) * 0.1
                for dst in dst_candidates[:5]:
                    if dst == src:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[dst] += delta
                    if a_try[src] < 0:
                        continue
                    a_try = a_try * norm / a_try.sum()
                    sc_try = compute_score_fast(a_try)
                    if sc_try < best_score * (1 - 1e-5):
                        a = a_try
                        best_score = sc_try
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Perturb
            n_perturb = max(5, n // 50)
            idx_p = rng.choice(n, size=n_perturb, replace=False)
            a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
            a = np.maximum(a, 0.0)
            a = a * norm / a.sum()
        else:
            no_improve = 0
        
        if verbose and no_improve % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakReduction ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with gradient-informed candidate selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_score_fast(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            peak_lags = np.where(conv >= cur_max * 0.80)[0]
            
            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i or j >= n:
                    continue
                aj = a[j]
                
                # Compute the change in convolution when moving delta from i to j
                g = np.zeros(size)
                
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                d_max = ai * 0.5
                n_samples = 15
                deltas = np.linspace(0.02, d_max, n_samples)
                
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max / (a.sum()**2) < best_score / (2*n) - 1e-12:
                        best_score = compute_score_fast(a)
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                n_perturb = max(10, n // 40)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                if no_improve >= 20:
                    break
        else:
            no_improve = 0
        
        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def edge_weighted_init(n, rng):
    """Initialize with larger values at edges to combat triangular convolution envelope."""
    x = np.linspace(0, 1, n)
    # Inverted envelope: large at edges, small in middle
    a = 1.0 + 0.5 * (2 * x - 1)**2  # Parabolic, larger at edges
    # Add some sinusoidal structure
    f1 = rng.integers(2, 8)
    f2 = rng.integers(8, 20)
    a += 0.2 * np.cos(2 * np.pi * f1 * x)
    a += 0.1 * np.cos(2 * np.pi * f2 * x)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    a += 0.005 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a
