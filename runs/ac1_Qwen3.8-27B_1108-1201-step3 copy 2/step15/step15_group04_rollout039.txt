"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP best-response with Nesterov acceleration (scipy HiGHS solver)
2. Adaptive constraint screening for LP speed
3. Dual-guided subgradient refinement

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

_fft_cache = {}

def get_fft_size(size):
    if size not in _fft_cache:
        _fft_cache[size] = 1 << (size - 1).bit_length()
    return _fft_cache[size]

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fs = get_fft_size(size)
    A = rfft(a, fs)
    return irfft(A * A, fs)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fs = get_fft_size(size)
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A * B, fs)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_conv_matrix(a, active=None):
    """Build T such that T @ b = conv(a, b), optionally only for active rows."""
    n = len(a)
    size = 2 * n - 1
    if active is not None:
        T = np.zeros((len(active), n))
        for col_idx, row_idx in enumerate(active):
            lo = max(0, row_idx - n + 1)
            hi = min(n, row_idx + 1)
            T[col_idx, lo:hi] = a[lo:hi][::-1]
    else:
        T = np.zeros((size, n))
        for i in range(n):
            T[i:i+n, i] = a
    return T

def lp_best_response(a, M, active=None, time_limit_lp=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    T = build_conv_matrix(a, active)
    c = -np.ones(n)
    b_ub = np.full(T.shape[0], M)
    
    t_start = time.monotonic()
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
    except Exception:
        return None, None
    
    if time.monotonic() - t_start > time_limit_lp:
        return None, None
    
    if res.status == 0:
        b = np.maximum(res.x, 0)
        duals = None
        try:
            if hasattr(res, 'ineqlin') and res.ineqlin is not None and res.ineqlin.marginals is not None:
                duals = np.maximum(res.ineqlin.marginals, 0)
                if active is not None:
                    full_duals = np.zeros(size)
                    full_duals[active] = duals
                    duals = full_duals
        except Exception:
            pass
        return b, duals
    return None, None

def nesterov_lp_optimize(a0, time_limit, verbose=True, use_screening=True):
    """Main optimization: Nesterov-accelerated LP best-response with adaptive screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    y = a.copy()
    no_improve = 0
    iteration = 0
    lp_success_count = 0
    avg_b = np.zeros(n)
    n_avg = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed = time.monotonic() - t0
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-15:
            break
        
        # Adaptive constraint screening
        if use_screening and iteration > 20:
            if iteration < 100:
                threshold = 0.5
            elif iteration < 500:
                threshold = 0.7
            else:
                threshold = 0.85
            active = np.where(conv_y >= M * threshold)[0]
            if len(active) < 20:
                active = np.where(conv_y >= M * 0.3)[0]
        else:
            active = None
        
        b, duals = lp_best_response(y, M, active)
        
        if b is None or b.sum() < 1e-15:
            # Fallback: subgradient step
            k_star = int(np.argmax(conv_y))
            sg = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    sg[i] = 2.0 * y[j]
            if k_star % 2 == 0:
                mid = k_star // 2
                if mid < n:
                    sg[mid] -= y[mid]
            
            sg_norm = np.linalg.norm(sg)
            if sg_norm > 1e-15:
                step = 0.05 * norm / sg_norm
                a_new = a - step * sg
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm / s_new
            else:
                a_new = a.copy()
        else:
            lp_success_count += 1
            b_sum = b.sum()
            g = b * norm / b_sum
            avg_b += g
            n_avg += 1
            
            # Check for constraint violations if we used screening
            if active is not None and len(active) < size:
                check_conv = fast_conv_cross(y, g)
                violated = np.where(check_conv > M * (1 + 1e-3))[0]
                if len(violated) > 0:
                    b_full, _ = lp_best_response(y, M, None)
                    if b_full is not None and b_full.sum() > 1e-15:
                        g = b_full * norm / b_full.sum()
            
            a_new = g
        
        # Nesterov update
        alpha = 2.0 / (iteration + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if iteration == 1:
            a_new = a_new.copy()
        else:
            t_val = float(iteration)
            momentum = t_val / (t_val + 2.0)
            a_nest = y_new + momentum * (y_new - y)
            a_nest = np.maximum(a_nest, 0)
            s_nest = a_nest.sum()
            if s_nest < 1e-15:
                a_nest = y_new.copy()
            else:
                a_nest = a_nest * norm / s_nest
            # Use the Nesterov point if it's better
            sc_nest = fast_conv_self(a_nest).max()
            sc_plain = fast_conv_self(a_new).max()
            a_new = a_nest if sc_nest <= sc_plain else a_new
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        
        # Periodic averaging check
        if n_avg > 0 and n_avg % 200 == 0:
            avg_check = avg_b / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
                    improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
        
        # Perturbation to escape local optima
        if no_improve in [80, 160, 320, 640]:
            rng = np.random.default_rng(iteration * 7 + 3)
            n_pert = max(5, n // 40)
            idx = rng.choice(n, size=n_pert, replace=False)
            scale = 0.05 + 0.15 * min(1.0, no_improve / 400.0)
            y[idx] *= (1.0 + scale * rng.standard_normal(n_pert))
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            a = y.copy()
            no_improve = 0
        
        # Reset LP failure tracking
        if no_improve > 500 and lp_success_count < 10:
            break
        
        if verbose and iteration % 200 == 0:
            print(f"    LP-Nest t={iteration} ({elapsed:.0f}s) lp_ok={lp_success_count}: best={best_score:.8f}")
        
        if no_improve > 2000:
            break
    
    return best_a, best_score

def subgradient_refine(a0, time_limit, verbose=False):
    """Direct subgradient method on the max-convolution objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    lr = 0.1
    momentum = 0.7
    v = np.zeros(n)
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad of f_{k*} at a
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = 2.0 * a[j]
        if k_star % 2 == 0:
            mid = k_star // 2
            if mid < n:
                sg[mid] -= a[mid]
        
        # Normalized step
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        # Polyak-like step size
        lr_t = lr / (1.0 + 0.01 * it)
        step = lr_t * norm / sg_norm
        
        v = momentum * v - step * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            # Reduce learning rate on improvement
            lr *= 0.95
        
        a = a_new
        
        if verbose and it % 500 == 0:
            print(f"    SubGrad it={it}: best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Coordinate descent: move mass between pairs to reduce peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Candidate j's: peak lags + random
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            peak_lags = np.where(conv >= cur_max * 0.85)[0]
            for kl in peak_lags[:6]:
                j1 = (kl - i) % n
                j2 = i + int(kl)
                if j1 < n:
                    j_candidates.add(j1)
                if j2 < n:
                    j_candidates.add(j2)
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gain vector (approximate)
                g = np.zeros(size)
                # Adding d to position j: conv increases by d * a shifted by j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Removing d from position i: conv decreases by d * a shifted by i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Find optimal delta via sampling
                n_samples = 15
                deltas = np.linspace(0.01, 0.8, n_samples) * ai
                for d in deltas:
                    new_conv = conv + d * g
                    # Quadratic corrections
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_val < cur_max - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_pert = max(5, n // 30)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lse_gradient_descent(a0, time_limit):
    """Log-sum-exp smoothing gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fs = get_fft_size(size)
    
    n_stages = 10
    temps = np.geomspace(50.0, 0.5, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fs)
            A = rfft(a, fs)
            g1 = irfft(np.conj(A) * W, fs)[:n]
            aw = irfft(A * W, fs)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)
            
            # Line search
            for trial in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        # Asymmetric ramp with oscillation
        a = (1.0 + 0.5 * x) * (1.0 + 0.2 * np.sin(8 * np.pi * x))
        a = np.maximum(a, 0.1)
    else:
        # Power-law decay from left
        a = (1.0 - x) ** 0.5 + 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: finds the best sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # Phase 0: Quick LSE warm-up
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = min(remaining * 0.04, 60.0)
    if t0_phase > 15:
        print(f"\n=== Phase 0: LSE warmup (budget={t0_phase:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, t0_phase)
        try_update(a_out, "lse_warmup")
    
    # Phase 1: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 1: LP-Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, verbose=True, use_screening=True)
        try_update(a_out, f"lp1_n{n1}")
    
    # Phase 2: Subgradient refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Subgradient n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = subgradient_refine(best_seq, t2, verbose=False)
        try_update(a_out, f"subgrad_n{n2}")
    
    # Phase 3: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: LP-Nesterov R2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, verbose=True, use_screening=True)
        try_update(a_out, f"lp2_n{n3}")
    
    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: Pair CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{n4}")
    
    # Phase 5: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 5: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.4, 180.0)
        a_out, _ = nesterov_lp_optimize(init_long, t_long, verbose=True, use_screening=True)
        try_update(a_out, f"lp_long_{n_long}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_long:
            t_long_cd = min(remaining * 0.4, 80.0)
            a_out2, _ = pair_cd_optimize(best_seq, t_long_cd, rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")
    
    # Phase 6: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        t_f1 = min(remaining * 0.3, 60.0)
        a_out, _ = lse_gradient_descent(best_seq, t_f1)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_f2 = min(remaining * 0.4, 80.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t_f2, verbose=False, use_screening=True)
            try_update(a_out, "lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_f3 = min(remaining * 0.5, 50.0)
            a_out, _ = pair_cd_optimize(best_seq, t_f3, rng, verbose=False)
            try_update(a_out, "pcd_final")
    
    # Phase 7: One more LP round if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 7: Extra LP (remaining={remaining:.0f}s) ===")
        t_extra = min(remaining * 0.5, 120.0)
        a_out, _ = nesterov_lp_optimize(best_seq, t_extra, verbose=False, use_screening=True)
        try_update(a_out, "lp_extra")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
