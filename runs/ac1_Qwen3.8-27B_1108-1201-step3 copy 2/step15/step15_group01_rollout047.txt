def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def compute_smooth_grad(a, w):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def lp_best_response(a, M, solver, active_indices=None):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    if active_indices is not None and len(active_indices) > 0:
        T_use = T[active_indices, :]
    else:
        T_use = T
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 5000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        conv_y = fast_autocorr(y)
        # Adaptive constraint selection: use top fraction
        frac = max(0.4, min(1.0, 200.0 / len(conv_y)))
        k_constraints = max(50, int(frac * len(conv_y)))
        top_idx = np.argsort(conv_y)[::-1][:k_constraints]
        active = top_idx

        b = lp_best_response(y, M_curr, solver, active_only=active)
        if b is None:
            b = lp_best_response(y, M_curr, solver, active_only=None)
            if b is None:
                stuck_count += 1
                if stuck_count > 50:
                    break
                continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 15 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Occasionally try a convex combination with uniform

                s_mix = mixed.sum()
                if s_mix > 1e-15:
                    mixed = mixed * norm / s_mix

        if verbose and n_iters % 30 == 0:

def smooth_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temps = np.array([30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001])

    for T in temps:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 0.3:
            break
        n_steps = max(20, min(800, int(remaining * 2.5)))

        y = a.copy()
        for it in range(n_steps):
            elapsed = time.monotonic() - t0
            if elapsed > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = compute_smooth_grad(y, w)

            step = norm / (n * T * (1.0 + 0.02 * it))
            step = min(step, norm / (n * 0.3))

            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Smooth T={T:.3f} done ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, window=50, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-8:
                continue

            for offset in range(1, window + 1):
                j = (i + offset) % n
                if j == i or a[i] < norm * 1e-8:
                    continue

                # Binary search for best transfer amount
                lo, hi = 0.0, a[i] * 0.8
                best_d = 0.0
                best_val = best_score
                for _ in range(6):
                    mid = (lo + hi) / 2
                    if mid <= 1e-15:
                        lo = mid
                        continue
                    a_t = a.copy()
                    a_t[i] -= mid
                    a_t[j] += mid
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        break
                    a_t = a_t * norm / s_t
                    conv_t = fast_autocorr(a_t)
                    if conv_t.max() < best_val - 1e-13:
                        best_val = conv_t.max()
                        best_d = mid
                        lo = mid
                    else:
                        hi = mid

                if best_d > 1e-15:
                    a[i] -= best_d
                    a[j] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    if conv.max() < best_score - 1e-13:
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def post_process(a, rng):
    """Aggressive post-processing: zeroing, symmetrization, local swaps."""
    n = len(a)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    norm = np.sqrt(2.0 * n)

    # Try zeroing elements below various thresholds
    for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
        threshold = frac * a.max()
        a_zeroed = a.copy()
        a_zeroed[a_zeroed < threshold] = 0.0
        s = a_zeroed.sum()
        if s < 1e-15:
            continue
        a_zeroed = a_zeroed * norm / s
        sc = fast_autocorr(a_zeroed).max()
        if sc < best_score:
            best_score = sc
            best_a = a_zeroed.copy()

    # Try symmetrization
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym * norm / s
        sc = fast_autocorr(a_sym).max()
        if sc < best_score:
            best_score = sc
            best_a = a_sym.copy()

    # Try block averaging (smooth nearby elements)
    for block_size in [2, 3, 5]:
        a_block = a.copy()
        for i in range(0, n - block_size + 1, block_size):
            chunk = a_block[i:i + block_size]
            a_block[i:i + block_size] = chunk.mean()
        s = a_block.sum()
        if s > 1e-15:
            a_block = a_block * norm / s
            sc = fast_autocorr(a_block).max()
            if sc < best_score:
                best_score = sc
                best_a = a_block.copy()

    # Quick local pair swaps on best
    t_start = time.monotonic()
    for _ in range(50):
        if time.monotonic() - t_start > 5:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or best_a[i] < 1e-10:
            continue
        for frac in [0.05, 0.1, 0.2, 0.5]:
            d = best_a[i] * frac
            a_t = best_a.copy()
            a_t[i] -= d
            a_t[j] += d
            s = a_t.sum()
            if s < 1e-15:
                break
            a_t = a_t * norm / s
            sc = fast_autocorr(a_t).max()
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_t.copy()
                break

    return best_a, best_score

        structures.append(np.exp(-4.0 * (xc - 0.3) ** 2) + np.exp(-4.0 * (xc + 0.3) ** 2))
        # Ramp
        structures.append(np.linspace(0.5, 1.5, n_c))
        # Comb (every other)
        comb = np.ones(n_c)
        comb[::3] = 0.3
        structures.append(comb)

        for s_idx, struct in enumerate(structures):
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 15:
                break
            t_sub = min(remaining_sub * 0.3, 40.0)
            # Mix with current best
            mix_frac = 0.5
            a_mix = (1.0 - mix_frac) * best_seq * np.sqrt(2.0 * n_c) / best_seq.sum() + \
                    mix_frac * struct * np.sqrt(2.0 * n_c) / struct.sum()
            a_mix = np.maximum(a_mix, 0)
            a_mix = a_mix * np.sqrt(2.0 * n_c) / a_mix.sum()

            # Quick smooth optimization
            a_out, _ = smooth_optimize(a_mix, t_sub, verbose=False)
            do_try_update(a_out, f"restart_{s_idx}")

            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 10 or best_seq is None:
                break
            # Quick LP
            t_lp = min(remaining_sub * 0.4, 30.0)
            if t_lp > 10:
                a_norm_lp = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
                a_out_lp, _ = nesterov_lp(a_norm_lp, t_lp, solver, verbose=False)
                do_try_update(a_out_lp, f"lp_restart_{s_idx}")

    # === Final post-processing ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10 and best_seq is not None:
        n_f = len(best_seq)
        a_final_norm = best_seq * np.sqrt(2.0 * n_f) / best_seq.sum()
        a_final, _ = post_process(a_final_norm, rng)
        do_try_update(a_final, "final_pp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
