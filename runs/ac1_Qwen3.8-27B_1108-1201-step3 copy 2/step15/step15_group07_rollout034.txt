import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

_global_best = None


def fast_autocorr(a):
    """Compute linear autocorrelation of a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} via bisection on lambda."""
    n = len(v)
    if n == 0:
        return v
    lo = v.min() - S
    hi = v.max()
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.maximum(v - mid, 0.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.maximum(v - lam, 0.0)
    s = result.sum()
    if s > 1e-12:
        result = result * S / s
    else:
        result = np.full(n, S / n)
    return result


def build_conv_matrix(a):
    """Build convolution matrix T: (T @ b)[k] = (a * b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_best_response_lp(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def subgrad_fista(a0, time_limit, verbose=True):
    """FISTA with hard subgradient of max autocorrelation."""
    global _global_best
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t = 1.0
    a_prev = a.copy()
    t0 = time.monotonic()
    step_scale = 1.0
    patience = 0
    rng = np.random.default_rng(42)

    for it in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        momentum = (t_prev - 1.0) / t
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y * S / s_y

        # Compute subgradient
        autocorr = fast_autocorr(y)
        k_star = int(np.argmax(autocorr))
        # g_i = 2 * y[k_star - i] if 0 <= k_star - i < n
        g = np.zeros(n)
        i_lo = max(0, k_star - n + 1)
        i_hi = min(k_star, n - 1)
        if i_hi >= i_lo:
            idx = np.arange(i_lo, i_hi + 1)
            g[idx] = 2.0 * y[k_star - idx]

        # Adaptive step size
        step = step_scale / np.sqrt(it + 1.0)
        # Scale step appropriately: g has magnitude ~ S, so step should be ~ S^2/||g||^2
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            step = min(step * S / g_norm, S * 0.5 / (np.linalg.norm(y) + 1e-10))

        z = y - step * g
        a_new = project_simplex(z, S)

        a_prev = a
        a = a_new

        sc = fast_autocorr(a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            patience = 0
            step_scale = min(step_scale * 1.05, 10.0)
        else:
            patience += 1
            if patience % 500 == 0:
                step_scale *= 0.7

        # Restart if stuck
        if patience > 3000:
            a = best_a + rng.normal(0, S * 0.01, n)
            a = np.maximum(a, 0.0)
            a = a * S / a.sum()
            a_prev = a.copy()
            t = 1.0
            patience = 0
            step_scale = 1.0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SF iter={it} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    if _global_best is None or best_score < _global_best[1]:
        _global_best = (best_a.copy(), best_score)

    # Initialize temperature
    autocorr_init = fast_autocorr(a)
    mu = max((autocorr_init.max() - autocorr_init.min()) / 20.0, S * 0.01)
    mu_min = S * 0.001

    t = 1.0
    a_prev = a.copy()
    t0 = time.monotonic()
    patience = 0

        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y * S / s_y

        autocorr = fast_autocorr(y)

        w_sum = w.sum()
        if w_sum < 1e-15:
