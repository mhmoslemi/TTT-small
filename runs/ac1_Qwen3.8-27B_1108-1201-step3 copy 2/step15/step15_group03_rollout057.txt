```python
import time

def fast_conv_cross(a, b):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def normalize_a(a, norm):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), norm / len(a))
    return a * norm / s

def build_conv_matrix(a, n):
    """Build the matrix T such that T @ b = conv(a, b)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

def lp_best_response(y, M, solver, active_indices=None):
    """Solve max sum(b) s.t. conv(y,b) <= M, b >= 0."""
    n = len(y)
    size = 2 * n - 1
    T = build_conv_matrix(y, n)
    b_var = cp.Variable(n, nonneg=True)
    
    if active_indices is not None and 0 < len(active_indices) < size:
        constraints = [T[active_indices, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

                # where h_m[k] = a[k-m] + a[m-k] (valid indices)
                h_i = np.zeros(size)
                h_j = np.zeros(size)
                for k in range(size):
                    if 0 <= k - i < n:
                        h_i[k] += a[k - i]
                    if 0 <= i - k < n:
                        h_i[k] += a[i - k]
                    if 0 <= k - j < n:
                        h_j[k] += a[k - j]
                    if 0 <= j - k < n:
                        h_j[k] += a[j - k]
                
                gain = h_j - h_i
                
                # For small δ, max(conv + δ*gain) is approximately:
                # either stays at cur_max (if the max coordinate isn't affected much)
                # or changes linearly
                # Try a few δ values
                best_delta = 0.0
                best_val = cur_max
                for frac in [0.1, 0.25, 0.5, 0.75]:
                    d = ai * frac
                    new_conv = conv + d * gain
                    # Correction for diagonal terms
                    new_conv[2*i] += d*d - 2*ai*d
                    new_conv[2*j] += d*d + 2*aj*d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                
                if best_delta > 0:
                    a[i] -= best_delta
                    a[j] += best_delta
                    a = normalize_a(a, norm)
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    conv = fast_conv_self(a)
                    cur_max = conv.max()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                # Perturb to escape local minimum
                n_pert = max(5, n // 50)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.08 * rng.standard_normal(n_pert))
                a = normalize_a(a, norm)
        else:
            no_improve = 0
    
    return best_a, best_score
```

Hmm, I realize the coord_descent_phase has a very expensive inner loop (computing h_i and h_j for each pair). Let me optimize this.

Actually, the gain vector for transferring δ from i to j is:
gain[k] = (a[k-j] if 0<=k-j<n else 0) + (a[j-k] if 0<=j-k<n else 0) - (a[k-i] if 0<=k-i<n else 0) - (a[i-k] if 0<=i-k<n else 0)

This is essentially:
gain = shift(a, -j) + reverse_shift(a, j) - shift(a, -i) - reverse_shift(a, i)

where shift(a, k)[m] = a[m-k] if valid, and reverse_shift(a, k)[m] = a[k-m] if valid.

I can compute this more efficiently using the fact that it's just shifted versions of a:
gain[k] = a[k-j]+a[j-k] - a[k-i]-a[i-k] (with zero-padding)

This can be computed as a

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (log-sum-exp) with momentum
2. LP best-response with Nesterov acceleration and time-average tracking
3. Vectorized pair coordinate descent polish
4. Multi-length exploration
Key innovation: tracking the time-average of LP iterates for better convergence.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize_a(a, norm):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), norm / len(a))
    return a * norm / s


def build_conv_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_best_response(y, M, solver, active_indices=None):
    n = len(y)
    size = 2 * n - 1
    T = build_conv_matrix(y, n)
    b_var = cp.Variable(n, nonneg=True)
    if active_indices is not None and 0 < len(active_indices) < size:
        constraints = [T[active_indices, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def smooth_gd_phase(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()

    momentum = np.zeros(n)
    beta = 0.9
    base_lr = norm / (2.0 * n)

    n_stages = 10
    temps = np.geomspace(60.0, 0.3, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(50, int(remaining * 100))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -100.0, 100.0)
            w = np.exp(shifted)
            w = w / w.sum()

            A = rfft(a, fft_size)
            W = rfft(w, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            lr = base_lr / (1.0 + 0.05 * T_val) * (1.0 / (1.0 + 0.01 * it))
            momentum = beta * momentum - lr * grad
            a_new = a + momentum
            a = normalize_a(a_new, norm)

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    SmoothGD stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score


def lp_nesterov_phase(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    avg_a = np.zeros(n)
    avg_count = 0
    best_avg_a = a.copy()
    best_avg_score = best_score
    last_avg_check = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        threshold_frac = max(0.3, 0.6 - 0.0001 * t)
        n_active = max(50, int(threshold_frac * size))
        sorted_conv = np.sort(conv_y)[::-1]
        threshold = sorted_conv[min(n_active, size - 1)]
        active = np.where(conv_y >= threshold * 0.99)[0]

        b = lp_best_response(y, M, solver, active)
        if b is None:
            b = lp_best_response(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints occasionally
        if len(active) < size and t % 200 == 0:
            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * 1.01:
                b_full = lp_best_response(y, M, solver, None)
                if b_full is not None:
                    b_sum2 = b_full.sum()
                    if b_sum2 > 1e-15:
                        g = b_full * norm / b_sum2

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = normalize_a(y_new, norm)

        momentum = (t - 1.0) / (t + 2.0)
        x_new = y_new + momentum * (y_new - y)
        x_new = normalize_a(x_new, norm)

        # Time-average
        avg_a += x_new
        avg_count += 1

        y = y_new

        sc_x = fast_conv_self(x_new).max()
        if sc_x < best_score:
            best_score = sc_x
            best_a = x_new.copy()

        # Check average every 200 iterations
        if t - last_avg_check >= 200:
            avg = avg_a / avg_count
            sc_avg = fast_conv_self(avg).max()
            if sc_avg < best_avg_score:
                best_avg_score = sc_avg
                best_avg_a = avg.copy()
            last_avg_check = t

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): ind={best_score:.8f}, avg={best_avg_score:.8f}")

    # Final comparison
    if best_avg_score < best_score:
        best_a = best_avg_a.copy()
        best_score = best_avg_score

    return best_a, best_score


def coord_descent_phase(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-10 * norm:
                continue

            peak_lags = np.where(conv >= cur_max * 0.93)[0]
            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j_candidates.add((kl - i + n) % n)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)

            for j in list(j_candidates)[:12]:
                if j == i or j < 0 or j >= n:
                    continue
                aj = a[j]
                if aj < 1e-15 and ai < 1e-15:
                    continue

                # Compute gain vector: change in conv per unit transfer from i to j
                # gain[k] = a[k-j]+a[j-k] - a[k-i]+a[i-k] (with bounds)
                gain = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    gain[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    gain[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    gain[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    gain[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                best_delta = 0.0
                best_val = cur_max
                for frac in [0.15, 0.3, 0.5, 0.7, 0.9]:
                    d = ai * frac
                    if d < 1e-14:
                        continue
                    new_conv = conv + d * gain
                    new_conv[2 * i] += d * d - 2.0 * ai * d
                    new_conv[2 * j] += d * d + 2.0 * aj * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d

                if best_delta > 0:
                    a[i] -= best_delta
                    a[j] += best_delta
                    a = normalize_a(a, norm)
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    conv = fast_conv_self(a)
                    cur_max = conv.max()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 2:
                n_pert = max(5, n // 40)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = normalize_a(a, norm)
        else:
            no_improve = 0

        if verbose and (idx + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.40, 650.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_phase(best_seq, t2, solver)
        try_update(a_out, "lp_nest")

    # Phase 3: Coord descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 3: Coord Descent n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = coord_descent_phase(best_seq, t3, rng)
        try_update(a_out, "cd1")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.17, 250.0)
    print(f"\n=== Phase 4: LP Round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lp_nesterov_phase(best_seq, t4, solver)
        try_update(a_out, "lp_r2")

    # Phase 5: Coord descent round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: CD Round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = coord_descent_phase(best_seq, t5, rng)
        try_update(a_out, "cd2")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)
    print(f"\n=== Phase 6: Multi-length (budget={remaining:.0f}s) ===")
    if remaining > 80:
        for delta in [300, -300, 500]:
            n_try = n_base + delta
            if n_try < 100 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t_sub = min(remaining * 0.35, 150.0)
            if t_sub < 15:
                continue
            print(f"  Trying n={n_try} (budget={t_sub:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lp_nesterov_phase(init_try, t_sub, solver, verbose=False)
            try_update(a_out, f"lp_n{n_try}")

    # Final polish with remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Final polish (budget={remaining:.0f}s) ===")
        t_fin = min(remaining * 0.6, 60.0)
        if t_fin > 10:
            a_out, _ = coord_descent_phase(best_seq, t_fin, rng, verbose=False)
            try_update(a_out, "final_cd")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_wrapper(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
