    n = len(a)
    size = 2 * n - 1
    T = build_conv_tight(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def compute_subgrad(a, active_set):
    """Compute subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    size = 2 * n - 1
    grad = np.zeros(n)
    for k in active_set:
        k = int(k)
        # grad[j] += 2*a[k-j] if 0 <= k-j < n
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            # a[k-j] for j in [j_lo, j_hi]
            # k-j ranges from k-j_lo to k-j_hi
            idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] += 2.0 * a[idx]
    return grad


def freq_perturb(a, rng, strength=0.1):
    """
    Frequency-domain perturbation that attacks the convolution peak.
    Reduces the spectral components that contribute most to the peak.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    
    A = np.fft.fft(a_padded)
    
    # Compute C = A^2 (this is the spectrum of the convolution)
    C = A * A
    # c = IFFT(C) is the convolution
    # Find the peak position
    c = np.fft.ifft(C).real[:size]
    peak_k = np.argmax(c)
    
    # For each frequency bin j, compute its contribution to c[peak_k]
    # c[peak_k] = (1/fft_size) * sum_j C[j] * exp(2*pi*i*j*peak_k/fft_size)
    # We want to reduce C[j] where Re(C[j] * exp(2*pi*i*j*peak_k/fft_size)) > 0
    
    phase = np.exp(2j * np.pi * np.arange(fft_size) * peak_k / fft_size)
    contrib = np.real(C * phase)
    
    # Reduce the bins with positive contribution
    reduction_mask = contrib > 0
    n_reduce = np.sum(reduction_mask)
    if n_reduce < 1:
        return a.copy()
    
    # Apply reduction
    C_new = C.copy()
    # Reduce by a fraction proportional to the contribution
    for j in range(fft_size):
        if reduction_mask[j]:
            C_new[j] *= (1.0 - strength * 0.5)
    
    # Restore conjugate symmetry (should be automatic since C = A^2 and A has conjugate symmetry)
    # But numerical errors might break it, so enforce:
    C_new[0] = np.real(C_new[0])
    for j in range(1, (fft_size + 1) // 2):
        C_new[fft_size - j] = np.conj(C_new[j])
    
    # Recover A_new from C_new: A_new = sqrt(C_new)
    # C_new = A_new^2, so A_new[j] = sqrt(C_new[j])
    # sqrt of complex: magnitude = sqrt(|C|), phase = angle(C)/2
    A_new = np.zeros(fft_size, dtype=complex)
    for j in range(fft_size):
        mag = np.sqrt(np.abs(C_new[j]) + 1e-30)
        ang = np.angle(C_new[j]) / 2.0
        A_new[j] = mag * np.exp(1j * ang)
    
    # Ensure conjugate symmetry of A_new
    A_new[0] = np.real(A_new[0])
    for j in range(1, (fft_size + 1) // 2):
        A_new[fft_size - j] = np.conj(A_new[j])
    
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * a.sum() / s
    return a_new


def freq_smooth_perturb(a, rng, n_bands=5, strength=0.3):
    """
    Alternative frequency perturbation: randomly reduce a few frequency bands.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    
    # Pick random bands to reduce
    n_bins = len(A)
    bands = rng.choice(n_bins - 1, size=min(n_bands, n_bins - 1), replace=False) + 1
    
    for b in bands:
        # Reduce a small window around bin b
        lo = max(1, b - 2)
        hi = min(n_bins, b + 3)
        A[lo:hi] *= (1.0 - strength * rng.uniform(0.3, 1.0))
    
    A[0] = A[0] * (a.sum() / max(np.sum(np.fft.irfft(A, fft_size)[:n].clip(0)), 1e-15))
    
    a_new = np.fft.irfft(A, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * a.sum() / s
    return a_new


def nesterov_lp(a0, time_limit, solver, verbose=True, freq_perturb_every=300):
    """LP best-response with Nesterov acceleration and periodic frequency perturbation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0  # normalize to sum=1
    
    best_a = a.copy()
    best_max = fast_conv(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    y = a.copy()
    momentum_t = 1.0
    patience = 0
    restart_count = 0
    n_iters = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        M_curr = fast_conv(y).max()
        if M_curr < 1e-14:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue
        
        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum  # normalize to sum=1
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        momentum_t += 1.0
        
        # Check improvement
        for candidate in [y, a]:
            sc = fast_conv(candidate).max()
            if sc < best_max - 1e-13:
                best_max = sc
                best_a = candidate.copy()
                patience = 0
        
        # Periodic frequency perturbation to escape local minima
        if patience > freq_perturb_every and n_iters % freq_perturb_every // 3 == 0:
            a_pert = freq_smooth_perturb(best_a, rng, n_bands=3, strength=0.2)
            a_pert = a_pert / a_pert.sum()
            sc_pert = fast_conv(a_pert).max()
            # Even if perturbation makes it worse, use it as new starting point
            y = a_pert.copy()
            a = a_pert.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP freq-perturb {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_max:.8f}")
        
        # Standard restart
        if patience > 200 and restart_count > 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_max:.8f}")
        
        if patience >= 0:
            patience += 1 if fast_conv(y).max() > best_max + 1e-13 else 0
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_max:.8f}, restarts={restart_count}")
    
    return best_a, best_max


def subgrad_optimize(a0, time_limit, verbose=True):
    """Subgradient descent with adaptive step size and frequency perturbation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_max = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    eta = 0.1 / np.sqrt(n)
    patience = 0
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        max_val = conv.max()
        
        # Active set: top-k lags
        k_active = min(20, len(conv))
        active_idx = np.argsort(conv)[-k_active:]
        # Weight by how close to max
        weights = np.exp(-(conv[active_idx] - max_val) * 50.0)  # sharpen
        weights = weights / weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active_idx):
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi+1] += 2.0 * w * a[idx]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Mirror descent (multiplicative weights)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * grad
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new / a_new.sum()
        
        # Line search: try a few step sizes
        best_sc = max_val
        best_a_try = a
        for factor in [0.5, 1.0, 2.0, 5.0]:
            eta_try = eta * factor
            la = log_a - eta_try * grad
            at = np.exp(la - la.max())
            at = at / at.sum()
            sc = fast_conv(at).max()
            if sc < best_sc:
                best_sc = sc
                best_a_try = at
                eta = eta_try
        
        a = best_a_try
        n_iters += 1
        
        if best_sc < best_max - 1e-12:
            best_max = best_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 200 == 0:
                eta *= 0.8
        
        # Frequency perturbation when stuck
        if patience > 500 and n_iters % 100 == 0:
            a_pert = freq_smooth_perturb(a, rng, n_bands=5, strength=0.3)
            a = a_pert
            patience = 0
            eta = 0.1 / np.sqrt(n)
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={n_iters} ({elapsed:.0f}s): best={best_max:.8f}, eta={eta:.6f}")
    
    return best_a, best_max


def cd_polish(a0, time_limit, verbose=False):
    """Pairwise coordinate descent for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv(a).max()
    a = a / s
    
    best_a = a.copy()
    conv = fast_conv(a)
    best_max = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-12:
                continue
            
            # Try moving mass from i to j
            for offset in [1, 2, 3, 5, 8, 13, 21, 34, 55]:
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    # Try a small move
                    d = a[i] * 0.3
                    a[i] -= d
                    a[j] += d
                    new_max = fast_conv(a).max()
                    if new_max < best_max - 1e-13:
                        best_max = new_max
                        best_a = a.copy()
                        conv = fast_conv(a)
                        improved = True
                    else:
                        a[i] += d
                        a[j] -= d
        
        if not improved and n_passes > 5:
            # Try a bigger perturbation
            noise = rng.laplace(0, 0.01, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            if a_pert.sum() > 1e-15:
                a = a_pert / a_pert.sum()
            else:
                break
        
        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_max:.8f}")
        
        if n_passes > 100:
            break
    
    return best_a, best_max


def try_different_n(a_template, n_candidates, time_per_n, verbose=True):
    """Try resizing the template to different n values and quick-optimize."""
    best_a = None
    best_score = float('inf')
    
    for n in n_candidates:
        # Resize template to n
        if len(a_template) == n:
            a = a_template.copy()
        else:
            # Interpolate
            x_old = np.linspace(0, 1, len(a_template))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, a_template)
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        
        # Quick subgradient optimization
        a_opt, sc = subgrad_optimize(a, time_per_n, verbose=False)
        
        score = 2 * n * sc  # since sum(a)=1, score = 2n*max_conv
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            if verbose:
                print(f"    n={n}: score={score:.6f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main entry point. Searches for the best sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    rng = np.random.default_rng(seed)
    
    print(f"Total time budget: {time_budget:.0f}s")
    
    solver = get_solver()
    
    # Initialize from parent if available
    best_a = None
    best_score = float('inf')
    n_best = 1600
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best_a = p / p.sum()
            n_best = len(best_a)
            sc = evaluate_safe(best_a)
            best_score = sc
            print(f"Parent: n={n_best}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_a is None:
        n_best = 1600
        best_a = np.ones(n_best) / n_best
        best_score = evaluate_safe(best_a)
        print(f"Default init: n={n_best}, score={best_score:.6f}")
    
    def update_best(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 1: LP Nesterov (40% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LP Nesterov n={n_best} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp(best_a, t1, solver, verbose=True, freq_perturb_every=200)
        update_best(a_out, "lp1")
    
    # Phase 2: Subgradient optimization (25% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    t2 = remaining *
