def nesterov_subgradient(a0, time_limit, n_peaks=10, verbose=False):

        weights = weights / weights.sum()

        # For subgradient: x_{t+1} = x_t - step_t * grad_t

def lp_best_response_sparse(a_fixed, M, active_set, solver):
    """LP best response with sparse constraint matrix."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    
    # Build sparse constraint matrix
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            rows.append(idx)
            cols.append(j)
            vals.append(a_fixed[k - j])
    
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, None

        top_k = np.argsort(conv_y)[-max(5, n // 100):]

                break
            
            aj = a[j]
            
            # Background: conv without terms involving a[j]
            # conv[k] = bg[k] + 2*aj*c[k] + aj^2*delta(k,2j)
            c = np.zeros(size)
            for k in range(size):
                val = 0.0
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    val += a[i1]
                i2 = j - k

                    val += a[i2]

def get_solver():

            if prob.status in ('optimal', 'optimal_inaccurate'):

                return s

def init_sequence(n, rng, mode='sin'):
    """Generate initial sequence."""
    if mode == 'uniform':
        a = np.ones(n)
    elif mode == 'sin':
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 'bell':
        x = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (x - 0.5) ** 2)
    elif mode == 'random':
        a = rng.random(n)
        a = np.maximum(a, 0.01)
    elif mode == 'cos2':
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    return a / a.sum()

        # Nesterov momentum update
        beta_t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        if t > 1:
            beta_prev = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * (t - 1) * (t - 1)))
        else:
            beta_prev = 1.0

        y_new = a_trial + (beta_prev / beta_t - 1.0) * (a_trial - y_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new * norm / s_y
        else:
            y_new = a_trial.copy()

        y_prev = y
        y = y_new
        a = a_trial

        # Track best
        if score_trial < best_score - 1e-14:
            best_score = score_trial
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Random diversity: occasionally perturb
        if patience % 200 == 100 and patience <= max_patience - 100:
            idx = rng.integers(0, n, size=max(1, n // 100))
            a_pert = a.copy()
            a_pert[idx] *= (0.5 + rng.random(len(idx)))
            a_pert = normalize(a_pert, norm)
            sc_pert = fast_autocorr(a_pert).max()
            if sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert.copy()
                y = a_pert.copy()
                y_prev = a_pert.copy()
                patience = 0

        # Restart if very stalled
        if patience > max_patience // 2:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            patience = max_patience // 4

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSub t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def build_sparse_conv_matrix(a_fixed, k_indices, n):
    """Build sparse matrix for constraints (a_fixed * b)[k] <= M."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            rows.append(idx)
            cols.append(j)
            vals.append(a_fixed[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    T = build_sparse_conv_matrix(a_fixed, active_set, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0,
                         tol_start=0.04, tol_end=0.003, max_active_frac=0.5):
    """
    Nesterov-accelerated LP best-response with adaptive active set.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = normalize(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    stall_threshold = 60
    n_iters = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()

        # Adaptive active set: tighten over time
        progress = min(t / 200.0, 1.0)
        tol = tol_start * (tol_end / tol_start) ** progress
        active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
        # Always include top lags
        n_top = max(5, n // 50)
        top_k = np.argsort(conv_y)[-n_top:]
        active_set = np.unique(np.concatenate([active_set, top_k]))

        # Cap active set for speed
        max_active = int(max_active_frac * size)
        if len(active_set) > max_active:
            conv_active = conv_y[active_set]
            threshold = np.percentile(conv_active, 60)
            active_set = active_set[conv_active >= threshold]
            if len(active_set) > max_active:
                active_set = np.argsort(conv_y)[-max_active:]

        b = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            # Try larger active set
            if len(active_set) < size:
                active_set = np.argsort(conv_y)[-min(size, 2 * n):]
                b = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                if t % 50 == 0:
                    # Fallback: use subgradient direction
                    top1 = int(np.argmax(conv_y))
                    grad = compute_subgradient(y, top1)
                    g = y - 0.1 * grad / np.linalg.norm(grad)
                    g = normalize(g, norm)
                else:
                    continue
        else:
            b_sum = b.sum()
            if b_sum < 1e-15:
                continue
            g = normalize(b, norm)
            n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.93)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
            else:
                a_new = y_new.copy()

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average every 50 iters
        if n_iters % 50 == 0 and n_iters > 100:
            # Quick check of averaging recent iterates
            pass

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at t={t} ({elapsed:.0f}s), score={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, |A|={len(active_set)}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False):
    """
    Coordinate descent with exact 1D optimization per coordinate.
    Uses efficient computation of the coordinate-wise score.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
    t0 = time.monotonic()
    size = 2 * n - 1
    conv = fast_autocorr(a)

    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]

            # Compute correlation c[k] = sum of a[i] for terms where changing a[j] affects conv[k]
            # conv[k] involves a[j] iff k-j is in [0,n-1] or j-k is in [0,n-1]
            # Actually: conv[k] = sum_i a[i]*a[k-i]. a[j] appears when i=j (gives a[k-j])
            # or when k-i=j (i=k-j, gives a[k-j]).
            # So the "interaction" coefficient for a[j] in conv[k] is:
            #   c[k] = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n) = 2*a[k-j] (if valid)
            # Wait, that's the gradient, not the interaction.
            #
            # Let me think differently. If we change a[j] from aj to x:
            # conv_new[k] = conv_old[k] + (x - aj) * (a[k-j] + a[j-k]) + 0
            # ... no, that's not right either because of the a[j]*a[j] term.
            #
            # Conv[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
            # Terms involving a[j]:
            #   i=j: a[j]*a[k-j] (if 0<=k-j<n)
            #   k-i=j, i=k-j: a[k-j]*a[j] (if 0<=k-j<n, and k-j != j)
            #   i=j and k-i=j (k=2j): a[j]*a[j] (counted once)
            #
            # So if k != 2j: terms involving a[j] are a[j]*a[k-j] + a[k-j]*a[j] = 2*aj*a[k-j]
            # If k = 2j: term is a[j]*a[j] = aj^2
            #
            # After change a[j] -> x:
            # If k != 2j: new terms are 2*x*a[k-j], change = 2*(x-aj)*a[k-j]
            # If k = 2j: new term is x^2, change = x^2 - aj^2
            #
            # So: conv_new[k] = conv_bg[k] + 2*x*c[k] + (x^2 if k==2j else 0)
            # where c[k] = a[k-j] (if 0<=k-j<n, and k!=2j), c[2j] = 0
            # and conv_bg[k] = conv[k] - 2*aj*c[k] - (aj^2 if k==2j else 0)

            c = np.zeros(size)
            lo = max(0, j + 1)  # k-j >= 1 means k >= j+1 (to exclude k=2j handled separately)
            # Actually: c[k] = a[k-j] for k where 0<=k-j<n and k-j != j
            # i.e., k in [j, j+n-1] excluding k=2j
            for k in range(max(0, j), min(size, j + n)):
                i = k - j
                if 0 <= i < n and i != j:
                    c[k] = a[i]

            bg = conv.copy()
            bg -= 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            s_others = a.sum() - aj

            # Score for a given x: 2*n*max(bg + 2*x*c + x^2*delta) / (s_others + x)^2
            # Try several candidates
            candidates = [0.0, aj * 0.3, aj * 0.7, aj, aj * 1.3, aj * 1.7,
                         s_others * 0.05, s_others * 0.3]
            candidates = sorted(set([x for x in candidates if 0 <= x <= 2 * norm]))

            best_x = aj
            best_obj = float('inf')

            for x in candidates:
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                new_conv = bg + 2.0 * x * c
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                mx = new_conv.max()
                obj = 2.0 * n * mx / (new_sum * new_sum)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            # Bisection refinement
            if best_x != aj and abs(best_x - aj) > 1e-10:
                lo_b = min(aj, best_x)
                hi_b = max(aj, best_x)
                obj_lo = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
                for _ in range(8):
                    mid = (lo_b + hi_b) / 2.0
                    new_sum = s_others + mid
                    if new_sum < 1e-15:
                        break
                    new_conv = bg + 2.0 * mid * c
                    if 2 * j < size:
                        new_conv[2 * j] += mid * mid
                    obj_mid = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                    if obj_mid < best_obj:
                        hi_b = mid
                        best_obj = obj_mid
                        best_x = mid
                    else:
                        lo_b = mid
                best_x = (lo_b + hi_b) / 2.0

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    conv = fast_autocorr(a)
                    new_score = 2.0 * n * conv.max() / (a.sum() ** 2)
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD pass done ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def get_solver():
    """Find the best available LP solver."""
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def init_sequence(n, rng, mode='sin'):
    """Generate various initialization sequences."""
    x = np.linspace(0.0, 1.0, n)
    if mode == 'uniform':
        a = np.ones(n)
    elif mode == 'sin':
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 'bell':
        a = np.exp(-1.5 * (x - 0.5) ** 2)
    elif mode == 'random':
        a = np.maximum(rng.random(n), 0.01)
    elif mode == 'cos2':
        a = np.maximum(np.cos(np.pi * x) ** 2, 0.01)
    elif mode == 'multiscale':
        a = 0.4 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(6 * np.pi * x) \
            + 0.1 * np.cos(10 * np.pi * x) + 0.05 * np.cos(14 * np.pi * x)
        a = np.maximum(a, 0.005)
    else:
        a = np.ones(n)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    1. Load parent or create initial sequence
    2. Explore multiple lengths with LP-based optimization
    3. Polish with corrected Nesterov subgradient
    4. Final refinement with coordinate descent and LP
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"\nStarting: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: LP optimization at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_parent}, budget={t1:.0f}s ===")
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_parent))
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0,
                                        tol_start=0.04, tol_end=0.003)
        try_update(a_out, f"lp1_n{n_parent}")

    # === Phase 2: Corrected subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 150.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Nesterov Subgradient n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = nesterov_subgradient(a_norm, t2, n_peaks=10, verbose=True, seed=42)
        try_update(a_out, f"subg_n{n_cur}")

    # === Phase 3: LP round 2 (tighter active set) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP round 2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=123,
                                        tol_start=0.02, tol_end=0.001)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 120.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Coordinate descent n={n_cur}, budget={t4:.0f}s ===")
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = coordinate_descent_fast(a_norm, t4, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    lengths_to_try = []
    n0 = len(best_seq)
    for dn in [-200, -100, 100, 200, 400]:
        n_new = n0 + dn
        if 800 <= n_new <= 2800 and n_new not in lengths_to_try and n_new != n0:
            lengths_to_try.append(n_new)
    lengths_to_try.sort()

    for n_try in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        t5 = min(remaining * 0.10, 100.0)
        if t5 < 30:
            break
        print(f"\n=== Phase 5: Trying n={n_try}, budget={t5:.0f}s ===")

        # Initialize by interpolation
        a_init = interpolate_sequence(best_seq, n_try)
        a_norm = normalize(a_init, np.sqrt(2.0 * n_try))

        # Run subgradient (faster than LP for exploration)
        a_out, _ = nesterov_subgradient(a_norm, t5 * 0.5, n_peaks=8, verbose=False, seed=n_try)
        try_update(a_out, f"subg_explore_n{n_try}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and a_out is not None:
            # Follow up with LP if subgradient found something good
            t_lp = min(remaining * 0.3, 60.0)
            if t_lp > 20:
                score_check = fast_autocorr(normalize(a_out, np.sqrt(2.0 * n_try))).max()
                if score_check < best_score * 1.01:
                    a_norm2 = normalize(a_out, np.sqrt(2.0 * n_try))
                    a_out2, _ = nesterov_lp_optimize(a_norm2, t_lp, solver, verbose=False,
                                                     seed=n_try, tol_start=0.02, tol_end=0.002)
                    try_update(a_out2, f"lp_explore_n{n_try}")

    # === Phase 6: Perturbation + LP restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    for r in range(3):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t6 = min(remaining * 0.15, 80.0)
        if t6 < 20:
            break
        print(f"\n=== Phase 6.{r+1}: Perturb+LP n={n_cur}, budget={t6:.0f}s ===")
        scale = 0.015 + 0.01 * r
        noise = 1.0 + scale * rng.standard_normal(n_cur)
        noise = np.maximum(noise, 0.5)
        a_pert = best_seq * noise
        a_pert = a_pert / a_pert.sum()
        a_norm = normalize(a_pert, np.sqrt(2.0 * n_cur))
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=False,
                                        seed=500 + r * 7, tol_start=0.03, tol_end=0.003)
        try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 7: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        t7 = min(remaining * 0.3, 80.0)
        print(f"\n=== Phase 7: Final subgradient n={n_cur}, budget={t7:.0f}s ===")
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = nesterov_subgradient(a_norm, t7, n_peaks=5, verbose=False, seed=999)
        try_update(a_out, f"subg_final_n{n_cur}")

    # === Phase 8: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_cur = len(best_seq)
        t8 = min(remaining * 0.4, 60.0)
        print(f"\n=== Phase 8: Final CD n={n_cur}, budget={t8:.0f}s ===")
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = coordinate_descent_fast(a_norm, t8, verbose=False)
        try_update(a_out, f"cd_final_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
